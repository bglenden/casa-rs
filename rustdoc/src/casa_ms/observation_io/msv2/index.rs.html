2//! The MSv2 adapter's opening index: every MAIN row's partition and its
3//! cell on that partition's time × baseline grid, built once at opening so
4//! that estimates are arithmetic and every read is addressed by cell.
5//!
6//! Rows are partitioned the way XRADIO's converter does by default
7//! (`DATA_DESC_ID` × `OBS_MODE` × `OBSERVATION_ID` × `EPHEMERIS_ID`), plus
8//! the feed pair, which MSv4 has no axis for, plus whatever the
9//! [`PartitionScheme`] adds. Within a partition the grid is the unique
10//! times (exact values; no tolerance merges jittered times) by the unique
11//! ordered baselines; a cell two rows claim is refused, naming both rows,
12//! and a cell no row fills is padding.
13//!
14//! MAIN's key columns are read [`ROWS_PER_CHUNK`] rows at a time, in two
15//! passes: the first gathers each partition's times, baselines and
16//! processors and each scan's states, the second places every row in its
17//! cell. Opening therefore holds the grids (four bytes a cell) and one
18//! chunk of columns, whatever the MeasurementSet's size; a MAIN that fits
19//! in one chunk is read once.
20

30/// The cell of a grid no row fills.
31pub(super) const NO_ROW: u32 = u32::MAX;
32
33/// MAIN rows whose key columns are read at once while the index is built:
34/// about 15 MiB of columns and row numbers.
35pub(super) const ROWS_PER_CHUNK: usize = 1 << 18;
36
37/// Bytes a chunk of key columns ([`KeyColumns`]) holds per row: a row
38/// number, a time and ten integer keys.
39pub(super) const CHUNK_ROW_BYTES: usize =
40    size_of::<usize>() + size_of::<f64>() + 10 * size_of::<i32>();
41
42/// At most the bytes a node of a B-tree of `entry`-byte entries holds:
43/// room for 11 entries, the 12 edges of an internal node, and its header.
44const fn node_bytes(entry: usize) -> usize {
45    11 * entry + 12 * size_of::<usize>() + 16
46}
47
48/// At most the bytes a B-tree holds an entry of `entry` bytes besides its
49/// root ([`node_bytes`]): a node but the root holds at least 5 entries.
50const fn entry_bytes(entry: usize) -> usize {
51    node_bytes(entry).div_ceil(5)
52}
53
54/// The step the opening asks its grant for beyond what it needs, so that
55/// a growing first pass asks it rarely.
56const GRANT_STEP: usize = 1 << 20;
57
58/// What the opening asks to hold ([`Index::build`]'s `grant`): the bytes
59/// the index will hold in all, in totals that only grow.
60pub(super) type OpeningGrant<'a> = dyn FnMut(usize) -> Result<(), ObservationError> + 'a;
61
62/// Bytes a grid holds a time while its rows are placed ([`Placing`]): the
63/// time, its field and its scan.
64const PLACING_TIME_BYTES: usize =
65    size_of::<f64>() + size_of::<Option<FieldId>>() + size_of::<Option<ScanId>>();

67/// One chunk of MAIN's key columns, refilled for every chunk.
68#[derive(Default)]
69struct KeyColumns {
70    /// The chunk's MAIN rows.
71    rows: Vec<usize>,
72    data_description: Vec<i32>,
73    time: Vec<f64>,
74    field: Vec<i32>,
75    scan_number: Vec<i32>,
76    state: Vec<i32>,
77    observation: Vec<i32>,
78    antenna1: Vec<i32>,
79    antenna2: Vec<i32>,
80    feed1: Vec<i32>,
81    feed2: Vec<i32>,
82    processor: Vec<i32>,
83}
84
85impl KeyColumns {
86    /// Read the key columns of MAIN rows `rows`.
87    fn read(&mut self, main: &Table, rows: Range<usize>) -> Result<(), ObservationError> {
88        let Self {
89            rows: indices,
90            data_description,
91            time,
92            field,
93            scan_number,
94            state,
95            observation,
96            antenna1,
97            antenna2,
98            feed1,
99            feed2,
100            processor,
101        } = self;
102        indices.clear();
103        indices.extend(rows);
104        let mut destinations = vec![RequiredScalarColumnDestination::new(
105            "TIME",
106            RequiredScalarColumnValuesMut::Float64(time),
107        )];
108        for (name, values) in [
109            ("DATA_DESC_ID", data_description),
110            ("FIELD_ID", field),
111            ("SCAN_NUMBER", scan_number),
112            ("STATE_ID", state),
113            ("OBSERVATION_ID", observation),
114            ("ANTENNA1", antenna1),
115            ("ANTENNA2", antenna2),
116            ("FEED1", feed1),
117            ("FEED2", feed2),
118            ("PROCESSOR_ID", processor),
119        ] {
120            destinations.push(RequiredScalarColumnDestination::new(
121                name,
122                RequiredScalarColumnValuesMut::Int32(values),
123            ));
124        }
125        main.required_scalar_columns_for_rows_into(indices, &mut destinations)
126            .map_err(table_error)
127    }
128
129    /// The baseline of chunk row `i`, MAIN row `row`.
130    fn baseline(&self, i: usize, row: usize) -> Result<OrderedBaseline, ObservationError> {
131        let antenna = |value: i32| {
132            u32::try_from(value)
133                .map(AntennaId)
134                .map_err(|_| invalid(format!("row {row} has a negative antenna {value}")))
135        };
136        Ok(OrderedBaseline {
137            antenna1: antenna(self.antenna1[i])?,
138            antenna2: antenna(self.antenna2[i])?,
139        })
140    }
141}

143/// What defines a partition of MAIN rows.
144#[derive(Clone, Debug, PartialEq, Eq, PartialOrd, Ord)]
145pub(super) struct PartitionKey {
146    pub(super) data_description: i32,
147    /// `STATE::OBS_MODE` of the rows' state; empty without a state.
148    pub(super) obs_mode: String,
149    pub(super) observation: i32,
150    /// The rows' field's `EPHEMERIS_ID`, when the column exists.
151    pub(super) ephemeris: Option<i32>,
152    pub(super) feeds: [i32; 2],
153    /// `FIELD_ID`, when the scheme splits by field.
154    pub(super) field: Option<i32>,
155    /// `SCAN_NUMBER`, when the scheme splits by scan.
156    pub(super) scan: Option<i32>,
157}

159/// A row's [`PartitionKey`] with its `OBS_MODE` as the mode's rank among
160/// the distinct modes: it orders as the key does and costs no allocation.
161#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
162struct RowKey {
163    data_description: i32,
164    obs_mode: usize,
165    observation: i32,
166    ephemeris: Option<i32>,
167    feeds: [i32; 2],
168    field: Option<i32>,
169    scan: Option<i32>,
170}
171
172/// The subtables' part of every row's key.
173struct Keys<'a> {
174    scheme: PartitionScheme,
175    /// Each `FIELD` row's `EPHEMERIS_ID`.
176    field_ephemeris: &'a [Option<i32>],
177    /// The distinct `OBS_MODE`s and the empty mode, ascending.
178    modes: Vec<String>,
179    /// Each `STATE` row's mode, by its rank in `modes`.
180    state_mode: Vec<usize>,
181    /// The empty mode's rank: a row's without a state.
182    no_mode: usize,
183}
184
185impl<'a> Keys<'a> {
186    fn new(
187        scheme: PartitionScheme,
188        state_modes: &[String],
189        field_ephemeris: &'a [Option<i32>],
190    ) -> Self {
191        let modes: Vec<String> = state_modes
192            .iter()
193            .cloned()
194            .chain([String::new()])
195            .collect::<BTreeSet<_>>()
196            .into_iter()
197            .collect();
198        let rank = |mode: &str| {
199            modes
200                .binary_search_by(|known| known.as_str().cmp(mode))
201                .expect("every mode is collected")
202        };
203        let state_mode = state_modes.iter().map(|mode| rank(mode)).collect();
204        let no_mode = rank("");
205        Self {
206            scheme,
207            field_ephemeris,
208            modes,
209            state_mode,
210            no_mode,
211        }
212    }
213
214    /// The key and field of chunk row `i`, MAIN row `row`.
215    fn of(
216        &self,
217        columns: &KeyColumns,
218        i: usize,
219        row: usize,
220    ) -> Result<(RowKey, FieldId), ObservationError> {
221        let field = columns.field[i];
222        let ephemeris = usize::try_from(field)
223            .ok()
224            .and_then(|index| self.field_ephemeris.get(index))
225            .ok_or_else(|| invalid(format!("row {row} has FIELD_ID {field} outside FIELD")))?;
226        let obs_mode = usize::try_from(columns.state[i])
227            .ok()
228            .and_then(|state| self.state_mode.get(state))
229            .copied()
230            .unwrap_or(self.no_mode);
231        let key = RowKey {
232            data_description: columns.data_description[i],
233            obs_mode,
234            observation: columns.observation[i],
235            ephemeris: *ephemeris,
236            feeds: [columns.feed1[i], columns.feed2[i]],
237            field: self.scheme.fields.then_some(field),
238            scan: self.scheme.scans.then_some(columns.scan_number[i]),
239        };
240        Ok((key, FieldId(field.unsigned_abs())))
241    }
242
243    fn partition_key(&self, key: RowKey) -> PartitionKey {
244        PartitionKey {
245            data_description: key.data_description,
246            obs_mode: self.modes[key.obs_mode].clone(),
247            observation: key.observation,
248            ephemeris: key.ephemeris,
249            feeds: key.feeds,
250            field: key.field,
251            scan: key.scan,
252        }
253    }
254}

256/// What the first pass learns of one partition.
257#[derive(Default)]
258struct Gathered {
259    /// Distinct `TIME`s, by their bits.
260    times: BTreeSet<u64>,
261    /// The last row's `TIME`: rows of one time usually follow each other.
262    last_time: Option<u64>,
263    baselines: BTreeSet<OrderedBaseline>,
264    processors: BTreeSet<i32>,
265    /// The `STATE_ID`s of its rows (#937).
266    states: BTreeSet<i32>,
267    /// The last row's `STATE_ID`: rows of one state usually follow each
268    /// other.
269    last_state: Option<i32>,
270    rows: usize,
271}
272
273impl Gathered {
274    /// At most the bytes a new partition's sets hold for their roots.
275    const ROOTS: usize = node_bytes(size_of::<u64>())
276        + node_bytes(size_of::<OrderedBaseline>())
277        + 2 * node_bytes(size_of::<i32>());
278
279    /// The bytes the partition's grid holds while its rows are placed: a
280    /// cell a time and baseline, each time's and baseline's tables, and its
281    /// place among the grids and their keys.
282    fn grid_bytes(&self) -> usize {
283        let (times, baselines) = (self.times.len(), self.baselines.len());
284        times
285            .saturating_mul(baselines)
286            .saturating_mul(size_of::<u32>())
287            .saturating_add(times * PLACING_TIME_BYTES)
288            .saturating_add(baselines * size_of::<OrderedBaseline>())
289            .saturating_add(self.states.len() * size_of::<i32>())
290            .saturating_add(size_of::<Placing>() + size_of::<RowKey>())
291    }
292}

294/// What the first pass learns of all of MAIN.
295#[derive(Default)]
296struct Survey {
297    partitions: BTreeMap<RowKey, Gathered>,
298    /// At most the bytes its sets and maps hold: each tree's root when it
299    /// is made, and [`entry_bytes`] an entry.
300    bytes: usize,
301    /// Each `SCAN_NUMBER`'s `STATE_ID`s.
302    scan_states: BTreeMap<i32, BTreeSet<i32>>,
303}
304
305impl Survey {
306    /// Gather the rows of `columns`, MAIN rows from `first`, asking
307    /// `ensure` for `base` and what the survey holds after each row that
308    /// added a key: at most one row's keys beyond what was granted are
309    /// held before it answers.
310    fn gather(
311        &mut self,
312        keys: &Keys<'_>,
313        (columns, first): (&KeyColumns, usize),
314        base: usize,
315        ensure: &mut dyn FnMut(usize) -> Result<(), ObservationError>,
316    ) -> Result<(), ObservationError> {
317        const PARTITION: usize = size_of::<(RowKey, Gathered)>();
318        const SCAN: usize = size_of::<(i32, BTreeSet<i32>)>();
319        let mut last_scan = None;
320        for i in 0..columns.time.len() {
321            let row = first + i;
322            let (key, _) = keys.of(columns, i, row)?;
323            let mut added = 0;
324            let first_partition = self.partitions.is_empty();
325            let partition = self.partitions.entry(key).or_insert_with(|| {
326                added += entry_bytes(PARTITION) + Gathered::ROOTS;
327                if first_partition {
328                    added += node_bytes(PARTITION);
329                }
330                Gathered::default()
331            });
332            let time = columns.time[i].to_bits();
333            if partition.last_time != Some(time) {
334                added += usize::from(partition.times.insert(time)) * entry_bytes(8);
335                partition.last_time = Some(time);
336            }
337            added += usize::from(partition.baselines.insert(columns.baseline(i, row)?))
338                * entry_bytes(size_of::<OrderedBaseline>());
339            added += usize::from(partition.processors.insert(columns.processor[i]))
340                * entry_bytes(size_of::<i32>());
341            if partition.last_state != Some(columns.state[i]) {
342                added += usize::from(partition.states.insert(columns.state[i]))
343                    * entry_bytes(size_of::<i32>());
344                partition.last_state = Some(columns.state[i]);
345            }
346            partition.rows += 1;
347            let scan = (columns.scan_number[i], columns.state[i]);
348            if last_scan != Some(scan) {
349                let first_scan = self.scan_states.is_empty();
350                let states = self.scan_states.entry(scan.0).or_insert_with(|| {
351                    added += entry_bytes(SCAN) + node_bytes(size_of::<i32>());
352                    if first_scan {
353                        added += node_bytes(SCAN);
354                    }
355                    BTreeSet::new()
356                });
357                added += usize::from(states.insert(scan.1)) * entry_bytes(size_of::<i32>());
358                last_scan = Some(scan);
359            }
360            if added > 0 {
361                self.bytes += added;
362                ensure(base + self.bytes)?;
363            }
364        }
365        Ok(())
366    }
367}

369/// The bytes the scans and their tables hold: each entry with its name
370/// and intents, the scan identifiers by number, and the states MAIN holds.
371fn scans_bytes(scans: &[ScanEntry], numbers: usize, states: usize) -> usize {
372    let entries: usize = scans
373        .iter()
374        .map(|scan| {
375            size_of::<ScanEntry>()
376                + scan.name.capacity()
377                + scan.intents.capacity() * size_of::<String>()
378                + scan.intents.iter().map(String::capacity).sum::<usize>()
379        })
380        .sum();
381    let numbers =
382        node_bytes(size_of::<(i32, ScanId)>()) + numbers * entry_bytes(size_of::<(i32, ScanId)>());
383    let states = node_bytes(size_of::<i32>()) + states * entry_bytes(size_of::<i32>());
384    entries + numbers + states
385}

387/// One partition's grid while the second pass places its rows.
388struct Placing {
389    times_s: Vec<f64>,
390    baselines: Vec<OrderedBaseline>,
391    cells: Vec<u32>,
392    field_per_time: Vec<Option<FieldId>>,
393    scan_per_time: Vec<Option<ScanId>>,
394    rows: usize,
395    /// The last row's `TIME` bits and time index.
396    last_time: Option<(u64, usize)>,
397    /// The partition's `STATE_ID`s, ascending, and its one `PROCESSOR_ID`.
398    states: Vec<i32>,
399    processor: Option<i32>,
400}
401
402impl Placing {
403    /// The empty grid of a partition the first pass gathered.
404    ///
405    /// # Errors
406    ///
407    /// When the partition's rows have several `PROCESSOR_ID`s (XRADIO
408    /// requires one per partition).
409    fn new(gathered: Gathered) -> Result<Self, ObservationError> {
410        if gathered.processors.len() > 1 {
411            return Err(ObservationError::Unsupported {
412                what: "rows of one partition with several PROCESSOR_IDs",
413            });
414        }
415        let mut times_s: Vec<f64> = gathered.times.into_iter().map(f64::from_bits).collect();
416        times_s.sort_by(f64::total_cmp);
417        let baselines: Vec<OrderedBaseline> = gathered.baselines.into_iter().collect();
418        let times = times_s.len();
419        Ok(Self {
420            cells: vec![NO_ROW; times * baselines.len()],
421            field_per_time: vec![None; times],
422            scan_per_time: vec![None; times],
423            times_s,
424            baselines,
425            rows: gathered.rows,
426            last_time: None,
427            states: gathered.states.into_iter().collect(),
428            processor: gathered.processors.first().copied(),
429        })
430    }
431
432    /// Place chunk row `i`, MAIN row `row`, of field `field`.
433    ///
434    /// # Errors
435    ///
436    /// When its cell already holds a row, or its time already holds another
437    /// field or scan.
438    fn place(
439        &mut self,
440        columns: &KeyColumns,
441        i: usize,
442        row: usize,
443        field: FieldId,
444        scans: &BTreeMap<i32, ScanId>,
445    ) -> Result<(), ObservationError> {
446        let bits = columns.time[i].to_bits();
447        let t = match self.last_time {
448            Some((last, t)) if last == bits => t,
449            _ => {
450                let t = self
451                    .times_s
452                    .binary_search_by(|time| time.total_cmp(&columns.time[i]))
453                    .expect("every row's time is on the grid");
454                self.last_time = Some((bits, t));
455                t
456            }
457        };
458        let b = self
459            .baselines
460            .binary_search(&columns.baseline(i, row)?)
461            .expect("every row's baseline is on the grid");
462        let cell = &mut self.cells[t * self.baselines.len() + b];
463        if *cell != NO_ROW {
464            return Err(ObservationError::InvalidStore {
465                reason: format!(
466                    "rows {} and {row} are two acquisitions of one time and baseline; \
467                     the partition keys do not separate them",
468                    *cell
469                ),
470            });
471        }
472        *cell = u32::try_from(row).expect("rows are counted below NO_ROW");
473        let scan = *scans
474            .get(&columns.scan_number[i])
475            .expect("every SCAN_NUMBER is catalogued");
476        one_per_time(
477            &mut self.field_per_time[t],
478            field,
479            "two fields at one time of one partition",
480        )?;
481        one_per_time(
482            &mut self.scan_per_time[t],
483            scan,
484            "two scans at one time of one partition",
485        )
486    }
487
488    fn finish(self) -> Grid {
489        Grid {
490            times_s: self.times_s,
491            field_per_time: self
492                .field_per_time
493                .into_iter()
494                .map(|field| field.expect("every time has a row"))
495                .collect(),
496            scan_per_time: self
497                .scan_per_time
498                .into_iter()
499                .map(|scan| scan.expect("every time has a row"))
500                .collect(),
501            baselines: self.baselines,
502            cells: self.cells,
503            rows: self.rows,
504            states: self.states,
505            processor: self.processor,
506        }
507    }
508}

510/// The opening index of a MAIN table.
511#[derive(Debug, PartialEq)]
512pub(super) struct Index {
513    /// Every partition's key and grid, in key order.
514    pub(super) partitions: Vec<(PartitionKey, Grid)>,
515    /// The scans, by ascending `SCAN_NUMBER`.
516    pub(super) scans: Vec<ScanEntry>,
517    /// The `STATE_ID`s MAIN's rows hold, -1 and those beyond `STATE`
518    /// included.
519    pub(super) states: BTreeSet<i32>,
520}
521
522impl Index {
523    /// Partition `main`'s rows and place them on their grids, reading the
524    /// key columns `rows_per_chunk` rows at a time.
525    ///
526    /// `state_modes` is each `STATE` row's `OBS_MODE` and `field_ephemeris`
527    /// each `FIELD` row's `EPHEMERIS_ID`.
528    ///
529    /// # Errors
530    ///
531    /// When a row's field is not in `FIELD` or an antenna is negative, two
532    /// rows share a (time, baseline) cell, a time holds rows of two fields
533    /// or two scans, a partition's rows have several `PROCESSOR_ID`s, or
534    /// MAIN has too many rows to number in 32 bits; whatever `grant`
535    /// returns when it refuses.
536    ///
537    /// Before it allocates them, the build asks `grant` for the bytes the
538    /// index will hold in all, in totals that only grow: its chunk of key
539    /// columns, the first pass's sets and maps (in steps, as keys come),
540    /// then the scans and the grids. A refusal stops it with nothing more
541    /// allocated.
542    pub(super) fn build(
543        main: &Table,
544        scheme: PartitionScheme,
545        (state_modes, field_ephemeris): (&[String], &[Option<i32>]),
546        rows_per_chunk: usize,
547        grant: &mut OpeningGrant<'_>,
548    ) -> Result<Self, ObservationError> {
549        let rows = main.row_count();
550        if rows >= NO_ROW as usize {
551            return Err(invalid("more than 2^32 - 1 MAIN rows"));
552        }
553        let chunks: Vec<Range<usize>> = (0..rows)
554            .step_by(rows_per_chunk)
555            .map(|start| start..start.saturating_add(rows_per_chunk).min(rows))
556            .collect();
557        let chunk_bytes = rows_per_chunk.min(rows) * CHUNK_ROW_BYTES;
558        let mut granted = 0;
559        let mut ensure = |needed: usize| -> Result<(), ObservationError> {
560            if needed > granted {
561                let step = needed.max(granted + granted / 2).max(granted + GRANT_STEP);
562                if grant(step).is_err() {
563                    grant(needed)?;
564                    granted = needed;
565                } else {
566                    granted = step;
567                }
568            }
569            Ok(())
570        };
571        ensure(chunk_bytes)?;
572        let keys = Keys::new(scheme, state_modes, field_ephemeris);
573        let mut columns = KeyColumns::default();
574        let mut survey = Survey::default();
575        for chunk in &chunks {
576            columns.read(main, chunk.clone())?;
577            survey.gather(&keys, (&columns, chunk.start), chunk_bytes, &mut ensure)?;
578        }
579        let states: BTreeSet<i32> = survey.scan_states.values().flatten().copied().collect();
580        let (scans, scan_ids) = catalogue::scans(survey.scan_states, state_modes);
581        let scans_bytes = scans_bytes(&scans, scan_ids.len(), states.len());
582        let grids: usize = survey.partitions.values().map(Gathered::grid_bytes).sum();
583        ensure(chunk_bytes + survey.bytes + scans_bytes + grids)?;
584        let order: Vec<RowKey> = survey.partitions.keys().copied().collect();
585        let mut grids = survey
586            .partitions
587            .into_values()
588            .map(Placing::new)
589            .collect::<Result<Vec<_>, _>>()?;
590        for chunk in &chunks {
591            if chunks.len() > 1 {
592                columns.read(main, chunk.clone())?;
593            }
594            let mut last: Option<(RowKey, usize)> = None;
595            for i in 0..columns.time.len() {
596                let row = chunk.start + i;
597                let (key, field) = keys.of(&columns, i, row)?;
598                let partition = match last {
599                    Some((known, partition)) if known == key => partition,
600                    _ => {
601                        let partition = order
602                            .binary_search(&key)
603                            .expect("the first pass saw every key");
604                        last = Some((key, partition));
605                        partition
606                    }
607                };
608                grids[partition].place(&columns, i, row, field, &scan_ids)?;
609            }
610        }
611        Ok(Self {
612            partitions: order
613                .into_iter()
614                .zip(grids)
615                .map(|(key, grid)| (keys.partition_key(key), grid.finish()))
616                .collect(),
617            scans,
618            states,
619        })
620    }
621}
622
623/// One partition's time × baseline grid and the row at each cell.
624#[derive(Debug, PartialEq)]
625pub(super) struct Grid {
626    /// Unique times, ascending (MAIN `TIME`, seconds from MJD 0).
627    pub(super) times_s: Vec<f64>,
628    pub(super) field_per_time: Vec<FieldId>,
629    pub(super) scan_per_time: Vec<ScanId>,
630    /// Unique ordered baselines, ascending.
631    pub(super) baselines: Vec<OrderedBaseline>,
632    /// The MAIN row at cell `t × baselines.len() + b`, or [`NO_ROW`].
633    pub(super) cells: Vec<u32>,
634    /// Rows placed: cells that are not padding.
635    pub(super) rows: usize,
636    /// Its rows' `STATE_ID`s, ascending (−1 for rows without a state).
637    pub(super) states: Vec<i32>,
638    /// Its rows' one `PROCESSOR_ID`.
639    pub(super) processor: Option<i32>,
640}
641
642impl Grid {
643    /// The cells of times `times`.
644    pub(super) fn cells(&self, times: Range<usize>) -> &[u32] {
645        let per_time = self.baselines.len();
646        &self.cells[times.start * per_time..times.end * per_time]
647    }
648
649    /// The first of its rows in time order, as CASA's iteration meets them:
650    /// the lowest MAIN row of its first time.
651    pub(super) fn first_time_row(&self) -> usize {
652        self.cells(0..1)
653            .iter()
654            .copied()
655            .filter(|cell| *cell != NO_ROW)
656            .min()
657            .expect("a partition's first time holds a row") as usize
658    }
659}
660
661/// Record `value` at a time, which every row of that time must share.
662fn one_per_time<T: Copy + PartialEq>(
663    slot: &mut Option<T>,
664    value: T,
665    what: &'static str,
666) -> Result<(), ObservationError> {
667    if slot.is_some_and(|known| known != value) {
668        return Err(ObservationError::Unsupported { what });
669    }
670    *slot = Some(value);
671    Ok(())
672}

682    /// The index does not depend on how many rows each chunk reads, with

694        .expect("unpack");
695        let ms = MeasurementSet::open_retained_read(root.path().join("ms_trial.ms")).expect("open");
696        let modes = catalogue::read_state_modes(&ms).expect("modes");
697        let ephemerides = catalogue::fields_and_uvw(&ms, &BTreeMap::new())
698            .expect("fields")
699            .ephemerides;
700        let main = ms.main_table();
701        assert!(main.row_count() > 64, "several chunks of 7 and 64");
702        for scheme in [
703            PartitionScheme::default(),
704            PartitionScheme {
705                fields: true,
706                scans: true,
707            },
708        ] {
709            let build = |rows_per_chunk| {

712                    scheme,
713                    (&modes, &ephemerides),
714                    rows_per_chunk,
715                    &mut |_| Ok(()),

717                .expect("index")
718            };
719            let whole = build(usize::MAX);
