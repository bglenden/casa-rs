23pub(super) fn write(
24    ms: &MeasurementSet,
25    request: &GainSolveRequest,
26    coefficients: &SplineCoefficients,
27    (mode, refresh_existing): (CalibrationOutputMode, bool),
28    fields: &BTreeSet<i32>,
29    (reference, antennas, frequency): (i32, &BTreeSet<usize>, f64),
30    poly_mode: &str,
31) -> Result<usize, GainSolveError> {
32    let mut columns = legacy_polynomial_columns(true);
33    columns.retain(|column| column.name() != "SIDEBAND_REF");
34    columns.extend(
35        ["N_KNOTS_AMP", "N_KNOTS_PHASE"]
36            .map(|name| ColumnSchema::scalar(name, PrimitiveType::Int32)),
37    );
38    columns.extend(
39        ["SPLINE_KNOTS_AMP", "SPLINE_KNOTS_PHASE"]
40            .map(|name| ColumnSchema::array_variable(name, PrimitiveType::Float64, Some(4))),
41    );
42    let ms_name = ms
43        .path()
44        .and_then(|path| path.file_name())
45        .and_then(|name| name.to_str())
46        .unwrap_or_default();
47    let output = CalibrationOutput::prepare(&request.output_table, mode)?;
48    let table_mode = if refresh_existing && mode == CalibrationOutputMode::Append {
49        if output.destination.exists() {
50            let existing = crate::output_merge::open(&output.destination)?;
51            if existing.info().sub_type != "GSPLINE" {
52                return Err(GainSolveError::Baseline(
53                    "append destination is not a GSPLINE table".into(),
54                ));
55            }
56        }
57        // Generic GSPLINE refreshes existing antenna/field entries.
58        CalibrationOutputMode::Replace
59    } else {
60        mode
61    };
62    let mut writer = CalibrationTableWriter::staged(
63        ms,
64        CalibrationTableDescriptor {
65            output: &request.output_table,
66            schema: TableSchema::new(columns).expect("GSPLINE schema"),
67            subtype: "GSPLINE",
68            parameter_type: None,
69            measurement_set_name: ms_name.to_owned(),
70            include_polarization_basis: false,
71            time_extra_precision_column: Some("TIME_EXTRA_PREC"),
72        },
73        output,
74        table_mode,
75    );
76    writer.preserve_row_order();
77    let table = writer.table_mut();
78    *table.keywords_mut() = RecordValue::new(vec![
79        RecordField::new("CAL_DESC", Value::table_ref("././CAL_DESC")),
80        RecordField::new("CAL_HISTORY", Value::table_ref("././CAL_HISTORY")),
81    ]);
82    set_fixed_unit_keyword(table, "REF_FREQUENCY", &["Hz"]);
83    set_measinfo_keyword(table, "REF_FREQUENCY", "frequency", Some("LSRK"));
84    set_fixed_unit_keyword(table, "REF_DIRECTION", &["rad", "rad"]);
85    set_measinfo_keyword(table, "REF_DIRECTION", "direction", Some("J2000"));
86    let row = SplineRow {
87        coefficients,
88        time: coefficients.knots.iter().sum::<f64>() / coefficients.knots.len() as f64,
89        reference,
90        frequency,
91        mode: poly_mode,
92        write_amplitude: poly_mode.contains("AMP") || poly_mode.contains("A&P"),
93        write_phase: poly_mode.contains("PHAS") || poly_mode.contains("A&P"),
94    };
95    for &field in fields {
96        for &antenna in antennas {
97            writer.append(row.values(field, antenna))?;
98        }
99    }
100    let description = writer.output_path().join("CAL_DESC");
101    write_legacy_cal_desc(
102        description.clone(),
103        &BTreeMap::from([(-1, 0)]),
104        |_| 0,
105        true,
106        ms_name,
107    )
108    .map_err(|source| GainSolveError::SaveCalibrationTable {
109        path: description.display().to_string(),
110        source: Box::new(source),
111    })?;
112    let history = writer.output_path().join("CAL_HISTORY");
113    write_legacy_history(&history).map_err(|source| GainSolveError::SaveCalibrationTable {
114        path: history.display().to_string(),
115        source: Box::new(source),
116    })?;
117    writer.save_main()?;
118    Ok(fields.len() * antennas.len())
119}

121struct SplineRow<'a> {
122    coefficients: &'a SplineCoefficients,
123    time: f64,
124    reference: i32,
125    frequency: f64,
126    mode: &'a str,
127    write_amplitude: bool,
128    write_phase: bool,
129}
130
131impl SplineRow<'_> {
132    fn values(&self, field: i32, antenna: usize) -> RecordValue {
133        let scalar = |name, value| RecordField::new(name, Value::Scalar(value));
134        let int = |name, value| scalar(name, ScalarValue::Int32(value));
135        let number = |name, value| scalar(name, ScalarValue::Float64(value));
136        let text = |name, value: &str| scalar(name, ScalarValue::String(value.to_owned()));
137        let mut fields = vec![
138            number("TIME", self.time),
139            number("TIME_EXTRA_PREC", 0.),
140            number("INTERVAL", 0.),
141            int("ANTENNA1", antenna as i32),
142            int("FIELD_ID", field),
143            int("CAL_DESC_ID", 0),
144            text("POLY_TYPE", "SPLINE"),
145            text("POLY_MODE", self.mode),
146            text("PHASE_UNITS", if self.write_phase { "RADIANS" } else { "" }),
147            scalar(
148                "SCALE_FACTOR",
149                ScalarValue::Complex32(Complex32::new(1., 0.)),
150            ),
151            scalar("TOTAL_SOLUTION_OK", ScalarValue::Bool(false)),
152            scalar("TOTAL_FIT", ScalarValue::Float32(0.)),
153            scalar("TOTAL_FIT_WEIGHT", ScalarValue::Float32(0.)),
154            int("MEAS_FREQ_REF", 1),
155            int("MEAS_DIR_REF", 0),
156        ];
157        for name in [
158            "FEED1",
159            "ARRAY_ID",
160            "OBSERVATION_ID",
161            "SCAN_NUMBER",
162            "PROCESSOR_ID",
163            "STATE_ID",
164            "PHASE_ID",
165            "PULSAR_BIN",
166            "PULSAR_GATE_ID",
167            "FREQ_GROUP",
168            "CALIBRATION_GROUP",
169            "CAL_HISTORY_ID",
170        ] {
171            fields.push(int(name, -1));
172        }
173        for name in [
174            "FREQ_GROUP_NAME",
175            "FIELD_NAME",
176            "FIELD_CODE",
177            "SOURCE_NAME",
178            "SOURCE_CODE",
179        ] {
180            fields.push(text(name, ""));
181        }
182        for (name, value) in [
183            ("REF_ANT", if self.write_phase { self.reference } else { 0 }),
184            ("REF_FEED", 0),
185            ("REF_RECEPTOR", 0),
186        ] {
187            fields.push(RecordField::new(
188                name,
189                Value::Array(ArrayValue::Int32(array(&[1, 1, 1], vec![value]))),
190            ));
191        }
192        fields.extend([
193            f64_field(
194                "REF_FREQUENCY",
195                &[1, 1, 1],
196                vec![if self.write_phase { self.frequency } else { 0. }],
197            ),
198            f64_field(
199                "REF_DIRECTION",
200                &[2, 1, 1, 1],
201                vec![0., std::f64::consts::FRAC_PI_2],
202            ),
203            RecordField::new(
204                "GAIN",
205                Value::Array(ArrayValue::Complex32(array(
206                    &[1, 1, 1, 1],
207                    vec![Complex32::default()],
208                ))),
209            ),
210        ]);
211        for (name, value) in [("FLAG", true), ("SOLUTION_OK", false)] {
212            fields.push(RecordField::new(
213                name,
214                Value::Array(ArrayValue::Bool(array(&[1, 1, 1, 1], vec![value]))),
215            ));
216        }
217        for name in ["FIT", "FIT_WEIGHT", "SNR"] {
218            fields.push(RecordField::new(
219                name,
220                Value::Array(ArrayValue::Float32(array(&[1, 1, 1, 1], vec![0.]))),
221            ));
222        }
223        for (suffix, part) in [
224            (
225                "AMP",
226                self.write_amplitude
227                    .then_some(self.coefficients.log_amplitude.as_ref())
228                    .flatten(),
229            ),
230            (
231                "PHASE",
232                self.write_phase
233                    .then_some(self.coefficients.phase.as_ref())
234                    .flatten(),
235            ),
236        ] {
237            let count = part.map_or(0, |_| self.coefficients.knots.len());
238            let knots = if count == 0 {
239                vec![0.]
240            } else {
241                self.coefficients.knots.clone()
242            };
243            let values = part.map_or_else(
244                || vec![0.],
245                |hands| {
246                    hands
247                        .iter()
248                        .flat_map(|hand| hand[antenna].iter().copied().chain([0.; 4]))
249                        .collect()
250                },
251            );
252            fields.extend([
253                RecordField::new(
254                    format!("N_KNOTS_{suffix}"),
255                    Value::Scalar(ScalarValue::Int32(count as i32)),
256                ),
257                RecordField::new(
258                    format!("N_POLY_{suffix}"),
259                    Value::Scalar(ScalarValue::Int32(count as i32)),
260                ),
261                f64_field(
262                    &format!("SPLINE_KNOTS_{suffix}"),
263                    &[1, 1, 1, knots.len()],
264                    knots,
265                ),
266                f64_field(
267                    &format!("POLY_COEFF_{suffix}"),
268                    &[1, 1, 1, values.len()],
269                    values,
270                ),
271            ]);
272        }
273        RecordValue::new(fields)
274    }
275}

277fn f64_field(name: &str, shape: &[usize], values: Vec<f64>) -> RecordField {
278    RecordField::new(
279        name,
280        Value::Array(ArrayValue::Float64(array(shape, values))),
281    )
282}
283
284fn array<T>(shape: &[usize], values: Vec<T>) -> ArrayD<T> {
285    ArrayD::from_shape_vec(IxDyn(shape).f(), values).expect("GSPLINE legacy array shape")
