36/// Supported native `gencal` families.
37#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
38#[serde(rename_all = "snake_case")]
39pub enum GencalType {
40    /// Multiplicative per-receptor amplitude (`G Jones`).
41    Amp,
42    /// Per-receptor phase in degrees (`G Jones`).
43    Ph,
44    /// Additive single-band delay in nanoseconds (`K Jones`).
45    Sbd,
46    /// Additive multi-band delay in nanoseconds (`K Jones`).
47    Mbd,
48    /// Antenna-position delay correction table (`KAntPos Jones`).
49    Antpos,
50    /// VLA gain-curve plus antenna-efficiency table (`EGainCurve`).
51    Gceff,
52    /// Zenith-opacity table (`TOpac`).
53    Opac,
54}
55
56impl GencalType {
57    fn table_subtype(self) -> &'static str {
58        match self {
59            Self::Amp | Self::Ph => "G Jones",
60            Self::Sbd | Self::Mbd => "K Jones",
61            Self::Antpos => "KAntPos Jones",
62            Self::Gceff => "EGainCurve",
63            Self::Opac => "TOpac",
64        }
65    }
66
67    fn parameter_column(self) -> (&'static str, PrimitiveType, &'static str) {
68        match self {
69            Self::Amp | Self::Ph => (COL_CPARAM, PrimitiveType::Complex32, "Complex"),
70            _ => (COL_FPARAM, PrimitiveType::Float32, "Float"),
71        }
72    }
73
74    fn is_specified(self) -> bool {
75        matches!(self, Self::Amp | Self::Ph | Self::Sbd | Self::Mbd)
76    }
77}
78
79impl std::str::FromStr for GencalType {
80    type Err = String;
81
82    fn from_str(value: &str) -> Result<Self, Self::Err> {
83        match value.to_ascii_lowercase().as_str() {
84            "amp" => Ok(Self::Amp),
85            "ph" => Ok(Self::Ph),
86            "sbd" => Ok(Self::Sbd),
87            "mbd" => Ok(Self::Mbd),
88            "antpos" => Ok(Self::Antpos),
89            "gceff" | "gc" => Ok(Self::Gceff),
90            "opac" | "opacity" => Ok(Self::Opac),
91            other => Err(format!(
92                "unsupported gencal caltype {other:?}; expected amp, ph, sbd, mbd, antpos, gceff, or opac"
93            )),
94        }
95    }
96}

98/// Request for writing one native prior-cal table.
99#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
100pub struct GencalRequest {
101    /// Input MeasurementSet root path.
102    pub measurement_set: PathBuf,
103    /// Output calibration-table path.
104    pub output_table: PathBuf,
105    /// Prior calibration family to generate.
106    pub caltype: GencalType,
107    /// CASA-style antenna selector for `antpos`.
108    #[serde(default)]
109    pub antenna: String,
110    /// CASA-style SPW selector for `opac`.
111    #[serde(default)]
112    pub spw: String,
113    /// Numeric parameters. `antpos` expects triples per selected antenna; `opac`
114    /// expects one opacity per selected SPW.
115    #[serde(default)]
116    pub parameter: Vec<f64>,
117    /// Optional explicit VLA GainCurves table path for `gceff`.
118    #[serde(default)]
119    pub gaincurve_table: Option<PathBuf>,
120}

139/// Errors from native prior-cal table generation.
140#[derive(Debug, Error)]
141pub enum GencalError {
142    /// Shared calibration-table persistence failed.
143    #[error(transparent)]
144    CalibrationTable(#[from] crate::CalibrationTableWriteError),
145    /// Opening the input MS failed.
146    #[error("failed to open measurement set {path}: {source}")]
147    OpenMeasurementSet {
148        /// Path that failed.
149        path: String,
150        /// Source error.
151        source: Box<casa_ms::MsError>,
152    },
153    /// Opening the VLA GainCurves table failed.
154    #[error("failed to open VLA GainCurves table {path}: {source}")]
155    OpenGainCurves {
156        /// Path that failed.
157        path: String,
158        /// Source error.
159        source: Box<casa_tables::TableError>,
160    },
161    /// Output table preparation failed.
162    #[error("failed to prepare output caltable {path}: {reason}")]
163    PrepareOutput {
164        /// Path that failed.
165        path: String,
166        /// Reason.
167        reason: String,
168    },
169    /// Saving a table failed.
170    #[error("failed to save caltable {path}: {source}")]
171    Save {
172        /// Path that failed.
173        path: String,
174        /// Source error.
175        source: Box<casa_tables::TableError>,
176    },
177    /// Copying an MS subtable failed.
178    #[error("failed to copy {subtable} into caltable {path}: {source}")]
179    CopySubtable {
180        /// Subtable name.
181        subtable: String,
182        /// Caltable root path.
183        path: String,
184        /// Source error.
185        source: Box<casa_tables::TableError>,
186    },
187    /// The request does not match the supported native surface.
188    #[error("{0}")]
189    InvalidRequest(String),
190    /// A needed value in a CASA table had an unsupported type or shape.
191    #[error("{0}")]
192    UnsupportedTableValue(String),
193}

206/// Generate one native prior calibration table.
207pub fn gencal(request: &GencalRequest) -> Result<GencalReport, GencalError> {
208    gencal_with_output_mode(request, "", crate::CalibrationOutputMode::Update)
209}

211pub(crate) fn gencal_with_output_mode(
212    request: &GencalRequest,
213    polarization: &str,
214    mode: crate::CalibrationOutputMode,
215) -> Result<GencalReport, GencalError> {
216    // Reserve before reading prior parameters, so concurrent updates cannot
217    // both accumulate from the same old product and silently lose a correction.
218    let output = crate::output::CalibrationOutput::prepare(&request.output_table, mode)?;
219    let ms = MeasurementSet::open(&request.measurement_set).map_err(|source| {
220        GencalError::OpenMeasurementSet {
221            path: request.measurement_set.display().to_string(),
222            source: Box::new(source),
223        }
224    })?;
225    let updating = mode == crate::CalibrationOutputMode::Update
226        && request.output_table.exists()
227        && request.caltype != GencalType::Gceff;
228    let mut rows = match request.caltype {
229        GencalType::Amp | GencalType::Ph | GencalType::Sbd | GencalType::Mbd => {
230            specified_rows(&ms, request, polarization, updating)?
231        }
232        GencalType::Antpos => antpos_rows(&ms, request)?,
233        GencalType::Gceff => gaincurve_rows(&ms, request)?,
234        GencalType::Opac => opacity_rows(&ms, request)?,
235    };
236    if updating && !request.caltype.is_specified() {
237        apply_existing_parameters(request, &mut rows)?;
238    }
239    write_prior_caltable(&ms, request, &rows, output, mode)
240}

242/// Load the existing `setSpecify` parameters and metadata. The ordered gain
243/// and delay selectors apply their corrections directly to the loaded cells.
244/// Other prior writers already contain one additive correction per cell.
245fn apply_existing_parameters(
246    request: &GencalRequest,
247    rows: &mut [PriorCalRow],
248) -> Result<(), GencalError> {
249    crate::output::validate_destination(&request.output_table, "Calibration")?;
250    let old = crate::output_merge::open(&request.output_table)?;
251    if old.keywords().get("VisCal")
252        != Some(&Value::Scalar(ScalarValue::String(
253            request.caltype.table_subtype().to_string(),
254        )))
255    {
256        return Err(GencalError::InvalidRequest(
257            "existing table has incompatible calibration type; prior product is unchanged"
258                .to_string(),
259        ));
260    }
261    let (column, parameter_type, _) = request.caltype.parameter_column();
262    let mut prior = BTreeMap::new();
263    for row in old
264        .rows()
265        .map_err(|e| GencalError::InvalidRequest(e.to_string()))?
266    {
267        let id = |column| match row.get(column) {
268            Some(Value::Scalar(ScalarValue::Int32(v))) => Ok(*v),
269            _ => Err(GencalError::InvalidRequest(format!(
270                "missing {column} in existing gencal table"
271            ))),
272        };
273        let key = (id(COL_SPECTRAL_WINDOW_ID)?, id(COL_ANTENNA1)?);
274        let valid = match row.get(column) {
275            Some(Value::Array(ArrayValue::Float32(_))) => parameter_type == PrimitiveType::Float32,
276            Some(Value::Array(ArrayValue::Complex32(_))) => {
277                parameter_type == PrimitiveType::Complex32
278            }
279            _ => false,
280        };
281        if !valid {
282            return Err(GencalError::InvalidRequest(format!(
283                "existing table has no {parameter_type:?} {column}"
284            )));
285        }
286        if prior.insert(key, row.clone()).is_some() {
287            return Err(GencalError::InvalidRequest(
288                "existing gencal table has duplicate antenna/SPW entries".to_string(),
289            ));
290        }
291    }
292    for row in rows {
293        // setSpecify's existing-table branch leaves currObs at its default
294        // -1, then keepNCT writes that for the regenerated rows.
295        row.observation_id = -1;
296        if let Some(metadata) = prior.remove(&(row.spw_id, row.antenna_id)) {
297            match (&mut row.parameters, metadata.get(column)) {
298                (ArrayValue::Float32(delta), Some(Value::Array(ArrayValue::Float32(previous))))
299                    if delta.shape() == previous.shape() =>
300                {
301                    if request.caltype.is_specified() {
302                        delta.assign(previous);
303                    } else {
304                        for (value, old) in delta.iter_mut().zip(previous) {
305                            *value += old;
306                        }
307                    }
308                }
309                (
310                    ArrayValue::Complex32(factor),
311                    Some(Value::Array(ArrayValue::Complex32(previous))),
312                ) if factor.shape() == previous.shape() => {
313                    factor.assign(previous);
314                }
315                _ => {
316                    return Err(GencalError::InvalidRequest(format!(
317                        "existing gencal {column} type or shape is incompatible",
318                    )));
319                }
320            }
321            row.preserved = Some(metadata);
322        }
323    }
324    if let Some((&(window, antenna), _)) = prior.first_key_value() {
325        return Err(GencalError::InvalidRequest(format!(
326            "Error in Calibrater::specifycal. Existing table has an entry for window {window}, antenna {antenna}, outside the input grid."
327        )));
328    }
329    Ok(())
330}

332fn specified_rows(
333    ms: &MeasurementSet,
334    request: &GencalRequest,
335    polarization: &str,
336    updating: bool,
337) -> Result<Vec<PriorCalRow>, GencalError> {
338    let invalid = |message: String| {
339        GencalError::InvalidRequest(format!("Error in Calibrater::specifycal. {message}"))
340    };
341    let antenna_count = ms
342        .antenna()
343        .map_err(|error| invalid(error.to_string()))?
344        .row_count();
345    let window_count = ms
346        .spectral_window()
347        .map_err(|error| invalid(error.to_string()))?
348        .row_count();
349
350    let repeat_windows = request.spw.is_empty() || request.spw == "*";
351    let windows = if repeat_windows {
352        (0..window_count).collect::<Vec<_>>()
353    } else {
354        ms.resolve_spw_selector(&request.spw)
355            .map_err(|error| invalid(error.to_string()))?
356            .channel_list()
357            .iter()
358            .map(|range| range.spw_id)
359            .collect()
360    };
361    let antennas = if request.antenna == "*" {
362        Vec::new()
363    } else {
364        ms.resolve_specified_antennas(&request.antenna)
365            .map_err(|error| invalid(error.to_string()))?
366    };
367    let antenna_groups = if antennas.is_empty() {
368        vec![(0..antenna_count).collect::<Vec<_>>()]
369    } else {
370        antennas.into_iter().map(|id| vec![id]).collect()
371    };
372    let receptor_groups: &[&[usize]] = match polarization {
373        "" => &[&[0, 1]],
374        "R" | "X" => &[&[0]],
375        "L" | "Y" => &[&[1]],
376        "R,L" | "X,Y" => &[&[0], &[1]],
377        "L,R" | "Y,X" => &[&[1], &[0]],
378        _ => return Err(invalid("Invalid pol specification".into())),
379    };
380
381    // An omitted axis broadcasts the parameter. Explicit windows, antennas
382    // and receptors use caller order, with the receptor varying fastest.
383    let per_window = antenna_groups.len() * receptor_groups.len();
384    let required = per_window * if repeat_windows { 1 } else { windows.len() };
385    let given = request.parameter.len();
386    if antenna_groups.is_empty()
387        || windows.is_empty()
388        || given == 0
389        || (given != 1 && given != required)
390    {
391        return Err(invalid(format!(
392            "Inconsistent number of parameters specified (expected 1 or {required}, got {given}).",
393        )));
394    }
395    let complex = matches!(request.caltype, GencalType::Amp | GencalType::Ph);
396    let identity = if complex {
397        complex_column([Complex32::new(1.0, 0.0); 2])
398    } else {
399        f32_column(&[0.0; 2])
400    };
401    let mut rows = (0..antenna_count * window_count)
402        .map(|row| PriorCalRow {
403            field_id: -1,
404            spw_id: (row / antenna_count) as i32,
405            antenna_id: (row % antenna_count) as i32,
406            observation_id: 0,
407            parameters: identity.clone(),
408            snr_value: 1.0,
409            preserved: None,
410        })
411        .collect::<Vec<_>>();
412    if updating {
413        apply_existing_parameters(request, &mut rows)?;
414    }
415    let factors = request
416        .parameter
417        .iter()
418        .map(|value| match request.caltype {
419            GencalType::Ph => {
420                let (sin, cos) = (value * std::f64::consts::PI / 180.0).sin_cos();
421                Complex32::new(cos as f32, sin as f32)
422            }
423            _ => Complex32::new(*value as f32, 0.0),
424        })
425        .collect::<Vec<_>>();
426    for (window_order, window) in windows.into_iter().enumerate() {
427        let window_offset = if repeat_windows {
428            0
429        } else {
430            window_order * per_window
431        };
432        for (antenna_order, antennas) in antenna_groups.iter().enumerate() {
433            for (receptor_order, receptors) in receptor_groups.iter().enumerate() {
434                let index = window_offset + antenna_order * receptor_groups.len() + receptor_order;
435                let value = factors[index % given];
436                for antenna in antennas {
437                    let parameters = &mut rows[window * antenna_count + antenna].parameters;
438                    for receptor in *receptors {
439                        match parameters {
440                            ArrayValue::Complex32(gains) => gains[[*receptor, 0]] *= value,
441                            ArrayValue::Float32(delays) => delays[[*receptor, 0]] += value.re,
442                            _ => unreachable!("specified prior columns contain gains or delays"),
443                        }
444                    }
445                }
446            }
447        }
448    }
449
450    Ok(rows)
451}

453fn complex_column(values: [Complex32; 2]) -> ArrayValue {
454    ArrayValue::Complex32(
455        ArrayD::from_shape_vec(IxDyn(&[2, 1]).f(), values.to_vec()).expect("two receptors"),
456    )

459fn antpos_rows(
460    ms: &MeasurementSet,
461    request: &GencalRequest,
462) -> Result<Vec<PriorCalRow>, GencalError> {
463    let antenna = ms
464        .antenna()
465        .map_err(|source| GencalError::OpenMeasurementSet {
466            path: "ANTENNA".to_string(),
467            source: Box::new(source),
468        })?;
469    let selected = parse_antenna_offsets(ms, &request.antenna, &request.parameter)?;
470    let mut rows = Vec::with_capacity(antenna.row_count());
471    for antenna_id in 0..antenna.row_count() {
472        let offsets = selected
473            .get(&(antenna_id as i32))
474            .copied()
475            .unwrap_or([0.0, 0.0, 0.0]);
476        rows.push(PriorCalRow {
477            field_id: -1,
478            spw_id: 0,
479            antenna_id: antenna_id as i32,
480            observation_id: 0,
481            parameters: f32_column(&offsets.map(|value| value as f32)),
482            snr_value: 1.0,
483            preserved: None,
484        });
485    }
486    Ok(rows)
487}

489fn gaincurve_rows(
490    ms: &MeasurementSet,
491    request: &GencalRequest,
492) -> Result<Vec<PriorCalRow>, GencalError> {
493    let source_path = request
494        .gaincurve_table
495        .clone()
496        .or_else(default_vla_gaincurve_table)
497        .ok_or_else(|| {
498            GencalError::InvalidRequest(
499                "gceff requires --gaincurve-table or a CASA data tree at ~/.casa/data/nrao/VLA/GainCurves".to_string(),
500            )
501        })?;
502    let gaincurves = Table::open(TableOptions::new(&source_path)).map_err(|source| {
503        GencalError::OpenGainCurves {
504            path: source_path.display().to_string(),
505            source: Box::new(source),
506        }
507    })?;
508    let antenna_count = ms
509        .antenna()
510        .map_err(|source| GencalError::OpenMeasurementSet {
511            path: "ANTENNA".to_string(),
512            source: Box::new(source),
513        })?
514        .row_count();
515    let antenna_keys = vla_gaincurve_antenna_keys(ms)?;
516    let spw = ms
517        .spectral_window()
518        .map_err(|source| GencalError::OpenMeasurementSet {
519            path: "SPECTRAL_WINDOW".to_string(),
520            source: Box::new(source),
521        })?;
522    let obs_time = first_observation_time(ms).or_else(|_| first_main_time(ms))?;
523    let mut rows = Vec::with_capacity(antenna_count * spw.row_count());
524    for spw_id in 0..spw.row_count() {
525        let center_hz = spw_center_frequency_hz(&spw, spw_id)?;
526        let band = vla_spw_band(&spw, spw_id)?;
527        let efficiency = vla_efficiency(center_hz / 1.0e9) as f32;
528        for (antenna_id, antenna_key) in antenna_keys.iter().enumerate().take(antenna_count) {
529            let mut fparam = matching_gaincurve(
530                &gaincurves,
531                obs_time,
532                center_hz,
533                band.as_deref(),
534                antenna_key,
535            )?;
536            for value in &mut fparam {
537                *value *= efficiency;
538            }
539            rows.push(PriorCalRow {
540                field_id: -1,
541                spw_id: spw_id as i32,
542                antenna_id: antenna_id as i32,
543                observation_id: -1,
544                parameters: f32_column(&fparam),
545                snr_value: 0.0,
546                preserved: None,
547            });
548        }
549    }
550    Ok(rows)
551}

553fn opacity_rows(
554    ms: &MeasurementSet,
555    request: &GencalRequest,
556) -> Result<Vec<PriorCalRow>, GencalError> {
557    let antenna_count = ms
558        .antenna()
559        .map_err(|source| GencalError::OpenMeasurementSet {
560            path: "ANTENNA".to_string(),
561            source: Box::new(source),
562        })?
563        .row_count();
564    let selected_spws = parse_spw_selector(ms, &request.spw)?;
565    // One opacity holds for every selected window, as in CASA's
566    // specifycal; otherwise there is one for each.
567    let opacity = match request.parameter.as_slice() {
568        [one] => vec![*one; selected_spws.len()],
569        values if values.len() == selected_spws.len() => values.to_vec(),
570        values => {
571            return Err(GencalError::InvalidRequest(format!(
572                "opac takes one opacity, or one for each selected SPW (got {} for {} SPWs)",
573                values.len(),
574                selected_spws.len()
575            )));
576        }
577    };
578    let opacities = selected_spws
579        .into_iter()
580        .zip(opacity)
581        .collect::<BTreeMap<_, _>>();
582    let selected_antennas = opacity_antennas(ms, &request.antenna, antenna_count)?;
583    let mut rows = Vec::new();
584    for spw_id in parse_spw_selector(ms, "")? {
585        for antenna_id in 0..antenna_count {
586            rows.push(PriorCalRow {
587                field_id: -1,
588                spw_id,
589                antenna_id: antenna_id as i32,
590                observation_id: 0,
591                parameters: f32_column(&[if selected_antennas.contains(&(antenna_id as i32)) {
592                    opacities.get(&spw_id).copied().unwrap_or(0.0) as f32
593                } else {
594                    0.0
595                }]),
596                snr_value: 1.0,
597                preserved: None,
598            });
599        }
600    }
601    Ok(rows)
602}

604fn opacity_antennas(
605    ms: &MeasurementSet,
606    selector: &str,
607    count: usize,
608) -> Result<BTreeSet<i32>, GencalError> {
609    if selector.trim().is_empty() {
610        return Ok((0..count as i32).collect());
611    }
612    let parameters = vec![0.0; selector.split(',').count() * 3];
613    let selected = parse_antenna_offsets(ms, selector, &parameters)?;
614    if selected.keys().any(|id| *id < 0 || *id >= count as i32) {
615        return Err(GencalError::InvalidRequest(
616            "antenna outside input MS".to_string(),
617        ));
618    }
619    Ok(selected.into_keys().collect())
620}

622fn write_prior_caltable(
623    ms: &MeasurementSet,
624    request: &GencalRequest,
625    rows: &[PriorCalRow],
626    output: crate::output::CalibrationOutput,
627    mode: crate::CalibrationOutputMode,
628) -> Result<GencalReport, GencalError> {
629    let (parameter_column, parameter_type, parameter_name) = request.caltype.parameter_column();
630    let schema = TableSchema::new(vec![
631        ColumnSchema::scalar(COL_TIME, casa_types::PrimitiveType::Float64),
632        ColumnSchema::scalar(COL_FIELD_ID, casa_types::PrimitiveType::Int32),
633        ColumnSchema::scalar(COL_SPECTRAL_WINDOW_ID, casa_types::PrimitiveType::Int32),
634        ColumnSchema::scalar(COL_ANTENNA1, casa_types::PrimitiveType::Int32),
635        ColumnSchema::scalar(COL_ANTENNA2, casa_types::PrimitiveType::Int32),
636        ColumnSchema::scalar(COL_INTERVAL, casa_types::PrimitiveType::Float64),
637        ColumnSchema::scalar(COL_SCAN_NUMBER, casa_types::PrimitiveType::Int32),
638        ColumnSchema::scalar(COL_OBSERVATION_ID, casa_types::PrimitiveType::Int32),
639        ColumnSchema::array_variable(parameter_column, parameter_type, Some(2)),
640        ColumnSchema::array_variable(COL_PARAMERR, casa_types::PrimitiveType::Float32, Some(2)),
641        ColumnSchema::array_variable(COL_FLAG, casa_types::PrimitiveType::Bool, Some(2)),
642        ColumnSchema::array_variable(COL_SNR, casa_types::PrimitiveType::Float32, Some(2)),
643        ColumnSchema::array_variable(COL_WEIGHT, casa_types::PrimitiveType::Float32, Some(2)),
644    ])
645    .expect("valid gencal caltable schema");
646    let subtype = request.caltype.table_subtype();
647    let mut writer = CalibrationTableWriter::staged(
648        ms,
649        CalibrationTableDescriptor {
650            output: &request.output_table,
651            schema,
652            subtype,
653            parameter_type: Some(parameter_name),
654            measurement_set_name: ms
655                .path()
656                .and_then(Path::file_name)
657                .map(|name| name.to_string_lossy().to_string())
658                .unwrap_or_else(|| "<in-memory>".to_string()),
659            include_polarization_basis: true,
660            time_extra_precision_column: None,
661        },
662        output,
663        mode,
664    );
665
666    for row in rows {
667        let len = row.parameters.shape()[0];
668        let zeros = vec![0.0_f32; len];
669        let flags = vec![false; len];
670        let snr = vec![row.snr_value; len];
671        let mut record = RecordValue::new(vec![
672            RecordField::new(COL_TIME, Value::Scalar(ScalarValue::Float64(0.0))),
673            RecordField::new(
674                COL_FIELD_ID,
675                Value::Scalar(ScalarValue::Int32(row.field_id)),
676            ),
677            RecordField::new(
678                COL_SPECTRAL_WINDOW_ID,
679                Value::Scalar(ScalarValue::Int32(row.spw_id)),
680            ),
681            RecordField::new(
682                COL_ANTENNA1,
683                Value::Scalar(ScalarValue::Int32(row.antenna_id)),
684            ),
685            RecordField::new(COL_ANTENNA2, Value::Scalar(ScalarValue::Int32(-1))),
686            RecordField::new(COL_INTERVAL, Value::Scalar(ScalarValue::Float64(0.0))),
687            RecordField::new(COL_SCAN_NUMBER, Value::Scalar(ScalarValue::Int32(-1))),
688            RecordField::new(
689                COL_OBSERVATION_ID,
690                Value::Scalar(ScalarValue::Int32(row.observation_id)),
691            ),
692            RecordField::new(parameter_column, Value::Array(row.parameters.clone())),
693            RecordField::new(COL_PARAMERR, Value::Array(f32_column(&zeros))),
694            RecordField::new(COL_FLAG, Value::Array(bool_column(&flags))),
695            RecordField::new(COL_SNR, Value::Array(f32_column(&snr))),
696        ]);
697        if let Some(prior) = &row.preserved {
698            for column in [COL_PARAMERR, COL_FLAG, COL_SNR] {
699                if let Some(value) = prior.get(column) {
700                    record.upsert(column, value.clone());
701                }
702            }
703        }
704        writer.append(record)?;
705    }
706
707    writer.copy_subtables(&[
708        (SubtableId::Observation, "OBSERVATION"),
709        (SubtableId::Antenna, "ANTENNA"),
710        (SubtableId::Field, "FIELD"),
711        (SubtableId::SpectralWindow, "SPECTRAL_WINDOW"),
712        (SubtableId::History, "HISTORY"),
713    ])?;
714    collapse_spectral_axis(&writer.output_path().join("SPECTRAL_WINDOW"), rows)?;
715    writer.save_main()?;
716
717    let spectral_window_ids = rows
718        .iter()
719        .map(|row| row.spw_id)
720        .collect::<BTreeSet<_>>()
721        .into_iter()
722        .collect();
723    let antenna_ids = rows
724        .iter()
725        .map(|row| row.antenna_id)
726        .collect::<BTreeSet<_>>()
727        .into_iter()
728        .collect();
729    Ok(GencalReport {
730        output_table: request.output_table.clone(),
731        caltype: request.caltype,
732        table_subtype: subtype.to_string(),
733        row_count: rows.len(),
734        spectral_window_ids,
735        antenna_ids,
736    })
737}

739fn parse_antenna_offsets(
740    ms: &MeasurementSet,
741    antenna_selector: &str,
742    parameters: &[f64],
743) -> Result<BTreeMap<i32, [f64; 3]>, GencalError> {
744    if antenna_selector.trim().is_empty() {
745        return Err(GencalError::InvalidRequest(
746            "antpos currently requires explicit --antenna and --parameter offsets; automatic VLA baseline lookup is not implemented".to_string(),
747        ));
748    }
749    let antenna = ms
750        .antenna()
751        .map_err(|source| GencalError::OpenMeasurementSet {
752            path: "ANTENNA".to_string(),
753            source: Box::new(source),
754        })?;
755    let mut name_to_id = BTreeMap::new();
756    for row in 0..antenna.row_count() {
757        let name = antenna.name(row).map_err(|source| {
758            GencalError::InvalidRequest(format!("failed to read ANTENNA.NAME row {row}: {source}"))
759        })?;
760        name_to_id.insert(name.to_ascii_lowercase(), row as i32);
761    }
762    let selected = antenna_selector
763        .split(',')
764        .map(str::trim)
765        .filter(|part| !part.is_empty())
766        .map(|part| {
767            if let Ok(index) = part.parse::<i32>() {
768                return Ok(index);
769            }
770            name_to_id
771                .get(&part.to_ascii_lowercase())
772                .copied()
773                .ok_or_else(|| GencalError::InvalidRequest(format!("unknown antenna {part:?}")))
774        })
775        .collect::<Result<Vec<_>, _>>()?;
776    if parameters.len() != selected.len() * 3 {
777        return Err(GencalError::InvalidRequest(format!(
778            "antpos requires three parameters per selected antenna (got {} parameters for {} antennas)",
779            parameters.len(),
780            selected.len()
781        )));
782    }
783    let mut offsets = BTreeMap::new();
784    for (index, antenna_id) in selected.iter().copied().enumerate() {
785        offsets.insert(
786            antenna_id,
787            [
788                parameters[index * 3],
789                parameters[index * 3 + 1],
790                parameters[index * 3 + 2],
791            ],
792        );
793    }
794    Ok(offsets)
795}

797fn parse_spw_selector(ms: &MeasurementSet, selector: &str) -> Result<Vec<i32>, GencalError> {
798    let count = ms
799        .spectral_window()
800        .map_err(|source| GencalError::OpenMeasurementSet {
801            path: "SPECTRAL_WINDOW".to_string(),
802            source: Box::new(source),
803        })?
804        .row_count() as i32;
805    if selector.trim().is_empty() {
806        return Ok((0..count).collect());
807    }
808    let mut ids = BTreeSet::new();
809    for part in selector.split(',').map(str::trim).filter(|p| !p.is_empty()) {
810        if let Some((start, end)) = part.split_once('~') {
811            let start = start.parse::<i32>().map_err(|_| {
812                GencalError::InvalidRequest(format!("invalid SPW selector {part:?}"))
813            })?;
814            let end = end.parse::<i32>().map_err(|_| {
815                GencalError::InvalidRequest(format!("invalid SPW selector {part:?}"))
816            })?;
817            for id in start.min(end)..=start.max(end) {
818                ids.insert(id);
819            }
820        } else {
821            ids.insert(part.parse::<i32>().map_err(|_| {
822                GencalError::InvalidRequest(format!("invalid SPW selector {part:?}"))
823            })?);
824        }
825    }
826    if ids.iter().any(|id| *id < 0 || *id >= count) {
827        return Err(GencalError::InvalidRequest(format!(
828            "SPW selector {selector:?} is outside 0..{}",
829            count.saturating_sub(1)
830        )));
831    }
832    Ok(ids.into_iter().collect())
833}

835fn matching_gaincurve(
836    table: &Table,
837    time_seconds: f64,
838    frequency_hz: f64,
839    band: Option<&str>,
840    antenna_key: &str,
841) -> Result<Vec<f32>, GencalError> {
842    if let Some(band) = band
843        && let Some(gain) = matching_gaincurve_from_rows(
844            table,
845            time_seconds,
846            |row| {
847                get_string(table, row, COL_BANDNAME)
848                    .map(|row_band| row_band == band)
849                    .unwrap_or(false)
850            },
851            antenna_key,
852        )?
853    {
854        return Ok(gain);
855    }
856    matching_gaincurve_from_rows(
857        table,
858        time_seconds,
859        |row| {
860            let Ok(bfreq) = get_f64(table, row, COL_BFREQ) else {
861                return false;
862            };
863            let Ok(efreq) = get_f64(table, row, COL_EFREQ) else {
864                return false;
865            };
866            bfreq <= frequency_hz && efreq > frequency_hz
867        },
868        antenna_key,
869    )?
870    .ok_or_else(|| {
871        GencalError::InvalidRequest(format!(
872            "no VLA GainCurves row covers antenna {antenna_key}, time {time_seconds}, frequency {frequency_hz}"
873        ))
874    })
875}
876
877fn matching_gaincurve_from_rows(
878    table: &Table,
879    time_seconds: f64,
880    mut row_matches: impl FnMut(usize) -> bool,
881    antenna_key: &str,
882) -> Result<Option<Vec<f32>>, GencalError> {
883    let nominal_key = "0";
884    let mut nominal = None;
885    for row in 0..table.row_count() {
886        if get_f64(table, row, COL_BTIME)? > time_seconds
887            || get_f64(table, row, COL_ETIME)? <= time_seconds
888            || !row_matches(row)
889        {
890            continue;
891        }
892        let antenna = get_string(table, row, COL_ANTENNA)?;
893        let gain = get_f32_array_fortran(table, row, COL_GAIN)?;
894        if antenna == antenna_key {
895            return Ok(Some(gain));
896        }
897        if antenna == nominal_key {
898            nominal = Some(gain);
899        }
900    }
901    Ok(nominal)
902}

904/// SolvableVisCal::initSVC collapses all spectral vectors for uniform
905/// parameters: mean channel frequency and summed widths/resolutions.
906/// Keep this inside the writer's staging directory, before publication.
907fn collapse_spectral_axis(path: &Path, solutions: &[PriorCalRow]) -> Result<(), GencalError> {
908    let mut table = Table::open(TableOptions::new(path))
909        .map_err(|error| GencalError::InvalidRequest(error.to_string()))?;
910    for row in 0..table.row_count() {
911        let absent = !solutions
912            .iter()
913            .any(|solution| solution.spw_id == row as i32);
914        table
915            .cell_accessor_mut(row, "FLAG_ROW")
916            .and_then(|mut cell| cell.set(Value::Scalar(ScalarValue::Bool(absent))))
917            .map_err(|e| GencalError::InvalidRequest(e.to_string()))?;
918
919        for column in ["CHAN_FREQ", "CHAN_WIDTH", "RESOLUTION", "EFFECTIVE_BW"] {
920            let ArrayValue::Float64(values) = table
921                .cell_accessor(row, column)
922                .and_then(|cell| cell.array())
923                .map_err(|error| GencalError::InvalidRequest(error.to_string()))?
924            else {
925                return Err(GencalError::InvalidRequest(format!(
926                    "{column} is not a double array"
927                )));
928            };
929            if values.is_empty() {
930                return Err(GencalError::InvalidRequest(format!(
931                    "SPECTRAL_WINDOW row {row} has empty {column}"
932                )));
933            }
934            let value = values.iter().sum::<f64>()
935                / if column == "CHAN_FREQ" {
936                    values.len() as f64
937                } else {
938                    1.
939                };
940            if column == "CHAN_WIDTH" {
941                table
942                    .cell_accessor_mut(row, "TOTAL_BANDWIDTH")
943                    .and_then(|mut cell| cell.set(Value::Scalar(ScalarValue::Float64(value.abs()))))
944                    .map_err(|error| GencalError::InvalidRequest(error.to_string()))?;
945            }
946            table
947                .cell_accessor_mut(row, column)
948                .and_then(|mut cell| {
949                    cell.set(Value::Array(ArrayValue::Float64(ArrayD::from_elem(
950                        IxDyn(&[1]),
951                        value,
952                    ))))
953                })
954                .map_err(|error| GencalError::InvalidRequest(error.to_string()))?;
955        }
956        table
957            .cell_accessor_mut(row, "NUM_CHAN")
958            .and_then(|mut cell| cell.set(Value::Scalar(ScalarValue::Int32(1))))
959            .map_err(|error| GencalError::InvalidRequest(error.to_string()))?;
960    }
961    table
962        .save(TableOptions::new(path))
963        .map_err(|source| GencalError::Save {
964            path: path.display().to_string(),
965            source: Box::new(source),
966        })
967}

1027fn vla_efficiency(frequency_ghz: f64) -> f64 {
1028    const FREQ: [f64; 42] = [
1029        1.0,
1030        1.1,
1031        1.2,
1032        1.3,
1033        1.4,
1034        1.5,
1035        1.6,
1036        1.7,
1037        1.8,
1038        1.9,
1039        2.0,
1040        2.0 + 1e-9,
1041        2.3,
1042        2.7,
1043        3.0,
1044        3.5,
1045        3.7,
1046        4.0,
1047        4.0 + 1e-9,
1048        5.0,
1049        6.0,
1050        7.0,
1051        8.0,
1052        8.0 + 1e-9,
1053        12.0,
1054        12.0 + 1e-9,
1055        13.0,
1056        14.0,
1057        15.0,
1058        16.0,
1059        17.0,
1060        18.0,
1061        19.0,
1062        24.0,
1063        26.0,
1064        26.5,
1065        28.0,
1066        33.0,
1067        38.0,
1068        40.0,
1069        43.0,
1070        48.0,
1071    ];
1072    const EFF: [f64; 42] = [
1073        0.45, 0.48, 0.48, 0.45, 0.46, 0.45, 0.43, 0.44, 0.44, 0.49, 0.48, 0.52, 0.52, 0.51, 0.53,
1074        0.55, 0.53, 0.54, 0.55, 0.54, 0.56, 0.62, 0.64, 0.60, 0.60, 0.65, 0.65, 0.62, 0.58, 0.59,
1075        0.60, 0.60, 0.57, 0.52, 0.48, 0.50, 0.49, 0.42, 0.35, 0.29, 0.28, 0.26,
1076    ];
1077    let scaled = EFF.map(|value| (value / 5.622).sqrt());
1078    if frequency_ghz <= FREQ[0] {
1079        return scaled[0];
1080    }
1081    for index in 1..FREQ.len() {
1082        if frequency_ghz <= FREQ[index] {
1083            let fraction = (frequency_ghz - FREQ[index - 1]) / (FREQ[index] - FREQ[index - 1]);
1084            return scaled[index - 1] + (scaled[index] - scaled[index - 1]) * fraction;
1085        }
1086    }
1087    scaled[scaled.len() - 1]
1088}

1143fn get_f32_array_fortran(table: &Table, row: usize, column: &str) -> Result<Vec<f32>, GencalError> {
1144    match table
1145        .cell_accessor(row, column)
1146        .and_then(|cell| cell.array())
1147        .map_err(|error| GencalError::UnsupportedTableValue(error.to_string()))?
1148    {
1149        ArrayValue::Float32(values) => {
1150            let shape = values.shape();
1151            if shape.len() != 2 {
1152                return Ok(values.iter().copied().collect());
1153            }
1154            let mut out = Vec::with_capacity(shape[0] * shape[1]);
1155            for j in 0..shape[1] {
1156                for i in 0..shape[0] {
1157                    out.push(values[[i, j]]);
1158                }
1159            }
1160            Ok(out)
1161        }
1162        other => Err(GencalError::UnsupportedTableValue(format!(
1163            "{column} row {row} has shape {:?}, expected Float32 array",
1164            other.shape()
1165        ))),
1166    }
1167}
