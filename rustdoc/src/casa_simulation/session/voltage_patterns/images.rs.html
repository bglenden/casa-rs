14const CACHE: usize = 64 << 10;
15
16fn reading_bytes(path: &Path) -> SimulatorResult<u64> {
17    let rows = casa_tables::Table::rows_on_disk(path).map_err(raised)? as u64;
18    let mut bytes = rows.saturating_mul(4096).saturating_add(256 << 10);
19    for entry in std::fs::read_dir(path).map_err(raised)? {
20        let entry = entry.map_err(raised)?;
21        let metadata = entry.metadata().map_err(raised)?;
22        if metadata.is_file() {
23            // Tiled pixel files contain fixed-width numeric data, unlike
24            // VPManager record columns. Budget a wire copy and decoded data;
25            // metadata keeps the record reader's larger expansion bound.
26            let factor = if entry.file_name().to_string_lossy().contains("_TSM") {
27                2
28            } else {
29                4
30            };
31            bytes = bytes.saturating_add(metadata.len().saturating_mul(factor));
32        }
33    }
34    Ok(bytes.saturating_add(CACHE as u64))
35}

37fn read<T: ImagePixel + Copy>(
38    session: &Simulator,
39    path: &Path,
40    convert: impl Fn(T) -> Complex32,
41) -> SimulatorResult<Admitted<ImagePlane>> {
42    let _reading = reserve(session, reading_bytes(path)?)?;
43    let image = PagedImage::<T>::open_with_cache(path, CACHE).map_err(raised)?;
44    let shape = image.shape();
45    let coords = image.coordinates();
46    if shape.len() != 4
47        || shape[0] == 0
48        || shape[1] == 0
49        || shape[2] != 4
50        || shape[3] == 0
51        || coords.find_coordinate(CoordinateType::Direction) != Some(0)
52        || coords.find_coordinate(CoordinateType::Stokes) != Some(1)
53        || coords.find_coordinate(CoordinateType::Spectral) != Some(2)
54    {
55        return Err(raised(
56            "Jones images require direction, four correlation and spectral axes",
57        ));
58    }
59    let codes = (0..4)
60        .map(|p| {
61            coords
62                .coordinate(1)
63                .to_world(&[p as f64])
64                .map(|v| v[0] as i32)
65                .map_err(raised)
66        })
67        .collect::<SimulatorResult<Vec<_>>>()?;
68    let (basis, first) = if codes.iter().all(|v| (5..=8).contains(v)) {
69        (Polarisation::Circular, 5)
70    } else if codes.iter().all(|v| (9..=12).contains(v)) {
71        (Polarisation::Linear, 9)
72    } else {
73        return Err(raised(
74            "Jones image polarization is neither circular nor linear",
75        ));
76    };
77    let positions: [usize; 4] = std::array::from_fn(|p| {
78        codes
79            .iter()
80            .position(|v| *v == first + p as i32)
81            .unwrap_or(4)
82    });
83    if positions.contains(&4) {
84        return Err(raised("Jones image must contain all four correlations"));
85    }
86    let CoordinateModel::Direction(direction) = coords.coordinate(0) else {
87        unreachable!()
88    };
89    let spectral = coords.coordinate(2);
90    let frequency_hz =
91        casa_types::quanta::Quantity::new(spectral.reference_value()[0], &spectral.axis_units()[0])
92            .and_then(|value| value.get_value_in(&casa_types::quanta::Unit::new("Hz")?))
93            .map_err(raised)?;
94    if !frequency_hz.is_finite() || frequency_hz <= 0. {
95        return Err(raised("invalid Jones image frequency"));
96    }
97    let count = shape[0]
98        .checked_mul(shape[1])
99        .ok_or_else(|| raised("Jones image size overflow"))?;
100    let bytes = (count as u64)
101        .checked_mul(size_of::<[Complex32; 4]>() as u64)
102        .ok_or_else(|| raised("Jones image size overflow"))?;
103    let mut reservation = reserve(session, 2 * bytes + (16 << 10))?;
104    let values = image
105        .get_slice(&[0, 0, 0, 0], &[shape[0], shape[1], 4, 1])
106        .map_err(raised)?;
107    let pixels: Vec<[Complex32; 4]> = (0..count)
108        .map(|i| {
109            std::array::from_fn(|p| {
110                convert(values[IxDyn(&[i % shape[0], i / shape[0], positions[p], 0])])
111            })
112        })
113        .collect();
114    if pixels
115        .iter()
116        .flatten()
117        .any(|v| !v.re.is_finite() || !v.im.is_finite())
118    {
119        return Err(raised("Jones image contains a non-finite voltage sample"));
120    }
121    reservation.retain(bytes + (16 << 10));
122    Ok(Admitted {
123        value: ImagePlane {
124            shape: [shape[0], shape[1]],
125            pixels,
126            direction: direction.clone(),
127            frequency_hz,
128            basis,
129        },
130        reservation,
131    })
132}

134pub(super) fn load(session: &Simulator, record: &RecordValue) -> SimulatorResult<Arc<JonesImage>> {
135    let (real, imaginary) = if record.get("compleximage").is_some() {
136        (
137            read::<Complex32>(session, Path::new(text(record, "compleximage")?), |v| v)?,
138            None,
139        )
140    } else {
141        let real = read::<f32>(session, Path::new(text(record, "realimage")?), |v| {
142            Complex32::new(v, 0.)
143        })?;
144        let path = Path::new(text(record, "imagimage")?);
145        let imaginary = if path.as_os_str().is_empty() || !path.exists() {
146            None
147        } else {
148            Some(read::<f32>(session, path, |v| Complex32::new(v, 0.))?)
149        };
150        if imaginary
151            .as_ref()
152            .is_some_and(|v| v.value.shape != real.value.shape)
153        {
154            return Err(raised(
155                "real and imaginary primary beam images have different shapes",
156            ));
157        }
158        (real, imaginary)
159    };
160    Ok(Arc::new(JonesImage { real, imaginary }))
161}
