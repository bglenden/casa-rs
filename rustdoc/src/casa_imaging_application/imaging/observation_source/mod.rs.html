2//! The observation's batches as a [`BoundedSource`] of native rows: the
3//! route every pass reads (#932).
4//!
5//! An observation is one or several stores (CASA's tclean images a list of
6//! MeasurementSets as one), each with its own partitions, selection and
7//! [`GeometryPlan`]. Each wave reads every store in turn, in the snapshot's
8//! order, and binds one [`AccessRequest`] per selected partition of each:
9//! the partition's selected channels that reach the wave's planes, the
10//! selected fields, and the roles the wave reads. Each batch is packed
11//! into a [`NativeBlock`] in one copy: its padding and the samples the UV
12//! selection excludes are dropped, and each kept sample's geometry comes
13//! from its store's plan. A block holds whole times of one partition.
14//! Samples the store flags (MSv2 `FLAG_ROW` included) are kept: geometry
15//! is formed for them, and the passes skip their flagged values.
16//!
17//! This module sees only the shared [`Observation`] interface; the
18//! selection was resolved against each store when the run was prepared
19//! (`compile::partitions`).

68/// One store's route: the store, its selection resolved to its partitions
69/// and its compiled geometry.
70pub(crate) struct StoreRoute {
71    pub(crate) store: RouteStore,
72    pub(crate) selection: PartitionSelection,
73    pub(crate) plan: GeometryPlan,
74    /// The bytes the pointing answers the source keeps may hold, under
75    /// which the store refuses no query for its size; zero when geometry
76    /// reads no pointing.
77    pub(crate) pointing_budget: usize,
78}

80/// What a store's route reads from: an observation, or the update session
81/// of a run that writes visibilities back.
82pub(crate) enum RouteStore {
83    Read(Box<dyn Observation>),
84    Write(Box<UpdateSession>),
85}
86
87impl StoreRoute {
88    /// The observation the route reads.
89    pub(crate) fn observation(&self) -> &dyn Observation {
90        match &self.store {
91            RouteStore::Read(observation) => observation.as_ref(),
92            RouteStore::Write(session) => session.observation(),
93        }
94    }
95
96    /// The update session of a run that writes visibilities.
97    fn session(&self) -> Option<&UpdateSession> {
98        match &self.store {
99            RouteStore::Read(_) => None,
100            RouteStore::Write(session) => Some(session),
101        }
102    }
103}

105/// One selected partition of the route: its store's and its own place.
106#[derive(Clone, Copy, Debug, PartialEq, Eq)]
107struct PartitionKey {
108    store: usize,
109    partition: usize,
110}

112/// A partition and the ordinals of its selected channels a wave reads;
113/// `None` when none reaches the wave.
114type PartitionWindow = (PartitionKey, Option<Range<usize>>);

116impl BatchRoute {
117    /// The keys of the route's stores ([`StoreKeys`]): every store's
118    /// aperture classes ([`GeometryPlan::aperture_classes`]) and where its
119    /// windows and fields start.
120    pub(crate) fn keys(&self) -> StoreKeys {
121        StoreKeys::new(
122            self.stores
123                .iter()
124                .map(|store| store.observation().catalogue().as_ref()),
125            self.stores
126                .iter()
127                .flat_map(|store| store.plan.aperture_classes())
128                .collect(),
129        )
130    }
131
132    /// The write-back of a run that writes visibilities.
133    pub(crate) fn write_back(&self) -> Option<&WriteBack> {
134        self.write_back.as_ref()
135    }
136
137    /// Every selected partition, store by store.
138    fn partitions(&self) -> impl Iterator<Item = PartitionKey> + '_ {
139        self.stores.iter().enumerate().flat_map(|(store, route)| {
140            (0..route.selection.partitions.len())
141                .map(move |partition| PartitionKey { store, partition })
142        })
143    }
144
145    /// The request reading partition `key`'s selected channels `window`
146    /// (ordinals of its selected channels), with the visibilities when
147    /// `values`, its batches holding the admitted times; and the layout of
148    /// its blocks.
149    fn request(
150        &self,
151        key: PartitionKey,
152        window: Range<usize>,
153        values: bool,
154    ) -> Result<(AccessRequest, Layout), SourceError> {
155        let route = &self.stores[key.store];
156        let partition = &route.selection.partitions[key.partition];
157        let entry = route.observation().catalogue().partition(partition.id);
158        let layout = Layout::new(partition, entry, window)?;
159        let probe = requests::payload(
160            (&route.selection, &route.plan),
161            entry,
162            partition,
163            layout.native_channels(),
164            values,
165            usize::MAX,
166        );
167        let estimate = route.observation().estimate(&probe)?;
168        let held = estimate
169            .partitions
170            .first()
171            .ok_or("an estimate without its partition")?
172            .held_bytes_per_time();
173        let request = AccessRequest {
174            batch_bytes: held
175                .saturating_mul(self.batching.times_per_batch(key.store, key.partition)),
176            ..probe
177        };
178        Ok((request, layout))
179    }
180
181    /// The most bytes the write-back of a writing pass holds beside the
182    /// passes', over every partition's traversal of its selected channels
183    /// ([`WriteBack::write_bytes`]): the largest updater's, the largest
184    /// patch's and the largest provenance's, since the sink's buffers keep
185    /// one partition's capacity while another's updater writes; zero when
186    /// the run writes nothing.
187    pub(crate) fn write_bytes(&self) -> Result<u64, SourceError> {
188        let Some(write_back) = self.write_back() else {
189            return Ok(0);
190        };
191        let charge = self
192            .partitions()
193            .try_fold(WriteCharge::default(), |most, key| {
194                let route = &self.stores[key.store];
195                let channels = route.selection.partitions[key.partition].channels.len();
196                let (request, _) = self.request(key, 0..channels, true)?;
197                let session = route
198                    .session()
199                    .ok_or("a writing run's store was opened to read only")?;
200                Ok::<_, SourceError>(most.most(write_back.write_bytes(session, &request)?))
201            })?;
202        Ok(charge.bytes())
203    }
204}

206/// The traversal of one wave.
207struct Stream<'a> {
208    values: bool,
209    /// Whether the traversal's blocks are written back: each partition it
210    /// binds and each block it fills is queued for the sink.
211    writing: bool,
212    /// Each partition's selected channels the wave reads, as ordinals of
213    /// its selected channels, store by store; `None` when none reaches the
214    /// wave.
215    windows: Vec<PartitionWindow>,
216    /// The next partition to bind.
217    next: usize,
218    /// The partition being read: its key, traversal and block layout.
219    current: Option<(PartitionKey, Traversal<'a>, Layout)>,
220    /// The geometry state carried from batch to batch of one store, and
221    /// that store.
222    geometry: (GeometryTraversal, Option<usize>),
223}
224
225enum State<'a> {
226    Idle,
227    Streaming(Box<Stream<'a>>),
228    Failed,
229}

231/// The observation's selected samples, projected onto every chart.
232///
233/// A restricted cube wave reads only the selected channels whose
234/// output-frame frequencies reach its planes at some selected time, with
235/// both interpolation partners at each edge (`window`); a compiled
236/// continuum transform fits each whole row, so its traversals are never
237/// restricted. A flags-and-weights traversal does not read the
238/// visibilities: its values are zero, and a continuum transform only flags
239/// the channels it keeps out of the output. A non-finite input in an
240/// unflagged sample fails the run with the sample named; one in a flagged
241/// sample is ignored.
242pub(crate) struct ObservationSource<'a> {
243    route: &'a BatchRoute,
244    packer: Packer<'a>,
245    planes: u32,
246    bounds: Option<PlaneBounds>,
247    /// Each store's partitions surveyed: at the first traversal when
248    /// geometry reads pointing, with the conversion factors at the first
249    /// restricted wave.
250    surveys: Option<Vec<Vec<PartitionSurvey>>>,
251    /// Each store's pointing answers.
252    pointing: Vec<PointingCache>,
253    maximum: BlockShape,
254    state: State<'a>,
255    buffers: BatchBuffers,
256    geometry: BatchGeometry,
257    keep: Vec<bool>,
258    scratch: PackScratch,
259}
260
261impl<'a> ObservationSource<'a> {
262    /// A source over `route` for `problem`, delivering rows projected on
263    /// every chart. `bounds` enables restricted traversals for a basis of
264    /// `planes` planes; `pointing_ramp` carries each row's pointing offset
265    /// for a kernel set that ramps to it.
266    pub(crate) fn new(
267        problem: &'a CompiledProblem,
268        route: &'a BatchRoute,
269        planes: u32,
270        bounds: Option<PlaneBounds>,
271        pointing_ramp: bool,
272    ) -> Self {
273        let layout = row_layout(problem);
274        let pointing: Vec<DirectionCoordinateSpec> = if pointing_ramp {
275            problem
276                .geometry()
277                .domains()
278                .iter()
279                .flat_map(|domain| std::iter::repeat_n(domain.direction(), domain.facets().len()))
280                .collect()
281        } else {
282            Vec::new()
283        };
284        Self {
285            route,
286            packer: Packer {
287                problem,
288                plans: route.stores.iter().map(|store| &store.plan).collect(),
289                charts: layout.domains,
290                pointing,
291                keys: route.keys(),
292            },
293            planes,
294            bounds,
295            surveys: None,
296            pointing: route
297                .stores
298                .iter()
299                .map(|_| PointingCache::default())
300                .collect(),
301            maximum: BlockShape {
302                rows: route.batching.rows(),
303                ..layout
304            },
305            state: State::Idle,
306            buffers: BatchBuffers::default(),
307            geometry: BatchGeometry::default(),
308            keep: Vec::new(),
309            scratch: PackScratch::default(),
310        }
311    }
312
313    /// Each partition's selected channels the wave over `planes` reads.
314    fn windows(
315        &mut self,
316        planes: PlaneRange,
317        restrict: bool,
318    ) -> Result<Vec<PartitionWindow>, SourceError> {
319        let problem = self.packer.problem;
320        let route: &'a BatchRoute = self.route;
321        let bounds = (restrict
322            && problem.visibility_transform().is_none()
323            && planes != PlaneRange::new(0, self.planes))
324        .then(|| self.bounds.as_ref().map(|bounds| bounds(planes)))
325        .flatten();
326        let Some(bounds) = bounds else {
327            return Ok(route
328                .partitions()
329                .map(|key| {
330                    let partition = &route.stores[key.store].selection.partitions[key.partition];
331                    (key, Some(0..partition.channels.len()))
332                })
333                .collect());
334        };
335        if !(bounds[0].is_finite() && bounds[1].is_finite() && bounds[0] != bounds[1]) {
336            return Err("a wave's frequency bounds are empty or not finite".into());
337        }
338        let surveys = self.surveyed(true)?;
339        Ok(route
340            .partitions()
341            .map(|key| {
342                let store = &route.stores[key.store];
343                let partition = &store.selection.partitions[key.partition];
344                let factors = surveys[key.store][key.partition]
345                    .factors
346                    .expect("factors surveyed");
347                let entry = store.observation().catalogue().partition(partition.id);
348                let native = partition
349                    .channels
350                    .iter()
351                    .map(|channel| entry.frequencies_hz[*channel as usize])
352                    .collect::<Vec<_>>();
353                let window = (!factors.is_empty())
354                    .then(|| wave_window(&native, factors, bounds))
355                    .flatten();
356                (key, window)
357            })
358            .collect())
359    }
360
361    /// Each store's partitions surveyed, with their conversion factors
362    /// when `factors`: surveyed again only when factors are first needed.
363    fn surveyed(&mut self, factors: bool) -> Result<&[Vec<PartitionSurvey>], SourceError> {
364        let current = self.surveys.as_ref().is_some_and(|surveys| {
365            !factors
366                || surveys
367                    .iter()
368                    .flatten()
369                    .all(|survey| survey.factors.is_some())
370        });
371        if !current {
372            let survey_bytes = self.route.batching.survey_bytes();
373            self.surveys = Some(
374                self.route
375                    .stores
376                    .iter()
377                    .map(|store| survey(store, survey_bytes, factors))
378                    .collect::<Result<_, _>>()?,
379            );
380        }
381        Ok(self.surveys.as_deref().expect("partitions surveyed"))
382    }
383
384    /// Bind partition `key`'s traversal of the wave's channels `window`,
385    /// its batches holding the admitted times; a writing traversal queues
386    /// the request for its sink.
387    fn bind(
388        &self,
389        key: PartitionKey,
390        window: Range<usize>,
391        (values, writing): (bool, bool),
392    ) -> Result<(Traversal<'a>, Layout), SourceError> {
393        let route: &'a BatchRoute = self.route;
394        let (request, layout) = route.request(key, window, values)?;
395        let traversal = route.stores[key.store].observation().bind_read(&request)?;
396        if writing && let Some(write_back) = route.write_back() {
397            write_back.deliver(Delivered::Partition {
398                store: key.store,
399                request,
400            });
401        }
402        Ok((traversal, layout))
403    }
404
405    /// Hold partition `key`'s pointing over its selected times, when its
406    /// store's geometry reads pointing.
407    fn hold_pointing(&mut self, key: PartitionKey) -> Result<(), SourceError> {
408        let route: &'a BatchRoute = self.route;
409        let store = &route.stores[key.store];
410        if !store.plan.reads_pointing() {
411            return Ok(());
412        }
413        let Some(span) = self.surveyed(false)?[key.store][key.partition].span else {
414            self.pointing[key.store].release();
415            return Ok(());
416        };
417        self.pointing[key.store].hold(
418            store.observation(),
419            store.pointing_budget,
420            (
421                pointing_direction(self.packer.problem),
422                store.selection.partitions[key.partition].id,
423                span,
424            ),
425        )
426    }

428    /// Read on until a batch packs a row into `out`; `Ok(false)` once every
429    /// partition is read.
430    fn advance(
431        &mut self,
432        stream: &mut Stream<'a>,
433        out: &mut NativeBlock,
434    ) -> Result<bool, SourceError> {
435        loop {
436            if stream.current.is_none() {
437                let Some(next) = (stream.next..stream.windows.len())
438                    .find(|index| stream.windows[*index].1.is_some())
439                else {
440                    return Ok(false);
441                };
442                let (key, window) = stream.windows[next].clone();
443                let window = window.expect("a window");
444                let (traversal, layout) =
445                    self.bind(key, window, (stream.values, stream.writing))?;
446                if stream.geometry.1 != Some(key.store) {
447                    // Each store's plan carries its own geometry state.
448                    stream.geometry = (GeometryTraversal::default(), Some(key.store));
449                }
450                stream.next = next + 1;
451                stream.current = Some((key, traversal, layout));
452                self.hold_pointing(key)?;
453            }
454            let (key, traversal, layout) = stream.current.as_mut().expect("a bound partition");
455            let route: &'a BatchRoute = self.route;
456            let store = &route.stores[key.store];
457            let partition = &store.selection.partitions[key.partition];
458            let Some(event) = traversal.next(&mut self.buffers)? else {
459                stream.current = None;
460                continue;
461            };
462            let ReadEvent::Batch(batch) = event else {
463                continue;
464            };
465            select_samples(&batch, &store.selection, partition, &mut self.keep)?;
466            if !self.keep.iter().any(|keep| *keep) {
467                continue;
468            }
469            store.plan.apply(
470                &batch,
471                &self.keep,
472                self.pointing[key.store].series(),
473                &mut stream.geometry.0,
474                &mut self.geometry,
475            )?;
476            self.packer.pack(
477                (&batch, key.store, partition, layout),
478                &self.geometry,
479                &self.keep,
480                stream.values,
481                &mut self.scratch,
482                out,
483            )?;
484            if stream.writing
485                && let Some(write_back) = route.write_back()
486            {
487                // The block's rows are the kept samples, in order.
488                let mut samples = Vec::with_capacity(batch.samples());
489                samples.extend(
490                    (0..batch.samples())
491                        .filter(|sample| self.keep[*sample])
492                        .map(|sample| u32::try_from(sample).expect("a batch's samples fit u32")),
493                );
494                write_back.deliver(Delivered::Block(Provenance {
495                    address: batch.address(),
496                    samples,
497                    spectral_window: self
498                        .packer
499                        .keys
500                        .window(key.store, partition.spectral_window),
501                }));
502            }
503            return Ok(true);
504        }
505    }
506}

508impl BoundedSource for ObservationSource<'_> {
509    fn begin(
510        &mut self,
511        planes: PlaneRange,
512        restrict: bool,
513        content: SampleContent,
514    ) -> Result<(), SourceError> {
515        if !matches!(self.state, State::Idle) {
516            return Err("a traversal began before the previous one finished".into());
517        }
518        self.state = State::Failed;
519        let windows = self.windows(planes, restrict)?;
520        let values = content == SampleContent::Visibilities;
521        let writing = match self.route.write_back() {
522            Some(write_back) if values => write_back.writing()?,
523            _ => false,
524        };
525        self.state = State::Streaming(Box::new(Stream {
526            values,
527            writing,
528            windows,
529            next: 0,
530            current: None,
531            geometry: (GeometryTraversal::default(), None),
532        }));
533        Ok(())
534    }
535
536    fn fill(&mut self, out: &mut NativeBlock) -> Result<bool, SourceError> {
537        let State::Streaming(mut stream) = std::mem::replace(&mut self.state, State::Failed) else {
538            return Err("no traversal is in progress".into());
539        };
540        let filled = self.advance(&mut stream, out)?;
541        self.state = if filled {
542            State::Streaming(stream)
543        } else {
544            State::Idle
545        };
546        Ok(filled)
547    }
548
549    fn maximum_block(&self) -> BlockShape {
550        self.maximum
551    }
