3//! Thin PyO3 `ia.insert` wrapper delegating to [`casa_imageanalysis::tool::ops_edit::insert_image`].
4
5use casa_imageanalysis::tool::ops_edit::insert_image;
6use pyo3::prelude::*;
7
8use crate::convert::DoubleList;
9
10use super::PyImageTool;
11use super::handle::{emit_tool_logs, raised};
12use super::subimage::Region;
13
14#[pymethods]
15impl PyImageTool {
16    #[pyo3(signature = (infile="", region=None, locate=None, verbose=false), text_signature = "($self, infile='', region={}, locate=[-1.0], verbose=False)")]
17    fn insert(
18        &self,
19        py: Python<'_>,
20        infile: &str,
21        region: Option<&Bound<'_, PyAny>>,
22        locate: Option<DoubleList>,
23        verbose: bool,
24    ) -> PyResult<bool> {
25        if self.state().image.is_none() {
26            return Ok(false);
27        }
28        let region = Region::argument(py, region, "insert")?;
29        let locate = locate.map_or_else(|| vec![-1.0], |values| values.0);
30        let (outcome, logs) = py.detach(|| -> PyResult<_> {
31            let state = self.state();
32            let Some(attachment) = &state.image else {
33                return Ok((false, Vec::new()));
34            };
35            let history_command = state.history.then(|| {
36                let region_hist = region.as_ref().map_or_else(|| "{...}".into(), Region::history);
37                let location = locate
38                    .iter()
39                    .map(|value| format!("{value:.6}"))
40                    .collect::<Vec<_>>()
41                    .join(",");
42                format!("ia.insert(infile=\"{infile}\", region={region_hist}, locate=[{location}], verbose={verbose})")
43            });
44            insert_image(
45                attachment,
46                infile,
47                region.as_ref().map(Region::as_region),
48                &locate,
49                verbose,
50                history_command,
51            )
52            .map_err(raised)
53        })?;
54        emit_tool_logs(py, &logs)?;
55        Ok(outcome)
56    }
57}
