15struct ChannelMatrix<'a, T: Clone> {
16    values: Cow<'a, [T]>,
17    strides: [usize; 2],
18}
19
20impl<'a, T: Copy> ChannelMatrix<'a, T> {
21    fn new(matrix: ArrayView2<'a, T>) -> Self {
22        let corr = matrix.strides()[0];
23        let chan = matrix.strides()[1];
24        // Keep either contiguous axis order. A sliced, non-contiguous cell
25        // is copied once in logical order before the inner averaging loop.
26        if let (Ok(corr), Ok(chan), Some(values)) = (
27            usize::try_from(corr),
28            usize::try_from(chan),
29            matrix.to_slice_memory_order(),
30        ) {
31            return Self {
32                values: Cow::Borrowed(values),
33                strides: [corr, chan],
34            };
35        }
36        Self {
37            values: Cow::Owned(matrix.iter().copied().collect()),
38            strides: [matrix.ncols(), 1],
39        }
40    }
41
42    #[inline(always)]
43    fn get(&self, corr: usize, chan: usize) -> T {
44        self.values[corr * self.strides[0] + chan * self.strides[1]]
45    }
46}

48#[derive(Clone, Copy, PartialEq, Eq)]
49pub(crate) enum WeightKind {
50    Observed,
51    Calibrated,
52    Unit,
53}
54
55pub(crate) fn visibility_weight_kind(
56    column: OptionalMainColumn,
57    has_observed: bool,
58    has_corrected: bool,
59    time_averaged: bool,
60) -> WeightKind {
61    use OptionalMainColumn::{Data, FloatData, LagData, ModelData};
62    match column {
63        Data => WeightKind::Observed,
64        FloatData if time_averaged && has_corrected => WeightKind::Observed,
65        ModelData if has_corrected => WeightKind::Calibrated,
66        ModelData if has_observed => WeightKind::Observed,
67        ModelData | LagData => WeightKind::Unit,
68        _ => WeightKind::Calibrated,
69    }
70}

72pub(super) struct InputStatistic<'a> {
73    scalar: ArrayView1<'a, f32>,
74    spectrum: Option<ChannelMatrix<'a, f32>>,
75}

77#[derive(Clone, Copy, Default)]
78pub(super) struct SpectrumCells<'a> {
79    pub(super) weight: Option<&'a ArrayValue>,
80    pub(super) sigma: Option<&'a ArrayValue>,
81}
82
83impl<'a> InputStatistic<'a> {
84    pub(super) fn from_cells(
85        scalar: &'a ArrayValue,
86        spectrum: Option<&'a ArrayValue>,
87        shape: [usize; 2],
88        label: &str,
89    ) -> Result<Self, TableError> {
90        let ArrayValue::Float32(scalar) = scalar else {
91            return Err(TableError::Schema(format!("{label} must hold Float32")));
92        };
93        let scalar = scalar
94            .view()
95            .into_dimensionality::<Ix1>()
96            .ok()
97            .filter(|value| value.len() == shape[0])
98            .ok_or_else(|| {
99                TableError::Schema(format!("{label} does not match the correlation axis"))
100            })?;
101        let invalid_spectrum = || {
102            TableError::Schema(format!(
103                "{label}_SPECTRUM does not match FLAG's correlation and channel axes"
104            ))
105        };
106        let spectrum = match spectrum {
107            None => None,
108            Some(ArrayValue::Float32(spectrum)) if spectrum.is_empty() => None,
109            Some(ArrayValue::Float32(spectrum)) if spectrum.shape() == shape => {
110                Some(ChannelMatrix::new(
111                    spectrum
112                        .view()
113                        .into_dimensionality::<Ix2>()
114                        .map_err(|_| invalid_spectrum())?,
115                ))
116            }
117            Some(_) => return Err(invalid_spectrum()),
118        };
119        Ok(Self { scalar, spectrum })
120    }
121
122    pub(super) fn value(&self, corr: usize, chan: usize) -> f32 {
123        self.spectrum
124            .as_ref()
125            .map_or_else(|| self.scalar[corr], |array| array.get(corr, chan))
126    }
127}

136pub(super) struct ChannelAverages<'a> {
137    flags: ChannelMatrix<'a, bool>,
138    input_shape: (usize, usize),
139    bins: &'a [Vec<usize>],
140    bin_statistics: Array2<BinStatistics>,
141    sigma: InputStatistic<'a>,
142    weight: InputStatistic<'a>,
143    data: bool,
144    corrected: bool,
145    model: bool,
146    floating: bool,
147    main: OptionalMainColumn,
148    propagate_weights: bool,
149    time_averaged: bool,
150}
151
152struct BinStatistics {
153    flagged: bool,
154    weights: [f32; 3],
155}
156
157impl<'a> ChannelAverages<'a> {
158    pub(super) fn new(
159        flags: &'a ArrayValue,
160        bins: &'a [Vec<usize>],
161        weight: &'a ArrayValue,
162        sigma: &'a ArrayValue,
163        spectra: SpectrumCells<'a>,
164        columns: &VisibilityColumns,
165        (propagate_weights, time_averaged): (bool, bool),
166    ) -> Result<Self, TableError> {
167        use OptionalMainColumn::{CorrectedData, Data, FloatData, LagData, ModelData};
168
169        let ArrayValue::Bool(flags) = flags else {
170            return Err(TableError::Schema("FLAG must hold Bool".to_string()));
171        };
172        let flags = flags
173            .view()
174            .into_dimensionality::<Ix2>()
175            .map_err(|_| TableError::Schema("FLAG must be rank-2 [corr, chan]".to_string()))?;
176        let (correlations, channels) = flags.dim();
177        let flags = ChannelMatrix::new(flags);
178        super::validate_channel_bins(bins, channels)?;
179        let shape = [correlations, channels];
180        let weight = InputStatistic::from_cells(weight, spectra.weight, shape, "WEIGHT")?;
181        let sigma = InputStatistic::from_cells(sigma, spectra.sigma, shape, "SIGMA")?;
182        let has = |source| columns.copies().iter().any(|copy| copy.source == source);
183        let data = has(Data);
184        let corrected = has(CorrectedData);
185        let model = has(ModelData);
186        let floating = has(FloatData);
187        let main = [Data, CorrectedData, ModelData, FloatData, LagData]
188            .into_iter()
189            .find(|&column| has(column))
190            .ok_or_else(|| {
191                TableError::Schema("channel averaging requires a visibility column".to_string())
192            })?;
193        let bin_statistics = Array2::from_shape_fn((correlations, bins.len()), |(corr, bin)| {
194            let flagged = bins[bin].iter().all(|&chan| flags.get(corr, chan));
195            let mut weights = [0.0; 3];
196            for &chan in &bins[bin] {
197                if flags.get(corr, chan) == flagged {
198                    weights[WeightKind::Observed as usize] +=
199                        weight_from_sigma(sigma.value(corr, chan));
200                    weights[WeightKind::Calibrated as usize] += weight.value(corr, chan);
201                    weights[WeightKind::Unit as usize] += 1.0;
202                }
203            }
204            BinStatistics { flagged, weights }
205        });
206        Ok(Self {
207            flags,
208            input_shape: (correlations, channels),
209            bins,
210            bin_statistics,
211            sigma,
212            weight,
213            data,
214            corrected,
215            model,
216            floating,
217            main,
218            propagate_weights,
219            time_averaged,
220        })
221    }
222
223    pub(super) fn visibility(
224        &self,
225        column: OptionalMainColumn,
226        value: ArrayValue,
227    ) -> Result<ArrayValue, TableError> {
228        let weight = self.visibility_weight(column);
229        match value {
230            ArrayValue::Complex32(array) => self
231                .mean_visibility(&array, weight)
232                .map(ArrayValue::Complex32),
233            ArrayValue::Float32(array) => self
234                .mean_visibility(&array, weight)
235                .map(ArrayValue::Float32),
236            _ => Err(TableError::Schema(
237                "visibility arrays must hold Complex32 or Float32".to_string(),
238            )),
239        }
240    }
241
242    pub(super) fn flags(&self) -> ArrayValue {
243        let kind = self.visibility_weight(self.main);
244        let output = self.bin_statistics.map(|bin| {
245            let weight = bin.weights[kind as usize];
246            bin.flagged || weight <= 0.0 || weight.is_nan()
247        });
248        ArrayValue::Bool(output.into_dyn())
249    }
250
251    pub(super) fn statistics(&self) -> AveragedStatistics {
252        let (weight, variance) = self.weight_spectra();
253        let sigma = variance.mapv(sigma_from_weight);
254        AveragedStatistics {
255            weight: ArrayValue::Float32(self.mean_bins(&weight).into_dyn()),
256            sigma: ArrayValue::Float32(self.mean_bins(&sigma).into_dyn()),
257            weight_spectrum: ArrayValue::Float32(weight.into_dyn()),
258            sigma_spectrum: ArrayValue::Float32(sigma.into_dyn()),
259        }
260    }
261
262    pub(super) fn weight_spectra(&self) -> (Array2<f32>, Array2<f32>) {
263        let observed = self.data || (self.time_averaged && self.floating);
264        if self.time_averaged && !(observed && self.corrected) {
265            let weights = self.summed_weights(WeightKind::Calibrated);
266            (weights.clone(), weights)
267        } else if self.corrected {
268            let calibrated = self.summed_weights(WeightKind::Calibrated);
269            let observed = if observed {
270                self.summed_weights(WeightKind::Observed)
271            } else {
272                calibrated.clone()
273            };
274            (calibrated, observed)
275        } else {
276            let kind = if self.data {
277                WeightKind::Observed
278            } else if self.model {
279                WeightKind::Unit
280            } else {
281                WeightKind::Calibrated
282            };
283            let weights = self.summed_weights(kind);
284            (weights.clone(), weights)
285        }
286    }
287
288    pub(super) fn bin_flags(&self) -> Array2<bool> {
289        self.bin_statistics.map(|bin| bin.flagged)
290    }
291
292    fn visibility_weight(&self, column: OptionalMainColumn) -> WeightKind {
293        // CASA uses the scalar statistics for output weights and sigmas, but
294        // uses an unweighted visibility average when weight spectra are absent.
295        if !self.propagate_weights {
296            return WeightKind::Unit;
297        }
298        visibility_weight_kind(column, self.data, self.corrected, self.time_averaged)
299    }
300
301    fn weight_at(&self, kind: WeightKind, corr: usize, chan: usize) -> f32 {
302        match kind {
303            WeightKind::Observed => weight_from_sigma(self.sigma.value(corr, chan)),
304            WeightKind::Calibrated => self.weight.value(corr, chan),
305            WeightKind::Unit => 1.0,
306        }
307    }
308
309    fn summed_weights(&self, kind: WeightKind) -> Array2<f32> {
310        self.bin_statistics.map(|bin| bin.weights[kind as usize])
311    }
312
313    fn mean_visibility<T>(
314        &self,
315        values: &ArrayD<T>,
316        kind: WeightKind,
317    ) -> Result<ArrayD<T>, TableError>
318    where
319        T: Copy + Default + AddAssign + Div<f32, Output = T> + Mul<f32, Output = T>,
320    {
321        let values = values
322            .view()
323            .into_dimensionality::<Ix2>()
324            .ok()
325            .filter(|values| values.dim() == self.input_shape)
326            .ok_or_else(|| TableError::Schema("visibility and FLAG shapes differ".to_string()))?;
327        let values = ChannelMatrix::new(values);
328        Ok(
329            Array2::from_shape_fn(self.bin_statistics.raw_dim(), |(corr, bin)| {
330                let statistics = &self.bin_statistics[[corr, bin]];
331                let normalization = statistics.weights[kind as usize];
332                if normalization > 0.0 {
333                    let mut sum = T::default();
334                    for &chan in &self.bins[bin] {
335                        if self.flags.get(corr, chan) == statistics.flagged {
336                            sum += values.get(corr, chan) * self.weight_at(kind, corr, chan);
337                        }
338                    }
339                    sum / normalization
340                } else {
341                    T::default()
342                }
343            })
344            .into_dyn(),
345        )
346    }
347
348    fn mean_bins(&self, values: &Array2<f32>) -> Array1<f32> {
349        Array1::from_shape_fn(values.nrows(), |corr| {
350            mean_weight_preferring_unflagged(
351                self.bin_statistics
352                    .row(corr)
353                    .iter()
354                    .zip(values.row(corr).iter().copied())
355                    .map(|(bin, value)| (bin.flagged, value)),
356            )
357        })
358    }
359}

361/// Reduce channel weights, using flagged channels only when all are flagged.
362pub(crate) fn mean_weight_preferring_unflagged(bins: impl Iterator<Item = (bool, f32)>) -> f32 {
363    let mut sums = [0.0, 0.0];
364    let mut counts = [0, 0];
365    for (flagged, value) in bins {
366        let index = usize::from(flagged);
367        sums[index] += value;
368        counts[index] += 1;
369    }
370    let index = usize::from(counts[0] == 0);
371    sums[index] / counts[index].max(1) as f32
372}

374pub(super) fn unaveraged_spectra(
375    weight: &ArrayValue,
376    sigma: &ArrayValue,
377    spectra: SpectrumCells<'_>,
378    flags: &ArrayValue,
379    bins: &[Vec<usize>],
380    corrected_becomes_data: bool,
381) -> Result<(ArrayValue, ArrayValue), TableError> {
382    let &[corr, chan] = flags.shape() else {
383        return Err(TableError::Schema(
384            "FLAG must be rank-2 [corr, chan]".to_string(),
385        ));
386    };
387    super::validate_channel_bins(bins, chan)?;
388    let weight = InputStatistic::from_cells(weight, spectra.weight, [corr, chan], "WEIGHT")?;
389    let sigma = InputStatistic::from_cells(sigma, spectra.sigma, [corr, chan], "SIGMA")?;
390    let weight = Array2::from_shape_fn((corr, bins.len()), |(corr, bin)| {
391        weight.value(corr, bins[bin][0])
392    });
393    let sigma = if corrected_becomes_data {
394        weight.mapv(sigma_from_weight)
395    } else {
396        Array2::from_shape_fn((corr, bins.len()), |(corr, bin)| {
397            sigma.value(corr, bins[bin][0])
398        })
399    };
400    Ok((
401        ArrayValue::Float32(weight.into_dyn()),
402        ArrayValue::Float32(sigma.into_dyn()),
403    ))
404}

406pub(crate) fn weight_from_sigma(sigma: f32) -> f32 {
407    if sigma > f32::MIN_POSITIVE {
408        1.0 / (sigma * sigma)
409    } else {
410        0.0
411    }
412}
413
414pub(crate) fn sigma_from_weight(weight: f32) -> f32 {
415    if weight > f32::MIN_POSITIVE {
416        1.0 / weight.sqrt()
417    } else {
418        -1.0
419    }
