2//! Recover FTMachine's moving-source rotation from a stored Fourier model.
3
4use casa_ms::{EphemerisBudget, PhaseCentreEphemeris, PlanetaryEphemeris};
5use casa_types::measures::MeasFrame;
6use casa_types::measures::direction::{DirectionRef, MDirection};
7use casa_types::measures::position::MPosition;
8use casa_types::measures::record::{direction_from_record, position_from_record};
9use casa_types::{RecordValue, ScalarValue, Value};
10
11use crate::ft::moving_source::ObservedSource;
12
13use super::{FourierRow, GainSolveError, model_error, record_number, record_text};
14
15pub(super) struct Tracking {
16    source: Source,
17    first_direction: MDirection,
18    image_direction: MDirection,
19    position: MPosition,
20    fixed_epoch: Option<f64>,
21    last_shift: Option<([u64; 3], [f64; 2])>,
22}
23
24enum Source {
25    Field,
26    Ephemeris(PhaseCentreEphemeris),
27    Legacy(ObservedSource),
28    Direction(MDirection),
29    Named(Box<PlanetaryEphemeris>),
30}
31
32impl Tracking {
33    pub(super) fn load(record: &RecordValue) -> Result<Option<Self>, GainSolveError> {
34        if !matches!(
35            record.get("fixmovingsource"),
36            Some(Value::Scalar(ScalarValue::Bool(true)))
37        ) {
38            return Ok(None);
39        }
40        let moving = measure(record, "movingdir_rec")?;
41        let reference = record_text(moving, "refer")?;
42        let source = match reference {
43            "COMET" => Source::Ephemeris(
44                PhaseCentreEphemeris::external(
45                    record_text(record, "ephemeristable")?,
46                    EphemerisBudget::new(64 << 20),
47                )
48                .map_err(model_error)?,
49            ),
50            _ if record.get("doconversion").is_some() => Source::Legacy(
51                ObservedSource::new(direction_from_record(moving).map_err(model_error)?)
52                    .map_err(model_error)?,
53            ),
54            "APP" => Source::Field,
55            _ if reference.parse::<DirectionRef>().is_ok() => {
56                Source::Direction(direction_from_record(moving).map_err(model_error)?)
57            }
58            _ => Source::Named(Box::new(
59                PlanetaryEphemeris::open_discovered(reference).map_err(model_error)?,
60            )),
61        };
62        Ok(Some(Self {
63            source,
64            first_direction: direction_from_record(measure(record, "firstmovingdir_rec")?)
65                .map_err(model_error)?,
66            image_direction: direction_from_record(measure(record, "mimage_rec")?)
67                .map_err(model_error)?,
68            position: position_from_record(measure(record, "mlocation_rec")?)
69                .map_err(model_error)?,
70            fixed_epoch: record_number(record, "phasecentertime").filter(|time| *time > 0.0),
71            last_shift: None,
72        }))
73    }
74
75    pub(super) fn observe(
76        &mut self,
77        row: &FourierRow<'_>,
78        virtual_anchor: Option<&mut Option<MDirection>>,
79    ) -> Result<(MeasFrame, [f64; 2]), GainSolveError> {
80        let frame = row
81            .engine
82            .spectral_frame_observatory_direction(row.row.time_seconds, row.field.clone())
83            .map_err(model_error)?
84            .with_position(self.position.clone());
85        let shift = self.shift(row, &frame, virtual_anchor)?;
86        Ok((frame, shift))
87    }
88
89    fn shift(
90        &mut self,
91        row: &FourierRow<'_>,
92        frame: &MeasFrame,
93        virtual_anchor: Option<&mut Option<MDirection>>,
94    ) -> Result<[f64; 2], GainSolveError> {
95        let key = [
96            row.row.time_seconds.to_bits(),
97            row.phase_time.to_bits(),
98            u64::from(row.row.field_id as u32),
99        ];
100        if let Some((previous, shift)) = self.last_shift
101            && previous == key
102        {
103            return Ok(shift);
104        }
105        let source = match &self.source {
106            Source::Field => row
107                .engine
108                .tracked_body_direction_j2000(
109                    self.fixed_epoch.unwrap_or(row.phase_time),
110                    row.row.field_id as usize,
111                )
112                .map_err(model_error)?,
113            Source::Ephemeris(ephemeris) => row
114                .engine
115                .ephemeris_direction_j2000(
116                    row.row.time_seconds,
117                    row.row.field_id as usize,
118                    "COMET",
119                    ephemeris,
120                )
121                .map_err(model_error)?,
122            Source::Direction(direction) => direction.clone(),
123            Source::Named(planet) => planet.direction(frame).map_err(model_error)?,
124            Source::Legacy(source) => source
125                .direction(frame, self.image_direction.refer())
126                .map_err(model_error)?,
127        };
128        let source = source
129            .convert_to(self.image_direction.refer(), frame)
130            .map_err(model_error)?;
131        let (longitude, latitude) = source.as_angles();
132        let anchor = match virtual_anchor {
133            // VisModelData calls initMaps on its first observing buffer.
134            Some(anchor) if self.is_legacy() => anchor.get_or_insert(source),
135            _ => &self.first_direction,
136        };
137        let (anchor_lon, anchor_lat) = anchor.as_angles();
138        let shift = [longitude - anchor_lon, latitude - anchor_lat];
139        self.last_shift = Some((key, shift));
140        Ok(shift)
141    }
142
143    pub(super) fn centre(&self, shift: [f64; 2]) -> MDirection {
144        let (image_lon, image_lat) = self.image_direction.as_angles();
145        let sign = if self.is_legacy() {
146            // The legacy rotateUVW phases on image + first - current.
147            -1.
148        } else {
149            1.
150        };
151        MDirection::from_angles(
152            image_lon + sign * shift[0],
153            image_lat + sign * shift[1],
154            self.image_direction.refer(),
155        )
156    }
157
158    pub(super) fn is_legacy(&self) -> bool {
159        matches!(self.source, Source::Legacy(_))
160    }
161}

163/// CASA indexes the correlator directions by integration time. A saved
164/// positive epoch uses the first integration at or after it, clamped to
165/// the last integration when it lies beyond the MeasurementSet.
166fn measure<'a>(record: &'a RecordValue, name: &str) -> Result<&'a RecordValue, GainSolveError> {
167    match record.get(name) {
168        Some(Value::Record(measure)) => Ok(measure),
169        _ => Err(model_error(format!(
170            "virtual tracking model has no {name} measure"
171        ))),
172    }
173}
