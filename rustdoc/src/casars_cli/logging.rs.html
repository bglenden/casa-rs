21/// CLI flag that enables CASA log-table output.
22pub const LOG_TABLE_FLAG: &str = "--log-table";
23/// CLI flag that controls the CASA log-table minimum priority.
24pub const LOG_TABLE_PRIORITY_FLAG: &str = "--log-table-priority";
25/// CLI flag that controls stderr logging, or disables it with `off`.
26pub const LOG_STDERR_PRIORITY_FLAG: &str = "--log-stderr-priority";
27
28/// Errors produced by CLI logging flag/environment parsing or subscriber setup.
29#[derive(Debug, Error)]
30pub enum CliLoggingError {
31    /// Missing value for a CLI logging flag.
32    #[error("missing value for {0}")]
33    MissingOptionValue(&'static str),
34    /// Non-UTF-8 value for a CLI logging flag.
35    #[error("invalid non-UTF-8 value for {0}")]
36    NonUtf8OptionValue(&'static str),
37    /// Invalid priority string in `CASA_RS_LOG_*` environment variables.
38    #[error("invalid {name} priority {value:?}")]
39    InvalidEnvPriority {
40        /// Environment variable name.
41        name: &'static str,
42        /// Invalid value text.
43        value: String,
44    },
45    /// Invalid priority or sink setup error from `casa-logging`.
46    #[error(transparent)]
47    Logging(#[from] CasaLoggingError),
48    /// Failed to install the global `tracing` subscriber.
49    #[error("tracing subscriber setup failed: {0}")]
50    Subscriber(String),
51}

53/// Application-level logging configuration for CLI binaries.
54#[derive(Debug, Clone)]
55pub struct CasaLoggingConfig {
56    /// Optional path to a CASA LOG table.
57    pub table_path: Option<PathBuf>,
58    /// Minimum priority written to the CASA LOG table.
59    pub table_priority: CasaPriority,
60    /// Minimum priority written to stderr (`None` disables stderr output).
61    pub stderr_priority: Option<CasaPriority>,
62}
63
64impl Default for CasaLoggingConfig {
65    fn default() -> Self {
66        Self::from_env().unwrap_or_else(|_| Self::fallback())
67    }
68}
69
70impl CasaLoggingConfig {
71    /// Whether the environment or command line explicitly selects an output sink.
72    ///
73    /// A priority for a log table alone does not select a table. An explicit
74    /// stderr priority (including `off`) or a table path does select output.
75    #[must_use]
76    pub fn output_was_configured(args: &[OsString]) -> bool {
77        output_was_configured(
78            args,
79            std::env::var_os("CASA_RS_LOG_TABLE").is_some(),
80            std::env::var_os("CASA_RS_LOG_STDERR_PRIORITY").is_some(),
81        )
82    }
83
84    /// Build logging configuration from `CASA_RS_LOG_*` environment variables.
85    ///
86    /// # Errors
87    /// Returns [`CliLoggingError::InvalidEnvPriority`] if an environment
88    /// variable holds an unknown priority name.
89    pub fn from_env() -> Result<Self, CliLoggingError> {
90        Ok(Self {
91            table_path: std::env::var_os("CASA_RS_LOG_TABLE").map(PathBuf::from),
92            table_priority: env_priority("CASA_RS_LOG_TABLE_PRIORITY")?
93                .unwrap_or(CasaPriority::Info),
94            stderr_priority: env_stderr_priority("CASA_RS_LOG_STDERR_PRIORITY")?,
95        })
96    }
97
98    /// Apply the shared logging flags in `args` to this configuration and
99    /// return the other arguments. Reads no environment: the caller chose the
100    /// configuration the flags refine.
101    ///
102    /// # Errors
103    /// Returns [`CliLoggingError`] if a logging flag lacks a value or specifies
104    /// an invalid priority.
105    pub fn with_cli_args(
106        self,
107        args: impl IntoIterator<Item = OsString>,
108    ) -> Result<(Vec<OsString>, Self), CliLoggingError> {
109        apply_cli_logging_args(args, self)
110    }
111
112    fn fallback() -> Self {
113        Self {
114            table_path: None,
115            table_priority: CasaPriority::Info,
116            stderr_priority: Some(CasaPriority::Warn),
117        }
118    }
119}
120
121fn output_was_configured(args: &[OsString], table: bool, stderr: bool) -> bool {
122    table
123        || stderr
124        || args.iter().filter_map(|arg| arg.to_str()).any(|arg| {
125            [LOG_TABLE_FLAG, LOG_STDERR_PRIORITY_FLAG]
126                .iter()
127                .any(|flag| {
128                    arg == *flag
129                        || arg
130                            .strip_prefix(*flag)
131                            .is_some_and(|rest| rest.starts_with('='))
132                })
133        })
134}
135
136fn env_priority(name: &'static str) -> Result<Option<CasaPriority>, CliLoggingError> {
137    std::env::var(name)
138        .ok()
139        .map(|value| {
140            CasaPriority::from_str(&value)
141                .map_err(|_| CliLoggingError::InvalidEnvPriority { name, value })
142        })
143        .transpose()
144}
145
146fn env_stderr_priority(name: &'static str) -> Result<Option<CasaPriority>, CliLoggingError> {
147    match std::env::var(name) {
148        Ok(value) if priority_is_off(&value) => Ok(None),
149        Ok(value) => CasaPriority::from_str(&value)
150            .map(Some)
151            .map_err(|_| CliLoggingError::InvalidEnvPriority { name, value }),
152        Err(_) => Ok(Some(CasaPriority::Warn)),
153    }
154}
155
156fn priority_is_off(value: &str) -> bool {
157    matches!(
158        value.trim().to_ascii_lowercase().as_str(),
159        "off" | "none" | "quiet" | "false" | "0"
160    )
161}

163/// Remove shared CASA logging flags without reading environment or initializing logging.
164///
165/// # Errors
166/// Returns [`CliLoggingError`] if a logging flag lacks a value or specifies an
167/// invalid priority.
168pub fn strip_cli_logging_args(
169    args: impl IntoIterator<Item = OsString>,
170) -> Result<Vec<OsString>, CliLoggingError> {
171    let (filtered, _) = apply_cli_logging_args(args, CasaLoggingConfig::fallback())?;
172    Ok(filtered)
173}
174
175fn apply_cli_logging_args(
176    args: impl IntoIterator<Item = OsString>,
177    mut config: CasaLoggingConfig,
178) -> Result<(Vec<OsString>, CasaLoggingConfig), CliLoggingError> {
179    let mut filtered = Vec::new();
180    let mut args = args.into_iter().peekable();
181    while let Some(arg) = args.next() {
182        if let Some(value) = split_option_value(&arg, LOG_TABLE_FLAG) {
183            config.table_path = Some(PathBuf::from(value));
184            continue;
185        }
186        if arg == LOG_TABLE_FLAG {
187            let value = next_option_value(&mut args, LOG_TABLE_FLAG)?;
188            config.table_path = Some(PathBuf::from(value));
189            continue;
190        }
191        if let Some(value) = split_option_value(&arg, LOG_TABLE_PRIORITY_FLAG) {
192            config.table_priority = CasaPriority::from_str(&value)?;
193            continue;
194        }
195        if arg == LOG_TABLE_PRIORITY_FLAG {
196            let value = next_option_value(&mut args, LOG_TABLE_PRIORITY_FLAG)?;
197            config.table_priority = parse_priority_option(&value, LOG_TABLE_PRIORITY_FLAG)?;
198            continue;
199        }
200        if let Some(value) = split_option_value(&arg, LOG_STDERR_PRIORITY_FLAG) {
201            config.stderr_priority = parse_optional_priority_str(&value)?;
202            continue;
203        }
204        if arg == LOG_STDERR_PRIORITY_FLAG {
205            let value = next_option_value(&mut args, LOG_STDERR_PRIORITY_FLAG)?;
206            config.stderr_priority = parse_optional_priority(&value)?;
207            continue;
208        }
209        filtered.push(arg);
210    }
211    Ok((filtered, config))
212}
213
214fn split_option_value(arg: &OsString, flag: &'static str) -> Option<String> {
215    let arg = arg.to_str()?;
216    let rest = arg.strip_prefix(flag)?;
217    rest.strip_prefix('=').map(str::to_string)
218}
219
220fn next_option_value(
221    args: &mut impl Iterator<Item = OsString>,
222    flag: &'static str,
223) -> Result<OsString, CliLoggingError> {
224    args.next().ok_or(CliLoggingError::MissingOptionValue(flag))
225}
226
227fn parse_optional_priority(value: &OsString) -> Result<Option<CasaPriority>, CliLoggingError> {
228    let value = value.to_str().ok_or(CliLoggingError::NonUtf8OptionValue(
229        LOG_STDERR_PRIORITY_FLAG,
230    ))?;
231    parse_optional_priority_str(value)
232}
233
234fn parse_optional_priority_str(value: &str) -> Result<Option<CasaPriority>, CliLoggingError> {
235    if priority_is_off(value) {
236        Ok(None)
237    } else {
238        Ok(Some(CasaPriority::from_str(value)?))
239    }
240}
241
242fn parse_priority_option(
243    value: &OsString,
244    flag: &'static str,
245) -> Result<CasaPriority, CliLoggingError> {
246    let value = value
247        .to_str()
248        .ok_or(CliLoggingError::NonUtf8OptionValue(flag))?;
249    Ok(CasaPriority::from_str(value)?)
250}

252/// Guard holding sink state for explicit flush at task exit.
253#[derive(Clone)]
254pub struct CasaLoggingGuard {
255    sink: Arc<dyn CasaLogSink>,
256}
257
258impl CasaLoggingGuard {
259    /// Flush the configured logging sinks.
260    ///
261    /// # Errors
262    /// Returns [`CasaLoggingError`] if flushing any sink fails.
263    pub fn flush(&self) -> Result<(), CasaLoggingError> {
264        self.sink.flush()
265    }
266}
267
268impl Drop for CasaLoggingGuard {
269    fn drop(&mut self) {
270        let _ = self.flush();
271    }
272}

274/// Initialize global logging once with stderr records written to `stderr`.
275///
276/// # Errors
277/// Returns [`CliLoggingError`] if installing the global `tracing` subscriber fails.
278pub fn init_global_with_stderr<W: Write + Send + 'static>(
279    config: CasaLoggingConfig,
280    stderr: Arc<Mutex<W>>,
281) -> Result<CasaLoggingGuard, CliLoggingError> {
282    let mut sinks: Vec<Arc<dyn CasaLogSink>> = Vec::new();
283    if let Some(path) = config.table_path {
284        sinks.push(Arc::new(CasaTableLogSink::new(
285            path,
286            config.table_priority,
287            TableLogOpenMode::CreateOrAppend,
288        )));
289    }
290    if let Some(priority) = config.stderr_priority {
291        sinks.push(Arc::new(WriterSink::new(stderr, priority)));
292    }
293    let sink: Arc<dyn CasaLogSink> = if sinks.is_empty() {
294        Arc::new(NullSink)
295    } else {
296        Arc::new(CompositeSink::new(sinks))
297    };
298    tracing::subscriber::set_global_default(
299        Registry::default().with(CasaLogLayer::filtered(sink.clone())),
300    )
301    .map_err(|error| CliLoggingError::Subscriber(error.to_string()))?;
302    Ok(CasaLoggingGuard { sink })
303}

305/// Run a [`TaskFrontEnd`] as a binary `main`: strip CLI logging flags,
306/// install process-global logging only for an executing action (`Run` / `JsonRun`),
307/// enter the workspace when running on the command line, and write stdout/stderr.
308pub fn run_front_end<F, W: Write + Send + 'static>(
309    front_end: &TaskFrontEnd<F>,
310    args: impl IntoIterator<Item = OsString>,
311    logging: Result<CasaLoggingConfig, CliLoggingError>,
312    enter_workspace: EnterWorkspace,
313    stdout: &mut dyn Write,
314    stderr: W,
315) -> ExitCode
316where
317    F: Fn(&TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure>,
318{
319    let raw_args: Vec<OsString> = args.into_iter().collect();
320    let default_stderr_logs = !CasaLoggingConfig::output_was_configured(&raw_args);
321    let stderr = Arc::new(Mutex::new(stderr));
322    let mut guard: Option<CasaLoggingGuard> = None;
323
324    let output = match strip_cli_logging_args(raw_args.clone()) {
325        Err(error) => casa_tasks::FrontEndOutput {
326            stdout: String::new(),
327            stderr: vec![format!("Error: {error}")],
328            code: casa_contracts::EXIT_VALIDATION_FAILED,
329        },
330        Ok(stripped) => {
331            front_end.dispatch_for_cli(&stripped, enter_workspace, default_stderr_logs, || {
332                let (_, mut config) = logging?.with_cli_args(raw_args)?;
333                // Named from where the command was run, before a run enters its workspace.
334                config.table_path = config
335                    .table_path
336                    .map(|path| std::path::absolute(&path).unwrap_or(path));
337                guard = Some(init_global_with_stderr(config, stderr.clone())?);
338                Ok::<_, CliLoggingError>(())
339            })
340        }
341    };
342
343    let _ = stdout.write_all(output.stdout.as_bytes());
344    let _ = stdout.flush();
345    let write_lines = |lines: &[String]| {
346        let mut stderr = stderr.lock().unwrap_or_else(PoisonError::into_inner);
347        for line in lines {
348            let _ = writeln!(stderr, "{line}");
349        }
350        let _ = stderr.flush();
351    };
352    write_lines(&output.stderr);
353    if let Some(Err(error)) = guard.as_ref().map(CasaLoggingGuard::flush) {
354        write_lines(&[format!("Error: failed to flush logging: {error}")]);
355    }
356    ExitCode::from(u8::try_from(output.code).unwrap_or(1))
357}

670    fn a_bad_logging_configuration_refuses_only_a_run() {
671        let calls = Calls::default();
672        let front_end = test_front_end(&calls);
673        let workspace = tempfile::tempdir().unwrap();
674        let run = |line: Vec<OsString>| {
675            let (mut stdout, stderr) = (Vec::new(), SharedBuffer::default());
676            let code = run_front_end(
677                &front_end,
678                line,
679                bad_logging(),
680                stay,
681                &mut stdout,
682                stderr.clone(),
683            );
684            (code, String::from_utf8(stdout).unwrap(), stderr.text())
685        };
686
687        for line in [
688            args(&["--template"]),
689            args(&["--describe"]),
690            args(&["--json-schema"]),
691            args(&["--log-stderr-priority", "INFO", "--show", "x.image"]),
692        ] {
693            let (code, stdout, stderr) = run(line.clone());
694            assert_eq!(code, ExitCode::SUCCESS, "{line:?}: {stderr}");
695            assert!(!stdout.is_empty(), "{line:?}");
696        }
697
698        let (code, stdout, stderr) = run(in_workspace(workspace.path(), &["x.image"]));
699        assert_eq!(code, ExitCode::from(2));
700        assert!(stdout.is_empty());
701        assert!(
702            stderr.contains("failed to initialize logging: invalid CASA_RS_LOG_TABLE_PRIORITY"),
703            "{stderr}"
704        );
705
706        let request = workspace.path().join("request.json");
707        std::fs::write(
708            &request,
709            r#"{"surface":"imstat","values":{"imagename":"x.image"}}"#,
710        )
711        .unwrap();
712        let (code, stdout, _) = run(args(&["--json-run", request.to_str().unwrap()]));
713        assert_eq!(code, ExitCode::from(2));
714        let reply: TaskReply = serde_json::from_str(&stdout).unwrap();
715        assert!(
716            matches!(&reply, TaskReply::Failed { kind: TaskFailureKind::Validation, message, .. }
717                if message.contains("failed to initialize logging")),
718            "{reply:?}"
719        );
720
721        // A malformed logging flag is refused on every action.
722        let (code, _, stderr) = run(args(&["--template", "--log-table"]));
723        assert_eq!(code, ExitCode::from(2));
724        assert!(stderr.contains("missing value for --log-table"), "{stderr}");
725
726        assert!(calls.borrow().is_empty());
727    }
