31/// Spectral imaging family requested at the task surface.
32#[derive(Debug, Clone, Copy, PartialEq, Eq)]
33pub enum SpectralMode {
34    /// Continuum multi-frequency synthesis.
35    Mfs,
36    /// Frame-aware spectral cube.
37    Cube,
38    /// Spectral cube in the native data frame.
39    Cubedata,
40    /// Source-rest-frame cube for a moving target.
41    Cubesource,
42    /// Multi-term continuum reconstruction through cube major cycles.
43    Mvc,
44}
45
46impl SpectralMode {
47    fn parse(value: &str) -> Result<Self, String> {
48        match value.to_ascii_lowercase().as_str() {
49            "mfs" => Ok(Self::Mfs),
50            "cube" => Ok(Self::Cube),
51            "cubedata" => Ok(Self::Cubedata),
52            "cubesource" => Ok(Self::Cubesource),
53            "mvc" => Ok(Self::Mvc),
54            _ => Err(format!("unsupported spectral mode {value:?}")),
55        }
56    }
57
58    pub(crate) const fn cube_specmode(self) -> casa_ms::spectral_selection::CubeSpecMode {
59        match self {
60            Self::Mfs | Self::Cube | Self::Cubesource | Self::Mvc => {
61                casa_ms::spectral_selection::CubeSpecMode::Cube
62            }
63            Self::Cubedata => casa_ms::spectral_selection::CubeSpecMode::Cubedata,
64        }
65    }
66}

411    /// Parse the direct CLI surface. Machine JSON requests are handled before
412    /// this method by `TaskCliHost`.
413    pub fn parse(args: impl IntoIterator<Item = OsString>) -> Result<Self, String> {
414        let mut config = Self::defaults();
415        let mut robust = None;
416        let args = args.into_iter().collect::<Vec<_>>();
417        let mut index = 0;
418        while index < args.len() {
419            let flag = args[index]
420                .to_str()
421                .ok_or_else(|| "CLI argument is not UTF-8".to_string())?;
422            let value = |offset: usize| -> Result<&str, String> {
423                args.get(index + offset)
424                    .and_then(|value| value.to_str())
425                    .ok_or_else(|| format!("{flag} requires a value"))
426            };
427            let mut consumed = 2;
428            match flag {
429                "--ms" => config.ms = PathBuf::from(value(1)?),
430                "--imagename" => config.imagename = PathBuf::from(value(1)?),
431                "--imsize" => config.imsize = parse(value(1)?, flag)?,
432                "--cell-arcsec" => config.cell_arcsec = parse(value(1)?, flag)?,
433                "--projection" if value(1)?.eq_ignore_ascii_case("sin") => {}
434                "--projection" => return Err("only SIN projection is supported".to_string()),
435                "--field" => config.field_ids = Some(parse_ids(value(1)?, flag)?),
436                "--uvrange" => config.uvrange = Some(value(1)?.to_string()),
437                "--intent" => config.intent = Some(value(1)?.to_string()),
438                "--phasecenter-field" => config.phasecenter_field = Some(parse(value(1)?, flag)?),
439                "--phasecenter" => config.phasecenter = Some(value(1)?.to_string()),
440                "--ddid" => config.ddid = Some(parse(value(1)?, flag)?),
441                "--spw" => {
442                    let text = value(1)?.to_string();
443                    config.spw = text.parse().ok();
444                    config.spw_selector = Some(text);
445                }
446                "--channel-start" => config.channel_start = Some(parse(value(1)?, flag)?),
447                "--channel-count" | "--nchan" => {
448                    config.channel_count = Some(parse(value(1)?, flag)?);
449                }
450                "--fitspw" => {
451                    let selector = value(1)?;
452                    config.continuum_fit_spw = (!selector.is_empty()
453                        && !selector.eq_ignore_ascii_case("none"))
454                    .then(|| selector.to_string());
455                }
456                "--fitorder" => config.continuum_fit_order = parse(value(1)?, flag)?,
457                "--datacolumn" => config.datacolumn = Some(value(1)?.to_string()),
458                "--savemodel" => config.save_model = parse_save_model(value(1)?)?,
459                "--save-continuum-residual" => {
460                    config.save_continuum_residual = true;
461                    consumed = 1;
462                }
463                "--startmodel" => config.start_model = Some(PathBuf::from(value(1)?)),

637                    aw_controls(&mut config).conjugate_beams = true;
638                    consumed = 1;
639                }
640                "--no-conjbeams" => {
641                    aw_controls(&mut config).conjugate_beams = false;
642                    consumed = 1;
643                }
644                "--computepastep" => {
645                    aw_controls(&mut config).compute_pa_step_deg = parse(value(1)?, flag)?
646                }
647                "--rotatepastep" => {
648                    aw_controls(&mut config).rotate_pa_step_deg = parse(value(1)?, flag)?
649                }
650                "--pointingoffsetsigdev" => {
651                    aw_controls(&mut config).pointing_offset_sigdev = parse_csv(value(1)?, flag)?
652                }
653                "--mosweight" => {
654                    aw_controls(&mut config).mosaic_weighting = true;
655                    consumed = 1;
656                }
657                "--no-mosweight" => {
658                    aw_controls(&mut config).mosaic_weighting = false;
659                    consumed = 1;
660                }
661                "--normtype" => {
662                    aw_controls(&mut config).normalization = parse_aw_normalization(value(1)?)?
663                }
664                "--imaging-fft-precision" => {
665                    config.imaging_fft_precision = parse_fft_precision(value(1)?)?
666                }
667                "--imaging-memory-pressure-policy" => {
668                    config.imaging_memory_pressure_policy = parse_memory_policy(value(1)?)?
669                }
670                "--imaging-memory-target-mb" => {
671                    config.imaging_memory_target_mb = Some(parse(value(1)?, flag)?)
672                }
673                "--imaging-prepare-buffer-mb" => {
674                    config.imaging_prepare_buffer_mb = Some(parse(value(1)?, flag)?)
675                }
676                "--imaging-row-block-rows" => {
677                    config.imaging_row_block_rows = Some(parse(value(1)?, flag)?)
678                }
679                "--imaging-prepare-workers" => {
680                    config.imaging_prepare_workers = Some(parse(value(1)?, flag)?)
681                }
682                "--imaging-read-ahead-blocks" => {
683                    config.imaging_read_ahead_blocks = Some(parse(value(1)?, flag)?)
684                }
685                "--standard-mfs-acceleration" => {
686                    config.standard_mfs_acceleration = parse_acceleration(value(1)?)?;
687                }
688                "--standard-mfs-backend" => {
689                    config.standard_mfs_backend = Some(value(1)?.to_string());
690                }
691                "--standard-mfs-grid-threads" => {
692                    let threads = value(1)?;
693                    config.standard_mfs_grid_threads =
694                        (threads != "auto").then(|| threads.to_string());
695                }
696                "--imaging-fft-backend" => {
697                    config.imaging_fft_backend = parse_fft_backend(value(1)?)?;
698                }
699                other => return Err(format!("unknown casars-imager option {other:?}")),
700            }
701            index += consumed;
702        }
703        config.weighting = match (config.weighting, robust) {
704            (WeightingMode::Briggs { robust }, None) => WeightingMode::Briggs { robust },
705            (WeightingMode::BriggsBwTaper { robust }, None) => {
706                WeightingMode::BriggsBwTaper { robust }
707            }
708            (WeightingMode::Briggs { .. }, Some(robust)) => WeightingMode::Briggs { robust },
709            (WeightingMode::BriggsBwTaper { .. }, Some(robust)) => {
710                WeightingMode::BriggsBwTaper { robust }
711            }
712            (weighting, _) => weighting,
713        };
714        if config.ms.as_os_str().is_empty() || config.imagename.as_os_str().is_empty() {
715            return Err("--ms and --imagename are required".to_string());
716        }
717        if config.imsize == 0
718            || config.facets == 0
719            || config.cell_arcsec <= 0.0
720            || !config.cell_arcsec.is_finite()
721        {
722            return Err("--imsize, --cell-arcsec, and --facets must be positive".to_string());
723        }
724        Ok(config)
725    }

999            usize::try_from(integer("fitorder")?).map_err(|error| error.to_string())?;

1002            return Err("vis and imagename are required".to_string());
1003        }
1004        Ok(config)

1009    values: &'a BTreeMap<String, ParameterValue>,
1010    name: &str,
1011) -> Result<&'a ParameterValue, String> {
1012    values
1013        .get(name)

1015}

1017fn parameter_text(values: &BTreeMap<String, ParameterValue>, name: &str) -> Result<String, String> {
1018    match parameter_value(values, name)? {
1019        ParameterValue::String(value) => Ok(value.clone()),
1020        ParameterValue::Array(values) if values.len() == 1 => match &values[0] {
1021            ParameterValue::String(value) => Ok(value.clone()),
1022            value => Err(format!(
1023                "resolved imager parameter {name:?} singleton must be text, found {value:?}"
1024            )),
1025        },
1026        value => Err(format!(
1027            "resolved imager parameter {name:?} must be text, found {value:?}"
1028        )),
1029    }
1030}

1035        value => Err(format!(
1036            "resolved imager parameter {name:?} must be integer, found {value:?}"
1037        )),
1038    }

1060fn optional_usize(
1061    values: &BTreeMap<String, ParameterValue>,
1062    name: &str,

1066            .map(Some)
1067            .map_err(|error| error.to_string()),
1068        ParameterValue::String(value)
1069            if matches!(value.to_ascii_lowercase().as_str(), "none" | "auto") =>
1070        {
1071            Ok(None)
1072        }
1073        ParameterValue::String(value) => value
1074            .parse::<usize>()
1075            .map(Some)
1076            .map_err(|error| error.to_string()),
1077        value => Err(format!(
1078            "resolved imager parameter {name:?} must be optional integer, found {value:?}"
1079        )),
1080    }

1096fn parameter_square_usize(
1097    values: &BTreeMap<String, ParameterValue>,
1098    name: &str,
1099) -> Result<usize, String> {
1100    match parameter_value(values, name)? {
1101        ParameterValue::Integer(value) => {
1102            usize::try_from(*value).map_err(|error| error.to_string())
1103        }
1104        ParameterValue::Array(items) if items.len() == 2 => {
1105            let [first, second] = items.as_slice() else {
1106                unreachable!()
1107            };
1108            if first != second {
1109                return Err(format!("{name} must be square"));
1110            }
1111            match first {
1112                ParameterValue::Integer(value) => {
1113                    usize::try_from(*value).map_err(|error| error.to_string())
1114                }
1115                value => Err(format!("{name} axis must be integer, found {value:?}")),
1116            }
1117        }
1118        value => Err(format!(
1119            "resolved imager parameter {name:?} must be integer or square pair, found {value:?}"
1120        )),
1121    }
1122}

1124fn parameter_square_quantity(
1125    values: &BTreeMap<String, ParameterValue>,
1126    name: &str,
1127    suffix: &str,
1128) -> Result<f64, String> {
1129    match parameter_value(values, name)? {
1130        ParameterValue::String(value) => parse_suffixed(value, suffix),
1131        ParameterValue::Array(items) if items.len() == 2 && items[0] == items[1] => match &items[0]
1132        {
1133            ParameterValue::String(value) => parse_suffixed(value, suffix),
1134            value => Err(format!("{name} axis must be text, found {value:?}")),
1135        },
1136        value => Err(format!(
1137            "resolved imager parameter {name:?} must be quantity or square pair, found {value:?}"
1138        )),
1139    }

1144    name: &str,

1150        ParameterValue::String(value) => parse_suffixed(value, suffix),
1151        value => Err(format!(
1152            "resolved imager parameter {name:?} must be quantity, found {value:?}"
1153        )),
1154    }

1157fn parse_suffixed(value: &str, suffix: &str) -> Result<f64, String> {

1227/// Run the machine or direct CLI surface.
1228pub fn run_with_cli_args(args: impl IntoIterator<Item = OsString>) -> Result<(), String> {
1229    let raw_args = args.into_iter().collect::<Vec<_>>();
1230    let mut managed_output = false;
1231    let mut args = Vec::with_capacity(raw_args.len());
1232    let mut index = 0;
1233    while index < raw_args.len() {
1234        if raw_args[index].to_str() == Some("--managed-output") {
1235            managed_output = raw_args
1236                .get(index + 1)
1237                .and_then(|value| value.to_str())
1238                .map(|value| parse_bool(value, "--managed-output"))
1239                .transpose()?
1240                .unwrap_or(true);
1241            index += usize::from(
1242                raw_args
1243                    .get(index + 1)
1244                    .and_then(|value| value.to_str())
1245                    .is_some_and(|value| matches!(value, "true" | "false")),
1246            ) + 1;
1247            continue;
1248        }
1249        args.push(raw_args[index].clone());
1250        index += 1;
1251    }
1252    let host = casa_task_runtime::TaskCliHost::new(
1253        imager_task_schema_bundle(),
1254        |request: ImagerTaskRequest| request.execute(),
1255    );
1256    if let Some(output) = host.dispatch(&args).map_err(|error| error.to_string())? {
1257        if managed_output
1258            && args
1259                .iter()
1260                .any(|argument| argument.to_str() == Some("--json-run"))
1261        {
1262            let result: ImagerTaskResult = serde_json::from_str(&output)
1263                .map_err(|error| format!("decode canonical imager task result: {error}"))?;
1264            let ImagerTaskResult::Run(result) = result;
1265            println!(
1266                "{}",
1267                serde_json::to_string_pretty(&ManagedImagingOutput::from_task_result(&result))
1268                    .map_err(|error| format!("serialize managed imaging output: {error}"))?
1269            );
1270        } else {
1271            println!("{output}");
1272        }
1273        return Ok(());
1274    }
1275    if args
1276        .iter()
1277        .any(|arg| matches!(arg.to_str(), Some("-h" | "--help")))
1278    {
1279        println!("{}", command_schema("casars-imager").render_help());
1280        return Ok(());
1281    }
1282    let request = request_from_parameter_cli_args(&args)?;
1283    let unsupported = request.unsupported_reasons()?;
1284    if !unsupported.is_empty() {
1285        return Err(format!(
1286            "imager request is unavailable in this installed build: {}",
1287            unsupported
1288                .iter()
1289                .map(|reason| format!("{}/{}", reason.kind, reason.id))
1290                .collect::<Vec<_>>()
1291                .join(", ")
1292        ));
1293    }
1294    let summary = run_from_request(&request)?;
1295    let result = ImagerRunTaskResult::from_run(request, &summary);
1296    if managed_output {
1297        println!(
1298            "{}",
1299            serde_json::to_string_pretty(&ManagedImagingOutput::from_task_result(&result))
1300                .map_err(|error| format!("serialize managed imaging output: {error}"))?
1301        );
1302    } else {
1303        for warning in &result.run.warnings {
1304            eprintln!("warning: {warning}");
1305        }
1306        println!(
1307            "Wrote CASA-compatible products at prefix {} ({} gridded samples, {} major cycles, {} reported minor iterations, {} actual components, stop={:?})",
1308            result.request.image_name.display(),
1309            result.run.gridded_samples,
1310            result.run.major_cycles,
1311            result.run.minor_iterations,
1312            result.run.actual_minor_iterations,
1313            result.run.clean_stop_reason,
1314        );
1315    }
1316    Ok(())
1317}
