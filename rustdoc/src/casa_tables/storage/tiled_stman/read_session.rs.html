12#[derive(Default)]
13pub(super) struct TileReadSession {
14    pub(super) table_id: Option<u64>,
15    pub(super) files: std::collections::HashMap<u32, TileReadSessionFile>,
16    pub(super) recent_tiles: Vec<(SharedTileKey, Arc<[u8]>)>,
17}
18
19impl TileReadSession {
20    fn table_id(&mut self, table_path: &Path) -> u64 {
21        if let Some(id) = self.table_id {
22            return id;
23        }
24        let mut cache = SHARED_TILE_CACHE
25            .lock()
26            .expect("shared tile cache lock poisoned");
27        let id = cache.table_id(table_path);
28        self.table_id = Some(id);
29        id
30    }
31
32    fn file(
33        &mut self,
34        table_path: &Path,
35        dm_seq_nr: u32,
36        file_seq_nr: u32,
37    ) -> Result<&mut TileReadSessionFile, StorageError> {
38        match self.files.entry(file_seq_nr) {
39            std::collections::hash_map::Entry::Occupied(entry) => Ok(entry.into_mut()),
40            std::collections::hash_map::Entry::Vacant(entry) => {
41                let path = tsm_data_path(table_path, dm_seq_nr, file_seq_nr);
42                let file = DataFile::open(&path).map_err(|err| {
43                    StorageError::FormatMismatch(format!("cannot read {}: {err}", path.display()))
44                })?;
45                Ok(entry.insert(TileReadSessionFile {
46                    file,
47                    file_position: 0,
48                    read_ahead_start: 0,
49                    read_ahead: Vec::new(),
50                }))
51            }
52        }
53    }
54
55    #[allow(clippy::too_many_arguments)]
56    fn read_tile_streamed(
57        &mut self,
58        table_path: &Path,
59        dm_seq_nr: u32,
60        file_seq_nr: u32,
61        file_pos: u64,
62        dst: &mut [u8],
63        file_tile_bytes: usize,
64        useful_read_ahead_bytes: usize,
65        dt: CasacoreDataType,
66        tile_nelem: usize,
67        needs_swap: bool,
68        swap_size: usize,
69    ) -> Result<(), StorageError> {
70        let session_file = self.file(table_path, dm_seq_nr, file_seq_nr)?;
71        let file_end = file_pos
72            .checked_add(file_tile_bytes as u64)
73            .ok_or_else(|| StorageError::FormatMismatch("tile read offset overflow".to_string()))?;
74        let read_ahead_end = session_file
75            .read_ahead_start
76            .saturating_add(session_file.read_ahead.len() as u64);
77        if file_pos < session_file.read_ahead_start || file_end > read_ahead_end {
78            if session_file.file_position != file_pos {
79                session_file.file.seek(SeekFrom::Start(file_pos))?;
80            }
81            // Prefetch only the contiguous selected tile run. The next row
82            // tile may start at a different channel or may not be selected.
83            let read_ahead_bytes = STREAMING_TILED_READ_AHEAD_BYTES
84                .min(useful_read_ahead_bytes)
85                .max(file_tile_bytes);
86            session_file.read_ahead.resize(read_ahead_bytes, 0);
87            let mut bytes_read = 0usize;
88            while bytes_read < read_ahead_bytes {
89                let read = session_file
90                    .file
91                    .read(&mut session_file.read_ahead[bytes_read..])?;
92                if read == 0 {
93                    break;
94                }
95                #[cfg(test)]
96                STREAMED_READ_COUNTS.with(|counts| {
97                    let (calls, bytes) = counts.get();
98                    counts.set((calls + 1, bytes + read));
99                });
100                bytes_read += read;
101            }
102            session_file.read_ahead.truncate(bytes_read);
103            session_file.read_ahead_start = file_pos;
104            session_file.file_position = file_pos + bytes_read as u64;
105            if bytes_read < file_tile_bytes {
106                return Err(StorageError::FormatMismatch(format!(
107                    "short tiled read at byte {file_pos}: expected {file_tile_bytes} bytes, found {bytes_read}"
108                )));
109            }
110        }
111        let buffered_start = usize::try_from(file_pos - session_file.read_ahead_start)
112            .map_err(|_| StorageError::FormatMismatch("tile buffer offset overflow".to_string()))?;
113        let buffered = &session_file.read_ahead[buffered_start..buffered_start + file_tile_bytes];
114        if file_tile_bytes == dst.len() {
115            dst.copy_from_slice(buffered);
116            if needs_swap {
117                swap_bytes_inplace(dst, swap_size);
118            }
119        } else {
120            debug_assert_eq!(dt, CasacoreDataType::TpBool);
121            debug_assert_eq!(dst.len(), tile_nelem);
122            for (pixel, value) in dst.iter_mut().enumerate() {
123                *value = u8::from((buffered[pixel / 8] & (1 << (pixel % 8))) != 0);
124            }
125        }
126        Ok(())
127    }
128
129    fn recent_tile(&self, key: &SharedTileKey) -> Option<Arc<[u8]>> {
130        self.recent_tiles
131            .iter()
132            .find_map(|(recent_key, data)| (recent_key == key).then(|| Arc::clone(data)))
133    }
134
135    fn remember_tile(&mut self, key: SharedTileKey, data: Arc<[u8]>) {
136        if let Some(position) = self
137            .recent_tiles
138            .iter()
139            .position(|(recent_key, _)| recent_key == &key)
140        {
141            let entry = self.recent_tiles.remove(position);
142            self.recent_tiles.push(entry);
143            return;
144        }
145        if self.recent_tiles.len() >= TILE_READ_SESSION_RECENT_TILE_COUNT {
146            self.recent_tiles.remove(0);
147        }
148        self.recent_tiles.push((key, data));
149    }
150}

152#[allow(clippy::too_many_arguments)]
153pub(super) fn load_streamed_column_tile(
154    table_path: &Path,
155    dm_seq_nr: u32,
156    header: &TiledStManHeader,
157    cube: &TsmCubeInfo,
158    bucket_size: usize,
159    col_offset_in_tile: usize,
160    dt: CasacoreDataType,
161    tile_index: usize,
162    contiguous_tiles: usize,
163    session: &mut TileReadSession,
164    tile: &mut Vec<u8>,
165) -> Result<(), StorageError> {
166    let tile_nelem: usize = cube.tile_shape.iter().product();
167    let tile_bytes = tile_nelem * tile_element_size(dt);
168    let file_tile_bytes = tile_storage_bytes(dt, tile_nelem);
169    let useful_read_ahead_bytes = read_ahead_span(contiguous_tiles, bucket_size, file_tile_bytes)
170        .ok_or_else(|| {
171        StorageError::FormatMismatch("selected tile run size overflow".into())
172    })?;
173    let swap_size = match dt {
174        CasacoreDataType::TpComplex => 4,
175        CasacoreDataType::TpDComplex => 8,
176        _ => tile_element_size(dt),
177    };
178    let needs_swap = header.big_endian != cfg!(target_endian = "big");
179    let file_pos =
180        (cube.file_offset as usize + tile_index * bucket_size + col_offset_in_tile) as u64;
181    tile.resize(tile_bytes, 0);
182    session.read_tile_streamed(
183        table_path,
184        dm_seq_nr,
185        cube.file_seq_nr as u32,
186        file_pos,
187        tile,
188        file_tile_bytes,
189        useful_read_ahead_bytes,
190        dt,
191        tile_nelem,
192        needs_swap,
193        swap_size,
194    )
195}

197pub(super) fn read_ahead_span(
198    tiles: usize,
199    bucket_bytes: usize,
200    column_tile_bytes: usize,
201) -> Option<usize> {
202    let additional_tiles = tiles.checked_sub(1)?;
203    // A bucket can contain large tiles from other columns. Read ahead only
204    // when the buffer can reach the end of a second tile of this column.
205    if bucket_bytes > STREAMING_TILED_READ_AHEAD_BYTES.saturating_sub(column_tile_bytes) {
206        return Some(column_tile_bytes);
207    }
208    additional_tiles
209        .checked_mul(bucket_bytes)?
210        .checked_add(column_tile_bytes)
211}

213/// The byte order of a shared tile's pixels: [`load_shared_column_tile`]
214/// and [`load_streamed_column_tile`] swap a tile to the host's order as
215/// they read it, so its pixels decode in that order, not the manager's.
216pub(super) const SHARED_TILE_BIG_ENDIAN: bool = cfg!(target_endian = "big");
217
218#[allow(clippy::too_many_arguments)]
219pub(super) fn load_shared_column_tile(
220    table_path: &Path,
221    dm_seq_nr: u32,
222    header: &TiledStManHeader,
223    cube_idx: usize,
224    cube: &TsmCubeInfo,
225    target_col_idx: usize,
226    bucket_size: usize,
227    col_offset_in_tile: usize,
228    dt: CasacoreDataType,
229    tile_index: usize,
230    session: &mut TileReadSession,
231) -> Result<Arc<[u8]>, StorageError> {
232    let table_id = session.table_id(table_path);
233    let tile_nelem: usize = cube.tile_shape.iter().product();
234    let key = SharedTileKey {
235        table_id,
236        dm_seq_nr,
237        cube_idx,
238        target_col_idx,
239        tile_index,
240        is_bool: dt == CasacoreDataType::TpBool,
241        single_pixel: tile_nelem <= 1,
242    };
243    if let Some(hit) = session.recent_tile(&key) {
244        return Ok(hit);
245    }
246    if let Some(hit) = SHARED_TILE_CACHE
247        .lock()
248        .expect("shared tile cache lock poisoned")
249        .get(&key)
250    {
251        session.remember_tile(key, Arc::clone(&hit));
252        return Ok(hit);
253    }
254
255    let tile_bytes = tile_nelem * tile_element_size(dt);
256    let file_tile_bytes = tile_storage_bytes(dt, tile_nelem);
257    let swap_size = match dt {
258        CasacoreDataType::TpComplex => 4,
259        CasacoreDataType::TpDComplex => 8,
260        _ => tile_element_size(dt),
261    };
262    let needs_swap = header.big_endian != cfg!(target_endian = "big");
263    let file_pos =
264        (cube.file_offset as usize + tile_index * bucket_size + col_offset_in_tile) as u64;
265    let session_file = session.file(table_path, dm_seq_nr, cube.file_seq_nr as u32)?;
266    let mut tile = vec![0u8; tile_bytes];
267    read_tiled_file_tile(
268        &mut session_file.file,
269        file_pos,
270        &mut tile,
271        file_tile_bytes,
272        dt,
273        tile_nelem,
274        needs_swap,
275        swap_size,
276    )?;
277    session_file.file_position = file_pos + file_tile_bytes as u64;
278    session_file.read_ahead.clear();
279    let data: Arc<[u8]> = Arc::from(tile);
280    let mut cache = SHARED_TILE_CACHE
281        .lock()
282        .expect("shared tile cache lock poisoned");
283    if let Some(hit) = cache.get(&key) {
284        session.remember_tile(key, Arc::clone(&hit));
285        return Ok(hit);
286    }
287    let cached = cache.insert(key.clone(), data);
288    session.remember_tile(key, Arc::clone(&cached));
289    Ok(cached)
290}

292#[allow(clippy::too_many_arguments)]
293pub(super) fn reconstruct_cube_column_from_shared_tiles(
294    table_path: &Path,
295    dm_seq_nr: u32,
296    header: &TiledStManHeader,
297    cube_idx: usize,
298    cube: &TsmCubeInfo,
299    target_col_idx: usize,
300    dt: CasacoreDataType,
301    elem_size: usize,
302    session: &mut TileReadSession,
303) -> Result<Vec<u8>, StorageError> {
304    let (bucket_size, col_offsets) = compute_tile_layout(&header.col_data_types, &cube.tile_shape);
305    let ndim = cube.cube_shape.len();
306    let tiles_per_dim: Vec<usize> = (0..ndim)
307        .map(|i| cube.cube_shape[i].div_ceil(cube.tile_shape[i]))
308        .collect();
309    let nr_tiles: usize = tiles_per_dim.iter().product();
310    let cube_nelem: usize = cube.cube_shape.iter().product();
311    let mut result = vec![0u8; cube_nelem * elem_size];
312
313    for tile_idx in 0..nr_tiles {
314        let tile_pos = linear_to_nd(tile_idx, &tiles_per_dim);
315        let cube_start: Vec<usize> = (0..ndim)
316            .map(|d| tile_pos[d] * cube.tile_shape[d])
317            .collect();
318        let actual_extent: Vec<usize> = (0..ndim)
319            .map(|d| std::cmp::min(cube.tile_shape[d], cube.cube_shape[d] - cube_start[d]))
320            .collect();
321        let tile = load_shared_column_tile(
322            table_path,
323            dm_seq_nr,
324            header,
325            cube_idx,
326            cube,
327            target_col_idx,
328            bucket_size,
329            col_offsets[target_col_idx],
330            dt,
331            tile_idx,
332            session,
333        )?;
334        copy_tile_to_cube(
335            tile.as_ref(),
336            &cube.tile_shape,
337            &mut result,
338            &cube.cube_shape,
339            &cube_start,
340            &actual_extent,
341            elem_size,
342        );
343    }
344
345    Ok(result)
346}

348#[allow(clippy::too_many_arguments)]
349pub(super) fn decode_selected_cube_row_2d_channel_range_typed_from_shared_tiles(
350    table_path: &Path,
351    dm_seq_nr: u32,
352    header: &TiledStManHeader,
353    cube_idx: usize,
354    cube: &TsmCubeInfo,
355    target_col_idx: usize,
356    col_desc: &ColumnDescContents,
357    dt: CasacoreDataType,
358    elem_size: usize,
359    selected: SelectedCubeRow,
360    channel_start: usize,
361    channel_count: usize,
362    corr_count: usize,
363    tile_corr_count: usize,
364    tile_channel_count: usize,
365    session: &mut TileReadSession,
366) -> Result<Option<ArrayValue>, StorageError> {
367    let channel_end = channel_start + channel_count;
368    let cell_ndim = cube.cube_shape.len().saturating_sub(1);
369    let cell_shape = &cube.cube_shape[..cell_ndim];
370    let cell_tile_shape = &cube.tile_shape[..cell_ndim];
371    let row_tile = selected.pos_in_cube / cube.tile_shape[cell_ndim];
372    let row_in_tile = selected.pos_in_cube % cube.tile_shape[cell_ndim];
373    let row_tile_nelem: usize = cell_tile_shape.iter().product();
374    let tiles_per_dim: Vec<usize> = cube
375        .cube_shape
376        .iter()
377        .zip(cube.tile_shape.iter())
378        .map(|(&shape, &tile)| shape.div_ceil(tile))
379        .collect();
380    let tile_grid_strides = fortran_order_strides(&tiles_per_dim);
381    let (bucket_size, col_offsets) = compute_tile_layout(&header.col_data_types, &cube.tile_shape);
382    let first_channel_tile = channel_start / tile_channel_count;
383    let last_channel_tile = (channel_end.saturating_sub(1)) / tile_channel_count;
384
385    match dt {
386        CasacoreDataType::TpComplex => {
387            let mut values = vec![Complex32::new(0.0, 0.0); corr_count * channel_count];
388            for channel_tile in first_channel_tile..=last_channel_tile {
389                let tile_channel_start = channel_tile * tile_channel_count;
390                let actual_tile_channels =
391                    std::cmp::min(tile_channel_count, cell_shape[1] - tile_channel_start);
392                let overlap_start = std::cmp::max(channel_start, tile_channel_start);
393                let overlap_end =
394                    std::cmp::min(channel_end, tile_channel_start + actual_tile_channels);
395                if overlap_start >= overlap_end {
396                    continue;
397                }
398
399                let full_tile_pos = [0usize, channel_tile, row_tile];
400                let tile_index: usize = full_tile_pos
401                    .iter()
402                    .zip(tile_grid_strides.iter())
403                    .map(|(pos, stride)| pos * stride)
404                    .sum();
405                let tile = load_shared_column_tile(
406                    table_path,
407                    dm_seq_nr,
408                    header,
409                    cube_idx,
410                    cube,
411                    target_col_idx,
412                    bucket_size,
413                    col_offsets[target_col_idx],
414                    dt,
415                    tile_index,
416                    session,
417                )?;
418                let src_start = row_in_tile * row_tile_nelem * elem_size;
419                let src_end = src_start + row_tile_nelem * elem_size;
420                let tile_row = &tile[src_start..src_end];
421
422                for channel in overlap_start..overlap_end {
423                    let src_channel = channel - tile_channel_start;
424                    let dst_channel = channel - channel_start;
425                    for corr in 0..corr_count {
426                        let src_elem = corr + src_channel * tile_corr_count;
427                        let dst_elem = corr + dst_channel * corr_count;
428                        let src_byte = src_elem * elem_size;
429                        values[dst_elem] = decode_complex32_scalar(
430                            &tile_row[src_byte..src_byte + elem_size],
431                            header.big_endian,
432                        );
433                    }
434                }
435            }
436            ArrayD::from_shape_vec(IxDyn(&[corr_count, channel_count]).f(), values)
437                .map(ArrayValue::Complex32)
438                .map(Some)
439                .map_err(|e| StorageError::FormatMismatch(format!("array shape: {e}")))
440        }
441        CasacoreDataType::TpBool => {
442            let mut values = vec![false; corr_count * channel_count];
443            for channel_tile in first_channel_tile..=last_channel_tile {
444                let tile_channel_start = channel_tile * tile_channel_count;
445                let actual_tile_channels =
446                    std::cmp::min(tile_channel_count, cell_shape[1] - tile_channel_start);
447                let overlap_start = std::cmp::max(channel_start, tile_channel_start);
448                let overlap_end =
449                    std::cmp::min(channel_end, tile_channel_start + actual_tile_channels);
450                if overlap_start >= overlap_end {
451                    continue;
452                }
453
454                let full_tile_pos = [0usize, channel_tile, row_tile];
455                let tile_index: usize = full_tile_pos
456                    .iter()
457                    .zip(tile_grid_strides.iter())
458                    .map(|(pos, stride)| pos * stride)
459                    .sum();
460                let tile = load_shared_column_tile(
461                    table_path,
462                    dm_seq_nr,
463                    header,
464                    cube_idx,
465                    cube,
466                    target_col_idx,
467                    bucket_size,
468                    col_offsets[target_col_idx],
469                    dt,
470                    tile_index,
471                    session,
472                )?;
473                let src_start = row_in_tile * row_tile_nelem * elem_size;
474                let src_end = src_start + row_tile_nelem * elem_size;
475                let tile_row = &tile[src_start..src_end];
476
477                for channel in overlap_start..overlap_end {
478                    let src_channel = channel - tile_channel_start;
479                    let dst_channel = channel - channel_start;
480                    for corr in 0..corr_count {
481                        let src_elem = corr + src_channel * tile_corr_count;
482                        let dst_elem = corr + dst_channel * corr_count;
483                        values[dst_elem] = tile_row[src_elem * elem_size] != 0;
484                    }
485                }
486            }
487            ArrayD::from_shape_vec(IxDyn(&[corr_count, channel_count]).f(), values)
488                .map(ArrayValue::Bool)
489                .map(Some)
490                .map_err(|e| StorageError::FormatMismatch(format!("array shape: {e}")))
491        }
492        other => Err(StorageError::FormatMismatch(format!(
493            "typed 2-D channel-range tiled read does not support {other:?} for column {}",
494            col_desc.col_name
495        ))),
496    }
497}

499#[allow(clippy::too_many_arguments)]
500pub(super) fn decode_tiled_cell_from_shared_tiles(
501    table_path: &Path,
502    dm_seq_nr: u32,
503    header: &TiledStManHeader,
504    cube_idx: usize,
505    cube: &TsmCubeInfo,
506    target_col_idx: usize,
507    col_desc: &ColumnDescContents,
508    dt: CasacoreDataType,
509    elem_size: usize,
510    session: &mut TileReadSession,
511) -> Result<Option<ArrayValue>, StorageError> {
512    let raw = reconstruct_cube_column_from_shared_tiles(
513        table_path,
514        dm_seq_nr,
515        header,
516        cube_idx,
517        cube,
518        target_col_idx,
519        dt,
520        elem_size,
521        session,
522    )?;
523    match decode_array_value(&raw, &cube.cube_shape, dt, SHARED_TILE_BIG_ENDIAN)? {
524        Value::Array(array) => Ok(Some(array)),
525        other => Err(StorageError::FormatMismatch(format!(
526            "tiled cell row '{}' expected array value, found {:?}",
527            col_desc.col_name,
528            other.kind()
529        ))),
530    }
