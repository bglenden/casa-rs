2//! Numerical solve kernel for `gaincal` and `bandpass`.
3//!
4//! Every solve is CASA's `VisCalSolver2::solve`: a Levenberg–Marquardt
5//! minimisation of the weighted complex residual Σ w |V − g_i conj(g_j)|²
6//! over antenna gains, with every receptor of one solution solved together
7//! and a first guess from `GJones::guessPar` or `TJones::guessPar`. A
8//! phase-only solve fits data already reduced to unit amplitude
9//! (`SolveDataBuffer::enforceAPonData`) and normalises the gains afterwards,
10//! as `SolvableVisCal::enforceAPonSoln` does; an amplitude-and-phase solve
11//! keeps them. Within one solution the visibilities of each baseline are
12//! collapsed to their weighted sum, which leaves the chi-square unchanged up
13//! to a constant because the gains are constant over the solution.
14

44/// One receptor's collapsed visibilities for one solution. For each directed
45/// baseline (i, j) `edges` holds the weighted sum Σ w V_ij of its samples and
46/// `weights` the weight sum Σ w; (j, i) holds the conjugate sum. `stats`
47/// carries what the absolute chi-square needs (Σ w |V|²) and the sample
48/// counts behind the parameter errors.
49#[derive(Debug, Clone, Default)]
50pub(crate) struct SolveGraph {
51    pub(crate) edges: HashMap<(i32, i32), Complex32>,
52    pub(crate) weights: HashMap<(i32, i32), f32>,
53    pub(crate) stats: HashMap<(i32, i32), SolveEdgeStats>,
54}
55
56impl SolveGraph {
57    /// Add one collapsed sample of baseline `(antenna1, antenna2)`: its
58    /// weight, its weighted visibility and its statistics.
59    pub(crate) fn accumulate(
60        &mut self,
61        (antenna1, antenna2): (i32, i32),
62        weight: f32,
63        weighted_sample: Complex32,
64        edge_stats: SolveEdgeStats,
65    ) {
66        for (baseline, value) in [
67            ((antenna1, antenna2), weighted_sample),
68            ((antenna2, antenna1), weighted_sample.conj()),
69        ] {
70            *self.edges.entry(baseline).or_default() += value;
71            *self.weights.entry(baseline).or_default() += weight;
72            self.stats
73                .entry(baseline)
74                .and_modify(|stats| stats.add(edge_stats))
75                .or_insert(edge_stats);
76        }
77    }
78
79    fn weight(&self, baseline: (i32, i32)) -> f32 {
80        self.weights.get(&baseline).copied().unwrap_or_default()
81    }
82
83    /// The baselines with positive weight between antennas of `antennas`.
84    fn restricted_to(&self, antennas: &BTreeSet<i32>) -> Self {
85        let keep = |&(antenna1, antenna2): &(i32, i32)| {
86            antennas.contains(&antenna1)
87                && antennas.contains(&antenna2)
88                && self.weight((antenna1, antenna2)) > f32::EPSILON
89        };
90        Self {
91            edges: self
92                .edges
93                .iter()
94                .filter(|(baseline, _)| keep(baseline))
95                .map(|(baseline, value)| (*baseline, *value))
96                .collect(),
97            weights: self
98                .weights
99                .iter()
100                .filter(|(baseline, _)| keep(baseline))
101                .map(|(baseline, value)| (*baseline, *value))
102                .collect(),
103            stats: self
104                .stats
105                .iter()
106                .filter(|(baseline, _)| keep(baseline))
107                .map(|(baseline, value)| (*baseline, *value))
108                .collect(),
109        }
110    }
111
112    /// Antennas with at least one positive-weight baseline.
113    pub(crate) fn antennas(&self) -> BTreeSet<i32> {
114        self.weights
115            .iter()
116            .filter(|((antenna1, antenna2), weight)| {
117                antenna1 != antenna2 && **weight > f32::EPSILON
118            })
119            .map(|((antenna1, _), _)| *antenna1)
120            .collect()
121    }
122}

184/// Solve one `gaincal` solution interval, or `None` where CASA keeps no
185/// solution for it (see [`solve_graphs`]). Every antenna of the array or
186/// the interval gets a row; antennas without a solution are flagged, as are
187/// solutions at or below `min_snr`.
188pub(crate) fn solve_group(
189    group: SolveAccumulator,
190    available_antennas: &BTreeSet<i32>,
191    gain_type: GainType,
192    solve_mode: GainSolveMode,
193    refant_id: i32,
194    min_snr: f32,
195    min_baselines_per_antenna: usize,
196) -> Option<Vec<SolutionRow>> {
197    let scan_number = *group.scan_numbers.iter().next().unwrap_or(&0);
198    let graphs = group.channel_graphs.first()?;
199    let solved = solve_graphs(
200        graphs,
201        gain_type,
202        solve_mode,
203        SolveGraphOptions {
204            refant_id,
205            min_baselines_per_antenna,
206        },
207        None,
208    )?;
209
210    let mut antenna_ids = available_antennas.clone();
211    antenna_ids.extend(group.antenna_ids.iter().copied());
212    antenna_ids.insert(refant_id);
213    let receptors = match gain_type {
214        GainType::G => &solved.receptors[..],
215        GainType::T => &solved.receptors[..1],
216    };
217    Some(
218        antenna_ids
219            .into_iter()
220            .map(|antenna_id| {
221                let mut row = SolutionRow {
222                    time_seconds: group.solution_time,
223                    field_id: group.field_id,
224                    spw_id: group.spw_id,
225                    antenna_id,
226                    scan_number,
227                    observation_id: group.observation_id,
228                    refant_id: solved.refant_id,
229                    gains: Vec::with_capacity(receptors.len()),
230                    flags: Vec::with_capacity(receptors.len()),
231                    param_errors: Vec::with_capacity(receptors.len()),
232                    snrs: Vec::with_capacity(receptors.len()),
233                    weights: Vec::with_capacity(receptors.len()),
234                };
235                for receptor in receptors {
236                    let snr = receptor.snr.get(&antenna_id).copied().unwrap_or_default();
237                    let param_error = receptor
238                        .param_error
239                        .get(&antenna_id)
240                        .copied()
241                        .unwrap_or_default();
242                    let flagged = !receptor.solved.contains(&antenna_id)
243                        || snr <= min_snr
244                        || !param_error.is_finite();
245                    row.gains.push(if flagged {
246                        Complex32::new(1.0, 0.0)
247                    } else {
248                        receptor.gains[&antenna_id]
249                    });
250                    row.flags.push(flagged);
251                    row.param_errors.push(param_error);
252                    row.snrs.push(snr);
253                    row.weights.push(
254                        receptor
255                            .hessian
256                            .get(&antenna_id)
257                            .copied()
258                            .unwrap_or_default(),
259                    );
260                }
261                row
262            })
263            .collect(),
264    )
265}

267/// Solve the receptor graphs of one solution together, as
268/// `VisCalSolver2::solve` does, or `None` where CASA records no solution:
269/// no antenna keeps `min_baselines_per_antenna` baselines with data
270/// (`SolvableVisCal::verifyConstraints`), or the solve has not converged
271/// after [`MAX_ITERATIONS`]. CASA also fails a solve whose chi-square is
272/// exactly zero; here an exact fit is a solution (`docs/CASA (C++) bugs.md`).
273///
274/// `start`, when given, seeds the antennas it solved in place of the first
275/// guess, as CASA's B solve starts each channel from the one solved before.
276pub(crate) fn solve_graphs(
277    graphs: &[SolveGraph],
278    gain_type: GainType,
279    solve_mode: GainSolveMode,
280    options: SolveGraphOptions,
281    start: Option<&SolvedGraphs>,
282) -> Option<SolvedGraphs> {
283    let constrained = baseline_constrained_antennas(graphs, options.min_baselines_per_antenna);
284    let refant_id = if constrained.contains(&options.refant_id) {
285        options.refant_id
286    } else {
287        *constrained.first()?
288    };
289    let active = graphs
290        .iter()
291        .map(|graph| graph.restricted_to(&constrained))
292        .collect::<Vec<_>>();
293    let solved_antennas = active
294        .iter()
295        .map(|graph| {
296            let with_data = graph.antennas();
297            reachable_antennas(graph, refant_id)
298                .into_iter()
299                .filter(|antenna_id| with_data.contains(antenna_id))
300                .collect::<BTreeSet<_>>()
301        })
302        .collect::<Vec<_>>();
303    let mut first = first_guess(&active, &solved_antennas, gain_type, refant_id);
304    if let Some(start) = start {
305        for (guess, receptor) in first.iter_mut().zip(&start.receptors) {
306            for (antenna_id, value) in guess.iter_mut() {
307                if receptor.solved.contains(antenna_id) {
308                    *value = complex32_to_64(receptor.gains[antenna_id]);
309                }
310            }
311        }
312    }
313    let gains = solve_lm(&active, &solved_antennas, first, refant_id)?;
314    let mut receptors = active
315        .iter()
316        .zip(solved_antennas)
317        .zip(gains)
318        .map(|((graph, solved), gains)| {
319            finish_receptor(graph, solved, gains, solve_mode, refant_id)
320        })
321        .collect::<Vec<_>>();
322    apply_group_reduced_chi_square(&mut receptors);
323    Some(SolvedGraphs {
324        refant_id,
325        receptors,
326    })
327}

329/// The antennas that keep at least `min_baselines_per_antenna` baselines
330/// with data once antennas below the threshold, and their baselines, are
331/// removed in turn (`SolvableVisCal::verifyConstraints`). A baseline counts
332/// when any receptor has weight on it.
333pub(crate) fn baseline_constrained_antennas(
334    graphs: &[SolveGraph],
335    min_baselines_per_antenna: usize,
336) -> BTreeSet<i32> {
337    let baselines = graphs
338        .iter()
339        .flat_map(|graph| graph.weights.iter())
340        .filter(|((antenna1, antenna2), weight)| antenna1 < antenna2 && **weight > f32::EPSILON)
341        .map(|(baseline, _)| *baseline)
342        .collect::<BTreeSet<_>>();
343    let mut active = baselines
344        .iter()
345        .flat_map(|&(antenna1, antenna2)| [antenna1, antenna2])
346        .collect::<BTreeSet<_>>();
347    loop {
348        let mut counts = HashMap::<i32, usize>::new();
349        for &(antenna1, antenna2) in &baselines {
350            if active.contains(&antenna1) && active.contains(&antenna2) {
351                *counts.entry(antenna1).or_default() += 1;
352                *counts.entry(antenna2).or_default() += 1;
353            }
354        }
355        let before = active.len();
356        active.retain(|antenna_id| {
357            counts.get(antenna_id).copied().unwrap_or_default() >= min_baselines_per_antenna.max(1)
358        });
359        if active.len() == before {
360            return active;
361        }
362    }
363}

365/// CASA's first guess (`GJones::guessPar`, `TJones::guessPar`): each antenna
366/// starts from its baseline to the reference antenna, conjugated (G) or as a
367/// unit phasor (T), divided by the square root of those baselines' mean
368/// amplitude, which is also the reference antenna's start. Antennas without
369/// a baseline to the reference start there too. Phase-only data have unit
370/// amplitude, so the guess is then the baselines' phases.
371fn first_guess(
372    graphs: &[SolveGraph],
373    antennas: &[BTreeSet<i32>],
374    gain_type: GainType,
375    refant_id: i32,
376) -> Vec<HashMap<i32, Complex64>> {
377    let mut amplitude_sum = 0.0_f64;
378    let mut amplitude_count = 0_usize;
379    let mut guesses = graphs
380        .iter()
381        .zip(antennas)
382        .map(|(graph, antennas)| {
383            antennas
384                .iter()
385                .filter(|antenna_id| **antenna_id != refant_id)
386                .filter_map(|&antenna_id| {
387                    let weight = f64::from(graph.weight((refant_id, antenna_id)));
388                    let edge = graph.edges.get(&(refant_id, antenna_id))?;
389                    let mean = complex32_to_64(*edge) / weight;
390                    let amplitude = mean.norm();
391                    (weight > f64::EPSILON && amplitude > 0.0).then(|| {
392                        amplitude_sum += amplitude;
393                        amplitude_count += 1;
394                        let guess = match gain_type {
395                            GainType::G => mean.conj(),
396                            GainType::T => mean.conj() / amplitude,
397                        };
398                        (antenna_id, guess)
399                    })
400                })
401                .collect::<HashMap<_, _>>()
402        })
403        .collect::<Vec<_>>();
404    let scale = if amplitude_count > 0 {
405        (amplitude_sum / amplitude_count as f64).sqrt()
406    } else {
407        1.0
408    };
409    for (guess, antennas) in guesses.iter_mut().zip(antennas) {
410        for value in guess.values_mut() {
411            *value /= scale;
412        }
413        for antenna_id in antennas {
414            guess
415                .entry(*antenna_id)
416                .or_insert(Complex64::new(scale, 0.0));
417        }
418        if antennas.contains(&refant_id) {
419            guess.insert(refant_id, Complex64::new(scale, 0.0));
420        }
421    }
422    guesses
423}

425/// The Levenberg–Marquardt iteration of `VisCalSolver2::solve`, with its
426/// step size optimisation (`optStepSize2`) and convergence test
427/// (`converged`): a step counts towards convergence when chi-square has not
428/// risen by more than 0.1% and changed by less than 10%, and six such steps
429/// converge. `None` when [`MAX_ITERATIONS`] pass without converging.
430fn solve_lm(
431    graphs: &[SolveGraph],
432    antennas: &[BTreeSet<i32>],
433    mut gains: Vec<HashMap<i32, Complex64>>,
434    refant_id: i32,
435) -> Option<Vec<HashMap<i32, Complex64>>> {
436    let mut last_gains = gains.clone();
437    let mut last_chi_square = f64::MAX;
438    let mut convergence_count = 0_i32;
439
440    for iteration in 0..=MAX_ITERATIONS {
441        let mut chi_square = chi_square_graphs(graphs, &gains);
442        if chi_square <= 0.0 {
443            return Some(gains);
444        }
445        let delta_chi_square = chi_square - last_chi_square;
446        let fractional_delta = delta_chi_square / chi_square;
447        if fractional_delta <= 0.001 {
448            if delta_chi_square.abs() < 0.1 * chi_square {
449                convergence_count += 1;
450            }
451            if convergence_count > 5 {
452                return Some(gains);
453            }
454        } else if delta_chi_square.abs() > 0.1 * chi_square {
455            convergence_count = 0;
456        } else {
457            convergence_count = (convergence_count - 1).max(0);
458        }
459
460        if delta_chi_square <= 0.0 {
461            last_chi_square = chi_square;
462        } else {
463            gains.clone_from(&last_gains);
464            chi_square = chi_square_graphs(graphs, &gains);
465        }
466
467        let mut gradients = Vec::with_capacity(graphs.len());
468        let mut deltas = Vec::with_capacity(graphs.len());
469        for ((graph, antennas), receptor_gains) in graphs.iter().zip(antennas).zip(&gains) {
470            let (gradient, hessian) = gradient_hessian(graph, antennas, receptor_gains);
471            let delta = antennas
472                .iter()
473                .map(|antenna_id| {
474                    let hessian = hessian.get(antenna_id).copied().unwrap_or_default();
475                    let value = if hessian > f64::EPSILON {
476                        -gradient[antenna_id] / hessian / 2.0
477                    } else {
478                        Complex64::new(0.0, 0.0)
479                    };
480                    (*antenna_id, value)
481                })
482                .collect::<HashMap<_, _>>();
483            gradients.push((gradient, hessian));
484            deltas.push(delta);
485        }
486
487        last_gains.clone_from(&gains);
488        let step = optimize_step(graphs, &last_gains, &mut gains, &mut deltas, chi_square);
489        for (receptor_gains, receptor_delta) in gains.iter_mut().zip(&deltas) {
490            for (antenna_id, gain) in receptor_gains.iter_mut() {
491                *gain += receptor_delta.get(antenna_id).copied().unwrap_or_default();
492            }
493        }
494        for (receptor, (antennas, (gradient, hessian))) in
495            antennas.iter().zip(&gradients).enumerate()
496        {
497            trace::trace_phase_solver_iteration(PhaseSolverTraceEvent {
498                iteration,
499                refant_id,
500                chi_square,
501                last_chi_square,
502                delta_chi_square,
503                fractional_delta,
504                convergence_count,
505                step: &step,
506                antenna_ids: antennas,
507                last_gains: &last_gains[receptor],
508                gains: &gains[receptor],
509                gradient,
510                hessian,
511                delta: &deltas[receptor],
512            });
513        }
514    }
515    None
516}

518/// Refer one receptor's converged gains to the reference antenna's phase and
519/// derive its diagnostics. Errors and SNRs come from the solved gains, before
520/// a phase-only solve normalises them (`SolvableVisCal::formSolveSNR` runs
521/// before `enforceAPonSoln`).
522fn finish_receptor(
523    graph: &SolveGraph,
524    solved: BTreeSet<i32>,
525    mut gains: HashMap<i32, Complex64>,
526    solve_mode: GainSolveMode,
527    refant_id: i32,
528) -> SolvedReceptor {
529    rereference_complex64(&mut gains, refant_id, false);
530    let error_gains = gains
531        .iter()
532        .map(|(antenna_id, gain)| (*antenna_id, complex64_to_32(*gain)))
533        .collect::<HashMap<_, _>>();
534    let snr_amplitude = gains
535        .iter()
536        .map(|(antenna_id, gain)| (*antenna_id, gain.norm() as f32))
537        .collect::<HashMap<_, _>>();
538    if solve_mode == GainSolveMode::Phase {
539        rereference_complex64(&mut gains, refant_id, true);
540    }
541    let gains = gains
542        .into_iter()
543        .map(|(antenna_id, gain)| (antenna_id, complex64_to_32(gain)))
544        .collect::<HashMap<_, _>>();
545
546    let hessian = hessian(graph, &error_gains, &solved);
547    let error_chi_square = chi_square_for_errors(graph, &error_gains);
548    let good_count = solved.len();
549    let sample_count = undirected_collapsed_count(&graph.stats);
550    let degrees_of_freedom = (2 * sample_count.saturating_sub(good_count)).max(1);
551    let param_error = param_errors(&hessian, error_chi_square / degrees_of_freedom as f64);
552    let snr = snrs(&gains, &solved, &param_error, &snr_amplitude);
553
554    SolvedReceptor {
555        gains,
556        solved,
557        hessian,
558        param_error,
559        snr,
560        error_chi_square,
561        sample_count,
562        good_count,
563        snr_amplitude,
564    }
565}

567/// Errors from the reduced chi-square of all receptors together, as one
568/// `VisCalSolver2::getErrors` covers every parameter of the solve.
569fn apply_group_reduced_chi_square(solved: &mut [SolvedReceptor]) {
570    let error_chi_square = solved
571        .iter()
572        .map(|per_receptor| per_receptor.error_chi_square)
573        .sum::<f64>();
574    let sample_count = solved
575        .iter()
576        .map(|per_receptor| per_receptor.sample_count)
577        .sum::<usize>();
578    let good_count = solved
579        .iter()
580        .map(|per_receptor| per_receptor.good_count)
581        .sum::<usize>();
582    let degrees_of_freedom = (2 * sample_count.saturating_sub(good_count)).max(1);
583    let reduced_chi_square = error_chi_square / degrees_of_freedom as f64;
584
585    for per_receptor in solved {
586        per_receptor.param_error = param_errors(&per_receptor.hessian, reduced_chi_square);
587        per_receptor.snr = snrs(
588            &per_receptor.gains,
589            &per_receptor.solved,
590            &per_receptor.param_error,
591            &per_receptor.snr_amplitude,
592        );
593    }
594}

596fn snrs(
597    gains: &HashMap<i32, Complex32>,
598    solved: &BTreeSet<i32>,
599    param_error: &HashMap<i32, f32>,
600    snr_amplitude: &HashMap<i32, f32>,
601) -> HashMap<i32, f32> {
602    gains
603        .iter()
604        .map(|(antenna_id, gain)| {
605            let error = param_error.get(antenna_id).copied().unwrap_or_default();
606            let amplitude = snr_amplitude
607                .get(antenna_id)
608                .copied()
609                .unwrap_or_else(|| gain.norm());
610            let value = if error > 0.0 {
611                amplitude / error
612            } else if solved.contains(antenna_id) {
613                9_999_999.0
614            } else {
615                0.0
616            };
617            (*antenna_id, value)
618        })
619        .collect()
620}

622/// `VisCalSolver2::optStepSize2`: scale the Gauss–Newton step by trying it,
623/// doubling it while chi-square falls or halving it until it does, then
624/// taking the vertex of the parabola through the last three chi-squares.
625/// `gains` returns equal to `last_gains`; `deltas` returns scaled.
626fn optimize_step(
627    graphs: &[SolveGraph],
628    last_gains: &[HashMap<i32, Complex64>],
629    gains: &mut [HashMap<i32, Complex64>],
630    deltas: &mut [HashMap<i32, Complex64>],
631    current_chi_square: f64,
632) -> PhaseStepTrace {
633    let mut step = 1.0_f64;
634    let mut iterations = 0_usize;
635    let x0 = current_chi_square;
636    apply_step(gains, last_gains, deltas, 1.0);
637    let mut x1 = chi_square_graphs(graphs, gains);
638    let expanded;
639
640    let x2 = if x1 < x0 {
641        expanded = true;
642        apply_step(gains, last_gains, deltas, 2.0 * step);
643        let mut trial_x2 = chi_square_graphs(graphs, gains);
644        while trial_x2 < x1 {
645            iterations += 1;
646            step *= 2.0;
647            x1 = trial_x2;
648            apply_step(gains, last_gains, deltas, 2.0 * step);
649            trial_x2 = chi_square_graphs(graphs, gains);
650        }
651        trial_x2
652    } else {
653        expanded = false;
654        step *= 0.5;
655        apply_step(gains, last_gains, deltas, step);
656        let mut trial_x2 = x1;
657        x1 = chi_square_graphs(graphs, gains);
658        while x1 > x0 {
659            iterations += 1;
660            step *= 0.5;
661            trial_x2 = x1;
662            apply_step(gains, last_gains, deltas, step);
663            x1 = chi_square_graphs(graphs, gains);
664        }
665        trial_x2
666    };
667
668    let denominator = x0 - 2.0 * x1 + x2;
669    let opt_factor = if denominator.abs() > 0.0 {
670        step * (1.5 - (x2 - x1) / denominator)
671    } else {
672        0.0
673    };
674
675    for (gain, last_gain) in gains.iter_mut().zip(last_gains) {
676        gain.clone_from(last_gain);
677    }
678    if opt_factor > 0.0 {
679        for receptor_delta in deltas {
680            for value in receptor_delta.values_mut() {
681                *value *= opt_factor;
682            }
683        }
684    }
685    PhaseStepTrace {
686        x0,
687        x1,
688        x2,
689        step,
690        opt_factor,
691        expanded,
692        iterations,
693    }
694}
695
696fn apply_step(
697    gains: &mut [HashMap<i32, Complex64>],
698    bases: &[HashMap<i32, Complex64>],
699    deltas: &[HashMap<i32, Complex64>],
700    factor: f64,
701) {
702    for ((receptor_gains, base), delta) in gains.iter_mut().zip(bases).zip(deltas) {
703        for (antenna_id, gain) in receptor_gains.iter_mut() {
704            *gain = base[antenna_id] + factor * delta.get(antenna_id).copied().unwrap_or_default();
705        }
706    }
707}

709fn chi_square_graphs(graphs: &[SolveGraph], gains: &[HashMap<i32, Complex64>]) -> f64 {
710    graphs
711        .iter()
712        .zip(gains)
713        .map(|(graph, gains)| chi_square_complex64(graph, gains))
714        .sum()
715}
716
717fn chi_square_complex64(graph: &SolveGraph, gains: &HashMap<i32, Complex64>) -> f64 {
718    let mut total = 0.0_f64;
719    for (&(antenna1, antenna2), stat) in &graph.stats {
720        if antenna1 > antenna2 {
721            continue;
722        }
723        let weight = f64::from(graph.weight((antenna1, antenna2)));
724        if weight <= f64::EPSILON {
725            continue;
726        }
727        let gain1 = gains
728            .get(&antenna1)
729            .copied()
730            .unwrap_or_else(|| Complex64::new(1.0, 0.0));
731        let gain2 = gains
732            .get(&antenna2)
733            .copied()
734            .unwrap_or_else(|| Complex64::new(1.0, 0.0));
735        let observed = graph
736            .edges
737            .get(&(antenna1, antenna2))
738            .copied()
739            .map(complex32_to_64)
740            .unwrap_or_default();
741        let model = gain1 * gain2.conj();
742        let cross = (model.conj() * observed).re;
743        let contribution = stat.weighted_sample_power - 2.0 * cross + weight * model.norm_sqr();
744        total += contribution.max(0.0);
745    }
746    total
747}

749/// The chi-square's gradient and diagonal Hessian with respect to each gain
750/// of `antennas` (`VisCalSolver2::accGradHess2`).
751fn gradient_hessian(
752    graph: &SolveGraph,
753    antennas: &BTreeSet<i32>,
754    gains: &HashMap<i32, Complex64>,
755) -> (HashMap<i32, Complex64>, HashMap<i32, f64>) {
756    let mut gradient = antennas
757        .iter()
758        .map(|antenna_id| (*antenna_id, Complex64::new(0.0, 0.0)))
759        .collect::<HashMap<_, _>>();
760    let mut hessian = antennas
761        .iter()
762        .map(|antenna_id| (*antenna_id, 0.0_f64))
763        .collect::<HashMap<_, _>>();
764
765    for (&(antenna1, antenna2), edge) in &graph.edges {
766        if antenna1 > antenna2 {
767            continue;
768        }
769        let weight = f64::from(graph.weight((antenna1, antenna2)));
770        if weight <= f64::EPSILON {
771            continue;
772        }
773        let gain1 = gains
774            .get(&antenna1)
775            .copied()
776            .unwrap_or_else(|| Complex64::new(1.0, 0.0));
777        let gain2 = gains
778            .get(&antenna2)
779            .copied()
780            .unwrap_or_else(|| Complex64::new(1.0, 0.0));
781        let residual_sum = weight * gain1 * gain2.conj() - complex32_to_64(*edge);
782        let deriv1 = gain2.conj();
783        let deriv2 = gain1;
784        if let Some(value) = gradient.get_mut(&antenna1) {
785            *value += residual_sum * deriv1.conj();
786            *hessian.entry(antenna1).or_default() += weight * deriv1.norm_sqr();
787        }
788        if let Some(value) = gradient.get_mut(&antenna2) {
789            *value += deriv2 * residual_sum.conj();
790            *hessian.entry(antenna2).or_default() += weight * deriv2.norm_sqr();
791        }
792    }
793
794    (gradient, hessian)
795}

797/// Rotate every gain by the reference antenna's phase; with `phase_only`
798/// also normalise each to unit amplitude.
799fn rereference_complex64(gains: &mut HashMap<i32, Complex64>, refant_id: i32, phase_only: bool) {
800    let ref_gain = gains
801        .get(&refant_id)
802        .copied()
803        .unwrap_or_else(|| Complex64::new(1.0, 0.0));
804    let ref_norm = ref_gain.norm();
805    if ref_norm > f64::EPSILON {
806        let anchor = ref_gain / ref_norm;
807        for gain in gains.values_mut() {
808            *gain /= anchor;
809        }
810    }
811    if phase_only {
812        for gain in gains.values_mut() {
813            let norm = gain.norm();
814            if norm > f64::EPSILON {
815                *gain /= norm;
816            }
817        }
818        if let Some(gain) = gains.get_mut(&refant_id) {
819            *gain = Complex64::new(1.0, 0.0);
820        }
821    }
822}

824fn complex32_to_64(value: Complex32) -> Complex64 {
825    Complex64::new(f64::from(value.re), f64::from(value.im))
826}

828fn complex64_to_32(value: Complex64) -> Complex32 {
829    Complex32::new(value.re as f32, value.im as f32)

832fn hessian(
833    graph: &SolveGraph,
834    gains: &HashMap<i32, Complex32>,
835    solved: &BTreeSet<i32>,
836) -> HashMap<i32, f32> {
837    let mut hessian = HashMap::new();
838    for &(antenna_id, other_id) in graph.edges.keys() {
839        if !solved.contains(&antenna_id) || !solved.contains(&other_id) {
840            continue;
841        }
842        let weight = graph.weight((antenna_id, other_id));
843        if weight <= f32::EPSILON {
844            continue;
845        }
846        let other_power = gains
847            .get(&other_id)
848            .map(|gain| gain.norm_sqr())
849            .unwrap_or(1.0);
850        *hessian.entry(antenna_id).or_insert(0.0) += weight * other_power;
851    }
852    hessian
853}

855fn chi_square_for_errors(graph: &SolveGraph, gains: &HashMap<i32, Complex32>) -> f64 {
856    let mut total = 0.0_f64;
857    for (&(antenna1, antenna2), stat) in &graph.stats {
858        if antenna1 > antenna2 {
859            continue;
860        }
861        let gain1 = gains
862            .get(&antenna1)
863            .copied()
864            .unwrap_or_else(|| Complex32::new(1.0, 0.0));
865        let gain2 = gains
866            .get(&antenna2)
867            .copied()
868            .unwrap_or_else(|| Complex32::new(1.0, 0.0));
869        let observed = graph
870            .edges
871            .get(&(antenna1, antenna2))
872            .copied()
873            .unwrap_or_default();
874        let weight = f64::from(graph.weight((antenna1, antenna2)));
875        let model = gain1 * gain2.conj();
876        let cross = (model.conj() * observed).re;
877        let contribution = stat.weighted_sample_power - 2.0 * f64::from(cross)
878            + weight * f64::from(model.norm_sqr());
879        total += contribution.max(0.0);
880    }
881    total
882}

893fn param_errors(hessian: &HashMap<i32, f32>, reduced_chi_square: f64) -> HashMap<i32, f32> {
894    hessian
895        .iter()
896        .map(|(antenna_id, hessian)| {
897            let value = if *hessian > 0.0 && reduced_chi_square.is_finite() {
898                (2.0 * reduced_chi_square / f64::from(*hessian)).sqrt() as f32
899            } else {
900                0.0
901            };
902            (*antenna_id, value)
903        })
904        .collect()
905}

907fn reachable_antennas(graph: &SolveGraph, refant_id: i32) -> BTreeSet<i32> {
908    let mut reachable = BTreeSet::from([refant_id]);
909    let mut queue = VecDeque::from([refant_id]);
910    while let Some(antenna_id) = queue.pop_front() {
911        for (from, to) in graph.edges.keys() {
912            if *from == antenna_id && reachable.insert(*to) {
913                queue.push_back(*to);
914            }
915        }
916    }
917    reachable
918}
