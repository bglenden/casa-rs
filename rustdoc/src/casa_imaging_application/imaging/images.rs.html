15/// The pass images of domain `domain` in the normal state's x-major layout.
16///
17/// Data images become the residual planes `[channel][term][pol]`; on an
18/// initial pass the PSF moments and their `sumwt` come along.
19pub(crate) fn pass_images(domain: usize, images: &NormalImages, initial: bool) -> PassImages {
20    let (height, width) = images.planes[0].data[0].dim();
21    let first = images.first_plane as usize;
22    let mut residual = Vec::with_capacity(images.planes.len() * images.data_terms * images.pols);
23    for plane in &images.planes {
24        for image in &plane.data {
25            push_x_major(&mut residual, image);
26        }
27    }
28    let (psf, sum_weights) = if initial {
29        let mut psf = Vec::new();
30        let mut sum_weights = Vec::new();
31        for (index, plane) in images.planes.iter().enumerate() {
32            for image in &plane.psf {
33                push_x_major(&mut psf, image);
34            }
35            for term in 0..images.psf_terms {
36                for pol in 0..images.pols {
37                    sum_weights.push(images.psf_sumwt(index, term, pol));
38                }
39            }
40        }
41        (Some(psf), sum_weights)
42    } else {
43        (None, Vec::new())
44    };
45    PassImages {
46        domain,
47        shape: [width, height],
48        channels: first..first + images.planes.len(),
49        polarizations: images.pols,
50        residual,
51        psf,
52        sum_weights,
53    }
54}

56fn push_x_major(out: &mut Vec<f32>, image: &Array2<f32>) {
57    let (height, width) = image.dim();
58    out.reserve(width * height);
59    for x in 0..width {
60        for y in 0..height {
61            out.push(image[(y, x)]);
62        }
63    }
64}

66/// Model images of `planes` on domain `domain` of `generation`: one plane
67/// per output channel of a channel-local basis, the Taylor terms of one
68/// plane otherwise. Pixels outside the model's support predict nothing.
69pub(crate) fn model_images(
70    generation: &ModelGeneration,
71    domain: usize,
72    planes: PlaneRange,
73    channel_local: bool,
74) -> Result<ModelImages, ImagingError> {
75    let shape = generation.shape();
76    let [width, height] = shape.domains()[domain].pixels();
77    let pols = shape.polarizations();
78    let terms = if channel_local {
79        1
80    } else {
81        shape.coefficients()
82    };
83    let mut model_planes = Vec::with_capacity(planes.len());
84    for plane in planes.start..planes.end {
85        let mut images = Vec::with_capacity(terms * pols);
86        for term in 0..terms {
87            let coefficient = if channel_local { plane as usize } else { term };
88            for pol in 0..pols {
89                let samples = generation.read_plane(domain, coefficient, pol)?;
90                images.push(Array2::from_shape_fn((height, width), |(y, x)| {
91                    let sample = samples[y * width + x];
92                    match sample.support() {
93                        ModelSupport::Valid => sample.value().value() as f32,
94                        ModelSupport::Invalid => 0.0,
95                    }
96                }));
97            }
98        }
99        model_planes.push(ModelPlane { images });
100    }
101    Ok(ModelImages {
102        first_plane: planes.start,
103        planes: model_planes,
104    })
105}

107/// Prepared model grids of `planes` of domain `domain`.
108pub(crate) fn prepare_model(
109    operator: &MeasurementOperator,
110    generation: &ModelGeneration,
111    domain: usize,
112    planes: PlaneRange,
113) -> Result<PreparedModelGrids, ImagingError> {
114    let channel_local = matches!(
115        operator.basis(),
116        casa_imaging_operator::Basis::ChannelLocal { .. }
117    );
118    let images = model_images(generation, domain, planes, channel_local)?;
119    Ok(operator.prepare_model(&images, ModelPrescale::Unit)?)
120}
