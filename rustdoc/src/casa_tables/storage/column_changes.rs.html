21impl<'a> StagedEdit<'a> {
22    pub(crate) fn stage_columns(
23        table: &'a Path,
24        mut dat: TableDatContents,
25        added: Vec<ColumnAddition>,
26        dminfo: Option<ManagerAddition>,
27        existing: &[DataManagerInfo],
28    ) -> Result<Self, StorageError> {
29        let edit = Self::new(table)?;
30        let mut groups = BTreeSet::new();
31        let mut defaults = RecordValue::default();
32        let mut names: HashSet<_> = dat
33            .table_desc
34            .columns
35            .iter()
36            .map(|c| c.col_name.clone())
37            .collect();
38        let mut manager_names: HashSet<_> = existing.iter().map(|dm| dm.name.clone()).collect();
39        let spec = dminfo.as_ref().map(|dm| &dm.spec);
40        let mut assigned = if let Some(dm) = &dminfo {
41            if !dm.name.is_empty() && !manager_names.insert(dm.name.clone()) {
42                return Err(StorageError::Schema(format!(
43                    "storage manager {} already exists",
44                    dm.name
45                )));
46            }
47            Some((
48                allocate(&mut dat, &dm.kind),
49                dm.kind.clone(),
50                dm.name.clone(),
51            ))
52        } else {
53            existing
54                .iter()
55                .find(|dm| admits_add(&dm.dm_type))
56                .map(|dm| (dm.seq_nr, dm.dm_type.clone(), dm.name.clone()))
57        };
58        for column in added {
59            let mut desc = ColumnDescContents::from_column_schema(&column.schema);
60            if !names.insert(desc.col_name.clone()) {
61                return Err(StorageError::Schema(format!(
62                    "column {} already exists",
63                    desc.col_name
64                )));
65            }
66            let (seq, kind, group) = match &assigned {
67                Some(binding) => binding.clone(),
68                None => {
69                    let kind = column
70                        .manager_type
71                        .unwrap_or_else(|| "StandardStMan".to_string());
72                    let requested = column.manager_group.unwrap_or_else(|| kind.clone());
73                    let mut name = requested.clone();
74                    let mut suffix = 1u64;
75                    while !manager_names.insert(name.clone()) {
76                        name = format!("{requested}_{suffix}");
77                        suffix += 1;
78                    }
79                    let chosen = (allocate(&mut dat, &kind), kind, name);
80                    if admits_add(&chosen.1) {
81                        assigned = Some(chosen.clone());
82                    }
83                    chosen
84                }
85            };
86            // A manager's stored name can be empty even when its column
87            // descriptions name a group (for example StManAipsIO).
88            // Joining a manager keeps that descriptor's declared binding.
89            if let Some((declared, _)) = dat
90                .table_desc
91                .columns
92                .iter()
93                .zip(&dat.column_set.columns)
94                .find(|(_, entry)| entry.dm_seq_nr == seq)
95            {
96                desc.data_manager_type = declared.data_manager_type.clone();
97                desc.data_manager_group = declared.data_manager_group.clone();
98            } else {
99                desc.data_manager_type = kind;
100                desc.data_manager_group = group;
101            }
102            desc.keywords = column.keywords;
103            if let Some(value) = column.default {
104                defaults.push(RecordField::new(&desc.col_name, value));
105            }
106            dat.column_set.columns.push(PlainColumnEntry {
107                original_name: desc.col_name.clone(),
108                dm_seq_nr: seq,
109                is_array: desc.is_array,
110            });
111            dat.table_desc.columns.push(desc);
112            groups.insert(seq);
113        }
114        for seq in groups {
115            let columns: Vec<_> = dat
116                .table_desc
117                .columns
118                .iter()
119                .zip(&dat.column_set.columns)
120                .filter(|(_, entry)| entry.dm_seq_nr == seq)
121                .map(|(column, _)| column.clone())
122                .collect();
123            let original = existing.iter().find(|dm| dm.seq_nr == seq);
124            let output =
125                edit.write_added_columns(&dat, seq, &columns, original, &defaults, spec)?;
126            if let Some(blob) = output {
127                dat.column_set
128                    .data_managers
129                    .iter_mut()
130                    .find(|dm| dm.seq_nr == seq)
131                    .expect("allocated manager")
132                    .data = blob;
133            }
134        }
135        edit.stage_control(&dat)?;
136        Ok(edit)
137    }

139    fn write_added_columns(
140        &self,
141        dat: &TableDatContents,
142        seq: u32,
143        columns: &[ColumnDescContents],
144        existing: Option<&DataManagerInfo>,
145        defaults: &RecordValue,
146        spec: Option<&RecordValue>,
147    ) -> Result<Option<Vec<u8>>, StorageError> {
148        let name = existing.map_or(&columns[0].data_manager_group, |dm| &dm.name);
149        let kind = existing.map_or(&columns[0].data_manager_type, |dm| &dm.dm_type);
150        let file = format!("table.f{seq}");
151        let dst = self.staged.path().join(&file);
152        let nrows = usize::try_from(dat.nrrow).map_err(|_| {
153            StorageError::FormatMismatch("column addition row count overflow".into())
154        })?;
155        let defaults = RecordValue::new(
156            defaults
157                .fields()
158                .iter()
159                .filter(|field| columns.iter().any(|column| column.col_name == field.name))
160                .cloned()
161                .collect(),
162        );
163        let mut rows = if let Some(existing) = existing {
164            let names: Vec<&str> = existing.columns.iter().map(String::as_str).collect();
165            CompositeStorage
166                .load_selected_columns_with_row_hint(self.table, &names, Some(dat.nrrow))?
167                .rows
168        } else {
169            vec![RecordValue::default(); nrows]
170        };
171        for row in &mut rows {
172            for field in defaults.fields() {
173                row.push(field.clone());
174            }
175        }
176        match kind.as_str() {
177            "StandardStMan" if existing.is_some() => Ok(Some(standard_stman::rewrite_ssm_rows(
178                &self.table.join(file),
179                &dst,
180                columns,
181                &rows,
182                name,
183            )?)),
184            "StandardStMan" => {
185                let layout = Some((
186                    uint_spec(spec, "BUCKETSIZE").filter(|&n| n >= 128),
187                    uint_spec(spec, "PERSCACHESIZE").unwrap_or(2),
188                ));
189                Ok(Some(standard_stman::write_ssm_with_bucket(
190                    &dst,
191                    columns,
192                    &rows,
193                    dat.big_endian,
194                    name,
195                    layout,
196                )?))
197            }
198            "StManAipsIO" => {
199                stman_aipsio::write_named_stman_file(
200                    &dst,
201                    seq,
202                    columns,
203                    &rows,
204                    ByteOrder::BigEndian,
205                    name,
206                )?;
207                Ok(None)
208            }
209            "IncrementalStMan" => {
210                incremental_stman::write_ism_with_layout(
211                    &dst,
212                    columns,
213                    &rows,
214                    dat.big_endian,
215                    uint_spec(spec, "BUCKETSIZE").filter(|&n| n > 0),
216                    uint_spec(spec, "PERSCACHESIZE").unwrap_or(1),
217                )?;
218                Ok(Some(incremental_stman::serialize_ism_dm_blob(name)?))
219            }
220            "TiledShapeStMan" | "TiledColumnStMan" | "TiledCellStMan" => {
221                let shape = tile_shape(spec)?;
222                if columns.iter().any(|col| !col.is_array || col.is_record()) {
223                    return Err(StorageError::Schema(format!(
224                        "{kind} requires array columns"
225                    )));
226                }
227                tiled_stman::save_tiled_columns(
228                    self.staged.path(),
229                    seq,
230                    kind,
231                    columns,
232                    &rows,
233                    dat.big_endian,
234                    shape.as_deref(),
235                    name,
236                    &dat.table_desc.private_keywords,
237                )?;
238                if let Some(cache) = uint_spec(spec, "MAXIMUMCACHESIZE") {
239                    tiled_stman::append::set_cache(&dst, u64::from(cache))?;
240                }
241                Ok(None)
242            }
243            other => Err(StorageError::UnsupportedDataManager(other.to_string())),
244        }
245    }
246}

248fn allocate(dat: &mut TableDatContents, kind: &str) -> u32 {
249    let seq = dat.column_set.seq_count;
250    dat.column_set.seq_count += 1;
251    dat.column_set.data_managers.push(DataManagerEntry {
252        type_name: kind.to_string(),
253        seq_nr: seq,
254        data: Vec::new(),
255    });
256    seq
257}
258
259fn admits_add(kind: &str) -> bool {
260    matches!(kind, "StandardStMan" | "StManAipsIO")
261}
262
263fn uint_spec(spec: Option<&RecordValue>, key: &str) -> Option<u32> {
264    match spec?.get(key)? {
265        Value::Scalar(ScalarValue::Int32(x)) => u32::try_from(*x).ok(),
266        Value::Scalar(ScalarValue::Int64(x)) => u32::try_from(*x).ok(),
267        Value::Scalar(ScalarValue::UInt32(x)) => Some(*x),
268        _ => None,
269    }
270}

272fn tile_shape(spec: Option<&RecordValue>) -> Result<Option<Vec<usize>>, StorageError> {
273    let Some(value) = spec.and_then(|spec| spec.get("DEFAULTTILESHAPE")) else {
274        return Ok(None);
275    };
276    let invalid = || StorageError::Schema("DEFAULTTILESHAPE must contain positive integers".into());
277    let dims: Vec<i64> = match value {
278        Value::Array(casa_types::ArrayValue::Int32(a)) => {
279            a.iter().copied().map(i64::from).collect()
280        }
281        Value::Array(casa_types::ArrayValue::Int64(a)) => a.iter().copied().collect(),
282        _ => return Err(invalid()),
283    };
284    dims.iter()
285        .map(|&n| {
286            usize::try_from(n)
287                .ok()
288                .filter(|&size| size > 0)
289                .ok_or_else(invalid)
290        })
291        .collect::<Result<Vec<_>, _>>()
292        .map(Some)
293}
