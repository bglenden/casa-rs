3//! the weight below which `tsdimaging` masks its image
4//! (`do_weight_mask`).

6use ndarray::{Array2, ArrayView2, ArrayView3};
7
8use crate::SingleDishError;

10/// `SDGrid::pickWeights` from `WEIGHT`, `[spectrum][correlation]`: each
11/// spectrum's weight, the same in each of its `channels` channels
12/// (`[spectrum][channel]`). One correlation's weight is taken as it is; two
13/// correlations' `w₀` and `w₁` combine as the weight of their mean,
14/// `4 w₀ w₁ / (w₀ + w₁)`, and as 0 when their sum is zero (CAS-9957).
15///
16/// # Errors
17///
18/// Weights of more than two correlations: CASA refuses to image full-Stokes
19/// data.
20pub fn weights_from_rows(
21    weight: ArrayView2<'_, f32>,
22    channels: usize,
23) -> Result<Array2<f32>, SingleDishError> {
24    let (spectra, correlations) = weight.dim();
25    check_correlations(correlations)?;
26    let combine = |spectrum: usize| match correlations {
27        1 => weight[[spectrum, 0]],
28        _ => combined(weight[[spectrum, 0]], weight[[spectrum, 1]]),
29    };
30    let mut weights = Array2::zeros((spectra, channels));
31    for (spectrum, mut row) in weights.rows_mut().into_iter().enumerate() {
32        row.fill(combine(spectrum));
33    }
34    Ok(weights)
35}

37/// `SDGrid::pickWeights` from `WEIGHT_SPECTRUM`,
38/// `[spectrum][channel][correlation]`: [`weights_from_rows`] for each
39/// channel.
40///
41/// # Errors
42///
43/// Weights of more than two correlations.
44pub fn weights_from_spectra(weight: ArrayView3<'_, f32>) -> Result<Array2<f32>, SingleDishError> {
45    let (spectra, channels, correlations) = weight.dim();
46    check_correlations(correlations)?;
47    Ok(Array2::from_shape_fn(
48        (spectra, channels),
49        |(spectrum, channel)| match correlations {
50            1 => weight[[spectrum, channel, 0]],
51            _ => combined(
52                weight[[spectrum, channel, 0]],
53                weight[[spectrum, channel, 1]],
54            ),
55        },
56    ))
57}
58
59fn check_correlations(correlations: usize) -> Result<(), SingleDishError> {
60    if correlations == 1 || correlations == 2 {
61        Ok(())
62    } else {
63        Err(SingleDishError::Correlations(format!(
64            "imaging single-dish data of {correlations} correlations is not supported; CASA \
65             images one or two (full-Stokes data, four correlations, are refused)"
66        )))
67    }
68}
69
70/// Two correlations' weights combined (`toStokesWeight`).
71fn combined(w0: f32, w1: f32) -> f32 {
72    let denominator = w0 + w1;
73    if denominator.abs() < f32::MIN_POSITIVE {
74        0.0
75    } else {
76        4.0 * (w0 * w1) / denominator
77    }
78}

80/// `do_weight_mask`: the weight a pixel must exceed to stay unmasked,
81/// `minweight` times the median of the positive weights through every
82/// plane (`ia.statistics` of the weight image where it is positive), or
83/// `None` when no weight is positive (CASA then sets no mask). The median
84/// of an even count is the mean of the middle two.
85///
86/// One departure (`docs/CASA (C++) bugs.md`): CASA writes the threshold into
87/// its mask expression with `%f`, six decimal places, so a threshold under
88/// 5e-7 masks nothing. This is the threshold itself.
89#[must_use]
90pub fn weight_mask_threshold(weights: &[f32], minweight: f64) -> Option<f64> {
91    let mut positive: Vec<f32> = weights.iter().copied().filter(|&w| w > 0.0).collect();
92    if positive.is_empty() {
93        return None;
94    }
95    let even = positive.len().is_multiple_of(2);
96    let middle = positive.len() / 2;
97    let (below, &mut upper, _) = positive.select_nth_unstable_by(middle, f32::total_cmp);
98    let median = if even {
99        // An even count: the lower middle is the largest below.
100        let lower = below.iter().copied().fold(f32::NEG_INFINITY, f32::max);
101        (f64::from(lower) + f64::from(upper)) / 2.0
102    } else {
103        f64::from(upper)
104    };
105    Some(median * minweight)
106}

108#[cfg(test)]
109mod tests {
110    use ndarray::{Array3, array};
111
112    use super::*;
113
114    #[test]
115    fn two_correlations_combine_as_the_weight_of_their_mean() {
116        let rows = weights_from_rows(array![[2.0_f32, 2.0], [1.0, 3.0], [0.0, 0.0]].view(), 3)
117            .expect("weights");
118        assert_eq!(rows.row(0).to_vec(), vec![4.0; 3]);
119        assert_eq!(rows.row(1).to_vec(), vec![3.0; 3]);
120        assert_eq!(rows.row(2).to_vec(), vec![0.0; 3]);
121        let one = weights_from_rows(array![[5.0_f32]].view(), 2).expect("weights");
122        assert_eq!(one.row(0).to_vec(), vec![5.0; 2]);
123    }
124
125    #[test]
126    fn spectra_combine_channel_by_channel() {
127        let mut spectrum = Array3::zeros((1, 2, 2));
128        spectrum[[0, 0, 0]] = 1.0_f32;
129        spectrum[[0, 0, 1]] = 1.0;
130        spectrum[[0, 1, 0]] = 2.0;
131        spectrum[[0, 1, 1]] = 6.0;
132        let weights = weights_from_spectra(spectrum.view()).expect("weights");
133        assert_eq!(weights.row(0).to_vec(), vec![2.0, 6.0]);
134    }
135
136    #[test]
137    fn four_correlations_are_refused() {
138        assert!(weights_from_rows(Array2::<f32>::ones((1, 4)).view(), 1).is_err());
139        assert!(weights_from_spectra(Array3::<f32>::ones((1, 1, 4)).view()).is_err());
140    }
141
142    #[test]
143    fn the_mask_threshold_is_minweight_times_the_median_positive_weight() {
144        assert_eq!(
145            weight_mask_threshold(&[0.0, 3.0, 1.0, 2.0, -1.0], 0.5),
146            Some(1.0)
147        );
148        assert_eq!(
149            weight_mask_threshold(&[4.0, 1.0, 0.0, 3.0, 2.0], 0.5),
150            Some(1.25)
151        );
152        assert_eq!(weight_mask_threshold(&[0.0, 0.0], 0.5), None);
153    }
