19pub(super) struct OpenedMs {
20    pub(super) measurement_set: MeasurementSet,
21    pub(super) lock: TableWriteLock,
22}
23
24impl OpenedMs {
25    pub(super) fn open(path: &Path) -> Result<Self, TaskFailure> {
26        let lock = TableWriteLock::acquire_to_reread(path, 0).map_err(execution)?;
27        Self::with_lock(path, lock)
28    }
29
30    fn with_lock(path: &Path, lock: TableWriteLock) -> Result<Self, TaskFailure> {
31        let measurement_set = MeasurementSet::open(path).map_err(execution)?;
32        Ok(Self {
33            measurement_set,
34            lock,
35        })
36    }
37
38    pub(super) fn initialize(mut self, column: VisibilityDataColumn) -> Result<Self, TaskFailure> {
39        self.lock = initialize_scratch_column(&mut self.measurement_set, column, self.lock)
40            .map_err(execution)?;
41        Ok(self)
42    }
43
44    pub(super) fn reload(self, path: &Path) -> Result<Self, TaskFailure> {
45        drop(self.measurement_set);
46        Self::with_lock(path, self.lock)
47    }
48
49    pub(super) fn release(self) -> Result<(), TaskFailure> {
50        drop(self.measurement_set);
51        self.lock.release().map_err(execution)
52    }
53}

55fn execution(error: impl std::fmt::Display) -> TaskFailure {
56    TaskFailure::execution(error.to_string())
57}

59pub(super) fn execute(
60    task: &TaskInvocation<'_>,
61    request: &ExecuteApplyTaskRequest,
62) -> Result<Option<ApplyMultiMsReport>, TaskFailure> {
63    let path = &request.measurement_set;
64    let Some(parts) = partition_paths(path)? else {
65        return Ok(None);
66    };
67    let backup = if request.flagbackup && request.apply_mode.writes_flags() {
68        Some(
69            save_next_partition_flag_backup(path, &parts, "applycal")
70                .map_err(|error| TaskFailure::execution(error.to_string()))?,
71        )
72    } else {
73        None
74    };
75
76    let mut session = task.session().clone();
77    let gain_paths = task
78        .texts("gaintable")?
79        .iter()
80        .map(|path| {
81            let path = if path.is_empty() {
82                Path::new(".")
83            } else {
84                Path::new(path)
85            };
86            table_cache_name(path)
87                .map(|absolute| ParameterValue::String(absolute.to_string_lossy().into_owned()))
88                .map_err(|error| TaskFailure::execution(error.to_string()))
89        })
90        .collect::<Result<Vec<_>, _>>()?;
91    session
92        .set("gaintable", ParameterValue::Array(gain_paths))
93        .map_err(|error| TaskFailure::execution(error.to_string()))?;
94    if backup.is_some() {
95        session
96            .set("flagbackup", ParameterValue::Bool(false))
97            .map_err(|error| TaskFailure::execution(error.to_string()))?;
98    }
99    let mut partitions = Vec::with_capacity(parts.len());
100    for path in parts {
101        session
102            .set(
103                "vis",
104                ParameterValue::String(path.to_string_lossy().into_owned()),
105            )
106            .map_err(|error| TaskFailure::execution(error.to_string()))?;
107        let result = run_partition(&TaskInvocation::new(&session, task.workspace()), &path);
108        if let Err(error) = &result {
109            tracing::warn!(partition = %path.display(), %error, "applycal failed for Multi-MS partition");
110        }
111        partitions.push(ApplyPartitionReport {
112            measurement_set: path,
113            result,
114        });
115    }
116    Ok(Some(ApplyMultiMsReport {
117        measurement_set: path.clone(),
118        flag_backup_version: backup,
119        partitions,
120    }))
121}

123pub(super) fn partition_paths(path: &Path) -> Result<Option<Vec<PathBuf>>, TaskFailure> {
124    let Ok(text) = std::fs::read_to_string(path.join("table.info")) else {
125        return Ok(None);
126    };
127    if TableInfo::parse(&text).sub_type != "CONCATENATED" {
128        return Ok(None);
129    }
130    let parts = Table::concatenated_source_paths(path)
131        .map_err(|error| TaskFailure::execution(error.to_string()))?
132        .ok_or_else(|| TaskFailure::execution("Multi-MS has no concatenation"))?;
133    // CASA opens the aggregate before dispatch, so a broken reference
134    // fails before any partition receives changes.
135    for part in &parts {
136        MeasurementSet::open(part).map_err(|error| {
137            TaskFailure::execution(format!(
138                "open Multi-MS partition {}: {error}",
139                part.display()
140            ))
141        })?;
142    }
143    Ok(Some(parts))
144}

146fn run_partition(task: &TaskInvocation<'_>, path: &Path) -> Result<ApplyExecutionReport, String> {
147    let CalibrationTaskRequest::ExecuteApply(request) =
148        super::casa_read::task_request(task).map_err(|failure| failure.message)?
149    else {
150        unreachable!("Multi-MS applycal uses an apply request");
151    };
152    let report = request.execute_apply().map_err(|error| error.to_string())?;
153    super::history::record_applycal(task, path);
154    Ok(report)
155}
