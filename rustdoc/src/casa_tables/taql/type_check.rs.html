10#[derive(Clone, Copy, Debug, PartialEq, Eq)]
11enum Kind {
12    Unknown,
13    Bool,
14    Int,
15    Real,
16    Complex,
17    Text,
18    Date,
19    Regex,
20}
21
22impl Kind {
23    fn numeric(self) -> bool {
24        matches!(self, Self::Int | Self::Real | Self::Complex)
25    }
26
27    fn combine(self, other: Self) -> Result<Self, TaqlError> {
28        match (self, other) {
29            (Self::Unknown, _) | (_, Self::Unknown) => Ok(Self::Unknown),
30            (left, right) if left == right => Ok(left),
31            (left, right) if left.numeric() && right.numeric() => {
32                if matches!(left, Self::Complex) || matches!(right, Self::Complex) {
33                    Ok(Self::Complex)
34                } else {
35                    Ok(Self::Real)
36                }
37            }
38            _ => Err(invalid(format!(
39                "cannot combine {self:?} and {other:?} operands"
40            ))),
41        }
42    }
43
44    fn require(self, allowed: &[Self], expression: &str) -> Result<(), TaqlError> {
45        if self == Self::Unknown || allowed.contains(&self) {
46            Ok(())
47        } else {
48            Err(invalid(format!(
49                "{expression} does not accept a {self:?} operand"
50            )))
51        }
52    }
53}
54
55const NUMERIC: &[Kind] = &[Kind::Int, Kind::Real, Kind::Complex];
56const REAL: &[Kind] = &[Kind::Int, Kind::Real];
57const ORDERED: &[Kind] = &[Kind::Int, Kind::Real, Kind::Complex, Kind::Text, Kind::Date];
58
59#[derive(Clone, Copy, Debug, PartialEq, Eq)]
60enum Shape {
61    Unknown,
62    Scalar,
63    Array(Option<usize>),
64}
65
66impl Shape {
67    fn combine(self, other: Self) -> Self {
68        match (self, other) {
69            (Self::Scalar, other) | (other, Self::Scalar) => other,
70            (Self::Array(left), Self::Array(right)) if left == right => Self::Array(left),
71            (Self::Array(_), Self::Array(_)) => Self::Array(None),
72            _ => Self::Unknown,
73        }
74    }
75}
76
77type ExprType = (Kind, Shape);
78
79fn scalar(kind: Kind) -> ExprType {
80    (kind, Shape::Scalar)
81}
82
83fn invalid(message: impl Into<String>) -> TaqlError {
84    TaqlError::TypeError {
85        message: message.into(),
86    }
87}

89/// Validate an expression parsed with casacore's record-selection rules.
90/// The expression itself may be non-Boolean. The result reports whether
91/// it has a type that can be used as a row predicate.
92/// # Errors
93/// A known operand type, column, array index, or function call is invalid.
94#[doc(hidden)]
95pub fn validate_record_expression(
96    expr: &Expr,
97    table: &Table,
98) -> Result<RecordExpressionType, TaqlError> {
99    super::calc::validate_calc_columns(expr, table)?;
100    let (kind, shape) = check(expr, table)?;
101    Ok(match (kind, shape) {
102        (Kind::Bool | Kind::Unknown, Shape::Array(_)) => RecordExpressionType::BooleanArray,
103        (Kind::Bool | Kind::Unknown, _) => RecordExpressionType::PossibleRowPredicate,
104        _ => RecordExpressionType::NonBoolean,
105    })
106}
107
108/// Whether an expression can participate in a CASA Boolean row selection.
109#[derive(Clone, Copy, PartialEq, Eq)]
110pub enum RecordExpressionType {
111    /// Scalar Boolean, including expressions with a dynamic result type.
112    PossibleRowPredicate,
113    /// Boolean elements, but an array result cannot directly select rows.
114    BooleanArray,
115    /// A known non-Boolean element type.
116    NonBoolean,
117}

119fn check(expr: &Expr, table: &Table) -> Result<ExprType, TaqlError> {
120    use Expr::*;
121    let expression_type = match expr {
122        Literal(value) => scalar(literal_kind(value)),
123        RowNumber => scalar(Kind::Int),
124        ColumnRef(column) => table
125            .schema()
126            .and_then(|schema| schema.column(&column.column))
127            .map(|column| {
128                let shape = match column.column_type() {
129                    ColumnType::Scalar => Shape::Scalar,
130                    ColumnType::Array(ArrayShapeContract::Fixed { shape }) => {
131                        Shape::Array(Some(shape.len()))
132                    }
133                    ColumnType::Array(ArrayShapeContract::Variable { ndim }) => Shape::Array(*ndim),
134                    ColumnType::Record => Shape::Unknown,
135                };
136                (
137                    column.data_type().map_or(Kind::Unknown, primitive_kind),
138                    shape,
139                )
140            })
141            .unwrap_or((Kind::Unknown, Shape::Unknown)),
142        Unary { op, operand } => {
143            let (kind, shape) = check(operand, table)?;
144            match op {
145                UnaryOp::Not => kind.require(&[Kind::Bool], "logical NOT")?,
146                UnaryOp::BitNot => kind.require(&[Kind::Int], "bitwise NOT")?,
147                UnaryOp::Negate => kind.require(NUMERIC, "negation")?,
148            }
149            (kind, shape)
150        }
151        Binary { left, op, right } => binary(*op, check(left, table)?, check(right, table)?)?,
152        ArrayIndex { array, indices } => index(check(array, table)?, indices, table)?,
153        ArrayLiteral(values) => {
154            let mut kinds = values
155                .iter()
156                .map(|value| check(value, table).map(|(kind, _)| kind));
157            let kind = kinds.next().transpose()?.unwrap_or(Kind::Unknown);
158            let kind = kinds.try_fold(kind, |left, right| left.combine(right?))?;
159            (kind, Shape::Array(None))
160        }
161        FunctionCall { name, args } => {
162            functions::validate_function_call(name, args.len())?;
163            let operands = args
164                .iter()
165                .map(|arg| check(arg, table))
166                .collect::<Result<Vec<_>, _>>()?;
167            function_type(name, &operands)?
168        }
169        Between {
170            expr, low, high, ..
171        } => {
172            let left = check(expr, table)?;
173            binary(BinaryOp::Ge, left, check(low, table)?)?;
174            binary(BinaryOp::Le, left, check(high, table)?)?
175        }
176        In { expr, values, .. } => {
177            let (left, shape) = check(expr, table)?;
178            for value in values {
179                left.combine(check(value, table)?.0)?;
180            }
181            (Kind::Bool, shape)
182        }
183        InSet { expr, elements, .. } => {
184            if elements.is_empty() {
185                return Err(invalid("set selection requires at least one element"));
186            }
187            let (left, shape) = check(expr, table)?;
188            for element in elements {
189                match element {
190                    InSetElement::Value(value) => {
191                        left.combine(check(value, table)?.0)?;
192                    }
193                    InSetElement::Range { start, end, step } => {
194                        for value in [start, end, step].into_iter().flatten() {
195                            left.combine(check(value, table)?.0)?;
196                        }
197                    }
198                }
199            }
200            (Kind::Bool, shape)
201        }
202        Like { expr, pattern, .. } | RegexMatch { expr, pattern, .. } => {
203            let (operand, shape) = check(expr, table)?;
204            operand.require(&[Kind::Text], "pattern match")?;
205            check(pattern, table)?
206                .0
207                .require(&[Kind::Text, Kind::Regex], "pattern match")?;
208            (Kind::Bool, shape)
209        }
210        IsNull { expr, .. } => (Kind::Bool, check(expr, table)?.1),
211        BoundRowLookup { index, .. } | Aggregate { arg: index, .. } => {
212            check(index, table)?;
213            (Kind::Unknown, Shape::Unknown)
214        }
215        PreparedConstant(_) | BoundRowFunction(_) | Star | Subquery(_) => {
216            (Kind::Unknown, Shape::Unknown)
217        }
218    };
219    Ok(expression_type)
220}

222fn primitive_kind(ty: PrimitiveType) -> Kind {
223    use PrimitiveType::*;
224    match ty {
225        Bool => Kind::Bool,
226        UInt8 | UInt16 | UInt32 | Int16 | Int32 | Int64 => Kind::Int,
227        Float32 | Float64 => Kind::Real,
228        Complex32 | Complex64 => Kind::Complex,
229        String => Kind::Text,
230    }
231}
232
233fn literal_kind(value: &Literal) -> Kind {
234    match value {
235        Literal::Bool(_) => Kind::Bool,
236        Literal::Int(_) => Kind::Int,
237        Literal::Float(_) | Literal::Quantity { .. } => Kind::Real,
238        Literal::Complex(_) => Kind::Complex,
239        Literal::String(_) => Kind::Text,
240        Literal::Regex { .. } => Kind::Regex,
241        Literal::Null => Kind::Unknown,
242    }
243}

245fn binary(op: BinaryOp, left: ExprType, right: ExprType) -> Result<ExprType, TaqlError> {
246    use BinaryOp::*;
247    let shape = left.1.combine(right.1);
248    if matches!(op, Eq | Ne)
249        && matches!(
250            (left.0, right.0),
251            (Kind::Text, Kind::Regex) | (Kind::Regex, Kind::Text)
252        )
253    {
254        return Ok((Kind::Bool, shape));
255    }
256    if matches!(op, Eq | Ne | Lt | Le | Gt | Ge)
257        && matches!(
258            (left.0, right.0),
259            (Kind::Date, Kind::Int | Kind::Real | Kind::Text)
260                | (Kind::Int | Kind::Real | Kind::Text, Kind::Date)
261        )
262    {
263        return Ok((Kind::Bool, shape));
264    }
265    let expression = format!("{op:?}");
266    if matches!(op, Add | Sub) && (left.0 == Kind::Date || right.0 == Kind::Date) {
267        let kinds = (left.0, right.0);
268        match kinds {
269            (Kind::Date, Kind::Date) if op == Sub => return Ok((Kind::Real, shape)),
270            (Kind::Date, Kind::Int | Kind::Real | Kind::Unknown) => return Ok((Kind::Date, shape)),
271            (Kind::Int | Kind::Real | Kind::Unknown, Kind::Date) if op == Add => {
272                return Ok((Kind::Date, shape));
273            }
274            _ => {}
275        }
276    }
277    let compatible = left.0.combine(right.0)?;
278    let inputs = match op {
279        And | Or => &[Kind::Bool][..],
280        BitAnd | BitOr | BitXor => &[Kind::Int][..],
281        Eq | Ne => return Ok((Kind::Bool, shape)),
282        Lt | Le | Gt | Ge => ORDERED,
283        Add if left.0 == Kind::Text || right.0 == Kind::Text => &[Kind::Text][..],
284        _ => NUMERIC,
285    };
286    left.0.require(inputs, &expression)?;
287    right.0.require(inputs, &expression)?;
288    let kind = match op {
289        And | Or | Lt | Le | Gt | Ge => Kind::Bool,
290        BitAnd | BitOr | BitXor => Kind::Int,
291        Div | IntDiv if compatible == Kind::Int => Kind::Real,
292        _ => compatible,
293    };
294    Ok((kind, shape))
295}

297fn index(array: ExprType, indices: &[IndexElement], table: &Table) -> Result<ExprType, TaqlError> {
298    if array.1 == Shape::Scalar {
299        return Err(invalid("indexing requires an array"));
300    }
301    let mut slices = 0;
302    for element in indices {
303        match element {
304            IndexElement::Single(expr) => {
305                let (kind, shape) = check(expr, table)?;
306                if kind == Kind::Bool && matches!(shape, Shape::Array(_)) && indices.len() == 1 {
307                    return Ok((array.0, Shape::Array(None)));
308                }
309                kind.require(&[Kind::Int], "array index")?;
310            }
311            IndexElement::Slice { start, end, step } => {
312                slices += 1;
313                for expr in [start, end, step].into_iter().flatten() {
314                    check(expr, table)?.0.require(&[Kind::Int], "array slice")?;
315                }
316                if matches!(step, Some(Expr::Literal(Literal::Int(stride))) if *stride <= 0) {
317                    return Err(invalid("array slice stride must be positive"));
318                }
319            }
320        }
321    }
322    if matches!(array.1, Shape::Array(Some(ndim)) if ndim != indices.len()) {
323        return Err(invalid(
324            "number of subscripts does not match array dimensionality",
325        ));
326    }
327    Ok((
328        array.0,
329        if slices == 0 {
330            Shape::Scalar
331        } else {
332            Shape::Array(Some(slices))
333        },
334    ))
335}

337fn function_type(name: &str, args: &[ExprType]) -> Result<ExprType, TaqlError> {
338    if functions::has_udf(name) {
339        return Ok((Kind::Unknown, Shape::Unknown));
340    }
341    let (first, shape) = args
342        .first()
343        .copied()
344        .unwrap_or((Kind::Unknown, Shape::Unknown));
345    let result = match name.to_ascii_lowercase().as_str() {
346        "abs" | "norm" => {
347            first.require(NUMERIC, name)?;
348            (
349                if first == Kind::Complex {
350                    Kind::Real
351                } else {
352                    first
353                },
354                shape,
355            )
356        }
357        "arg" | "imag" => {
358            first.require(NUMERIC, name)?;
359            (Kind::Real, shape)
360        }
361        "asin" | "acos" | "atan" | "sin" | "cos" | "tan" | "sinh" | "cosh" | "tanh" | "exp"
362        | "log" | "ln" | "log10" | "sqrt" | "conj" => {
363            first.require(NUMERIC, name)?;
364            (
365                if first == Kind::Complex {
366                    Kind::Complex
367                } else {
368                    Kind::Real
369                },
370                shape,
371            )
372        }
373        "square" | "cube" => {
374            first.require(NUMERIC, name)?;
375            (first, shape)
376        }
377        "floor" | "ceil" | "round" | "sign" => {
378            first.require(REAL, name)?;
379            (first, shape)
380        }
381        "any" | "all" => {
382            first.require(&[Kind::Bool], name)?;
383            scalar(Kind::Bool)
384        }
385        "ntrue" | "nfalse" => {
386            first.require(&[Kind::Bool], name)?;
387            scalar(Kind::Int)
388        }
389        "isnan" | "isinf" | "isfinite" => {
390            first.require(NUMERIC, name)?;
391            (Kind::Bool, shape)
392        }
393        "real" | "double" | "float" => (Kind::Real, shape),
394        "int" => (Kind::Int, shape),
395        "bool" => (Kind::Bool, shape),
396        "rownumber" | "rownr" | "rowid" | "nelements" | "ndim" => scalar(Kind::Int),
397        "date" | "datetime" => (Kind::Date, shape),
398        _ => (Kind::Unknown, Shape::Unknown),
399    };
400    Ok(result)
