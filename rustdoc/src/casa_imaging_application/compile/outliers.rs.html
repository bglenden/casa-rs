14/// One outlier image domain as the outlier file defines it.
15pub(super) struct OutlierDomain {
16    pub(super) name: String,
17    pub(super) output: PathBuf,
18    pub(super) image_size: usize,
19    pub(super) cell_arcsec: f64,
20    pub(super) phase_center: String,
21    pub(super) mask: DomainMask,
22}
23
24pub(super) fn read_outlier_domains(
25    path: &Path,
26    default_size: usize,
27    default_cell_arcsec: f64,
28) -> Result<Vec<OutlierDomain>, ApplicationError> {
29    let text = fs::read_to_string(path).map_err(|error| {
30        boxed(format!(
31            "cannot read outlier file {}: {error}",
32            path.display()
33        ))
34    })?;
35    let records = parse_records(path, &text)?;
36    if records.is_empty() {
37        return Err(boxed("outlier file did not define any image domains"));
38    }
39    records
40        .into_iter()
41        .enumerate()
42        .map(|(ordinal, fields)| {
43            compile_record(path, ordinal, fields, default_size, default_cell_arcsec)
44        })
45        .collect()
46}
47
48fn parse_records(
49    path: &Path,
50    text: &str,
51) -> Result<Vec<BTreeMap<String, String>>, ApplicationError> {
52    let mut records = Vec::new();
53    let mut current = BTreeMap::new();
54    for (line_index, raw) in text.lines().enumerate() {
55        let line = raw.trim();
56        if line.is_empty() || line.starts_with('#') {
57            continue;
58        }
59        let Some((key, value)) = line.split_once('=') else {
60            return Err(boxed(format!(
61                "outlier file {} line {} must contain one parameter=value pair",
62                path.display(),
63                line_index + 1
64            )));
65        };
66        if value.contains('=') {
67            return Err(boxed(format!(
68                "outlier file {} line {} contains more than one '='",
69                path.display(),
70                line_index + 1
71            )));
72        }
73        let key = key.trim().to_ascii_lowercase();
74        if key == "imagename" && !current.is_empty() {
75            records.push(current);
76            current = BTreeMap::new();
77        }
78        if current
79            .insert(key.clone(), trim_string(value).to_owned())
80            .is_some()
81        {
82            return Err(boxed(format!(
83                "outlier file {} line {} repeats field {key:?}",
84                path.display(),
85                line_index + 1
86            )));
87        }
88    }
89    if !current.is_empty() {
90        records.push(current);
91    }
92    Ok(records)
93}

95fn compile_record(
96    path: &Path,
97    ordinal: usize,
98    mut fields: BTreeMap<String, String>,
99    default_size: usize,
100    default_cell_arcsec: f64,
101) -> Result<OutlierDomain, ApplicationError> {
102    let name = take_required(&mut fields, "imagename", ordinal)?;
103    let output = resolve_output(path, &name);
104    let image_size = fields
105        .remove("imsize")
106        .map(|value| parse_square_size(&value))
107        .transpose()?
108        .unwrap_or(default_size);
109    let cell_arcsec = fields
110        .remove("cell")
111        .map(|value| parse_square_cell(&value))
112        .transpose()?
113        .unwrap_or(default_cell_arcsec);
114    let phase_center = take_required(&mut fields, "phasecenter", ordinal)?;
115    let mask = fields
116        .remove("mask")
117        .filter(|value| !value.is_empty())
118        .map(|value| circle_mask(&value, image_size))
119        .transpose()?
120        .unwrap_or(DomainMask::FullPlane);
121
122    admit_default(&mut fields, "usemask", &["", "user"])?;
123    admit_default(&mut fields, "specmode", &["", "mfs", "cont"])?;
124    admit_default(&mut fields, "nchan", &["", "1"])?;
125    admit_default(&mut fields, "nterms", &["", "1"])?;
126    admit_default(&mut fields, "gridder", &["", "standard", "gridft", "ft"])?;
127    admit_default(&mut fields, "deconvolver", &["", "hogbom"])?;
128    admit_default(&mut fields, "wprojplanes", &["", "1"])?;
129    for unsupported in ["startmodel", "start", "width", "reffreq"] {
130        if fields
131            .remove(unsupported)
132            .is_some_and(|value| !value.is_empty())
133        {
134            return Err(boxed(format!(
135                "outlier image {ordinal} sets unsupported field {unsupported:?}"
136            )));
137        }
138    }
139    if !fields.is_empty() {
140        return Err(boxed(format!(
141            "outlier image {ordinal} contains unsupported field(s): {}",
142            fields.keys().cloned().collect::<Vec<_>>().join(", ")
143        )));
144    }
145    Ok(OutlierDomain {
146        name,
147        output,
148        image_size,
149        cell_arcsec,
150        phase_center,
151        mask,
152    })
153}

155fn take_required(
156    fields: &mut BTreeMap<String, String>,
157    key: &str,
158    ordinal: usize,
159) -> Result<String, ApplicationError> {
160    fields
161        .remove(key)
162        .filter(|value| !value.is_empty())
163        .ok_or_else(|| boxed(format!("outlier image {ordinal} is missing required {key}")))
164}
165
166fn admit_default(
167    fields: &mut BTreeMap<String, String>,
168    key: &str,
169    admitted: &[&str],
170) -> Result<(), ApplicationError> {
171    let Some(value) = fields.remove(key) else {
172        return Ok(());
173    };
174    if admitted
175        .iter()
176        .any(|candidate| value.eq_ignore_ascii_case(candidate))
177    {
178        Ok(())
179    } else {
180        Err(boxed(format!(
181            "outlier field {key}={value:?} is outside the installed multi-domain slice"
182        )))
183    }
184}

186fn parse_square_size(text: &str) -> Result<usize, ApplicationError> {
187    let values = parse_list(text)
188        .into_iter()
189        .map(|value| value.parse::<usize>())
190        .collect::<Result<Vec<_>, _>>()?;
191    match values.as_slice() {
192        [size] if *size > 0 => Ok(*size),
193        [width, height] if *width > 0 && width == height => Ok(*width),
194        _ => Err(boxed(
195            "outlier imsize must be a positive square scalar or pair",
196        )),
197    }
198}

200fn parse_square_cell(text: &str) -> Result<f64, ApplicationError> {
201    let values = parse_list(text)
202        .into_iter()
203        .map(|value| parse_arcsec(&value))
204        .collect::<Result<Vec<_>, _>>()?;
205    match values.as_slice() {
206        [cell] if cell.is_finite() && *cell > 0.0 => Ok(*cell),
207        [x, y] if x.is_finite() && *x > 0.0 && (*x - *y).abs() <= f64::EPSILON => Ok(*x),
208        _ => Err(boxed(
209            "outlier cell must be one positive arcsec value or an equal pair",
210        )),
211    }
212}

214fn parse_arcsec(text: &str) -> Result<f64, ApplicationError> {
215    let lower = trim_string(text).to_ascii_lowercase();
216    lower
217        .strip_suffix("arcsec")
218        .unwrap_or(&lower)
219        .trim()
220        .parse::<f64>()
221        .map_err(|error| boxed(format!("invalid outlier cell {text:?}: {error}")))
222}

224fn circle_mask(text: &str, image_size: usize) -> Result<DomainMask, ApplicationError> {
225    let compact = trim_string(text)
226        .chars()
227        .filter(|character| !character.is_whitespace())
228        .collect::<String>()
229        .to_ascii_lowercase();
230    let body = compact
231        .strip_prefix("circle[[")
232        .and_then(|value| value.strip_suffix(']'))
233        .ok_or_else(|| boxed("outlier mask must use circle[[xpix,ypix],rpix]"))?;
234    let (centre, radius) = body
235        .split_once("],")
236        .ok_or_else(|| boxed("outlier mask must use circle[[xpix,ypix],rpix]"))?;
237    let (x, y) = centre
238        .split_once(',')
239        .ok_or_else(|| boxed("outlier mask must use circle[[xpix,ypix],rpix]"))?;
240    let centre = [parse_pixels(x)?, parse_pixels(y)?];
241    let radius = parse_pixels(radius)?;
242    if radius < 0.0
243        || centre
244            .iter()
245            .any(|value| *value < 0.0 || *value >= image_size as f64)
246    {
247        return Err(boxed("outlier circle mask exceeds its image domain"));
248    }
249    let radius_squared = radius * radius;
250    let support_len = image_size
251        .checked_mul(image_size)
252        .ok_or_else(|| boxed("outlier mask support size overflowed"))?;
253    let mut support = Vec::with_capacity(support_len);
254    for x in 0..image_size {
255        for y in 0..image_size {
256            let dx = x as f64 - centre[0];
257            let dy = y as f64 - centre[1];
258            support.push(dx * dx + dy * dy <= radius_squared);
259        }
260    }
261    Ok(DomainMask::PixelSupport(support.into_boxed_slice()))
262}
263
264fn parse_pixels(text: &str) -> Result<f64, ApplicationError> {
265    let value = text
266        .strip_suffix("pix")
267        .ok_or_else(|| boxed("outlier circle coordinates must use pix units"))?
268        .parse::<f64>()?;
269    if value.is_finite() {
270        Ok(value)
271    } else {
272        Err(boxed("outlier circle coordinates must be finite"))
273    }
274}
275
276fn parse_list(text: &str) -> Vec<String> {
277    text.trim()
278        .trim_start_matches('[')
279        .trim_end_matches(']')
280        .split(',')
281        .map(trim_string)
282        .filter(|value| !value.is_empty())
283        .map(str::to_owned)
284        .collect()
285}
286
287fn trim_string(value: &str) -> &str {
288    value
289        .trim()
290        .trim_matches(',')
291        .trim()
292        .trim_matches(|character| character == '"' || character == '\'')
293        .trim()
294}

296fn resolve_output(outlier_file: &Path, name: &str) -> PathBuf {
297    let output = PathBuf::from(name);
298    if output.is_absolute() {
299        output
300    } else {
301        outlier_file
302            .parent()
303            .filter(|parent| !parent.as_os_str().is_empty())
304            .map_or(output.clone(), |parent| parent.join(output))
305    }
306}
307
308fn boxed(message: impl Into<String>) -> ApplicationError {
309    Box::new(std::io::Error::new(
310        std::io::ErrorKind::InvalidInput,
311        message.into(),
312    ))
313}

315#[cfg(test)]
316mod tests {
317    use super::*;
318
319    #[test]
320    fn parses_casa_multifield_fixture_shape_and_circle_support() {
321        let directory = tempfile::tempdir().expect("temporary directory");
322        let path = directory.path().join("outlier.txt");
323        std::fs::write(
324            &path,
325            "imagename='outlier'\nimsize=[80,80]\ncell=['8arcsec','8arcsec']\nphasecenter='J2000 19:58:40.895 +40.55.58.543'\nmask='circle[[40pix,40pix],10pix]'\n",
326        )
327        .expect("write outlier fixture");
328        let domains = read_outlier_domains(&path, 100, 4.0).expect("compile outlier");
329        assert_eq!(domains.len(), 1);
330        assert_eq!(domains[0].image_size, 80);
331        assert_eq!(domains[0].cell_arcsec, 8.0);
332        assert_eq!(domains[0].output, directory.path().join("outlier"));
333        let DomainMask::PixelSupport(support) = &domains[0].mask else {
334            panic!("expected pixel support");
335        };
336        assert!(support[40 * 80 + 40]);
337        assert!(!support[0]);
338    }
