13fn planned_vec<T>(capacity: usize) -> Result<Vec<T>, SpectralOperatorError> {
14    let mut values = Vec::new();
15    values
16        .try_reserve_exact(capacity)
17        .map_err(|_| SpectralOperatorError::ResidencyOverflow)?;
18    if values.capacity() != capacity {
19        return Err(SpectralOperatorError::ResidencyOverflow);
20    }
21    Ok(values)
22}

24fn planned_prediction_lane(
25    group_capacity: usize,
26    record_layout: GriddedNormalRecordLayout,
27) -> Result<RwLock<GriddedNormalPredictionLane>, SpectralOperatorError> {
28    let prediction_width = record_layout.prediction_width();
29    let value_capacity = group_capacity
30        .checked_mul(prediction_width)
31        .ok_or(SpectralOperatorError::ResidencyOverflow)?;
32    let (model_scratch, moment_scratch) = match record_layout {
33        GriddedNormalRecordLayout::Scalar
34        | GriddedNormalRecordLayout::ChannelLocal { .. }
35        | GriddedNormalRecordLayout::TaylorWithCoordinates(_)
36        | GriddedNormalRecordLayout::TaylorViaChannelMajor { .. }
37        | GriddedNormalRecordLayout::Joint { .. } => (planned_vec(0)?, planned_vec(0)?),
38        GriddedNormalRecordLayout::Taylor(_) => {
39            let mut model_scratch = planned_vec(prediction_width)?;
40            model_scratch.resize(prediction_width, Complex64::default());
41            let mut moment_scratch = planned_vec(record_layout.normal_moments())?;
42            moment_scratch.resize(record_layout.normal_moments(), 0.0);
43            (model_scratch, moment_scratch)
44        }
45    };
46    Ok(RwLock::new(GriddedNormalPredictionLane {
47        groups: 0..0,
48        values: planned_vec(value_capacity)?,
49        model_scratch,
50        moment_scratch,
51    }))
52}

68impl GriddedNormalTileGeometry {
69    fn new(
70        grid_shape: [usize; 2],
71        key: GriddedNormalTileKey,
72        support: usize,
73    ) -> Result<Self, SpectralOperatorError> {
74        let axis = |extent: usize, tile: i32| -> Result<(usize, usize), SpectralOperatorError> {
75            let center =
76                i64::try_from(extent / 2).map_err(|_| SpectralOperatorError::ResidencyOverflow)?;
77            let raw_start = center + i64::from(tile) * GRIDDED_NORMAL_TILE_EDGE as i64;
78            let raw_end = raw_start + GRIDDED_NORMAL_TILE_EDGE as i64;
79            let extent =
80                i64::try_from(extent).map_err(|_| SpectralOperatorError::ResidencyOverflow)?;
81            let core_start = usize::try_from(raw_start.clamp(0, extent))
82                .map_err(|_| SpectralOperatorError::ResidencyOverflow)?;
83            let core_end = usize::try_from(raw_end.clamp(0, extent))
84                .map_err(|_| SpectralOperatorError::ResidencyOverflow)?;
85            if core_start >= core_end {
86                return Err(SpectralOperatorError::UnsupportedGeometry);
87            }
88            Ok((core_start, core_end))
89        };
90        let (x0, x1) = axis(grid_shape[0], key.x)?;
91        let (y0, y1) = axis(grid_shape[1], key.y)?;
92        let origin = [x0.saturating_sub(support), y0.saturating_sub(support)];
93        let end = [
94            x1.saturating_add(support).min(grid_shape[0]),
95            y1.saturating_add(support).min(grid_shape[1]),
96        ];
97        Ok(Self {
98            key,
99            origin,
100            shape: [end[0] - origin[0], end[1] - origin[1]],
101            support,
102        })
103    }

105    pub(super) fn translated_taps(
106        self,
107        taps: SampleTaps,
108    ) -> Result<SampleTaps, SpectralOperatorError> {
109        let translated = SampleTaps {
110            x: TapSpan {
111                start: taps
112                    .x
113                    .start
114                    .checked_sub(self.origin[0])
115                    .ok_or(SpectralOperatorError::InvalidGriddedRecord)?,
116                weight_index: taps.x.weight_index,
117            },
118            y: TapSpan {
119                start: taps
120                    .y
121                    .start
122                    .checked_sub(self.origin[1])
123                    .ok_or(SpectralOperatorError::InvalidGriddedRecord)?,
124                weight_index: taps.y.weight_index,
125            },
126        };
127        if translated
128            .x
129            .start
130            .checked_add(2 * self.support)
131            .is_none_or(|end| end >= self.shape[0])
132            || translated
133                .y
134                .start
135                .checked_add(2 * self.support)
136                .is_none_or(|end| end >= self.shape[1])
137        {
138            return Err(SpectralOperatorError::InvalidGriddedRecord);
139        }
140        Ok(translated)
141    }

176    pub(super) fn new_for_projection(
177        grid_shapes: impl IntoIterator<Item = [usize; 2]>,
178        support: usize,
179        aw_projection: bool,
180    ) -> Result<Self, SpectralOperatorError> {
181        let catalogs = grid_shapes
182            .into_iter()
183            .map(|shape| {
184                GriddedNormalTileCatalog::new_for_projection(shape, support, aw_projection)
185            })
186            .collect::<Result<Vec<_>, _>>()?;
187        if catalogs.is_empty() {
188            return Err(SpectralOperatorError::UnsupportedGeometry);
189        }
190        let mut offsets: Vec<usize> = Vec::with_capacity(catalogs.len() + 1);
191        offsets.push(0);
192        for catalog in &catalogs {
193            offsets.push(
194                offsets
195                    .last()
196                    .copied()
197                    .and_then(|offset| offset.checked_add(catalog.geometries.len()))
198                    .ok_or(SpectralOperatorError::ResidencyOverflow)?,
199            );
200        }
201        Ok(Self { catalogs, offsets })
202    }

264    pub(super) fn accumulators(
265        &self,
266        core_depth: usize,
267        record_capacity: usize,
268    ) -> Result<Vec<Mutex<GriddedNormalTileAccumulator>>, SpectralOperatorError> {
269        let count = task_capacity(self.tile_count(), record_capacity)?;
270        let maximum_cells = self.maximum_cell_count()?;
271        let mut accumulators = Vec::new();
272        accumulators
273            .try_reserve_exact(count)
274            .map_err(|_| SpectralOperatorError::ResidencyOverflow)?;
275        for _ in 0..count {
276            accumulators.push(Mutex::new(GriddedNormalTileAccumulator::new(
277                maximum_cells,
278                core_depth,
279            )));
280        }
281        if accumulators.len() != count || accumulators.capacity() != count {
282            return Err(SpectralOperatorError::ResidencyOverflow);
283        }
284        Ok(accumulators)
285    }

310pub(super) fn storage_layout_for_projection(
311    grid_shapes: impl IntoIterator<Item = [usize; 2]>,
312    coefficient_terms: usize,
313    support: usize,
314    aw_projection: bool,
315) -> Result<GriddedNormalStorageLayout, SpectralOperatorError> {
316    if coefficient_terms == 0 {
317        return Err(SpectralOperatorError::InvalidSlab);
318    }
319    let catalogs =
320        GriddedNormalDomainTileCatalogs::new_for_projection(grid_shapes, support, aw_projection)?;
321    let maximum_cells = catalogs.maximum_cell_count()?;
322    let merge_complex_values = catalogs
323        .catalogs
324        .iter()
325        .try_fold(0_usize, |total, catalog| {
326            catalog.grid_shape[0]
327                .checked_mul(catalog.grid_shape[1])
328                .and_then(|cells| total.checked_add(cells))
329                .ok_or(SpectralOperatorError::ResidencyOverflow)
330        })?
331        .checked_mul(coefficient_terms)
332        .and_then(|values| values.checked_mul(2))
333        .ok_or(SpectralOperatorError::ResidencyOverflow)?;
334    let tile_count = catalogs.tile_count();
335    let catalog_bytes = catalogs
336        .catalogs
337        .capacity()
338        .checked_mul(size_of::<GriddedNormalTileCatalog>())
339        .and_then(|bytes| {
340            catalogs
341                .offsets
342                .capacity()
343                .checked_mul(size_of::<usize>())
344                .and_then(|offsets| bytes.checked_add(offsets))
345        })
346        .and_then(|bytes| {
347            catalogs.catalogs.iter().try_fold(bytes, |total, catalog| {
348                catalog
349                    .geometries
350                    .capacity()
351                    .checked_mul(size_of::<GriddedNormalTileGeometry>())
352                    .and_then(|geometry| total.checked_add(geometry))
353            })
354        })
355        .ok_or(SpectralOperatorError::ResidencyOverflow)?;
356    let routing_bytes = tile_count
357        .checked_mul(size_of::<u32>() * 2)
358        .and_then(|bytes| {
359            tile_count
360                .checked_add(1)
361                .and_then(|count| count.checked_mul(size_of::<u32>()))
362                .and_then(|offsets| bytes.checked_add(offsets))
363        })
364        .ok_or(SpectralOperatorError::ResidencyOverflow)?;
365    let slot_metadata_bytes = coefficient_terms
366        .checked_mul(size_of::<Array2<Complex64>>() * 2)
367        .and_then(|bytes| bytes.checked_add(size_of::<Mutex<GriddedNormalTileAccumulator>>()))
368        .and_then(|bytes| bytes.checked_add(size_of::<GriddedNormalTileTask>()))
369        .ok_or(SpectralOperatorError::ResidencyOverflow)?;
370    let domain_count = catalogs.domain_count();
371    let merge_descriptor_bytes = domain_count
372        .checked_mul(size_of::<Vec<Array2<Complex64>>>() * 2)
373        .and_then(|bytes| {
374            domain_count
375                .checked_mul(coefficient_terms)
376                .and_then(|planes| planes.checked_mul(size_of::<Array2<Complex64>>() * 2))
377                .and_then(|planes| bytes.checked_add(planes))
378        })
379        .ok_or(SpectralOperatorError::ResidencyOverflow)?;
380    let fixed_metadata_bytes = catalog_bytes
381        .checked_add(routing_bytes)
382        .and_then(|bytes| bytes.checked_add(merge_descriptor_bytes))
383        .ok_or(SpectralOperatorError::ResidencyOverflow)?;
384    Ok(GriddedNormalStorageLayout {
385        tile_count,
386        maximum_tile_cells: maximum_cells,
387        accumulation_width: coefficient_terms,
388        convolution_support: support,
389        aw_projection,
390        merge_complex_values,
391        fixed_metadata_bytes,
392        slot_metadata_bytes,
393    })
394}

397    #[cfg(test)]
398    pub(super) fn new(
399        grid_shape: [usize; 2],
400        support: usize,
401    ) -> Result<Self, SpectralOperatorError> {
402        Self::new_for_projection(grid_shape, support, false)

404
405    fn new_for_projection(
406        grid_shape: [usize; 2],
407        support: usize,
408        aw_projection: bool,
409    ) -> Result<Self, SpectralOperatorError> {
410        let (minimum_key, maximum_key) =
411            Self::key_bounds_for_projection(grid_shape, support, aw_projection)?;
412        let tiles_x = usize::try_from(i64::from(maximum_key.x) - i64::from(minimum_key.x) + 1)
413            .map_err(|_| SpectralOperatorError::ResidencyOverflow)?;
414        let tiles_y = usize::try_from(i64::from(maximum_key.y) - i64::from(minimum_key.y) + 1)
415            .map_err(|_| SpectralOperatorError::ResidencyOverflow)?;
416        let tile_count = tiles_x
417            .checked_mul(tiles_y)
418            .ok_or(SpectralOperatorError::ResidencyOverflow)?;
419        let mut geometries = Vec::new();
420        geometries
421            .try_reserve_exact(tile_count)
422            .map_err(|_| SpectralOperatorError::ResidencyOverflow)?;
423        for x in minimum_key.x..=maximum_key.x {
424            for y in minimum_key.y..=maximum_key.y {
425                geometries.push(GriddedNormalTileGeometry::new(
426                    grid_shape,
427                    GriddedNormalTileKey { x, y },
428                    support,
429                )?);
430            }
431        }
432        if geometries.capacity() != tile_count || geometries.len() != tile_count {
433            return Err(SpectralOperatorError::ResidencyOverflow);
434        }
435        Ok(Self {
436            grid_shape,
437            minimum_key,
438            tiles_y,
439            geometries,
440            support,
441        })
442    }
443
444    fn key_bounds_for_projection(
445        grid_shape: [usize; 2],
446        support: usize,
447        aw_projection: bool,
448    ) -> Result<(GriddedNormalTileKey, GriddedNormalTileKey), SpectralOperatorError> {
449        if grid_shape
450            .into_iter()
451            .any(|extent| extent <= support.saturating_mul(2))
452        {
453            return Err(SpectralOperatorError::UnsupportedGeometry);
454        }
455        let axis = |coordinate: usize, extent: usize| -> Result<i32, SpectralOperatorError> {
456            let coordinate =
457                i64::try_from(coordinate).map_err(|_| SpectralOperatorError::ResidencyOverflow)?;
458            let center =
459                i64::try_from(extent / 2).map_err(|_| SpectralOperatorError::ResidencyOverflow)?;
460            i32::try_from((coordinate - center).div_euclid(GRIDDED_NORMAL_TILE_EDGE as i64))
461                .map_err(|_| SpectralOperatorError::ResidencyOverflow)
462        };
463        Ok((
464            GriddedNormalTileKey {
465                x: axis(if aw_projection { 0 } else { support }, grid_shape[0])?,
466                y: axis(if aw_projection { 0 } else { support }, grid_shape[1])?,
467            },
468            GriddedNormalTileKey {
469                x: axis(
470                    if aw_projection {
471                        grid_shape[0] - 1
472                    } else {
473                        grid_shape[0] - support - 1
474                    },
475                    grid_shape[0],
476                )?,
477                y: axis(
478                    if aw_projection {
479                        grid_shape[1] - 1
480                    } else {
481                        grid_shape[1] - support - 1
482                    },
483                    grid_shape[1],
484                )?,
485            },
486        ))
487    }
488
489    fn tile_ordinal(&self, taps: SampleTaps) -> Result<usize, SpectralOperatorError> {
490        let center = [
491            taps.x
492                .start
493                .checked_add(self.support)
494                .ok_or(SpectralOperatorError::InvalidGriddedRecord)?,
495            taps.y
496                .start
497                .checked_add(self.support)
498                .ok_or(SpectralOperatorError::InvalidGriddedRecord)?,
499        ];
500        self.tile_ordinal_center(center)
501    }
502
503    fn tile_ordinal_center(&self, center: [usize; 2]) -> Result<usize, SpectralOperatorError> {
504        if center[0] >= self.grid_shape[0] || center[1] >= self.grid_shape[1] {
505            return Err(SpectralOperatorError::InvalidGriddedRecord);
506        }
507        let axis = |coordinate: usize, extent: usize| -> Result<i32, SpectralOperatorError> {
508            let coordinate =
509                i64::try_from(coordinate).map_err(|_| SpectralOperatorError::ResidencyOverflow)?;
510            let anchor =
511                i64::try_from(extent / 2).map_err(|_| SpectralOperatorError::ResidencyOverflow)?;
512            i32::try_from((coordinate - anchor).div_euclid(GRIDDED_NORMAL_TILE_EDGE as i64))
513                .map_err(|_| SpectralOperatorError::ResidencyOverflow)
514        };
515        let key = GriddedNormalTileKey {
516            x: axis(center[0], self.grid_shape[0])?,
517            y: axis(center[1], self.grid_shape[1])?,
518        };
519        let x = usize::try_from(i64::from(key.x) - i64::from(self.minimum_key.x))
520            .map_err(|_| SpectralOperatorError::InvalidGriddedRecord)?;
521        let y = usize::try_from(i64::from(key.y) - i64::from(self.minimum_key.y))
522            .map_err(|_| SpectralOperatorError::InvalidGriddedRecord)?;
523        let ordinal = x
524            .checked_mul(self.tiles_y)
525            .and_then(|base| base.checked_add(y))
526            .ok_or(SpectralOperatorError::ResidencyOverflow)?;
527        if self
528            .geometries
529            .get(ordinal)
530            .is_none_or(|geometry| geometry.key != key)
531        {
532            return Err(SpectralOperatorError::InvalidGriddedRecord);
533        }
534        Ok(ordinal)
535    }
536}

538pub(super) struct GriddedNormalTileAccumulator {
539    cell_capacity: usize,
540    pub(super) grids: Vec<Array2<Complex64>>,
541    pub(super) compensations: Vec<Array2<Complex64>>,
542}
543
544impl GriddedNormalTileAccumulator {
545    fn new(cell_capacity: usize, core_depth: usize) -> Self {
546        let shape = (1, cell_capacity);
547        let planes = || (0..core_depth).map(|_| Array2::zeros(shape)).collect();
548        Self {
549            cell_capacity,
550            grids: planes(),
551            compensations: planes(),
552        }
553    }
554
555    fn bind_geometry(&mut self, shape: [usize; 2]) -> Result<(), SpectralOperatorError> {
556        let cells = shape[0]
557            .checked_mul(shape[1])
558            .ok_or(SpectralOperatorError::ResidencyOverflow)?;
559        if cells > self.cell_capacity {
560            return Err(SpectralOperatorError::ResidencyOverflow);
561        }
562        for plane in self.grids.iter_mut().chain(&mut self.compensations) {
563            let (mut values, offset) = std::mem::take(plane).into_raw_vec_and_offset();
564            if offset.is_some_and(|offset| offset != 0) || values.capacity() != self.cell_capacity {
565                return Err(SpectralOperatorError::ResidencyOverflow);
566            }
567            values.resize(cells, Complex64::default());
568            values.fill(Complex64::default());
569            *plane = Array2::from_shape_vec((shape[0], shape[1]), values)
570                .map_err(|_| SpectralOperatorError::UnsupportedGeometry)?;
571        }
572        Ok(())
573    }
574}

736    fn prepare<'a, I>(
737        &mut self,
738        first_sequence: u64,
739        descriptors: &[BlockDescriptor],
740        frames: I,
741        catalogs: &GriddedNormalDomainTileCatalogs,
742        operators: &[SpectralSlabOperator],
743        output_channels: usize,
744    ) -> Result<(), SpectralOperatorError>
745    where
746        I: IntoIterator<Item = (u64, &'a [u8])>,
747    {
748        if self.active_frames != 0 {
749            return Err(SpectralOperatorError::BlockSequence);
750        }
751        self.frame_sequences.clear();
752        self.frame_record_counts.clear();
753        self.groups.clear();
754        self.classifications.clear();
755        self.routes.clear();
756        self.tasks.clear();
757        self.tile_counts.fill(0);
758        self.tile_cursors.fill(0);
759        self.tile_offsets.fill(0);
760        self.lane_record_counts.fill(0);
761        self.lane_tap_visit_counts.fill(0);
762
763        let prepared = (|| {
764            for (frame_ordinal, (sequence, encoded)) in frames.into_iter().enumerate() {
765                let frame_ordinal_u64 = u64::try_from(frame_ordinal)
766                    .map_err(|_| SpectralOperatorError::CoverageOverflow)?;
767                let expected = first_sequence
768                    .checked_add(frame_ordinal_u64)
769                    .ok_or(SpectralOperatorError::CoverageOverflow)?;
770                if sequence != expected {
771                    return Err(SpectralOperatorError::BlockSequence);
772                }
773                let descriptor = descriptors
774                    .get(
775                        usize::try_from(sequence)
776                            .map_err(|_| SpectralOperatorError::GriddedRecordMismatch)?,
777                    )
778                    .ok_or(SpectralOperatorError::GriddedRecordMismatch)?;
779                validate_encoded_block(descriptor, encoded, self.record_bytes)?;
780                let record_count = encoded.len() / self.record_bytes;
781                if record_count
782                    > *self
783                        .frame_capacities
784                        .get(frame_ordinal)
785                        .ok_or(SpectralOperatorError::ResidencyOverflow)?
786                {
787                    return Err(SpectralOperatorError::ResidencyOverflow);
788                }
789                let frame_ordinal_u32 = u32::try_from(frame_ordinal)
790                    .map_err(|_| SpectralOperatorError::CoverageOverflow)?;
791                match self.record_layout {
792                    GriddedNormalRecordLayout::Scalar
793                    | GriddedNormalRecordLayout::ChannelLocal { .. }
794                    | GriddedNormalRecordLayout::TaylorWithCoordinates(_)
795                    | GriddedNormalRecordLayout::TaylorViaChannelMajor { .. }
796                    | GriddedNormalRecordLayout::Joint { .. } => {
797                        let mut group_start = 0usize;
798                        for (record_ordinal, bytes) in
799                            encoded.chunks_exact(self.record_bytes).enumerate()
800                        {
801                            let record = decode_domain_record(bytes, catalogs, output_channels)?;
802                            let group_ordinal = u32::try_from(self.groups.len())
803                                .map_err(|_| SpectralOperatorError::CoverageOverflow)?;
804                            let (tile_ordinal, tap_count) = if let Some(aw) = record.aw {
805                                let (center, tap_count) = operators
806                                    .get(record.chart_ordinal)
807                                    .ok_or(SpectralOperatorError::InvalidGriddedRecord)?
808                                    .aw_gridded_grid_footprint(aw)?;
809                                (
810                                    catalogs.tile_ordinal_center(record.chart_ordinal, center)?,
811                                    tap_count,
812                                )
813                            } else {
814                                (
815                                    catalogs.tile_ordinal(record.chart_ordinal, record.taps)?,
816                                    usize::try_from(GRIDDED_NORMAL_TAPS_PER_RECORD)
817                                        .map_err(|_| SpectralOperatorError::CoverageOverflow)?,
818                                )
819                            };
820                            self.classifications.push(GriddedNormalClassification {
821                                tile_ordinal: u32::try_from(tile_ordinal)
822                                    .map_err(|_| SpectralOperatorError::CoverageOverflow)?,
823                                group_ordinal,
824                                frame_ordinal: frame_ordinal_u32,
825                                record_ordinal: u32::try_from(record_ordinal)
826                                    .map_err(|_| SpectralOperatorError::CoverageOverflow)?,
827                                tap_count: u32::try_from(tap_count)
828                                    .map_err(|_| SpectralOperatorError::CoverageOverflow)?,
829                            });
830                            self.tile_counts[tile_ordinal] = self.tile_counts[tile_ordinal]
831                                .checked_add(1)
832                                .ok_or(SpectralOperatorError::CoverageOverflow)?;
833                            if record.group_end {
834                                self.groups.push(GriddedNormalGroupSpan {
835                                    frame_ordinal: frame_ordinal_u32,
836                                    records: u32::try_from(group_start)
837                                        .map_err(|_| SpectralOperatorError::CoverageOverflow)?
838                                        ..u32::try_from(record_ordinal + 1)
839                                            .map_err(|_| SpectralOperatorError::CoverageOverflow)?,
840                                    prediction_lane: 0,
841                                    prediction_index: 0,
842                                });
843                                group_start = record_ordinal + 1;
844                            }
845                        }
846                        if group_start != record_count {
847                            return Err(SpectralOperatorError::InvalidGriddedRecord);
848                        }
849                    }
850                    GriddedNormalRecordLayout::Taylor(plan) => {
851                        for (record_ordinal, bytes) in
852                            encoded.chunks_exact(self.record_bytes).enumerate()
853                        {
854                            let record = decode_taylor_record(
855                                bytes,
856                                catalogs
857                                    .grid_shape(0)
858                                    .ok_or(SpectralOperatorError::InvalidGriddedRecord)?,
859                                plan.normal_moment_count(),
860                            )?;
861                            let group_ordinal = u32::try_from(self.groups.len())
862                                .map_err(|_| SpectralOperatorError::CoverageOverflow)?;
863                            let tile_ordinal = catalogs.tile_ordinal(0, record.taps)?;
864                            self.classifications.push(GriddedNormalClassification {
865                                tile_ordinal: u32::try_from(tile_ordinal)
866                                    .map_err(|_| SpectralOperatorError::CoverageOverflow)?,
867                                group_ordinal,
868                                frame_ordinal: frame_ordinal_u32,
869                                record_ordinal: u32::try_from(record_ordinal)
870                                    .map_err(|_| SpectralOperatorError::CoverageOverflow)?,
871                                tap_count: u32::try_from(GRIDDED_NORMAL_TAPS_PER_RECORD)
872                                    .map_err(|_| SpectralOperatorError::CoverageOverflow)?,
873                            });
874                            self.tile_counts[tile_ordinal] = self.tile_counts[tile_ordinal]
875                                .checked_add(1)
876                                .ok_or(SpectralOperatorError::CoverageOverflow)?;
877                            self.groups.push(GriddedNormalGroupSpan {
878                                frame_ordinal: frame_ordinal_u32,
879                                records: u32::try_from(record_ordinal)
880                                    .map_err(|_| SpectralOperatorError::CoverageOverflow)?
881                                    ..u32::try_from(record_ordinal + 1)
882                                        .map_err(|_| SpectralOperatorError::CoverageOverflow)?,
883                                prediction_lane: 0,
884                                prediction_index: 0,
885                            });
886                        }
887                    }
888                }
889                self.frame_sequences.push(sequence);
890                self.frame_record_counts.push(
891                    u32::try_from(record_count)
892                        .map_err(|_| SpectralOperatorError::CoverageOverflow)?,
893                );
894            }
895            if self.frame_sequences.is_empty() {
896                return Err(SpectralOperatorError::IncompleteCoverage);
897            }
898            self.prepare_prediction_lanes()?;
899            self.prepare_tile_routes()?;
900            self.active_frames = self.frame_sequences.len();
901            self.first_sequence = Some(first_sequence);
902            self.record_count = u64::try_from(self.routes.len())
903                .map_err(|_| SpectralOperatorError::CoverageOverflow)?;
904            self.prediction_groups = self
905                .prediction_groups
906                .checked_add(
907                    u64::try_from(self.groups.len())
908                        .map_err(|_| SpectralOperatorError::CoverageOverflow)?,
909                )
910                .ok_or(SpectralOperatorError::CoverageOverflow)?;
911            Ok(())
912        })();
913        if prepared.is_err() {
914            self.reset_active()?;
915        }
916        prepared
917    }

919    fn prepare_prediction_lanes(&mut self) -> Result<(), SpectralOperatorError> {
920        let total_records: usize = self
921            .groups
922            .iter()
923            .map(GriddedNormalGroupSpan::record_count)
924            .sum();
925        let mut starts = [0usize; GRIDDED_NORMAL_LANE_COUNT + 1];
926        starts[GRIDDED_NORMAL_LANE_COUNT] = self.groups.len();
927        let mut group = 0usize;
928        let mut records = 0usize;
929        for (lane, start) in starts
930            .iter_mut()
931            .enumerate()
932            .take(GRIDDED_NORMAL_LANE_COUNT)
933            .skip(1)
934        {
935            let target = total_records
936                .saturating_mul(lane)
937                .div_ceil(GRIDDED_NORMAL_LANE_COUNT);
938            while group < self.groups.len() && records < target {
939                records = records
940                    .checked_add(self.groups[group].record_count())
941                    .ok_or(SpectralOperatorError::CoverageOverflow)?;
942                group += 1;
943            }
944            *start = group;
945        }
946        for lane in 0..GRIDDED_NORMAL_LANE_COUNT {
947            let range = starts[lane]..starts[lane + 1];
948            let owner = self.predictions[lane]
949                .get_mut()
950                .map_err(|_| SpectralOperatorError::GriddedSectorPoisoned)?;
951            let value_count = range
952                .len()
953                .checked_mul(self.prediction_width)
954                .ok_or(SpectralOperatorError::ResidencyOverflow)?;
955            if value_count > owner.values.capacity() {
956                return Err(SpectralOperatorError::ResidencyOverflow);
957            }
958            owner.groups = range.clone();
959            owner.values.resize(value_count, Complex64::default());
960            for (local, group) in self.groups[range].iter_mut().enumerate() {
961                group.prediction_lane =
962                    u8::try_from(lane).map_err(|_| SpectralOperatorError::CoverageOverflow)?;
963                group.prediction_index =
964                    u32::try_from(local).map_err(|_| SpectralOperatorError::CoverageOverflow)?;
965            }
966        }
967        Ok(())
968    }

970    fn prepare_tile_routes(&mut self) -> Result<(), SpectralOperatorError> {
971        if self.classifications.len() > self.maximum_records {
972            return Err(SpectralOperatorError::ResidencyOverflow);
973        }
974        for tile in 0..self.tile_counts.len() {
975            self.tile_offsets[tile + 1] = self.tile_offsets[tile]
976                .checked_add(self.tile_counts[tile])
977                .ok_or(SpectralOperatorError::CoverageOverflow)?;
978            self.tile_cursors[tile] = self.tile_offsets[tile];
979        }
980        let record_count = self.classifications.len();
981        if usize::try_from(self.tile_offsets[self.tile_counts.len()])
982            .map_err(|_| SpectralOperatorError::CoverageOverflow)?
983            != record_count
984        {
985            return Err(SpectralOperatorError::IncompleteCoverage);
986        }
987        self.routes
988            .resize(record_count, GriddedNormalRoute::default());
989        for classification in &self.classifications {
990            let tile = usize::try_from(classification.tile_ordinal)
991                .map_err(|_| SpectralOperatorError::CoverageOverflow)?;
992            let route = usize::try_from(self.tile_cursors[tile])
993                .map_err(|_| SpectralOperatorError::CoverageOverflow)?;
994            self.routes[route] = GriddedNormalRoute {
995                group_ordinal: classification.group_ordinal,
996                frame_ordinal: classification.frame_ordinal,
997                record_ordinal: classification.record_ordinal,
998                tap_count: classification.tap_count,
999            };
1000            self.tile_cursors[tile] = self.tile_cursors[tile]
1001                .checked_add(1)
1002                .ok_or(SpectralOperatorError::CoverageOverflow)?;
1003        }
1004        self.classifications.clear();
1005
1006        let target = record_count.div_ceil(GRIDDED_NORMAL_LANE_COUNT).max(1);
1007        let mut duplicate = 0usize;
1008        for tile in 0..self.tile_counts.len() {
1009            let start = usize::try_from(self.tile_offsets[tile])
1010                .map_err(|_| SpectralOperatorError::CoverageOverflow)?;
1011            let end = usize::try_from(self.tile_offsets[tile + 1])
1012                .map_err(|_| SpectralOperatorError::CoverageOverflow)?;
1013            let count = end - start;
1014            if count == 0 {
1015                continue;
1016            }
1017            let shards = count.div_ceil(target).clamp(1, GRIDDED_NORMAL_LANE_COUNT);
1018            let chunk = count.div_ceil(shards);
1019            for (shard, route_start) in (start..end).step_by(chunk).enumerate() {
1020                let route_end = route_start.saturating_add(chunk).min(end);
1021                if shard != 0 {
1022                    duplicate += 1;
1023                }
1024                if self.tasks.len() == self.tasks.capacity() {
1025                    return Err(SpectralOperatorError::ResidencyOverflow);
1026                }
1027                self.tasks.push(GriddedNormalTileTask {
1028                    tile_ordinal: u32::try_from(tile)
1029                        .map_err(|_| SpectralOperatorError::CoverageOverflow)?,
1030                    shard: u8::try_from(shard)
1031                        .map_err(|_| SpectralOperatorError::CoverageOverflow)?,
1032                    lane: 0,
1033                    accumulator_ordinal: u32::try_from(self.tasks.len())
1034                        .map_err(|_| SpectralOperatorError::CoverageOverflow)?,
1035                    routes: u32::try_from(route_start)
1036                        .map_err(|_| SpectralOperatorError::CoverageOverflow)?
1037                        ..u32::try_from(route_end)
1038                            .map_err(|_| SpectralOperatorError::CoverageOverflow)?,
1039                });
1040            }
1041        }
1042        if duplicate > GRIDDED_NORMAL_HOT_TILE_DUPLICATES {
1043            return Err(SpectralOperatorError::ResidencyOverflow);
1044        }
1045        self.tasks.sort_unstable_by_key(|task| {
1046            (Reverse(task.record_count()), task.tile_ordinal, task.shard)
1047        });
1048        let mut lane_record_loads = [0_u64; GRIDDED_NORMAL_LANE_COUNT];
1049        let mut lane_tap_loads = [0_u64; GRIDDED_NORMAL_LANE_COUNT];
1050        for task in &mut self.tasks {
1051            let lane = lane_tap_loads
1052                .iter()
1053                .enumerate()
1054                .min_by_key(|(lane, load)| (**load, lane_record_loads[*lane], *lane))
1055                .map(|(lane, _)| lane)
1056                .expect("four logical lanes");
1057            let records = u64::try_from(task.record_count())
1058                .map_err(|_| SpectralOperatorError::CoverageOverflow)?;
1059            let route_start = usize::try_from(task.routes.start)
1060                .map_err(|_| SpectralOperatorError::CoverageOverflow)?;
1061            let route_end = usize::try_from(task.routes.end)
1062                .map_err(|_| SpectralOperatorError::CoverageOverflow)?;
1063            let taps = self.routes[route_start..route_end]
1064                .iter()
1065                .try_fold(0_u64, |total, route| {
1066                    total.checked_add(u64::from(route.tap_count))
1067                })
1068                .ok_or(SpectralOperatorError::CoverageOverflow)?;
1069            lane_record_loads[lane] = lane_record_loads[lane]
1070                .checked_add(records)
1071                .ok_or(SpectralOperatorError::CoverageOverflow)?;
1072            lane_tap_loads[lane] = lane_tap_loads[lane]
1073                .checked_add(taps)
1074                .ok_or(SpectralOperatorError::CoverageOverflow)?;
1075            task.lane = u8::try_from(lane).map_err(|_| SpectralOperatorError::CoverageOverflow)?;
1076        }
1077        self.tasks
1078            .sort_unstable_by_key(|task| (task.tile_ordinal, task.shard));
1079        for (ordinal, task) in self.tasks.iter_mut().enumerate() {
1080            task.accumulator_ordinal =
1081                u32::try_from(ordinal).map_err(|_| SpectralOperatorError::CoverageOverflow)?;
1082        }
1083        self.lane_record_counts = lane_record_loads;
1084        self.lane_tap_visit_counts = lane_tap_loads;
1085        Ok(())
1086    }

1088    fn reset_active(&mut self) -> Result<(), SpectralOperatorError> {
1089        for owner in &mut self.predictions {
1090            let owner = owner
1091                .get_mut()
1092                .map_err(|_| SpectralOperatorError::GriddedSectorPoisoned)?;
1093            owner.groups = 0..0;
1094            owner.values.clear();
1095        }
1096        self.frame_sequences.clear();
1097        self.frame_record_counts.clear();
1098        self.groups.clear();
1099        self.classifications.clear();
1100        self.routes.clear();
1101        self.tasks.clear();
1102        self.tile_counts.fill(0);
1103        self.tile_cursors.fill(0);
1104        self.tile_offsets.fill(0);
1105        self.lane_record_counts.fill(0);
1106        self.lane_tap_visit_counts.fill(0);
1107        self.active_frames = 0;
1108        self.first_sequence = None;
1109        self.record_count = 0;
1110        Ok(())
1111    }

1137    pub(super) fn route_capacity_bytes(&self) -> Result<u64, SpectralOperatorError> {
1138        let capacity_bytes = |capacity: usize, value_bytes: usize| {
1139            capacity
1140                .checked_mul(value_bytes)
1141                .ok_or(SpectralOperatorError::ResidencyOverflow)
1142        };
1143        let mut bytes = capacity_bytes(self.frame_capacities.len(), size_of::<usize>())?
1144            .checked_add(capacity_bytes(
1145                self.frame_sequences.capacity(),
1146                size_of::<u64>(),
1147            )?)
1148            .and_then(|bytes| {
1149                bytes.checked_add(
1150                    capacity_bytes(self.frame_record_counts.capacity(), size_of::<u32>()).ok()?,
1151                )
1152            })
1153            .and_then(|bytes| {
1154                bytes.checked_add(
1155                    capacity_bytes(self.groups.capacity(), size_of::<GriddedNormalGroupSpan>())
1156                        .ok()?,
1157                )
1158            })
1159            .and_then(|bytes| {
1160                bytes.checked_add(
1161                    capacity_bytes(
1162                        self.classifications.capacity(),
1163                        size_of::<GriddedNormalClassification>(),
1164                    )
1165                    .ok()?,
1166                )
1167            })
1168            .and_then(|bytes| {
1169                bytes.checked_add(
1170                    capacity_bytes(self.routes.capacity(), size_of::<GriddedNormalRoute>()).ok()?,
1171                )
1172            })
1173            .ok_or(SpectralOperatorError::ResidencyOverflow)?;
1174        for owner in &self.predictions {
1175            let owner = owner
1176                .read()
1177                .map_err(|_| SpectralOperatorError::GriddedSectorPoisoned)?;
1178            bytes = bytes
1179                .checked_add(capacity_bytes(
1180                    owner.values.capacity(),
1181                    size_of::<Complex64>(),
1182                )?)
1183                .and_then(|bytes| {
1184                    bytes.checked_add(
1185                        capacity_bytes(owner.model_scratch.capacity(), size_of::<Complex64>())
1186                            .ok()?,
1187                    )
1188                })
1189                .and_then(|bytes| {
1190                    bytes.checked_add(
1191                        capacity_bytes(owner.moment_scratch.capacity(), size_of::<f64>()).ok()?,
1192                    )
1193                })
1194                .ok_or(SpectralOperatorError::ResidencyOverflow)?;
1195        }
1196        u64::try_from(bytes).map_err(|_| SpectralOperatorError::ResidencyOverflow)
1197    }

1335    ) -> Result<GriddedNormalWork, SpectralOperatorError> {
1336        if local_ordinal >= GRIDDED_NORMAL_PARTITION_COUNT
1337            || local_ordinal < self.next_partition_commit
1338        {
1339            return Err(SpectralOperatorError::IncompleteCoverage);
1340        }

1342            .two_domain
1343            .read()
1344            .map_err(|_| SpectralOperatorError::GriddedSectorPoisoned)?;

1346            || prepared.active_frames != frame_count
1347            || first_sequence != self.next_block_sequence

1675    /// Commit one lane in canonical phase/lane order and merge the completed window.
1676    pub fn commit_two_domain(
1677        &mut self,
1678        partial: GriddedNormalPartial,
1679    ) -> Result<(), SpectralOperatorError> {
1680        let work = partial.work;
1681        if work.first_block_sequence != self.next_block_sequence
1682            || work.partition_key() as usize != self.next_partition_commit
1683        {
1684            return Err(SpectralOperatorError::BlockSequence);
1685        }
1686        self.next_partition_commit += 1;
1687        if self.next_partition_commit == GRIDDED_NORMAL_LANE_COUNT {
1688            if let Some(trace) = self.science_prediction_trace.as_mut() {
1689                let prepared = self
1690                    .two_domain
1691                    .read()
1692                    .map_err(|_| SpectralOperatorError::GriddedSectorPoisoned)?;
1693                prepared.with_published_predictions(|predictions| {
1694                    for prediction in predictions {
1695                        for value in &prediction.values {
1696                            trace.push_complex(*value);
1697                        }
1698                    }
1699                    Ok(())
1700                })?;
1701            }
1702        }
1703        if self.next_partition_commit != GRIDDED_NORMAL_PARTITION_COUNT {
1704            return Ok(());
1705        }
1706        {
1707            let prepared = self
1708                .two_domain
1709                .read()
1710                .map_err(|_| SpectralOperatorError::GriddedSectorPoisoned)?;
1711            for task in &prepared.tasks {
1712                let (domain_ordinal, geometry) = self.tile_catalogs.geometry(
1713                    usize::try_from(task.tile_ordinal)
1714                        .map_err(|_| SpectralOperatorError::CoverageOverflow)?,
1715                )?;
1716                let accumulator = self.tile_accumulators[usize::try_from(task.accumulator_ordinal)
1717                    .map_err(|_| SpectralOperatorError::CoverageOverflow)?]
1718                .lock()
1719                .map_err(|_| SpectralOperatorError::GriddedSectorPoisoned)?;
1720                for plane in 0..self.normal_grids[domain_ordinal].len() {
1721                    for local_x in 0..geometry.shape[0] {
1722                        for local_y in 0..geometry.shape[1] {
1723                            let value = accumulator.grids[plane][(local_x, local_y)];
1724                            if value == Complex64::default() {
1725                                continue;
1726                            }
1727                            let target =
1728                                (geometry.origin[0] + local_x, geometry.origin[1] + local_y);
1729                            let cell = &mut self.normal_grids[domain_ordinal][plane][target];
1730                            let compensation =
1731                                &mut self.normal_compensations[domain_ordinal][plane][target];
1732                            let contribution =
1733                                value - accumulator.compensations[plane][(local_x, local_y)];
1734                            let updated = *cell + contribution;
1735                            *compensation = (updated - *cell) - contribution;
1736                            *cell = updated;
1737                        }
1738                    }
1739                }
1740            }
1741        }
1742        let applied_records = self
1743            .applied_records
1744            .checked_add(work.window_record_count)
1745            .ok_or(SpectralOperatorError::CoverageOverflow)?;
1746        let next_block_sequence = self
1747            .next_block_sequence
1748            .checked_add(work.frame_count)
1749            .ok_or(SpectralOperatorError::CoverageOverflow)?;
