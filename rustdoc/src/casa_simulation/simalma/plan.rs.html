57pub(super) fn prepare(p: &Parameters, request: &SimalmaRequest) -> Result<Plan, SimobserveError> {
58    let directory =
59        std::path::absolute(request.workspace.join(p.text("project"))).map_err(failure)?;
60    let project = directory
61        .file_name()
62        .ok_or_else(|| failure("project must have a directory name"))?
63        .to_string_lossy()
64        .into_owned();
65    if directory.exists() && !p.flag("overwrite") {
66        return Err(failure("project directory already exists"));
67    }
68    for key in ["skymodel", "complist"] {
69        if let Ok(input) = std::fs::canonicalize(p.text(key)) {
70            let output = std::fs::canonicalize(&directory).unwrap_or(directory.clone());
71            if input.starts_with(output) {
72                return Err(failure(format!(
73                    "{key} would be deleted with the existing project"
74                )));
75            }
76        }
77    }
78    let mut model = model(p, request)?;
79    let primary_beam =
80        1.13 * 0.2997924 / (model.frequency / 1e9) / 12. * 3600. * 180. / std::f64::consts::PI;
81    let super::configuration::Selection {
82        arrays,
83        tp_antennas,
84        tp_minutes,
85        tp_configuration,
86    } = super::configuration::select(p, request, &directory, &project, &model, primary_beam)?;
87    protect_pointings(p, &request.workspace, &directory, &arrays)?;
88    if model.cell[0] == 0. {
89        let cell = arrays
90            .iter()
91            .map(|a| a.resolution)
92            .fold(f64::INFINITY, f64::min)
93            / 20.;
94        model.cell = [cell; 2];
95    }
96    let mapsize = p.strings("mapsize");
97    let mapsize = if mapsize.first().is_none_or(String::is_empty) {
98        model.size.map(|v| format!("{v}arcsec")).to_vec()
99    } else {
100        mapsize
101    };
102    let directions = p.strings("direction");
103    let direction = match directions.len() {
104        0 => Directions::One(String::new()),
105        1 => Directions::One(directions[0].clone()),
106        _ => Directions::List(directions),
107    };
108    let noisy = p.0["pwv"].as_f64().unwrap() > 0.;
109    let observation = SimobserveParameters {
110        project: directory.to_string_lossy().into_owned(),
111        skymodel: p.text("skymodel").to_owned(),
112        inbright: p.text("inbright").to_owned(),
113        indirection: p.text("indirection").to_owned(),
114        incell: p.text("incell").to_owned(),
115        incenter: p.text("incenter").to_owned(),
116        inwidth: p.text("inwidth").to_owned(),
117        complist: p.text("complist").to_owned(),
118        compwidth: p.text("compwidth").to_owned(),
119        setpointings: p.flag("setpointings"),
120        ptgfile: p.text("ptgfile").to_owned(),
121        integration: p.text("integration").to_owned(),
122        direction,
123        mapsize,
124        maptype: "ALMA".into(),
125        pointingspacing: format!("{}PB", 1. / 3f64.sqrt() / 1.13),
126        hourangle: p.text("hourangle").to_owned(),
127        thermalnoise: if noisy {
128            "tsys-atm".into()
129        } else {
130            String::new()
131        },
132        user_pwv: p.0["pwv"].as_f64().unwrap(),
133        t_ground: 270.,
134        calflux: "0Jy".into(),
135        graphics: "none".into(),
136        verbose: p.flag("verbose"),
137        overwrite: true,
138        ..Default::default()
139    };
140    let tp_stem = antennas::stem(&project, &tp_configuration)?;
141    Ok(Plan {
142        directory,
143        project,
144        arrays,
145        model,
146        observation,
147        tp_antennas,
148        tp_minutes,
149        tp_configuration,
150        tp_stem,
151        tp_vis: Vec::new(),
152        noisy,
153        primary_beam,
154    })
155}

157fn model(p: &Parameters, request: &SimalmaRequest) -> Result<Model, SimobserveError> {
158    let path = Path::new(p.text("skymodel"));
159    if path.exists() {
160        let shape = if path.is_file() {
161            casa_images::fits_image_shape(path).map_err(failure)?
162        } else {
163            AnyPagedImage::open(path).map_err(failure)?.shape().to_vec()
164        };
165        let memory = shape
166            .iter()
167            .try_fold(64u64, |bytes, &n| bytes.checked_mul(n as u64))
168            .and_then(|bytes| bytes.checked_add(32 << 20))
169            .ok_or_else(|| failure("sky model dimensions overflow"))?;
170        let r = request.options.resources;
171        let _memory = admit(
172            r.host(),
173            r.policy(),
174            &Demand {
175                phase: "ALMA sky-model inspection",
176                memory,
177            },
178        )
179        .map_err(failure)?;
180        let staging = tempfile::tempdir().map_err(failure)?;
181        let one = |s: &str| {
182            if s.is_empty() {
183                Vec::new()
184            } else {
185                vec![s.to_owned()]
186            }
187        };
188        let summary = modify_model(
189            path,
190            &staging.path().join("model"),
191            &ModelOverrides {
192                inbright: p.text("inbright").to_owned(),
193                indirection: one(p.text("indirection")),
194                incell: one(p.text("incell")),
195                incenter: p.text("incenter").to_owned(),
196                inwidth: p.text("inwidth").to_owned(),
197            },
198        )?;
199        return Ok(Model {
200            direction: summary.refdir,
201            cell: [
202                qa::value_in(&summary.cell[0], "arcsec")?,
203                qa::value_in(&summary.cell[1], "arcsec")?,
204            ],
205            size: [
206                qa::value_in(&summary.size[0], "arcsec")?,
207                qa::value_in(&summary.size[1], "arcsec")?,
208            ],
209            frequency: qa::value_in(&summary.specrefval.quantity()?, "Hz")?,
210            channels: summary.nchan,
211            preparation_bytes: memory,
212        });
213    }
214    let list = ComponentList::read_table(p.text("complist")).map_err(failure)?;
215    let component =
216        crate::simutil::skymodel::ComponentModel::from_list(&list, p.text("compwidth"), 1)?;
217    // simalma's extent is 2*max offset, versus simobserve's 3*max offset.
218    let size = qa::value_in(&qa::read(&component.size[0])?, "arcsec")? * 2. / 3.;
219    Ok(Model {
220        direction: component.refdir,
221        cell: [0.; 2],
222        size: [size; 2],
223        frequency: qa::value_in(&component.specrefval, "Hz")?,
224        channels: 1,
225        preparation_bytes: 32 << 20,
226    })
227}

229pub(super) fn publish_directory(p: &Parameters, plan: &Plan) -> Result<(), SimobserveError> {
230    if std::fs::symlink_metadata(&plan.directory).is_ok() {
231        if !p.flag("overwrite") {
232            return Err(failure("project directory already exists"));
233        }
234        if plan.directory.is_symlink() {
235            std::fs::remove_file(&plan.directory).map_err(failure)?;
236        } else {
237            std::fs::remove_dir_all(&plan.directory).map_err(failure)?;
238        }
239    }
240    std::fs::create_dir_all(&plan.directory).map_err(failure)
241}

243// Match simobserve's stem substitution and outside/project search order
244// before overwriting the directory that may hold the selected input.
245fn protect_pointings(
246    p: &Parameters,
247    workspace: &Path,
248    directory: &Path,
249    arrays: &[Array],
250) -> Result<(), SimobserveError> {
251    if p.flag("setpointings") {
252        return Ok(());
253    }
254    let project = std::fs::canonicalize(directory).unwrap_or(directory.to_path_buf());
255    for array in arrays {
256        let name = p.text("ptgfile").replace("$project", &array.stem);
257        let in_workspace = workspace.join(&name);
258        let path = if in_workspace.exists() {
259            in_workspace
260        } else {
261            directory.join(name)
262        };
263        let input = std::fs::canonicalize(&path).map_err(failure)?;
264        if path.starts_with(directory) || input.starts_with(&project) {
265            return Err(failure(format!(
266                "pointing input {} would be deleted with the existing project",
267                path.display()
268            )));
269        }
270    }
271    Ok(())
