2//! The initial major cycle, from an empty model or a start model (#841
3//! Phase 6b, #970 review).
4//!
5//! A start model is the model's first generation, in the units a run
6//! publishes its `.model` in. Under a direction-independent response, or
7//! flat sky, those are the physical units the pass predicts from, and the
8//! initial pass grids the residual of the start model. Under flat noise or
9//! pbsquare (a mosaic's or AW kernel set's `normtype`), `.model` is the
10//! physical model times the PB (flat noise) or PB² (pbsquare), and CASA
11//! divides a model through the PB or PB² of the weights just before it
12//! predicts (`SIImageStore::divideModelByWeight`), once `makePSF` has
13//! formed them. Such a run takes CASA's order: the initial pass forms the
14//! PSF, weights and sums with no model, and the start model, made physical
15//! with those weights, is the first generation of a refresh of that pass's
16//! residual. That costs one more pass over the data, for such runs only.
17//! Either way, a run started from another run's model with no minor cycle
18//! gives that run's products (the T1 laws; CASA 6.7.6 does the same on a
19//! 14-pointing mosaic).
20

31/// The response a start model is made physical with before the first
32/// major cycle predicts it: a flat-noise or pbsquare one, through the PB
33/// (flat noise) or PB² (pbsquare); `None` without a start model, or when
34/// its units are physical already.
35pub(super) fn reseed_response(
36    start_model: Option<&StartModel>,
37    response: Option<MinorCycleImageResponse>,
38) -> Option<MinorCycleImageResponse> {
39    start_model.and(response).filter(|response| {
40        matches!(
41            response.normalization(),
42            ProductNormalization::FlatNoise | ProductNormalization::PbSquare
43        )
44    })
45}

47/// Each main-image plane's flux in a start model's `samples`, in `.model`'s
48/// units, as `SIImageStore::getModelFlux` sums it; a Taylor family's is its
49/// first term's, under its first channel. Sorted by `[domain, channel,
50/// polarization]`.
51pub(super) fn plane_fluxes(
52    problem: &CompiledProblem,
53    start: &StartModel,
54    samples: &[f32],
55) -> Vec<([usize; 3], f64)> {
56    let polarizations = start.polarizations();
57    let [width, height] = start.image_size();
58    samples
59        .chunks_exact(width * height)
60        .enumerate()
61        .filter_map(|(index, plane)| {
62            plane_key(problem, polarizations, index)
63                .map(|key| (key, plane.iter().map(|value| f64::from(*value)).sum()))
64        })
65        .collect()
66}
67
68/// The `[domain, channel, polarization]` of a start model's plane `index`;
69/// `None` for a Taylor term after the first.
70fn plane_key(problem: &CompiledProblem, polarizations: usize, index: usize) -> Option<[usize; 3]> {
71    let cube = matches!(
72        problem.reconstruction().basis(),
73        ReconstructionBasis::ChannelLocal { .. }
74    );
75    let (coefficient, polarization) = (index / polarizations, index % polarizations);
76    (cube || coefficient == 0).then_some([0, if cube { coefficient } else { 0 }, polarization])
77}

79impl Run<'_> {
80    /// The initial major cycle: data and PSF, from the start model when there
81    /// is one (its residual) and from an empty model otherwise; it writes
82    /// visibilities when it is also final. A run that reads its PSF or
83    /// residual forms what it does not read ([`super::continuation`]).
84    pub(super) fn initial(&mut self, last: bool, mvc_start: bool) -> Result<Major, ImagingError> {
85        if mvc_start {
86            self.measure_row_work()?;
87            return self.psf_pass();
88        }
89        match (self.forms.psf, self.forms.residual) {
90            (true, true) => {}
91            (true, false) => {
92                self.measure_row_work()?;
93                return self.psf_pass();
94            }
95            (false, true) => {
96                self.measure_row_work()?;
97                let read = self.read_existing(self.reseed.is_none())?;
98                if let Some(response) = self.reseed {
99                    return self.reseed(read, response, last);
100                }
101                return self.refresh(
102                    read.completion,
103                    (Vec::new(), Vec::new(), Reservation::none()),
104                    last,
105                );
106            }
107            (false, false) => return self.read_existing(true),
108        }
109        self.measure_row_work()?;
110        match self.reseed {
111            Some(response) => {
112                let unseeded = self.initial_pass(false, false)?;
113                self.reseed(unseeded, response, last)
114            }
115            None => self.initial_pass(self.start_model.is_some(), last),
116        }
117    }

119    /// The initial pass, from the start model when `seeded`.
120    fn initial_pass(&mut self, seeded: bool, last: bool) -> Result<Major, ImagingError> {
121        // The model, and a start model's seeding beside it, are admitted
122        // with the pass before either is allocated, so the pass plans its
123        // waves and chunks in what seeding leaves (#970 review).
124        let seeding = if seeded { self.seeding_bytes() } else { 0 };
125        let Admitted {
126            value: plan,
127            reservation: mut pass,
128        } = self.admit_pass(
129            self.initial_modes(),
130            seeded,
131            self.lifecycle.resident_bytes() + self.write_bytes(last) + seeding,
132            self.writes(last),
133        )?;
134        let generation = match &self.start_model {
135            Some(start) if seeded => {
136                // Released once the generation holds the samples.
137                let _seeding = pass.split(seeding);
138                let samples = start.main_samples().map_err(ImagingError::StartModel)?;
139                self.start_model_flux = plane_fluxes(self.problem, start, &samples);
140                self.lifecycle.initial_from_start_model(&samples)?
141            }
142            _ => self.lifecycle.initial_empty()?,
143        };
144        let model = self.lifecycle.prepare_final_model(generation, [])?;
145        let cycle = MajorCycle::initial(self.problem, model, self.normal_storage(plan.residency)?)?;
146        let major = self.reconcile(cycle, self.initial_modes(), seeded, last, plan)?;
147        self.retain(&major, pass)?;
148        Ok(major)
149    }

151    /// The residual of the start model, made physical with `unseeded`'s
152    /// weights under `response`: a refresh of the initial pass, whose PSF,
153    /// weights and sums carry over.
154    fn reseed(
155        &mut self,
156        unseeded: Major,
157        response: MinorCycleImageResponse,
158        last: bool,
159    ) -> Result<Major, ImagingError> {
160        let start = self
161            .start_model
162            .clone()
163            .expect("a run that reseeds has a start model");
164        // The retained charge covers the normal state, and the empty model,
165        // which the seeded generation replaces at the same size.
166        let (normal_state, empty) = unseeded.completion.into_parts();
167        drop(empty);
168        let seeding = self.seeding_bytes();
169        let Admitted {
170            value: plan,
171            reservation: mut pass,
172        } = self.admit_pass(
173            ModeSet::DATA,
174            true,
175            self.write_bytes(last) + seeding,
176            self.writes(last),
177        )?;
178        let generation = {
179            let _seeding = pass.split(seeding);
180            self.physical_generation(&start, &normal_state, response)?
181        };
182        let model = self.lifecycle.prepare_final_model(generation, [])?;
183        let cycle = MajorCycle::refresh(
184            self.problem,
185            normal_state,
186            model,
187            self.normal_storage(plan.residency)?,
188        )?;
189        let major = self.reconcile(cycle, ModeSet::DATA, true, last, plan)?;
190        self.retain(&major, pass)?;
191        Ok(major)
192    }

194    /// The first generation from `start`, its apparent samples divided
195    /// through the PB (flat noise) or PB² (pbsquare) of `normal`'s weights
196    /// where that supports them and zero elsewhere (`divideModelByWeight`),
197    /// as the minor cycle maps solved values to the physical model.
198    pub(super) fn physical_generation(
199        &mut self,
200        start: &StartModel,
201        normal: &FinalNormalState,
202        response: MinorCycleImageResponse,
203    ) -> Result<ModelGeneration, ImagingError> {
204        let [nx, ny] = start.image_size();
205        let polarizations = start.polarizations();
206        let apparent = start.main_samples().map_err(ImagingError::StartModel)?;
207        let units = ImagingError::StartModelUnits;
208        let mut physical = Vec::with_capacity(apparent.len());
209        let mut fluxes = Vec::new();
210        // Canonical order: spectral plane, polarization, y, x (x fastest);
211        // the normal state's planes are x-major (x · ny + y). Every Taylor
212        // term divides by term 0's weights of its polarization
213        // (`SIImageStoreMultiTerm::divideModelByWeight`).
214        let taylor = (normal.basis() == casa_imaging_deconvolution::NormalBasis::Taylor)
215            .then(|| normal.read_window(normal.slab().core_range()))
216            .transpose()?;
217        for (plane, samples) in apparent.chunks(nx * ny).enumerate() {
218            let principal = taylor.as_ref().map(|window| {
219                window
220                    .normal_moment(0, plane % polarizations)
221                    .expect("the zeroth normal moment")
222            });
223            let (sensitivity, sum_weight) = match principal {
224                Some(moment) => (
225                    std::borrow::Cow::Borrowed(moment.sensitivity()),
226                    moment.sum_weight(),
227                ),
228                None => {
229                    let weights =
230                        normal.read_plane(0, plane / polarizations, plane % polarizations)?;
231                    (weights.read_sensitivity()?, weights.sum_weight())
232                }
233            };
234            // `SIImageStore::divideModelByWeight` skips a plane whose PB's
235            // peak is zero: no sample reached it, or its rows all miss the
236            // grid (an AW weight image with no sum, which
237            // `divideImageByWeightVal` zeroes). Its model stays as it is.
238            let key = plane_key(self.problem, polarizations, plane);
239            let bound = MosaicSensitivity::try_new(&sensitivity).map_err(units)?;
240            let Some(bound) = bound.filter(|_| sum_weight > 0.0) else {
241                physical.extend_from_slice(samples);
242                fluxes.extend(
243                    key.map(|key| (key, samples.iter().map(|value| f64::from(*value)).sum())),
244                );
245                continue;
246            };
247            let bound = bound.with_normal_sum_weight(sum_weight).map_err(units)?;
248            let mut flux = 0.0;
249            for y in 0..ny {
250                for x in 0..nx {
251                    let value = bound
252                        .apparent_to_physical(
253                            f64::from(samples[y * nx + x]),
254                            x * ny + y,
255                            response.normalization(),
256                            response.policy(),
257                        )
258                        .map_err(units)?;
259                    physical.push(value as f32);
260                    // `.model` keeps what `divideModelByWeight` cuts: its
261                    // multiplication back can't restore it.
262                    if value != 0.0 {
263                        flux += f64::from(samples[y * nx + x]);
264                    }
265                }
266            }
267            fluxes.extend(key.map(|key| (key, flux)));
268        }
269        self.start_model_flux = fluxes;
270        Ok(self.lifecycle.initial_from_start_model(&physical)?)
271    }

273    /// What seeding the model from the start model holds besides the model
274    /// ([`StartModel::seeding_bytes`], and its conversion when it is made
275    /// physical); 0 without one.
276    pub(super) fn seeding_bytes(&self) -> u64 {
277        self.start_model.as_ref().map_or(0, |start| {
278            start.seeding_bytes()
279                + if self.reseed.is_some() {
280                    start.conversion_bytes()
281                } else {
282                    0
283                }
284        })
285    }
