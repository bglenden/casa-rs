2//! The names CASA gives shapes, spectral models and polarisation
3//! representations, and its rules for parsing them (`ComponentType`).
4//!
5//! A component-list table stores each name as a string. CASA writes the
6//! canonical names below and reads a name back by capitalising every word
7//! (casacore's `String::capitalize`) and then:
8//!
9//! - for a **shape**, taking the first canonical name that *contains* the
10//!   capitalised string, in the order point, Gaussian, disk, limb-darkened
11//!   disk (a minimum match: `"gaus"` is a Gaussian, `"disk"` a disk);
12//! - for a **spectral model** or a **polarisation**, requiring the
13//!   capitalised string to *equal* a canonical name.
14
15use crate::error::ComponentError;
16
17/// The kind of a component's shape, CASA's `ComponentType::Shape`.
18#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
19pub enum ShapeType {
20    /// A point source (`POINT`).
21    Point,
22    /// An elliptical Gaussian (`GAUSSIAN`).
23    Gaussian,
24    /// A uniform elliptical disk (`DISK`).
25    Disk,
26    /// A limb-darkened elliptical disk (`LDISK`).
27    LimbDarkenedDisk,
28}
29
30impl ShapeType {
31    /// Every shape, in CASA's enumeration order (the order names are matched).
32    pub const ALL: [Self; 4] = [
33        Self::Point,
34        Self::Gaussian,
35        Self::Disk,
36        Self::LimbDarkenedDisk,
37    ];
38
39    /// The name CASA writes in a component list's `Shape` column
40    /// (`ComponentType::name`).
41    pub fn casa_name(self) -> &'static str {
42        match self {
43            Self::Point => "Point",
44            Self::Gaussian => "Gaussian",
45            Self::Disk => "Disk",
46            Self::LimbDarkenedDisk => "Limbdarkeneddisk",
47        }
48    }
49
50    /// Parse a shape name as `ComponentType::shape` does: capitalise each
51    /// word, then take the first canonical name containing the result.
52    ///
53    /// # Errors
54    ///
55    /// [`ComponentError::UnknownShape`] when no canonical name contains it.
56    pub fn parse_casa(name: &str) -> Result<Self, ComponentError> {
57        let canonical = capitalize_words(name);
58        Self::ALL
59            .into_iter()
60            .find(|shape| shape.casa_name().contains(canonical.as_str()))
61            .ok_or_else(|| ComponentError::UnknownShape(name.to_string()))
62    }
63}

65/// The kind of a component's spectral model, CASA's
66/// `ComponentType::SpectralShape`.
67#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
68pub enum SpectralType {
69    /// A flat spectrum (`CONSTANT_SPECTRUM`).
70    Constant,
71    /// A power law in frequency (`SPECTRAL_INDEX`).
72    SpectralIndex,
73    /// Flux tabulated against frequency (`TABULAR_SPECTRUM`).
74    Tabular,
75    /// A power law whose exponent is a polynomial in the logarithm of the
76    /// frequency ratio (`PLP`).
77    PowerLogPolynomial,
78}
79
80impl SpectralType {
81    /// Every spectral model, in CASA's enumeration order.
82    pub const ALL: [Self; 4] = [
83        Self::Constant,
84        Self::SpectralIndex,
85        Self::Tabular,
86        Self::PowerLogPolynomial,
87    ];
88
89    /// The name CASA writes in a component list's `Spectrum_Shape` column and
90    /// in a spectral record's `type` field (`ComponentType::name`).
91    pub fn casa_name(self) -> &'static str {
92        match self {
93            Self::Constant => "Constant",
94            Self::SpectralIndex => "Spectral Index",
95            Self::Tabular => "Tabular Spectrum",
96            Self::PowerLogPolynomial => "Power Logarithmic Polynomial",
97        }
98    }
99
100    /// Parse a spectral-model name as `ComponentType::spectralShape` does:
101    /// capitalise each word, then require an exact match.
102    ///
103    /// # Errors
104    ///
105    /// [`ComponentError::UnknownSpectrum`] when no canonical name matches.
106    pub fn parse_casa(name: &str) -> Result<Self, ComponentError> {
107        let canonical = capitalize_words(name);
108        Self::ALL
109            .into_iter()
110            .find(|spectrum| spectrum.casa_name() == canonical)
111            .ok_or_else(|| ComponentError::UnknownSpectrum(name.to_string()))
112    }
113}

115/// How a flux's four values represent the polarisation, CASA's
116/// `ComponentType::Polarisation`.
117#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
118pub enum Polarisation {
119    /// I, Q, U and V (`STOKES`). The values are real.
120    Stokes,
121    /// XX, XY, YX and YY (`LINEAR`), complex.
122    Linear,
123    /// RR, RL, LR and LL (`CIRCULAR`), complex.
124    Circular,
125}
126
127impl Polarisation {
128    /// Every representation, in CASA's enumeration order.
129    pub const ALL: [Self; 3] = [Self::Stokes, Self::Linear, Self::Circular];
130
131    /// The name CASA writes in a component list's `Flux_Polarisation` column
132    /// (`ComponentType::name`).
133    pub fn casa_name(self) -> &'static str {
134        match self {
135            Self::Stokes => "Stokes",
136            Self::Linear => "Linear",
137            Self::Circular => "Circular",
138        }
139    }
140
141    /// Parse a representation name as `ComponentType::polarisation` does:
142    /// capitalise each word, then require an exact match.
143    ///
144    /// # Errors
145    ///
146    /// [`ComponentError::UnknownPolarisation`] when no canonical name
147    /// matches.
148    pub fn parse_casa(name: &str) -> Result<Self, ComponentError> {
149        let canonical = capitalize_words(name);
150        Self::ALL
151            .into_iter()
152            .find(|polarisation| polarisation.casa_name() == canonical)
153            .ok_or_else(|| ComponentError::UnknownPolarisation(name.to_string()))
154    }
155}

157/// casacore's `String::capitalize`: the first letter or digit of every word
158/// (a run of ASCII letters and digits) upper case, the rest lower case.
159fn capitalize_words(text: &str) -> String {
160    let mut out = String::with_capacity(text.len());
161    let mut in_word = false;
162    for character in text.chars() {
163        if character.is_ascii_alphanumeric() {
164            if in_word {
165                out.push(character.to_ascii_lowercase());
166            } else {
167                in_word = true;
168                out.push(character.to_ascii_uppercase());
169            }
170        } else {
171            in_word = false;
172            out.push(character);
173        }
174    }
175    out
176}

178#[cfg(test)]
179mod tests {
180    use super::*;
181
182    #[test]
183    fn capitalisation_follows_casacore_words() {
184        assert_eq!(capitalize_words("spectral index"), "Spectral Index");
185        assert_eq!(capitalize_words("LIMB-darkened DISK"), "Limb-Darkened Disk");
186        assert_eq!(capitalize_words("tabular  spectrum"), "Tabular  Spectrum");
187        assert_eq!(capitalize_words("2nd"), "2nd");
188    }
189
190    #[test]
191    fn shapes_match_by_containment_in_enumeration_order() {
192        assert_eq!(ShapeType::parse_casa("gaus"), Ok(ShapeType::Gaussian));
193        assert_eq!(ShapeType::parse_casa("POINT"), Ok(ShapeType::Point));
194        // "Disk" is contained in "Disk" before "Limbdarkeneddisk" is tried.
195        assert_eq!(ShapeType::parse_casa("disk"), Ok(ShapeType::Disk));
196        assert_eq!(
197            ShapeType::parse_casa("limbdarkeneddisk"),
198            Ok(ShapeType::LimbDarkenedDisk)
199        );
200        // casacore's capitalisation makes "Limb-Darkened Disk", which no
201        // canonical name contains.
202        assert!(ShapeType::parse_casa("limb-darkened disk").is_err());
203        assert!(ShapeType::parse_casa("ring").is_err());
204        for shape in ShapeType::ALL {
205            assert_eq!(ShapeType::parse_casa(shape.casa_name()), Ok(shape));
206        }
207    }
208
209    #[test]
210    fn spectra_and_polarisations_need_exact_matches() {
211        assert_eq!(
212            SpectralType::parse_casa("spectral index"),
213            Ok(SpectralType::SpectralIndex)
214        );
215        assert!(SpectralType::parse_casa("spectral").is_err());
216        assert_eq!(
217            Polarisation::parse_casa("CIRCULAR"),
218            Ok(Polarisation::Circular)
219        );
220        assert!(Polarisation::parse_casa("lin").is_err());
221        for spectrum in SpectralType::ALL {
222            assert_eq!(SpectralType::parse_casa(spectrum.casa_name()), Ok(spectrum));
223        }
224        for polarisation in Polarisation::ALL {
225            assert_eq!(
226                Polarisation::parse_casa(polarisation.casa_name()),
227                Ok(polarisation)
228            );
229        }
230    }
