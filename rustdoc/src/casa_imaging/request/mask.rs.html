16impl ImagingRequest {
17    /// Whether an image is combined with a CRTF region of limited spectral range.
18    pub(crate) fn has_ranged_image_mask(&self) -> bool {
19        self.mask_images.len() == 1 && self.mask_box.iter().any(|region| region.range.is_some())
20    }
21
22    /// Keep image mask combinations within their compared geometries.
23    pub(crate) fn validate_mask_image_list(&self) -> Result<(), InvalidRequest> {
24        let standard_hogbom = self.stokes.as_slice() == [PolarizationCoordinate::StokesI]
25            && matches!(self.gridder, Gridder::Standard { facets } if facets.get() == 1)
26            && self.deconvolver == Deconvolver::Hogbom
27            && self.niter > 0
28            && self.outlierfile.is_none();
29        if self.mask_images.len() > 1
30            && (!self.mask_box.is_empty()
31                || self.mask_polygon.is_some()
32                || self.specmode != SpecMode::Mfs
33                || !standard_hogbom)
34        {
35            return Err(InvalidRequest {
36                parameters: "mask_image",
37                reason: "multiple mask images support standard Stokes I MFS Högbom cleaning without region masks or outliers (#823)",
38            });
39        }
40        if self.has_ranged_image_mask() && (!standard_hogbom || self.specmode != SpecMode::Cube) {
41            return Err(InvalidRequest {
42                parameters: "mask_image",
43                reason: "a mask image with a spectral-range region supports standard Stokes I Högbom cubes (#823)",
44            });
45        }
46        Ok(())
47    }
48}

50/// The native single mask path, or the list of paths resolved by tclean.
51pub(super) fn mask_images<'de, D: Deserializer<'de>>(
52    deserializer: D,
53) -> Result<Vec<PathBuf>, D::Error> {
54    match Value::deserialize(deserializer)? {
55        value @ Value::Array(_) => Vec::<PathBuf>::deserialize(value).map_err(D::Error::custom),
56        value => optional::<_, PathBuf>(value)
57            .map(|path| path.into_iter().collect())
58            .map_err(D::Error::custom),
59    }
60}

62/// Boxes `x0,y0,x1,y1` separated by `;`, each limited to a spectral range
63/// by an optional `@<low>~<high>`, both ends in `Hz` or both in `pix`.
64pub(super) fn mask_boxes<'de, D: Deserializer<'de>>(
65    deserializer: D,
66) -> Result<Vec<UserMaskBox>, D::Error> {
67    let value = Value::deserialize(deserializer)?;
68    let text = value
69        .as_str()
70        .ok_or_else(|| unexpected(&value, "pixel boxes"))?;
71    if unset(text) {
72        return Ok(Vec::new());
73    }
74    text.split(';')
75        .map(|item| {
76            let (corners, range) = match item.split_once('@') {
77                Some((corners, range)) => (corners, Some(spectral_mask_range(range)?)),
78                None => (item, None),
79            };
80            let corners = comma_list::<i64>(corners)?
81                .try_into()
82                .map_err(|_| format!("a mask box has four corners, not {corners:?}"))?;
83            Ok(UserMaskBox { corners, range })
84        })
85        .collect::<Result<_, String>>()
86        .map_err(D::Error::custom)
87}

89/// A mask box's `<low>~<high>` range: two finite values, both in `Hz` or
90/// both in `pix`.
91fn spectral_mask_range(text: &str) -> Result<SpectralMaskRange, String> {
92    let (low, high) = text
93        .split_once('~')
94        .ok_or_else(|| format!("a mask box range is <low>~<high>, not {text:?}"))?;
95    let end = |end: &str, unit: &str| {
96        end.trim()
97            .strip_suffix(unit)
98            .and_then(|number| number.trim().parse::<f64>().ok())
99            .filter(|number| number.is_finite())
100    };
101    match (
102        end(low, "Hz"),
103        end(high, "Hz"),
104        end(low, "pix"),
105        end(high, "pix"),
106    ) {
107        (Some(low), Some(high), _, _) => Ok(SpectralMaskRange::Frequency([low, high])),
108        (_, _, Some(low), Some(high)) => Ok(SpectralMaskRange::Pixel([low, high])),
109        _ => Err(format!(
110            "a mask box range has two finite ends, both in Hz or both in pix, not {text:?}"
111        )),
112    }
113}
