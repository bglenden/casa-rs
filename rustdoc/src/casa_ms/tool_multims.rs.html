23/// Assemble a portable CASA MultiMS from the input tables in order.
24///
25/// Members move into `SUBMSS` on success. Referencing a physical member
26/// twice retains both logical occurrences and moves it once. Copying
27/// subtables replaces the selected links with copies from the first member.
28///
29/// # Errors
30/// A missing or incompatible member, a conflicting output, or a failed
31/// publication. On failure member moves are rolled back. If a concurrent
32/// change prevents restoration, surviving originals are retained for recovery.
33pub fn create_multi_ms(
34    output: &Path,
35    inputs: &[PathBuf],
36    joined_subtables: &[String],
37    copy_subtables: Option<&[String]>,
38) -> MsResult<()> {
39    create_multi_ms_then(output, inputs, joined_subtables, copy_subtables, || Ok(()))
40}
41
42/// Create a MultiMS and retain rollback state until the caller accepts it.
43///
44/// # Errors
45/// A MultiMS creation error or an error returned by the acceptance callback.
46pub fn create_multi_ms_then<T>(
47    output: &Path,
48    inputs: &[PathBuf],
49    joined_subtables: &[String],
50    copy_subtables: Option<&[String]>,
51    accept: impl FnOnce() -> MsResult<T>,
52) -> MsResult<T> {
53    let output = output_path(output)?;
54    let destination = Destination::open(&output)?;
55    let workspace = tempfile::Builder::new()
56        .prefix(".casa-rs-multims-")
57        .tempdir_in(output.parent().expect("canonical output parent"))
58        .map_err(invalid)?;
59    let product = workspace.path().join("mms");
60    let (mut members, order) = plan_members(inputs, &output, &product)?;
61    // Take the locks in path order. The row order comes from the caller.
62    let ordered = members
63        .iter()
64        .map(|member| &member.physical)
65        .collect::<BTreeSet<_>>();
66    let mut locks = Vec::new();
67    for member in ordered {
68        check_write_access(&member.join("table.dat")).map_err(invalid)?;
69        locks.extend(lock_table_tree_for_replacement(member)?);
70    }
71    if matches!(&destination, Destination::Table(_)) {
72        locks.extend(lock_table_tree_for_replacement(&output)?);
73    }
74    let locations = members
75        .iter()
76        .filter(|member| member.source == member.physical)
77        .map(|member| (member.source.clone(), member.target.clone()))
78        .collect();
79    let first = Table::open(TableOptions::new(&members[order[0]].physical))?;
80    let original = order
81        .iter()
82        .map(|&index| Table::open(TableOptions::new(&members[index].physical)))
83        .collect::<Result<Vec<_>, _>>()?;
84    if matches!(&destination, Destination::Table(_)) {
85        for table in &original {
86            check_named_table_output(table, &output)?;
87        }
88    }
89    let _ = Table::concat(original)?;
90    let mut published = None;
91    let result = assemble(Assembly {
92        members: &mut members,
93        order: &order,
94        first: &first,
95        product: &product,
96        joined_subtables,
97        copy_subtables,
98        locks: &locks,
99        locations: &locations,
100        output: &output,
101    })
102    .and_then(|()| {
103        let new = product.symlink_metadata().map_err(invalid)?;
104        destination.publish(&product, &output)?;
105        published = Some(new);
106        destination.verify_publication(&product, &output, published.as_ref().unwrap())
107    });
108    let mut rollback_locks = Some(locks);
109    let result = result.and_then(|()| {
110        drop(rollback_locks.take());
111        let accepted = accept()?;
112        destination.verify_publication(&product, &output, published.as_ref().unwrap())?;
113        Ok(accepted)
114    });
115    if let Err(error) = result {
116        let mut failed = Vec::new();
117        let restored_output = published.as_ref().map_or(Ok(()), |identity| {
118            destination.restore(&product, &output, identity)
119        });
120        match restored_output {
121            Ok(()) => {
122                for member in members.iter_mut().rev() {
123                    if let Err(error) = member.restore() {
124                        failed.push(error.to_string());
125                    }
126                }
127            }
128            Err(error) => failed.push(error.to_string()),
129        }
130        if !failed.is_empty() {
131            let recovery = workspace.keep();
132            for member in &mut members {
133                if member.location.is_some()
134                    && let Some(retired) = member.retired.take()
135                {
136                    let _ = retired.keep();
137                }
138            }
139            return Err(MsError::MultiMsRecovery {
140                cause: error.to_string(),
141                restore: failed.join(", "),
142                staging: recovery,
143            });
144        }
145        return Err(error);
146    }
147    result
148}

150struct Assembly<'a> {
151    members: &'a mut [Member],
152    order: &'a [usize],
153    first: &'a Table,
154    product: &'a Path,
155    joined_subtables: &'a [String],
156    copy_subtables: Option<&'a [String]>,
157    locks: &'a [TableWriteLock],
158    locations: &'a relocation::Locations,
159    output: &'a Path,
160}
161
162fn assemble(stage: Assembly<'_>) -> MsResult<()> {
163    let Assembly {
164        members,
165        order,
166        first,
167        product,
168        joined_subtables,
169        copy_subtables,
170        locks,
171        locations,
172        output,
173    } = stage;
174    fs::create_dir_all(product.join("SUBMSS")).map_err(invalid)?;
175    let first_path = first.path().expect("persisted member");
176    let (copy_modes, private_bases) = if copy_subtables.is_some() {
177        copy_modes(members, first_path, locations)?
178    } else {
179        (vec![CopyMode::Move; members.len()], HashSet::new())
180    };
181    for (member, &mode) in members.iter_mut().zip(&copy_modes) {
182        let edit = relocation::MemberEdit::prepare(member, locations)?;
183        let private_base = private_bases.contains(&member.physical);
184        let links = if member.source == member.physical || mode != CopyMode::Move {
185            relocation::filesystem_links(member, locations, output)?
186        } else {
187            Vec::new()
188        };
189        member.stage(edit, mode, private_base, locations, links)?;
190    }
191    if let Some(omit) = copy_subtables {
192        for (member, mode) in members.iter().zip(copy_modes) {
193            if mode != CopyMode::Move {
194                subtables::copy_from_first(
195                    first,
196                    &member.target,
197                    omit,
198                    locks,
199                    mode != CopyMode::CopyNested,
200                    product,
201                )?;
202            }
203        }
204    }
205    for member in members.iter() {
206        if !locks
207            .iter()
208            .any(|lock| lock.matches_table_path(&member.physical))
209            || member.source.canonicalize().ok().as_ref() != Some(&member.physical)
210        {
211            return Err(invalid(format!(
212                "MultiMS member changed: {}",
213                member.source.display()
214            )));
215        }
216    }
217    for member in members.iter_mut() {
218        member.relocate()?;
219    }
220    let tables = order
221        .iter()
222        .map(|&index| Table::open(TableOptions::new(&members[index].target)))
223        .collect::<Result<Vec<_>, _>>()?;
224    subtables::validate_joins(&tables, joined_subtables)?;
225    let mut info = first.info().clone();
226    info.sub_type = "CONCATENATED".into();
227    Table::concat(tables)?.save_with_subtables(
228        TableOptions::new(product),
229        &info,
230        joined_subtables,
231    )?;
232    Ok(())
233}

235#[derive(Clone, Copy, PartialEq, Eq)]
236enum CopyMode {
237    Move,
238    CopySubtables,
239    CopyNested,
240    Materialize,
241}
242
243fn copy_modes(
244    members: &[Member],
245    first: &Path,
246    locations: &relocation::Locations,
247) -> MsResult<(Vec<CopyMode>, HashSet<PathBuf>)> {
248    let mut modes = Vec::with_capacity(members.len());
249    let mut private_bases = HashSet::new();
250    for member in members {
251        let path = &member.physical;
252        if path == first {
253            modes.push(CopyMode::Move);
254            continue;
255        }
256        let mode = match Table::immediate_part_names(path, path)? {
257            TablePartNames::Plain(_) => CopyMode::CopySubtables,
258            TablePartNames::Reference(_) => {
259                if let Some(base_member) = relocation::owned_reference_base(path, locations)? {
260                    // RefTable keyword writes go to its base. Stage that base
261                    // before changing the view or its inherited keywords.
262                    private_bases.insert(base_member);
263                    CopyMode::CopySubtables
264                } else {
265                    CopyMode::Materialize
266                }
267            }
268            TablePartNames::Concatenation { .. } => CopyMode::CopyNested,
269        };
270        modes.push(mode);
271    }
272    Ok((modes, private_bases))
273}

275struct Member {
276    source: PathBuf,
277    physical: PathBuf,
278    target: PathBuf,
279    identity: fs::Metadata,
280    copied: bool,
281    location: Option<PathBuf>,
282    retired: Option<TempDir>,
283}
284
285impl Member {
286    fn stage(
287        &mut self,
288        edit: relocation::MemberEdit,
289        mode: CopyMode,
290        private_base: bool,
291        locations: &relocation::Locations,
292        links: relocation::FilesystemLinks,
293    ) -> MsResult<()> {
294        if self.source != self.physical && mode == CopyMode::Move {
295            relocation::stage_alias(self, locations)?;
296        } else if mode == CopyMode::Materialize {
297            relocation::materialize_member(self, locations)?;
298        } else if mode != CopyMode::Move || private_base || !edit.is_empty() || !links.is_empty() {
299            copy_locked_table_tree(&self.physical, &self.target).map_err(invalid)?;
300            relocation::apply_filesystem_links(links)?;
301            relocation::privatize_descriptor(&self.target)?;
302            edit.apply(&self.target)?;
303        } else {
304            return Ok(());
305        }
306        self.copied = true;
307        Ok(())
308    }
309
310    fn relocate(&mut self) -> MsResult<()> {
311        self.verify_location(&self.source)?;
312        if !self.copied {
313            match rename_without_replacing(&self.source, &self.target) {
314                Ok(()) => {
315                    self.location = Some(self.target.clone());
316                    return self.verify_location(&self.target);
317                }
318                Err(error) if error.kind() == ErrorKind::CrossesDevices => {
319                    copy_locked_table_tree(&self.physical, &self.target).map_err(invalid)?;
320                }
321                Err(error) => return Err(invalid(error)),
322            }
323        }
324        let retired = tempfile::Builder::new()
325            .prefix(".casa-rs-multims-member-")
326            .tempdir_in(self.source.parent().expect("canonical member parent"))
327            .map_err(invalid)?;
328        let location = retired.path().join("original.ms");
329        rename_without_replacing(&self.source, &location).map_err(invalid)?;
330        self.location = Some(location);
331        self.retired = Some(retired);
332        self.verify_location(self.location.as_ref().unwrap())
333    }
334
335    fn verify_location(&self, location: &Path) -> MsResult<()> {
336        if same_entry(
337            &self.identity,
338            &location.symlink_metadata().map_err(invalid)?,
339        ) {
340            return Ok(());
341        }
342        Err(invalid(format!(
343            "MultiMS member changed: {}",
344            self.source.display()
345        )))
346    }
347
348    fn restore(&mut self) -> MsResult<()> {
349        if let Some(location) = &self.location {
350            rename_without_replacing(location, &self.source).map_err(|error| {
351                invalid(format!(
352                    "restore {} from {}: {error}",
353                    self.source.display(),
354                    location.display(),
355                ))
356            })?;
357            self.location = None;
358        }
359        Ok(())
360    }
361}

363fn plan_members(
364    paths: &[PathBuf],
365    output: &Path,
366    product: &Path,
367) -> MsResult<(Vec<Member>, Vec<usize>)> {
368    if paths.is_empty() {
369        return Err(invalid("MultiMS requires at least one member"));
370    }
371    let mut distinct: HashMap<PathBuf, usize> = HashMap::new();
372    let mut names: HashSet<OsString> = HashSet::new();
373    let mut members: Vec<Member> = Vec::new();
374    let mut order = Vec::with_capacity(paths.len());
375    for path in paths {
376        let identity = path.symlink_metadata().map_err(invalid)?;
377        let physical = path.canonicalize().map_err(invalid)?;
378        let source = if identity.is_symlink() {
379            output_path(path)?
380        } else {
381            physical.clone()
382        };
383        if output.starts_with(&source)
384            || source.starts_with(output)
385            || output.starts_with(&physical)
386            || physical.starts_with(output)
387        {
388            return Err(invalid("the MultiMS destination overlaps an input"));
389        }
390        if let Some(&index) = distinct.get(&source) {
391            order.push(index);
392            continue;
393        }
394        if members.iter().any(|member| {
395            source.starts_with(&member.source)
396                || member.source.starts_with(&source)
397                || (physical != member.physical
398                    && (physical.starts_with(&member.physical)
399                        || member.physical.starts_with(&physical)))
400        }) {
401            return Err(invalid("MultiMS members overlap"));
402        }
403        let name = source
404            .file_name()
405            .ok_or_else(|| invalid("member has no file name"))?;
406        if !names.insert(name.to_owned()) {
407            return Err(invalid(format!(
408                "MultiMS members share the file name {}",
409                name.to_string_lossy()
410            )));
411        }
412        let table = Table::open(TableOptions::new(&physical))?;
413        if !table
414            .schema()
415            .is_some_and(|schema| schema.contains_column("DATA_DESC_ID"))
416            || table.keywords().get("MS_VERSION").is_none()
417        {
418            return Err(invalid(format!(
419                "{} is not a MeasurementSet",
420                source.display()
421            )));
422        }
423        let index = members.len();
424        let target = product.join("SUBMSS").join(name);
425        distinct.insert(source.clone(), index);
426        order.push(index);
427        members.push(Member {
428            source,
429            physical,
430            target,
431            identity,
432            copied: false,
433            location: None,
434            retired: None,
435        });
436    }
437    Ok((members, order))
438}

440enum Destination {
441    Absent,
442    Empty(fs::Metadata),
443    Table(fs::Metadata),
444}
445
446impl Destination {
447    fn open(path: &Path) -> MsResult<Self> {
448        match path.symlink_metadata() {
449            Err(error) if error.kind() == ErrorKind::NotFound => Ok(Self::Absent),
450            Err(error) => Err(invalid(error)),
451            Ok(metadata) if metadata.is_dir() => {
452                if path.read_dir().map_err(invalid)?.next().is_none() {
453                    Ok(Self::Empty(metadata))
454                } else if Table::open(TableOptions::new(path)).is_ok() {
455                    Ok(Self::Table(metadata))
456                } else {
457                    Err(invalid("existing MultiMS output directory is not a table"))
458                }
459            }
460            Ok(_) => Err(invalid("existing MultiMS output is not a table directory")),
461        }
462    }
463
464    fn publish(&self, product: &Path, output: &Path) -> MsResult<()> {
465        let before = match self {
466            Self::Absent => return rename_without_replacing(product, output).map_err(invalid),
467            Self::Empty(before) | Self::Table(before) => before,
468        };
469        let current = output.symlink_metadata().map_err(invalid)?;
470        if !same_entry(before, &current) {
471            return Err(invalid("MultiMS destination changed during creation"));
472        }
473        match self {
474            Self::Empty(_) | Self::Table(_) => {
475                exchange_directories(product, output).map_err(invalid)
476            }
477            Self::Absent => unreachable!("handled above"),
478        }
479    }
480
481    fn verify_publication(
482        &self,
483        product: &Path,
484        output: &Path,
485        new: &fs::Metadata,
486    ) -> MsResult<()> {
487        self.verify_output(output, new)?;
488        match self {
489            Self::Absent => Ok(()),
490            Self::Empty(old) | Self::Table(old) => {
491                if !same_entry(old, &product.symlink_metadata().map_err(invalid)?)
492                    || (matches!(self, Self::Empty(_))
493                        && product.read_dir().map_err(invalid)?.next().is_some())
494                {
495                    return Err(invalid("MultiMS destination changed during creation"));
496                }
497                Ok(())
498            }
499        }
500    }
501
502    fn verify_output(&self, output: &Path, identity: &fs::Metadata) -> MsResult<()> {
503        if same_entry(identity, &output.symlink_metadata().map_err(invalid)?) {
504            return Ok(());
505        }
506        Err(invalid(format!(
507            "published MultiMS directory changed: {}",
508            output.display()
509        )))
510    }
511
512    fn restore(&self, product: &Path, output: &Path, new: &fs::Metadata) -> MsResult<()> {
513        self.verify_output(output, new)?;
514        match self {
515            Self::Absent => rename_without_replacing(output, product).map_err(invalid),
516            Self::Empty(_) | Self::Table(_) => {
517                exchange_directories(product, output).map_err(invalid)
518            }
519        }
520    }
521}
522
523fn same_entry(before: &fs::Metadata, current: &fs::Metadata) -> bool {
524    if before.file_type() != current.file_type() {
525        return false;
526    }
527    #[cfg(unix)]
528    {
529        use std::os::unix::fs::MetadataExt;
530        (before.dev(), before.ino()) == (current.dev(), current.ino())
531    }
532    #[cfg(not(unix))]
533    {
534        before
535            .created()
536            .ok()
537            .zip(current.created().ok())
538            .is_some_and(|(a, b)| a == b)
539    }
540}

542fn output_path(path: &Path) -> MsResult<PathBuf> {
543    let name = path
544        .file_name()
545        .ok_or_else(|| invalid("MultiMS output must name a directory"))?;
546    let parent = path
547        .parent()
548        .filter(|path| !path.as_os_str().is_empty())
549        .unwrap_or(Path::new("."));
550    Ok(parent.canonicalize().map_err(invalid)?.join(name))
551}

562    #[test]
563    fn restores_a_displaced_member_after_detecting_changed_identity() {
564        let directory = tempfile::tempdir().unwrap();
565        let source = directory.path().join("source");
566        let target = directory.path().join("stage");
567        let original = directory.path().join("original");
568        fs::create_dir(&source).unwrap();
569        let mut member = Member {
570            source: source.clone(),
571            physical: source.clone(),
572            target: target.clone(),
573            identity: source.symlink_metadata().unwrap(),
574            copied: false,
575            location: None,
576            retired: None,
577        };
578        // Reproduce the final state of a race between checking the input
579        // name and moving its directory entry.
580        fs::rename(&source, &original).unwrap();
581        fs::create_dir(&source).unwrap();
582        fs::write(source.join("unrelated"), b"keep").unwrap();
583        fs::rename(&source, &target).unwrap();
584        member.location = Some(target.clone());
585        assert!(member.verify_location(&target).is_err());
586        member.restore().unwrap();
587        assert!(original.is_dir());
588        assert_eq!(fs::read(source.join("unrelated")).unwrap(), b"keep");
589    }

591    #[test]
592    fn restores_a_displaced_output_after_detecting_changed_identity() {
593        let directory = tempfile::tempdir().unwrap();
594        let output = directory.path().join("output");
595        let product = directory.path().join("product");
596        let original = directory.path().join("original");
597        fs::create_dir(&output).unwrap();
598        fs::create_dir(&product).unwrap();
599        let expected = Destination::Table(output.symlink_metadata().unwrap());
600        let new = product.symlink_metadata().unwrap();
601        fs::rename(&output, &original).unwrap();
602        fs::create_dir(&output).unwrap();
603        fs::write(output.join("unrelated"), b"keep").unwrap();
604        exchange_directories(&product, &output).unwrap();
605        assert!(
606            expected

610        expected.restore(&product, &output, &new).unwrap();
611        assert!(original.is_dir());
612        assert_eq!(fs::read(output.join("unrelated")).unwrap(), b"keep");
613    }
