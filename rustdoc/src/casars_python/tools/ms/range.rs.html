2//! Selected MAIN column ranges through the public MS tool.
3

6use pyo3::prelude::*;
7
8use super::legacy_selection::deprecation;
9use super::metadata_link::compose_rows;
10use super::owned_record::visibility_record;
11use super::visibility::requested_strings;
12use super::{PyMsTool, runtime_error, strict_bool};

15#[pymethods]
16impl PyMsTool {
17    #[pyo3(signature = (items = None, useflags = true, blocksize = 10), text_signature = "($self, items=[], useflags=True, blocksize=10)")]
18    fn range(
19        &self,
20        py: Python<'_>,
21        items: Option<&Bound<'_, PyAny>>,
22        #[pyo3(from_py_with = strict_bool)] useflags: bool,
23        #[pyo3(from_py_with = blocksize_argument)] blocksize: i64,
24    ) -> PyResult<Py<PyAny>> {
25        let items = requested_strings(items, "items")?;
26        let request = ToolRangeRequest::new(&items, useflags, blocksize);
27        let reply = py.detach(|| {
28            let mut attached = self.attached.lock().map_err(runtime_error)?;
29            match attached.as_mut() {
30                Some(ms) => ms.visibility_range(&request),
31                None => Ok(CasaValue::empty_record()),
32            }
33        })?;
34        reply.into_python(py)
35    }
36
37    #[pyo3(signature = (items = None, useflags = true, blocksize = 10), text_signature = "($self, items=[], useflags=True, blocksize=10)")]
38    fn rangeold(
39        &self,
40        py: Python<'_>,
41        items: Option<&Bound<'_, PyAny>>,
42        #[pyo3(from_py_with = strict_bool)] useflags: bool,
43        #[pyo3(from_py_with = legacy_blocksize_argument)] blocksize: i64,
44    ) -> PyResult<Py<PyAny>> {
45        let items = requested_strings(items, "items")?;
46        deprecation(py, "rangeold", "range")?;
47        let request = ToolRangeRequest::new(&items, useflags, i64::from(blocksize as i32));
48        let reply = py.detach(|| {
49            let mut attached = self.attached.lock().map_err(runtime_error)?;
50            let Some(ms) = attached.as_mut() else {
51                return Ok(CasaValue::empty_record());
52            };
53            let ms = ms.legacy_view_mut();
54            let link = ms.working_metadata_link();
55            let base = link.selected_rows().map_err(runtime_error)?;
56            let rows = compose_rows(base.as_deref(), ms.legacy_selection.rows());
57            let options = ms.legacy_selection.visibility_options();
58            if let Some(memory) = &ms.memory_ms {
59                let rows = rows.map_or_else(
60                    || (0..memory.row_count()).collect(),
61                    |rows| rows.into_owned(),
62                );
63                if rows.is_empty() {
64                    return Ok(CasaValue::empty_record());
65                }
66                let record = request
67                    .read(memory.main_table(), &[], None, &rows, options, |ids| {
68                        ids.iter()
69                            .map(|&id| {
70                                memory
71                                    .subtable(id)
72                                    .map(|table| (id, std::slice::from_ref(table)))
73                                    .ok_or_else(|| {
74                                        casa_ms::MsError::MissingSubtable(id.name().to_owned())
75                                    })
76                            })
77                            .collect()
78                    })
79                    .map_err(runtime_error)?;
80                return Ok(visibility_record(record));
81            }
82            if let Some(main) = ms
83                .legacy_selection
84                .positioned_iteration()
85                .and_then(|iteration| iteration.main.as_deref())
86            {
87                link.verify_directory().map_err(runtime_error)?;
88                main.verify_iterator_layout()?;
89                return ms.visibility.range_from_main(
90                    main,
91                    &link,
92                    rows.as_deref(),
93                    &request,
94                    options,
95                );
96            }
97            ms.visibility
98                .range_for_rows(&ms.path, &link, rows.as_deref(), &request, options)
99        })?;
100        reply.into_python(py)
101    }
102}

104fn blocksize_argument(value: &Bound<'_, PyAny>) -> PyResult<i64> {
105    parse_blocksize(value, "ms_range")
106}
107
108fn legacy_blocksize_argument(value: &Bound<'_, PyAny>) -> PyResult<i64> {
109    parse_blocksize(value, "ms_rangeold")
110}
111
112fn parse_blocksize(value: &Bound<'_, PyAny>, method: &str) -> PyResult<i64> {
113    if value.is_none() {
114        return Err(PySystemError::new_err(format!(
115            "<built-in function {method}> returned a result with an exception set"
116        )));
117    }
118    value
119        .extract()
120        .map_err(|_| PyTypeError::new_err("argument _blocksize must be an integer"))
121}
