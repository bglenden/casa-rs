2//! The host, memory, workers and device a session runs on.
3//!
4//! A session has no budget or worker model of its own (plan §4,
5//! Prediction): `close` admits its memory against the imaging runtime's
6//! [`ResourcePolicy`] on the detected [`HostResources`], and runs its
7//! geometry and noise on the runtime's worker team.
8//!
9//! The controls are the Python tool's constructor arguments (ADR-0019 item
10//! 3), [`ExecutionControls`]:
11//! - with neither a memory budget nor a worker count, the session takes the
12//!   balanced policy: three quarters of the host's free memory and of its
13//!   threads, at most its performance cores, except that a `close` that
14//!   predicts on Metal streams on every thread but the writer's
15//!   ([`SessionResources::close_cores`]);
16//! - with either, an explicit policy whose other half is the balanced
17//!   policy's on the host.
18//!
19//! The budget caps everything `close` holds at once: the MeasurementSet
20//! writer gets half of what the policy leaves free, and the batch being
21//! filled, its geometry and the worker scratch come out of the rest
22//! (`write::resources`). The worker count is the runtime's: threads kept
23//! busy, the MeasurementSet writer's among them.
24//!
25//! The backend is where a session predicts its model
26//! ([`SessionResources::prediction_backend`]); noise never runs on a device,
27//! so a session that predicts nothing writes the same MeasurementSet
28//! whatever the request. Every kernel set a session predicts with is the
29//! standard `GridFT` one, which the imaging runtime's Metal laws and the
30//! simulator's own (`tests/backends.rs`) cover, so `Auto` takes the host's
31//! Metal device whenever there is one.
32

41/// The device a session asks to predict on.
42#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
43pub enum BackendRequest {
44    /// The host's Metal device where it predicts the session's model, and
45    /// the CPU otherwise; the CPU while a session predicts nothing.
46    #[default]
47    Auto,
48    /// The CPU.
49    Cpu,
50    /// The host's Metal device; a host without one is refused when the
51    /// session's resources are built.
52    Metal,
53}
54
55impl BackendRequest {
56    /// The request's name, as the Python tool takes it.
57    #[must_use]
58    pub const fn name(self) -> &'static str {
59        match self {
60            Self::Auto => "auto",
61            Self::Cpu => "cpu",
62            Self::Metal => "metal",
63        }
64    }
65}
66
67/// A backend name other than `auto`, `cpu` or `metal`.
68#[derive(Clone, Debug, PartialEq, Eq, thiserror::Error)]
69#[error("backend must be 'auto', 'cpu' or 'metal', not '{0}'")]
70pub struct UnknownBackend(pub String);
71
72impl FromStr for BackendRequest {
73    type Err = UnknownBackend;
74
75    fn from_str(name: &str) -> Result<Self, Self::Err> {
76        [Self::Auto, Self::Cpu, Self::Metal]
77            .into_iter()
78            .find(|request| request.name() == name)
79            .ok_or_else(|| UnknownBackend(name.to_string()))
80    }
81}

83/// Controls CASA's `sm` lacks: the Python tool's keyword-only constructor
84/// arguments (ADR-0019 item 3).
85#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
86pub struct ExecutionControls {
87    /// Bytes the session may hold at once while `close` writes, capped at
88    /// the host's free memory; `None` for the balanced policy's three
89    /// quarters of it.
90    pub memory_budget: Option<u64>,
91    /// Threads `close` keeps busy, the MeasurementSet writer's among them,
92    /// capped at the host's; `None` for the balanced policy's.
93    pub workers: Option<NonZeroUsize>,
94    /// The device to predict on.
95    pub backend: BackendRequest,
96}
97
98/// The host, resource policy and backend of a session: what `close` admits
99/// its memory against and sizes its worker team from.
100#[derive(Clone, Copy, Debug, PartialEq, Eq)]
101pub struct SessionResources {
102    host: HostResources,
103    policy: ResourcePolicy,
104    backend: BackendRequest,
105}
106
107impl SessionResources {
108    /// The resources `controls` ask for on this host, detected once per
109    /// process.
110    ///
111    /// # Errors
112    ///
113    /// The host's threads or free memory could not be read, or `Metal` was
114    /// asked for on a host without a Metal device.
115    pub fn from_controls(controls: &ExecutionControls) -> SimulatorResult<Self> {
116        let host = HostResources::detect()
117            .map_err(|error| SimulatorError::raised(format!("host resources: {error}")))?;
118        Self::on_host(host, controls)
119    }
120
121    /// The resources `controls` ask for on `host`.
122    ///
123    /// # Errors
124    ///
125    /// `Metal` was asked for on a host without a Metal device.
126    pub fn on_host(host: HostResources, controls: &ExecutionControls) -> SimulatorResult<Self> {
127        let balanced = ResourcePolicy::Balanced;
128        let policy = match (controls.memory_budget, controls.workers) {
129            (None, None) => balanced,
130            (memory, workers) => ResourcePolicy::Explicit {
131                workers: workers.map_or_else(|| balanced.workers(&host), NonZeroUsize::get),
132                memory: memory.unwrap_or_else(|| balanced.memory(&host)),
133            },
134        };
135        Self::pinned(host, policy, controls.backend)
136    }
137
138    /// The resources of `policy` on `host` with `backend`, as given: tests
139    /// and benchmarks pin the host rather than detect it.
140    ///
141    /// # Errors
142    ///
143    /// `Metal` was asked for on a host without a Metal device.
144    pub fn pinned(
145        host: HostResources,
146        policy: ResourcePolicy,
147        backend: BackendRequest,
148    ) -> SimulatorResult<Self> {
149        if backend == BackendRequest::Metal && !host.metal {
150            return Err(SimulatorError::raised(
151                "backend 'metal' needs a Metal device, and this host has none",
152            ));
153        }
154        Ok(Self {
155            host,
156            policy,
157            backend,
158        })
159    }
160
161    /// The host.
162    #[must_use]
163    pub const fn host(&self) -> &HostResources {
164        &self.host
165    }
166
167    /// The resource policy.
168    #[must_use]
169    pub const fn policy(&self) -> &ResourcePolicy {
170        &self.policy
171    }
172
173    /// The backend asked for.
174    #[must_use]
175    pub const fn backend(&self) -> BackendRequest {
176        self.backend
177    }
178
179    /// The policy's workers on the host, before `close` admits its memory:
180    /// `close` streams MAIN to the writer's thread, so it runs the
181    /// streaming workers beside it.
182    #[must_use]
183    pub fn cores(&self) -> CoreAllocation {
184        self.policy.cores(&self.host)
185    }
186
187    /// The workers `close` runs when it predicts on Metal (`metal`) or
188    /// not: the policy's ([`Self::cores`]), except that under the balanced
189    /// policy a `close` that predicts on Metal streams on every thread but
190    /// the writer's, efficiency cores included (owner, #855). The device
191    /// degrids; what is left for the host, placing the samples and adding
192    /// the noise, gains from the efficiency cores where the CPU's own
193    /// degridding does not (#659): at 4096 channels on an M4, nine
194    /// streaming workers closed in 55.8 s against 70.0 s for three.
195    #[must_use]
196    pub fn close_cores(&self, metal: bool) -> CoreAllocation {
197        match self.policy {
198            ResourcePolicy::Balanced if metal => {
199                let streaming_workers = self.host.threads.saturating_sub(STREAM_PRODUCERS).max(1);
200                CoreAllocation {
201                    workers: self.policy.workers(&self.host).max(streaming_workers),
202                    streaming_workers,
203                }
204            }
205            _ => self.cores(),
206        }
207    }
208
209    /// The backend a session runs on: `Auto` takes Metal only where the
210    /// host has a device and the session predicts a model.
211    #[must_use]
212    pub fn resolved_backend(&self, predicts: bool) -> BackendChoice {
213        match self.backend {
214            BackendRequest::Cpu => BackendChoice::Cpu,
215            BackendRequest::Metal => BackendChoice::Metal,
216            BackendRequest::Auto if predicts && self.host.metal => BackendChoice::Metal,
217            BackendRequest::Auto => BackendChoice::Cpu,
218        }
219    }
220
221    /// The backend `close` predicts a session's model on, once it predicts
222    /// one: [`Self::resolved_backend`] of a session that predicts. `Auto`
223    /// is Metal on a host with a device and the CPU otherwise; `Metal` and
224    /// `Cpu` are what they ask for. On Metal, `close` also admits the
225    /// device copies of the model grids and the device predictor's bytes.
226    #[must_use]
227    pub fn prediction_backend(&self) -> BackendChoice {
228        self.resolved_backend(true)
229    }
230}

232#[cfg(test)]
233mod tests {
234    use super::*;

236    const HOST: HostResources = HostResources {
237        threads: 10,
238        performance_cores: 4,
239        available_memory: 8 << 30,
240        metal: false,

243    #[test]
244    fn backend_names_round_trip_and_others_are_refused() {
245        for request in [
246            BackendRequest::Auto,
247            BackendRequest::Cpu,
248            BackendRequest::Metal,
249        ] {
250            assert_eq!(request.name().parse::<BackendRequest>(), Ok(request));
251        }
252        let error = "gpu".parse::<BackendRequest>().expect_err("not a backend");
253        assert_eq!(
254            error.to_string(),
255            "backend must be 'auto', 'cpu' or 'metal', not 'gpu'"
256        );
257        assert!("CPU".parse::<BackendRequest>().is_err());
258    }
259
260    #[test]
261    fn metal_is_refused_without_a_device_and_auto_stays_on_the_cpu() {
262        let error = SessionResources::pinned(HOST, ResourcePolicy::Balanced, BackendRequest::Metal)
263            .expect_err("no device");
264        assert!(error.message().contains("Metal device"), "{error}");
265        let with_device = HostResources {
266            metal: true,
267            ..HOST
268        };
269        let metal =
270            SessionResources::pinned(with_device, ResourcePolicy::Balanced, BackendRequest::Metal)
271                .expect("a device");
272        assert_eq!(metal.resolved_backend(false), BackendChoice::Metal);
273        let auto =
274            SessionResources::pinned(with_device, ResourcePolicy::Balanced, BackendRequest::Auto)
275                .expect("auto");
276        assert_eq!(auto.resolved_backend(false), BackendChoice::Cpu);
277        assert_eq!(auto.resolved_backend(true), BackendChoice::Metal);
278        assert_eq!(auto.prediction_backend(), BackendChoice::Metal);
279        assert_eq!(metal.prediction_backend(), BackendChoice::Metal);
280        let cpu = SessionResources::pinned(HOST, ResourcePolicy::Balanced, BackendRequest::Auto)
281            .expect("auto");
282        assert_eq!(cpu.resolved_backend(true), BackendChoice::Cpu);
283        assert_eq!(cpu.prediction_backend(), BackendChoice::Cpu);
284        let asked =
285            SessionResources::pinned(with_device, ResourcePolicy::Balanced, BackendRequest::Cpu)
286                .expect("cpu");
287        assert_eq!(asked.prediction_backend(), BackendChoice::Cpu);
288    }

290    #[test]
291    fn the_balanced_policy_leaves_the_writer_a_performance_core() {
292        let balanced =
293            SessionResources::pinned(HOST, ResourcePolicy::Balanced, BackendRequest::Cpu)
294                .expect("resources");
295        assert_eq!(
296            balanced.cores(),
297            CoreAllocation {
298                workers: 4,
299                streaming_workers: 3,
300            }
301        );
302        let one = SessionResources::pinned(
303            HOST,
304            ResourcePolicy::Explicit {
305                workers: 1,
306                memory: 1 << 30,
307            },
308            BackendRequest::Cpu,
309        )
310        .expect("resources");
311        assert_eq!(one.cores().streaming_workers, 1);
312    }

314    #[test]
315    fn a_metal_close_streams_on_every_thread_but_the_writers_under_the_balanced_policy() {
316        let balanced =
317            SessionResources::pinned(HOST, ResourcePolicy::Balanced, BackendRequest::Cpu)
318                .expect("resources");
319        assert_eq!(
320            balanced.close_cores(true),
321            CoreAllocation {
322                workers: 9,
323                streaming_workers: 9,
324            }
325        );
326        assert_eq!(balanced.close_cores(false), balanced.cores());
327        let explicit = SessionResources::pinned(
328            HOST,
329            ResourcePolicy::Explicit {
330                workers: 4,
331                memory: 1 << 30,
332            },
333            BackendRequest::Cpu,
334        )
335        .expect("resources");
336        assert_eq!(explicit.close_cores(true), explicit.cores());
337    }

339    #[test]
340    fn either_control_makes_the_policy_explicit_with_the_other_balanced() {
341        let on_host = |controls: ExecutionControls| {
342            *SessionResources::on_host(HOST, &controls)
343                .expect("resources")
344                .policy()
345        };
346        assert_eq!(
347            on_host(ExecutionControls::default()),
348            ResourcePolicy::Balanced
349        );
350        let balanced = ResourcePolicy::Balanced;
351        assert_eq!(
352            on_host(ExecutionControls {
353                memory_budget: Some(1 << 28),
354                ..ExecutionControls::default()
355            }),
356            ResourcePolicy::Explicit {
357                workers: balanced.workers(&HOST),
358                memory: 1 << 28,
359            }
360        );
361        assert_eq!(
362            on_host(ExecutionControls {
363                workers: NonZeroUsize::new(2),
364                ..ExecutionControls::default()
365            }),
366            ResourcePolicy::Explicit {
367                workers: 2,
368                memory: balanced.memory(&HOST),
369            }
370        );
