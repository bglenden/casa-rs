15const MISSING_FRAME: &str = "Missing information in Frame specified for conversion";

17pub(super) fn epoch_text(mjd: MjdHighPrec) -> String {
18    let fraction = mjd.frac();
19    let hours = (24.0 * fraction).floor() as u32;
20    let minutes = (60.0 * (24.0 * fraction - f64::from(hours))).floor() as u32;
21    let seconds = 86_400.0 * fraction - f64::from(minutes) * 60.0 - f64::from(hours) * 3_600.0;
22    format!(
23        "{}::{hours:02}:{minutes:02}:{seconds:07.4}",
24        stream_double(mjd.day())
25    )
26}
27
28pub(super) fn vector<const N: usize>(values: [f64; N]) -> String {
29    format!("[{}]", values.map(stream_double).join(", "))
30}

32fn radial_error(error: MeasureError) -> String {
33    if let MeasureError::MissingFrameData { what } = error {
34        let name = what.split_whitespace().next().unwrap_or(what);
35        format!("No MeasFrame {name} specified for conversion of Radialvelocity")
36    } else {
37        error.to_string()
38    }
39}

41fn render(state: &mut State, mut native: MeasFrame) -> Result<String, String> {
42    let mut elements = Vec::new();
43    if let Some(epoch) = native.epoch() {
44        let to = |refer| match epoch.convert_to(refer, &native) {
45            Ok(value) => Ok(stream_double(value.value().as_mjd())),
46            Err(MeasureError::MissingFrameData { what }) if what.starts_with("dUT1") => {
47                Ok("unavailable".to_owned())
48            }
49            Err(error) => Err(error.to_string()),
50        };
51        let (tdb, ut1, tt) = (to(EpochRef::TDB)?, to(EpochRef::UT1)?, to(EpochRef::TT)?);
52        let Some(MeasureValue::Epoch(raw)) =
53            state.frame.epoch.as_ref().map(|epoch| epoch.measure.value)
54        else {
55            unreachable!("the explicit frame has an epoch");
56        };
57        elements.push(format!(
58            "Epoch: {} (TDB = {tdb}, UT1 = {ut1}, TT = {tt})",
59            epoch_text(raw)
60        ));
61    }
62    if let (Some(MeasureValue::Position(xyz)), Some(position)) = (
63        state
64            .frame
65            .position
66            .as_ref()
67            .map(|position| position.measure.value),
68        native.position(),
69    ) {
70        let (longitude, latitude, _) = spatial_spherical(position.as_itrf());
71        elements.push(format!(
72            "Position: {}\n        (Longitude = {} Latitude = {})",
73            vector(xyz),
74            stream_double(longitude),
75            stream_double(latitude)
76        ));
77    }
78    if let Some(direction) = native.direction() {
79        let converted = frame::convert_direction(state, direction, DirectionRef::J2000, &native)
80            .map_err(|error| match error {
81                MeasureError::MissingFrameData {
82                    what: COMET_UNAVAILABLE,
83                } => COMET_UNAVAILABLE.to_owned(),
84                MeasureError::MissingFrameData { .. } => MISSING_FRAME.to_owned(),
85                error => error.to_string(),
86            })?;
87        let (longitude, latitude) = direction_angles(converted.cosines());
88        let Some(MeasureValue::Direction(raw)) = state
89            .frame
90            .direction
91            .as_ref()
92            .map(|direction| direction.measure.value)
93        else {
94            unreachable!("the explicit frame has a direction");
95        };
96        elements.push(format!(
97            "Direction: {}\n        (J2000 = {} deg)",
98            vector(raw),
99            vector([longitude.to_degrees(), latitude.to_degrees()])
100        ));
101        if native.radial_velocity().is_some() {
102            native = frame::with_resolved_direction(state, native);
103        }
104    }
105    if let (Some(record), Some(velocity)) = (&state.frame.radial_velocity, native.radial_velocity())
106    {
107        let MeasureValue::RadialVelocity(raw) = record.measure.value else {
108            unreachable!("the frame holds a radial velocity");
109        };
110        if matches!(
111            velocity.refer(),
112            RadialVelocityRef::GEO | RadialVelocityRef::TOPO
113        ) && native.epoch().is_none()
114        {
115            return Err(radial_error(MeasureError::MissingFrameData {
116                what: "epoch",
117            }));
118        }
119        let lsr = velocity
120            .convert_to(RadialVelocityRef::LSRK, &native)
121            .map_err(radial_error)?;
122        elements.push(format!(
123            "Radialvelocity: {}\n        (LSR velocity = {} km/s)",
124            stream_double(raw),
125            stream_double(lsr.ms() / 1e3)
126        ));
127    }
128    if let Some(comet) = &state.frame.comet {
129        let [start, end] = comet.span_mjd().map(stream_double);
130        elements.push(format!(
131            "{} comet between MJD {start} and {end}",
132            comet.name()
133        ));
134    }
135    Ok(format!("Frame: {}", elements.join("\n       ")))
136}

138#[pymethods]
139impl PyMeasures {
140    /// Display the explicit frame and its derived coordinates.
141    fn showframe(&self, py: Python<'_>) -> PyResult<Py<PyAny>> {
142        self.call(py, |state, lines| {
143            if let Some(text) = &state.frame.rendered {
144                return Ok(CasaValue::Text(text.clone()));
145            }
146            let frame = if state.frame.epoch.is_some() {
147                frame::native_frame(state)?
148            } else {
149                frame::materialize(&state.frame, MeasFrame::new())
150            };
151            frame
152                .and_then(|frame| render(state, frame))
153                .map(|text| {
154                    state.frame.rendered = Some(text.clone());
155                    CasaValue::Text(text)
156                })
157                .map_err(|error| {
158                    lines.push(LogLine::error(format!("Exception Reports: {error}")));
159                    PyRuntimeError::new_err(error)
160                })
161        })
162    }
163}
