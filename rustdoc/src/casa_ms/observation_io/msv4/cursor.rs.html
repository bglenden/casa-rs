25/// Reusable decode storage for what cannot be decoded in place: the flag
26/// bytes before their conversion to `bool`, the stored weights the flags
27/// read when weights are not delivered, and the full correlation axis when
28/// a subset is delivered.
29#[derive(Default)]
30struct Scratch {
31    flag_bytes: Vec<u8>,
32    values: Vec<Complex32>,
33    weights: Vec<f32>,
34}

36/// A bound traversal of one MSv4 dataset: whole times in time order.
37pub struct Msv4Cursor<'o> {
38    observation: &'o Msv4Observation,
39    dataset: &'o Dataset,
40    arrays: &'o GroupArrays,
41    partition: PartitionId,
42    roles: Roles,
43    channels: Range<usize>,
44    resolved: ResolvedRequest,
45    /// The next dataset time to deliver, within `resolved.times`.
46    next_time: usize,
47    groups: Option<GroupIndex>,
48    estimate: PartitionEstimate,
49    counters: IoCounters,
50    scratch: Scratch,
51}
52
53impl<'o> Msv4Cursor<'o> {
54    pub(super) fn bind(
55        observation: &'o Msv4Observation,
56        request: &AccessRequest,
57        estimate: PartitionEstimate,
58        resolved: ResolvedRequest,
59    ) -> Result<Self, ObservationError> {
60        let (dataset, _) = observation.dataset(estimate.partition)?;
61        let arrays = &dataset.groups[resolved.data_group];
62        let groups = GroupIndex::new(
63            estimate.partition,
64            request,
65            observation.open.group_closure,
66            &dataset.times,
67            &arrays.fields,
68            dataset.scans.as_deref(),
69            &resolved.times,
70        )?;
71        Ok(Self {
72            observation,
73            dataset,
74            arrays,
75            partition: estimate.partition,
76            roles: request.roles,
77            channels: request.channels.clone(),
78            next_time: resolved.times.first().unwrap_or(0),
79            resolved,
80            groups,
81            estimate,
82            counters: IoCounters::default(),
83            scratch: Scratch::default(),
84        })
85    }
86}

88/// What a request reads: each variable decoded, whether it spans
89/// frequency, or is UVW (its last axis whole), or is `[time, baseline]`, and
90/// whether it is delivered (`WEIGHT` is decoded for the flags alone when
91/// flags are read without weights).
92#[derive(Clone, Copy)]
93enum Extent {
94    Cube,
95    Uvw,
96    TimeBaseline,
97}
98
99fn requested<'a>(
100    dataset: &'a Dataset,
101    arrays: &'a GroupArrays,
102    roles: Roles,
103) -> Vec<(&'a Array<FilesystemStore>, Extent, bool)> {
104    let mut read = Vec::new();
105    for (decoded, delivered, array, extent) in [
106        (roles.values, true, Some(&arrays.values), Extent::Cube),
107        (roles.flags, true, Some(&arrays.flags), Extent::Cube),
108        (
109            roles.weights || roles.flags,
110            roles.weights,
111            Some(&arrays.weights),
112            Extent::Cube,
113        ),
114        (roles.uvw, true, Some(&arrays.uvw), Extent::Uvw),
115        (
116            roles.time_centroid,
117            true,
118            dataset.centroid.as_ref(),
119            Extent::TimeBaseline,
120        ),
121        (
122            roles.exposure,
123            true,
124            dataset.exposure.as_ref(),
125            Extent::TimeBaseline,
126        ),
127    ] {
128        if let (true, Some(array)) = (decoded, array) {
129            read.push((array, extent, delivered));
130        }
131    }
132    read
133}

135/// What `request` costs on partition `id`: the chunks the traversal
136/// touches and the elements they hold, modelled from the chunk grid, and
137/// the grid's shape.
138///
139/// # Errors
140///
141/// When the request names an unknown partition, data group, channel or
142/// product, asks for a role the dataset does not hold (`TIME_CENTROID`,
143/// `EFFECTIVE_INTEGRATION_TIME`, `time.integration_time` or `scan_name`),
144/// or budgets less than one time; unsupported when it asks for centroids
145/// the dataset splits with `TIME_CENTROID_EXTRA_PRECISION`.
146pub(super) fn estimate(
147    observation: &Msv4Observation,
148    request: &AccessRequest,
149    id: PartitionId,
150) -> Result<(PartitionEstimate, ResolvedRequest), ObservationError> {
151    let (dataset, entry) = observation.dataset(id)?;
152    // The requested group's field at each time; a request naming no group
153    // of the partition is refused by the resolution.
154    let fields = entry
155        .data_group(&request.data_group)
156        .map_or(&[][..], |(index, _)| &dataset.groups[index].fields);
157    let resolved = request.resolve(entry, &dataset.times, fields)?;
158    let arrays = &dataset.groups[resolved.data_group];
159    let roles = request.roles;
160    if roles.time_centroid && dataset.centroid_extra_precision {
161        return Err(ObservationError::Unsupported {
162            what: "an MSv4 TIME_CENTROID split with TIME_CENTROID_EXTRA_PRECISION: \
163                   the profile carries one f64 offset per centroid",
164        });
165    }
166    for (wanted, present, variable) in [
167        (
168            roles.time_centroid,
169            dataset.centroid.is_some(),
170            "TIME_CENTROID",
171        ),
172        (
173            roles.exposure,
174            dataset.exposure.is_some(),
175            "EFFECTIVE_INTEGRATION_TIME",
176        ),
177        (
178            roles.interval,
179            dataset.interval_s.is_some(),
180            "time.integration_time",
181        ),
182        (roles.scan, dataset.scans.is_some(), "scan_name"),
183    ] {
184        if wanted && !present {
185            return Err(ObservationError::InvalidRequest {
186                reason: format!("the dataset has no {variable}, which the request reads"),
187            });
188        }
189    }
190    // An MSv4 dataset holds weights and no sigmas: the sigma rule takes
191    // its fallback, recorded, or is refused.
192    let weights = match request.weights {
193        WeightRequest::InverseSigmaSquared {
194            fallback: SigmaFallback::Refuse,
195        } if roles.weights => {
196            return Err(ObservationError::InvalidRequest {
197                reason: "an MSv4 dataset holds no sigmas for 1/sigma^2 weights, and the request \
198                         names no fallback"
199                    .to_string(),
200            });
201        }
202        _ => WeightSource {
203            rule: WeightRule::Stored,
204            variable: entry.data_groups[resolved.data_group].weights.clone(),
205            per_channel: true,
206        },
207    };
208    let (times, baselines) = (resolved.times.len(), entry.baselines.len());
209    let channels = request.channels.len();
210    let scalars = [roles.time_centroid, roles.exposure]
211        .into_iter()
212        .filter(|wanted| *wanted)
213        .count();
214    let delivered = resolved.correlations.products(entry).len();
215    let bytes_per_sample = roles.sample_bytes(channels, delivered, true, scalars);
216    // The cursor's scratch: the flag bytes before their conversion; the
217    // values of every stored product when a subset is delivered; and the
218    // stored weights, unless they are decoded straight into the batch (the
219    // flags read them too).
220    let stored = channels * entry.correlations.len();
221    let gathered = resolved.correlations.indices().is_some();
222    let weights_in_scratch = (roles.flags && !roles.weights) || (roles.weights && gathered);
223    let scratch_bytes_per_sample = stored
224        * (usize::from(roles.flags)
225            + usize::from(roles.values && gathered) * 8
226            + usize::from(weights_in_scratch) * 4);
227    // `time.integration_time` is one constant: no interval array.
228    let bytes_per_time = roles.time_bytes(0);
229    let times_per_batch = times_per_batch(
230        request.batch_bytes,
231        baselines * (bytes_per_sample + scratch_bytes_per_sample) + bytes_per_time,
232        times,
233    )?;
234    let blocks: Vec<Range<usize>> = if times_per_batch == 0 {
235        Vec::new()
236    } else {
237        resolved.times.blocks(times_per_batch).collect()
238    };
239    let batches = blocks.len();
240    let (mut decode_units, mut decoded, mut selected) = (0, 0, 0);
241    let variables = requested(dataset, arrays, roles);
242    for rows in blocks {
243        for &(array, extent, delivered) in &variables {
244            let subset = subset(array, rows.clone(), &request.channels, extent);
245            let (units, elements) = chunk_cost(array, &subset)?;
246            decode_units += units;
247            decoded += elements;
248            selected += match (delivered, extent, resolved.correlations.indices()) {
249                (false, ..) => 0,
250                (true, Extent::Cube, Some(indices)) => {
251                    subset.num_elements_usize() / entry.correlations.len() * indices.len()
252                }
253                (true, ..) => subset.num_elements_usize(),
254            };
255        }
256    }
257    let chunk = arrays.values.chunk_shape(&[0, 0, 0, 0]).map_err(storage)?;
258    let estimate = PartitionEstimate {
259        partition: id,
260        times,
261        time_runs: resolved.times.clone(),
262        weights: roles.weights.then_some(weights),
263        baselines,
264        bytes_per_sample,
265        bytes_per_time,
266        scratch_bytes_per_sample,
267        times_per_batch,
268        batches,
269        decode_units,
270        decoded_elements: decoded,
271        selected_elements: selected,
272        natural_blocks: NaturalBlocks {
273            times: chunk[0].get() as usize,
274            channels: chunk[2].get() as usize,
275        },
276        padded_fraction: 0.0,
277    };
278    Ok((estimate, resolved))
279}

281/// The array subset of times `rows` and, for a cube, channels `channels`.
282fn subset(
283    array: &Array<FilesystemStore>,
284    rows: Range<usize>,
285    channels: &Range<usize>,
286    extent: Extent,
287) -> ArraySubset {
288    let shape = array.shape();
289    let rows = rows.start as u64..rows.end as u64;
290    match extent {
291        Extent::Cube => ArraySubset::new_with_ranges(&[
292            rows,
293            0..shape[1],
294            channels.start as u64..channels.end as u64,
295            0..shape[3],
296        ]),
297        Extent::Uvw => ArraySubset::new_with_ranges(&[rows, 0..shape[1], 0..3]),
298        Extent::TimeBaseline => ArraySubset::new_with_ranges(&[rows, 0..shape[1]]),
299    }
300}

302/// The chunks `subset` touches and the elements those chunks hold.
303fn chunk_cost(
304    array: &Array<FilesystemStore>,
305    subset: &ArraySubset,
306) -> Result<(usize, usize), ObservationError> {
307    let chunks = array
308        .chunks_in_array_subset(subset)
309        .map_err(storage)?
310        .ok_or_else(|| invalid("the chunk grid cannot be indexed"))?;
311    let mut units = 0;
312    let mut elements = 0;
313    for indices in &chunks.indices() {
314        let shape = array.chunk_shape(&indices).map_err(storage)?;
315        units += 1;
316        elements += shape.iter().map(|n| n.get() as usize).product::<usize>();
317    }
318    Ok((units, elements))
319}

321/// Decode `subset` of `array`, whose Zarr data type must be `data_type`,
322/// straight into `destination`, whose length is the subset's element
323/// count: zarrs writes each chunk's selected elements into place, with no
324/// intermediate whole-subset copy. The decode copies bytes, so the type
325/// check here (and the layout check at opening) is what makes the bytes
326/// `T`s.
327fn decode_into<T: bytemuck::Pod>(
328    array: &Array<FilesystemStore>,
329    data_type: &str,
330    subset: &ArraySubset,
331    destination: &mut [T],
332    options: &CodecOptions,
333) -> Result<(), ObservationError> {
334    let ArrayMetadata::V3(metadata) = array.metadata() else {
335        return Err(invalid("not a Zarr v3 array"));
336    };
337    if metadata.data_type.name() != data_type
338        || array.data_type().fixed_size() != Some(size_of::<T>())
339        || destination.len() != subset.num_elements_usize()
340    {
341        return Err(invalid(format!(
342            "decode of {} into {data_type} elements",
343            metadata.data_type.name()
344        )));
345    }
346    let shape: Vec<u64> = subset.shape().to_vec();
347    let bytes = bytemuck::cast_slice_mut::<T, u8>(destination);
348    let cells = unsafe_cell_slice::UnsafeCellSlice::new(bytes);
349    // SAFETY: the view covers the whole destination, which this function
350    // borrows exclusively, so no other view of these bytes exists; the
351    // element size is the array's, checked above.
352    let mut view = unsafe {
353        ArrayBytesFixedDisjointView::new(
354            cells,
355            size_of::<T>(),
356            &shape,
357            ArraySubset::new_with_shape(shape.clone()),
358        )
359    }
360    .map_err(storage)?;
361    array
362        .retrieve_array_subset_into_opt(
363            subset,
364            ArrayBytesDecodeIntoTarget::Fixed(&mut view),
365            options,
366        )
367        .map_err(storage)
368}

370/// Keep products `keep` of every `[channel][stored]` cell of `source`.
371fn gather<T: Copy>(source: &[T], stored: usize, keep: &[usize], target: &mut Vec<T>) {
372    target.clear();
373    target.reserve(source.len() / stored * keep.len());
374    for cell in source.chunks_exact(stored) {
375        target.extend(keep.iter().map(|&product| cell[product]));
376    }
377}

379impl Msv4Cursor<'_> {
380    fn header(&self, times: usize) -> BatchHeader {
381        BatchHeader {
382            partition: self.partition,
383            data_group: self.resolved.data_group,
384            times,
385            first_channel: self.channels.start,
386            channels: self.channels.len(),
387            correlations: self.resolved.correlations.clone(),
388            roles: self.roles,
389            weights_per_channel: true,
390            // An MSv4 holds no sigmas: its weights are always the stored ones.
391            weight_rule: WeightRule::Stored,
392            interval: Sharing::Constant(self.dataset.interval_s.unwrap_or(f64::NAN)),
393            exposure: Sharing::PerSample,
394        }
395    }
396
397    fn fill_batch(
398        &mut self,
399        buffers: &mut BatchBuffers,
400        rows: Range<usize>,
401    ) -> Result<(), ObservationError> {
402        let dataset = self.dataset;
403        let arrays = self.arrays;
404        let catalogue = &self.observation.catalogue;
405        let capacity = buffers.capacity_bytes();
406        let header = self.header(rows.len());
407        let mut fill = buffers.begin(header, catalogue)?;
408        let batch = fill.arrays();
409        batch
410            .time_offsets_s
411            .extend_from_slice(&dataset.times[rows.clone()]);
412        batch.field.extend_from_slice(&arrays.fields[rows.clone()]);
413        if self.roles.scan {
414            let scans = dataset.scans.as_ref().expect("checked at binding");
415            batch.scan.extend_from_slice(&scans[rows.clone()]);
416        }
417        self.decode_batch(batch, &rows)?;
418        if let Some(groups) = self.groups.as_mut() {
419            groups.fragments(
420                rows,
421                catalogue.partition(self.partition).baselines.len(),
422                &mut batch.fragments,
423            );
424        }
425        self.counters.delivered_bytes += batch.values.len() * size_of::<Complex32>()
426            + batch.flags.len()
427            + batch.weights.len() * size_of::<f32>()
428            + (batch.uvw.len() + batch.centroid_offsets_s.len() + batch.exposure_s.len())
429                * size_of::<f64>();
430        fill.finish()?;
431        if buffers.capacity_bytes() > capacity {
432            self.counters.buffer_growths += 1;
433        }
434        Ok(())
435    }
436
437    /// Decode each requested variable's subset of times `rows` into
438    /// `batch`: in place when every product is delivered, else the whole
439    /// correlation axis into scratch and the delivered products gathered.
440    fn decode_batch(
441        &mut self,
442        batch: &mut BatchArrays,
443        rows: &Range<usize>,
444    ) -> Result<(), ObservationError> {
445        let (dataset, arrays) = (self.dataset, self.arrays);
446        let options = &self.observation.options;
447        let roles = self.roles;
448        let channels = &self.channels;
449        let stored = arrays.values.shape()[3] as usize;
450        let keep = self.resolved.correlations.indices();
451        let counters = &mut self.counters;
452        let scratch = &mut self.scratch;
453        let mut prepare = |wanted: bool,
454                           array: Option<&Array<FilesystemStore>>,
455                           extent: Extent|
456         -> Result<Option<ArraySubset>, ObservationError> {
457            let (true, Some(array)) = (wanted, array) else {
458                return Ok(None);
459            };
460            let subset = subset(array, rows.clone(), channels, extent);
461            counters.read_calls += 1;
462            counters.decode_units += chunk_cost(array, &subset)?.0;
463            Ok(Some(subset))
464        };
465        if let Some(subset) = prepare(roles.values, Some(&arrays.values), Extent::Cube)? {
466            let target = match keep {
467                None => &mut batch.values,
468                Some(_) => &mut scratch.values,
469            };
470            target.resize(subset.num_elements_usize(), Complex32::default());
471            decode_into(&arrays.values, "complex64", &subset, target, options)?;
472            if let Some(keep) = keep {
473                gather(&scratch.values, stored, keep, &mut batch.values);
474                counters.copied_elements += batch.values.len();
475            }
476        }
477        // The stored weights come first whenever flags are read: XRADIO pads
478        // a cell no row filled with NaN and leaves its FLAG false, so a
479        // non-finite stored weight marks an absent acquisition, and every
480        // delivered flag carries it.
481        let weights_in_place = roles.weights && keep.is_none();
482        if let Some(subset) = prepare(
483            roles.weights || roles.flags,
484            Some(&arrays.weights),
485            Extent::Cube,
486        )? {
487            let target = if weights_in_place {
488                &mut batch.weights
489            } else {
490                &mut scratch.weights
491            };
492            target.resize(subset.num_elements_usize(), 0.0);
493            decode_into(&arrays.weights, "float32", &subset, target, options)?;
494        }
495        if let Some(subset) = prepare(roles.flags, Some(&arrays.flags), Extent::Cube)? {
496            scratch.flag_bytes.resize(subset.num_elements_usize(), 0);
497            decode_into(
498                &arrays.flags,
499                "bool",
500                &subset,
501                &mut scratch.flag_bytes,
502                options,
503            )?;
504            let weights: &[f32] = if weights_in_place {
505                &batch.weights
506            } else {
507                &scratch.weights
508            };
509            let flag = |byte: u8, weight: f32| match byte {
510                0 => Ok(!weight.is_finite()),
511                1 => Ok(true),
512                _ => Err(invalid(format!("FLAG byte {byte} is not a boolean"))),
513            };
514            let cells = scratch
515                .flag_bytes
516                .chunks_exact(stored)
517                .zip(weights.chunks_exact(stored));
518            match keep {
519                None => {
520                    for (&byte, &weight) in scratch.flag_bytes.iter().zip(weights) {
521                        batch.flags.push(flag(byte, weight)?);
522                    }
523                }
524                Some(keep) if self.resolved.flags_from_all => {
525                    for (bytes, weights) in cells {
526                        let mut any = false;
527                        for (&byte, &weight) in bytes.iter().zip(weights) {
528                            any |= flag(byte, weight)?;
529                        }
530                        batch.flags.extend(std::iter::repeat_n(any, keep.len()));
531                    }
532                }
533                Some(keep) => {
534                    for (bytes, weights) in cells {
535                        for &product in keep {
536                            batch.flags.push(flag(bytes[product], weights[product])?);
537                        }
538                    }
539                }
540            }
541            counters.copied_elements += scratch.flag_bytes.len();
542        }
543        if let (true, Some(keep)) = (roles.weights, keep) {
544            gather(&scratch.weights, stored, keep, &mut batch.weights);
545            counters.copied_elements += batch.weights.len();
546        }
547        for (wanted, array, extent, target) in [
548            (roles.uvw, Some(&arrays.uvw), Extent::Uvw, &mut batch.uvw),
549            (
550                roles.time_centroid,
551                dataset.centroid.as_ref(),
552                Extent::TimeBaseline,
553                &mut batch.centroid_offsets_s,
554            ),
555            (
556                roles.exposure,
557                dataset.exposure.as_ref(),
558                Extent::TimeBaseline,
559                &mut batch.exposure_s,
560            ),
561        ] {
562            if let Some(subset) = prepare(wanted, array, extent)? {
563                target.resize(subset.num_elements_usize(), 0.0);
564                decode_into(
565                    array.expect("prepared"),
566                    "float64",
567                    &subset,
568                    target,
569                    options,
570                )?;
571            }
572        }
573        Ok(())
574    }
575}

577impl VisibilityCursor for Msv4Cursor<'_> {
578    fn catalogue(&self) -> &Arc<Catalogue> {
579        &self.observation.catalogue
580    }
581
582    fn estimate(&self) -> &PartitionEstimate {
583        &self.estimate
584    }
585
586    fn step(&mut self) -> Step {
587        if let Some(key) = self.groups.as_mut().and_then(GroupIndex::pop_end) {
588            return Step::GroupEnd(key);
589        }
590        if self.resolved.times.next_from(self.next_time).is_none() {
591            Step::End
592        } else {
593            Step::Batch
594        }
595    }
596
597    fn fill(&mut self, buffers: &mut BatchBuffers) -> Result<(), ObservationError> {
598        let Some(rows) = self
599            .resolved
600            .times
601            .block_at(self.next_time, self.estimate.times_per_batch)
602        else {
603            return Err(ObservationError::InvalidRequest {
604                reason: "fill after the end of the traversal".to_string(),
605            });
606        };
607        self.fill_batch(buffers, rows.clone())?;
608        self.next_time = rows.end;
609        self.counters.batches += 1;
610        Ok(())
611    }
612
613    fn counters(&self) -> IoCounters {
614        self.counters
615    }
