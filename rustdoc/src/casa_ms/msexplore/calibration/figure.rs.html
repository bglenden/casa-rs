19pub(super) fn prepare(
20    task: &TaskInvocation<'_>,
21    payload: &Path,
22    destination: Option<PathBuf>,
23    show: bool,
24    overwrite: bool,
25    panels: &PlotPanels,
26    header: impl Fn(&[Panel]) -> Vec<String>,
27) -> Result<Vec<PlotFigure>, TaskFailure> {
28    let pages = panels.pages().collect::<Vec<_>>();
29    let destinations = pages
30        .iter()
31        .enumerate()
32        .map(|(page, page_panels)| {
33            destination
34                .as_deref()
35                .map(|path| panels.filename(path, page, page_panels))
36        })
37        .collect::<Vec<_>>();
38    let output_paths = destinations
39        .iter()
40        .map(|path| ("plotfile", path.as_deref()))
41        .collect::<Vec<_>>();
42    crate::msexplore::task_contract::refuse_unwritable_outputs(
43        &output_paths,
44        overwrite,
45        "set overwrite=True to replace it",
46    )
47    .map_err(TaskFailure::validation)?;
48    let mut payload = PlotPayloadWriter::open(payload).map_err(write_failure)?;
49    let ranges = AxisRanges::new(task, panels)?;
50    let title = task.text("title")?.unwrap_or_default();
51    let mut figures = Vec::new();
52    for (page, (source, destination)) in pages.iter().zip(destinations).enumerate() {
53        let content = source
54            .iter()
55            .enumerate()
56            .map(|(index, panel)| make_panel(task, &mut payload, panels, panel, index, &ranges))
57            .collect::<Result<Vec<_>, _>>()?;
58        let [rows, columns] = panels.grid;
59        let mut document = PlotDocument {
60            id: format!("plotms-calibration-{page}"),
61            title: title.clone(),
62            subtitle: String::new(),
63            header_lines: header(source),
64            axes: Vec::new(),
65            layers: Vec::new(),
66            annotations: Vec::new(),
67            panels: content,
68            show_legend: task.flag("showlegend")?,
69            layout: Some(PlotLayout {
70                rows,
71                columns,
72                first_panel: None,
73                aspect: None,
74                margins: None,
75                font_size: None,
76            }),
77        };
78        if rows == 1 && columns == 1 && panels.axis_name().is_none() {
79            let panel = document.panels.pop().expect("one panel");
80            document.axes = panel.axes;
81            document.layers = panel.layers;
82            document.annotations = panel.annotations;
83        }
84        figures.push(PlotFigure {
85            output_file: destination,
86            show: show && page == 0,
87            document,
88        });
89    }
90    let bytes = payload.finish().map_err(write_failure)?;
91    for figure in &figures {
92        figure
93            .document
94            .validate_payload(bytes)
95            .map_err(|error| TaskFailure::execution(error.to_string()))?;
96    }
97    Ok(figures)
98}

100fn make_panel(
101    task: &TaskInvocation<'_>,
102    payload: &mut PlotPayloadWriter,
103    panels: &PlotPanels,
104    panel: &Panel,
105    index: usize,
106    ranges: &AxisRanges<'_>,
107) -> Result<PlotDocumentPanel, TaskFailure> {
108    let series = &panel.series;
109    let xlabel = task.text("xlabel")?.filter(|label| !label.is_empty());
110    let ylabel = task.text("ylabel")?.filter(|label| !label.is_empty());
111    let manual = plot_limits(task.value("plotrange"))?;
112    let color = task
113        .text("coloraxis")?
114        .filter(|value| !value.is_empty())
115        .map(|value| Axis::parse(&value))
116        .transpose()
117        .map_err(TaskFailure::validation)?;
118    let on_right = |index| right_side(&ranges.locations, index);
119    let x = &series[0].x;
120    let [x_range, left_range, right_range] = ranges.for_panel(panel);
121    let mut axes = vec![axis(
122        "x",
123        *x,
124        xlabel.as_deref(),
125        x_range,
126        [manual[0], manual[1]],
127    )];
128    for right in [false, true] {
129        let selected = series
130            .iter()
131            .enumerate()
132            .filter(|(index, _)| on_right(*index) == right)
133            .map(|(_, series)| series)
134            .collect::<Vec<_>>();
135        let Some(first) = selected.first() else {
136            continue;
137        };
138        let mut ordinate = axis(
139            ordinate_id(right),
140            first.y,
141            ylabel.as_deref(),
142            if right { right_range } else { left_range },
143            [manual[2], manual[3]],
144        );
145        if selected.len() > 1 && ylabel.is_none() {
146            ordinate.label = selected
147                .iter()
148                .map(|line| line.y.label())
149                .collect::<Vec<_>>()
150                .join(", ");
151        }
152        ordinate.draws_on_trailing_edge = right;
153        axes.push(ordinate);
154    }
155    let connect = task
156        .text("xconnector")?
157        .is_some_and(|name| name == "line" || name == "step");
158    let connect_times = task.flag("timeconnector")?;
159    let mut layers = Vec::new();
160    for (id, series) in series.iter().enumerate() {
161        let yid = ordinate_id(on_right(id));
162        let mut groups = BTreeMap::<(bool, String, u64), Vec<&Point>>::new();
163        for point in series.points.iter().filter(|point| {
164            point.x.is_finite()
165                && point.y.is_finite()
166                && (!point.flagged || shows_flagged(ranges.flagged_symbols, id))
167        }) {
168            let label = color.map_or_else(
169                || series.y.label().to_string(),
170                |color| {
171                    format!(
172                        "{} {}",
173                        color.label(),
174                        color.value(&point.coordinates, Default::default(), 0.0)
175                    )
176                },
177            );
178            let epoch = if connect && !connect_times {
179                point.coordinates.time.to_bits()
180            } else {
181                0
182            };
183            groups
184                .entry((point.flagged, label, epoch))
185                .or_default()
186                .push(point);
187        }
188        for ((flagged, label, epoch), points) in groups {
189            layers.push(layer(
190                payload,
191                (index, id, yid, epoch),
192                flagged,
193                label,
194                points,
195                connect,
196            )?);
197        }
198    }
199    let colorize_overlay = if task.flag("colorizeoverlay")? {
200        color
201    } else {
202        None
203    };
204    append_atmosphere(
205        payload,
206        panel,
207        index,
208        AtmosphereStyle {
209            flagged_symbols: ranges.flagged_symbols,
210            color: colorize_overlay,
211            connect,
212        },
213        &mut axes,
214        &mut layers,
215    )?;
216    let title = panels
217        .axis_name()
218        .map_or_else(String::new, |axis| format!("{axis} {}", panel.label));
219    Ok(PlotDocumentPanel {
220        id: format!("calibration-panel-{index}"),
221        title,
222        title_font_size: None,
223        axes,
224        layers,
225        annotations: Vec::new(),
226        aspect: None,
227        frame_visible: None,
228    })
229}

231struct AtmosphereStyle<'a> {
232    flagged_symbols: Option<&'a ParameterValue>,
233    color: Option<Axis>,
234    connect: bool,
235}

237fn append_atmosphere(
238    payload: &mut PlotPayloadWriter,
239    panel: &Panel,
240    index: usize,
241    style: AtmosphereStyle<'_>,
242    axes: &mut Vec<PlotDocumentAxis>,
243    layers: &mut Vec<PlotDocumentLayer>,
244) -> Result<(), TaskFailure> {
245    let Some(first) = panel.overlay.first() else {
246        return Ok(());
247    };
248    let mut bounds = [f64::INFINITY, f64::NEG_INFINITY];
249    for (overlay_id, series) in panel.overlay.iter().enumerate() {
250        let series_index = panel.series.len() + overlay_id;
251        let show_flagged = shows_flagged(style.flagged_symbols, series_index);
252        let mut groups = BTreeMap::<String, Vec<&Point>>::new();
253        for point in series.points.iter().filter(|point| {
254            point.x.is_finite() && point.y.is_finite() && (show_flagged || !point.flagged)
255        }) {
256            bounds = [bounds[0].min(point.y), bounds[1].max(point.y)];
257            let group = style.color.map_or_else(
258                || series.y.label().to_string(),
259                |axis| {
260                    format!(
261                        "{} {}",
262                        axis.label(),
263                        axis.value(&point.coordinates, Default::default(), 0.0)
264                    )
265                },
266            );
267            groups.entry(group).or_default().push(point);
268        }
269        for (group, points) in groups {
270            let mut curve = layer(
271                payload,
272                (index, series_index, "atmosphere", 0),
273                false,
274                group,
275                points,
276                style.connect,
277            )?;
278            curve.title = series.y.label().into();
279            curve.provenance_summary = "Atmospheric model over the selected channels".into();
280            if let Some(layer_style) = &mut curve.style {
281                layer_style.color = style.color.is_none().then(|| {
282                    crate::msexplore::plotms_figure::default_overlay_color(overlay_id).into()
283                });
284            }
285            layers.push(curve);
286        }
287    }
288    let mut overlay_axis = axis("atmosphere", first.y, None, bounds, [0.0; 2]);
289    overlay_axis.draws_on_trailing_edge = true;
290    axes.push(overlay_axis);
291    Ok(())
292}

294fn ordinate_id(right: bool) -> &'static str {
295    if right { "right" } else { "left" }
296}

298fn layer(
299    payload: &mut PlotPayloadWriter,
300    (panel, index, ordinate, epoch): (usize, usize, &str, u64),
301    flagged: bool,
302    label: String,
303    points: Vec<&Point>,
304    connect: bool,
305) -> Result<PlotDocumentLayer, TaskFailure> {
306    let x = payload
307        .values(points.iter().map(|point| point.x))
308        .map_err(write_failure)?;
309    let y = payload
310        .values(points.iter().map(|point| point.y))
311        .map_err(write_failure)?;
312    let kind = if connect {
313        PlotLayerKind::Line
314    } else {
315        PlotLayerKind::Scatter
316    };
317    Ok(PlotDocumentLayer {
318        id: format!("{index}-{panel}-{ordinate}-{flagged}-{epoch}-{label}"),
319        title: label.clone(),
320        kind,
321        x_axis_id: "x".into(),
322        y_axis_id: ordinate.into(),
323        x_values: Vec::new(),
324        y_values: Vec::new(),
325        point_labels: Vec::new(),
326        point_symbol_sizes: Vec::new(),
327        interval_x_start: Vec::new(),
328        interval_x_end: Vec::new(),
329        interval_y: Vec::new(),
330        interval_height: Vec::new(),
331        interval_labels: Vec::new(),
332        provenance: Vec::new(),
333        color_group: if flagged { "flagged".into() } else { label },
334        symbol_size: 2.0,
335        line_width: 1.0,
336        opacity: 1.0,
337        source_sample_count: points.len() as u64,
338        payload_strategy: PlotPayloadStrategy::BinaryArrays,
339        provenance_summary: "Calibration table solutions".into(),
340        data: Some(PlotLayerData::Points { x, y }),
341        style: Some(PlotLayerStyle {
342            color: flagged.then(|| "#ff0000".into()),
343            marker: Some("o".into()),
344            line_style: connect.then(|| "-".into()),
345        }),
346    })
347}

349struct AxisRanges<'a> {
350    locations: Vec<&'a str>,
351    flagged_symbols: Option<&'a ParameterValue>,
352    global: [[f64; 2]; 3],
353    local_x: bool,
354    local_y: bool,
355}
356
357impl<'a> AxisRanges<'a> {
358    fn new(task: &'a TaskInvocation<'_>, layout: &PlotPanels) -> Result<Self, TaskFailure> {
359        let locations =
360            super::axis_names(task.value("yaxislocation")).map_err(TaskFailure::validation)?;
361        if locations
362            .iter()
363            .any(|location| !matches!(*location, "" | "left" | "right"))
364        {
365            return Err(TaskFailure::validation(
366                "yaxislocation accepts left or right",
367            ));
368        }
369        let flagged_symbols = task.value("customflaggedsymbol");
370        Ok(Self {
371            global: data_ranges(&layout.panels, &locations, flagged_symbols),
372            locations,
373            flagged_symbols,
374            local_x: task.flag("xselfscale")? && !task.flag("xsharedaxis")?,
375            local_y: task.flag("yselfscale")? && !task.flag("ysharedaxis")?,
376        })
377    }
378
379    fn for_panel(&self, panel: &Panel) -> [[f64; 2]; 3] {
380        let mut ranges = self.global;
381        if self.local_x || self.local_y {
382            let local = data_ranges(
383                std::slice::from_ref(panel),
384                &self.locations,
385                self.flagged_symbols,
386            );
387            if self.local_x {
388                ranges[0] = local[0];
389            }
390            if self.local_y {
391                ranges[1..].copy_from_slice(&local[1..]);
392            }
393        }
394        ranges
395    }
396}

398fn right_side(locations: &[&str], index: usize) -> bool {
399    locations
400        .get(index)
401        .or_else(|| (locations.len() == 1).then(|| &locations[0]))
402        == Some(&"right")
403}

405pub(in crate::msexplore) fn shows_flagged(symbols: Option<&ParameterValue>, index: usize) -> bool {
406    let enabled = match symbols {
407        Some(ParameterValue::Array(values)) => values.get(index),
408        other => other,
409    };
410    matches!(enabled, Some(ParameterValue::Bool(true)))
411}
412
413fn data_ranges(
414    panels: &[Panel],
415    locations: &[&str],
416    flagged_symbols: Option<&ParameterValue>,
417) -> [[f64; 2]; 3] {
418    let mut ranges = [[f64::INFINITY, f64::NEG_INFINITY]; 3];
419    let add = |range: &mut [f64; 2], value: f64| {
420        if value.is_finite() {
421            *range = [range[0].min(value), range[1].max(value)];
422        }
423    };
424    for panel in panels {
425        for (index, series) in panel.series.iter().enumerate() {
426            let y_index = if right_side(locations, index) { 2 } else { 1 };
427            let show_flags = shows_flagged(flagged_symbols, index);
428            for point in &series.points {
429                if show_flags || !point.flagged {
430                    add(&mut ranges[0], point.x);
431                    add(&mut ranges[y_index], point.y);
432                }
433            }
434        }
435    }
436    ranges
437}

439fn axis(
440    id: &str,
441    axis: Axis,
442    label: Option<&str>,
443    [mut lower, mut upper]: [f64; 2],
444    limits: [f64; 2],
445) -> PlotDocumentAxis {
446    if lower > upper {
447        [lower, upper] = [0.0, 1.0];
448    }
449    let padding = if lower == upper {
450        if axis == Axis::Time {
451            0.5
452        } else {
453            lower.abs().mul_add(0.01, 0.5)
454        }
455    } else {
456        0.05 * (upper - lower)
457    };
458    if limits != [0.0, 0.0] {
459        [lower, upper] = limits;
460    } else {
461        lower -= padding;
462        upper += padding;
463    }
464    let unit = axis.unit();
465    let label = label.map_or_else(
466        || {
467            if unit.is_empty() {
468                axis.label().into()
469            } else {
470                format!("{} ({unit})", axis.label())
471            }
472        },
473        str::to_owned,
474    );
475    PlotDocumentAxis {
476        id: id.into(),
477        label,
478        unit: unit.into(),
479        lower,
480        upper,
481        scale: PlotAxisScale::Linear,
482        lane_labels: Vec::new(),
483        draws_on_trailing_edge: false,
484    }
485}

487pub(in crate::msexplore) fn plot_limits(
488    value: Option<&ParameterValue>,
489) -> Result<[f64; 4], TaskFailure> {
490    let Some(ParameterValue::Array(values)) = value else {
491        return Ok([0.0; 4]);
492    };
493    if values.is_empty() {
494        return Ok([0.0; 4]);
495    }
496    let limits = values
497        .iter()
498        .map(|value| match value {
499            ParameterValue::Float(value) if value.is_finite() => Ok(*value),
500            ParameterValue::Integer(value) => Ok(*value as f64),
501            _ => Err(TaskFailure::validation(
502                "plotrange requires finite axis bounds",
503            )),
504        })
505        .collect::<Result<Vec<_>, _>>()?;
506    limits
507        .try_into()
508        .map_err(|_| TaskFailure::validation("plotrange requires four axis bounds"))
509}
