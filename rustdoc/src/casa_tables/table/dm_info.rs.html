20fn plain_table_dat(
21    start: &std::path::Path,
22) -> Result<Option<(PathBuf, TableDatContents)>, TableError> {
23    let mut path = start.to_path_buf();
24    let mut seen = HashSet::new();
25    for _ in 0..32 {
26        let real =
27            std::fs::canonicalize(&path).map_err(|error| TableError::Storage(error.to_string()))?;
28        if !seen.insert(real) {
29            return Err(TableError::Storage("reference table cycle".into()));
30        }
31        match read_table_dat_dispatch(&path.join(TABLE_CONTROL_FILE))? {
32            TableDatResult::Plain(table_dat) => return Ok(Some((path, table_dat))),
33            TableDatResult::Ref(reference) => {
34                path = add_directory_for_link(&reference.parent_relative_path, &path);
35            }
36            TableDatResult::Concat(_) => return Ok(None),
37        }
38    }
39    Err(TableError::Storage(
40        "reference table chain is too deep".into(),
41    ))
42}

44impl Table {
45    /// Return the `SPEC` record of data manager `seq_nr` on disk, or `None` for
46    /// memory, concatenated, or unimplemented storage manager types.
47    ///
48    /// # C++ equivalent
49    ///
50    /// `DataManager::dataManagerSpec()` exposed through `Table::dataManagerInfo()`.
51    pub fn data_manager_spec(&self, seq_nr: u32) -> Result<Option<RecordValue>, TableError> {
52        if self.is_memory() {
53            return Ok(self
54                .data_manager_info()
55                .iter()
56                .find(|manager| manager.seq_nr == seq_nr && manager.dm_type == "MemoryStMan")
57                .map(|_| RecordValue::default()));
58        }
59        let Some(source) = self.source_path.as_ref() else {
60            return Ok(None);
61        };
62        let Some((root_path, table_dat)) = plain_table_dat(source)? else {
63            return Ok(None);
64        };
65        let Some(dm) = table_dat
66            .column_set
67            .data_managers
68            .iter()
69            .find(|entry| entry.seq_nr == seq_nr)
70        else {
71            return Ok(None);
72        };
73        match dm.type_name.as_str() {
74            "StManAipsIO" | "MemoryStMan" => Ok(Some(RecordValue::default())),
75            "StandardStMan" => ssm_spec(&root_path, seq_nr)
76                .map(Some)
77                .map_err(TableError::from),
78            "IncrementalStMan" => ism_spec(&root_path, seq_nr)
79                .map(Some)
80                .map_err(TableError::from),
81            "TiledColumnStMan" | "TiledShapeStMan" | "TiledCellStMan" | "TiledDataStMan" => {
82                tiled_spec(&root_path, seq_nr, &table_dat)
83                    .map(Some)
84                    .map_err(TableError::from)
85            }
86            _ => Ok(None),
87        }
88    }
89
90    /// Return full-save options preserving the layout when the table is a
91    /// single `StManAipsIO` or single `StandardStMan` group whose manager name
92    /// the writer reproduces.
93    pub fn single_rewrite_options(&self) -> Result<Option<TableOptions>, TableError> {
94        if let Some(options) = self.single_stman_aipsio_options()? {
95            return Ok(Some(options));
96        }
97        let Some(path) = self.source_path.as_ref() else {
98            return Ok(None);
99        };
100        let TableDatResult::Plain(table_dat) =
101            read_table_dat_dispatch(&path.join(TABLE_CONTROL_FILE))?
102        else {
103            return Ok(None);
104        };
105        let [manager] = table_dat.column_set.data_managers.as_slice() else {
106            return Ok(None);
107        };
108        let groups: HashSet<&str> = table_dat
109            .table_desc
110            .columns
111            .iter()
112            .map(|column| column.data_manager_group.as_str())
113            .collect();
114        let (Some(group), 1) = (groups.iter().next(), groups.len()) else {
115            return Ok(None);
116        };
117        if manager.type_name != "StandardStMan"
118            || manager.seq_nr != 0
119            || table_dat
120                .table_desc
121                .columns
122                .iter()
123                .any(|column| column.data_manager_type != "StandardStMan")
124        {
125            return Ok(None);
126        }
127        if dm_name_from_blob(&manager.data).is_none() {
128            return Ok(None);
129        }
130        let data_path = path.join(format!(
131            "{}{}",
132            crate::storage::TABLE_DATA_FILE_PREFIX,
133            manager.seq_nr
134        ));
135        let casacore_default = crate::storage::standard_stman::ssm_has_small_buckets(&data_path)?;
136        let format = if table_dat.big_endian {
137            EndianFormat::BigEndian
138        } else {
139            EndianFormat::LittleEndian
140        };
141        Ok(Some(
142            TableOptions::new(path)
143                .with_data_manager(DataManagerKind::StandardStMan)
144                .with_data_manager_group(*group)
145                .with_casacore_default_ssm(casacore_default)
146                .with_endian_format(format),
147        ))
148    }
149}

151#[cfg(test)]
152mod tests {
153    use casa_core::{PrimitiveType, RecordField, ScalarValue, Value};
154    use tempfile::tempdir;
155
156    use crate::{ColumnSchema, TableSchema};
157
158    use super::*;
159
160    #[test]
161    fn single_rewrite_preserves_32k_bucket_size_and_group_name() {
162        let dir = tempdir().expect("tempdir");
163        let path = dir.path().join("big_ssm.table");
164        let schema = TableSchema::new(vec![ColumnSchema::scalar("id", PrimitiveType::Int32)])
165            .expect("schema");
166        let mut table = Table::with_schema(schema);
167        table
168            .add_row(RecordValue::new(vec![RecordField::new(
169                "id",
170                Value::Scalar(ScalarValue::Int32(7)),
171            )]))
172            .expect("add row");
173        table
174            .save(
175                TableOptions::new(&path)
176                    .with_data_manager(DataManagerKind::StandardStMan)
177                    .with_data_manager_group("StandardStMan"),
178            )
179            .expect("save");
180
181        let mut reopened = Table::open(TableOptions::new(&path)).expect("open");
182        let spec_before = reopened
183            .data_manager_spec(0)
184            .expect("spec")
185            .expect("some spec");
186        assert_eq!(
187            spec_before.get("BUCKETSIZE"),
188            Some(&Value::Scalar(ScalarValue::Int32(32_768)))
189        );
190
191        let opts = reopened
192            .single_rewrite_options()
193            .expect("rewrite opts")
194            .expect("single group");
195        assert!(!opts.casacore_default_ssm());
196        reopened
197            .add_row(RecordValue::new(vec![RecordField::new(
198                "id",
199                Value::Scalar(ScalarValue::Int32(9)),
200            )]))
201            .expect("add row 2");
202        reopened.save(opts).expect("resave");
203
204        let again = Table::open(TableOptions::new(&path)).expect("reopen");
205        let dm = again.data_manager_info();
206        assert_eq!(dm.len(), 1);
207        assert_eq!(dm[0].name, "StandardStMan");
208        let spec_after = again
209            .data_manager_spec(0)
210            .expect("spec after")
211            .expect("some spec");
212        assert_eq!(
213            spec_after.get("BUCKETSIZE"),
214            Some(&Value::Scalar(ScalarValue::Int32(32_768)))
215        );
216    }
