35/// The compiled spectral axis and the source channels it reads.
36pub(super) struct PreparedSpectralAxis<'a> {
37    /// Each store's source channels the axis reads, in store order.
38    pub(super) selected_source_channels: Cow<'a, [WindowChannels]>,
39    /// How each store's windows read beyond their selected channels at
40    /// every time ([`WindowRead`]), in store order.
41    pub(super) window_reads: Vec<WindowReads>,
42    pub(super) source_frame: FrequencyFrame,
43    pub(super) output_frequency_reference: FrequencyRef,
44    pub(super) output_frame: FrequencyFrame,
45    pub(super) anchor: SpectralFrameAnchor,
46    pub(super) wcs: SpectralWcs,
47    source_lsrk_grid: Option<SourceLsrkGrid>,
48    pub(super) rest_frequency: RestFrequency,
49    /// The spectral axis the image header records: the gridded axis, but
50    /// for an MFS image's frame and rest frequencies.
51    pub(super) image: ImageSpectralCoordinate,
52    pub(super) doppler: DopplerConvention,
53    pub(super) sampling: SpectralSamplingLaw,
54    pub(super) basis: ReconstructionBasis,
55    pub(super) output_channels: usize,
56    /// Centre of the first output channel.
57    pub(super) reference_frequency_hz: f64,
58    /// Channel increment; a continuum plane's full band.
59    pub(super) increment_hz: f64,
60}

62impl PreparedSpectralAxis<'_> {
63    /// The model's spectral coordinate of this axis.
64    pub(super) fn coordinate(&self) -> SpectralCoordinateSpec {
65        SpectralCoordinateSpec::new(
66            self.source_frame,
67            self.output_frame,
68            self.anchor,
69            self.wcs.clone(),
70            self.rest_frequency,
71            self.doppler,
72        )
73        .with_source_lsrk_grid(self.source_lsrk_grid)
74    }
75
76    /// The source channels the axis reads from surveyed window `spw_id` of
77    /// store `store`.
78    pub(super) fn window_channels(&self, store: usize, spw_id: usize) -> &[usize] {
79        self.selected_source_channels[store]
80            .get(&spw_id)
81            .expect("the spectral axis selects channels of every surveyed window")
82    }
83}

148/// The request's spectral axis over the surveyed selection of `stores`,
149/// laid on store `reference`'s metadata (CASA's `msobj`): a continuum
150/// plane over `continuum`, the band of every store, or a cube on the
151/// reference store's window that reads each store's window where it reaches
152/// the cube.
153pub(super) fn prepare_axis<'a>(
154    request: &ImagingRequest,
155    (stores, reference): (&[SurveyedStore], usize),
156    continuum: Option<&'a ContinuumBand>,
157    frame: &FrameContext<'_>,
158    moving_rest_frame: Option<&MeasFrame>,
159    continuum_reference_hz: Option<f64>,
160) -> Result<PreparedSpectralAxis<'a>, PrepareError> {
161    let store = &stores[reference];
162    let (msv2, catalogue) = (
163        store
164            .msv2()
165            .map(|observation| observation.measurement_set()),
166        store.catalogue(),
167    );
168    let survey = &store.survey;
169    let source_reference = survey.source_frequency_reference;
170    let law = match request.specmode {
171        SpecMode::Mfs => continuum_law(
172            request,
173            (msv2, catalogue),
174            survey,
175            continuum.expect("prepare spans the band of every continuum run"),
176            continuum_reference_hz,
177        )?,
178        SpecMode::Cube | SpecMode::Cubedata => cube_law(request, (stores, reference), frame)?,
179        SpecMode::Cubesource => {
180            super::stores::one_store(stores, "a source-frame cube (specmode cubesource)")?;
181            let frame_with_velocity = moving_rest_frame
182                .expect("prepare resolves the moving rest frame of every source-frame cube");
183            source_frame_cube_law(request, store, frame, frame_with_velocity)?
184        }
185    };
186    let source_frame = imaging_frequency_frame(source_reference)?;
187    let output_frame = imaging_frequency_frame(law.output_frequency_reference)?;
188    Ok(PreparedSpectralAxis {
189        selected_source_channels: law.selected_source_channels,
190        window_reads: law.window_reads,
191        source_frame,
192        output_frequency_reference: law.output_frequency_reference,
193        output_frame,
194        anchor: spectral_frame_anchor(source_frame, output_frame, request.specmode, frame)?,
195        wcs: SpectralWcs::Linear {
196            channels: law.output_channels,
197            reference_pixel: 0.0,
198            reference_frequency_hz: law.reference_frequency_hz,
199            increment_hz: law.increment_hz,
200        },
201        source_lsrk_grid: law.source_lsrk_grid,
202        rest_frequency: law.rest_frequency,
203        // CASA lists tclean's further rest frequencies after the image's.
204        image: ImageSpectralCoordinate {
205            additional_rest_frequencies_hz: request.restfreq.iter().skip(1).copied().collect(),
206            ..law.image
207        },
208        doppler: law.doppler,
209        sampling: law.sampling,
210        basis: law.basis,
211        output_channels: law.output_channels,
212        reference_frequency_hz: law.reference_frequency_hz,
213        increment_hz: law.increment_hz,
214    })
215}

217/// One plane over the selected band's envelope over every store (`band`),
218/// gridded in LSRK as CASA's `FTMachine::matchChannel` grids it whatever
219/// the image's `outframe`. Its header records the envelope in `outframe`
220/// (`buildCoordinateSystem`'s `MSUtil::getFreqRangeInSpw`) and `restfreq`'s
221/// first rest frequency, else the reference store's (`survey`'s) first
222/// visibility buffer's source line ([`continuum_rest_frequency_hz`] on an
223/// MSv2, whose `SOURCE` and `DOPPLER` tables CASA reads; the catalogue's
224/// first line on an MSv4), else the recorded band's midpoint (`getImFreq`).
225fn continuum_law<'a>(
226    request: &ImagingRequest,
227    (msv2, catalogue): (Option<&MeasurementSet>, &Catalogue),
228    survey: &Survey,
229    band: &'a ContinuumBand,
230    continuum_reference_hz: Option<f64>,
231) -> Result<AxisLaw<'a>, PrepareError> {
232    let (channels, envelope, recorded) = (&band.channels, &band.gridded, &band.recorded);
233    let [lower_hz, upper_hz] = envelope.edges_hz();
234    let [recorded_lower_hz, recorded_upper_hz] = recorded.edges_hz();
235    // `getImFreq`: a given reffreq is the reference at pixel 0, with the
236    // band widened by twice its offset from the band's centre (CAS-9518).
237    let reference = |midpoint_hz: f64, width_hz: f64| match continuum_reference_hz {
238        Some(reference_hz) => (
239            reference_hz,
240            width_hz + 2.0 * (reference_hz - midpoint_hz).abs(),
241        ),
242        None => (midpoint_hz, width_hz),
243    };
244    if continuum_reference_hz.is_some() && request.outframe != FrequencyRef::LSRK {
245        return Err(PrepareError::ReferenceFrequencyOutsideLsrk {
246            outframe: request.outframe,
247        });
248    }
249    let (reference_frequency_hz, increment_hz) =
250        reference(envelope.midpoint_hz(), upper_hz - lower_hz);
251    let (image_reference_hz, image_increment_hz) = reference(
252        recorded.midpoint_hz(),
253        recorded_upper_hz - recorded_lower_hz,
254    );
255    let field_id = survey.first_visibility_field_id;
256    let spw_id = survey.spectral_windows[0].spw_id;
257    let image_rest_frequency_hz = match (request.restfreq.first(), msv2) {
258        (Some(&hertz), _) => hertz,
259        (None, Some(ms)) => {
260            continuum_rest_frequency_hz(ms, field_id, spw_id, recorded.midpoint_hz())?
261        }
262        (None, None) => catalogue
263            .source_records(
264                FieldId(u32::try_from(field_id).expect("a FIELD row is an u32 id")),
265                WindowId(u32::try_from(spw_id).expect("a window id fits u32")),
266            )
267            .first()
268            .and_then(|record| record.rest_frequencies.as_ref()?.first())
269            .map(casa_types::measures::frequency::MFrequency::hz)
270            .filter(|value| *value != 0.0)
271            .unwrap_or(recorded.midpoint_hz()),
272    };
273    Ok(AxisLaw {
274        selected_source_channels: continuum_channels(channels),
275        window_reads: vec![WindowReads::new(); channels.len()],
276        output_frequency_reference: FrequencyRef::LSRK,
277        source_lsrk_grid: None,
278        reference_frequency_hz,
279        increment_hz,
280        output_channels: 1,
281        rest_frequency: RestFrequency::NotApplicable,
282        image: ImageSpectralCoordinate {
283            frequency_reference: request.outframe,
284            reference_frequency_hz: image_reference_hz,
285            increment_hz: image_increment_hz,
286            rest_frequency_hz: image_rest_frequency_hz,
287            additional_rest_frequencies_hz: Vec::new(),
288            undefined_frame: false,
289        },
290        doppler: DopplerConvention::NotApplicable,
291        sampling: SpectralSamplingLaw::IDENTITY,
292        basis: ReconstructionBasis::Constant,
293    })
294}

296/// CASA `SynthesisParamsImage::buildCoordinateSystemCore` asks
297/// `MSDopplerUtil::dopplerInfo` for the first selected SPW and field. Its
298/// first rest frequency wins; an absent or zero value uses the MFS band's
299/// midpoint. This header label does not change the continuum/Taylor basis.
300fn continuum_rest_frequency_hz(
301    ms: &MeasurementSet,
302    field_id: usize,
303    spw_id: usize,
304    band_midpoint_hz: f64,
305) -> Result<f64, PrepareError> {
306    let source_id = ms.field()?.source_id(field_id)?;
307    let window = ms
308        .subtable(SubtableId::SpectralWindow)
309        .expect("required SPW table");
310    let doppler_id = if window
311        .schema()
312        .is_some_and(|schema| schema.contains_column("DOPPLER_ID"))
313    {
314        let scalar = window
315            .cell_accessor(spw_id, "DOPPLER_ID")
316            .map_err(casa_ms::MsError::from)?
317            .scalar()
318            .map_err(casa_ms::MsError::from)?;
319        match scalar {
320            casa_types::ScalarValue::Int32(value) => *value,
321            _ => {
322                return Err(
323                    casa_ms::MsError::InvalidInput("DOPPLER_ID must be Int32".into()).into(),
324                );
325            }
326        }
327    } else {
328        -1
329    };
330    if doppler_id >= 0 && ms.subtable(SubtableId::Doppler).is_some() {
331        return Err(casa_ms::MsError::InvalidInput(
332            "MFS rest-frequency metadata with a DOPPLER table is not supported".into(),
333        )
334        .into());
335    }
336    if ms.subtable(SubtableId::Source).is_none() {
337        return Ok(band_midpoint_hz);
338    }
339    let resolved = continuum_source_line(ms, source_id, spw_id)?;
340    Ok(resolved
341        .filter(|value| *value != 0.0)
342        .unwrap_or(band_midpoint_hz))
343}

345/// The first SOURCE line within a matching key group; -1 is a wildcard.
346/// CASA binary-searches sorted unique keys using its wildcard comparator.
347/// With multiple key groups, wildcards can select or hide a matching group;
348/// refuse that unresolved case instead of merging physical row order.
349fn continuum_source_line(
350    ms: &MeasurementSet,
351    source_id: i32,
352    spw_id: usize,
353) -> Result<Option<f64>, PrepareError> {
354    let source = ms.source()?;
355    let spw_id = i32::try_from(spw_id).expect("SPW ids are stored i32 values");
356    let mut first_key = None;
357    let mut multiple_keys = false;
358    let mut wildcard = false;
359    let mut matching_key = None;
360    let mut resolved = None;
361    for row in 0..source.row_count() {
362        let row_source = source.i32(row, "SOURCE_ID")?;
363        let row_spw = source.i32(row, "SPECTRAL_WINDOW_ID")?;
364        let key = (row_source, row_spw);
365        multiple_keys |= first_key.is_some_and(|prior| prior != key);
366        first_key.get_or_insert(key);
367        wildcard |= row_source == -1 || row_spw == -1;
368        if (row_source != source_id && row_source != -1) || (row_spw != spw_id && row_spw != -1) {
369            continue;
370        }
371        if matching_key.is_some_and(|prior| prior != key) {
372            return Err(casa_ms::MsError::InvalidInput(
373                "overlapping SOURCE key groups for MFS rest-frequency metadata are not supported"
374                    .into(),
375            )
376            .into());
377        }
378        matching_key = Some(key);
379        if resolved.is_some() {
380            continue;
381        }
382        let Some(ArrayValue::Float64(values)) = source.optional_array(row, "REST_FREQUENCY")?
383        else {
384            continue;
385        };
386        if values.is_empty() {
387            continue;
388        }
389        resolved = values.first().copied();
390    }
391    if wildcard && multiple_keys {
392        return Err(casa_ms::MsError::InvalidInput(
393            "multiple SOURCE key groups with wildcard ids for MFS rest-frequency metadata are not supported".into(),
394        ).into());
395    }
396    Ok(resolved)
397}

399/// Whether CASA tunes a `cube`'s frequency selection to the image's band
400/// (`SynthesisImagerVi2::tuneSelectData`): when the window of its first
401/// visibility buffer has 30 or more channels, or when the model is saved to
402/// `MODEL_DATA`, `SynthesisImager::tuneSelectData` narrows each window's
403/// bands to the LSRK frequencies from pixel −1.5 to `nchan + 0.5` of the
404/// image axis ([`frequency_bands::cube_bands`]); a `cubedata` or
405/// `cubesource` cube keeps the selection. The tuned band keeps every
406/// channel a plane is interpolated from, since CASA refuses planes
407/// narrower than a channel (`calcChanFreqs`), so tuning changes which
408/// channels are read but not the image: a window the axis misses is still
409/// read (its bands are kept where the image's overlaps none of them), and
410/// the density `VisImagingWeight` forms is the untuned selection's.
411/// casa-rs's visibility-domain continuum fit (`fitspw`, which CASA's
412/// tclean lacks) keeps the named channels as its output channels.
413fn tunes_selection(request: &ImagingRequest, first_window: &SourceSpectralWindow) -> bool {
414    request.specmode == SpecMode::Cube
415        && request.fitspw.is_none()
416        && (first_window.frequencies_hz.len() >= 30 || request.savemodel == SaveModel::ModelColumn)
417}

419/// CASA's cube axis over the reference store's selected windows
420/// (`buildCoordinateSystemCore` on `msobj`'s windows): one window's
421/// channels, or several windows' merged into one axis
422/// ([`casa_ms::combine_spectral_windows`]), with `axis`'s grid laid over
423/// them; and the band the `spw` selection spans over `band_surveys`, in each
424/// one's data frame ([`casa_ms::casa_selected_band_hz`]; CASA's `gdatafstart`
425/// and `gdatafend`). Every window's samples are
426/// gridded onto the one axis (`FTMachine::matchChannel`).
427fn cube_setup(
428    request: &ImagingRequest,
429    (survey, band_surveys): (&Survey, &[&Survey]),
430    frame: &FrameContext<'_>,
431    mut axis: CubeAxisConfig,
432    input_frequency_factor: f64,
433) -> Result<(CubeSpectralSetup, [f64; 2]), PrepareError> {
434    fn selected<'w>(
435        survey: &'w Survey,
436        window: &'w SourceSpectralWindow,
437    ) -> (&'w [f64], &'w [f64], Option<&'w [usize]>) {
438        (
439            window.frequencies_hz.as_slice(),
440            window.channel_widths_hz.as_slice(),
441            survey.spw_channels.get(&window.spw_id).map(Vec::as_slice),
442        )
443    }
444    let windows = &survey.spectral_windows;
445    // Name the window whose selection spans no band.
446    if let Some(window) = band_surveys
447        .iter()
448        .flat_map(|survey| {
449            survey
450                .spectral_windows
451                .iter()
452                .map(move |window| (*survey, window))
453        })
454        .find(|(survey, window)| {
455            casa_ms::casa_selected_band_hz([selected(survey, window)]).is_none()
456        })
457        .map(|(_, window)| window)
458    {
459        return Err(PrepareError::SpectralWindowCoordinates {
460            spw_id: window.spw_id,
461        });
462    }
463    // Over every MeasurementSet (`buildCoordinateSystem`'s `gdatafstart` and
464    // `gdatafend`), though the axis lies on `survey`'s channels.
465    let band_hz = casa_ms::casa_selected_band_hz(band_surveys.iter().flat_map(|survey| {
466        survey
467            .spectral_windows
468            .iter()
469            .map(move |window| selected(survey, window))
470    }))
471    .ok_or(PrepareError::NoSelectedSamples)?;
472    axis.selected_band_hz = Some(band_hz);
473    let (mut frequencies_hz, mut widths_hz) = match windows.as_slice() {
474        [window] => (
475            window.frequencies_hz.clone(),
476            window.channel_widths_hz.clone(),
477        ),
478        _ => casa_ms::combine_spectral_windows(
479            &windows
480                .iter()
481                .map(|window| {
482                    (
483                        window.frequencies_hz.as_slice(),
484                        window.channel_widths_hz.as_slice(),
485                    )
486                })
487                .collect::<Vec<_>>(),
488        )?,
489    };
490    if input_frequency_factor != 1.0 {
491        // CASA cuts the stored axis at the selected edges before shifting it.
492        if axis.start.is_none()
493            && let Some(cut) = casa_ms::spectral_selection::casa_default_start_channels(
494                &frequencies_hz,
495                &widths_hz,
496                band_hz,
497            )
498        {
499            frequencies_hz = frequencies_hz[cut.clone()].to_vec();
500            widths_hz = widths_hz[cut].to_vec();
501        }
502        axis.selected_band_hz = None;
503        for hz in frequencies_hz.iter_mut().chain(&mut widths_hz) {
504            *hz *= input_frequency_factor;
505        }
506    }
507    let setup = CubeSpectralSetup::for_casa_cube_axis(
508        survey.source_frequency_reference,
509        &frequencies_hz,
510        &widths_hz,
511        request.cube_output_channel_count(),
512        &axis,
513        frame.anchor_time_mjd_seconds,
514        frame.field_id,
515        Some(frame.phase.clone()),
516        frame.engine,
517    )?;
518    Ok((setup, band_hz))
519}

521/// What a cube on `setup` reads of each store, in store order: the
522/// channels of each window, and the bands of the windows whose times read
523/// only some of them ([`store_reads`]). Whether CASA tunes the selection
524/// to the image's band is decided once, from the iterator's first buffer,
525/// the first store's ([`tunes_selection`]), and applies to every store
526/// (`SynthesisImager::tuneSelectData`).
527///
528/// The reference store's windows are those the axis is laid on; another
529/// store's one window is read where it reaches the axis, as CASA's cube
530/// gridding selects each MeasurementSet's channels in the image's band.
531///
532/// # Errors
533///
534/// An axis none of the reference store's windows reaches; another store
535/// of several windows, or of another source frame than the reference
536/// store's; a frame conversion that fails.
537fn cube_store_reads(
538    request: &ImagingRequest,
539    (stores, reference): (&[SurveyedStore], usize),
540    frame: &FrameContext<'_>,
541    setup: &CubeSpectralSetup,
542) -> Result<(Vec<WindowChannels>, Vec<WindowReads>), PrepareError> {
543    let first = &stores[0].survey;
544    let first_window = first
545        .spectral_windows
546        .iter()
547        .find(|window| window.spw_id == first.first_visibility_spw_id)
548        .expect("the first visibility is in a surveyed window");
549    let tuned = tunes_selection(request, first_window);
550    let image_band = image_band(frame, setup, tuned)?;
551    let mut channels = Vec::with_capacity(stores.len());
552    let mut bands = Vec::with_capacity(stores.len());
553    for (index, store) in stores.iter().enumerate() {
554        let survey = &store.survey;
555        let field = if index == reference {
556            frame.field_id
557        } else {
558            one_window(&survey.spectral_windows, "native cube imaging")?;
559            if survey.source_frequency_reference != setup.source_freq_ref {
560                return Err(PrepareError::MixedSourceFrames);
561            }
562            usize::try_from(*survey.fields.first().expect("a surveyed store's field"))
563                .expect("a surveyed field is a FIELD row")
564        };
565        let reads = store_reads(
566            request,
567            ((store.catalogue(), &store.engine), survey, field),
568            setup,
569            (image_band, tuned),
570        )?;
571        if index == reference && !reads.reaches_axis {
572            return Err(PrepareError::NoCommonCubeChannels);
573        }
574        channels.push(reads.channels);
575        bands.push(reads.windows);
576    }
577    Ok((channels, bands))
578}

588/// The channels a cube on `setup` reads of each window of one store's
589/// `survey`, its samples' frequencies converted towards `field`: those
590/// whose samples reach the axis
591/// ([`CubeSpectralSetup::source_channel_support`]) and that the window's
592/// frequency bands, tuned to the image's band `image_band` where CASA
593/// tunes the selection ([`tunes_selection`]), select at some epoch of its
594/// samples ([`cube_bands`]); a `fitspw` fit, which has no bands, reads
595/// those `spw` names. Where the bands do not
596/// select every channel read at every epoch, the window carries them and
597/// each time reads only the channels they select then, as CASA's iterator
598/// does. For tracked cubic AW with per-plane density, retain every
599/// iterator-selected channel of a contributing window. The spacing of the
600/// first pair determines CASA's resampling branch after Doppler conversion.
601///
602/// A window none of whose channels reaches the axis is read as `spw`
603/// selects it, tuned or not, as CASA does, by the two passes that need it:
604/// its samples fall outside every plane, but `VisImagingWeight`'s density
605/// counts them, since CASA forms it over the untuned selection of the imager
606/// that defines the cube, and a final pass writes the cube model's
607/// prediction there, zero, into its `MODEL_DATA` (test_modelvis_29). The
608/// other passes skip it where their source confirms this test, at a few
609/// epochs and over the whole window's channels, for its rows
610/// ([`WindowRead::density_only`]).
611fn store_reads(
612    request: &ImagingRequest,
613    ((catalogue, engine), survey, field): ((&Catalogue, &MsCalEngine), &Survey, usize),
614    setup: &CubeSpectralSetup,
615    (image_band, tuned): (Option<[f64; 2]>, bool),
616) -> Result<StoreReads, PrepareError> {
617    let bands = cube_bands(request, (catalogue, engine), survey, image_band)?;
618    let keep_iterator_channels = tracked_aw_cube_density(
619        request,
620        survey.source_frequency_reference,
621        setup.output_freq_ref,
622    );
623    let density_bands = if whole_selection_density(request) {
624        cube_bands(request, (catalogue, engine), survey, None)?
625    } else {
626        None
627    };
628    let prediction_bands =
629        if request.savemodel == SaveModel::ModelColumn && request.predicts_final_model() {
630            cube_bands(request, (catalogue, engine), survey, None)?
631        } else {
632            None
633        };
634    let predicts = request.savemodel == SaveModel::ModelColumn && request.predicts_final_model();
635    let mut reads = StoreReads {
636        channels: WindowChannels::new(),
637        windows: WindowReads::new(),
638        reaches_axis: false,
639    };
640    for window in &survey.spectral_windows {
641        let mut channels = setup
642            .source_channel_support(
643                &window.frequencies_hz,
644                &window.channel_widths_hz,
645                survey.time_bounds_mjd_seconds,
646                field,
647                engine,
648            )?
649            .indices;
650        let selection = match &bands {
651            Some(bands) => bands.epoch_selection((catalogue, engine), survey, window)?,
652            None => None,
653        };
654        let explicit = survey.spw_channels.get(&window.spw_id);
655        match (&selection, explicit) {
656            (Some(selection), _) => channels.retain(|channel| selection.ever(*channel)),
657            (None, Some(explicit)) if !tuned => {
658                channels.retain(|channel| explicit.contains(channel));
659            }
660            _ => {}
661        }
662        if channels.is_empty() {
663            // `spw`'s channels, or the whole window when it names none.
664            channels = explicit
665                .cloned()
666                .unwrap_or_else(|| (0..window.frequencies_hz.len()).collect());
667            reads.windows.entry(window.spw_id).or_default().density_only = true;
668        } else {
669            reads.reaches_axis = true;
670            if keep_iterator_channels && let Some(selection) = &selection {
671                // FTM first compares the iterator's channel spacing with
672                // the image step. Trimming even an unused leading channel
673                // can change that comparison after frequency conversion.
674                channels = (0..window.frequencies_hz.len())
675                    .filter(|&channel| selection.ever(channel))
676                    .collect();
677            }
678            if let (Some(bands), Some(selection)) = (&bands, &selection)
679                && channels.iter().any(|channel| !selection.always(*channel))
680                && let Some(window_bands) = bands.frequency_bands(window.spw_id)
681            {
682                reads.windows.entry(window.spw_id).or_default().bands = Some(window_bands.clone());
683            }
684        }
685        if let Some(density) = &density_bands
686            && let Some(read) =
687                window_density((catalogue, engine), survey, window, density, &channels)?
688        {
689            reads.windows.entry(window.spw_id).or_default().density = Some(read);
690        }
691        if predicts {
692            let prediction = match &prediction_bands {
693                Some(bands) => {
694                    let selection = bands.epoch_selection((catalogue, engine), survey, window)?;
695                    match selection {
696                        Some(selection) => {
697                            let read = (0..window.frequencies_hz.len())
698                                .filter(|channel| selection.ever(*channel))
699                                .collect::<Vec<_>>();
700                            let by_time = (!read.iter().all(|channel| selection.always(*channel)))
701                                .then(|| bands.frequency_bands(window.spw_id).cloned())
702                                .flatten();
703                            WindowDensity {
704                                channels: read,
705                                bands: by_time,
706                            }
707                        }
708                        None => WindowDensity {
709                            channels: Vec::new(),
710                            bands: None,
711                        },
712                    }
713                }
714                None => WindowDensity {
715                    channels: explicit
716                        .cloned()
717                        .unwrap_or_else(|| (0..window.frequencies_hz.len()).collect()),
718                    bands: None,
719                },
720            };
721            reads.windows.entry(window.spw_id).or_default().prediction = Some(prediction);
722        }
723        reads.channels.insert(window.spw_id, channels);
724    }
725    Ok(reads)
726}

728/// What a density over the whole selection counts of `window`, where that
729/// isn't simply `channels`: the channels `bands`, untuned, select at some
730/// time, so a range with a stride whole and a time's channels moving with its
731/// Doppler shift; None for a window without bands.
732fn window_density(
733    (catalogue, engine): (&Catalogue, &MsCalEngine),
734    survey: &Survey,
735    window: &SourceSpectralWindow,
736    bands: &frequency_bands::CubeBands,
737    channels: &[usize],
738) -> Result<Option<WindowDensity>, PrepareError> {
739    let Some(selection) = bands.epoch_selection((catalogue, engine), survey, window)? else {
740        return Ok(None);
741    };
742    let read = (0..window.frequencies_hz.len())
743        .filter(|channel| selection.ever(*channel))
744        .collect::<Vec<_>>();
745    let by_time = (!read.iter().all(|channel| selection.always(*channel)))
746        .then(|| bands.frequency_bands(window.spw_id).cloned())
747        .flatten();
748    Ok(
749        (read != channels || by_time.is_some()).then_some(WindowDensity {
750            channels: read,
751            bands: by_time,
752        }),
753    )
754}

787/// The bands a cube's gridding selects one store's windows' channels by at
788/// each time ([`frequency_bands::cube_bands`]): a `cube`'s in `outframe`,
789/// or tuned to the image's band `image_band` where CASA tunes the
790/// selection; in the data's frame where tclean's selection frame is
791/// `Undefined` (`cubedata`) or `REST` (`cubesource`, or a `cube` whose
792/// `outframe` is `REST`), which select each named range's channels at
793/// every time, a range with a stride whole. None for a `fitspw` fit,
794/// which reads its channels as named.
795fn cube_bands(
796    request: &ImagingRequest,
797    (catalogue, engine): (&Catalogue, &MsCalEngine),
798    survey: &Survey,
799    image_band: Option<[f64; 2]>,
800) -> Result<Option<frequency_bands::CubeBands>, PrepareError> {
801    use frequency_bands::SelectionFrame;
802    if request.fitspw.is_some() {

808            SelectionFrame::Converted(request.outframe)

812        }

817/// A cube in its output frame (`cube`) or the data's (`cubedata`), laid
818/// on the windows of store `reference` (CASA's `msobj`), reading each
819/// store's channels where they reach it ([`cube_store_reads`]).
820fn cube_law(
821    request: &ImagingRequest,
822    (stores, reference): (&[SurveyedStore], usize),
823    frame: &FrameContext<'_>,
824) -> Result<AxisLaw<'static>, PrepareError> {
825    let store = &stores[reference];
826    let (catalogue, survey) = (store.catalogue(), &store.survey);
827    let band_surveys = stores.iter().map(|store| &store.survey).collect::<Vec<_>>();
828    let mut axis = cube_axis(request)?;
829    let exact_aw_grid =
830        tracked_aw_cube_density(request, survey.source_frequency_reference, axis.outframe);
831    if exact_aw_grid
832        || (request.specmode == SpecMode::Cubedata
833            && matches!(request.gridder, Gridder::Awproject(_)))
834    {
835        casa_cube_frequency(&mut axis.start, request.start.as_deref());
836        casa_cube_frequency(&mut axis.width, request.width.as_deref());
837    }
838    let (setup, band_hz) = cube_setup(request, (survey, &band_surveys), frame, axis.clone(), 1.0)?;
839    let (selected_source_channels, window_reads) =
840        cube_store_reads(request, (stores, reference), frame, &setup)?;
841    let output_channels = setup.output_channel_frequencies_hz.len();
842    let (reference_frequency_hz, increment_hz) = first_and_increment(&setup, output_channels)?;
843    let source_rest_hz =
844        source_rest_frequency(catalogue, frame.field_id, &survey.spectral_windows)?;
845    let (resolved_rest_hz, image_rest_frequency_hz) =
846        cube_rest_frequency_hz(request.restfreq.first().copied(), source_rest_hz, band_hz);
847    let (rest_frequency, doppler) = match resolved_rest_hz {
848        None => (
849            RestFrequency::NotApplicable,
850            DopplerConvention::NotApplicable,
851        ),
852        Some(hertz) => (
853            RestFrequency::Line { hertz },
854            doppler_convention(request.veltype)?,
855        ),
856    };
857    let source_lsrk_grid = if exact_aw_grid && output_channels > 1 {
858        Some(tracked_cube_grid(
859            frame,
860            reference_frequency_hz,
861            increment_hz,
862        )?)
863    } else {
864        topo_source_grid(request, &axis, &band_surveys, output_channels)
865    };
866    Ok(AxisLaw {
867        selected_source_channels: Cow::Owned(selected_source_channels),
868        window_reads,
869        output_frequency_reference: setup.output_freq_ref,
870        source_lsrk_grid,
871        reference_frequency_hz,
872        increment_hz,
873        output_channels,
874        rest_frequency,
875        image: ImageSpectralCoordinate {
876            frequency_reference: setup.output_freq_ref,
877            reference_frequency_hz,
878            increment_hz,
879            rest_frequency_hz: image_rest_frequency_hz,
880            additional_rest_frequencies_hz: Vec::new(),
881            undefined_frame: request.specmode == SpecMode::Cubedata
882                && setup.output_freq_ref != FrequencyRef::REST,
883        },
884        doppler,
885        sampling: sampling_law(setup.interpolation),
886        basis: ReconstructionBasis::ChannelLocal {
887            channels: output_channels,
888        },
889    })
890}

908/// `MSTransformRegridder::convertGridPars` reads frequency quantities
909/// through `Quantum::getValue`, which divides by the ratio of unit scales.
910/// The last bit matters when a converted centre falls between two density
911/// weights. Channel and velocity requests keep their existing axis law.
912fn casa_cube_frequency(value: &mut Option<CubeAxisValue>, text: Option<&str>) {
913    if let (Some(CubeAxisValue::FrequencyHz { hz, .. }), Some(text)) = (value, text) {
914        let quantity = text
915            .parse::<Quantity>()
916            .expect("already parsed the frequency");
917        *hz = quantity.value() / (1.0 / quantity.unit().val().factor);
918    }
919}

921/// `FTMachine` samples the image coordinate in LSRK and subtracts the
922/// first two world frequencies for its interpolation step. Retain that
923/// axis for a tracked cubic AW cube using per-channel density.
924fn tracked_cube_grid(
925    frame: &FrameContext<'_>,
926    first_hz: f64,
927    increment_hz: f64,
928) -> Result<SourceLsrkGrid, PrepareError> {
929    let reference = frame.direction.reference_direction();
930    let direction = MDirection::from_angles(
931        reference.longitude_rad(),
932        reference.latitude_rad(),
933        reference.frame().direction_ref(),
934    );
935    let anchor = frame
936        .engine
937        .spectral_frame_observatory_direction(frame.anchor_time_mjd_seconds, direction)?;
938    let conversion = PreparedFrequencyConversion::new(
939        FrequencyRef::TOPO,
940        FrequencyRef::LSRK,
941        Some(&anchor),
942        Some(&anchor),
943    )?;
944    let first_lsrk = conversion.convert_hz(first_hz);
945    let second_lsrk = conversion.convert_hz(first_hz + increment_hz);
946    Ok(SourceLsrkGrid {
947        first_hz: first_lsrk,
948        increment_hz: second_lsrk - first_lsrk,
949        reporting_conversions: Some([
950            conversion,
951            PreparedFrequencyConversion::new(
952                FrequencyRef::LSRK,
953                FrequencyRef::TOPO,
954                Some(&anchor),
955                Some(&anchor),
956            )?,
957        ]),
958        terminal_plane_outside: false,
959    })
960}

962/// Source LSRK grid for a TOPO cube whose channel-index request spans one SPW.
963/// The terminal plane's exact signed span determines coverage when its
964/// rounded centre coincides with the stored edge (#1198). The ordinary
965/// CASA coordinate construction supplies the published TOPO WCS.
966fn topo_source_grid(
967    request: &ImagingRequest,
968    axis: &CubeAxisConfig,
969    surveys: &[&Survey],
970    output_channels: usize,
971) -> Option<SourceLsrkGrid> {
972    let [survey] = surveys else { return None };
973    let [window] = survey.spectral_windows.as_slice() else {
974        return None;
975    };
976    if !matches!(request.gridder, Gridder::Standard { .. })
977        || request.weighting != Weighting::Natural
978        || axis.specmode != CubeSpecMode::Cube
979        || survey.source_frequency_reference != FrequencyRef::LSRK
980        || axis.outframe != FrequencyRef::TOPO
981        || axis.veltype != DopplerRef::RADIO
982        || axis.interpolation == CubeInterpolation::Nearest
983        || !matches!(axis.start, None | Some(CubeAxisValue::Channel(0)))
984        || !matches!(axis.width, None | Some(CubeAxisValue::Channel(1)))
985        || output_channels != window.frequencies_hz.len()
986        || survey
987            .spw_channels
988            .get(&window.spw_id)
989            .is_some_and(|channels| !channels.iter().copied().eq(0..output_channels))
990    {
991        return None;
992    }
993    source_grid_at_terminal(&window.frequencies_hz, &window.channel_widths_hz)
994}

996fn source_grid_at_terminal(frequencies: &[f64], widths: &[f64]) -> Option<SourceLsrkGrid> {
997    let [first, _, ..] = *frequencies else {
998        return None;
999    };
1000    let last = *frequencies.last()?;
1001    let width = widths.first()?.abs();
1002    if !width.is_normal() || widths.iter().any(|stored| stored.abs() != width) {
1003        return None;
1004    }
1005    let (high, low) = (first.max(last), first.min(last));
1006    let span = high - low;
1007    let steps = (frequencies.len() - 1) as f64;
1008    let step = width.copysign(last - first);
1009    let rounded_span = width * steps;
1010    // Both the executable grid and a single rounding of the ideal centre
1011    // agree at this boundary. Compare the exact spans before admitting it.
1012    if !span.is_normal() || first + step * steps != last || step.mul_add(steps, first) != last {
1013        return None;
1014    }
1015    // Rounded values preserve order when they differ. For a tie, use the
1016    // FMA product residual and the FastTwoSum subtraction residual.
1017    let terminal_plane_outside = rounded_span > span
1018        || (rounded_span == span && width.mul_add(steps, -rounded_span) > (high - span) - low);
1019    let half = step / 2.0;
1020    // A wave can begin at an interior channel. Its first interval still
1021    // gives one sample per plane, and either FineGrid path starts at first.
1022    if (first - half) + half != first
1023        || frequencies.windows(2).any(|pair| {
1024            let gap = pair[1] - pair[0];
1025            gap.signum() != step.signum() || width / gap.abs() >= 2.0
1026        })
1027    {
1028        return None;
1029    }
1030    Some(SourceLsrkGrid {
1031        first_hz: first,
1032        increment_hz: step,
1033        reporting_conversions: None,
1034        terminal_plane_outside,
1035    })
1036}

1038/// A cube in the rest frame of a moving source: regrid the shifted
1039/// data channels (`cubesource`), on `store`'s one
1040/// window, reading its channels as [`store_reads`] reads a `cubedata`
1041/// cube's.
1042///
1043/// # Errors
1044///
1045/// Several selected windows: CASA's source-frame cube over several
1046/// windows is not yet ported (#1059).
1047fn source_frame_cube_law(
1048    request: &ImagingRequest,
1049    store: &SurveyedStore,
1050    frame: &FrameContext<'_>,
1051    moving_rest_frame: &MeasFrame,
1052) -> Result<AxisLaw<'static>, PrepareError> {
1053    let (catalogue, survey) = (store.catalogue(), &store.survey);
1054    if survey.spectral_windows.len() > 1 {
1055        return Err(PrepareError::SourceFrameCubeWindows {
1056            selected: survey.spectral_windows.len(),
1057        });
1058    }
1059    let source_reference = survey.source_frequency_reference;
1060    let factor = casa_ms::convert_frequency_to_frame_with_frame(
1061        source_reference,
1062        FrequencyRef::REST,
1063        1.0,
1064        Some(moving_rest_frame),
1065    )?;
1066    let mut axis = CubeAxisConfig {
1067        specmode: CubeSpecMode::Cubedata,
1068        outframe: source_reference,
1069        ..cube_axis(request)?
1070    };
1071    casa_cube_frequency(&mut axis.start, request.start.as_deref());
1072    casa_cube_frequency(&mut axis.width, request.width.as_deref());
1073    // CASA shifts the input centres and widths before regridding.
1074    let (mut setup, _) = cube_setup(request, (survey, &[survey]), frame, axis, factor)?;
1075    let output_channels = setup.output_channel_frequencies_hz.len();
1076    let (reference_frequency_hz, increment_hz) = first_and_increment(&setup, output_channels)?;
1077    // The selection helper reads stored frequencies. Put its support
1078    // bounds in that same frame before choosing the source channels.
1079    for hz in setup
1080        .output_channel_frequencies_hz
1081        .iter_mut()
1082        .chain(&mut setup.output_channel_widths_hz)
1083    {
1084        *hz /= factor;
1085    }
1086    let reads = store_reads(
1087        request,
1088        ((catalogue, &store.engine), survey, frame.field_id),
1089        &setup,
1090        (None, false),
1091    )?;
1092    if !reads.reaches_axis {
1093        return Err(PrepareError::NoCommonCubeChannels);
1094    }
1095    let rest_frequency_hz = request
1096        .restfreq
1097        .first()
1098        .copied()
1099        .or(source_rest_frequency(
1100            catalogue,
1101            frame.field_id,
1102            &survey.spectral_windows,
1103        )?)
1104        .ok_or(PrepareError::RestFrequencyMissing)?;
1105    Ok(AxisLaw {
1106        selected_source_channels: Cow::Owned(vec![reads.channels]),
1107        window_reads: vec![reads.windows],
1108        output_frequency_reference: FrequencyRef::REST,
1109        source_lsrk_grid: None,
1110        reference_frequency_hz,
1111        increment_hz,
1112        output_channels,
1113        rest_frequency: RestFrequency::Line {
1114            hertz: rest_frequency_hz,
1115        },
1116        image: ImageSpectralCoordinate {
1117            frequency_reference: FrequencyRef::REST,
1118            reference_frequency_hz,
1119            increment_hz,
1120            rest_frequency_hz,
1121            additional_rest_frequencies_hz: Vec::new(),
1122            undefined_frame: false,
1123        },
1124        doppler: doppler_convention(request.veltype)?,
1125        sampling: sampling_law(request.interpolation),
1126        basis: ReconstructionBasis::ChannelLocal {
1127            channels: output_channels,
1128        },
1129    })
1130}

1132/// CASA's cube axis of the request: `start` (else `channel_start`) and
1133/// `width` in the request's velocity convention; [`cube_setup`] sets the
1134/// band `spw` selects.
1135fn cube_axis(request: &ImagingRequest) -> Result<CubeAxisConfig, PrepareError> {
1136    let start = match (&request.start, request.channel_start) {
1137        (Some(start), _) => Some(CubeAxisValue::parse(start, request.veltype)?),
1138        (None, Some(channel)) => Some(CubeAxisValue::Channel(
1139            i32::try_from(channel).map_err(|_| PrepareError::CubeStartChannel { channel })?,
1140        )),
1141        (None, None) => None,
1142    };
1143    let width = request
1144        .width
1145        .as_deref()
1146        .map(|width| CubeAxisValue::parse(width, request.veltype))
1147        .transpose()?;
1148    Ok(CubeAxisConfig {
1149        specmode: if request.specmode == SpecMode::Cubedata {
1150            CubeSpecMode::Cubedata
1151        } else {
1152            CubeSpecMode::Cube
1153        },
1154        outframe: request.outframe,
1155        veltype: request.veltype,
1156        interpolation: request.interpolation,
1157        rest_frequency_hz: request.restfreq.first().copied(),
1158        start,
1159        width,
1160        selected_band_hz: None,
1161    })
1162}

1164/// The first output channel's centre and the channel increment.
1165/// A one-channel cube's increment is its width.
1166fn first_and_increment(
1167    setup: &CubeSpectralSetup,
1168    output_channels: usize,
1169) -> Result<(f64, f64), PrepareError> {
1170    let first_hz = setup.output_channel_frequencies_hz[0];
1171    let increment_hz = if output_channels > 1 {
1172        setup.output_channel_frequencies_hz[1] - first_hz
1173    } else {
1174        setup.output_channel_widths_hz[0]
1175    };
1176    if !increment_hz.is_finite() || increment_hz == 0.0 {
1177        return Err(PrepareError::CubeIncrement { increment_hz });
1178    }
1179    Ok((first_hz, increment_hz))
1180}

1202/// The image's epoch, reference direction and observatory, where the
1203/// rows' frame and the output frame differ, or where a `cube` is gridded
1204/// through LSRK ([`FrequencyFrame::gridded_through_lsrk`]): CASA grids a
1205/// `cube` (not `cubedata`, whose rows keep their frequencies) in LSRK
1206/// whatever its `outframe`, so a TOPO cube of TOPO rows is fixed in LSRK at
1207/// the first time and each row is Doppler tracked from it.
1208fn spectral_frame_anchor(
1209    source_frame: FrequencyFrame,
1210    output_frame: FrequencyFrame,
1211    specmode: SpecMode,
1212    frame: &FrameContext<'_>,
1213) -> Result<SpectralFrameAnchor, PrepareError> {
1214    let tracked = specmode == SpecMode::Cube && output_frame.gridded_through_lsrk();
1215    if source_frame == output_frame && !tracked {
1216        return Ok(SpectralFrameAnchor::NotApplicable);
1217    }
1218    let [x_metres, y_metres, z_metres] = frame.engine.observatory_position().as_itrf();
1219    Ok(SpectralFrameAnchor::Conversion {
1220        // As `MEpoch::from_mjd_seconds` reads a row's time, so a row at the
1221        // first time is at the anchor's epoch exactly.
1222        epoch: Epoch::new(
1223            frame.anchor_time_mjd_seconds * (1.0 / 86_400.0),
1224            imaging_time_scale(frame.engine.time_reference())?,
1225        ),
1226        direction: frame.direction.reference_direction(),
1227        observatory_position: ItrfPosition::new(x_metres, y_metres, z_metres),
1228    })
1229}

1231pub(super) fn imaging_frequency_frame(
1232    reference: FrequencyRef,
1233) -> Result<FrequencyFrame, PrepareError> {
1234    match reference {
1235        FrequencyRef::REST => Ok(FrequencyFrame::Rest),
1236        FrequencyRef::TOPO => Ok(FrequencyFrame::Topocentric),
1237        FrequencyRef::BARY => Ok(FrequencyFrame::Barycentric),
1238        FrequencyRef::LSRK => Ok(FrequencyFrame::Lsrk),
1239        _ => Err(PrepareError::UnsupportedFrequencyFrame { reference }),
1240    }
1241}

1246        EpochRef::TAI => Ok(TimeScale::Tai),
1247        EpochRef::TT => Ok(TimeScale::Tt),
1248        EpochRef::TDB => Ok(TimeScale::Tdb),
1249        _ => Err(PrepareError::UnsupportedEpochReference { reference }),
1250    }

1253/// The rest frequency the field's source states over the selected windows
1254/// (MSv2 `SOURCE::REST_FREQUENCY`, MSv4 `LINE_REST_FREQUENCY`), if any: the
1255/// first line of each record for a selected window or for every window
1256/// ([`Catalogue::source_records`]), which must agree.
1257fn source_rest_frequency(
1258    catalogue: &Catalogue,
1259    field_id: usize,
1260    spectral_windows: &[SourceSpectralWindow],
1261) -> Result<Option<f64>, PrepareError> {
1262    let field = FieldId(u32::try_from(field_id).expect("a FIELD row is an u32 id"));
1263    let mut resolved = None;
1264    for window in spectral_windows {
1265        let window = WindowId(u32::try_from(window.spw_id).expect("a window id fits u32"));
1266        for record in catalogue.source_records(field, window) {
1267            let Some(value) = record
1268                .rest_frequencies
1269                .as_ref()
1270                .and_then(|lines| lines.first())
1271                .map(casa_types::measures::frequency::MFrequency::hz)
1272                .filter(|value| *value > 0.0)
1273            else {
1274                continue;
1275            };
1276            if resolved.is_some_and(|prior: f64| prior.to_bits() != value.to_bits()) {
1277                return Err(PrepareError::RestFrequencyConflict);
1278            }
1279            resolved = Some(value);
1280        }
1281    }
1282    Ok(resolved)
1283}

1285/// CASA's precedence for a cube's rest frequency: the request's, else the
1286/// source's; the header records the centre of the selected band `band_hz`
1287/// when neither exists (`buildCoordinateSystemCore`'s provisional rest
1288/// frequency, `(datafend + datafstart)/2`).
1289fn cube_rest_frequency_hz(
1290    explicit_hz: Option<f64>,
1291    source_hz: Option<f64>,
1292    band_hz: [f64; 2],
1293) -> (Option<f64>, f64) {
1294    let resolved = explicit_hz.or(source_hz);
1295    (
1296        resolved,
1297        resolved.unwrap_or((band_hz[1] + band_hz[0]) / 2.0),
1298    )
1299}
