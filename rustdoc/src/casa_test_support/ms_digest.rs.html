2//! The Rust side of the whole-MeasurementSet digest that C++ casacore also
3//! computes ([`MeasurementSetOracle::digest_manifest`]).
4//!
5//! The manifest covers the main table, the standard MS subtables, table and
6//! column keywords, and every row value, plus any table reachable through a
7//! `TpTable` keyword. Each table contributes compact digests rather than a
8//! textual dump, so two manifests are equal exactly when C++ and Rust read
9//! the same schema, keywords and cells. The tables are opened with
10//! `casa-tables` alone, as subtables named by the main table's keywords, so
11//! any crate can compare a MeasurementSet it wrote with what casacore reads.
12//!
13//! [`MeasurementSetOracle::digest_manifest`]: crate::ms_interop::MeasurementSetOracle::digest_manifest
14
15use std::collections::BTreeSet;

22/// The subtables every MeasurementSet has, in manifest order.
23const REQUIRED_SUBTABLES: [&str; 12] = [
24    "ANTENNA",
25    "DATA_DESCRIPTION",
26    "FEED",
27    "FIELD",
28    "FLAG_CMD",
29    "HISTORY",
30    "OBSERVATION",
31    "POINTING",
32    "POLARIZATION",
33    "PROCESSOR",
34    "SPECTRAL_WINDOW",
35    "STATE",
36];
37
38/// The optional subtables, digested when the main table names them.
39const OPTIONAL_SUBTABLES: [&str; 5] = ["DOPPLER", "FREQ_OFFSET", "SOURCE", "SYSCAL", "WEATHER"];
40
41/// The digest manifest of the MeasurementSet at `ms_path`, line for line as
42/// the C++ oracle writes it.
43///
44/// # Panics
45///
46/// A table that cannot be opened or a required subtable that is missing.
47pub fn measurement_set_manifest(ms_path: &Path) -> String {
48    let root_path = normalize_existing_path(ms_path);
49    let main = open_table(&root_path);
50    let mut visited = BTreeSet::new();
51    let mut manifest = Vec::new();
52    digest_table_recursive(
53        &main,
54        "MAIN".to_string(),
55        &root_path,
56        &mut visited,
57        &mut manifest,
58    );
59    for label in REQUIRED_SUBTABLES {
60        let table = subtable(&main, &root_path, label)
61            .unwrap_or_else(|| panic!("MeasurementSet has no {label} subtable"));
62        digest_table_recursive(
63            &table,
64            label.to_string(),
65            &root_path,
66            &mut visited,
67            &mut manifest,
68        );
69    }
70    for label in OPTIONAL_SUBTABLES {
71        if let Some(table) = subtable(&main, &root_path, label) {
72            digest_table_recursive(
73                &table,
74                label.to_string(),
75                &root_path,
76                &mut visited,
77                &mut manifest,
78            );
79        }
80    }
81
82    let mut global = StableDigest::default();
83    for line in &manifest {
84        global.write_string(line);
85    }
86    let mut out = String::new();
87    for line in &manifest {
88        out.push_str(line);
89        out.push('\n');
90    }
91    out.push_str(&format!("GLOBAL {}\n", global.hex()));
92    out
93}

95/// The table a manifest label names: `"MAIN"` or a subtable such as
96/// `"SOURCE"`, or `None` when the MeasurementSet has no such subtable.
97///
98/// # Panics
99///
100/// A table that cannot be opened.
101pub fn labelled_table(ms_path: &Path, label: &str) -> Option<Table> {
102    let root_path = normalize_existing_path(ms_path);
103    let main = open_table(&root_path);
104    if label == "MAIN" {
105        return Some(main);
106    }
107    subtable(&main, &root_path, label)
108}

110/// The digest of every row of `table`, a table of the MeasurementSet at
111/// `ms_path`, as the C++ oracle's MAIN-row benchmarks return it.
112pub fn table_rows_digest(table: &Table, ms_path: &Path) -> String {
113    let root_path = normalize_existing_path(ms_path);
114    let mut discovered_refs = BTreeSet::new();
115    digest_table_rows(table, &root_path, &mut discovered_refs).hex()
116}

118/// One row's digest, as [`MeasurementSetOracle::table_row_digest`] returns
119/// it, for a table of the MeasurementSet at `ms_path`.
120///
121/// [`MeasurementSetOracle::table_row_digest`]: crate::ms_interop::MeasurementSetOracle::table_row_digest
122pub fn table_row_digest(table: &Table, row_index: usize, ms_path: &Path) -> String {
123    let root_path = normalize_existing_path(ms_path);
124    let mut digest = StableDigest::default();
125    let mut discovered_refs = BTreeSet::new();
126    write_row_digest_for_table(
127        &mut digest,
128        table,
129        row_index,
130        &root_path,
131        &mut discovered_refs,
132    );
133    digest.hex()
134}

136/// One row's per-field manifest, as
137/// [`MeasurementSetOracle::table_row_field_manifest`] returns it: a line per
138/// column with its type, shape and digest.
139///
140/// [`MeasurementSetOracle::table_row_field_manifest`]: crate::ms_interop::MeasurementSetOracle::table_row_field_manifest
141pub fn table_row_field_manifest(table: &Table, row_index: usize, ms_path: &Path) -> String {
142    let root_path = normalize_existing_path(ms_path);
143    let owner_path = normalize_existing_path(table.path().expect("disk-backed table path"));
144    let row = table.row_accessor().row(row_index).expect("row present");
145    let mut out = String::new();
146    for name in sorted_row_columns(table, row) {
147        let defined = table
148            .is_cell_defined(row_index, &name)
149            .expect("column exists for row manifest");
150        let synthesized;
151        let value = match row.get(&name) {
152            Some(value) => value,
153            None => {
154                synthesized = synthesized_value(table, &name);
155                &synthesized
156            }
157        };
158        let mut digest = StableDigest::default();
159        let mut discovered_refs = BTreeSet::new();
160        digest.write_string(&name);
161        digest.write_string(value_type_tag(value));
162        digest.write_bool(defined);
163        out.push_str(&name);
164        out.push('\t');
165        out.push_str(value_type_tag(value));
166        if defined {
167            digest_value(
168                &mut digest,
169                value,
170                &root_path,
171                &mut discovered_refs,
172                &owner_path,
173            );
174        } else {
175            let ndim = undefined_cell_ndim(table, &name);
176            digest.write_i64(ndim);
177            out.push('\t');
178            out.push_str(&format!("undefined(ndim={ndim})"));
179        }
180        if defined && let Value::Array(array) = value {
181            let shape = array
182                .shape()
183                .iter()
184                .map(usize::to_string)
185                .collect::<Vec<_>>();
186            out.push_str(&format!("\t[{}]", shape.join(",")));
187        }
188        out.push('\t');
189        out.push_str(&digest.hex());
190        out.push('\n');
191    }
192    out
193}

195#[derive(Debug, Clone, Copy)]
196struct StableDigest {
197    a: u64,
198    b: u64,
199}
200
201impl Default for StableDigest {
202    fn default() -> Self {
203        Self {
204            a: 1_469_598_103_934_665_603,
205            b: 1_099_511_628_211,
206        }
207    }
208}
209
210impl StableDigest {
211    fn write_bytes(&mut self, bytes: &[u8]) {
212        for &byte in bytes {
213            self.a ^= u64::from(byte);
214            self.a = self.a.wrapping_mul(1_099_511_628_211);
215            self.b ^= u64::from(byte);
216            self.b = self.b.wrapping_mul(14_029_467_366_897_019_727);
217        }
218    }
219
220    fn write_u8(&mut self, value: u8) {
221        self.write_bytes(&[value]);
222    }
223
224    fn write_bool(&mut self, value: bool) {
225        self.write_u8(u8::from(value));
226    }
227
228    fn write_u32(&mut self, value: u32) {
229        self.write_bytes(&value.to_le_bytes());
230    }
231
232    fn write_u64(&mut self, value: u64) {
233        self.write_bytes(&value.to_le_bytes());
234    }
235
236    fn write_i64(&mut self, value: i64) {
237        self.write_bytes(&value.to_le_bytes());
238    }
239
240    fn write_string(&mut self, value: &str) {
241        self.write_u64(value.len() as u64);
242        self.write_bytes(value.as_bytes());
243    }
244
245    fn hex(self) -> String {
246        format!("{:016x}{:016x}", self.a, self.b)
247    }
248}

250fn open_table(path: &Path) -> Table {
251    Table::open(TableOptions::new(path))
252        .unwrap_or_else(|error| panic!("open {}: {error}", path.display()))

255/// The subtable the main table's `label` keyword names, if it has one.
256fn subtable(main: &Table, root_path: &Path, label: &str) -> Option<Table> {
257    let Some(Value::TableRef(reference)) = main.keywords().get(label) else {
258        return None;
259    };
260    Some(open_table(&resolve_table_ref(root_path, reference)))
261}

263fn normalize_existing_path(path: &Path) -> PathBuf {
264    fs::canonicalize(path).unwrap_or_else(|_| path.to_path_buf())

267fn relative_label(root: &Path, path: &Path) -> String {
268    if let Ok(rel) = path.strip_prefix(root) {
269        return rel.to_string_lossy().replace('\\', "/");
270    }
271    path.to_string_lossy().replace('\\', "/")
272}

291fn schema_true_type_tag(col: &casa_tables::ColumnSchema) -> String {
292    match col.column_type() {
293        ColumnType::Scalar => primitive_tag(col.data_type().expect("scalar primitive")).to_string(),
294        ColumnType::Array(_) => {
295            format!(
296                "Array{}",
297                primitive_tag(col.data_type().expect("array primitive"))
298            )
299        }
300        ColumnType::Record => "Record".to_string(),
301    }
302}

304fn schema_options_bits(col: &casa_tables::ColumnSchema) -> u32 {
305    let opts = col.options();
306    let mut bits = 0_u32;
307    if opts.direct {
308        bits |= 1;
309    }
310    if opts.undefined {
311        bits |= 2;
312    }
313    if matches!(
314        col.column_type(),
315        ColumnType::Array(casa_tables::ArrayShapeContract::Fixed { .. })
316    ) {
317        bits |= 4;
318    }
319    bits
320}

322fn schema_ndim_and_shape(col: &casa_tables::ColumnSchema) -> (i64, Vec<usize>) {
323    match col.column_type() {
324        ColumnType::Scalar | ColumnType::Record => (0, Vec::new()),
325        ColumnType::Array(casa_tables::ArrayShapeContract::Fixed { shape }) => {
326            (shape.len() as i64, shape.clone())
327        }
328        ColumnType::Array(casa_tables::ArrayShapeContract::Variable { ndim }) => {
329            (ndim.map_or(-1, |n| n as i64), Vec::new())
330        }
331    }
332}

334fn write_schema_column(digest: &mut StableDigest, col: &casa_tables::ColumnSchema) {
335    digest.write_string(&schema_true_type_tag(col));
336    digest.write_u32(schema_options_bits(col));
337    let (ndim, shape) = schema_ndim_and_shape(col);
338    digest.write_i64(ndim);
339    digest.write_u64(shape.len() as u64);
340    for dim in shape {
341        digest.write_u64(dim as u64);
342    }
343}

345fn write_unknown_schema_column(digest: &mut StableDigest) {
346    digest.write_string("Unknown");
347    digest.write_u32(0);
348    digest.write_i64(-1);
349    digest.write_u64(0);
350}

352fn digest_record(
353    digest: &mut StableDigest,
354    record: &RecordValue,
355    root_path: &Path,
356    discovered_refs: &mut BTreeSet<PathBuf>,
357    owner_path: &Path,
358) {
359    let mut fields: Vec<_> = record.fields().iter().collect();
360    fields.sort_by(|a, b| a.name.cmp(&b.name));
361    digest.write_u64(fields.len() as u64);
362    for field in fields {
363        digest.write_string(&field.name);
364        digest.write_string(value_type_tag(&field.value));
365        digest_value(digest, &field.value, root_path, discovered_refs, owner_path);
366    }
367}

400/// Digest an array's shape and then its elements in Fortran order.
401fn digest_array_with<T>(
402    digest: &mut StableDigest,
403    shape: &[usize],
404    get: impl Fn(&[usize]) -> T,
405    mut emit: impl FnMut(&mut StableDigest, T),
406) {
407    digest.write_u64(shape.len() as u64);
408    for &dim in shape {
409        digest.write_u64(dim as u64);
410    }
411    let nelem = if shape.is_empty() {
412        1
413    } else {
414        shape.iter().product()
415    };
416    if nelem == 0 {
417        return;
418    }
419    let ndim = shape.len();
420    let mut index = vec![0_usize; ndim];
421    loop {
422        emit(digest, get(&index));
423        let mut axis = 0;
424        while axis < ndim {
425            index[axis] += 1;
426            if index[axis] < shape[axis] {
427                break;
428            }
429            index[axis] = 0;
430            axis += 1;
431        }
432        if axis == ndim {
433            break;
434        }
435    }
436}

438fn digest_array(digest: &mut StableDigest, value: &ArrayValue) {
439    use ndarray::IxDyn;
440    match value {
441        ArrayValue::Bool(arr) => digest_array_with(
442            digest,
443            arr.shape(),
444            |idx| arr[IxDyn(idx)],
445            StableDigest::write_bool,
446        ),
447        ArrayValue::UInt8(arr) => digest_array_with(
448            digest,
449            arr.shape(),
450            |idx| arr[IxDyn(idx)],
451            StableDigest::write_u8,
452        ),
453        ArrayValue::UInt16(arr) => digest_array_with(
454            digest,
455            arr.shape(),
456            |idx| arr[IxDyn(idx)],
457            |d, v| d.write_bytes(&v.to_le_bytes()),
458        ),
459        ArrayValue::UInt32(arr) => digest_array_with(
460            digest,
461            arr.shape(),
462            |idx| arr[IxDyn(idx)],
463            StableDigest::write_u32,
464        ),
465        ArrayValue::Int16(arr) => digest_array_with(
466            digest,
467            arr.shape(),
468            |idx| arr[IxDyn(idx)],
469            |d, v| d.write_bytes(&v.to_le_bytes()),
470        ),
471        ArrayValue::Int32(arr) => digest_array_with(
472            digest,
473            arr.shape(),
474            |idx| arr[IxDyn(idx)],
475            |d, v| d.write_bytes(&v.to_le_bytes()),
476        ),
477        ArrayValue::Int64(arr) => digest_array_with(
478            digest,
479            arr.shape(),
480            |idx| arr[IxDyn(idx)],
481            StableDigest::write_i64,
482        ),
483        ArrayValue::Float32(arr) => digest_array_with(
484            digest,
485            arr.shape(),
486            |idx| arr[IxDyn(idx)],
487            |d, v| d.write_u32(v.to_bits()),
488        ),
489        ArrayValue::Float64(arr) => digest_array_with(
490            digest,
491            arr.shape(),
492            |idx| arr[IxDyn(idx)],
493            |d, v| d.write_u64(v.to_bits()),
494        ),
495        ArrayValue::Complex32(arr) => digest_array_with(
496            digest,
497            arr.shape(),
498            |idx| arr[IxDyn(idx)],
499            |d, v| {
500                d.write_u32(v.re.to_bits());
501                d.write_u32(v.im.to_bits());
502            },
503        ),
504        ArrayValue::Complex64(arr) => digest_array_with(
505            digest,
506            arr.shape(),
507            |idx| arr[IxDyn(idx)],
508            |d, v| {
509                d.write_u64(v.re.to_bits());
510                d.write_u64(v.im.to_bits());
511            },
512        ),
513        ArrayValue::String(arr) => digest_array_with(
514            digest,
515            arr.shape(),
516            |idx| arr[IxDyn(idx)].clone(),
517            |d, v| d.write_string(&v),
518        ),
519    }
520}

526fn digest_value(
527    digest: &mut StableDigest,
528    value: &Value,
529    root_path: &Path,
530    discovered_refs: &mut BTreeSet<PathBuf>,
531    owner_path: &Path,
532) {
533    match value {
534        Value::Scalar(ScalarValue::Bool(v)) => digest.write_bool(*v),
535        Value::Scalar(ScalarValue::UInt8(v)) => digest.write_u8(*v),
536        Value::Scalar(ScalarValue::UInt16(v)) => digest.write_bytes(&v.to_le_bytes()),
537        Value::Scalar(ScalarValue::UInt32(v)) => digest.write_u32(*v),
538        Value::Scalar(ScalarValue::Int16(v)) => digest.write_bytes(&v.to_le_bytes()),
539        Value::Scalar(ScalarValue::Int32(v)) => digest.write_bytes(&v.to_le_bytes()),
540        Value::Scalar(ScalarValue::Int64(v)) => digest.write_i64(*v),
541        Value::Scalar(ScalarValue::Float32(v)) => digest.write_u32(v.to_bits()),
542        Value::Scalar(ScalarValue::Float64(v)) => digest.write_u64(v.to_bits()),
543        Value::Scalar(ScalarValue::Complex32(v)) => {
544            digest.write_u32(v.re.to_bits());
545            digest.write_u32(v.im.to_bits());
546        }
547        Value::Scalar(ScalarValue::Complex64(v)) => {
548            digest.write_u64(v.re.to_bits());
549            digest.write_u64(v.im.to_bits());
550        }
551        Value::Scalar(ScalarValue::String(v)) => digest.write_string(v),
552        Value::Array(arr) => digest_array(digest, arr),
553        Value::Record(record) => {
554            digest_record(digest, record, root_path, discovered_refs, owner_path);
555        }
556        Value::TableRef(path) => {
557            let resolved = resolve_table_ref(owner_path, path);
558            discovered_refs.insert(resolved.clone());
559            digest.write_string(&relative_label(root_path, &resolved));
560        }
561    }
562}

564/// The table's column names, from its schema or else from its first row.
565fn column_names(table: &Table) -> Vec<String> {
566    if let Some(schema) = table.schema() {
567        return schema
568            .columns()
569            .iter()
570            .map(|col| col.name().to_string())
571            .collect();
572    }
573    table
574        .row_accessor()
575        .row(0)
576        .map(|row| {
577            row.fields()
578                .iter()
579                .map(|field| field.name.clone())
580                .collect()
581        })
582        .unwrap_or_default()
583}

585/// A row's column names, sorted.
586fn sorted_row_columns(table: &Table, row: &RecordValue) -> Vec<String> {
587    let mut names: Vec<String> = match table.schema() {
588        Some(schema) => schema
589            .columns()
590            .iter()
591            .map(|col| col.name().to_string())
592            .collect(),
593        None => row
594            .fields()
595            .iter()
596            .map(|field| field.name.clone())
597            .collect(),
598    };
599    names.sort();
600    names
601}

603/// The value of a field a row omits: the empty record of a record column,
604/// or, for an undefined array cell, an empty array of the column's type,
605/// which gives the manifest its type tag (its elements are not digested).
606fn synthesized_value(table: &Table, name: &str) -> Value {
607    let schema = table
608        .schema()
609        .unwrap_or_else(|| panic!("missing row field {name} without schema"));
610    let column = schema
611        .column(name)
612        .unwrap_or_else(|| panic!("missing row field {name} not in the schema"));
613    match (column.column_type(), column.data_type()) {
614        (ColumnType::Record, _) => Value::Record(RecordValue::default()),
615        (ColumnType::Array(_), Some(data_type)) => Value::Array(empty_array(data_type)),
616        _ => panic!("missing value for defined row field {name}"),
617    }
618}

620/// An empty array of `data_type`.
621fn empty_array(data_type: PrimitiveType) -> ArrayValue {
622    match data_type {
623        PrimitiveType::Bool => ArrayValue::from_bool_vec(Vec::new()),
624        PrimitiveType::UInt8 => ArrayValue::from_u8_vec(Vec::new()),
625        PrimitiveType::UInt16 => ArrayValue::from_u16_vec(Vec::new()),
626        PrimitiveType::UInt32 => ArrayValue::from_u32_vec(Vec::new()),
627        PrimitiveType::Int16 => ArrayValue::from_i16_vec(Vec::new()),
628        PrimitiveType::Int32 => ArrayValue::from_i32_vec(Vec::new()),
629        PrimitiveType::Int64 => ArrayValue::from_i64_vec(Vec::new()),
630        PrimitiveType::Float32 => ArrayValue::from_f32_vec(Vec::new()),
631        PrimitiveType::Float64 => ArrayValue::from_f64_vec(Vec::new()),
632        PrimitiveType::Complex32 => ArrayValue::from_complex32_vec(Vec::new()),
633        PrimitiveType::Complex64 => ArrayValue::from_complex64_vec(Vec::new()),
634        PrimitiveType::String => ArrayValue::from_string_vec(Vec::new()),
635    }
636}

638fn undefined_cell_ndim(table: &Table, name: &str) -> i64 {
639    table
640        .schema()
641        .and_then(|schema| schema.column(name))
642        .map_or(-1, |schema| schema_ndim_and_shape(schema).0)
643        .max(0)
644}
645
646fn digest_table_info(table: &Table) -> StableDigest {
647    let mut digest = StableDigest::default();
648    digest.write_string(&table.info().table_type);
649    digest.write_string(&table.info().sub_type);
650    digest
651}
652
653fn digest_table_schema(table: &Table) -> StableDigest {
654    let mut digest = StableDigest::default();
655    if let Some(schema) = table.schema() {
656        digest.write_u64(schema.columns().len() as u64);
657        for col in schema.columns() {
658            digest.write_string(col.name());
659            write_schema_column(&mut digest, col);
660        }
661    } else {
662        let names = column_names(table);
663        digest.write_u64(names.len() as u64);
664        for name in names {
665            digest.write_string(&name);
666            write_unknown_schema_column(&mut digest);
667        }
668    }
669    digest
670}
671
672fn append_table_schema_column_lines(manifest: &mut Vec<String>, table: &Table, label: &str) {
673    for name in column_names(table) {
674        let mut digest = StableDigest::default();
675        digest.write_string(&name);
676        match table.schema().and_then(|s| s.column(&name)) {
677            Some(schema) => write_schema_column(&mut digest, schema),
678            None => write_unknown_schema_column(&mut digest),
679        }
680        manifest.push(format!("{label}:SCHEMACOL:{name} {}", digest.hex()));
681    }
682}

684fn digest_table_column_keywords(
685    table: &Table,
686    root_path: &Path,
687    discovered_refs: &mut BTreeSet<PathBuf>,
688) -> StableDigest {
689    let mut digest = StableDigest::default();
690    let owner_path = normalize_existing_path(table.path().expect("disk-backed table path"));
691    let names = column_names(table);
692    digest.write_u64(names.len() as u64);
693    for name in names {
694        digest.write_string(&name);
695        let keywords = table.column_keywords(&name).cloned().unwrap_or_default();
696        digest_record(
697            &mut digest,
698            &keywords,
699            root_path,
700            discovered_refs,
701            &owner_path,
702        );
703    }
704    digest
705}

707fn digest_table_keywords(
708    table: &Table,
709    root_path: &Path,
710    discovered_refs: &mut BTreeSet<PathBuf>,
711) -> StableDigest {
712    let mut digest = StableDigest::default();
713    let owner_path = normalize_existing_path(table.path().expect("disk-backed table path"));
714    digest_record(
715        &mut digest,
716        table.keywords(),
717        root_path,
718        discovered_refs,
719        &owner_path,
720    );
721    digest
722}
723
724fn write_row_digest_for_table(
725    digest: &mut StableDigest,
726    table: &Table,
727    row_index: usize,
728    root_path: &Path,
729    discovered_refs: &mut BTreeSet<PathBuf>,
730) {
731    let owner_path = normalize_existing_path(table.path().expect("disk-backed table path"));
732    let row = table.row_accessor().row(row_index).expect("row present");
733    let names = sorted_row_columns(table, row);
734    digest.write_u64(names.len() as u64);
735    for name in names {
736        let defined = table
737            .is_cell_defined(row_index, &name)
738            .expect("column exists for row digest");
739        let synthesized;
740        let value = match row.get(&name) {
741            Some(value) => value,
742            None => {
743                synthesized = synthesized_value(table, &name);
744                &synthesized
745            }
746        };
747        digest.write_string(&name);
748        digest.write_string(value_type_tag(value));
749        digest.write_bool(defined);
750        if defined {
751            digest_value(digest, value, root_path, discovered_refs, &owner_path);
752        } else {
753            digest.write_i64(undefined_cell_ndim(table, &name));
754        }
755    }
756}
757
758fn digest_table_rows(
759    table: &Table,
760    root_path: &Path,
761    discovered_refs: &mut BTreeSet<PathBuf>,
762) -> StableDigest {
763    let mut digest = StableDigest::default();
764    digest.write_u64(table.row_count() as u64);
765    for row_index in 0..table.row_count() {
766        write_row_digest_for_table(&mut digest, table, row_index, root_path, discovered_refs);
767    }
768    digest
769}
770
771fn digest_table_recursive(
772    table: &Table,
773    label: String,
774    root_path: &Path,
775    visited: &mut BTreeSet<PathBuf>,
776    manifest: &mut Vec<String>,
777) {
778    let table_path = normalize_existing_path(table.path().expect("disk-backed table path"));
779    if !visited.insert(table_path) {
780        return;
781    }
782
783    let mut discovered_refs = BTreeSet::new();
784    manifest.push(format!("{label}:INFO {}", digest_table_info(table).hex()));
785    manifest.push(format!(
786        "{label}:SCHEMA {}",
787        digest_table_schema(table).hex()
788    ));
789    append_table_schema_column_lines(manifest, table, &label);
790    manifest.push(format!(
791        "{label}:COLKW {}",
792        digest_table_column_keywords(table, root_path, &mut discovered_refs).hex()
793    ));
794    manifest.push(format!(
795        "{label}:TABLEKW {}",
796        digest_table_keywords(table, root_path, &mut discovered_refs).hex()
797    ));
798    manifest.push(format!(
799        "{label}:ROWS {}",
800        digest_table_rows(table, root_path, &mut discovered_refs).hex()
801    ));
802
803    for ref_path in discovered_refs {
804        if visited.contains(&ref_path) {
805            continue;
806        }
807        if !ref_path.exists() {
808            let ref_label = format!("EXTRA_MISSING:{}", relative_label(root_path, &ref_path));
809            manifest.push(format!("{ref_label} MISSING"));
810            continue;
811        }
812        let ref_table = open_table(&ref_path);
813        let ref_label = format!("EXTRA:{}", relative_label(root_path, &ref_path));
814        digest_table_recursive(&ref_table, ref_label, root_path, visited, manifest);
815    }
816}
