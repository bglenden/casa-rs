11type Coordinates = (i32, usize, usize);
12type Metric = &'static str;
13type Metrics<T> = BTreeMap<Metric, T>;
14type Samples = BTreeMap<Coordinates, Metrics<Vec<f64>>>;
15type WindowValues<T> = BTreeMap<i32, BTreeMap<usize, BTreeMap<usize, T>>>;
16type Platforming = BTreeMap<String, BTreeMap<i32, BTreeMap<usize, Vec<CasaValue>>>>;
17
18pub(super) struct StatisticsGrid {
19    antennas: BTreeSet<String>,
20    windows: BTreeSet<i32>,
21    times: BTreeSet<usize>,
22    receptors: usize,
23    metrics: Vec<Metric>,
24}
25
26impl StatisticsGrid {
27    pub(super) fn new(
28        rows: impl Iterator<Item = (String, i32, usize, usize)>,
29        ordinates: &[Ordinate],
30    ) -> Self {
31        let mut grid = Self {
32            antennas: BTreeSet::new(),
33            windows: BTreeSet::new(),
34            times: BTreeSet::new(),
35            receptors: 0,
36            metrics: ordinates.iter().map(|axis| metric(*axis)).collect(),
37        };
38        for (antenna, window, time, receptors) in rows {
39            grid.antennas.insert(antenna);
40            grid.windows.insert(window);
41            grid.times.insert(time);
42            grid.receptors = grid.receptors.max(receptors);
43        }
44        grid
45    }
46
47    fn coordinates(&self) -> impl Iterator<Item = Coordinates> + '_ {
48        self.windows.iter().flat_map(move |&window| {
49            self.times
50                .iter()
51                .flat_map(move |&time| (0..self.receptors).map(move |pol| (window, time, pol)))
52        })
53    }
54}

56pub(crate) fn derivative_record(plots: &BandpassPlots) -> Option<CasaValue> {
57    let grid = plots.statistics.as_ref()?;
58    let mut antennas = BTreeMap::<&str, WindowValues<Metrics<Option<f64>>>>::new();
59    for antenna in &grid.antennas {
60        let mut values = WindowValues::new();
61        for address in grid.coordinates() {
62            *coordinates_mut(&mut values, address) =
63                grid.metrics.iter().map(|&metric| (metric, None)).collect();
64        }
65        antennas.insert(antenna, values);
66    }
67
68    let mut samples = Samples::new();
69    let mut platforming = Platforming::new();
70    for panel in &plots.panels {
71        let metric = metric(panel.ordinate);
72        let std_metric = if metric == "phase" {
73            "phasestd"
74        } else {
75            "ampstd"
76        };
77        for series in &panel.series {
78            let Some(DerivativeStatistics {
79                mad,
80                stdev,
81                platforming: event,
82            }) = series.statistics
83            else {
84                continue;
85            };
86            let address = (series.window, series.time_index, series.receptor);
87            let values = antennas.entry(&series.antenna_name).or_default();
88            let cell = coordinates_mut(values, address);
89            cell.insert(metric, Some(mad));
90            cell.insert(std_metric, Some(stdev));
91            if let Some(event) = event {
92                add_event(&mut platforming, &series.antenna_name, address, event);
93            }
94        }
95    }
96    for antenna in antennas.values() {
97        for address in grid.coordinates() {
98            let (window, time, receptor) = address;
99            for &metric in &grid.metrics {
100                if let Some(mad) = antenna[&window][&time][&receptor][metric] {
101                    samples
102                        .entry(address)
103                        .or_default()
104                        .entry(metric)
105                        .or_default()
106                        .push(mad);
107                }
108            }
109        }
110    }
111
112    let mut entries = antennas
113        .into_iter()
114        .map(|(antenna, windows)| (antenna.to_string(), window_dict(windows, encode_optional)))
115        .collect::<Vec<_>>();
116    entries.push((
117        "platforming".into(),
118        CasaValue::dict(platforming.into_iter().map(|(antenna, windows)| {
119            (
120                antenna,
121                int_dict(windows, |receptors| {
122                    int_dict(receptors, |items| CasaValue::List { items })
123                }),
124            )
125        })),
126    ));
127    entries.push(("median".into(), median_record(grid, &samples)));
128    Some(CasaValue::dict(entries))
129}

131fn metric(ordinate: Ordinate) -> Metric {
132    if ordinate == Ordinate::Phase {
133        "phase"
134    } else {
135        "amp"
136    }
137}

139fn coordinates_mut<T: Default>(
140    values: &mut WindowValues<T>,
141    (window, time, receptor): Coordinates,
142) -> &mut T {
143    values
144        .entry(window)
145        .or_default()
146        .entry(time)
147        .or_default()
148        .entry(receptor)
149        .or_default()
150}
151
152fn add_event(
153    platforming: &mut Platforming,
154    antenna: &str,
155    (window, _, receptor): Coordinates,
156    event: PlatformingTime,
157) {
158    let value = match event {
159        PlatformingTime::Epoch(time) => CasaValue::float(time),
160        PlatformingTime::Index(index) => CasaValue::Int {
161            value: index as i64,
162        },
163    };
164    platforming
165        .entry(antenna.into())
166        .or_default()
167        .entry(window)
168        .or_default()
169        .entry(receptor)
170        .or_default()
171        .push(value);
172}

174fn median_record(grid: &StatisticsGrid, samples: &Samples) -> CasaValue {
175    let mut windows = WindowValues::<Metrics<Option<f64>>>::new();
176    let mut all = Metrics::<Vec<f64>>::new();
177    let mut last_address = None;
178    for address in grid.coordinates() {
179        let cell = coordinates_mut(&mut windows, address);
180        last_address = Some(address);
181        for &metric in &grid.metrics {
182            let values = samples
183                .get(&address)
184                .and_then(|metrics| metrics.get(metric))
185                .map_or(&[][..], Vec::as_slice);
186            all.entry(metric).or_default().extend_from_slice(values);
187            cell.insert(metric, Some(median(values)));
188        }
189    }
190    let overall = all
191        .into_iter()
192        .map(|(metric, values)| (metric, Some(median(&values))))
193        .collect();
194    if grid.metrics == ["phase"] {
195        // CASA puts the overall phase median in the final cell in phase-only mode.
196        if let Some(address) = last_address {
197            *coordinates_mut(&mut windows, address) = overall;
198        }
199        return window_dict(windows, encode_optional);
200    }
201    let CasaValue::Dict { mut entries } = window_dict(windows, encode_optional) else {
202        unreachable!("window dictionaries always encode as dictionaries");
203    };
204    entries.push(CasaEntry {
205        key: CasaKey::String("median".into()),
206        value: encode_optional(overall),
207    });
208    CasaValue::Dict { entries }
209}

211fn window_dict<T>(values: WindowValues<T>, mut encode: impl FnMut(T) -> CasaValue) -> CasaValue {
212    int_dict(values, |times| {
213        int_dict(times, |receptors| int_dict(receptors, &mut encode))
214    })
215}
216
217fn int_dict<I, T>(items: BTreeMap<I, T>, mut encode: impl FnMut(T) -> CasaValue) -> CasaValue
218where
219    I: TryInto<i64>,
220{
221    CasaValue::Dict {
222        entries: items
223            .into_iter()
224            .filter_map(|(key, item)| {
225                Some(CasaEntry {
226                    key: CasaKey::Int(key.try_into().ok()?),
227                    value: encode(item),
228                })
229            })
230            .collect(),
231    }
232}
233
234fn encode_optional(values: Metrics<Option<f64>>) -> CasaValue {
235    CasaValue::dict(
236        values
237            .into_iter()
238            .map(|(key, value)| (key, value.map_or(CasaValue::None, CasaValue::float))),
239    )
