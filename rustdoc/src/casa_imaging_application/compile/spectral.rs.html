30/// The compiled spectral axis and the source channels it reads.
31pub(super) struct PreparedSpectralAxis {
32    pub(super) selected_source_channels: WindowChannels,
33    pub(super) source_frame: FrequencyFrame,
34    pub(super) output_frequency_reference: FrequencyRef,
35    pub(super) output_frame: FrequencyFrame,
36    pub(super) anchor: SpectralFrameAnchor,
37    pub(super) wcs: SpectralWcs,
38    pub(super) rest_frequency: RestFrequency,
39    /// The rest frequency the image header records.
40    pub(super) image_rest_frequency_hz: f64,
41    pub(super) doppler: DopplerConvention,
42    pub(super) sampling: SpectralSamplingLaw,
43    pub(super) basis: ReconstructionBasis,
44    pub(super) output_channels: usize,
45    /// Centre of the first output channel.
46    pub(super) reference_frequency_hz: f64,
47    /// Channel increment; a continuum plane's full band.
48    pub(super) increment_hz: f64,
49}

51impl PreparedSpectralAxis {
52    /// The model's spectral coordinate of this axis.
53    pub(super) fn coordinate(&self) -> SpectralCoordinateSpec {
54        SpectralCoordinateSpec::new(
55            self.source_frame,
56            self.output_frame,
57            self.anchor,
58            self.wcs.clone(),
59            self.rest_frequency,
60            self.doppler,
61        )
62    }
63
64    /// The source channels the axis reads from surveyed window `spw_id`.
65    pub(super) fn window_channels(&self, spw_id: usize) -> &[usize] {
66        self.selected_source_channels
67            .get(&spw_id)
68            .expect("the spectral axis selects channels of every surveyed window")
69    }

82/// The output axis of one spectral mode, before its frame anchor.
83struct AxisLaw {
84    selected_source_channels: WindowChannels,
85    output_frequency_reference: FrequencyRef,
86    reference_frequency_hz: f64,
87    increment_hz: f64,
88    output_channels: usize,
89    rest_frequency: RestFrequency,
90    image_rest_frequency_hz: f64,
91    doppler: DopplerConvention,
92    sampling: SpectralSamplingLaw,
93    basis: ReconstructionBasis,

96/// The request's spectral axis over the surveyed selection.
97pub(super) fn prepare_axis(
98    request: &ImagingRequest,
99    ms: &MeasurementSet,
100    survey: &Survey,
101    frame: &FrameContext<'_>,
102    moving_rest_frame: Option<&MeasFrame>,
103) -> Result<PreparedSpectralAxis, PrepareError> {
104    let source_reference = survey.source_frequency_reference;
105    let law = match request.specmode {
106        SpecMode::Mfs => continuum_law(survey),
107        SpecMode::Cube | SpecMode::Cubedata => cube_law(request, ms, survey, frame)?,
108        SpecMode::Cubesource => {
109            let frame_with_velocity = moving_rest_frame
110                .expect("prepare resolves the moving rest frame of every source-frame cube");
111            source_frame_cube_law(request, ms, survey, frame, frame_with_velocity)?
112        }
113    };
114    let source_frame = imaging_frequency_frame(source_reference)?;
115    let output_frame = imaging_frequency_frame(law.output_frequency_reference)?;
116    Ok(PreparedSpectralAxis {
117        selected_source_channels: law.selected_source_channels,
118        source_frame,
119        output_frequency_reference: law.output_frequency_reference,
120        output_frame,
121        anchor: spectral_frame_anchor(source_frame, output_frame, frame)?,
122        wcs: SpectralWcs::Linear {
123            channels: law.output_channels,
124            reference_pixel: 0.0,
125            reference_frequency_hz: law.reference_frequency_hz,
126            increment_hz: law.increment_hz,
127        },
128        rest_frequency: law.rest_frequency,
129        image_rest_frequency_hz: law.image_rest_frequency_hz,
130        doppler: law.doppler,
131        sampling: law.sampling,
132        basis: law.basis,
133        output_channels: law.output_channels,
134        reference_frequency_hz: law.reference_frequency_hz,
135        increment_hz: law.increment_hz,
136    })
137}

139/// One LSRK plane over the selected band's envelope.
140fn continuum_law(survey: &Survey) -> AxisLaw {
141    let (channels, envelope) = survey
142        .continuum
143        .as_ref()
144        .expect("the survey reduces the spectral envelope of every continuum run");
145    let [lower_hz, upper_hz] = envelope.edges_hz();
146    let reference_frequency_hz = envelope.midpoint_hz();
147    AxisLaw {
148        selected_source_channels: channels.clone(),
149        output_frequency_reference: FrequencyRef::LSRK,
150        reference_frequency_hz,
151        increment_hz: upper_hz - lower_hz,
152        output_channels: 1,
153        rest_frequency: RestFrequency::NotApplicable,
154        image_rest_frequency_hz: reference_frequency_hz,
155        doppler: DopplerConvention::NotApplicable,
156        sampling: SpectralSamplingLaw::IDENTITY,
157        basis: ReconstructionBasis::Constant,
158    }
159}

161/// A cube in its output frame (`cube`) or the data's (`cubedata`).
162fn cube_law(
163    request: &ImagingRequest,
164    ms: &MeasurementSet,
165    survey: &Survey,
166    frame: &FrameContext<'_>,
167) -> Result<AxisLaw, PrepareError> {
168    let window = one_window(&survey.spectral_windows, "native cube imaging")?;
169    let axis = cube_axis(request)?;
170    let output_channels = request.channel_count.unwrap_or(window.frequencies_hz.len());
171    let (setup, support) = cube_setup(survey, window, output_channels, &axis, frame)?;
172    let mut selected = support.indices;
173    let explicit = explicit_spw_channels(request, window.spw_id, &window.frequencies_hz)?
174        .map(|channels| channels.into_iter().collect::<BTreeSet<_>>());
175    if let Some(explicit) = &explicit {
176        selected.retain(|channel| explicit.contains(channel));
177    }
178    if selected.is_empty() {
179        return Err(PrepareError::NoCommonCubeChannels);
180    }
181    let (reference_frequency_hz, increment_hz) = first_and_increment(&setup, output_channels, 1.0)?;
182    let source_rest_hz = source_rest_frequency(ms, frame.field_id, &survey.spectral_windows)?;
183    let (resolved_rest_hz, image_rest_frequency_hz) =
184        cube_rest_frequency_hz(request.restfreq, source_rest_hz, window, explicit.as_ref());
185    let (rest_frequency, doppler) = match resolved_rest_hz {
186        None => (
187            RestFrequency::NotApplicable,
188            DopplerConvention::NotApplicable,
189        ),
190        Some(hertz) => (
191            RestFrequency::Line { hertz },
192            doppler_convention(request.veltype)?,
193        ),
194    };
195    Ok(AxisLaw {
196        selected_source_channels: BTreeMap::from([(window.spw_id, selected)]),
197        output_frequency_reference: setup.output_freq_ref,
198        reference_frequency_hz,
199        increment_hz,
200        output_channels,
201        rest_frequency,
202        image_rest_frequency_hz,
203        doppler,
204        sampling: sampling_law(setup.interpolation),
205        basis: ReconstructionBasis::ChannelLocal {
206            channels: output_channels,
207        },
208    })
209}

211/// A cube in the rest frame of a moving source: the data-frame axis
212/// scaled into the source's rest frame (`cubesource`).
213fn source_frame_cube_law(
214    request: &ImagingRequest,
215    ms: &MeasurementSet,
216    survey: &Survey,
217    frame: &FrameContext<'_>,
218    moving_rest_frame: &MeasFrame,
219) -> Result<AxisLaw, PrepareError> {
220    let window = one_window(&survey.spectral_windows, "source-frame cube imaging")?;
221    let source_reference = survey.source_frequency_reference;
222    let mut axis = cube_axis(request)?;
223    axis.specmode = CubeSpecMode::Cubedata;
224    axis.outframe = source_reference;
225    let output_channels = request.channel_count.unwrap_or(window.frequencies_hz.len());
226    let (setup, support) = cube_setup(survey, window, output_channels, &axis, frame)?;
227    let factor = casa_ms::convert_frequency_to_frame_with_frame(
228        source_reference,
229        FrequencyRef::REST,
230        1.0,
231        Some(moving_rest_frame),
232    )?;
233    let (reference_frequency_hz, increment_hz) =
234        first_and_increment(&setup, output_channels, factor)?;
235    let rest_frequency_hz = request
236        .restfreq
237        .or(source_rest_frequency(
238            ms,
239            frame.field_id,
240            &survey.spectral_windows,
241        )?)
242        .ok_or(PrepareError::RestFrequencyMissing)?;
243    Ok(AxisLaw {
244        selected_source_channels: BTreeMap::from([(window.spw_id, support.indices)]),
245        output_frequency_reference: FrequencyRef::REST,
246        reference_frequency_hz,
247        increment_hz,
248        output_channels,
249        rest_frequency: RestFrequency::Line {
250            hertz: rest_frequency_hz,
251        },
252        image_rest_frequency_hz: rest_frequency_hz,
253        doppler: doppler_convention(request.veltype)?,
254        sampling: sampling_law(request.interpolation),
255        basis: ReconstructionBasis::ChannelLocal {
256            channels: output_channels,
257        },
258    })
259}

261/// CASA's cube axis of the request: `start` (else `channel_start`) and
262/// `width` in the request's velocity convention.
263fn cube_axis(request: &ImagingRequest) -> Result<CubeAxisConfig, PrepareError> {
264    let start = match (&request.start, request.channel_start) {
265        (Some(start), _) => Some(CubeAxisValue::parse(start, request.veltype)?),
266        (None, Some(channel)) => Some(CubeAxisValue::Channel(
267            i32::try_from(channel).map_err(|_| PrepareError::CubeStartChannel { channel })?,
268        )),
269        (None, None) => None,
270    };
271    let width = request
272        .width
273        .as_deref()
274        .map(|width| CubeAxisValue::parse(width, request.veltype))
275        .transpose()?;
276    Ok(CubeAxisConfig {
277        specmode: if request.specmode == SpecMode::Cubedata {
278            CubeSpecMode::Cubedata
279        } else {
280            CubeSpecMode::Cube
281        },
282        outframe: request.outframe,
283        veltype: request.veltype,
284        interpolation: request.interpolation,
285        rest_frequency_hz: request.restfreq,
286        start,
287        width,
288    })
289}

291fn cube_setup(
292    survey: &Survey,
293    window: &SourceSpectralWindow,
294    output_channels: usize,
295    axis: &CubeAxisConfig,
296    frame: &FrameContext<'_>,
297) -> Result<(CubeSpectralSetup, casa_ms::ResolvedChannelSelection), PrepareError> {
298    Ok(CubeSpectralSetup::for_casa_cube_axis(
299        survey.source_frequency_reference,
300        &window.frequencies_hz,
301        &window.channel_widths_hz,
302        output_channels,
303        axis,
304        frame.anchor_time_mjd_seconds,
305        frame.field_id,
306        Some(frame.phase.clone()),
307        frame.time_bounds_mjd_seconds,
308        frame.engine,
309    )?)
310}

312/// The first output channel's centre and the channel increment, scaled by
313/// `factor`; a one-channel cube's increment is its width.
314fn first_and_increment(
315    setup: &CubeSpectralSetup,
316    output_channels: usize,
317    factor: f64,
318) -> Result<(f64, f64), PrepareError> {
319    let first_hz = setup.output_channel_frequencies_hz[0];
320    let increment_hz = if output_channels > 1 {
321        setup.output_channel_frequencies_hz[1] - first_hz
322    } else {
323        setup.output_channel_widths_hz[0]
324    } * factor;
325    if !increment_hz.is_finite() || increment_hz == 0.0 {
326        return Err(PrepareError::CubeIncrement { increment_hz });
327    }
328    Ok((first_hz * factor, increment_hz))
329}

331/// The model's Doppler convention of CASA `veltype`.
332fn doppler_convention(veltype: DopplerRef) -> Result<DopplerConvention, PrepareError> {
333    match veltype {
334        DopplerRef::RADIO => Ok(DopplerConvention::Radio),
335        DopplerRef::Z => Ok(DopplerConvention::Optical),
336        DopplerRef::BETA => Ok(DopplerConvention::Relativistic),
337        DopplerRef::RATIO | DopplerRef::GAMMA => Err(PrepareError::UnsupportedDoppler {
338            convention: veltype,
339        }),
340    }
341}

343const fn sampling_law(interpolation: CubeInterpolation) -> SpectralSamplingLaw {
344    match interpolation {
345        CubeInterpolation::Nearest => SpectralSamplingLaw::NEAREST,
346        CubeInterpolation::Linear => SpectralSamplingLaw::LINEAR,
347        CubeInterpolation::Cubic => SpectralSamplingLaw::CUBIC,
348    }

351fn spectral_frame_anchor(
352    source_frame: FrequencyFrame,
353    output_frame: FrequencyFrame,
354    frame: &FrameContext<'_>,
355) -> Result<SpectralFrameAnchor, PrepareError> {
356    if source_frame == output_frame {
357        return Ok(SpectralFrameAnchor::NotApplicable);
358    }
359    let [x_metres, y_metres, z_metres] = frame.engine.observatory_position().as_itrf();
360    Ok(SpectralFrameAnchor::Conversion {
361        epoch: Epoch::new(
362            frame.anchor_time_mjd_seconds / 86_400.0,
363            imaging_time_scale(frame.engine.time_reference())?,
364        ),
365        direction: frame.direction.reference_direction(),
366        observatory_position: ItrfPosition::new(x_metres, y_metres, z_metres),
367    })
368}

370fn imaging_frequency_frame(reference: FrequencyRef) -> Result<FrequencyFrame, PrepareError> {
371    match reference {
372        FrequencyRef::REST => Ok(FrequencyFrame::Rest),
373        FrequencyRef::TOPO => Ok(FrequencyFrame::Topocentric),
374        FrequencyRef::BARY => Ok(FrequencyFrame::Barycentric),
375        FrequencyRef::LSRK => Ok(FrequencyFrame::Lsrk),
376        _ => Err(PrepareError::UnsupportedFrequencyFrame { reference }),
377    }
378}

380pub(super) fn imaging_time_scale(reference: EpochRef) -> Result<TimeScale, PrepareError> {
381    match reference {
382        EpochRef::UTC => Ok(TimeScale::Utc),
383        EpochRef::TAI => Ok(TimeScale::Tai),
384        EpochRef::TT => Ok(TimeScale::Tt),
385        EpochRef::TDB => Ok(TimeScale::Tdb),
386        _ => Err(PrepareError::UnsupportedEpochReference { reference }),
387    }
388}

390/// The `REST_FREQUENCY` the `SOURCE` table gives the field over the
391/// selected windows, if any.
392fn source_rest_frequency(
393    measurement_set: &MeasurementSet,
394    field_id: usize,
395    spectral_windows: &[SourceSpectralWindow],
396) -> Result<Option<f64>, PrepareError> {
397    let source_id = measurement_set.field()?.source_id(field_id)?;
398    if source_id < 0 || measurement_set.subtable(SubtableId::Source).is_none() {
399        return Ok(None);
400    }
401    let source = measurement_set.source()?;
402    let selected = spectral_windows
403        .iter()
404        .map(|window| i32::try_from(window.spw_id).expect("SPW ids are stored i32 values"))
405        .collect::<BTreeSet<_>>();
406    let mut resolved = None;
407    for row in 0..source.row_count() {
408        if source.i32(row, "SOURCE_ID")? != source_id {
409            continue;
410        }
411        let spectral_window_id = source.i32(row, "SPECTRAL_WINDOW_ID")?;
412        if spectral_window_id >= 0 && !selected.contains(&spectral_window_id) {
413            continue;
414        }
415        let Some(ArrayValue::Float64(values)) = source.optional_array(row, "REST_FREQUENCY")?
416        else {
417            continue;
418        };
419        let Some(value) = values.first().copied().filter(|value| *value > 0.0) else {
420            continue;
421        };
422        if resolved.is_some_and(|prior: f64| prior.to_bits() != value.to_bits()) {
423            return Err(PrepareError::RestFrequencyConflict);
424        }
425        resolved = Some(value);
426    }
427    Ok(resolved)
428}

430/// CASA's precedence for a cube's rest frequency: the request's, else the
431/// source's; the header records the selected band's centre when neither
432/// exists.
433fn cube_rest_frequency_hz(
434    explicit_hz: Option<f64>,
435    source_hz: Option<f64>,
436    spectral_window: &SourceSpectralWindow,
437    selected_channels: Option<&BTreeSet<usize>>,
438) -> (Option<f64>, f64) {
439    let resolved = explicit_hz.or(source_hz);
440    let image_hz = resolved.unwrap_or_else(|| {
441        let (lower, upper) = spectral_window
442            .frequencies_hz
443            .iter()
444            .zip(&spectral_window.channel_widths_hz)
445            .enumerate()
446            .filter(|(index, _)| selected_channels.is_none_or(|channels| channels.contains(index)))
447            .map(|(_, (&centre, &width))| (centre - width.abs() / 2.0, centre + width.abs() / 2.0))
448            .fold(
449                (f64::INFINITY, f64::NEG_INFINITY),
450                |(lower, upper), (lo, hi)| (lower.min(lo), upper.max(hi)),
451            );
452        lower + (upper - lower) / 2.0
453    });
454    (resolved, image_hz)
455}
