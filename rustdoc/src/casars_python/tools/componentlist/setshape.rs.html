29/// Parse all parameters in CASA's order and report shape refusals as false.
30pub(super) fn prepare(
31    request: &Request,
32    lines: &mut Vec<LogLine>,
33) -> ConvertResult<Option<Prepared>> {
34    let shape_type = match ShapeType::parse_casa(&request.name) {
35        Ok(ShapeType::LimbDarkenedDisk) => {
36            lines.push(LogLine::error(
37                "Limbdarkeneddisk is not implemented yet. Shape not changed.",
38            ));
39            return Ok(None);
40        }
41        Ok(kind) => kind,
42        Err(_) => {
43            lines.push(LogLine::error("Could not translate the shape type to a known value. Known types are: Point, Gaussian, Disk. Shape not changed."));
44            return Ok(None);
45        }
46    };
47    let axes = [
48        quantity(&request.axes[0])?,
49        quantity(&request.axes[1])?,
50        quantity(&request.axes[2])?,
51    ];
52    let empty = |value: &Variant| {
53        value.is_empty_sequence() || matches!(value, Variant::Text(text) if text == "[]")
54    };
55    let errors: [Quantity; 3] = if request.errors.iter().all(|value| !empty(value)) {
56        [
57            quantity(&request.errors[0])?,
58            quantity(&request.errors[1])?,
59            quantity(&request.errors[2])?,
60        ]
61    } else {
62        let rad = Unit::new("rad").expect("rad is a unit");
63        std::array::from_fn(|_| Quantity::with_unit(0.0, rad.clone()))
64    };
65    let direction = MDirection::from_angles(0.0, 0.0, DirectionRef::J2000);
66    if shape_type == ShapeType::Point {
67        return Ok(Some(Prepared {
68            shape: ComponentShape::point(direction),
69            units: None,
70        }));
71    }
72    for (name, q) in ["majoraxis", "minoraxis", "positionangle"]
73        .into_iter()
74        .zip(&axes)
75    {
76        if q.unit().val().dim != UnitDim::basis(Dimension::Angle) {
77            lines.push(LogLine::error(format!("Could not parse the shape parameters. The {name} field must have angular units. Shape not changed.")));
78            return Ok(None);
79        }
80    }
81    let [major, mut minor, angle] = axes.each_ref().map(Quantity::get_si_value);
82    if let Some(message) = super::add::constructor_error(shape_type, major, minor, angle) {
83        return Err(ConvertError::runtime(message));
84    }
85    if shape_type == ShapeType::Disk && (major - minor).abs() < 1e-17 {
86        minor = major;
87    }
88    if [major, minor, angle].iter().any(|value| !value.is_finite()) {
89        return Err(ConvertError::runtime(
90            "non-finite component shape refused (#1053)",
91        ));
92    }
93    for (name, q) in ["majoraxiserror", "minoraxiserror", "positionangleerror"]
94        .into_iter()
95        .zip(&errors)
96    {
97        if q.unit().val().dim != UnitDim::basis(Dimension::Angle) {
98            lines.push(LogLine::error(format!("Could not parse the shape parameters. The {name} field must have angular units. Shape not changed.")));
99            return Ok(None);
100        }
101    }
102    let [major_error, minor_error, angle_error] = errors.each_ref().map(Quantity::get_si_value);
103    if [major_error, minor_error, angle_error]
104        .iter()
105        .any(|value| !value.is_finite())
106    {
107        return Err(ConvertError::runtime(
108            "non-finite component shape error refused (#1053)",
109        ));
110    }
111    if [major_error, minor_error, angle_error]
112        .iter()
113        .any(|value| *value < 0.0)
114    {
115        return Err(ConvertError::runtime(
116            "The errors must be non-negative angular quantities.",
117        ));
118    }
119    let axes_value =
120        Ellipse::new(major, minor, angle).map_err(|e| ConvertError::runtime(e.to_string()))?;
121    let error_values = EllipseErrors {
122        major_axis: major_error,
123        minor_axis: minor_error,
124        position_angle: angle_error,
125    };
126    let kind = match shape_type {
127        ShapeType::Gaussian => ShapeKind::Gaussian {
128            axes: axes_value,
129            errors: error_values,
130        },
131        ShapeType::Disk => ShapeKind::Disk {
132            axes: axes_value,
133            errors: error_values,
134        },
135        _ => unreachable!("point and limb-darkened disk handled above"),
136    };
137    let shape =
138        ComponentShape::new(direction, kind).map_err(|e| ConvertError::runtime(e.to_string()))?;
139    let units = [
140        axes[0].unit().clone(),
141        axes[1].unit().clone(),
142        axes[2].unit().clone(),
143        errors[0].unit().clone(),
144        errors[1].unit().clone(),
145        errors[2].unit().clone(),
146    ];
147    Ok(Some(Prepared {
148        shape,
149        units: Some(units),
150    }))
151}

153enum OptionalFrame<'py> {
154    Value(Bound<'py, PyAny>),
155    Leave(usize),
156}
157
158/// CASA converts, then ignores, this double vector; cycles crash the binding.
159pub(super) fn optional_pending(value: &Bound<'_, PyAny>) -> PyResult<bool> {
160    let ndarray = value.py().import("numpy")?.getattr("ndarray")?;
161    if value.is_instance(&ndarray)? {
162        return Ok(false);
163    }
164    let sequence = value.is_instance_of::<PyList>() || value.is_instance_of::<PyTuple>();
165    if !sequence {
166        if value.is_instance_of::<PyInt>() {
167            return Ok(value.extract::<i64>().is_err());
168        }
169        return Ok(!value.is_instance_of::<PyFloat>()
170            && !value.is_instance_of::<PyString>()
171            && !value.is_instance_of::<PyBytes>());
172    }
173    let mut active = HashSet::new();
174    let mut stack = vec![OptionalFrame::Value(value.clone())];
175    let mut pending = false;
176    while let Some(frame) = stack.pop() {
177        let object = match frame {
178            OptionalFrame::Value(object) => object,
179            OptionalFrame::Leave(identity) => {
180                active.remove(&identity);
181                continue;
182            }
183        };
184        if object.is_instance_of::<PyList>() || object.is_instance_of::<PyTuple>() {
185            let identity = object.as_ptr() as usize;
186            if !active.insert(identity) {
187                return Err(PyValueError::new_err(
188                    "cyclic component double-vector argument (CASA crash: #1053)",
189                ));
190            }
191            let items = object.try_iter()?.collect::<PyResult<Vec<_>>>()?;
192            stack.push(OptionalFrame::Leave(identity));
193            stack.extend(items.into_iter().rev().map(OptionalFrame::Value));
194        } else if object.is_instance_of::<PyInt>() {
195            pending |= object.extract::<i64>().is_err();
196        }
197    }
198    Ok(pending)
199}

201#[pymethods]
202impl PyComponentList {
203    /// Replace one component's shape, preserving its direction and errors.
204    #[pyo3(
205        signature = (which, r#type = None, majoraxis = None, minoraxis = None, positionangle = None, majoraxiserror = None, minoraxiserror = None, positionangleerror = None, optionalparms = None, log = None),
206        text_signature = "($self, which, type='Point', majoraxis=[], minoraxis=[], positionangle=[], majoraxiserror=[], minoraxiserror=[], positionangleerror=[], optionalparms=[0.0], log=True)"
207    )]
208    #[allow(clippy::too_many_arguments)]
209    fn setshape(
210        &self,
211        py: Python<'_>,
212        which: &Bound<'_, PyAny>,
213        #[pyo3(from_py_with = supplied)] r#type: Option<Bound<'_, PyAny>>,
214        #[pyo3(from_py_with = supplied)] majoraxis: Option<Bound<'_, PyAny>>,
215        #[pyo3(from_py_with = supplied)] minoraxis: Option<Bound<'_, PyAny>>,
216        #[pyo3(from_py_with = supplied)] positionangle: Option<Bound<'_, PyAny>>,
217        #[pyo3(from_py_with = supplied)] majoraxiserror: Option<Bound<'_, PyAny>>,
218        #[pyo3(from_py_with = supplied)] minoraxiserror: Option<Bound<'_, PyAny>>,
219        #[pyo3(from_py_with = supplied)] positionangleerror: Option<Bound<'_, PyAny>>,
220        #[pyo3(from_py_with = supplied)] optionalparms: Option<Bound<'_, PyAny>>,
221        #[pyo3(from_py_with = supplied)] log: Option<Bound<'_, PyAny>>,
222    ) -> PyResult<Py<PyAny>> {
223        let (which, mut pending) = match which.extract::<i64>() {
224            Ok(index) => (i64::from(index as i32), false),
225            Err(error) if which.is_none() || error.is_instance_of::<PyOverflowError>(py) => {
226                (-1, true)
227            }
228            Err(_) => return Err(PyTypeError::new_err("argument _which must be an integer")),
229        };
230        let name: String = match r#type {
231            None => "Point".to_string(),
232            Some(value) if value.is_instance_of::<PyString>() => value.extract()?,
233            Some(value) if value.is_instance_of::<PyBytes>() => {
234                let bytes = value.cast::<PyBytes>()?.as_bytes();
235                let end = bytes
236                    .iter()
237                    .position(|&byte| byte == 0)
238                    .unwrap_or(bytes.len());
239                PyBytes::new(py, &bytes[..end])
240                    .call_method1("decode", ("utf-8",))?
241                    .extract()?
242            }
243            Some(_) => return Err(PyTypeError::new_err("argument _type must be a string")),
244        };
245        let name = name.split('\0').next().unwrap_or_default().to_string();
246        let mut convert = |arg| {
247            let (value, has_pending) = setrefdir_args::argument(arg)?;
248            pending |= has_pending;
249            PyResult::Ok(value)
250        };
251        let axes = [
252            convert(majoraxis)?,
253            convert(minoraxis)?,
254            convert(positionangle)?,
255        ];
256        let errors = [
257            convert(majoraxiserror)?,
258            convert(minoraxiserror)?,
259            convert(positionangleerror)?,
260        ];
261        if let Some(value) = optionalparms {
262            pending |= optional_pending(&value)?;
263        }
264        if log.is_some_and(|value| !value.is_instance_of::<PyBool>()) {
265            return Err(PyTypeError::new_err(
266                "in method 'componentlist_setshape', argument 11 of type 'bool'",
267            ));
268        }
269        let request = Request { name, axes, errors };
270        let result = self.call(py, move |state, lines| {
271            state
272                .setshape(which, &request, pending, lines)
273                .map(CasaValue::Bool)
274        })?;
275        if pending {
276            return Err(PySystemError::new_err(
277                "<built-in function componentlist_setshape> returned a result with an exception set",
278            ));
279        }
280        Ok(result)
281    }
