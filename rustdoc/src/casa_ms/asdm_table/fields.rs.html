3use casa_core::{ArrayValue, PrimitiveType, ScalarValue, Value};
4use casa_tables::ColumnSchema;
5use ndarray::{ArrayD, IxDyn};
6use num_complex::Complex32;

8use super::{Result, invalid};

10const SCHEMAS: &str = include_str!("tables.txt");
11
12pub(super) struct Field<'a> {
13    pub name: &'a str,
14    kind: PrimitiveType,
15    array: bool,
16}
17
18pub(super) fn read_schema(name: &str) -> Result<Vec<Field<'static>>> {
19    let row = SCHEMAS
20        .lines()
21        .filter_map(|line| line.split_once(' '))
22        .find_map(|(root, columns)| (root == name).then_some(columns))
23        .ok_or_else(|| invalid(format!("unsupported table {name}")))?;
24    row.split_ascii_whitespace()
25        .map(|spec| {
26            let (name, ty) = spec
27                .split_once(':')
28                .ok_or_else(|| invalid("incorrect built-in field schema"))?;
29            let kind = match ty.to_ascii_uppercase().as_str() {
30                "B" => PrimitiveType::Bool,
31                "I" => PrimitiveType::Int32,
32                "R" => PrimitiveType::Float32,
33                "D" => PrimitiveType::Float64,
34                "X" => PrimitiveType::Complex32,
35                "A" => PrimitiveType::String,
36                _ => return Err(invalid(format!("incorrect built-in type {ty}"))),
37            };
38            Ok(Field {
39                name,
40                kind,
41                array: ty.as_bytes()[0].is_ascii_lowercase(),
42            })
43        })
44        .collect()
45}
46
47impl Field<'_> {
48    pub(super) fn column(&self) -> ColumnSchema {
49        if self.array {
50            ColumnSchema::array_variable(self.name, self.kind, None)
51        } else {
52            ColumnSchema::scalar(self.name, self.kind)
53        }
54    }
55
56    pub(super) fn scalar_default(&self) -> Option<Value> {
57        if self.array {
58            return None;
59        }
60        use ScalarValue as S;
61        Some(Value::Scalar(match self.kind {
62            PrimitiveType::Bool => S::Bool(false),
63            PrimitiveType::Int32 => S::Int32(0),
64            PrimitiveType::Float32 => S::Float32(0.0),
65            PrimitiveType::Float64 => S::Float64(0.0),
66            PrimitiveType::String => S::String(String::new()),
67            _ => unreachable!("ASDM schema declares no other scalars"),
68        }))
69    }
70
71    pub(super) fn value(&self, text: &str) -> Result<Value> {
72        let text = text.trim();
73        if self.array {
74            return self.array_value(text).map(Value::Array);
75        }
76        use ScalarValue as S;
77        Ok(Value::Scalar(match self.kind {
78            PrimitiveType::Bool => S::Bool(self.parse_boolean(text)?),
79            PrimitiveType::Int32 => S::Int32(self.parse(text)?),
80            PrimitiveType::Float32 => S::Float32(self.parse(text)?),
81            PrimitiveType::Float64 => S::Float64(self.parse(text)?),
82            PrimitiveType::String => S::String(text.into()),
83            _ => unreachable!("ASDM schema declares no other scalars"),
84        }))
85    }
86
87    fn array_value(&self, text: &str) -> Result<ArrayValue> {
88        let mut tokens = Tokens(text);
89        let rank: usize = self.parse(tokens.next()?)?;
90        if !(1..=4).contains(&rank) {
91            return Err(invalid(format!("{}: invalid array rank {rank}", self.name)));
92        }
93        let shape = (0..rank)
94            .map(|_| self.parse(tokens.next()?))
95            .collect::<Result<Vec<usize>>>()?;
96        let count = shape
97            .iter()
98            .try_fold(1_usize, |size, len| size.checked_mul(*len))
99            .ok_or_else(|| invalid(format!("{}: array shape overflows", self.name)))?;
100        let malformed = || {
101            invalid(format!(
102                "{}: array dimensions do not match value count",
103                self.name
104            ))
105        };
106        if count > text.len() {
107            return Err(malformed());
108        }
109        macro_rules! build {
110            ($variant:ident, $read:expr) => {{
111                let values = (0..count).map(|_| $read).collect::<Result<Vec<_>>>()?;
112                ArrayValue::$variant(
113                    ArrayD::from_shape_vec(IxDyn(&shape), values).map_err(|_| malformed())?,
114                )
115            }};
116        }
117        let data = match self.kind {
118            PrimitiveType::Bool => build!(Bool, self.parse_boolean(tokens.next()?)),
119            PrimitiveType::Int32 => build!(Int32, self.parse(tokens.next()?)),
120            PrimitiveType::Float64 => build!(Float64, self.parse(tokens.next()?)),
121            PrimitiveType::String => build!(String, Ok(tokens.next()?.into())),
122            PrimitiveType::Complex32 => build!(
123                Complex32,
124                Ok(Complex32::new(
125                    self.parse(tokens.next()?)?,
126                    self.parse(tokens.next()?)?
127                ))
128            ),
129            _ => unreachable!("ASDM schema declares no other arrays"),
130        };
131        if !tokens.0.trim().is_empty() {
132            return Err(malformed());
133        }
134        Ok(data)
135    }
136
137    fn parse<T: std::str::FromStr>(&self, value: &str) -> Result<T> {
138        value
139            .parse()
140            .map_err(|_| invalid(format!("{}: invalid value '{value}'", self.name)))
141    }
142
143    fn parse_boolean(&self, value: &str) -> Result<bool> {
144        match value {
145            "true" | "1" => Ok(true),
146            "false" | "0" => Ok(false),
147            _ => Err(invalid(format!("{}: invalid boolean '{value}'", self.name))),
148        }
149    }
150}
151
152struct Tokens<'a>(&'a str);
153
154impl<'a> Tokens<'a> {
155    fn next(&mut self) -> Result<&'a str> {
156        let rest = self.0.trim_start();
157        if let Some(quoted) = rest.strip_prefix('"') {
158            let (token, remainder) = quoted
159                .split_once('"')
160                .ok_or_else(|| invalid("unclosed quoted array element"))?;
161            self.0 = remainder;
162            return Ok(token);
163        }
164        if rest.is_empty() {
165            return Err(invalid("missing array element"));
166        }
167        let end = rest.find(char::is_whitespace).unwrap_or(rest.len());
168        let (token, remainder) = rest.split_at(end);
169        self.0 = remainder;
170        Ok(token)
171    }
172}
