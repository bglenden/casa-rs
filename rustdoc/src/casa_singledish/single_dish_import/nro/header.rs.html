2//! Instrument and array metadata from the NOSTAR observation header.
3
4use super::binary::{
5    ARRAYS, ByteOrder, OBSERVATION_HEADER_BYTES, Record, SCAN_HEADER_BYTES, invalid,
6};
7use casa_ms::MsResult;
8
9const FIRST_ARRAY_MAP: usize = 15_028;
10const POL_NAMES: usize = 15_100;
11const SPEED_OF_LIGHT: f64 = 299_792_458.;
12
13#[derive(Clone, Copy)]
14pub(super) struct Array {
15    pub beam: i32,
16    pub window: i32,
17    pub code: i32,
18    pub plane: usize,
19    pub name: &'static str,
20    pub scale: f64,
21    pub rest: f64,
22    pub frequency: [f64; 2],
23    pub channel: [f64; 2],
24}
25
26pub(super) struct Header {
27    pub order: ByteOrder,
28    pub n_arrays: usize,
29    pub n_scans: usize,
30    pub n_beam: usize,
31    pub n_windows: usize,
32    pub n_channels: usize,
33    pub record_bytes: usize,
34    pub interval: f64,
35    pub arrays: [Option<Array>; ARRAYS],
36    pub direction_frame: &'static str,
37    pub direction: [f64; 2],
38    pub frequency_frame: i32,
39    pub velocity: f64,
40    pub doppler_factor: f64,
41    pub site: String,
42    pub observer: String,
43    pub project: String,
44    pub object: String,
45}
46
47impl Header {
48    pub(super) fn read(bytes: &[u8; OBSERVATION_HEADER_BYTES], file_size: u64) -> MsResult<Self> {
49        let record = Record::observation(bytes)?;
50        let n_arrays = record.i32(144) as usize;
51        let n_scans = positive(record.i32(148), "scan count")?;
52        let n_beam = positive(record.i32(14_972), "beam count")?;
53        let n_windows = positive(record.i32(14_980), "window count")?;
54        let n_channels = positive(record.i32(620), "channel count")?;
55        let record_bytes = positive(record.i32(14_936), "scan record length")?;
56        let bits = positive(record.i32(14_944), "sample bit width")?;
57        if bits != 12 {
58            return Err(invalid("sample bit width (expected packed 12-bit samples)"));
59        }
60        let payload = n_channels
61            .checked_mul(bits)
62            .ok_or_else(|| invalid("spectrum size"))?
63            .div_ceil(8);
64        if record_bytes < SCAN_HEADER_BYTES || payload > record_bytes - SCAN_HEADER_BYTES {
65            return Err(invalid("spectrum size versus scan record length"));
66        }
67        let raw_bytes = n_arrays
68            .checked_mul(n_scans)
69            .and_then(|rows| rows.checked_mul(record_bytes))
70            .and_then(|data_size| data_size.checked_add(OBSERVATION_HEADER_BYTES));
71        if raw_bytes.is_none_or(|length| length as u64 > file_size) {
72            return Err(invalid("truncated scan records"));
73        }
74        if n_beam > n_arrays || n_windows > n_arrays {
75            return Err(invalid("array, beam or window count"));
76        }
77        let interval = record.f64(640);
78        if !interval.is_finite() || interval <= 0. {
79            return Err(invalid("integration interval"));
80        }
81        let direction_frame = direction_frame(record)?;
82        let velocity = record.f64(456);
83        let doppler_factor = match record.text(468, 4)? {
84            "RAD" => 1. - velocity / SPEED_OF_LIGHT,
85            "OPT" => (1. + velocity / SPEED_OF_LIGHT).recip(),
86            _ => 1.,
87        };
88        let mut arrays = [None; ARRAYS];
89        for (index, item) in arrays.iter_mut().enumerate() {
90            *item = array_mapping(record, index, n_beam, n_windows)?;
91        }
92        Ok(Self {
93            order: record.order(),
94            n_arrays,
95            n_scans,
96            n_beam,
97            n_windows,
98            n_channels,
99            record_bytes,
100            interval,
101            arrays,
102            direction_frame,
103            direction: [record.f64(296), record.f64(304)],
104            frequency_frame: match record.text(464, 4)? {
105                "LSR" => 1,
106                "HEL" => 3,
107                "GAL" => 6,
108                _ => 64,
109            },
110            velocity,
111            doppler_factor,
112            site: record.text(14_948, 8)?.into(),
113            observer: record.text(72, 40)?.into(),
114            project: record.text(32, 16)?.into(),
115            object: record.text(272, 16)?.into(),
116        })
117    }
118}
119
120fn array_mapping(
121    record: Record<'_>,
122    index: usize,
123    beams: usize,
124    windows: usize,
125) -> MsResult<Option<Array>> {
126    let mapping = i32::from(record.i16(FIRST_ARRAY_MAP + 2 * index));
127    if mapping < 1101 {
128        return Ok(None);
129    }
130    let beam = mapping / 1000 - 1;
131    let window = mapping % 100 - 1;
132    let pol = (mapping / 100) % 10 - 1;
133    if !(0..beams as i32).contains(&beam) || !(0..windows as i32).contains(&window) || pol < 0 {
134        return Err(invalid("array mapping"));
135    }
136    let (plane, code, name) = match record.text(POL_NAMES + 3 * pol as usize, 3)? {
137        "XX" => (0, 9, "XX"),
138        "YY" => (1, 12, "YY"),
139        "XY" => (2, 10, "XY"),
140        "YX" => (3, 11, "YX"),
141        _ => return Err(invalid("array polarization")),
142    };
143    let calibration = index * 10 * 8;
144    Ok(Some(Array {
145        beam,
146        window,
147        code,
148        plane,
149        name,
150        scale: record.f64(4576 + 8 * index),
151        rest: record.f64(6256 + 8 * index),
152        frequency: [
153            record.f64(6536 + calibration),
154            record.f64(6544 + calibration),
155        ],
156        channel: [
157            record.f64(9336 + calibration),
158            record.f64(9344 + calibration),
159        ],
160    }))
161}
162
163fn direction_frame(record: Record<'_>) -> MsResult<&'static str> {
164    match (record.i32(332), record.text(288, 8)?) {
165        (0, "J2000") => Ok("J2000"),
166        (0, "B1950") => Ok("B1950"),
167        (1, _) => Ok("GALACTIC"),
168        (2, _) => Ok("AZEL"),
169        _ => Err(invalid("direction reference frame")),
170    }
171}
172
173fn positive(value: i32, field: &str) -> MsResult<usize> {
174    usize::try_from(value)
175        .ok()
176        .filter(|&value| value != 0)
177        .ok_or_else(|| invalid(field))
178}
