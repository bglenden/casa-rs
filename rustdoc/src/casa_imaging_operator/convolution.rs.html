9/// Taps of one convolution-function cell in the layout the kernels read.
10///
11/// Fine-offset rows are indexed by [`CellLocation`](crate::CellLocation)'s
12/// `ox`/`oy`, which run over `0..=oversampling`, so every layout holds
13/// `oversampling + 1` rows per axis and `oversampling` must be even.
14#[derive(Clone, Copy, Debug)]
15pub enum TapLayout<'a> {
16    /// Separable real kernel (standard spheroidal). `rows[o * support + i]`
17    /// is tap `i`, at grid offset `i − support/2` from the anchor, for fine
18    /// offset row `o`; one fractional offset is one contiguous row
19    /// (oversampling-major, Obit `ConvFunc`).
20    SeparableReal {
21        /// `(oversampling + 1) × support` tap values.
22        rows: &'a [f32],
23        /// Taps per axis (odd).
24        support: u16,
25        /// Fine offsets per cell (even).
26        oversampling: u16,
27    },
28    /// Dense complex kernel (W, AW, mosaic) ordered
29    /// `[oy][ox][mueller][iy][ix]`, x fastest, with one contiguous
30    /// `mueller × sy × sx` tile per fractional offset. Tap `(ix, iy)` sits
31    /// at grid offset `(ix − sx/2, iy − sy/2)` from the anchor.
32    Dense {
33        /// `(oversampling + 1)² × mueller_planes × sy × sx` tap values.
34        data: &'a [Complex32],
35        /// Taps per axis `[sx, sy]` (odd).
36        support: [u16; 2],
37        /// Fine offsets per cell (even).
38        oversampling: u16,
39        /// Mueller planes per tile; 1 for a scalar kernel.
40        mueller_planes: u8,
41    },
42}
43
44impl TapLayout<'_> {
45    /// Taps on each side of the anchor, `[x, y]`.
46    #[must_use]
47    pub fn half_support(&self) -> [u16; 2] {
48        match self {
49            Self::SeparableReal { support, .. } => [support / 2, support / 2],
50            Self::Dense { support, .. } => [support[0] / 2, support[1] / 2],
51        }
52    }
53
54    /// Fine offsets per cell.
55    #[must_use]
56    pub fn oversampling(&self) -> u16 {
57        match self {
58            Self::SeparableReal { oversampling, .. } | Self::Dense { oversampling, .. } => {
59                *oversampling
60            }
61        }
62    }
63
64    /// Mueller planes per cell.
65    #[must_use]
66    pub fn mueller_planes(&self) -> u8 {
67        match self {
68            Self::SeparableReal { .. } => 1,
69            Self::Dense { mueller_planes, .. } => *mueller_planes,
70        }
71    }
72}

74/// Which kernel Mueller plane serves each (grid polarization, visibility
75/// polarization) pair.
76///
77/// `direct` is the adjoint table for `w > 0`; `conjugate` its partner for
78/// `w ≤ 0`. The kernel swaps the tables on the sign of `w` and on the
79/// direction of the transform and conjugates the taps (HPG
80/// `mueller_indexes` / `conjugate_mueller_indexes`). `None` skips a pair.
81/// Both tables are `[grid pol][visibility pol]`.
82#[derive(Clone, Debug, PartialEq, Eq)]
83pub struct MuellerRouting {
84    /// Adjoint table for `w > 0`.
85    pub direct: Vec<Vec<Option<u8>>>,
86    /// Adjoint table for `w ≤ 0`.
87    pub conjugate: Vec<Vec<Option<u8>>>,
88}
89
90impl MuellerRouting {
91    /// Scalar routing: Mueller plane 0 for every pair `pol_map` joins
92    /// (visibility polarization `p` feeds grid polarization `pol_map[p]`),
93    /// with identical direct and conjugate tables.
94    #[must_use]
95    pub fn scalar(pol_map: &[Option<u8>], grid_pols: usize) -> Self {
96        let direct = (0..grid_pols)
97            .map(|gpol| {
98                pol_map
99                    .iter()
100                    .map(|target| (*target == Some(gpol as u8)).then_some(0))
101                    .collect::<Vec<_>>()
102            })
103            .collect::<Vec<_>>();
104        Self {
105            conjugate: direct.clone(),
106            direct,
107        }
108    }
109
110    /// Number of grid polarizations.
111    #[must_use]
112    pub fn grid_pols(&self) -> usize {
113        self.direct.len()
114    }
115
116    /// Number of visibility polarizations.
117    #[must_use]
118    pub fn visibility_pols(&self) -> usize {
119        self.direct.first().map_or(0, Vec::len)
120    }
121
122    /// The table for one transform: the adjoint uses `direct` for `w > 0`,
123    /// the forward transform swaps them.
124    #[must_use]
125    pub fn table(&self, w_positive: bool, forward: bool) -> &[Vec<Option<u8>>] {
126        if w_positive != forward {
127            &self.direct
128        } else {
129            &self.conjugate
130        }
131    }
132}

134/// Per-row context a kernel set keys its cells on.
135#[derive(Clone, Copy, Debug, PartialEq)]
136pub struct RowContext {
137    /// Row time in seconds (MeasurementSet `TIME`).
138    pub time_s: f64,
139    /// Antenna pair.
140    pub antennas: [u32; 2],
141    /// Parallactic angle of each antenna in radians.
142    pub parallactic_angle_rad: [f64; 2],
143    /// Field (pointing) identifier.
144    pub field: u32,
145    /// Pointing offset from the image phase centre in radians `[Δl, Δm]`.
146    pub pointing_offset_rad: [f64; 2],
147}

149/// Paired image-domain gridding correction: one real vector per grid axis,
150/// applied as `x[gx] · y[gy]` on both sides of the operator (CASA divides
151/// the model and the dirty image by the same spheroidal response).
152#[derive(Clone, Debug, PartialEq)]
153pub struct ImageCorrection {
154    x: Box<[f64]>,
155    y: Box<[f64]>,
156}
157
158impl ImageCorrection {
159    /// Correction vectors over the padded grid axes.
160    #[must_use]
161    pub fn new(x: Vec<f64>, y: Vec<f64>) -> Self {
162        Self {
163            x: x.into_boxed_slice(),
164            y: y.into_boxed_slice(),
165        }
166    }
167
168    /// Correction along the grid x axis.
169    #[must_use]
170    pub fn x(&self) -> &[f64] {
171        &self.x
172    }
173
174    /// Correction along the grid y axis.
175    #[must_use]
176    pub fn y(&self) -> &[f64] {
177        &self.y
178    }
179
180    /// Correction at grid cell `(gx, gy)`.
181    #[must_use]
182    pub fn at(&self, gx: usize, gy: usize) -> f64 {
183        self.x[gx] * self.y[gy]
184    }
185}

187/// A kernel set: pure cell selection plus the taps, routing and correction
188/// the kernels read. Implementations: the standard spheroidal set, W-planes,
189/// the AW catalog and the mosaic primary beam.
190pub trait ConvolutionFunctionSet: Send + Sync {
191    /// Cell for one row at `freq_hz` and `w_lambda`. Pure; CASA-pinned
192    /// rounding rules for w-planes, frequency and parallactic-angle cells
193    /// live here.
194    fn key(&self, row: &RowContext, freq_hz: f64, w_lambda: f64) -> CfKey;
195
196    /// Imaging taps of a cell.
197    fn taps(&self, key: CfKey) -> TapLayout<'_>;
198
199    /// `FT[PB²]` taps for the weight (sensitivity) image, placed at the uv
200    /// origin; `None` for sets without a weight image.
201    fn weight_taps(&self, key: CfKey) -> Option<TapLayout<'_>>;
202
203    /// Mueller plane routing for every cell of the set.
204    fn mueller(&self) -> &MuellerRouting;
205
206    /// The paired image-domain correction.
207    fn image_correction(&self) -> &ImageCorrection;
208}
