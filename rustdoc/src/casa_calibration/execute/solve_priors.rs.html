2//! Place diagonal prior calibration on the correct side of a delay solve.
3
4use super::{
5    ApplyExecutionError, ApplyPlan, ArrayValue, EvaluatedApplyRow, HashMap, MeasurementSet,
6    evaluate_apply_rows, for_each_apply_row,
7};
8
9pub(crate) fn evaluate_delay_priors(
10    ms: &MeasurementSet,
11    mut plan: ApplyPlan,
12    share_input_flags: bool,
13) -> Result<HashMap<usize, EvaluatedApplyRow>, ApplyExecutionError> {
14    // VisEquation::collapse2 corrects DATA with earlier terms and corrupts
15    // MODEL_DATA with terms at or after K. This distinction matters even for
16    // diagonal terms because CASA leaves data over a zero model undivided.
17    let (data_tables, mut model_tables): (Vec<_>, Vec<_>) =
18        std::mem::take(&mut plan.calibration_tables)
19            .into_iter()
20            .partition(|table| {
21                matches!(
22                    table.summary.table_subtype.as_str(),
23                    "B Jones" | "BPOLY" | "KAntPos Jones"
24                )
25            });
26    plan.calibration_tables = data_tables;
27    let mut rows = evaluate_apply_rows(ms, &plan, share_input_flags)?;
28    if model_tables.is_empty() {
29        return Ok(rows);
30    }
31    for table in &mut model_tables {
32        table.calwt = false;
33    }
34    plan.calibration_tables = model_tables;
35    plan.parang = false;
36    for_each_apply_row(ms, &plan, true, share_input_flags, |index, row| {
37        let data = rows.get_mut(&index).expect("same selection for both sides");
38        let ArrayValue::Bool(data_flags) = &mut data.flags else {
39            unreachable!("apply flags are boolean")
40        };
41        let ArrayValue::Bool(model_flags) = row.flags else {
42            unreachable!("model calibration flags are boolean")
43        };
44        data_flags.zip_mut_with(&model_flags, |data, &model| *data |= model);
45        let ArrayValue::Complex32(gains) = row.corrected_data else {
46            unreachable!("unit visibilities are complex")
47        };
48        data.model_gains = Some(gains);
49    })?;
50    Ok(rows)
51}
