6use casa_tables::Table;
7use casa_types::{ArrayValue, RecordValue, Value};

9use super::asap_row::{doubles, integer, invalid, number, string};
10use super::{DAY, direction, tag_rows};

14pub(super) fn split_sources(
15    rows: &[&RecordValue],
16    molecules: &Table,
17) -> MsResult<(Vec<Source>, HashMap<String, i32>)> {
18    let molecule_rows = tag_rows(molecules)?;
19    let mut sources = BTreeMap::new();
20    for &row in rows {
21        let key = (string(row, "SRCNAME")?, integer(row, "IFNO")?);
22        let time = number(row, "TIME")?;
23        let interval = number(row, "INTERVAL")?;
24        let times = sources
25            .entry(key)
26            .or_insert((row, time, time, interval, interval));
27        if time < times.1 {
28            times.1 = time;
29            times.3 = interval;
30        }
31        if time > times.2 {
32            times.2 = time;
33            times.4 = interval;
34        }
35    }
36    let mut ids = HashMap::new();
37    let mut result = Vec::with_capacity(sources.len());
38    for ((name, window), (row, first, last, first_interval, last_interval)) in sources {
39        let next_id = ids.len() as i32;
40        let id = *ids.entry(name.to_string()).or_insert(next_id);
41        let before = first * DAY - 0.5 * first_interval;
42        let after = last * DAY + 0.5 * last_interval;
43        let molecule = molecule_rows.get(&integer(row, "MOLECULE_ID")?);
44        let rest_frequencies = molecule
45            .and_then(|value| doubles(value, "RESTFREQUENCY").ok())
46            .unwrap_or_default();
47        let transitions = match molecule.and_then(|value| value.get("NAME")) {
48            Some(Value::Array(ArrayValue::String(values))) => values.iter().cloned().collect(),
49            _ => Vec::new(),
50        };
51        result.push(Source {
52            id,
53            name: name.into(),
54            window,
55            time: 0.5 * (before + after),
56            interval: after - before,
57            direction: doubles(row, "SRCDIRECTION")?
58                .try_into()
59                .map_err(|_| invalid("SRCDIRECTION"))?,
60            proper_motion: doubles(row, "SRCPROPERMOTION")?
61                .try_into()
62                .map_err(|_| invalid("SRCPROPERMOTION"))?,
63            rest_frequencies,
64            transitions,
65            velocity: number(row, "SRCVELOCITY")?,
66        });
67    }
68    Ok((result, ids))
69}

71pub(super) fn fields(
72    rows: &[&RecordValue],
73    sources: &HashMap<String, i32>,
74) -> MsResult<(Vec<Field>, HashMap<String, i32>)> {
75    let mut by_name = BTreeMap::new();
76    for &row in rows {
77        by_name.entry(string(row, "FIELDNAME")?).or_insert(row);
78    }
79    let mut available = (0..by_name.len() as i32).collect::<BTreeSet<_>>();
80    let mut ids = HashMap::new();
81    let mut result = Vec::with_capacity(by_name.len());
82    for (label, row) in by_name {
83        let (name, claim) = label.split_once("__").unwrap_or((label, ""));
84        let requested = claim.parse::<i32>().ok().filter(|id| available.remove(id));
85        let id = requested
86            .or_else(|| available.pop_first())
87            .ok_or_else(|| invalid("FIELDNAME"))?;
88        ids.insert(label.to_string(), id);
89        result.push(Field {
90            id,
91            name: name.into(),
92            source_id: sources[string(row, "SRCNAME")?],
93            time: number(row, "TIME")? * DAY,
94            direction: direction(row, "SRCDIRECTION")?,
95        });
96    }
97    result.sort_unstable_by_key(|field| field.id);
98    Ok((result, ids))
99}
