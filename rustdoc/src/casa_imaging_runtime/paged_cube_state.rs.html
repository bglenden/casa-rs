25/// Geometry and exact physical values layout for one reconstruction-owned array.
26///
27/// Scalar cardinality comes from the reconstruction owner. Cache tiles and the
28/// maximum accessed window are independent: sequential I/O can retain one
29/// plane while the caller holds a multi-plane window.
30#[derive(Debug, Clone)]
31pub(crate) struct CubeArrayLayout {
32    logical_scalars: usize,
33    window_scalars: usize,
34    values: TiledArrayStorageLayout,
35}
36
37impl CubeArrayLayout {
38    pub(crate) fn new(
39        logical_scalars: usize,
40        plane_scalars: usize,
41        window_scalars: usize,
42        cache_tiles: usize,
43    ) -> Result<Self, String> {
44        if logical_scalars == 0 || plane_scalars == 0 || window_scalars == 0 || cache_tiles == 0 {
45            return Err("cube array dimensions, window and cache must be positive".into());
46        }
47        let tile_scalars = plane_scalars.min(logical_scalars);
48        let cache_tiles = cache_tiles.min(logical_scalars.div_ceil(tile_scalars));
49        let cache_bytes = tile_scalars
50            .checked_mul(cache_tiles)
51            .and_then(|n| n.checked_mul(size_of::<f64>()))
52            .ok_or("cube array cache capacity overflow")?;
53        let shape = TiledShape::with_tile_shape(vec![logical_scalars], vec![tile_scalars])
54            .map_err(|error| error.to_string())?;
55        let values = PagedArray::<f64>::storage_layout(shape, cache_bytes)
56            .map_err(|error| error.to_string())?;
57        Ok(Self {
58            logical_scalars,
59            window_scalars: window_scalars.min(logical_scalars),
60            values,
61        })
62    }
63
64    fn support_layout(&self) -> Result<TiledArrayStorageLayout, String> {
65        let shape = TiledShape::with_tile_shape(
66            self.values.cube_shape().to_vec(),
67            self.values.tile_shape().to_vec(),
68        )
69        .map_err(|error| error.to_string())?;
70        PagedArray::<bool>::storage_layout(
71            shape,
72            self.values.cache_budget_bytes() / size_of::<f64>(),
73        )
74        .map_err(|error| error.to_string())
75    }
76
77    pub(crate) fn normal_ledger(&self, parent: &Path) -> Result<CubeArrayLedger, String> {
78        let directory = planned_directory(parent, NORMAL_DIRECTORY_PREFIX);
79        let values = directory.join("values");
80        let retained = PagedArray::<f64>::planned_persistent_heap_bytes(&self.values, &values)
81            .map_err(|error| error.to_string())?;
82        let window = checked_bytes(self.window_scalars, size_of::<f64>())?;
83        Ok(CubeArrayLedger {
84            retained_bytes: checked_sum(&[
85                size_of::<PagedNormalArray>(),
86                directory.as_os_str().len(),
87                retained,
88            ])?,
89            cache_bytes: self
90                .values
91                .cache_payload_bytes()
92                .map_err(|error| error.to_string())?,
93            cache_index_bytes: self
94                .values
95                .cache_metadata_bytes()
96                .map_err(|error| error.to_string())?,
97            read_scratch_bytes: checked_sum(&[
98                window,
99                self.values
100                    .slice_scratch_bytes()
101                    .map_err(|error| error.to_string())?,
102            ])?,
103            write_scratch_bytes: checked_sum(&[
104                window,
105                self.values
106                    .slice_scratch_bytes()
107                    .map_err(|error| error.to_string())?,
108            ])?,
109            flush_scratch_bytes: self
110                .values
111                .flush_scratch_bytes()
112                .map_err(|error| error.to_string())?,
113            storage_bytes: self
114                .values
115                .storage_bytes()
116                .map_err(|error| error.to_string())?,
117            file_handles: 1,
118        })
119    }
120}

122/// Complete owned payload/metadata projection for a created private backing.
123///
124/// Prepared factory metadata is counted separately by the factory. Read/write
125/// scratch excludes the reconstruction owner's input/destination window; it
126/// includes this adapter's ndarray copy and underlying typed conversion/stride
127/// buffers. File lengths include standard metadata and packed edge-tile padding,
128/// not filesystem inode/block-allocation overhead or allocator bookkeeping.
129#[derive(Debug, Clone, Copy, PartialEq, Eq)]

131    pub(crate) retained_bytes: usize,
132    pub(crate) cache_bytes: usize,
133    pub(crate) cache_index_bytes: usize,
134    pub(crate) read_scratch_bytes: usize,
135    pub(crate) write_scratch_bytes: usize,
136    pub(crate) flush_scratch_bytes: usize,
137    pub(crate) storage_bytes: usize,
138    pub(crate) file_handles: usize,
139}

188impl CubeBackingMetrics {
189    pub(crate) fn snapshot(&self) -> CubeBackingSnapshot {
190        *self.0.lock().expect("cube backing metrics lock poisoned")
191    }
192
193    pub(crate) fn register(self: &Arc<Self>, ledger: CubeArrayLedger) -> BackingObservation {
194        let mut state = self.0.lock().expect("cube backing metrics lock poisoned");
195        state.live_owned_bytes += ledger.retained_bytes;
196        state.peak_owned_bytes = state.peak_owned_bytes.max(state.live_owned_bytes);
197        state.live_cache_bytes += ledger.cache_bytes;
198        state.peak_cache_bytes = state.peak_cache_bytes.max(state.live_cache_bytes);
199        state.live_cache_index_bytes += ledger.cache_index_bytes;
200        state.peak_cache_index_bytes = state
201            .peak_cache_index_bytes
202            .max(state.live_cache_index_bytes);
203        state.live_storage_bytes += ledger.storage_bytes;
204        state.peak_storage_bytes = state.peak_storage_bytes.max(state.live_storage_bytes);
205        state.live_file_handles += ledger.file_handles;
206        state.peak_file_handles = state.peak_file_handles.max(state.live_file_handles);
207        state.live_backings += 1;
208        state.peak_backings = state.peak_backings.max(state.live_backings);
209        BackingObservation {
210            metrics: self.clone(),
211            ledger,
212        }
213    }
214}

216#[derive(Debug)]
217pub(crate) struct BackingObservation {
218    metrics: Arc<CubeBackingMetrics>,
219    ledger: CubeArrayLedger,
220}
221
222impl BackingObservation {
223    pub(crate) fn record(&self, scalars: usize, delta: TiledFileIoStats) {
224        let mut state = self
225            .metrics
226            .0
227            .lock()
228            .expect("cube backing metrics lock poisoned");
229        state.max_access_scalars = state.max_access_scalars.max(scalars);
230        state.read_bytes +=
231            delta.flat_bulk_read_bytes + delta.lru_read_bytes + delta.lru_batch_load_bytes;
232        state.write_bytes += delta.flat_flush_write_bytes
233            + delta.lru_flush_write_bytes
234            + delta.lru_batch_flush_bytes
235            + delta.direct_tile_write_bytes;
236    }
237}
238
239impl Drop for BackingObservation {
240    fn drop(&mut self) {
241        let mut state = self
242            .metrics
243            .0
244            .lock()
245            .expect("cube backing metrics lock poisoned");
246        state.live_owned_bytes -= self.ledger.retained_bytes;
247        state.live_cache_bytes -= self.ledger.cache_bytes;
248        state.live_cache_index_bytes -= self.ledger.cache_index_bytes;
249        state.live_storage_bytes -= self.ledger.storage_bytes;
250        state.live_file_handles -= self.ledger.file_handles;
251        state.live_backings -= 1;
252    }
253}

255fn array_file_bytes(directory: &Path) -> Result<usize, String> {
256    ["table.dat", "table.info", "table.f0", "table.f0_TSM0"]
257        .iter()
258        .try_fold(0usize, |sum, name| {
259            let bytes = std::fs::metadata(directory.join(name))
260                .map_err(|error| error.to_string())?
261                .len();
262            let bytes =
263                usize::try_from(bytes).map_err(|_| "cube backing file length exceeds usize")?;
264            checked_sum(&[sum, bytes])
265        })
266}
267
268fn planned_directory(parent: &Path, prefix: &str) -> PathBuf {
269    parent.join(format!("{prefix}{}", "x".repeat(DIRECTORY_RANDOM_CHARS)))
270}
271
272fn checked_bytes(count: usize, bytes: usize) -> Result<usize, String> {
273    count
274        .checked_mul(bytes)
275        .ok_or_else(|| "cube backing byte count overflow".into())
276}
277
278fn checked_sum(values: &[usize]) -> Result<usize, String> {
279    values.iter().try_fold(0usize, |sum, &value| {
280        sum.checked_add(value)
281            .ok_or_else(|| "cube backing byte count overflow".into())
282    })
283}

294impl PagedNormalStorageFactory {
295    pub(crate) fn new(
296        parent: &Path,
297        layouts: Box<[(usize, CubeArrayLayout)]>,
298        retentions: Box<[(usize, Arc<dyn std::fmt::Debug + Send + Sync>)]>,
299        metrics: Arc<CubeBackingMetrics>,
300    ) -> Self {
301        Self {
302            parent: parent.into(),
303            layouts,
304            retentions,
305            metrics,
306        }
307    }
308
309    pub(crate) fn owned_metadata_bytes(&self) -> Result<usize, String> {
310        let mut bytes = checked_sum(&[
311            size_of::<Self>(),
312            self.parent.as_os_str().len(),
313            checked_bytes(self.layouts.len(), size_of::<(usize, CubeArrayLayout)>())?,
314            size_of_val(self.retentions.as_ref()),
315        ])?;
316        for (_, layout) in &self.layouts {
317            bytes = checked_sum(&[
318                bytes,
319                layout
320                    .values
321                    .owned_heap_bytes()
322                    .map_err(|error| error.to_string())?,
323            ])?;
324        }
325        Ok(bytes)
326    }

329impl NormalStorageFactory for PagedNormalStorageFactory {
330    fn create(
331        &self,
332        allocation_ordinal: usize,
333        scalars: usize,
334    ) -> Result<Box<dyn NormalArrayStorage>, SpectralOperatorError> {
335        let layout = self
336            .layouts
337            .iter()
338            .find_map(|(ordinal, layout)| (*ordinal == allocation_ordinal).then_some(layout))
339            .filter(|layout| scalars > 0 && scalars <= layout.logical_scalars)
340            .ok_or_else(|| {
341                normal_storage_error("normal backing does not match its domain layout")
342            })?;
343        Ok(Box::new(PagedNormalArray::create(
344            &self.parent,
345            layout,
346            scalars,
347            self.retentions
348                .iter()
349                .find(|(ordinal, _)| *ordinal == allocation_ordinal)
350                .ok_or_else(|| normal_storage_error("normal allocation lacks its retention owner"))?
351                .1
352                .clone(),
353            self.metrics.clone(),
354        )?))
355    }
356}

358/// Exact scalar backing with a fixed tile cache and private file lifetime.
359///
360/// Each read additionally allocates one `f64` slice-result array, and each write
361/// allocates one `f64` ndarray copy of the supplied window. These transient
362/// arrays occupy `8 * window.len()` bytes each, beyond the cache and caller's
363/// window. Construction borrows pre-encoded standard table metadata.
364/// The caller must admit these allocations and bound the windows it supplies.
365#[derive(Debug)]
366pub(crate) struct PagedNormalArray {
367    array: Mutex<PagedArray<f64>>,
368    scalars: usize,
369    window_scalars: usize,
370    // Fields drop in declaration order: close the array before removing its files.
371    #[allow(
372        dead_code,
373        reason = "RAII guard removing the private backing files on drop"
374    )]
375    directory: TempDir,
376    observation: BackingObservation,
377    _retention: Arc<dyn std::fmt::Debug + Send + Sync>,
378}
379
380impl PagedNormalArray {
381    pub(crate) fn create(
382        parent: &Path,
383        layout: &CubeArrayLayout,
384        scalars: usize,
385        retention: Arc<dyn std::fmt::Debug + Send + Sync>,
386        metrics: Arc<CubeBackingMetrics>,
387    ) -> Result<Self, SpectralOperatorError> {
388        if scalars == 0 || scalars > layout.logical_scalars {
389            return Err(normal_storage_error(
390                "normal backing exceeds its planned scalar capacity",
391            ));
392        }
393        let directory = tempfile::Builder::new()
394            .prefix(NORMAL_DIRECTORY_PREFIX)
395            .rand_bytes(DIRECTORY_RANDOM_CHARS)
396            .tempdir_in(parent)
397            .map_err(normal_storage_error)?;
398        let array = PagedArray::create_planned(&layout.values, directory.path().join("values"))
399            .map_err(normal_storage_error)?;
400        let mut ledger = layout.normal_ledger(parent).map_err(normal_storage_error)?;
401        ledger.retained_bytes = checked_sum(&[
402            size_of::<Self>(),
403            directory.path().as_os_str().len(),
404            array
405                .owned_persistent_heap_bytes()
406                .map_err(normal_storage_error)?,
407        ])
408        .map_err(normal_storage_error)?;
409        ledger.storage_bytes =
410            array_file_bytes(&directory.path().join("values")).map_err(normal_storage_error)?;
411        ledger.file_handles = array.owned_file_handles();
412        Ok(Self {
413            array: Mutex::new(array),
414            scalars,
415            window_scalars: layout.window_scalars.min(scalars),
416            directory,
417            observation: metrics.register(ledger),
418            _retention: retention,
419        })
420    }
421
422    #[cfg(test)]
423    pub(crate) fn measurements(&self) -> Result<CubeArrayMeasurements, SpectralOperatorError> {
424        let array = self.array.lock().map_err(normal_storage_error)?;
425        Ok(CubeArrayMeasurements {
426            owned_bytes: checked_sum(&[
427                size_of::<Self>(),
428                self.directory.path().as_os_str().len(),
429                array
430                    .owned_persistent_heap_bytes()
431                    .map_err(normal_storage_error)?,
432            ])
433            .map_err(normal_storage_error)?,
434            storage_bytes: array_file_bytes(&self.directory.path().join("values"))
435                .map_err(normal_storage_error)?,
436            file_handles: array.owned_file_handles(),
437            values_io: array.io_stats(),
438            support_io: None,
439        })
440    }
441
442    fn check_window(&self, start: usize, len: usize) -> Result<(), SpectralOperatorError> {
443        let end = start
444            .checked_add(len)
445            .ok_or(SpectralOperatorError::ResidencyOverflow)?;
446        if end > self.scalars || len > self.window_scalars {
447            return Err(SpectralOperatorError::InvalidSlab);
448        }
449        Ok(())
450    }
451}

453impl NormalArrayStorage for PagedNormalArray {
454    fn len(&self) -> usize {
455        self.scalars
456    }
457
458    fn read(
459        &self,
460        start: usize,
461        len: usize,
462    ) -> Result<std::borrow::Cow<'_, [f64]>, SpectralOperatorError> {
463        self.check_window(start, len)?;
464        if len == 0 {
465            return Ok(std::borrow::Cow::Owned(Vec::new()));
466        }
467        let array = self.array.lock().map_err(normal_storage_error)?;
468        let before = array.io_stats();
469        let window = array
470            .get_slice(&[start], &[len], &[1])
471            .map_err(normal_storage_error)?;
472        if window.shape() != [len] || !window.is_standard_layout() {
473            return Err(normal_storage_error(
474                "normal backing slice is not contiguous",
475            ));
476        }
477        let (values, offset) = window.into_raw_vec_and_offset();
478        if offset != Some(0) || values.len() != len {
479            return Err(normal_storage_error(
480                "normal backing slice has invalid bounds",
481            ));
482        }
483        self.observation
484            .record(len, array.io_stats().delta_since(before));
485        Ok(std::borrow::Cow::Owned(values))
486    }
487
488    fn write(&mut self, start: usize, values: &[f64]) -> Result<(), SpectralOperatorError> {
489        self.check_window(start, values.len())?;
490        if values.is_empty() {
491            return Ok(());
492        }
493        let array = self.array.get_mut().map_err(normal_storage_error)?;
494        let before = array.io_stats();
495        let window = ArrayD::from_shape_vec(IxDyn(&[values.len()]), values.to_vec())
496            .map_err(normal_storage_error)?;
497        let result = array
498            .put_slice(&window, &[start])
499            .map_err(normal_storage_error);
500        self.observation
501            .record(values.len(), array.io_stats().delta_since(before));
502        result
503    }
504}
505
506impl Drop for PagedNormalArray {
507    fn drop(&mut self) {
508        if let Ok(array) = self.array.get_mut() {
509            let before = array.io_stats();
510            let _ = array.flush();
511            self.observation
512                .record(0, array.io_stats().delta_since(before));
513        }
514    }
515}
516
517fn normal_storage_error(error: impl std::fmt::Display) -> SpectralOperatorError {
518    SpectralOperatorError::NormalStorage(error.to_string())
519}

548    pub(crate) fn ledger(&self) -> Result<CubeArrayLedger, String> {
549        let directory = planned_directory(&self.parent, MODEL_DIRECTORY_PREFIX);
550        let values = PagedArray::<f64>::planned_persistent_heap_bytes(
551            &self.layout.values,
552            &directory.join("values"),
553        )
554        .map_err(|error| error.to_string())?;
555        let support = PagedArray::<bool>::planned_persistent_heap_bytes(
556            &self.support,
557            &directory.join("support"),
558        )
559        .map_err(|error| error.to_string())?;
560        let value_window = checked_bytes(self.layout.window_scalars, size_of::<f64>())?;
561        let support_window = checked_bytes(self.layout.window_scalars, size_of::<bool>())?;
562        let value_scratch = self
563            .layout
564            .values
565            .slice_scratch_bytes()
566            .map_err(|error| error.to_string())?;
567        let support_scratch = self
568            .support
569            .slice_scratch_bytes()
570            .map_err(|error| error.to_string())?;
571        Ok(CubeArrayLedger {
572            retained_bytes: checked_sum(&[
573                size_of::<PagedModelSamples>(),
574                directory.as_os_str().len(),
575                values,
576                support,
577            ])?,
578            cache_bytes: checked_sum(&[
579                self.layout
580                    .values
581                    .cache_payload_bytes()
582                    .map_err(|error| error.to_string())?,
583                self.support
584                    .cache_payload_bytes()
585                    .map_err(|error| error.to_string())?,
586            ])?,
587            cache_index_bytes: checked_sum(&[
588                self.layout
589                    .values
590                    .cache_metadata_bytes()
591                    .map_err(|error| error.to_string())?,
592                self.support
593                    .cache_metadata_bytes()
594                    .map_err(|error| error.to_string())?,
595            ])?,
596            read_scratch_bytes: checked_sum(&[value_window, value_scratch])?.max(checked_sum(&[
597                value_window,
598                support_window,
599                support_scratch,
600            ])?),
601            write_scratch_bytes: checked_sum(&[value_window, value_scratch])?
602                .max(checked_sum(&[support_window, support_scratch])?),
603            flush_scratch_bytes: self
604                .layout
605                .values
606                .flush_scratch_bytes()
607                .map_err(|error| error.to_string())?
608                .max(
609                    self.support
610                        .flush_scratch_bytes()
611                        .map_err(|error| error.to_string())?,
612                ),
613            storage_bytes: checked_sum(&[
614                self.layout
615                    .values
616                    .storage_bytes()
617                    .map_err(|error| error.to_string())?,
618                self.support
619                    .storage_bytes()
620                    .map_err(|error| error.to_string())?,
621            ])?,
622            file_handles: 2,
623        })
624    }

663/// One logical model retains its private files, not its full pixel payload.
664/// Values and support remain independently represented at full precision.
665#[derive(Debug)]
666pub(crate) struct PagedModelSamples {
667    arrays: Mutex<ModelArrays>,
668    samples: usize,
669    window_samples: usize,
670    // Delete after the array handles have flushed and closed.
671    #[allow(
672        dead_code,
673        reason = "RAII guard removing the private backing files on drop"
674    )]
675    directory: TempDir,
676    observation: BackingObservation,
677    _retention: Arc<dyn std::fmt::Debug + Send + Sync>,
678}
679
680#[derive(Debug)]
681struct ModelArrays {
682    values: PagedArray<f64>,
683    support: PagedArray<bool>,
684}
685
686impl PagedModelSamples {
687    pub(crate) fn create(
688        directory: &Path,
689        layout: &CubeArrayLayout,
690        support_layout: &TiledArrayStorageLayout,
691        retention: Arc<dyn std::fmt::Debug + Send + Sync>,
692        metrics: Arc<CubeBackingMetrics>,
693        mut ledger: CubeArrayLedger,
694    ) -> Result<Self, ModelLifecycleError> {
695        let directory = tempfile::Builder::new()
696            .prefix(MODEL_DIRECTORY_PREFIX)
697            .rand_bytes(DIRECTORY_RANDOM_CHARS)
698            .tempdir_in(directory)
699            .map_err(storage_error)?;
700        let values = PagedArray::create_planned(&layout.values, directory.path().join("values"))
701            .map_err(storage_error)?;
702        let support = PagedArray::create_planned(support_layout, directory.path().join("support"))
703            .map_err(storage_error)?;
704        ledger.retained_bytes = checked_sum(&[
705            size_of::<Self>(),
706            directory.path().as_os_str().len(),
707            values
708                .owned_persistent_heap_bytes()
709                .map_err(storage_error)?,
710            support
711                .owned_persistent_heap_bytes()
712                .map_err(storage_error)?,
713        ])
714        .map_err(storage_error)?;
715        ledger.storage_bytes = checked_sum(&[
716            array_file_bytes(&directory.path().join("values")).map_err(storage_error)?,
717            array_file_bytes(&directory.path().join("support")).map_err(storage_error)?,
718        ])
719        .map_err(storage_error)?;
720        ledger.file_handles = values.owned_file_handles() + support.owned_file_handles();
721        Ok(Self {
722            arrays: Mutex::new(ModelArrays { values, support }),
723            samples: layout.logical_scalars,
724            window_samples: layout.window_scalars,
725            directory,
726            observation: metrics.register(ledger),
727            _retention: retention,
728        })
729    }
730
731    #[cfg(test)]
732    pub(crate) fn measurements(&self) -> Result<CubeArrayMeasurements, ModelLifecycleError> {
733        let arrays = self.arrays.lock().map_err(storage_error)?;
734        Ok(CubeArrayMeasurements {
735            owned_bytes: checked_sum(&[
736                size_of::<Self>(),
737                self.directory.path().as_os_str().len(),
738                arrays
739                    .values
740                    .owned_persistent_heap_bytes()
741                    .map_err(storage_error)?,
742                arrays
743                    .support
744                    .owned_persistent_heap_bytes()
745                    .map_err(storage_error)?,
746            ])
747            .map_err(storage_error)?,
748            storage_bytes: checked_sum(&[
749                array_file_bytes(&self.directory.path().join("values")).map_err(storage_error)?,
750                array_file_bytes(&self.directory.path().join("support")).map_err(storage_error)?,
751            ])
752            .map_err(storage_error)?,
753            file_handles: arrays.values.owned_file_handles() + arrays.support.owned_file_handles(),
754            values_io: arrays.values.io_stats(),
755            support_io: Some(arrays.support.io_stats()),
756        })
757    }
758}

760impl ModelSampleStorage for PagedModelSamples {
761    fn sample_count(&self) -> usize {
762        self.samples
763    }
764
765    fn read(
766        &self,
767        start: usize,
768        destination: &mut [ModelSample],
769    ) -> Result<(), ModelLifecycleError> {
770        if start
771            .checked_add(destination.len())
772            .is_none_or(|end| end > self.samples || destination.len() > self.window_samples)
773        {
774            return Err(ModelLifecycleError::CellOutsideShape);
775        }
776        if destination.is_empty() {
777            return Ok(());
778        }
779        let arrays = self.arrays.lock().map_err(storage_error)?;
780        let before_values = arrays.values.io_stats();
781        let before_support = arrays.support.io_stats();
782        let values = arrays
783            .values
784            .get_slice(&[start], &[destination.len()], &[1])
785            .map_err(storage_error)?;
786        let support = arrays
787            .support
788            .get_slice(&[start], &[destination.len()], &[1])
789            .map_err(storage_error)?;
790        self.observation.record(
791            destination.len(),
792            arrays.values.io_stats().delta_since(before_values),
793        );
794        self.observation.record(
795            destination.len(),
796            arrays.support.io_stats().delta_since(before_support),
797        );
798        let values = values
799            .as_slice()
800            .ok_or_else(|| storage_error("model value window is not contiguous"))?;
801        let support = support
802            .as_slice()
803            .ok_or_else(|| storage_error("model support window is not contiguous"))?;
804        for ((destination, value), supported) in destination
805            .iter_mut()
806            .zip(values.iter().copied())
807            .zip(support.iter().copied())
808        {
809            *destination = if supported {
810                ModelSample::valid(ModelValue::new(value)?)
811            } else if value == 0.0 {
812                ModelSample::invalid()
813            } else {
814                return Err(ModelLifecycleError::InvalidSupportPayload);
815            };
816        }
817        Ok(())
818    }
819
820    fn write(&mut self, start: usize, samples: &[ModelSample]) -> Result<(), ModelLifecycleError> {
821        if start
822            .checked_add(samples.len())
823            .is_none_or(|end| end > self.samples || samples.len() > self.window_samples)
824        {
825            return Err(ModelLifecycleError::CellOutsideShape);
826        }
827        if samples.is_empty() {
828            return Ok(());
829        }
830        let arrays = self.arrays.get_mut().map_err(storage_error)?;
831        let before_values = arrays.values.io_stats();
832        let before_support = arrays.support.io_stats();
833        let values = ArrayD::from_shape_vec(
834            IxDyn(&[samples.len()]),
835            samples
836                .iter()
837                .map(|sample| sample.value().value())
838                .collect(),
839        )
840        .map_err(storage_error)?;
841        arrays
842            .values
843            .put_slice(&values, &[start])
844            .map_err(storage_error)?;
845        drop(values);
846        let support = ArrayD::from_shape_vec(
847            IxDyn(&[samples.len()]),
848            samples
849                .iter()
850                .map(|sample| sample.support() == ModelSupport::Valid)
851                .collect(),
852        )
853        .map_err(storage_error)?;
854        arrays
855            .support
856            .put_slice(&support, &[start])
857            .map_err(storage_error)?;
858        self.observation.record(
859            samples.len(),
860            arrays.values.io_stats().delta_since(before_values),
861        );
862        self.observation.record(
863            samples.len(),
864            arrays.support.io_stats().delta_since(before_support),
865        );
866        Ok(())
867    }
868}
869
870impl Drop for PagedModelSamples {
871    fn drop(&mut self) {
872        if let Ok(arrays) = self.arrays.get_mut() {
873            let before = arrays.values.io_stats();
874            let _ = arrays.values.flush();
875            self.observation
876                .record(0, arrays.values.io_stats().delta_since(before));
877            let before = arrays.support.io_stats();
878            let _ = arrays.support.flush();
879            self.observation
880                .record(0, arrays.support.io_stats().delta_since(before));
881        }
882    }
883}

886    ModelLifecycleError::Storage(error.to_string())
