2//! Small normal equations with CASA's collinearity and initial-parameter rules.
3
4use casa_numerics::invert_symmetric_positive_definite;
5
6const MAX_TERMS: usize = 3;
7type Coefficients = [f64; MAX_TERMS];
8type Matrix = [Coefficients; MAX_TERMS];
9
10pub(super) struct System {
11    terms: usize,
12    matrix: Matrix,
13    known: Coefficients,
14}
15
16pub(super) struct Solution {
17    pub(super) coefficients: Vec<f64>,
18    pub(super) covariance: Vec<Vec<f64>>,
19}
20
21impl System {
22    pub(super) fn new(terms: usize) -> Self {
23        Self {
24            terms,
25            matrix: [[0.; MAX_TERMS]; MAX_TERMS],
26            known: [0.; MAX_TERMS],
27        }
28    }
29
30    pub(super) fn accumulate(&mut self, x: f64, y: f64, sigma: f64) -> Option<()> {
31        let basis = [1., x, x * x];
32        let weight = sigma.recip().powi(2);
33        if !weight.is_finite() {
34            return None;
35        }
36        // LinearFit starts with all scaled coefficients at one. In a
37        // deficient fit, the minimum-norm update preserves that initial
38        // polynomial's component in the numerical null space.
39        let delta = y - basis[..self.terms].iter().sum::<f64>();
40        for (row, &value) in basis[..self.terms].iter().enumerate() {
41            let weighted_value = weight * value;
42            self.known[row] += weighted_value * delta;
43            for (column, &source) in basis[..self.terms].iter().enumerate().skip(row) {
44                self.matrix[row][column] += weighted_value * source;
45                self.matrix[column][row] = self.matrix[row][column];
46            }
47        }
48        Some(())
49    }
50
51    pub(super) fn solve(&self) -> Option<Solution> {
52        let independent = self.independent();
53        let rank = independent.len();
54        let reduced = independent
55            .iter()
56            .flat_map(|&i| independent.iter().map(move |&j| self.matrix[i][j]))
57            .collect::<Vec<_>>();
58        let inverse = invert_symmetric_positive_definite(&reduced, rank)?;
59        let distribute = self.distribute(&independent, &inverse)?;
60        let correction = distribute
61            .iter()
62            .map(|row| {
63                row.iter()
64                    .zip(&independent)
65                    .map(|(value, &index)| value * self.known[index])
66                    .sum()
67            })
68            .collect::<Vec<f64>>();
69        let covariance = if rank == self.terms {
70            distribute
71        } else {
72            distribute
73                .iter()
74                .map(|row| {
75                    distribute
76                        .iter()
77                        .map(|other| {
78                            row.iter()
79                                .zip(&independent)
80                                .map(|(left, &i)| {
81                                    left * other
82                                        .iter()
83                                        .zip(&independent)
84                                        .map(|(right, &j)| self.matrix[i][j] * right)
85                                        .sum::<f64>()
86                                })
87                                .sum::<f64>()
88                        })
89                        .collect()
90                })
91                .collect()
92        };
93        let coefficients = correction.into_iter().map(|value| value + 1.).collect();
94        Some(Solution {
95            coefficients,
96            covariance,
97        })
98    }
99
100    fn independent(&self) -> Vec<usize> {
101        let mut normal = self.matrix;
102        let mut terms = (0..self.terms).collect::<Vec<_>>();
103        let mut rank = 0;
104        while rank < terms.len() {
105            let index = terms[rank];
106            let original = self.matrix[index][index];
107            let pivot = normal[index][index];
108            // LSQFit compares the squared Schur pivot to the original
109            // diagonal, using the LinearFitSVD collinearity factor 1e-8 squared.
110            // Rank can change when uncertainty weights change.
111            if original == 0. || pivot.powi(2) / original <= 1e-16 {
112                terms.swap_remove(rank);
113                continue;
114            }
115            for &row in &terms[rank + 1..] {
116                for &column in &terms[rank + 1..] {
117                    normal[row][column] -= normal[index][row] * normal[index][column] / pivot;
118                }
119            }
120            rank += 1;
121        }
122        terms
123    }
124
125    fn distribute(&self, independent: &[usize], inverse: &[f64]) -> Option<Vec<Vec<f64>>> {
126        let rank = independent.len();
127        if rank == self.terms {
128            return Some(inverse.chunks_exact(rank).map(<[f64]>::to_vec).collect());
129        }
130        // Write every original column as C_j in the independent basis.
131        // The minimum-norm update is C^T (C C^T)^-1 A^-1 b, where A is
132        // the independent normal matrix. Keep the same map for covariance.
133        let basis = (0..self.terms)
134            .map(|original| {
135                (0..rank)
136                    .map(|row| {
137                        independent
138                            .iter()
139                            .enumerate()
140                            .map(|(column, &index)| {
141                                inverse[row * rank + column] * self.matrix[index][original]
142                            })
143                            .sum::<f64>()
144                    })
145                    .collect::<Vec<_>>()
146            })
147            .collect::<Vec<_>>();
148        let gram = (0..rank)
149            .flat_map(|row| {
150                let basis = &basis;
151                (0..rank).map(move |column| basis.iter().map(|term| term[row] * term[column]).sum())
152            })
153            .collect::<Vec<_>>();
154        let normalized = invert_symmetric_positive_definite(&gram, rank)?;
155        let map = basis
156            .iter()
157            .map(|term| {
158                (0..rank)
159                    .map(|column| {
160                        (0..rank)
161                            .map(|row| {
162                                term[row]
163                                    * (0..rank)
164                                        .map(|inner| {
165                                            normalized[row * rank + inner]
166                                                * inverse[inner * rank + column]
167                                        })
168                                        .sum::<f64>()
169                            })
170                            .sum()
171                    })
172                    .collect()
173            })
174            .collect();
175        Some(map)
176    }
177}
