12/// A scientific group: the samples a consumer treats together, such as one
13/// gain solution interval. Batches and storage chunks do not bound it.
14///
15/// Keyed as production `gaincal` keys solutions: by partition (spectral
16/// window, observation and intent), field unless the request combines
17/// fields, scan unless it combines scans, and the time bin counted from the
18/// first time of the (field, scan) run.
19#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
20pub struct GroupKey {
21    /// The partition. Groups never span partitions; a consumer that
22    /// combines spectral windows merges groups by (field, scan, bin) across
23    /// cursors.
24    pub partition: PartitionId,
25    /// The field, unless the request combines fields.
26    pub field: Option<FieldId>,
27    /// The scan, unless the request combines scans.
28    pub scan: Option<ScanId>,
29    /// The time bin within the (field, scan) run.
30    pub bin: u64,
31}

33/// Where a fragment lies in its group.
34#[derive(Clone, Copy, Debug, PartialEq, Eq)]
35pub enum FragmentPosition {
36    /// The whole group: no earlier or later fragments.
37    Whole,
38    /// The group's first fragment; more follow.
39    First,
40    /// Neither first nor last.
41    Middle,
42    /// The group's last fragment.
43    Last,
44}
45
46impl FragmentPosition {
47    /// Whether this fragment completes its group.
48    #[must_use]
49    pub fn closes(self) -> bool {
50        matches!(self, Self::Whole | Self::Last)
51    }
52}

54/// The samples of one group within one batch: whole times, so a contiguous
55/// range of the batch's samples.
56#[derive(Clone, Debug, PartialEq, Eq)]
57pub struct GroupFragment {
58    /// The group.
59    pub key: GroupKey,
60    /// The batch's samples that belong to it: whole times, in order.
61    pub samples: Range<usize>,
62    /// Where the fragment lies in its group. `Last` or `Whole` is given only
63    /// when the source has proven no later sample belongs to the group;
64    /// otherwise a later [`ReadEvent::GroupEnd`] closes it.
65    ///
66    /// Fragments belong to the delivered batch, never to compute windows of
67    /// it: [`VisibilityBatch::channel_window`] and
68    /// [`VisibilityBatch::time_window`] carry none, so a closing position
69    /// is seen once, however many work units share the batch.
70    pub position: FragmentPosition,
71}

73/// Joins the two completions of a group that concurrent work consumes: its
74/// input (a closing fragment or [`ReadEvent::GroupEnd`]) and the work units
75/// begun on its fragments or windows. A group is ready to finalize only
76/// when both have happened.
77///
78/// A consumer calls [`Self::begin`] for each unit of work it starts on a
79/// group, which hands out a [`WorkToken`]; gives the token to
80/// [`Self::finish`] when that unit is done; and calls [`Self::close`] when
81/// the source completes the group's input. Whichever call makes the group
82/// ready returns `true`, once. Tokens cannot be copied, so a unit finishes
83/// once and only begun work can finish.
84///
85/// A token dropped unfinished (its unit panicked, failed or was cancelled)
86/// abandons its group: the group never becomes ready, and the next
87/// [`Self::finish`], [`Self::close`] or [`Self::is_idle`] reports
88/// [`ObservationError::WorkAbandoned`] rather than waiting for work that
89/// will not come.
90///
91/// One completion serves one traversal. A source completes the input of a
92/// (partition, field, scan) series' groups in bin order, since it delivers
93/// whole times in time order, so the completion keeps only each series'
94/// latest closed bin to refuse a second closure or work begun after one:
95/// its memory is the series and the groups in flight, however long the
96/// traversal.
97#[derive(Debug, Default)]
98pub struct GroupCompletion {
99    pending: BTreeMap<GroupKey, usize>,
100    /// Groups whose input is complete and whose work is still pending.
101    closed: BTreeSet<GroupKey>,
102    /// Each series' latest bin whose input completed.
103    closed_through: BTreeMap<Series, u64>,
104    /// Groups whose tokens were dropped unfinished, until counted.
105    abandoned: Arc<Mutex<Vec<GroupKey>>>,
106    /// Groups with abandoned work.
107    failed: BTreeSet<GroupKey>,
108}
109
110/// A group's series: its key without the bin.
111type Series = (PartitionId, Option<FieldId>, Option<ScanId>);
112
113fn series(key: GroupKey) -> Series {
114    (key.partition, key.field, key.scan)
115}

117/// One unit of work begun on a group ([`GroupCompletion::begin`]). It ends
118/// when it is handed to [`GroupCompletion::finish`]; it is `Send`, so it can
119/// travel with the work and come back with its result. Dropping it
120/// unfinished abandons the group.
121#[derive(Debug)]
122#[must_use = "a unit of work ends only when its token is finished"]
123pub struct WorkToken {
124    key: GroupKey,
125    /// Where an unfinished drop reports the group; `None` once finished.
126    abandoned: Option<Arc<Mutex<Vec<GroupKey>>>>,
127}
128
129impl WorkToken {
130    /// The group the work is on.
131    #[must_use]
132    pub fn key(&self) -> GroupKey {
133        self.key
134    }
135}
136
137impl Drop for WorkToken {
138    fn drop(&mut self) {
139        if let Some(abandoned) = self.abandoned.take() {
140            abandoned
141                .lock()
142                .unwrap_or_else(PoisonError::into_inner)
143                .push(self.key);
144        }
145    }
146}
147
148impl GroupCompletion {
149    /// A unit of work on `key` has started; finish it with the token.
150    ///
151    /// # Errors
152    ///
153    /// If the group's input has already completed (no work can begin on a
154    /// group after its closure), or its work was abandoned.
155    pub fn begin(&mut self, key: GroupKey) -> Result<WorkToken, ObservationError> {
156        self.settle()?;
157        if self.is_closed(key) {
158            return Err(ObservationError::InvalidBatch {
159                reason: format!("work begun on {key:?} after its input completed"),
160            });
161        }
162        *self.pending.entry(key).or_default() += 1;
163        Ok(WorkToken {
164            key,
165            abandoned: Some(Arc::clone(&self.abandoned)),
166        })
167    }
168
169    /// The unit of work `token` names has finished; `true` if that makes
170    /// its group ready.
171    ///
172    /// # Errors
173    ///
174    /// [`ObservationError::WorkAbandoned`] if a unit of work was abandoned.
175    ///
176    /// # Panics
177    ///
178    /// If `token` was issued by another `GroupCompletion`.
179    pub fn finish(&mut self, mut token: WorkToken) -> Result<bool, ObservationError> {
180        token.abandoned = None;
181        let key = token.key;
182        self.release(key);
183        self.settle()?;
184        Ok(!self.pending.contains_key(&key) && self.closed.remove(&key))
185    }
186
187    /// The group's input is complete; `true` if no work on it is pending,
188    /// so the group is ready now.
189    ///
190    /// # Errors
191    ///
192    /// If the group was already closed, or a later group of its series;
193    /// [`ObservationError::WorkAbandoned`] if a unit of work was abandoned.
194    pub fn close(&mut self, key: GroupKey) -> Result<bool, ObservationError> {
195        self.settle()?;
196        if self.is_closed(key) {
197            return Err(ObservationError::InvalidBatch {
198                reason: format!("{key:?} closed twice, or after a later group of its series"),
199            });
200        }
201        self.closed_through.insert(series(key), key.bin);
202        if self.pending.contains_key(&key) {
203            self.closed.insert(key);
204            Ok(false)
205        } else {
206            Ok(true)
207        }
208    }
209
210    /// Units of work pending on `key`.
211    #[must_use]
212    pub fn pending(&self, key: GroupKey) -> usize {
213        self.pending.get(&key).copied().unwrap_or(0)
214    }
215
216    /// Whether no work is pending and no closed group awaits its work.
217    ///
218    /// # Errors
219    ///
220    /// [`ObservationError::WorkAbandoned`] if a unit of work was abandoned:
221    /// the completion then never becomes idle.
222    pub fn is_idle(&mut self) -> Result<bool, ObservationError> {
223        self.settle()?;
224        Ok(self.pending.is_empty() && self.closed.is_empty())
225    }
226
227    /// Whether `key`'s input has completed: its bin is at or before its
228    /// series' latest closed bin.
229    fn is_closed(&self, key: GroupKey) -> bool {
230        self.closed_through
231            .get(&series(key))
232            .is_some_and(|bin| key.bin <= *bin)
233    }
234
235    /// One unit of work on `key` is over.
236    ///
237    /// # Panics
238    ///
239    /// If none was pending.
240    fn release(&mut self, key: GroupKey) {
241        let pending = self
242            .pending
243            .get_mut(&key)
244            .expect("a token's work is pending in the completion that issued it");
245        *pending -= 1;
246        if *pending == 0 {
247            self.pending.remove(&key);
248        }
249    }
250
251    /// Count the tokens dropped unfinished, and report the first group with
252    /// abandoned work.
253    fn settle(&mut self) -> Result<(), ObservationError> {
254        let abandoned = std::mem::take(
255            &mut *self
256                .abandoned
257                .lock()
258                .unwrap_or_else(PoisonError::into_inner),
259        );
260        for key in abandoned {
261            self.release(key);
262            self.closed.remove(&key);
263            self.failed.insert(key);
264        }
265        match self.failed.first() {
266            Some(&group) => Err(ObservationError::WorkAbandoned { group }),
267            None => Ok(()),
268        }
269    }
270}

272/// What a cursor delivers.
273#[derive(Debug)]
274#[expect(
275    clippy::large_enum_variant,
276    reason = "a batch is a few hundred bytes of views, returned once per batch; boxing it would allocate per batch"
277)]
278pub enum ReadEvent<'a> {
279    /// The next batch.
280    Batch(VisibilityBatch<'a>),
281    /// No further samples of this group will arrive. The group's earlier
282    /// fragments may still be in a consumer's work units: input is complete,
283    /// the work need not be.
284    GroupEnd(GroupKey),
285}

287#[cfg(test)]
288mod tests {
289    use super::*;
290
291    fn key(bin: u64) -> GroupKey {
292        GroupKey {
293            partition: PartitionId(0),
294            field: None,
295            scan: None,
296            bin,
297        }
298    }
299
300    #[test]
301    fn a_group_is_ready_once_after_its_input_and_its_work() {
302        let mut completion = GroupCompletion::default();
303        let first = completion.begin(key(0)).expect("begin");
304        let second = completion.begin(key(0)).expect("begin");
305        assert_eq!(second.key(), key(0));
306        assert!(!completion.close(key(0)).expect("close"), "work pending");
307        assert!(completion.begin(key(0)).is_err(), "no work after closure");
308        assert!(!completion.finish(first).expect("finish"));
309        assert!(completion.finish(second).expect("finish"), "ready once");
310        assert!(completion.close(key(0)).is_err(), "closed twice");
311        assert!(
312            completion.close(key(1)).expect("close"),
313            "no work: ready now"
314        );
315        assert!(completion.close(key(0)).is_err(), "behind its series");
316        assert!(completion.is_idle().expect("idle"));
317        assert_eq!(
318            completion.closed_through.len(),
319            1,
320            "one series, however many groups"
321        );
322    }
323
324    /// A unit that panics with its token, or drops it, abandons the group:
325    /// the join reports it instead of waiting for work that will not come.
326    #[test]
327    fn abandoned_work_is_an_error_not_a_wait() {
328        let mut completion = GroupCompletion::default();
329        let kept = completion.begin(key(0)).expect("begin");
330        let lost = completion.begin(key(0)).expect("begin");
331        let unit = std::thread::spawn(move || {
332            let _token = lost;
333            panic!("the unit fails holding its token");
334        });
335        assert!(unit.join().is_err());
336        assert!(matches!(
337            completion.is_idle(),
338            Err(ObservationError::WorkAbandoned { group }) if group == key(0)
339        ));
340        assert!(completion.close(key(0)).is_err());
341        assert!(
342            completion.finish(kept).is_err(),
343            "its group never becomes ready"
344        );
345        assert!(completion.begin(key(1)).is_err());
346    }
