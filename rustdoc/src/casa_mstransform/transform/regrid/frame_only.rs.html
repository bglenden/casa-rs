30/// Stage a locked MS for a frame-only `regridspw` and retain the unchanged
31/// storage payloads outside the DATA_DESC_ID manager. The source locks must
32/// be held until the caller publishes or removes the staged table.
33///
34/// # Errors
35/// Returns an error if the input descriptor or any copied file is unreadable.
36pub fn stage_frame_only_ms(source: &Path, stage: &Path) -> Result<(), MsTransformError> {
37    let main = Table::open_metadata_only(TableOptions::new(source)).map_err(ms_error)?;
38    let reusable = main
39        .data_manager_info()
40        .iter()
41        .filter(|manager| !manager.columns.iter().any(|name| name == DDID))
42        .map(|manager| manager.seq_nr)
43        .collect::<BTreeSet<_>>();
44    copy_locked_table_tree_with_shared_files(source, stage, |name| {
45        table_manager_file_sequence(name).is_some_and(|manager| reusable.contains(&manager))
46    })
47    .map_err(ms_error)?;
48    private_table_file_metadata(&stage.join("table.dat")).map_err(ms_error)?;
49    Ok(())
50}

52/// Transform the spectral metadata of a private writable MeasurementSet.
53/// This implements the frame-only branch of CASA 6.7.6.14's `SubMS::regridSpw`.
54/// Each changed field/window pair gets its own spectral coordinates at the
55/// first integration of the pair. MAIN rows retain their order and arrays.
56///
57/// # Errors
58/// The private stage is malformed, can't resolve its measures, has an
59/// externally linked table that would need mutation, or cannot be saved.
60pub fn reframe_staged_ms(
61    stage: &Path,
62    options: &MsRegridOptions,
63) -> Result<bool, MsTransformError> {
64    let started = Instant::now();
65    let (mut ms, paths) = open_frame_tables(stage)?;
66    let Some(layout) = Layout::new(&ms, options, None)? else {
67        return Ok(false);
68    };
69    maybe_log_transform_progress("frame_only_plan", started.elapsed(), started.elapsed());
70    let metadata_started = Instant::now();
71    let windows = ms
72        .subtable(SubtableId::SpectralWindow)
73        .ok_or_else(|| invalid("Missing SPECTRAL_WINDOW.".into()))?;
74    let spectral = MsSpectralWindow::new(windows);
75    let geometry = FrameGeometry::open(&ms, options, true)?;
76    let records = layout.window_records(windows, |_, origin, record| {
77        if let Some(frame) = origin.frame_at_reference(options, &geometry)? {
78            update_window(
79                record,
80                &spectral,
81                origin.spw as usize,
82                origin.reference,
83                &frame,
84            )?;
85        }
86        Ok(())
87    })?;
88    let descriptions = ms
89        .subtable(SubtableId::DataDescription)
90        .ok_or_else(|| invalid("Missing DATA_DESCRIPTION.".into()))?;
91    layout.write_subtables(&paths, windows, descriptions, records)?;
92    maybe_log_transform_progress(
93        "frame_only_subtables",
94        metadata_started.elapsed(),
95        started.elapsed(),
96    );
97
98    let descriptions_started = Instant::now();
99    write_main_descriptions(
100        ms.main_table_mut(),
101        &layout.original_descriptions,
102        &layout.row_descriptions,
103    )?;
104    maybe_log_transform_progress(
105        "frame_only_main_descriptions",
106        descriptions_started.elapsed(),
107        started.elapsed(),
108    );
109    Ok(true)
110}

112fn open_frame_tables(stage: &Path) -> Result<(MeasurementSet, LocalTables), MsTransformError> {
113    let main = Table::open(TableOptions::new(stage)).map_err(ms_error)?;
114    let paths = local_tables(&main, stage)?;
115    let read_links = [
116        SubtableId::Antenna,
117        SubtableId::Field,
118        SubtableId::Observation,
119    ];
120    let mut links = HashMap::from([
121        (SubtableId::SpectralWindow, paths.windows.clone()),
122        (SubtableId::DataDescription, paths.descriptions.clone()),
123    ]);
124    if let Some(source) = &paths.source {
125        links.insert(SubtableId::Source, source.clone());
126    }
127    for id in read_links {
128        if let Some(path) = casa_ms::strict_subtable_path(&main, stage, id).map_err(ms_error)? {
129            links.insert(id, path);
130        }
131    }
132    let ms = MeasurementSet::with_open_main(main, stage.to_owned(), &links).map_err(ms_error)?;
133    Ok((ms, paths))
134}

136fn write_main_descriptions(
137    main: &mut Table,
138    original: &[i32],
139    ddids: &[i32],
140) -> Result<(), MsTransformError> {
141    if original == ddids {
142        return Ok(());
143    }
144    let path = main.path().expect("a staged MAIN has a table path");
145    if replace_private_int32_column(path, DDID, ddids).map_err(ms_error)? {
146        return Ok(());
147    }
148    let managers = main
149        .data_manager_info()
150        .iter()
151        .filter(|manager| manager.columns.iter().any(|column| column == DDID))
152        .map(|manager| manager.seq_nr)
153        .collect::<BTreeSet<_>>();
154    for entry in fs::read_dir(path).map_err(ms_error)? {
155        let entry = entry.map_err(ms_error)?;
156        if table_manager_file_sequence(&entry.file_name())
157            .is_some_and(|sequence| managers.contains(&sequence))
158        {
159            private_table_file_metadata(&entry.path()).map_err(ms_error)?;
160        }
161    }
162    let mut changed = Vec::new();
163    {
164        let mut writer = main.row_accessor_mut().prepare(&[DDID]).map_err(ms_error)?;
165        for (row, (&previous, &value)) in original.iter().zip(ddids).enumerate() {
166            if previous == value {
167                continue;
168            }
169            writer.seek(row).map_err(ms_error)?;
170            writer
171                .set_value_at(0, Value::Scalar(ScalarValue::Int32(value)))
172                .map_err(ms_error)?;
173            changed.push(row);
174        }
175    }
176    if !changed.is_empty() {
177        main.prepare_write()
178            .save_selected_rows(&[DDID], &changed)
179            .map_err(ms_error)?;
180    }
181    Ok(())
182}

190fn local_tables(main: &Table, stage: &Path) -> Result<LocalTables, MsTransformError> {
191    let root = stage.canonicalize().map_err(ms_error)?;
192    if !matches!(
193        Table::immediate_part_names(&root, &root).map_err(ms_error)?,
194        TablePartNames::Plain(_)
195    ) {
196        return Err(invalid(
197            "Frame-only regridding needs a physical MAIN table.".into(),
198        ));
199    }
200    let path = |id| {
201        let linked = casa_ms::strict_subtable_path(main, &root, id).map_err(ms_error)?;
202        linked
203            .map(|path| {
204                let path = path.canonicalize().map_err(ms_error)?;
205                if path == root || !path.starts_with(&root) {
206                    return Err(invalid(format!(
207                        "{} lies outside the staging directory.",
208                        id.name()
209                    )));
210                }
211                if !matches!(
212                    Table::immediate_part_names(&path, &path).map_err(ms_error)?,
213                    TablePartNames::Plain(_)
214                ) {
215                    return Err(invalid(format!("{} must be a physical table.", id.name())));
216                }
217                Ok(path)
218            })
219            .transpose()
220    };
221    let required =
222        |id: SubtableId| path(id)?.ok_or_else(|| invalid(format!("Missing {}.", id.name())));
223    path(SubtableId::History)?;
224    Ok(LocalTables {
225        windows: required(SubtableId::SpectralWindow)?,
226        descriptions: required(SubtableId::DataDescription)?,
227        source: path(SubtableId::Source)?,
228    })
229}

238impl WindowOrigin {
239    pub(super) fn frame_at_reference(
240        &self,
241        options: &MsRegridOptions,
242        geometry: &Arc<FrameGeometry>,
243    ) -> Result<Option<Frame>, MsTransformError> {
244        let Some(field) = self.field else {
245            return Ok(None);
246        };
247        let mut options = options.clone();
248        options.phasecenter = field.to_string();
249        Frame::new(&options, self.reference_time, field as usize, geometry).map(Some)
250    }

253pub(super) struct Layout {
254    pub(super) windows: Vec<WindowOrigin>,
255    descriptions: Vec<RecordValue>,
256    description_windows: Vec<i32>,
257    sources: Vec<(i32, i32, i32)>,
258    row_descriptions: Vec<i32>,
259    original_descriptions: Vec<i32>,
260}
261
262impl Layout {
263    pub(super) fn new(
264        ms: &MeasurementSet,
265        options: &MsRegridOptions,
266        rows: Option<&[usize]>,
267    ) -> Result<Option<Self>, MsTransformError> {
268        let source_frame = options.outframe.eq_ignore_ascii_case("SOURCE");
269        let target = if source_frame {
270            FrequencyRef::GEO
271        } else if let Ok(reference) = options.outframe.parse::<FrequencyRef>() {
272            reference
273        } else if options.outframe.is_empty() {
274            return Ok(None);
275        } else {
276            return Err(super::invalid_choice("outframe", &options.outframe));
277        };
278        let desc = ms
279            .subtable(SubtableId::DataDescription)
280            .ok_or_else(|| invalid("Missing DATA_DESCRIPTION.".into()))?;
281        let windows = ms
282            .subtable(SubtableId::SpectralWindow)
283            .ok_or_else(|| invalid("Missing SPECTRAL_WINDOW.".into()))?;
284        let spectral = MsSpectralWindow::new(windows);
285        let descriptions_to_windows = (0..desc.row_count())
286            .map(|row| MsDataDescription::new(desc).spectral_window_id(row))
287            .collect::<Result<Vec<_>, _>>()
288            .map_err(ms_error)?;
289        let references = (0..spectral.row_count())
290            .map(|row| {
291                let code = spectral.meas_freq_ref(row).map_err(ms_error)?;
292                FrequencyRef::from_casacore_code(code)
293                    .ok_or_else(|| invalid(format!("Invalid MEAS_FREQ_REF {code}.")))
294            })
295            .collect::<Result<Vec<_>, _>>()?;
296        if !source_frame && references.iter().all(|&reference| target == reference) {
297            return Ok(None);
298        }
299        let (descriptions, fields, times) = layout_columns(ms.main_table(), rows)?;
300        let mut layout = Self {
301            windows: Vec::new(),
302            descriptions: Vec::new(),
303            description_windows: Vec::new(),
304            sources: Vec::new(),
305            row_descriptions: vec![-1; ms.row_count()],
306            original_descriptions: Vec::new(),
307        };
308        let mut window_ids = BTreeMap::new();
309        let mut description_ids = BTreeMap::new();
310        let mut sorted = (0..times.len()).collect::<Vec<_>>();
311        sorted.sort_by(|&left, &right| times[left].total_cmp(&times[right]));
312        let mut changed = false;
313        let mut source_rows = BTreeSet::new();
314        let field_table = ms.field().map_err(ms_error)?;
315        let has_source = ms.subtable(SubtableId::Source).is_some();
316        let mut previous_pair = None;
317        for slot in sorted {
318            let row = rows.map_or(slot, |rows| rows[slot]);
319            let ddid = descriptions[slot] as usize;
320            let field = fields[slot];
321            if let Some((previous_ddid, previous_field, output)) = previous_pair
322                && ddid == previous_ddid
323                && field == previous_field
324            {
325                layout.row_descriptions[row] = output;
326                continue;
327            }
328            let window = *descriptions_to_windows
329                .get(ddid)
330                .ok_or_else(|| invalid(format!("Invalid MAIN DATA_DESC_ID {ddid}.")))?
331                as usize;
332            let reference = *references
333                .get(window)
334                .ok_or_else(|| invalid(format!("Invalid DATA_DESCRIPTION {SPW_ID} {window}.")))?;
335            let convert = reference != target || source_frame;
336            changed |= convert;
337            let key = (window, convert.then_some(field));
338            let output_spw = match window_ids.entry(key) {
339                Entry::Occupied(entry) => *entry.get(),
340                Entry::Vacant(entry) => {
341                    let output = i32::try_from(layout.windows.len()).map_err(ms_error)?;
342                    layout.windows.push(WindowOrigin {
343                        spw: window as i32,
344                        reference,
345                        field: key.1,
346                        reference_time: times[slot],
347                    });
348                    *entry.insert(output)
349                }
350            };
351            let output_ddid = match description_ids.entry((ddid, output_spw)) {
352                Entry::Occupied(entry) => *entry.get(),
353                Entry::Vacant(entry) => {
354                    let mut record = desc.row_accessor().row(ddid).map_err(ms_error)?.clone();
355                    set_i32(&mut record, SPW_ID, output_spw);
356                    let output = i32::try_from(layout.descriptions.len()).map_err(ms_error)?;
357                    layout.descriptions.push(record);
358                    layout.description_windows.push(output_spw);
359                    *entry.insert(output)
360                }
361            };
362            if has_source && source_rows.insert((field, output_spw)) {
363                let source = field_table.source_id(field as usize).map_err(ms_error)?;
364                if source >= 0 {
365                    layout.sources.push((source, window as i32, output_spw));
366                }
367            }
368            layout.row_descriptions[row] = output_ddid;
369            previous_pair = Some((ddid, field, output_ddid));
370        }
371        if rows.is_none() {
372            layout.original_descriptions = descriptions;
373        }
374        Ok(changed.then_some(layout))
375    }

377    pub(super) fn description(&self, row: usize) -> Option<i32> {
378        self.row_descriptions
379            .get(row)
380            .copied()
381            .filter(|&id| id >= 0)
382    }
383
384    pub(super) fn window(&self, row: usize) -> Option<i32> {
385        self.description(row)
386            .map(|id| self.description_windows[id as usize])
387    }
388
389    fn window_records(
390        &self,
391        windows: &Table,
392        mut populate: impl FnMut(usize, &WindowOrigin, &mut RecordValue) -> Result<(), MsTransformError>,
393    ) -> Result<Vec<RecordValue>, MsTransformError> {
394        let descriptor = windows.row_accessor();
395        self.windows
396            .iter()
397            .enumerate()
398            .map(|(id, origin)| {
399                let mut record = descriptor
400                    .row(origin.spw as usize)
401                    .map_err(ms_error)?
402                    .clone();
403                populate(id, origin, &mut record)?;
404                Ok(record)
405            })
406            .collect()
407    }

409    pub(super) fn write_resampled(
410        &self,
411        stage: &Path,
412        mut populate: impl FnMut(usize, &WindowOrigin, &mut RecordValue) -> Result<(), MsTransformError>,
413    ) -> Result<(), MsTransformError> {
414        let main = Table::open(TableOptions::new(stage)).map_err(ms_error)?;
415        let paths = local_tables(&main, stage)?;
416        let windows = Table::open(TableOptions::new(&paths.windows)).map_err(ms_error)?;
417        let descriptions = Table::open(TableOptions::new(&paths.descriptions)).map_err(ms_error)?;
418        let records = self.window_records(&windows, &mut populate)?;
419        self.write_subtables(&paths, &windows, &descriptions, records)
420    }
421
422    fn write_subtables(
423        &self,
424        paths: &LocalTables,
425        original_windows: &Table,
426        original_descriptions: &Table,
427        windows: Vec<RecordValue>,
428    ) -> Result<(), MsTransformError> {
429        replace_staged_rows(&paths.windows, original_windows, windows)?;
430        replace_staged_rows(
431            &paths.descriptions,
432            original_descriptions,
433            self.descriptions.clone(),
434        )?;
435        if let Some(source) = &paths.source {
436            rewrite_sources(source, &self.sources)?;
437        }
438        Ok(())
439    }
440}

444fn layout_columns(main: &Table, rows: Option<&[usize]>) -> Result<LayoutColumns, MsTransformError> {
445    let columns = [DDID, "FIELD_ID", "TIME"];
446    let mut scalars = match rows {
447        Some(rows) => main.required_scalar_columns_owned_for_rows(&columns, rows),
448        None => main.required_scalar_columns_owned(&columns),
449    }
450    .map_err(ms_error)?;
451    match (
452        scalars.remove(DDID),
453        scalars.remove("FIELD_ID"),
454        scalars.remove("TIME"),
455    ) {
456        (
457            Some(RequiredScalarColumnValues::Int32(descriptions)),
458            Some(RequiredScalarColumnValues::Int32(fields)),
459            Some(RequiredScalarColumnValues::Float64(times)),
460        ) => Ok((descriptions, fields, times)),
461        _ => Err(invalid(
462            "Invalid time, field or description column for regridding.".into(),
463        )),
464    }
465}

467fn update_window(
468    record: &mut RecordValue,
469    spectral: &MsSpectralWindow<'_>,
470    window: usize,
471    reference: FrequencyRef,
472    frame: &Frame,
473) -> Result<(), MsTransformError> {
474    let convert = frame.prepare(reference, &frame.target)?;
475    let shift = radial_shift(-frame.source_velocity.unwrap_or(0.));
476    let apply = |frequency| shift.shift_frequency_hz(convert.convert_hz(frequency));
477    let input = spectral.chan_freq(window).map_err(ms_error)?;
478    let interval = |widths: Vec<f64>| {
479        if widths.len() != input.len() {
480            return Err(invalid(
481                "Spectral metadata axes have different lengths.".into(),
482            ));
483        }
484        Ok(input
485            .iter()
486            .zip(widths)
487            .map(|(&centre, width)| {
488                shift.shift_frequency_hz(
489                    convert.convert_hz(centre + width / 2.)
490                        - convert.convert_hz(centre - width / 2.),
491                )
492            })
493            .collect::<Vec<_>>())
494    };
495    let centres = input.iter().copied().map(apply).collect::<Vec<_>>();
496    let widths = interval(
497        spectral
498            .chan_width(window)
499            .map_err(ms_error)?
500            .into_iter()
501            .map(f64::abs)
502            .collect(),
503    )?;
504    let resolution = interval(spectral.resolution(window).map_err(ms_error)?)?;
505    let Some((&first, &last, &first_width, &last_width)) = centres
506        .first()
507        .zip(centres.last())
508        .zip(widths.first().zip(widths.last()))
509        .map(|((first, last), (first_width, last_width))| (first, last, first_width, last_width))
510    else {
511        return Err(invalid("The spectral window has no channels.".into()));
512    };
513    let reference_hz = apply(spectral.ref_frequency(window).map_err(ms_error)?);
514    let bandwidth = (last - first).abs() + (first_width.abs() + last_width.abs()) / 2.;
515    for (column, values) in [
516        ("CHAN_FREQ", centres),
517        ("CHAN_WIDTH", widths),
518        ("RESOLUTION", resolution),
519    ] {
520        record.upsert(
521            column,
522            Value::Array(ArrayValue::Float64(Array1::from_vec(values).into_dyn())),
523        );
524    }
525    record.upsert(
526        "REF_FREQUENCY",
527        Value::Scalar(ScalarValue::Float64(reference_hz)),
528    );
529    record.upsert(
530        "TOTAL_BANDWIDTH",
531        Value::Scalar(ScalarValue::Float64(bandwidth)),
532    );
533    set_i32(
534        record,
535        "MEAS_FREQ_REF",
536        frame.metadata_reference.casacore_code(),
537    );
538    Ok(())
539}

545fn replace_staged_rows(
546    path: &Path,
547    original: &Table,
548    records: Vec<RecordValue>,
549) -> Result<(), MsTransformError> {
550    let schema = original
551        .schema()
552        .ok_or_else(|| invalid("Missing table schema".into()))?;
553    let mut table = Table::with_schema(schema.clone());
554    table.copy_info(original);
555    *table.keywords_mut() = original.keywords().clone();
556    for column in schema.columns() {
557        if let Some(keywords) = original.column_keywords(column.name()) {
558            table.set_column_keywords(column.name(), keywords.clone());
559        }
560    }
561    for record in records {
562        table.add_row(record).map_err(ms_error)?;
563    }
564    // The private stage is discarded if the table write fails.
565    write_private_table_preserving_modes(path, || {
566        table
567            .prepare_write()
568            .save(TableOptions::new(path))
569            .map_err(io::Error::other)
570    })
571    .map_err(ms_error)
572}

574fn rewrite_sources(path: &Path, links: &[(i32, i32, i32)]) -> Result<(), MsTransformError> {
575    let table = Table::open(TableOptions::new(path)).map_err(ms_error)?;
576    if table.row_count() == 0 || links.is_empty() {
577        return Ok(());
578    }
579    let pairs = (0..table.row_count())
580        .map(|row| {
581            Ok((
582                get_i32(&table, row, SOURCE_ID)?,
583                get_i32(&table, row, SPW_ID)?,
584            ))
585        })
586        .collect::<casa_ms::MsResult<Vec<_>>>()
587        .map_err(ms_error)?;
588    let replaced = links
589        .iter()
590        .map(|&(source, _, window)| (source, window))
591        .collect::<BTreeSet<_>>();
592    let mut retained = BTreeSet::new();
593    let mut records = Vec::new();
594    for (row, &pair) in pairs.iter().enumerate() {
595        if !replaced.contains(&pair) && retained.insert(pair) {
596            records.push(table.row_accessor().row(row).map_err(ms_error)?.clone());
597        }
598    }
599    for &(source, previous, output) in links {
600        let row = pairs
601            .iter()
602            .position(|&(id, window)| id == source && (window == previous || window == -1))
603            .ok_or_else(|| {
604                invalid(format!(
605                    "No SOURCE row for source {source} and SPW {previous}."
606                ))
607            })?;
608        let mut record = table.row_accessor().row(row).map_err(ms_error)?.clone();
609        set_i32(&mut record, SPW_ID, output);
610        records.push(record);
611    }
612    replace_staged_rows(path, &table, records)
613}
