2//! Where a task's executable is found.
3//!
4//! Python starts each task's executable directly (ADR-0017), so one search
5//! decides which file runs. It looks, in order, at:
6//!
7//! 1. the task's override, `CASARS_<APP>_BIN`
8//!    ([`ApplicationLaunchDescriptor::override_env`]);
9//! 2. a suite root: `CASARS_SUITE_ROOT`, or else the installed suite the
10//!    caller belongs to (see [`installed_suite_root`]), whose `bin/` holds
11//!    the executable;
12//! 3. a development checkout: the built executable under the workspace this
13//!    library was built in, in `$CARGO_TARGET_DIR` or `target/`.
14//!
15//! The first step that applies decides. A step that applies but finds no
16//! file is an error naming what is missing; it never falls through to a
17//! later step, so an installed suite never runs a checkout's build. Nothing
18//! is built or run through `cargo`: a checkout's missing executable is
19//! reported with the `cargo build` that makes it. The caller's own path
20//! (`current_exe`) is not consulted; from Python it is the interpreter.
21
22use std::ffi::OsString;
23use std::path::{Path, PathBuf};
24use std::time::SystemTime;
25
26use crate::ApplicationLaunchDescriptor;
27
28/// The environment variable that names an installed suite's root.
29pub const SUITE_ROOT_ENV: &str = "CASARS_SUITE_ROOT";
30
31/// The file that marks an installed suite's root (`scripts/install-suite.sh`).
32const SUITE_MANIFEST: &str = "bundle-manifest.json";
33
34/// The workspace this library was built in, fixed at build time.
35const BUILD_WORKSPACE: &str = concat!(env!("CARGO_MANIFEST_DIR"), "/../..");
36
37/// Find the executable `launch` names.
38///
39/// `environment` reads one environment variable; a caller passes its own
40/// process environment, or a test its own. `installed_suite` is the root of
41/// the installed suite the caller belongs to, when it belongs to one: from
42/// Python, [`installed_suite_root`] of `sys.prefix`. The search order is the
43/// module's. The path found is absolute: a relative override or suite root
44/// names a file in the caller's working directory, wherever the task then
45/// runs. The error says which step applied and what it lacked.
46pub fn resolve_task_executable(
47    launch: &ApplicationLaunchDescriptor,
48    environment: &dyn Fn(&str) -> Option<OsString>,
49    installed_suite: Option<&Path>,
50) -> Result<PathBuf, String> {
51    let caller =
52        std::env::current_dir().map_err(|error| format!("read the working directory: {error}"))?;
53    resolve_in(
54        launch,
55        environment,
56        installed_suite,
57        development_workspace(Path::new(BUILD_WORKSPACE)).as_deref(),
58        &caller,
59    )
60}
61
62/// The root of the installed suite whose Python environment is `prefix`, if
63/// it is one: the installer puts the environment at `<root>/python`, beside
64/// `<root>/bin` and the bundle manifest.
65pub fn installed_suite_root(prefix: &Path) -> Option<PathBuf> {
66    let root = prefix.parent()?;
67    (prefix.file_name()? == "python" && root.join(SUITE_MANIFEST).is_file())
68        .then(|| root.to_path_buf())
69}

71/// [`resolve_task_executable`] with the development checkout and the
72/// caller's working directory given.
73fn resolve_in(
74    launch: &ApplicationLaunchDescriptor,
75    environment: &dyn Fn(&str) -> Option<OsString>,
76    installed_suite: Option<&Path>,
77    development: Option<&Path>,
78    caller: &Path,
79) -> Result<PathBuf, String> {
80    let name = format!("{}{}", launch.executable, std::env::consts::EXE_SUFFIX);
81    if let Some(path) = environment(&launch.override_env).filter(|path| !path.is_empty()) {
82        let path = caller.join(path);
83        return if path.is_file() {
84            Ok(path)
85        } else {
86            Err(format!(
87                "{} names {}, which is not a file",
88                launch.override_env,
89                path.display()
90            ))
91        };
92    }
93    let suite = environment(SUITE_ROOT_ENV)
94        .filter(|root| !root.is_empty())
95        .map(|root| (caller.join(root), SUITE_ROOT_ENV))
96        .or_else(|| installed_suite.map(|root| (caller.join(root), "the installed suite")));
97    if let Some((root, source)) = suite {
98        let path = root.join("bin").join(&name);
99        return if path.is_file() {
100            Ok(path)
101        } else {
102            Err(format!(
103                "the casa-rs suite at {} ({source}) has no {name} in bin/; reinstall the \
104                 suite, or set {} to the executable",
105                root.display(),
106                launch.override_env
107            ))
108        };
109    }
110    if let Some(workspace) = development {
111        let target = match environment("CARGO_TARGET_DIR").filter(|dir| !dir.is_empty()) {
112            Some(dir) => workspace.join(dir),
113            None => workspace.join("target"),
114        };
115        return newest_build(&target, &name).ok_or_else(|| {
116            format!(
117                "{name} is not built in this development checkout ({}); build it with \
118                 `cargo build -p {} --bin {}`, or set {} to the executable",
119                target.display(),
120                launch.cargo_package,
121                launch.executable,
122                launch.override_env
123            )
124        });
125    }
126    Err(format!(
127        "no {name} for this task: set {SUITE_ROOT_ENV} to an installed casa-rs suite, or {} \
128         to the executable (docs/install.md)",
129        launch.override_env
130    ))
131}

133/// `workspace`, made absolute, when it is the casa-rs checkout this library
134/// was built in.
135fn development_workspace(workspace: &Path) -> Option<PathBuf> {
136    workspace
137        .join("crates/casa-provider-contracts/Cargo.toml")
138        .is_file()
139        .then(|| workspace.canonicalize().ok())
140        .flatten()
141}

143/// The most recently built of `target`'s debug and release `name`.
144fn newest_build(target: &Path, name: &str) -> Option<PathBuf> {
145    let modified = |path: &Path| -> Option<SystemTime> {
146        let metadata = path.metadata().ok()?;
147        metadata
148            .is_file()
149            .then(|| metadata.modified().ok())
150            .flatten()
151    };
152    ["debug", "release"]
153        .into_iter()
154        .map(|profile| target.join(profile).join(name))
155        .filter_map(|path| Some((modified(&path)?, path)))
156        .max_by_key(|(modified, _)| *modified)
157        .map(|(_, path)| path)
158}
