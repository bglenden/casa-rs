19enum CreationInput {
20    Shape(Option<IntList>),
21    Pixels(PyResult<Pixels>),
22}

24struct Create {
25    method: &'static str,
26    outfile: String,
27    shape: Vec<usize>,
28    record: RecordValue,
29    linear: bool,
30    overwrite: bool,
31    log: bool,
32    kind: String,
33}
34
35impl Create {
36    fn build(
37        &self,
38        pixels: Option<Pixels>,
39        path: Option<&Path>,
40        history: bool,
41    ) -> PyResult<ImageHandle> {
42        let record = if self.record.fields().is_empty() {
43            super::default_record::for_shape(&self.shape, self.linear)?
44        } else if let [field] = self.record.fields()
45            && let Value::Record(record) = &field.value
46        {
47            record.clone()
48        } else {
49            self.record.clone()
50        };
51        let coordinates = CoordinateSystem::from_record(&record)
52            .map_err(|_| raised("Exception: Invalid coordinate system record"))?;
53        if coordinates.n_pixel_axes() != self.shape.len() {
54            return Err(raised(
55                "Exception: Supplied CoordinateSystem and image shape are inconsistent.",
56            ));
57        }
58        let kind = self.kind.to_ascii_lowercase();
59        let shape = self.shape.clone();
60        let mut image = match (
61            kind.as_str(),
62            pixels.as_ref().is_some_and(Pixels::is_complex),
63        ) {
64            ("f", false) => ImageHandle::Float(
65                Storage::create(
66                    shape,
67                    coordinates,
68                    path,
69                    pixels.as_ref().map(Pixels::f32).as_ref(),
70                )
71                .map_err(raised)?,
72            ),
73            ("d", false) => ImageHandle::Double(
74                Storage::create(
75                    shape,
76                    coordinates,
77                    path,
78                    pixels.as_ref().map(Pixels::f64).as_ref(),
79                )
80                .map_err(raised)?,
81            ),
82            ("c", _) | ("f", true) => ImageHandle::Complex(
83                Storage::create(
84                    shape,
85                    coordinates,
86                    path,
87                    pixels.as_ref().map(Pixels::c32).as_ref(),
88                )
89                .map_err(raised)?,
90            ),
91            ("cd", _) | ("d", true) => ImageHandle::DComplex(
92                Storage::create(
93                    shape,
94                    coordinates,
95                    path,
96                    pixels.as_ref().map(Pixels::c64).as_ref(),
97                )
98                .map_err(raised)?,
99            ),
100            _ => {
101                return Err(raised(
102                    "Exception: Input parm type must be either 'f', 'c', 'd', or 'cd'",
103                ));
104            }
105        };
106        image.set_coordinate_record(record)?;
107        self.history(&mut image, pixels.as_ref(), history)?;
108        image.save().map_err(raised)?;
109        Ok(image)
110    }
111
112    fn history(
113        &self,
114        image: &mut ImageHandle,
115        pixels: Option<&Pixels>,
116        enabled: bool,
117    ) -> PyResult<()> {
118        let (storage, filename) = if self.outfile.is_empty() {
119            ("Temp", String::new())
120        } else {
121            ("Paged", format!("'{}'", self.outfile))
122        };
123        let pixel_type = match image {
124            ImageHandle::Float(_) => "float",
125            ImageHandle::Double(_) => "double",
126            ImageHandle::Complex(_) => "Complex",
127            ImageHandle::DComplex(_) => "DComplex",
128        };
129        let creation = format!(
130            "Created {storage} image {filename} of shape {:?} with {pixel_type} valued pixels.",
131            self.shape
132        );
133        image.append_history_at("::ImageFactory::createImage", &[creation])?;
134        if enabled {
135            let args = match pixels {
136                Some(pixels) if self.method == "newimagefromarray" => {
137                    format!("outfile={:?}, pixels={}", self.outfile, pixels.history())
138                }
139                Some(pixels) => format!("pixels={}", pixels.history()),
140                None => format!(
141                    "outfile={:?}, shape={}",
142                    self.outfile,
143                    format!("{:?}", self.shape).replace(' ', ""),
144                ),
145            };
146            let kind = if self.method == "newimagefromarray" {
147                String::new()
148            } else {
149                format!(", type={:?}", self.kind)
150            };
151            let call = format!(
152                "ia.{}({args}, csys={{...}}, linear={}, overwrite={}, log={}{kind})",
153                self.method, self.linear, self.overwrite, self.log
154            );
155            image.record_history(self.method, call)?;
156        }
157        Ok(())
158    }
159
160    fn image_shape(shape: Option<IntList>) -> PyResult<Vec<usize>> {
161        let shape = shape.unwrap_or_else(|| IntList::of(&[0])).0;
162        if shape.is_empty() {
163            return Err(raised(
164                "Exception: The shape must have more than zero elements.",
165            ));
166        }
167        if shape.iter().any(|&length| length <= 0) {
168            return Err(raised("Exception: All elements of shape must be positive."));
169        }
170        Ok(shape.into_iter().map(|length| length as usize).collect())
171    }
172
173    fn run(mut self, input: CreationInput, history: bool) -> PyResult<AttachedImage> {
174        let pixels = match input {
175            CreationInput::Shape(shape) => {
176                self.shape = Self::image_shape(shape)?;
177                None
178            }
179            CreationInput::Pixels(pixels) => {
180                if !matches!(self.kind.to_ascii_lowercase().as_str(), "f" | "d") {
181                    return Err(raised(format!(
182                        "Exception: Unsupported value for type: {:?}. Supported values are \"d\" and \"f\"",
183                        self.kind
184                    )));
185                }
186                let pixels = pixels?;
187                self.shape = pixels.shape().to_vec();
188                Some(pixels)
189            }
190        };
191        publish_image(&self.outfile, self.overwrite, |path| {
192            self.build(pixels, path, history)
193        })
194    }
195}

197pub(super) fn publish_image(
198    outfile: &str,
199    overwrite: bool,
200    build: impl FnOnce(Option<&Path>) -> PyResult<ImageHandle>,
201) -> PyResult<AttachedImage> {
202    if outfile.is_empty() {
203        return build(None).map(AttachedImage::temporary);
204    }
205    let path = std::path::absolute(outfile).map_err(raised)?;
206    let parent = path
207        .parent()
208        .ok_or_else(|| raised("Invalid output image path"))?;
209    let stage = tempfile::tempdir_in(parent).map_err(raised)?;
210    let product = stage.path().join("image");
211    drop(build(Some(&product))?);
212    let mut registry = attachment::registry();
213    let existed = match path.symlink_metadata() {
214        Ok(_) if !overwrite => {
215            return Err(raised(format!(
216                "Exception: File {} exists, and the user does not want to remove it..",
217                outfile
218            )));
219        }
220        Ok(_) => {
221            if registry
222                .get(&attachment::key(&path)?)
223                .is_some_and(|image| image.strong_count() > 0)
224            {
225                return Err(raised(format!(
226                    "Invalid Table operation: SetupNewTable {} is already opened (is in the table cache)",
227                    path.display()
228                )));
229            }
230            let empty = path
231                .read_dir()
232                .is_ok_and(|mut entries| entries.next().is_none());
233            if !empty
234                && casa_tables::Table::open_metadata_only(casa_tables::TableOptions::new(&path))
235                    .is_err()
236            {
237                return Err(raised(format!(
238                    "Table {} already exists (and is not a true table directory)",
239                    path.display()
240                )));
241            }
242            true
243        }
244        Err(error) if error.kind() == std::io::ErrorKind::NotFound => false,
245        Err(error) => return Err(raised(error)),
246    };
247    if existed {
248        exchange_directories(&product, &path).map_err(raised)?;
249    } else {
250        rename_without_replacing(&product, &path).map_err(raised)?;
251    }
252    match AttachedImage::open(&mut registry, &path) {
253        Ok(image) => Ok(image),
254        Err(error) => {
255            let restore = if existed {
256                exchange_directories(&product, &path)
257            } else {
258                rename_without_replacing(&path, &product)
259            };
260            if let Err(restore) = restore {
261                let backup_dir = stage.keep();
262                return Err(raised(format!(
263                    "{error}. Image publication could not be rolled back; inspect {}: {restore}",
264                    backup_dir.display()
265                )));
266            }
267            Err(error)
268        }
269    }
270}

272impl PyImageTool {
273    fn replace_from(
274        &self,
275        py: Python<'_>,
276        request: Create,
277        input: CreationInput,
278    ) -> PyResult<bool> {
279        py.detach(|| {
280            let mut state = self.state();
281            state.image = None;
282            state.image = Some(request.run(input, state.history)?);
283            Ok(true)
284        })
285    }
286
287    fn new_from(
288        &self,
289        py: Python<'_>,
290        request: Create,
291        input: CreationInput,
292    ) -> PyResult<Py<Self>> {
293        let image = py.detach(|| {
294            let history = self.state().history;
295            request.run(input, history)
296        })?;
297        Py::new(py, Self::attached(image))
298    }
299}
