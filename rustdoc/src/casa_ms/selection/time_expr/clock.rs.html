2//! The calendar arithmetic `MSTimeParse` does: casacore's `Time` class
3//! (`casa/OS/Time.cc`), `MVEpoch`'s whole-and-fractional day sum
4//! (`casa/Quanta/MVEpoch.cc`) and `MVTime`'s days and seconds
5//! (`casa/Quanta/MVTime.cc`).
6//!
7//! Every operation is the C++ one in IEEE double precision, unfused, in the
8//! same order, so a default or a bound comes out bit for bit as CASA's x86-64
9//! build computes it. Where the C++ converts a double that is out of range for
10//! an unsigned integer (undefined behaviour, which differs between CASA's
11//! platforms) the port saturates, as Rust's casts and CASA's arm64 build do.
12
13use casa_types::quanta::format::CalendarDate;
14
15use super::grammar::{TimeFields, UNSET};

17/// Seconds in a day, casacore's `C::day`.
18const DAY_S: f64 = 86_400.0;
19
20/// casacore `Time`: an MJD day number held in a C `uInt` and a day fraction.
21#[derive(Clone, Copy, Debug, PartialEq)]
22pub(super) struct CasacoreTime {
23    day: u32,
24    fraction: f64,
25}
26
27impl CasacoreTime {
28    /// `Time::Time(double jdn)` (Time.cc:61-72), or `None` where its `(int)`
29    /// cast of `jdn` would be undefined.
30    pub(super) fn from_julian_day(jdn: f64) -> Option<Self> {
31        if !(jdn > -2_147_483_649.0 && jdn < 2_147_483_648.0) {
32            return None;
33        }
34        #[expect(
35            clippy::cast_possible_truncation,
36            reason = "C's (int) cast truncates; the range is checked above"
37        )]
38        let whole = jdn as i32;
39        let fraction = jdn - f64::from(whole);
40        // `mJulianDay` is a uInt; an int below the offset wraps.
41        Some(if fraction >= 0.5 {
42            Self {
43                day: whole.wrapping_sub(2_400_000).cast_unsigned(),
44                fraction: fraction - 0.5,
45            }
46        } else {
47            Self {
48                day: whole.wrapping_sub(2_400_001).cast_unsigned(),
49                fraction: fraction + 0.5,
50            }
51        })
52    }
53
54    /// `Time::Time(uInt year, uInt month, uInt day, uInt hour, uInt min,
55    /// double sec)`, which runs `Time::setDate` (Time.cc:344-381). The `Int`
56    /// fields of the caller pass through `uInt` parameters unchanged in their
57    /// bits. `None` where the C `int` day-number arithmetic would overflow
58    /// (undefined behaviour), which only a year beyond about 1.4 million does.
59    ///
60    /// The hour, minute and second are not range-checked, so 25:00 or 60 s
61    /// roll into the next day or minute, as they do in CASA.
62    pub(super) fn from_date(
63        year: i32,
64        month: i32,
65        day: i32,
66        hour: i32,
67        minute: i32,
68        sec: f64,
69    ) -> Option<Self> {
70        let (y, m, d) = (year, month, day);
71        // md = (1461*(y+4800+(m-14)/12))/4+(367*(m-2-12*((m-14)/12)))/12
72        //      -(3*((y+4900+(m-14)/12)/100))/4+d-32075, in C int arithmetic.
73        let a = m.checked_sub(14)? / 12;
74        let years = 1461_i32.checked_mul(y.checked_add(4800)?.checked_add(a)?)? / 4;
75        let months =
76            367_i32.checked_mul(m.checked_sub(2)?.checked_sub(12_i32.checked_mul(a)?)?)? / 12;
77        let centuries = 3_i32.checked_mul(y.checked_add(4900)?.checked_add(a)? / 100)? / 4;
78        let md = years
79            .checked_add(months)?
80            .checked_sub(centuries)?
81            .checked_add(d)?
82            .checked_sub(32_075)?;
83        let fraction = f64::from(hour.cast_unsigned()) / 24.0
84            + f64::from(minute.cast_unsigned()) / 1440.0
85            + sec / DAY_S;
86        Some(Self {
87            // `uInt md` and `mJulianDay = md - 2400001` in uInt arithmetic.
88            day: md.cast_unsigned().wrapping_sub(2_400_001),
89            fraction,
90        })
91    }
92
93    /// `Time::modifiedJulianDay` (Time.cc:91-95).
94    pub(super) fn modified_julian_day(self) -> f64 {
95        f64::from(self.day) + self.fraction
96    }
97
98    /// `Time::hours` (Time.cc:419-423).
99    fn hours(self) -> u32 {
100        uint(self.fraction * 24.0 + 2e-5 / 3600.0)
101    }
102
103    /// `Time::minutes` (Time.cc:411-417).
104    fn minutes(self) -> u32 {
105        let hour = self.fraction * 24.0;
106        let minutes = (hour - f64::from(self.hours())) * 60.0;
107        uint(minutes + 2e-5 / 60.0)
108    }
109
110    /// `Time::seconds` (Time.cc:398-402) of `Time::dseconds` (Time.cc:404-409).
111    fn seconds(self) -> u32 {
112        let hour = self.fraction * 24.0;
113        let seconds = (hour - f64::from(self.hours())) * 3600.0 - f64::from(self.minutes()) * 60.0;
114        uint(seconds + 2e-5)
115    }
116
117    /// `Time::operator-` (Time.cc:104-119): `self - begin` in seconds.
118    fn seconds_since(self, begin: Self) -> f64 {
119        let mut fraction = self.fraction - begin.fraction;
120        let days = if fraction < 0.0 {
121            fraction += 1.0;
122            f64::from(self.day) - f64::from(begin.day) - 1.0
123        } else {
124            f64::from(self.day) - f64::from(begin.day)
125        };
126        (days + fraction) * DAY_S
127    }
128}

130/// C++'s `uInt(x)` for the values `Time`'s accessors convert. A negative or
131/// too large `x` is undefined behaviour there; Rust's cast saturates.
132#[expect(
133    clippy::cast_possible_truncation,
134    clippy::cast_sign_loss,
135    reason = "C's uInt cast truncates toward zero"
136)]
137fn uint(value: f64) -> u32 {
138    value as u32
139}

141/// `MVEpoch(inday, infrac).get()` (MVEpoch.cc:54-59, 156-172, 249-253): the
142/// whole days and the fractions of both arguments summed apart, then added.
143pub(super) fn epoch_days(day: f64, fraction: f64) -> f64 {
144    let mut whole = 0.0;
145    let mut part = 0.0;
146    for value in [day, fraction] {
147        let floor = value.floor();
148        whole += floor;
149        part += value - floor;
150    }
151    // `adjust()` loops, but for the finite days here `part` is in [0, 2), so
152    // its loops run at most once.
153    if part < 0.0 {
154        part += 1.0;
155        whole -= 1.0;
156    }
157    if part >= 1.0 {
158        part -= 1.0;
159        whole += 1.0;
160    }
161    whole + part
162}

164/// `MSTimeParse::toTAIInSec` (MSTimeParse.cc:339-344): `MVTime(epoch) *
165/// 86400`. Its conversion to TAI is commented out, so the seconds stay in the
166/// epoch's own scale, which is that of the `TIME` column they are compared to.
167pub(super) fn epoch_seconds(days: f64) -> f64 {
168    days * DAY_S
169}

171/// The fields and exposure `MSTimeParse::getDefaults` (MSTimeParse.cc:114-184)
172/// derives from one MAIN row's `TIME` and `EXPOSURE`.
173#[derive(Clone, Copy, Debug, PartialEq)]
174pub(super) struct RowDefaults {
175    pub(super) fields: TimeFields,
176    pub(super) exposure_s: f64,
177}
178
179/// `getDefaults` for a row with `time_s` (MJD seconds) and `exposure_s`, or
180/// `None` where the `TIME` is outside the days casacore's `Time` converts.
181pub(super) fn row_defaults(time_s: f64, exposure_s: f64) -> Option<RowDefaults> {
182    // `MVTime(timeQuant()(row))`: the base value in seconds over `C::day`.
183    let days = time_s / DAY_S;
184    // `Time t0(firstRowTime.getTime())`, `MVTime::getTime` (MVTime.cc:130-132).
185    let first = CasacoreTime::from_julian_day(days + 2_400_000.5)?;
186    // `MVTime::year`, `month` and `monthday` (MVTime.cc:160-176, 209-232).
187    let date = CalendarDate::of(days);
188    let year = i32::try_from(date.year).ok()?;
189    let month = i32::try_from(date.month).ok()?;
190    let day = i32::try_from(date.monthday).ok()?;
191    let hour = first.hours().cast_signed();
192    let minute = first.minutes().cast_signed();
193    let sec = first.seconds().cast_signed();
194    let whole = CasacoreTime::from_date(year, month, day, hour, minute, f64::from(sec))?;
195    // `defaultFractionalSec = (Int)((t0-t1)*1E3)`.
196    #[expect(
197        clippy::cast_possible_truncation,
198        reason = "C's (Int) cast truncates toward zero"
199    )]
200    let millisec = (first.seconds_since(whole) * 1e3) as i32;
201    Some(RowDefaults {
202        fields: TimeFields {
203            year,
204            month,
205            day,
206            hour,
207            minute,
208            sec,
209            millisec,
210        },
211        exposure_s,
212    })
213}

215/// `MSTimeParse::setDefaults(tf, True)` (MSTimeParse.cc:357-367).
216pub(super) fn fill_from_row(fields: TimeFields, row: &RowDefaults) -> TimeFields {

220/// `MSTimeParse::setDefaults(tf, False)` (MSTimeParse.cc:368-382): the MJD
221/// reference date (`Time(2400000.5)`, 1858/11/17) and zero time of day.
222pub(super) fn fill_from_mjd_reference(fields: TimeFields) -> TimeFields {

227            month: 11,

230            minute: 0,
231            sec: 0,
232            millisec: 0,

234    )

237/// `MSTimeParse::copyDefaults(target, source)` (MSTimeParse.cc:415-424).
238pub(super) fn fill(target: TimeFields, source: TimeFields) -> TimeFields {
239    let pick = |value: i32, default: i32| if value == UNSET { default } else { value };
240    TimeFields {
241        year: pick(target.year, source.year),
242        month: pick(target.month, source.month),
243        day: pick(target.day, source.day),
244        hour: pick(target.hour, source.hour),
245        minute: pick(target.minute, source.minute),
246        sec: pick(target.sec, source.sec),
247        millisec: pick(target.millisec, source.millisec),
248    }
249}

251/// The `Time` of complete fields: `Double s = Double(tf.sec) +
252/// Double(tf.fsec)/1000.0; Time t(...)` (MSTimeParse.cc:322-323).
253pub(super) fn time_of(fields: TimeFields) -> Option<CasacoreTime> {
254    let sec = f64::from(fields.sec) + f64::from(fields.millisec) / 1000.0;
255    CasacoreTime::from_date(
256        fields.year,
257        fields.month,
258        fields.day,
259        fields.hour,
260        fields.minute,
261        sec,
262    )
263}

265#[cfg(test)]
266mod tests {
267    use super::*;
268
269    const DAY_57813_S: f64 = 57_813.0 * DAY_S;
270
271    #[test]
272    fn set_date_counts_days_from_the_mjd_epoch() {
273        // 1858/11/17 is MJD 0; 2017/03/01 is MJD 57813.
274        let epoch = CasacoreTime::from_date(1858, 11, 17, 0, 0, 0.0).unwrap();
275        assert_eq!(epoch.modified_julian_day(), 0.0);
276        let day = CasacoreTime::from_date(2017, 3, 1, 9, 0, 0.0).unwrap();
277        assert_eq!(day.modified_julian_day(), 57_813.375);
278        // Fields past their range roll over: Feb 29 2017 is Mar 1, 24 h the next day.
279        let rolled = CasacoreTime::from_date(2017, 2, 29, 24, 0, 0.0).unwrap();
280        assert_eq!(rolled.modified_julian_day(), 57_814.0);
281        // A date before MJD 0 wraps the uInt day number, as in C++.
282        let early = CasacoreTime::from_date(1858, 11, 16, 0, 0, 0.0).unwrap();
283        assert_eq!(early.modified_julian_day(), f64::from(u32::MAX));
284        assert!(CasacoreTime::from_date(2_000_000, 1, 1, 0, 0, 0.0).is_none());
285    }
286
287    #[test]
288    fn epoch_days_sums_whole_days_and_fractions_apart() {
289        assert_eq!(epoch_days(57_813.375, 0.0), 57_813.375);
290        assert_eq!(epoch_days(57_813.75, 0.5), 57_814.25);
291        assert_eq!(epoch_days(57_813.375, 2.0), 57_815.375);
292    }

294    #[test]
295    fn defaults_split_the_first_row_time_into_casas_fields() {
296        // 2017/03/01/09:00:00.5, EXPOSURE 2 s.
297        let row = row_defaults(DAY_57813_S + 32_400.5, 2.0).unwrap();
298        assert_eq!(
299            row.fields,
300            TimeFields {
301                year: 2017,
302                month: 3,
303                day: 1,
304                hour: 9,
305                minute: 0,
306                sec: 0,
307                millisec: 500,
308            }
309        );
310        assert_eq!(row.exposure_s, 2.0);
311        // The milliseconds truncate: 0.1234 s keeps 123 ms.
312        let row = row_defaults(DAY_57813_S + 32_400.123_4, 2.0).unwrap();
313        assert_eq!((row.fields.sec, row.fields.millisec), (0, 123));
314        // 23:59:59.75 on 2017/02/28.
315        let row = row_defaults(DAY_57813_S - 0.25, 1.0).unwrap();
316        assert_eq!(
317            (row.fields.year, row.fields.month, row.fields.day),
318            (2017, 2, 28)
319        );
320        assert_eq!(
321            (row.fields.hour, row.fields.minute, row.fields.sec),
322            (23, 59, 59)
323        );
324        assert!(row_defaults(f64::NAN, 1.0).is_none());
325        assert!(row_defaults(1e20, 1.0).is_none());
326    }
