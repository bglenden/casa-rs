10/// Exact MeasurementSet visibility column read for selected-observation input.
11#[derive(Debug, Clone, Copy, PartialEq, Eq)]
12pub(crate) enum SelectedVisibilityColumn {
13    /// MAIN `DATA` complex visibility values.
14    Data,
15    /// MAIN `CORRECTED_DATA` complex visibility values.
16    CorrectedData,
17    /// MAIN `FLOAT_DATA` real visibility values.
18    FloatData,
19}
20
21impl SelectedVisibilityColumn {
22    const fn name(self) -> &'static str {
23        match self {
24            Self::Data => VisibilityDataColumn::Data.name(),
25            Self::CorrectedData => VisibilityDataColumn::CorrectedData.name(),
26            Self::FloatData => "FLOAT_DATA",
27        }
28    }
29
30    /// Bytes of one stored sample.
31    const fn sample_bytes(self) -> usize {
32        match self {
33            Self::FloatData => size_of::<f32>(),
34            Self::Data | Self::CorrectedData => size_of::<casa_types::Complex32>(),
35        }
36    }
37}

139pub(crate) struct SelectedObservationReadStaging {
140    /// Held per selected row.
141    pub(crate) bytes_per_row: usize,
142    /// Held once per fill.
143    pub(crate) fixed_bytes: usize,

146/// Project what the column reads of one block fill hold besides the buffer,
147/// for MAIN cells of `correlations` x `stored_channels` samples: the row
148/// walk's `UVW` read, then the visibility, `FLAG` and weight reads of
149/// [`MeasurementSet::fill_selected_observation_buffer`].
150///
151/// A column its data manager streams holds nothing per row. A column read
152/// cell by cell holds the whole stored cells of the selected rows, all
153/// channels, however narrow the selected channel range
154/// ([`casa_tables::SelectedReadFootprint`]). The fill reads one column at a
155/// time, so the staging is the largest any one read holds.
156pub(crate) fn selected_observation_read_staging(
157    main: &casa_tables::Table,
158    visibility: SelectedVisibilityColumn,
159    weight: SelectedWeightColumn,
160    correlations: usize,
161    stored_channels: usize,
162) -> Option<SelectedObservationReadStaging> {
163    let samples = correlations.checked_mul(stored_channels)?;
164    let visibility_bytes = visibility.sample_bytes();
165    let weight_read = match weight {
166        SelectedWeightColumn::Weight => (
167            main.selected_cell_read_footprint("WEIGHT"),
168            correlations.checked_mul(size_of::<f32>())?,
169        ),
170        SelectedWeightColumn::WeightSpectrum => (
171            main.selected_channel_read_footprint("WEIGHT_SPECTRUM"),
172            samples.checked_mul(size_of::<f32>())?,
173        ),
174    };
175    [
176        (
177            main.selected_channel_read_footprint(visibility.name()),
178            samples.checked_mul(visibility_bytes)?,
179        ),
180        (
181            main.selected_channel_read_footprint("FLAG"),
182            samples.checked_mul(size_of::<bool>())?,
183        ),
184        weight_read,
185        (
186            main.selected_cell_read_footprint("UVW"),
187            size_of::<[f64; 3]>(),
188        ),
189    ]
190    .into_iter()
191    .try_fold(
192        SelectedObservationReadStaging::default(),
193        |staging, (footprint, stored_cell_bytes)| {
194            Some(SelectedObservationReadStaging {
195                bytes_per_row: staging
196                    .bytes_per_row
197                    .max(footprint.staging_bytes_per_row(stored_cell_bytes)?),
198                fixed_bytes: staging
199                    .fixed_bytes
200                    .max(footprint.staging_fixed_bytes(stored_cell_bytes)?),
201            })
202        },
203    )
204}

206/// The storage a [`SelectedObservationBuffer`] is allocated with: `rows`
207/// selected rows, each with room for the packed samples and input weight
208/// values of the widest selected DATA_DESCRIPTION.
209///
210/// The buffer is refilled in place, one DATA_DESCRIPTION per block, so it
211/// retains the largest of each of its vectors over every block, whichever
212/// DATA_DESCRIPTION that block came from. A capacity is therefore the
213/// componentwise largest ([`Self::union`]) of its rows' needs, and
214/// [`Self::retained_bytes`] is what a buffer of it holds after any fill:
215/// [`MeasurementSet::fill_selected_observation_buffer`] copies the stored
216/// rows and reads the channelized columns straight into its vectors, so
217/// this is also the fill's peak besides the read staging
218/// ([`selected_observation_read_staging`]).
219#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
220pub(crate) struct SelectedObservationBufferCapacity {
221    rows: usize,
222    samples_per_row: usize,
223    weight_values_per_row: usize,
224    sample_bytes: usize,
225}
226
227impl SelectedObservationBufferCapacity {
228    /// Bytes of row metadata per row: the physical row index, the four
229    /// `i32` ids, the two times, `FLAG_ROW` and UVW.
230    const METADATA_BYTES_PER_ROW: usize = size_of::<usize>()
231        + 4 * size_of::<i32>()
232        + 2 * size_of::<f64>()
233        + size_of::<bool>()
234        + size_of::<[f64; 3]>();
235
236    /// One row of a DATA_DESCRIPTION whose blocks read `channels` channels of
237    /// `correlations` correlations from `visibility`, with input weights
238    /// from `weight`; `None` if the sample count overflows.
239    pub(crate) fn row(
240        visibility: SelectedVisibilityColumn,
241        weight: SelectedWeightColumn,
242        correlations: usize,
243        channels: usize,
244    ) -> Option<Self> {
245        let samples_per_row = correlations.checked_mul(channels)?;
246        Some(Self {
247            rows: 1,
248            samples_per_row,
249            weight_values_per_row: match weight {
250                SelectedWeightColumn::Weight => correlations,
251                SelectedWeightColumn::WeightSpectrum => samples_per_row,
252            },
253            sample_bytes: visibility.sample_bytes(),
254        })
255    }
256
257    /// Room for both `self` and `other`: the larger of each.
258    #[must_use]
259    pub(crate) fn union(self, other: Self) -> Self {
260        Self {
261            rows: self.rows.max(other.rows),
262            samples_per_row: self.samples_per_row.max(other.samples_per_row),
263            weight_values_per_row: self.weight_values_per_row.max(other.weight_values_per_row),
264            sample_bytes: self.sample_bytes.max(other.sample_bytes),
265        }
266    }
267
268    /// This capacity with room for `rows` rows.
269    #[must_use]
270    pub(crate) const fn with_rows(self, rows: usize) -> Self {
271        Self { rows, ..self }
272    }
273
274    /// The rows this capacity holds.
275    #[must_use]
276    pub(crate) const fn rows(self) -> usize {
277        self.rows
278    }
279
280    /// The bytes a buffer of this capacity retains; `None` on overflow.
281    pub(crate) fn retained_bytes(self) -> Option<usize> {
282        let samples = self.rows.checked_mul(self.samples_per_row)?;
283        let weights = self.rows.checked_mul(self.weight_values_per_row)?;
284        self.rows
285            .checked_mul(Self::METADATA_BYTES_PER_ROW)?
286            .checked_add(samples.checked_mul(self.sample_bytes + size_of::<bool>())?)?
287            .checked_add(weights.checked_mul(size_of::<f32>())?)
288    }
289
290    fn samples(self) -> usize {
291        self.rows
292            .checked_mul(self.samples_per_row)
293            .expect("an allocated buffer capacity fits usize")
294    }
295
296    fn weight_values(self) -> usize {
297        self.rows
298            .checked_mul(self.weight_values_per_row)
299            .expect("an allocated buffer capacity fits usize")
300    }
301}

303impl SelectedObservationBuffer {
304    /// An empty buffer allocated for blocks within `capacity`.
305    pub(crate) fn with_capacity(capacity: SelectedObservationBufferCapacity) -> Self {
306        let rows = capacity.rows;
307        Self {
308            capacity,
309            row_indices: Vec::with_capacity(rows),
310            channel_range: VisibilityChannelReadRange::new(0, 0),
311            correlation_count: 0,
312            // Installed at this capacity by the first fill, which knows
313            // the stored type.
314            visibility: None,
315            visibility_read: false,
316            flags: Vec::with_capacity(capacity.samples()),
317            weights: None,
318            row_flag: Vec::with_capacity(rows),
319            uvw_m: Vec::with_capacity(rows * 3),
320            data_description_ids: Vec::with_capacity(rows),
321            field_ids: Vec::with_capacity(rows),
322            antenna1: Vec::with_capacity(rows),
323            antenna2: Vec::with_capacity(rows),
324            time_mjd_seconds: Vec::with_capacity(rows),
325            time_centroid_mjd_seconds: Vec::with_capacity(rows),
326        }
327    }
328
329    /// The bytes this buffer's vectors hold allocated: each one's capacity,
330    /// not its length.
331    pub(crate) fn allocated_bytes(&self) -> usize {
332        fn bytes<T>(values: &Vec<T>) -> usize {
333            values.capacity() * size_of::<T>()
334        }
335        let visibility = match &self.visibility {
336            None => 0,
337            Some(SelectedStoredVisibilities::Float32(values)) => bytes(values),
338            Some(SelectedStoredVisibilities::Complex32(values)) => bytes(values),
339        };
340        let weights = match &self.weights {
341            None => 0,
342            Some(
343                SelectedStoredWeights::PerRow(values) | SelectedStoredWeights::PerChannel(values),
344            ) => bytes(values),
345        };
346        visibility
347            + weights
348            + bytes(&self.row_indices)
349            + bytes(&self.flags)
350            + bytes(&self.row_flag)
351            + bytes(&self.uvw_m)
352            + bytes(&self.data_description_ids)
353            + bytes(&self.field_ids)
354            + bytes(&self.antenna1)
355            + bytes(&self.antenna2)
356            + bytes(&self.time_mjd_seconds)
357            + bytes(&self.time_centroid_mjd_seconds)
358    }

360    pub(crate) fn numeric_columns(&self) -> Option<SelectedObservationNumericColumns<'_>> {
361        let visibility = match (self.visibility_read, self.visibility.as_ref()) {
362            (false, _) => SelectedNumericVisibility::Unread,
363            (true, Some(SelectedStoredVisibilities::Float32(values))) => {
364                SelectedNumericVisibility::Float32(values)
365            }
366            (true, Some(SelectedStoredVisibilities::Complex32(values))) => {
367                SelectedNumericVisibility::Complex32(values)
368            }
369            (true, None) => return None,
370        };
371        let weights = match self.weights.as_ref()? {
372            SelectedStoredWeights::PerRow(values) => SelectedNumericWeights::PerRow(values),
373            SelectedStoredWeights::PerChannel(values) => SelectedNumericWeights::PerChannel(values),
374        };
375        Some(SelectedObservationNumericColumns {
376            physical_rows: &self.row_indices,
377            channel_range: self.channel_range,
378            correlation_count: self.correlation_count,
379            visibility,
380            flags: &self.flags,
381            weights,
382            row_flags: &self.row_flag,
383        })
384    }
385    /// Number of selected MAIN rows in this block.
386    #[must_use]
387    pub(crate) fn row_count(&self) -> usize {
388        self.row_indices.len()
389    }
390
391    /// Return the MAIN metadata of one row of this block.
392    #[must_use]
393    pub(crate) fn row(&self, row_offset: usize) -> Option<SelectedStoredRow> {
394        let uvw_start = row_offset.checked_mul(3)?;
395        let uvw = self.uvw_m.get(uvw_start..uvw_start + 3)?;
396        Some(SelectedStoredRow {
397            physical_row: *self.row_indices.get(row_offset)?,
398            data_description_id: *self.data_description_ids.get(row_offset)?,
399            row_flag: *self.row_flag.get(row_offset)?,
400            uvw_m: [uvw[0], uvw[1], uvw[2]],
401            time_mjd_seconds: *self.time_mjd_seconds.get(row_offset)?,
402            time_centroid_mjd_seconds: *self.time_centroid_mjd_seconds.get(row_offset)?,
403            field_id: *self.field_ids.get(row_offset)?,
404            antenna1: *self.antenna1.get(row_offset)?,
405            antenna2: *self.antenna2.get(row_offset)?,
406        })
407    }

466impl MeasurementSet {
467    /// Fill one bounded block with the closed selected-observation storage column set.
468    ///
469    /// The row metadata (`DATA_DESC_ID`, `FIELD_ID`, the antennas, `TIME`,
470    /// `TIME_CENTROID`, `FLAG_ROW` and `UVW`) is copied from the request's
471    /// stored rows, which the row walk read; only the visibility, `FLAG` and
472    /// weight columns are read here, and the visibility column only when the
473    /// request names one.
474    ///
475    /// This selected-channel operation requires a lazily reopened disk-backed MeasurementSet with
476    /// no pending array-cell writes. It never falls back to cloning complete array cells.
477    ///
478    /// A request within the buffer's capacity grows none of its vectors.
479    pub(crate) fn fill_selected_observation_buffer(
480        &self,
481        request: &SelectedObservationBufferRequest<'_>,
482        buffer: &mut SelectedObservationBuffer,
483    ) -> MsResult<()> {
484        validate_request(self, request)?;
485        buffer.copy_stored_rows(request.rows);
486        buffer.channel_range = request.channel_range;
487        let samples = buffer.capacity.samples();
488        let weight_values = buffer.capacity.weight_values();
489        let row_indices = buffer.row_indices.as_slice();
490
491        buffer.visibility_read = false;
492        let visibility_correlations = match request.visibility {
493            Some(column) => {
494                let shape = self.fill_selected_visibilities(
495                    column,
496                    row_indices,
497                    request.channel_range,
498                    samples,
499                    &mut buffer.visibility,
500                )?;
501                require_shape(
502                    column.name(),
503                    shape.row_count,
504                    shape.channel_count,
505                    shape.axis0_count,
506                    request,
507                    shape.axis0_count,
508                )?;
509                buffer.visibility_read = true;
510                Some(shape.axis0_count)
511            }
512            None => None,
513        };
514
515        let flag_shape = self
516            .main_table()
517            .column_accessor("FLAG")?
518            .fill_array_cells_2d_channel_range_typed_uncached(
519                row_indices,
520                request.channel_range.start,
521                request.channel_range.count,
522                SelectedArray2DCellsMut::RowChannelBool(&mut buffer.flags),
523            )?
524            .ok_or_else(|| invalid("required MAIN FLAG cells are undefined"))?;
525        // Without the visibilities, FLAG gives the stored correlations.
526        let correlation_count = visibility_correlations.unwrap_or(flag_shape.axis0_count);
527        buffer.correlation_count = correlation_count;
528        require_shape(
529            "FLAG",
530            flag_shape.row_count,
531            flag_shape.channel_count,
532            flag_shape.axis0_count,
533            request,
534            correlation_count,
535        )?;
536
537        let weight_shape = match request.weight {
538            SelectedWeightColumn::Weight => {
539                if !matches!(buffer.weights, Some(SelectedStoredWeights::PerRow(_))) {
540                    buffer.weights = Some(SelectedStoredWeights::PerRow(Vec::with_capacity(
541                        weight_values,
542                    )));
543                }
544                let Some(SelectedStoredWeights::PerRow(values)) = buffer.weights.as_mut() else {
545                    unreachable!("per-row weight destination installed above")
546                };
547                let shape = self
548                    .main_table()
549                    .column_accessor("WEIGHT")?
550                    .fill_array_cells_1d_typed_uncached(
551                        row_indices,
552                        SelectedArray1DCellsMut::Float32(values),
553                    )?;
554                if shape.row_count != row_indices.len() || shape.axis0_count != correlation_count {
555                    return Err(invalid(
556                        "MAIN WEIGHT shape differs from selected visibility shape",
557                    ));
558                }
559                None
560            }
561            SelectedWeightColumn::WeightSpectrum => {
562                if !matches!(buffer.weights, Some(SelectedStoredWeights::PerChannel(_))) {
563                    buffer.weights = Some(SelectedStoredWeights::PerChannel(Vec::with_capacity(
564                        weight_values,
565                    )));
566                }
567                let Some(SelectedStoredWeights::PerChannel(values)) = buffer.weights.as_mut()
568                else {
569                    unreachable!("per-channel weight destination installed above")
570                };
571                Some(
572                    self.main_table()
573                        .column_accessor("WEIGHT_SPECTRUM")?
574                        .fill_array_cells_2d_channel_range_typed_uncached(
575                            row_indices,
576                            request.channel_range.start,
577                            request.channel_range.count,
578                            SelectedArray2DCellsMut::RowChannelFloat32(values),
579                        )?
580                        .ok_or_else(|| {
581                            invalid("required MAIN WEIGHT_SPECTRUM cells are undefined")
582                        })?,
583                )
584            }
585        };
586        if let Some(shape) = weight_shape {
587            require_shape(
588                "WEIGHT_SPECTRUM",
589                shape.row_count,
590                shape.channel_count,
591                shape.axis0_count,
592                request,
593                correlation_count,
594            )?;
595        }
596        debug_assert!(
597            buffer.allocated_bytes() <= buffer.capacity.retained_bytes().unwrap_or(usize::MAX),
598            "a selected-observation fill grew its buffer past the planned capacity"
599        );
600        Ok(())
601    }

603    /// Read `column`'s selected channels of `row_indices` into `destination`,
604    /// installing a vector of the stored type at `samples` the first time.
605    fn fill_selected_visibilities(
606        &self,
607        column: SelectedVisibilityColumn,
608        row_indices: &[usize],
609        channel_range: VisibilityChannelReadRange,
610        samples: usize,
611        destination: &mut Option<SelectedStoredVisibilities>,
612    ) -> MsResult<casa_tables::SelectedArray2DShape> {
613        let values = match column {
614            SelectedVisibilityColumn::FloatData => {
615                if !matches!(destination, Some(SelectedStoredVisibilities::Float32(_))) {
616                    *destination = Some(SelectedStoredVisibilities::Float32(Vec::with_capacity(
617                        samples,
618                    )));
619                }
620                let Some(SelectedStoredVisibilities::Float32(values)) = destination.as_mut() else {
621                    unreachable!("FLOAT_DATA destination installed above")
622                };
623                SelectedArray2DCellsMut::RowChannelFloat32(values)
624            }
625            SelectedVisibilityColumn::Data | SelectedVisibilityColumn::CorrectedData => {
626                if !matches!(destination, Some(SelectedStoredVisibilities::Complex32(_))) {
627                    *destination = Some(SelectedStoredVisibilities::Complex32(Vec::with_capacity(
628                        samples,
629                    )));
630                }
631                let Some(SelectedStoredVisibilities::Complex32(values)) = destination.as_mut()
632                else {
633                    unreachable!("complex visibility destination installed above")
634                };
635                SelectedArray2DCellsMut::RowChannelComplex32(values)
636            }
637        };
638        self.main_table()
639            .column_accessor(column.name())?
640            .fill_array_cells_2d_channel_range_typed_uncached(
641                row_indices,

643                channel_range.count,
644                values,
645            )?
646            .ok_or_else(|| {
647                invalid(format!(

649                    column.name()

655impl SelectedObservationBuffer {
656    /// Replace the row indices and row metadata with those of `rows`,
657    /// keeping the vectors' storage.
658    fn copy_stored_rows(&mut self, rows: &[SelectedObservationRow]) {
659        debug_assert!(
660            rows.len() <= self.capacity.rows,
661            "a selected-observation block exceeds its buffer's planned rows"
662        );
663        self.row_indices.clear();
664        self.data_description_ids.clear();
665        self.field_ids.clear();
666        self.antenna1.clear();
667        self.antenna2.clear();
668        self.time_mjd_seconds.clear();
669        self.time_centroid_mjd_seconds.clear();
670        self.row_flag.clear();
671        self.uvw_m.clear();
672        for row in rows {
673            self.row_indices.push(row.physical_row);
674            self.data_description_ids.push(row.data_description_id);
675            self.field_ids.push(row.field_id);
676            self.antenna1.push(row.antenna1);
677            self.antenna2.push(row.antenna2);
678            self.time_mjd_seconds.push(row.time_mjd_seconds);
679            self.time_centroid_mjd_seconds
680                .push(row.time_centroid_mjd_seconds);
681            self.row_flag.push(row.flag_row);
682            self.uvw_m.extend_from_slice(&row.uvw_m);
683        }
684    }
685}

687fn validate_request(
688    ms: &MeasurementSet,
689    request: &SelectedObservationBufferRequest<'_>,
690) -> MsResult<()> {
691    if request.rows.is_empty() {
692        return Err(invalid(
693            "selected-observation buffer requires at least one row",
694        ));
695    }
696    if request.channel_range.count == 0 {
697        return Err(invalid(
698            "selected-observation buffer requires at least one channel",
699        ));
700    }
701    if request
702        .rows
703        .iter()
704        .any(|row| row.physical_row >= ms.row_count())
705    {
706        return Err(invalid("selected-observation buffer row lies outside MAIN"));
707    }
708    Ok(())
709}

711fn require_shape(
712    column: &str,
713    rows: usize,
714    channels: usize,
715    correlations: usize,
716    request: &SelectedObservationBufferRequest<'_>,
717    expected_correlations: usize,
718) -> MsResult<()> {
719    if rows != request.rows.len()
720        || channels != request.channel_range.count
721        || correlations != expected_correlations
722    {
723        return Err(invalid(format!(
724            "MAIN {column} shape differs from selected visibility shape"
725        )));
726    }
727    Ok(())
728}

748    /// A buffer with room for the rows of `request`, each holding the
749    /// fixture's whole 2-correlation x 4-channel cells.
750    fn buffer_for(request: &SelectedObservationBufferRequest<'_>) -> SelectedObservationBuffer {
751        let visibility = request.visibility.unwrap_or(SelectedVisibilityColumn::Data);
752        SelectedObservationBuffer::with_capacity(
753            SelectedObservationBufferCapacity::row(visibility, request.weight, 2, 4)
754                .unwrap()
755                .with_rows(request.rows.len()),
756        )

759    /// The stored MAIN rows at `physical_rows`, in that order, as the row
760    /// walk reads them.
761    fn stored_rows(ms: &MeasurementSet, physical_rows: &[usize]) -> Vec<SelectedObservationRow> {
762        let plan = MsReadPlan::new(
763            ms.row_count(),
764            MsSelectionIoBudget {
765                available_bytes: ms.row_count() * SelectedObservationRow::STORAGE_BYTES_PER_ROW,
766                maximum_live_blocks: 1,
767                requested_bytes_per_row: SelectedObservationRow::STORAGE_BYTES_PER_ROW,
768                storage_alignment_rows: None,
769            },
770        )
771        .unwrap();
772        let mut cursor = ms.main_row_selection_cursor(plan).unwrap();
773        let mut rows = Vec::new();
774        while let Some(row) = cursor.next(ms).unwrap() {
775            rows.push(row);
776        }
777        physical_rows.iter().map(|row| rows[*row]).collect()
778    }
