2//! The rows a table holds on disk, which a partial in-place save must leave
3//! `table.dat` describing (#880).
4//!
5//! casacore rewrites `table.dat` only when a table's structure changed or
6//! the table is closed (`PlainTable::putFile`). Adding rows does not count
7//! as a change, so between those writes it publishes new rows in
8//! `table.lock`'s sync data alone, and `table.dat` trails them; a process
9//! that never closes the table, as CASA 6.7.6's simulator after a mosaic
10//! predict, leaves it so. casa-rs opens such a table with the published
11//! count. A partial save reads `table.dat` back from disk and writes it
12//! again, so it must bring the file's counts up to the table's: casacore
13//! checks every storage manager's rows against them.
14
15use super::*;
16use crate::storage::table_control::TableDatContents;
17
18/// The rows the table at `table_dir` holds as persisted, counted as
19/// [`Table::open`] counts them: the largest of `table.dat`'s two counts and
20/// the count the sync data publishes.
21pub(super) fn persisted_row_count(table_dir: &Path, table_dat: &TableDatContents) -> u64 {
22    table_dat
23        .nrrow
24        .max(table_dat.column_set.nrrow)
25        .max(published_row_count(table_dir))
26}
27
28/// The row count `table.lock`'s sync data publishes for the table at
29/// `table_dir`, 0 when there is none to read.
30pub(super) fn published_row_count(table_dir: &Path) -> u64 {
31    #[cfg(unix)]
32    {
33        crate::lock::read_sync_data_from_table_dir(table_dir)
34            .ok()
35            .flatten()
36            .map_or(0, |sync| sync.nrrow)
37    }
38    #[cfg(not(unix))]
39    {
40        let _ = table_dir;
41        0
42    }
43}
44
45impl Table {
46    /// Check that a partial in-place save may write the table at
47    /// `table_dir`, whose `table.dat` is `table_dat`, and return the rows the
48    /// table holds there.
49    ///
50    /// A partial save rewrites some data managers and leaves the others, so
51    /// this handle's rows must be the ones on disk: none added, inserted or
52    /// removed since the table was opened or last saved in full, and as many
53    /// as the disk holds. Otherwise the rewritten managers would disagree
54    /// with the others, or with `table.dat`, about the rows, and casacore
55    /// refuses the table. The check writes nothing, so a refused save leaves
56    /// the disk as it was.
57    pub(super) fn require_persisted_rows(
58        &self,
59        table_dir: &Path,
60        table_dat: &TableDatContents,
61    ) -> Result<u64, TableError> {
62        if self.inner.rows_reshaped() {
63            return Err(TableError::Storage(format!(
64                "a partial in-place save cannot write {}: rows were added or removed since \
65                 the table was opened or saved in full; save the whole table instead",
66                table_dir.display()
67            )));
68        }
69        let persisted = persisted_row_count(table_dir, table_dat);
70        if self.row_count() as u64 != persisted {
71            return Err(TableError::Storage(format!(
72                "a partial in-place save cannot write {}: it holds {persisted} rows on disk \
73                 and this handle {}; reopen the table",
74                table_dir.display(),
75                self.row_count()
76            )));
77        }
78        Ok(persisted)
79    }
80
81    /// Record a successful full save to `path`: when that is the table's own
82    /// directory, its rows on disk now match this handle's.
83    pub(super) fn note_full_save(&self, path: &Path) {
84        if self
85            .source_path
86            .as_deref()
87            .is_some_and(|source| locking::same_directory(source, path))
88        {
89            self.inner.note_rows_persisted();
90        }
91    }
92}
