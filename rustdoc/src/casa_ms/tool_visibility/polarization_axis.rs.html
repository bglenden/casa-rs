18/// In-row selection of recorded correlation products or derived Stokes values.
19pub struct VisibilityPolarization {
20    project: Projection,
21}
22
23type RecordedCorrelations = Arc<HashMap<i32, ToolCorrelationSelection>>;
24
25enum Projection {
26    Stored(RecordedCorrelations),
27    Converted {
28        types: Vec<StokesType>,
29        plans: HashMap<i32, Conversion>,
30        recorded: Option<RecordedCorrelations>,
31    },
32}
33
34impl VisibilityPolarization {
35    /// Use the recorded products selected by the CASA MS selection grammar.
36    pub fn stored(matches: Vec<ToolCorrelationSelection>) -> Self {
37        Self {
38            project: Projection::Stored(Arc::new(
39                matches
40                    .into_iter()
41                    .map(|plan| (plan.description_id, plan))
42                    .collect(),
43            )),
44        }
45    }
46
47    /// Convert the selected rows to the requested Stokes or correlation types.
48    /// # Errors
49    /// The DATA_DESCRIPTION or POLARIZATION rows cannot be read.
50    pub fn converted(ms: &MeasurementSet, types: Vec<StokesType>) -> MsResult<Self> {
51        let descriptions = ms.data_description()?;
52        let polarization = ms.polarization()?;
53        let mut plans = HashMap::with_capacity(descriptions.row_count());
54        for id in 0..descriptions.row_count() {
55            if descriptions.flag_row(id)? {
56                continue;
57            }
58            let pol_id = usize::try_from(descriptions.polarization_id(id)?)
59                .map_err(|_| input("negative POLARIZATION_ID"))?;
60            let codes = polarization.corr_type(pol_id)?;
61            plans.insert(id as i32, Conversion::new(&codes, &types));
62        }
63        Ok(Self {
64            project: Projection::Converted {
65                types,
66                plans,
67                recorded: None,
68            },
69        })
70    }
71
72    /// Apply this conversion to products already selected by an iterator.
73    /// Each conversion uses the physical correlation labels of its input.
74    ///
75    /// # Errors
76    /// Requires a conversion and a stored-product selection respectively.
77    pub fn over_recorded_selection(&self, selection: &Self) -> MsResult<Self> {
78        let (Projection::Converted { types, .. }, Projection::Stored(recorded)) =
79            (&self.project, &selection.project)
80        else {
81            return Err(input("expected converted and recorded polarization plans"));
82        };
83        let plans = recorded
84            .iter()
85            .map(|(&id, selection)| {
86                let codes = selection
87                    .types
88                    .iter()
89                    .map(|kind| kind.code())
90                    .collect::<Vec<_>>();
91                (id, Conversion::new(&codes, types))
92            })
93            .collect();
94        Ok(Self {
95            project: Projection::Converted {
96                types: types.clone(),
97                plans,
98                recorded: Some(Arc::clone(recorded)),
99            },
100        })
101    }
102
103    /// Whether complex visibilities are converted to derived polarization values.
104    pub fn is_conversion(&self) -> bool {
105        matches!(self.project, Projection::Converted { .. })
106    }
107
108    pub(super) fn write_indices(&self, id: i32) -> MsResult<&[usize]> {
109        let Projection::Stored(plans) = &self.project else {
110            return Err(input("putdata cannot write derived polarization"));
111        };
112        plans
113            .get(&id)
114            .map(|plan| plan.indices.as_slice())
115            .ok_or_else(|| missing_description(id))
116    }
117
118    pub(super) fn types(&self, id: i32) -> MsResult<&[StokesType]> {
119        match &self.project {
120            Projection::Stored(plans) => plans
121                .get(&id)
122                .map(|plan| plan.types.as_slice())
123                .ok_or_else(|| missing_description(id)),
124            Projection::Converted { types, .. } => Ok(types),
125        }
126    }
127
128    pub(super) fn gather(&self, values: ArrayValue, ids: &[i32]) -> MsResult<ArrayValue> {
129        let plans = match &self.project {
130            Projection::Stored(plans) => plans,
131            Projection::Converted {
132                recorded: Some(plans),
133                ..
134            } => plans,
135            Projection::Converted { recorded: None, .. } => return Ok(values),
136        };
137        macro_rules! gather {
138            ($($kind:ident),+) => {
139                match values {
140                    $(ArrayValue::$kind(values) => gather_correlations(&values, ids, plans)
141                        .map(ArrayValue::$kind),)+
142                    _ => Err(input("invalid correlation array element type")),
143                }
144            };
145        }
146        gather!(Bool, Float32, Complex32, Complex64)
147    }
148
149    pub(super) fn apply_conversion(&self, values: ArrayValue, ids: &[i32]) -> MsResult<ArrayValue> {
150        let Projection::Converted { types, plans, .. } = &self.project else {
151            return Ok(values);
152        };
153        let ArrayValue::Complex32(values) = values else {
154            return Err(input(
155                "polarization conversion requires Complex visibility data",
156            ));
157        };
158        project_rows(&values, ids, types.len(), plans, |plan, samples, output| {
159            plan.append(samples, output)
160        })
161        .map(ArrayValue::Complex32)
162    }
163
164    pub(super) fn converted_flags(
165        &self,
166        flags: &ArrayD<bool>,
167        ids: &[i32],
168    ) -> MsResult<ArrayD<bool>> {
169        let Projection::Converted { types, plans, .. } = &self.project else {
170            return Err(input("Stokes projection is missing"));
171        };
172        project_rows(flags, ids, types.len(), plans, Conversion::append_flags)
173    }
174
175    pub(super) fn averaging_inputs(
176        &self,
177        flags: ArrayValue,
178        weights: ArrayValue,
179        ids: &[i32],
180    ) -> MsResult<(ArrayValue, ArrayValue)> {
181        let Projection::Converted { types, plans, .. } = &self.project else {
182            return Ok((flags, weights));
183        };
184        let (ArrayValue::Bool(flags), ArrayValue::Float32(weights)) = (flags, weights) else {
185            return Err(input(
186                "polarization averaging requires Bool FLAG and Float WEIGHT",
187            ));
188        };
189        let width = types.len();
190        let [_, channels, _] = flags.shape() else {
191            return Err(input("polarization averaging requires a FLAG cube"));
192        };
193        let row_size = width * channels;
194        if row_size == 0 {
195            return Err(input(
196                "polarization averaging requires nonempty correlation and channel axes",
197            ));
198        }
199        let mut flags = self.converted_flags(&flags, ids)?;
200        let weights = project_rows(&weights, ids, width, plans, Conversion::append_weights)?;
201        let flags_mem = flags
202            .as_slice_memory_order_mut()
203            .expect("projected flags are contiguous");
204        let weights_mem = weights
205            .as_slice_memory_order()
206            .expect("projected weights are contiguous");
207        for (row_flags, row_weights) in flags_mem
208            .chunks_exact_mut(row_size)
209            .zip(weights_mem.chunks_exact(width))
210        {
211            for cell in row_flags.chunks_exact_mut(width) {
212                for (flag, &weight) in cell.iter_mut().zip(row_weights) {
213                    *flag |= weight <= 0.0;
214                }
215            }
216        }
217        Ok((ArrayValue::Bool(flags), ArrayValue::Float32(weights)))
218    }
219}

221fn gather_correlations<T: Copy>(
222    values: &ArrayD<T>,
223    ids: &[i32],
224    plans: &HashMap<i32, ToolCorrelationSelection>,
225) -> MsResult<ArrayD<T>> {
226    let id = ids
227        .first()
228        .ok_or_else(|| input("no selected polarization rows"))?;
229    let width = plans
230        .get(id)
231        .ok_or_else(|| missing_description(*id))?
232        .indices
233        .len();
234    project_rows(values, ids, width, plans, |plan, samples, output| {
235        if plan.indices.len() != width {
236            return Err(input(
237                "selected data descriptions have different correlation counts",
238            ));
239        }
240        for &index in &plan.indices {
241            output.push(
242                *samples
243                    .get(index)
244                    .ok_or_else(|| input("selected correlation is outside the visibility axis"))?,
245            );
246        }
247        Ok(())
248    })
249}

251fn project_rows<T: Copy, P>(
252    values: &ArrayD<T>,
253    ids: &[i32],
254    width: usize,
255    plans: &HashMap<i32, P>,
256    mut project: impl FnMut(&P, &[T], &mut Vec<T>) -> MsResult<()>,
257) -> MsResult<ArrayD<T>> {
258    let mut shape = values.shape().to_vec();
259    let input_width = shape.first().copied().unwrap_or(0);
260    if shape.len() < 2 || input_width == 0 || shape.last().copied() != Some(ids.len()) {
261        return Err(input(
262            "visibility array has invalid correlation or row dimensions",
263        ));
264    }
265    let memory = values
266        .as_slice_memory_order()
267        .ok_or_else(|| input("visibility correlation array is not contiguous"))?;
268    let cells_per_row = shape[1..shape.len() - 1].iter().product::<usize>();
269    shape[0] = width;
270    let mut output = Vec::with_capacity(super::baseline_axis::element_count(&shape)?);
271    if cells_per_row == 0 {
272        return shaped(&shape, output);
273    }
274    for (&id, row) in ids
275        .iter()
276        .zip(memory.chunks_exact(input_width * cells_per_row))
277    {
278        let plan = plans.get(&id).ok_or_else(|| missing_description(id))?;
279        for samples in row.chunks_exact(input_width) {
280            project(plan, samples, &mut output)?;
281        }
282    }
283    shaped(&shape, output)
284}
285
286fn missing_description(id: i32) -> crate::MsError {
287    input(format!(
288        "no polarization selection for data description {id}"
289    ))
