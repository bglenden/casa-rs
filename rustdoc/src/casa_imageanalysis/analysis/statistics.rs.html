5pub(super) fn imstat_typed<T>(
6    path: &Path,
7    image: &PagedImage<T>,
8    request: &ImstatRequest,
9) -> Result<ImstatOutcome, ImageError>
10where
11    T: ImagePixel + ImageExprValue + PartialOrd + ExprValueConvert + StatsElement,
12{
13    let algorithm = parse_imstat_algorithm(request).map_err(ImageError::InvalidMetadata)?;
14    let axes =
15        parse_axes(request.axes.as_deref(), image.ndim()).map_err(ImageError::CasaSelection)?;
16    let region_selection = region_pixels(image, request.region.as_deref())?;
17    if request
18        .box_pixels
19        .as_deref()
20        .is_some_and(|text| !text.trim().is_empty())
21        && region_selection.is_some()
22    {
23        return Err(ImageError::InvalidMetadata(
24            "imstat accepts either box or region, not both".to_string(),
25        ));
26    }
27    let explicit_box = match request
28        .box_pixels
29        .as_deref()
30        .filter(|text| !text.trim().is_empty())
31    {
32        Some(box_text) => Some(parse_box_region(
33            box_text,
34            image.shape(),
35            direction_plane_axes(image)?,
36        )?),
37        None => None,
38    };
39    let effective_region = explicit_box.as_ref().or(region_selection.as_ref());
40    let selection = Selection::new(
41        image,
42        effective_region,
43        request.chans.as_deref(),
44        request.stokes.as_deref(),
45    )?;
46    let pixel_mask = selection_pixel_mask(
47        image,
48        effective_region,
49        &selection,
50        request.mask.as_deref(),
51        request.stretch,
52    )?;
53    let coordinates = image.coordinates();
54    let world_text = |position: &[usize]| -> Result<String, ImageError> {
55        let pixel: Vec<f64> = position.iter().map(|&value| value as f64).collect();
56        Ok(casa_coordinates::format_world_display(coordinates, &pixel)?)
57    };
58    let blc = selection.start.clone();
59    let trc = trc_pixels(&selection.start, &selection.shape);
60    let blcf = world_text(&blc)?;
61    let trcf = world_text(&trc)?;
62
63    let display_axes: Vec<usize> = match &axes {
64        AxisSelection::All => Vec::new(),
65        AxisSelection::Collapsed(collapse) => (0..image.ndim())
66            .filter(|axis| !collapse.contains(axis))
67            .collect(),
68    };
69    if display_axes.is_empty() {
70        let summary = imstat_global_summary(
71            path,
72            image,
73            request,
74            &algorithm,
75            &selection,
76            pixel_mask.as_ref(),
77            blc,
78            trc,
79            blcf,
80            trcf,
81            &world_text,
82        )?;
83        Ok(ImstatOutcome::Global(summary))
84    } else {
85        let summary = imstat_planes_summary(
86            path,
87            image,
88            request,
89            &algorithm,
90            &selection,
91            pixel_mask.as_ref(),
92            &display_axes,
93            blc,
94            trc,
95            blcf,
96            trcf,
97        )?;
98        Ok(ImstatOutcome::Planes(summary))
99    }
100}

102/// The whole selection's statistics, as `imstat` reports them with every
103/// axis collapsed.
104#[allow(clippy::too_many_arguments)]
105pub(super) fn imstat_global_summary<T>(
106    path: &Path,
107    image: &PagedImage<T>,
108    request: &ImstatRequest,
109    algorithm: &StatisticsAlgorithm,
110    selection: &Selection,
111    pixel_mask: Option<&ndarray::ArrayD<bool>>,
112    blc: Vec<usize>,
113    trc: Vec<usize>,
114    blcf: String,
115    trcf: String,
116    world_text: &dyn Fn(&[usize]) -> Result<String, ImageError>,
117) -> Result<ImageStatisticsSummary, ImageError>
118where
119    T: ImagePixel + StatsElement,
120{
121    // The statistics the algorithm reports, with the extremum positions
122    // relative to the original image.
123    let computed = match algorithm {
124        StatisticsAlgorithm::Classic
125            if !selection.stokes_has_gaps() && !selection.channels_have_gaps() =>
126        {
127            // The classical moments stream through the lattice
128            // statistics; a gapped Stokes set needs the sample route.
129            let sub = image.sub_image(selection.start.clone(), selection.shape.clone())?;
130            let mut stats = LatticeStatistics::new(&sub);
131            if let Some([min, max]) = request.includepix {
132                stats.set_include_range(min, max);
133            }
134            if let Some(mask) = pixel_mask.cloned() {
135                stats.set_pixel_mask(mask);
136            }
137            let (minpos, maxpos) = stats.get_min_max_pos()?;
138            ComputedStats {
139                npts: scalar_stat(&stats, Statistic::Npts)?,
140                sum: scalar_stat(&stats, Statistic::Sum)?,
141                min: scalar_stat(&stats, Statistic::Min)?,
142                max: scalar_stat(&stats, Statistic::Max)?,
143                sumsq: scalar_stat(&stats, Statistic::SumSq)?,
144                mean: scalar_stat(&stats, Statistic::Mean)?,
145                rms: scalar_stat(&stats, Statistic::Rms)?,
146                sigma: scalar_stat(&stats, Statistic::Sigma)?,
147                median: scalar_stat(&stats, Statistic::Median)?,
148                medabsdevmed: scalar_stat(&stats, Statistic::MedAbsDevMed)?,
149                q1: scalar_stat(&stats, Statistic::Q1)?,
150                q3: scalar_stat(&stats, Statistic::Q3)?,
151                quartile: scalar_stat(&stats, Statistic::Quartile)?,
152                minpos: minpos.map(|pos| offset_pos(&selection.start, pos)),
153                maxpos: maxpos.map(|pos| offset_pos(&selection.start, pos)),
154            }
155        }
156        robust => {
157            let pixels = image.get_slice(&selection.start, &selection.shape)?;
158            let sample = sample_points(
159                &pixels,
160                pixel_mask,
161                request.includepix,
162                selection.channel_axis,
163                selection.channel_indices.as_deref(),
164                selection.stokes_axis,
165                selection.stokes_indices.as_deref(),
166                &selection.start,
167            );
168            match compute_robust_stats(&sample, robust)? {
169                RobustStats::Sample(stats) => ComputedStats {
170                    npts: stats.npts,
171                    sum: stats.sum,
172                    min: stats.min,
173                    max: stats.max,
174                    sumsq: stats.sumsq,
175                    mean: stats.mean,
176                    rms: stats.rms,
177                    sigma: stats.sigma,
178                    median: stats.median,
179                    medabsdevmed: stats.medabsdevmed,
180                    q1: stats.q1,
181                    q3: stats.q3,
182                    quartile: stats.quartile,
183                    minpos: stats.minpos.map(|linear| {
184                        offset_pos(
185                            &selection.start,
186                            selection_position(linear, &selection.shape),
187                        )
188                    }),
189                    maxpos: stats.maxpos.map(|linear| {
190                        offset_pos(
191                            &selection.start,
192                            selection_position(linear, &selection.shape),
193                        )
194                    }),
195                },
196                RobustStats::Biweight { location, scale } => {
197                    // The biweight reports only its location and scale,
198                    // with the sample's count and extrema.
199                    let npts = sample.len() as f64;
200                    let min = sample.iter().map(|p| p.value).fold(f64::NAN, f64::min);
201                    let max = sample.iter().map(|p| p.value).fold(f64::NAN, f64::max);
202                    ComputedStats {
203                        npts,
204                        sum: f64::NAN,
205                        min,
206                        max,
207                        sumsq: f64::NAN,
208                        mean: location,
209                        rms: f64::NAN,
210                        sigma: scale,
211                        median: f64::NAN,
212                        medabsdevmed: f64::NAN,
213                        q1: f64::NAN,
214                        q3: f64::NAN,
215                        quartile: f64::NAN,
216                        minpos: None,
217                        maxpos: None,
218                    }
219                }
220            }
221        }
222    };
223
224    let minposf = match &computed.minpos {
225        Some(position) => Some(world_text(position)?),
226        None => None,
227    };
228    let maxposf = match &computed.maxpos {
229        Some(position) => Some(world_text(position)?),
230        None => None,
231    };
232    let flux = if algorithm.kind() == StatisticsAlgorithmKind::Biweight {
233        None
234    } else {
235        image_flux(image, computed.sum, selection)
236    };
237    Ok(ImageStatisticsSummary {
238        blcf,
239        trcf,
240        npts: computed.npts,
241        min: computed.min,
242        max: computed.max,
243        sum: computed.sum,
244        sumsq: computed.sumsq,
245        mean: computed.mean,
246        rms: computed.rms,
247        sigma: computed.sigma,
248        median: computed.median,
249        medabsdevmed: computed.medabsdevmed,
250        q1: computed.q1,
251        q3: computed.q3,
252        quartile: computed.quartile,
253        minposf,
254        maxposf,
255        minpos: computed.minpos,
256        maxpos: computed.maxpos,
257        imagename: path.display().to_string(),
258        blc,
259        trc,
260        flux,
261        units: image.units().to_string(),
262    })
263}

265/// Each display plane's statistics, as `imstat` reports them with only
266/// some axes collapsed: one value per display position, in display
267/// order, with CASA's degenerate display axes dropped from the record's
268/// shape.
269#[allow(clippy::too_many_arguments)]
270pub(super) fn imstat_planes_summary<T>(
271    path: &Path,
272    image: &PagedImage<T>,
273    request: &ImstatRequest,
274    algorithm: &StatisticsAlgorithm,
275    selection: &Selection,
276    pixel_mask: Option<&ndarray::ArrayD<bool>>,
277    display_axes: &[usize],
278    blc: Vec<usize>,
279    trc: Vec<usize>,
280    blcf: String,
281    trcf: String,
282) -> Result<ImstatPlanesSummary, ImageError>
283where
284    T: ImagePixel + StatsElement,
285{
286    let display_positions: Vec<Vec<usize>> = display_axes
287        .iter()
288        .map(|&axis| selection.axis_positions(axis))
289        .collect();
290    let display_counts: Vec<usize> = display_positions.iter().map(|p| p.len()).collect();
291    let display_shape: Vec<usize> = display_counts
292        .iter()
293        .copied()
294        .filter(|&count| count > 1)
295        .collect();
296    let plane_count: usize = display_counts.iter().product();
297
298    // One pass over the selection's pixels, bucketed by display
299    // position: each pixel's plane is its display coordinates' index.
300    let pixels = image.get_slice(&selection.start, &selection.shape)?;
301    let mut buckets: Vec<Vec<SamplePoint>> = vec![Vec::new(); plane_count];
302    let stokes_collapsed = selection
303        .stokes_axis
304        .is_some_and(|axis| display_axes.iter().all(|&display| display != axis));
305    let mut strides = Vec::with_capacity(pixels.shape().len());
306    let mut stride = 1u64;
307    for &axis_length in pixels.shape() {
308        strides.push(stride);
309        stride *= axis_length as u64;
310    }
311    for (index, value) in pixels.indexed_iter() {
312        if let Some(mask) = pixel_mask {
313            let mask_index: &[usize] = index.slice();
314            if !mask[mask_index] {
315                continue;
316            }
317        }
318        let number = value.to_f64_stats();
319        if let Some([min, max]) = request.includepix
320            && !(min <= number && number <= max)
321        {
322            continue;
323        }
324        let mut linear = 0u64;
325        for (axis, &axis_stride) in strides.iter().enumerate() {
326            linear += index[axis] as u64 * axis_stride;
327        }
328        let mut bucket = 0usize;
329        // The display positions run in Fortran order, the order casacore
330        // keeps array data in and casa-rs's records carry.
331        let mut slot_stride = 1usize;
332        let mut in_selection = true;
333        for (slot, &axis) in display_axes.iter().enumerate() {
334            let absolute = index[axis] + selection.start[axis];
335            let Some(slot_index) = display_positions[slot]
336                .iter()
337                .position(|&position| position == absolute)
338            else {
339                in_selection = false;
340                break;
341            };
342            bucket += slot_index * slot_stride;
343            slot_stride *= display_counts[slot];
344        }
345        if !in_selection {
346            continue;
347        }
348        if stokes_collapsed {
349            let axis = selection.stokes_axis.expect("the axis was found above");
350            let kept = selection.stokes_indices.as_deref().unwrap_or(&[]);
351            let absolute = index[axis] + selection.start[axis];
352            if !kept.contains(&absolute) {
353                continue;
354            }
355        }
356        buckets[bucket].push(SamplePoint {
357            value: number,
358            position: linear,
359        });
360    }
361
362    let biweight = algorithm.kind() == StatisticsAlgorithmKind::Biweight;
363    let spectral_collapsed = image
364        .find_axis(CoordinateType::Spectral)
365        .is_some_and(|axis| display_axes.iter().all(|&display| display != axis));
366    // CASA's flux density exists per plane only when the direction plane
367    // is at least partly collapsed, as casacore beam-integrates the sky
368    // it sums: a plane that still varies over the whole sky has none.
369    let [plane_x, plane_y] = selection.plane;
370    let flux_over_sky = !display_axes.contains(&plane_x) || !display_axes.contains(&plane_y);
371    let mut planes = ImstatPlanesSummary {
372        imagename: path.display().to_string(),
373        blc,
374        trc,
375        blcf,
376        trcf,
377        display_shape,
378        plane_log: String::new(),
379        npts: Vec::with_capacity(plane_count),
380        min: Vec::with_capacity(plane_count),
381        max: Vec::with_capacity(plane_count),
382        sum: Vec::with_capacity(plane_count),
383        sumsq: Vec::with_capacity(plane_count),
384        mean: Vec::with_capacity(plane_count),
385        rms: Vec::with_capacity(plane_count),
386        sigma: Vec::with_capacity(plane_count),
387        median: Vec::with_capacity(plane_count),
388        medabsdevmed: Vec::with_capacity(plane_count),
389        q1: Vec::with_capacity(plane_count),
390        q3: Vec::with_capacity(plane_count),
391        quartile: Vec::with_capacity(plane_count),
392        flux: None,
393        biweight,
394    };
395    let mut iterations = Vec::new();
396    for bucket in &buckets {
397        let mut sample = bucket.clone();
398        sample.sort_by_key(|point| point.position);
399        match compute_robust_stats(&sample, algorithm)? {
400            RobustStats::Sample(stats) => {
401                iterations.push(stats.chauvenet_iterations);
402                planes.npts.push(stats.npts);
403                planes.min.push(stats.min);
404                planes.max.push(stats.max);
405                planes.sum.push(stats.sum);
406                planes.sumsq.push(stats.sumsq);
407                planes.mean.push(stats.mean);
408                planes.rms.push(stats.rms);
409                planes.sigma.push(stats.sigma);
410                planes.median.push(stats.median);
411                planes.medabsdevmed.push(stats.medabsdevmed);
412                planes.q1.push(stats.q1);
413                planes.q3.push(stats.q3);
414                planes.quartile.push(stats.quartile);
415                let flux = if flux_over_sky {
416                    image_flux_with_integration(image, stats.sum, selection, spectral_collapsed)
417                } else {
418                    None
419                };
420                if let Some(flux) = flux {
421                    planes
422                        .flux
423                        .get_or_insert_with(|| Vec::with_capacity(plane_count))
424                        .push(flux);
425                }
426            }
427            RobustStats::Biweight { location, scale } => {
428                let count = sample.len() as f64;
429                let min = sample.iter().map(|p| p.value).fold(f64::NAN, f64::min);
430                let max = sample.iter().map(|p| p.value).fold(f64::NAN, f64::max);
431                planes.npts.push(count);
432                planes.min.push(min);
433                planes.max.push(max);
434                planes.sum.push(f64::NAN);
435                planes.sumsq.push(f64::NAN);
436                planes.mean.push(location);
437                planes.rms.push(f64::NAN);
438                planes.sigma.push(scale);
439                planes.median.push(f64::NAN);
440                planes.medabsdevmed.push(f64::NAN);
441                planes.q1.push(f64::NAN);
442                planes.q3.push(f64::NAN);
443                planes.quartile.push(f64::NAN);
444            }
445        }
446    }
447    if biweight {
448        planes.flux = None;
449    }
450    if request.listit
451        || request
452            .logfile
453            .as_deref()
454            .is_some_and(|name| !name.is_empty())
455    {
456        planes.plane_log = imstat_log::plane_table(
457            image,
458            &planes,
459            display_axes,
460            &display_positions,
461            &iterations,
462        )?;
463    }
464    Ok(planes)
465}

487/// Converts a sum of image pixels to the flux reported by CASA statistics.
488///
489/// The factor uses square arcseconds for temperature and pixels per beam for
490/// per-beam brightness. If the spectral axis is collapsed over multiple
491/// channels, `channels` gives the first two measured spectral pixels.
492pub fn statistics_flux_scale(
493    coordinates: &CoordinateSystem,
494    units: &str,
495    beam: Option<GaussianBeam>,
496    channels: Option<[usize; 2]>,
497) -> Option<f64> {
498    let pixel_area = direction_pixel_area(coordinates)?;
499    let flux_factor = if units.contains('K') {
500        let arcsec_per_rad = 180.0 * 3600.0 / std::f64::consts::PI;
501        pixel_area * arcsec_per_rad * arcsec_per_rad
502    } else if units.contains("/beam") {
503        let beam_area = beam?.area();
504        if beam_area <= 0.0 {
505            return None;
506        }
507        pixel_area / beam_area
508    } else {
509        return None;
510    };
511    let channel_width = channels
512        .and_then(|pixels| spectral_integration(coordinates, pixels))
513        .unwrap_or(1.0);
514    Some(flux_factor * channel_width)
515}
516
517/// The width casacore integrates `flux` over: one channel of the selected
518/// spectral axis, when the selection holds more than one.
519pub(super) fn spectral_integration(
520    coordinates: &CoordinateSystem,
521    [first, second]: [usize; 2],
522) -> Option<f64> {
523    let coordinate = coordinates.coordinate(coordinates.find_coordinate(CoordinateType::Spectral)?);
524    let rest = spectral_rest_frequency(coordinates).unwrap_or(0.0);
525    if rest > 0.0 {
526        // The selection's own first two channels, as casacore's sub-image.
527        let f0 = coordinate.to_world(&[first as f64]).ok()?[0];
528        let f1 = coordinate.to_world(&[second as f64]).ok()?[0];
529        Some(SPEED_OF_LIGHT_KM_S * (f1 - f0).abs() / rest)
530    } else {
531        Some(coordinate.increment()[0] * second.abs_diff(first) as f64)
532    }
533}
534
535pub(super) fn direction_pixel_area(coords: &CoordinateSystem) -> Option<f64> {
536    for idx in 0..coords.n_coordinates() {
537        let coord = coords.coordinate(idx);
538        if coord.coordinate_type() == CoordinateType::Direction && coord.n_pixel_axes() >= 2 {
539            let increment = coord.increment();
540            return Some(increment[0].abs() * increment[1].abs());
541        }
542    }
543    None
