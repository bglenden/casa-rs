2//! The process allocator of the imaging apps, which returns freed working
3//! buffers to the system at once so that a run's footprint follows the memory
4//! its phases admit (plan section 7).
5//!
6//! Admission ([`crate::admit`]) charges the memory each phase holds live. The
7//! system allocators also keep large blocks after they are freed:
8//!
9//! - macOS's libmalloc caches them dirty in the process. `vmmap` lists them as
10//!   "Malloc Large (empty)"; `malloc_zone_statistics` does not count them and
11//!   `malloc_zone_pressure_relief` does not release them.
12//! - glibc raises its mmap threshold after such a block is freed, then serves
13//!   later blocks from an arena that it seldom trims.
14//!
15//! A run that allocates and frees per-plane buffers therefore holds its live
16//! charge plus the buffers it freed. The 512-channel cube's footprint reached
17//! 15.8 GB against 12.9 GB admitted; with libmalloc's large cache disabled it
18//! was 12.7 GB.
19//!
20//! [`ReturningAllocator`] maps each block of at least
21//! [`RETURNED_BLOCK_BYTES`] from the system and unmaps it when it is freed.
22//! Smaller blocks go to the system allocator, where the space a freed block
23//! leaves is bounded by its size class. The process that runs imaging
24//! (`casars-imager`; the other frontends start it) declares the allocator:
25//!
26//! ```ignore
27//! #[global_allocator]
28//! static ALLOCATOR: casa_imaging_runtime::ReturningAllocator =
29//!     casa_imaging_runtime::ReturningAllocator;
30//! ```
31//!
32//! On glibc this is what `mallopt(M_MMAP_THRESHOLD, 1 MiB)` with a matching
33//! `M_TRIM_THRESHOLD` would do; one mechanism serves both systems so that
34//! the `returning_allocator` law tests the same thing on each. Launch-time
35//! settings such as libmalloc's `MallocLargeCache=0` take effect only if
36//! they are in the environment before the process starts, which a binary
37//! run from a shell cannot ensure for itself.
38//!
39//! What this does not reach: memory C libraries allocate with `malloc`,
40//! FFTW's plans and planner scratch among them, still goes to the system
41//! allocator and its cache. Admission charges FFTW's plan cache; planner
42//! scratch freed into libmalloc's cache is outside both.
43//!
44//! A returned block's pages fault in afresh when they are allocated again. On
45//! the cube that costs about 3% of the run, mostly system time; pooling the
46//! pass's per-plane buffers (IF-10) removes most of it. Each live mapped
47//! block is one or more map entries: under a 16 GiB ceiling at most 16 Ki
48//! blocks, below Linux's default `vm.max_map_count` of 65,530. If pooling
49//! alone keeps a run's footprint within its admission with the system
50//! allocator, this allocator is deleted (#657).
51
52use std::alloc::{GlobalAlloc, Layout, System};
53
54/// The smallest block [`ReturningAllocator`] maps from the system; smaller
55/// blocks use the system allocator.
56pub const RETURNED_BLOCK_BYTES: usize = 1 << 20;
57
58/// A global allocator that returns every block of at least
59/// [`RETURNED_BLOCK_BYTES`] to the system when it is freed (module docs).
60///
61/// Mapped blocks are aligned to the system page, which serves every layout
62/// whose alignment is at most a page; a block with a larger alignment uses
63/// the system allocator. Fresh mappings are zeroed by the system, so zeroed
64/// allocation costs nothing beyond the mapping. Resizing a mapped block
65/// keeps it in place when its pages suffice and releases the pages it no
66/// longer needs when it shrinks. To grow, Linux's `mremap` extends the
67/// mapping in place or moves its pages without copying them; elsewhere the
68/// block grows in place where the following address range is free, and
69/// otherwise is copied to a new mapping, briefly holding both. A failed
70/// resize returns null and leaves the block as it was. A failed unmap
71/// aborts the process (see `unmap`). On platforms without `mmap` every
72/// block uses the system allocator.
73#[derive(Clone, Copy, Debug, Default)]
74pub struct ReturningAllocator;
75
76// SAFETY: every block is either the system allocator's, with the caller's
77// layout passed through unchanged, or a private anonymous mapping of at
78// least `layout.size()` bytes aligned to the page and at least to
79// `layout.align()`. Which of the two a block is depends only on its layout,
80// which `dealloc` and `realloc` receive unchanged, so each block returns to
81// the allocator that made it.
82unsafe impl GlobalAlloc for ReturningAllocator {
83    unsafe fn alloc(&self, layout: Layout) -> *mut u8 {
84        if mapped(layout) {
85            map(layout.size())
86        } else {
87            // SAFETY: the caller's contract for `alloc`.
88            unsafe { System.alloc(layout) }
89        }
90    }
91
92    unsafe fn alloc_zeroed(&self, layout: Layout) -> *mut u8 {
93        if mapped(layout) {
94            // Fresh anonymous pages read as zero.
95            map(layout.size())
96        } else {
97            // SAFETY: the caller's contract for `alloc_zeroed`.
98            unsafe { System.alloc_zeroed(layout) }
99        }
100    }
101
102    unsafe fn dealloc(&self, pointer: *mut u8, layout: Layout) {
103        if mapped(layout) {
104            // SAFETY: `alloc` mapped this block for the same layout.
105            unsafe { unmap(pointer, layout.size()) };
106        } else {
107            // SAFETY: the caller's contract for `dealloc`.
108            unsafe { System.dealloc(pointer, layout) };
109        }
110    }
111
112    unsafe fn realloc(&self, pointer: *mut u8, layout: Layout, new_size: usize) -> *mut u8 {
113        // SAFETY: the caller guarantees `new_size`, rounded up to the
114        // alignment, does not overflow `isize`.
115        let resized = unsafe { Layout::from_size_align_unchecked(new_size, layout.align()) };
116        match (mapped(layout), mapped(resized)) {
117            // SAFETY: the caller's contract for `realloc`.
118            (false, false) => unsafe { System.realloc(pointer, layout, new_size) },
119            // SAFETY: `pointer` is a mapping made for `layout.size()` bytes.
120            (true, true) => unsafe { remap(pointer, layout.size(), new_size) },
121            _ => {
122                // SAFETY: `resized` has a nonzero size and the caller's
123                // alignment.
124                let moved = unsafe { self.alloc(resized) };
125                if !moved.is_null() {
126                    // SAFETY: both blocks hold at least the bytes copied and
127                    // are distinct allocations.
128                    unsafe {
129                        std::ptr::copy_nonoverlapping(pointer, moved, layout.size().min(new_size));
130                        self.dealloc(pointer, layout);
131                    }
132                }
133                moved
134            }
135        }
136    }
137}

139/// Whether `layout`'s block is mapped from the system.
140#[cfg(unix)]
141fn mapped(layout: Layout) -> bool {
142    layout.size() >= RETURNED_BLOCK_BYTES && layout.align() <= page_size()
143}
144
145#[cfg(not(unix))]
146fn mapped(_layout: Layout) -> bool {
147    false
148}
149
150/// The system page size, read once.
151#[cfg(unix)]
152fn page_size() -> usize {
153    use std::sync::atomic::{AtomicUsize, Ordering};
154    static PAGE: AtomicUsize = AtomicUsize::new(0);
155    match PAGE.load(Ordering::Relaxed) {
156        0 => {
157            // SAFETY: `sysconf` has no preconditions.
158            let page = usize::try_from(unsafe { libc::sysconf(libc::_SC_PAGESIZE) })
159                .ok()
160                .filter(|page| page.is_power_of_two())
161                .unwrap_or(4096);
162            PAGE.store(page, Ordering::Relaxed);
163            page
164        }
165        page => page,
166    }
167}

169/// `bytes` rounded up to whole pages; `None` past the address space.
170#[cfg(unix)]
171fn pages(bytes: usize) -> Option<usize> {
172    let page = page_size();
173    Some(bytes.checked_add(page - 1)? & !(page - 1))
174}

176/// A fresh private mapping of at least `bytes` bytes, or null.
177#[cfg(unix)]
178fn map(bytes: usize) -> *mut u8 {
179    map_at(std::ptr::null_mut(), bytes)
180}
181
182/// A fresh private mapping of at least `bytes` bytes, at `hint` if that range
183/// is free and elsewhere otherwise; null when the system refuses.
184#[cfg(unix)]
185fn map_at(hint: *mut u8, bytes: usize) -> *mut u8 {
186    let Some(length) = pages(bytes) else {
187        return std::ptr::null_mut();
188    };
189    // SAFETY: an anonymous private mapping without MAP_FIXED replaces
190    // nothing; the hint is advisory.
191    let mapping = unsafe {
192        libc::mmap(
193            hint.cast(),
194            length,
195            libc::PROT_READ | libc::PROT_WRITE,
196            libc::MAP_PRIVATE | libc::MAP_ANON,
197            -1,
198            0,
199        )
200    };
201    if mapping == libc::MAP_FAILED {
202        std::ptr::null_mut()
203    } else {
204        mapping.cast()
205    }
206}

208/// Unmap the mapping of `bytes` bytes at `pointer`, or stop the process.
209///
210/// `munmap` can fail for a valid mapping: on Linux, freeing the middle of
211/// mappings the kernel merged needs one more map entry, which fails with
212/// `ENOMEM` once the process is at `vm.max_map_count`. A global allocator
213/// must not unwind, and a block whose pages stay mapped would hold memory
214/// that nothing charges, so the process aborts.
215///
216/// # Safety
217///
218/// `pointer` is a mapping [`map`] made for `bytes` bytes, or its tail from a
219/// page boundary, and nothing uses it after this call.
220#[cfg(unix)]
221unsafe fn unmap(pointer: *mut u8, bytes: usize) {
222    // `map` mapped these pages, so their length fits the address space.
223    let length = bytes.next_multiple_of(page_size());
224    // SAFETY: the caller's contract; the range is whole pages of one mapping.
225    if unsafe { libc::munmap(pointer.cast(), length) } != 0 {
226        abort_with(b"casa-rs: munmap of a freed block failed; aborting\n");
227    }
228}

230/// Write `message` to standard error without allocating, then abort.
231#[cfg(unix)]
232fn abort_with(message: &[u8]) -> ! {
233    // SAFETY: `message` is a live buffer of `message.len()` bytes; a failed
234    // write changes nothing.
235    unsafe { libc::write(libc::STDERR_FILENO, message.as_ptr().cast(), message.len()) };
236    std::process::abort()
237}

239/// Resize the mapping of `old` bytes at `pointer` to hold `new` bytes; null,
240/// leaving the block unchanged, when the system refuses.
241///
242/// # Safety
243///
244/// `pointer` is a mapping [`map`] made for `old` bytes, and `new` is at
245/// least [`RETURNED_BLOCK_BYTES`].
246#[cfg(unix)]
247unsafe fn remap(pointer: *mut u8, old: usize, new: usize) -> *mut u8 {
248    let (Some(held), Some(needed)) = (pages(old), pages(new)) else {
249        return std::ptr::null_mut();
250    };
251    if needed <= held {
252        if needed < held {
253            // SAFETY: the tail from a page boundary of this mapping.
254            unsafe { unmap(pointer.add(needed), held - needed) };
255        }
256        return pointer;
257    }
258    #[cfg(target_os = "linux")]
259    {
260        // SAFETY: the whole mapping, moved if it cannot grow in place.
261        let moved = unsafe { libc::mremap(pointer.cast(), held, needed, libc::MREMAP_MAYMOVE) };
262        if moved == libc::MAP_FAILED {
263            std::ptr::null_mut()
264        } else {
265            moved.cast()
266        }
267    }
268    #[cfg(not(target_os = "linux"))]
269    {
270        // Grow in place when the pages after the block are free.
271        // SAFETY: `held` bytes past the start of the mapping is its end.
272        let end = unsafe { pointer.add(held) };
273        let extension = map_at(end, needed - held);
274        if extension == end {
275            return pointer;
276        }
277        if !extension.is_null() {
278            // SAFETY: a fresh mapping of `needed - held` bytes, unused.
279            unsafe { unmap(extension, needed - held) };
280        }
281        let moved = map(needed);
282        if !moved.is_null() {
283            // SAFETY: both mappings hold `old` bytes and are distinct.
284            unsafe {
285                std::ptr::copy_nonoverlapping(pointer, moved, old);
286                unmap(pointer, old);
287            }
288        }
289        moved
290    }
291}

308#[cfg(all(test, unix))]
309mod tests {
310    use super::*;
311
312    const LARGE: usize = 3 * RETURNED_BLOCK_BYTES + 5;
313
314    fn layout(size: usize, align: usize) -> Layout {
315        Layout::from_size_align(size, align).expect("a valid layout")
316    }
317
318    /// Fill `bytes` bytes at `pointer` with their offsets modulo 251.
319    unsafe fn fill(pointer: *mut u8, bytes: usize) {
320        for offset in 0..bytes {
321            // SAFETY: the caller's block holds `bytes` bytes.
322            unsafe { pointer.add(offset).write((offset % 251) as u8) };
323        }
324    }
325
326    /// Whether the first `bytes` bytes at `pointer` still hold [`fill`]'s
327    /// pattern.
328    unsafe fn filled(pointer: *const u8, bytes: usize) -> bool {
329        // SAFETY: the caller's block holds `bytes` bytes.
330        (0..bytes).all(|offset| unsafe { pointer.add(offset).read() } == (offset % 251) as u8)
331    }
332
333    #[test]
334    fn large_blocks_are_zeroed_aligned_and_writable() {
335        let allocator = ReturningAllocator;
336        for align in [1, 8, 64, 4096] {
337            let layout = layout(LARGE, align);
338            // SAFETY: a nonzero layout, freed with the same layout.
339            unsafe {
340                let block = allocator.alloc_zeroed(layout);
341                assert!(!block.is_null());
342                assert_eq!(block as usize % align, 0, "aligned to {align}");
343                assert!((0..LARGE).all(|offset| block.add(offset).read() == 0));
344                fill(block, LARGE);
345                assert!(filled(block, LARGE));
346                allocator.dealloc(block, layout);
347            }
348        }
349    }
350
351    #[test]
352    fn resizing_keeps_the_contents_across_the_threshold() {
353        let allocator = ReturningAllocator;
354        let small = 1000;
355        let sizes = [
356            LARGE,
357            LARGE + 7,
358            4 * LARGE,
359            LARGE - 3,
360            RETURNED_BLOCK_BYTES,
361            small,
362            2 * LARGE,
363        ];
364        // SAFETY: each step resizes the live block with its current layout
365        // and the final block is freed with its own.
366        unsafe {
367            let mut current = layout(small, 16);
368            let mut block = allocator.alloc(current);
369            fill(block, small);
370            for size in sizes {
371                let kept = current.size().min(size);
372                block = allocator.realloc(block, current, size);
373                assert!(!block.is_null(), "{} -> {size}", current.size());
374                assert!(filled(block, kept), "{} -> {size}", current.size());
375                current = layout(size, 16);
376                fill(block, size);
377            }
378            allocator.dealloc(block, current);
379        }
380    }
381
382    #[test]
383    fn alignment_beyond_a_page_uses_the_system_allocator() {
384        let allocator = ReturningAllocator;
385        let align = 4 * page_size();
386        let layout = layout(LARGE, align);
387        assert!(!mapped(layout));
388        // SAFETY: a nonzero layout, freed with the same layout.
389        unsafe {
390            let block = allocator.alloc(layout);
391            assert!(!block.is_null());
392            assert_eq!(block as usize % align, 0);
393            allocator.dealloc(block, layout);
394        }
395    }
