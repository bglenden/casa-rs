21/// The frequency `componentlist.addcomponent` gives a component when none
22/// is given, LSRK.
23const ADDCOMPONENT_FREQUENCY_HZ: f64 = 1.42e9;
24
25/// The phase calibrator observed between passes over the map.
26pub(super) struct Calibrator {
27    /// `caldirection`.
28    pub(super) direction: String,
29    /// `<stem>.cal.cclist`: the calibrator as a point of `calflux`.
30    pub(super) list: PathBuf,
31}
32
33/// The calibrator, when `calflux` is positive, `caldirection` is set and
34/// the array is an interferometer; its component list is written.
35pub(super) fn calibrator(run: &mut Run<'_>) -> Result<Option<Calibrator>, SimobserveError> {
36    let parameters = run.parameters;
37    if !qa::is_quantity(&parameters.calflux) {
38        return Err(SimutilError::Quantity(parameters.calflux.clone()).into());
39    }
40    let flux_jy = qa::value_in(&qa::read(&parameters.calflux)?, "Jy")?;
41    let mut direction = parameters.caldirection.clone();
42    if direction.chars().count() < 4 {
43        direction.clear();
44    }
45    if !(flux_jy > 0. && !direction.is_empty() && run.uvmode) {
46        return Ok(None);
47    }
48    let path = run.file(".cal.cclist");
49    remove(&path)?;
50    if !is_direction(&direction) {
51        return Err(SimutilError::Direction(direction).into());
52    }
53    let (refer, [longitude, latitude]) = direction_measure(&direction)?;
54    let mut list = ComponentList::new();
55    list.push(
56        SkyComponent::new(
57            Flux::stokes_jy(flux_jy, 0., 0., 0.),
58            ComponentShape::point(MDirection::from_angles(longitude, latitude, refer)),
59            SpectralModel::new(
60                MFrequency::new(ADDCOMPONENT_FREQUENCY_HZ, FrequencyRef::LSRK),
61                SpectralKind::Constant,
62            ),
63        )
64        .with_label("phase calibrator"),
65    );
66    list.write_table(&path).map_err(SkyModelError::from)?;
67    run.wrote(&path);
68    Ok(Some(Calibrator {
69        direction,
70        list: path,
71    }))
72}

83/// A step of the schedule, as the task calls the session.
84#[derive(Debug, Clone, PartialEq)]
85pub(super) enum Step {
86    /// `observemany` of the map's pointings.
87    Targets(Vec<Target>),
88    /// `observe` of the calibrator.
89    Calibrator {
90        /// The scan's start, s.
91        start_s: f64,
92        /// Its stop, s.
93        stop_s: f64,
94    },
95}

97/// The observation in time: the reference epoch and the steps.
98#[derive(Debug, Clone)]
99pub(super) struct Schedule {
100    /// `refdate` at 00:00 UTC.
101    pub(super) reference: MEpoch,
102    /// The steps, in time order, in seconds from the transit of the first
103    /// field at the reference date.
104    pub(super) steps: Vec<Step>,
105}

107/// The schedule CASA's observing loop makes of `pointings` from
108/// `refdate`, `hourangle` and `totaltime`, with a calibrator when
109/// `calibrator`; but a pass the time ends inside is observed as far as the
110/// time goes, with or without one (see the module documentation of
111/// [`super`]).
112pub(super) fn schedule(
113    run: &Run<'_>,
114    pointings: &Pointings,
115    calibrator: bool,
116) -> Result<Schedule, SimobserveError> {
117    let parameters = run.parameters;
118    let reference = reference_date(&parameters.refdate, &parameters.hourangle)?;
119    let offset_s = hour_angle(&parameters.hourangle)?;
120    tracing::info!(
121        "You desire an hour angle of {} hours",
122        python(offset_s / 3600.)
123    );
124    let calibrator_s = qa::value_in(&qa::read(&pointings.integration)?, "s")?;
125    let times = &pointings.times_s;
126    let fields = pointings.pointings.len();
127    let total_s = total_time(
128        &parameters.totaltime,
129        times,
130        calibrator.then_some(calibrator_s),
131    )?;
132    let all_scans_s = times.iter().fold(0., |sum, time| sum + time);
133    if total_s < all_scans_s {
134        tracing::warn!(
135            "Not all pointings in the mosaic will be observed - check mosaic setup and exposure \
136             time parameters!"
137        );
138    }
139    let pass_s = times[..fields].iter().fold(0., |sum, time| sum + time)
140        + if calibrator { calibrator_s } else { 0. };
141    if total_s > 0. && pass_s <= 0. {
142        return Err(SimobserveError::raised(format!(
143            "a pass over the {fields} pointings takes {}s, so totaltime {} never ends",
144            python(pass_s),
145            parameters.totaltime
146        )));
147    }
148    let steps = steps(
149        &times[..fields],
150        total_s,
151        offset_s,
152        calibrator.then_some(calibrator_s),
153    );
154    Ok(Schedule { reference, steps })
155}

157/// CASA's observing loop: from `offset_s` before the middle of `total_s`,
158/// each pointing for its time in turn, pass after pass, until the time is
159/// used; with a calibrator, its scan of `calibrator_s` before the first
160/// pass and after each. The pointings of a pass the time ends inside are
161/// observed, calibrator or not (CASA drops them with a calibrator).
162fn steps(times_s: &[f64], total_s: f64, offset_s: f64, calibrator_s: Option<f64>) -> Vec<Step> {
163    let mut start_s = -total_s / 2.0 + offset_s;
164    let first_s = start_s;
165    let mut steps = Vec::new();
166    if let Some(calibrator_s) = calibrator_s {
167        steps.push(Step::Calibrator {
168            start_s,
169            stop_s: start_s + calibrator_s,
170        });
171        start_s += calibrator_s;
172    }
173    let mut targets = Vec::new();
174    let mut field = 0;
175    while start_s - first_s < total_s {
176        let stop_s = start_s + times_s[field];
177        targets.push(Target {
178            field,
179            start_s,
180            stop_s,
181        });
182        field += 1;
183        start_s = stop_s;
184        if field == times_s.len() {
185            if let Some(calibrator_s) = calibrator_s {
186                let stop_s = start_s + calibrator_s;
187                steps.push(Step::Targets(std::mem::take(&mut targets)));
188                steps.push(Step::Calibrator { start_s, stop_s });
189                start_s = stop_s;
190            }
191            field = 0;
192        }
193    }
194    if !targets.is_empty() {
195        steps.push(Step::Targets(targets));
196    }
197    steps
198}

200/// `refdate` at 00:00 UTC: its date, `(\d*/\d+/\d+)`, with any time after
201/// it (`[/:\d]*`) discarded in favour of `hourangle`.
202fn reference_date(refdate: &str, hourangle: &str) -> Result<MEpoch, SimobserveError> {
203    let (date, time) = split_date(refdate)
204        .ok_or_else(|| SimobserveError::raised(format!("Invalid reference date {refdate}")))?;
205    if time.chars().count() > 1 {
206        tracing::info!(
207            "Discarding time part of refdate, '{time}', in favor of hourangle parameter = \
208             {hourangle}"
209        );
210    }
211    let days = qa::value_in(&qa::read(&format!("{date}/00:00:00"))?, "d")?;
212    Ok(MEpoch::from_mjd(days, EpochRef::UTC))
213}

215/// The groups Python's `re.match('(\d*/\d+/\d+)([/:\d]*)', refdate)`
216/// captures. Each digit run is greedy, and none can give a digit back.
217fn split_date(text: &str) -> Option<(String, String)> {
218    let chars: Vec<char> = text.chars().collect();
219    let digits = |from: usize| {
220        chars[from..]
221            .iter()
222            .take_while(|c| c.is_ascii_digit())
223            .count()
224    };
225    let mut end = digits(0);
226    for least in [1, 1] {
227        if chars.get(end) != Some(&'/') {
228            return None;
229        }
230        let run = digits(end + 1);
231        if run < least {
232            return None;
233        }
234        end += 1 + run;
235    }
236    let time = chars[end..]
237        .iter()
238        .take_while(|c| c.is_ascii_digit() || **c == '/' || **c == ':')
239        .count();
240    Some((
241        chars[..end].iter().collect(),
242        chars[end..end + time].iter().collect(),
243    ))
244}

246/// `hourangle` in seconds: 0 for `transit`, else as hours when a number,
247/// or as the time it is; CASA's message for anything else.
248fn hour_angle(hourangle: &str) -> Result<f64, SimobserveError> {
249    if hourangle == "transit" {
250        return Ok(0.);
251    }
252    let mut offset_s = None;
253    let as_hours = format!("{hourangle}h");
254    if qa::is_quantity(&as_hours) {
255        let hours = qa::read(&as_hours)?;
256        if qa::conforms(&hours, "s") {
257            offset_s = Some(qa::value_in(&hours, "s")?);
258        }
259    }
260    // casacore converts an angle to a time at 360° a day, and keeps any
261    // other unit's dimensions (a number becomes `s.s-1`, not a time).
262    if qa::is_quantity(hourangle) {
263        let seconds = qa::convert(&qa::read(hourangle)?, "s")?;
264        if qa::conforms(&seconds, "s") {
265            offset_s = Some(seconds.value());
266        }
267    }
268    offset_s.ok_or_else(|| {
269        SimobserveError::raised(format!(
270            "Cannot interpret your hourangle parameter {hourangle} as a time quantity e.g. '5h', \
271             30min'"
272        ))
273    })
274}

276/// `totaltime` in seconds: a time, or with no unit a number of passes over
277/// the map (the calibrator's scan, `calibrator_s`, included in each).
278fn total_time(
279    totaltime: &str,
280    times_s: &[f64],
281    calibrator_s: Option<f64>,
282) -> Result<f64, SimobserveError> {
283    let not_a_time = || {
284        SimobserveError::raised(format!(
285            "totaltime {totaltime} does not appear to represent a time interval (use \
286             's','min','h'; not 'sec','m','hr')"
287        ))
288    };
289    if !qa::is_quantity(totaltime) {
290        return Err(not_a_time());
291    }
292    let total = qa::read(totaltime)?;
293    if total.value() < 0. {
294        return Err(SimobserveError::raised(
295            "Negative totaltime is not allowed.",
296        ));
297    }
298    if total.unit().name().is_empty() {
299        let pass_s = times_s.iter().fold(0., |sum, time| sum + time) + calibrator_s.unwrap_or(0.);
300        let passes: f64 = totaltime.trim().parse().unwrap_or(total.value());
301        let total_s = passes * pass_s;
302        tracing::info!("Total observing time = {}s.", python(total_s));
303        return Ok(total_s);
304    }
305    if !qa::conforms(&total, "s") {
306        return Err(not_a_time());
307    }
308    Ok(qa::value_in(&total, "s")?)
309}

315    #[test]
316    fn reference_dates_split_as_python_splits_them() {
317        let split = |text: &str| split_date(text);
318        let owned = |date: &str, time: &str| Some((date.to_string(), time.to_string()));
319        assert_eq!(split("2014/01/01"), owned("2014/01/01", ""));
320        assert_eq!(
321            split("2014/01/01/12:00:00"),
322            owned("2014/01/01", "/12:00:00")
323        );
324        assert_eq!(split("/1/2x"), owned("/1/2", ""));
325        assert_eq!(split("2014-01-01"), None);
326        assert_eq!(split("2014/01/"), None);
327        let epoch = reference_date("2014/01/01/12:00:00", "transit").expect("a date");
328        assert_eq!(epoch.value().as_mjd(), 56_658.0);
329    }

342    fn fields(step: &Step) -> Vec<usize> {
343        match step {
344            Step::Targets(targets) => targets.iter().map(|target| target.field).collect(),
345            Step::Calibrator { .. } => Vec::new(),
346        }
347    }

350    fn passes_repeat_until_the_time_is_used() {
351        // 3 pointings of 10 s for 50 s, centred on the transit: one
352        // observemany of 1.67 passes, as CASA calls it.
353        let steps = steps(&[10., 10., 10.], 50., 0., None);

363    #[test]
364    fn a_calibrator_brackets_each_pass_and_the_last_pass_is_kept() {
365        // Two whole passes (totaltime "2"): calibrator, pass, calibrator,
366        // pass, calibrator, as CASA observes them.
367        let whole = steps(&[10., 10.], 2. * 30., 0., Some(10.));
368        let kinds: Vec<_> = whole.iter().map(fields).collect();
369        assert_eq!(kinds, [vec![], vec![0, 1], vec![], vec![0, 1], vec![]]);
370        // Time ending inside the second pass: CASA drops its pointing,
371        // casa-rs observes it.
372        let partial = steps(&[10., 10.], 45., 0., Some(10.));
373        let kinds: Vec<_> = partial.iter().map(fields).collect();
374        assert_eq!(kinds, [vec![], vec![0, 1], vec![], vec![0]]);
375        // Less time than a pass: CASA observes the calibrator alone.
376        let short = steps(&[10., 10., 10.], 15., 0., Some(10.));
377        let kinds: Vec<_> = short.iter().map(fields).collect();
378        assert_eq!(kinds, [vec![], vec![0]]);
379    }
