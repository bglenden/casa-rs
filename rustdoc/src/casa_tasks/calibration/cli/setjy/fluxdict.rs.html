24pub(super) fn run(
25    task: &TaskInvocation<'_>,
26    mut opened: OpenedMs,
27    parts: &mut [OpenedMs],
28    selection: MsSelection,
29    selected: &ResolvedMsSelection,
30    scratch: bool,
31    imager_history: &mut Vec<String>,
32) -> Result<CasaValue, TaskFailure> {
33    let image = model_files::task_image(task)?;
34    let dictionary = FluxDictionary::new(task.value("fluxdict"))?;
35    let selected_fields = selected
36        .selected_rows
37        .iter()
38        .map(|row| row.field_id)
39        .collect::<BTreeSet<_>>();
40    let fields = dictionary.fields(&selected_fields, task.argument_text("fluxdict"))?;
41    let selected_windows = selection
42        .spw
43        .as_deref()
44        .map(|spw| {
45            requested_windows(&opened.measurement_set, Some(spw))
46                .map(|windows| windows.into_iter().collect::<BTreeSet<_>>())
47        })
48        .transpose()?;
49    let by_channel = task.flag("scalebychan")?;
50    let mut combined = Vec::new();
51    let mut scratch_ready = !scratch;
52    for (id, record) in fields {
53        let ParameterValue::Table(record) = record else {
54            return Err(invalid(format!("fluxdict field {id} must be a dictionary")));
55        };
56        let index = spectral_index(record, id)?;
57        let windows = dictionary.windows(record, selected_windows.as_ref())?;
58        if windows.is_empty() {
59            append_values(&mut combined, flux_dictionary(Vec::new()));
60            continue;
61        }
62        let (flux, hz) = dictionary.reference(record, &windows)?;
63        let Some(model) = model(task, flux, hz, index)? else {
64            continue;
65        };
66        let mut field_request = selection.clone();
67        if !scratch {
68            add_time_selection(task, &mut field_request)?;
69        }
70        field_request.field = Some(id.to_string());
71        field_request.spw = Some(
72            windows
73                .iter()
74                .map(i32::to_string)
75                .collect::<Vec<_>>()
76                .join(","),
77        );
78        let mut field_rows = opened.resolve(&field_request).map_err(execution)?;
79        if field_rows.selected_rows.is_empty() {
80            append_values(&mut combined, flux_dictionary(Vec::new()));
81            continue;
82        }
83        field_rows.channel_selections.clear();
84        let spws = requested_windows(&opened.measurement_set, field_request.spw.as_deref())?;
85        let super::PreparedModels {
86            models: prepared,
87            fluxes: values,
88            manual_samples,
89            ..
90        } = prepare(
91            &opened.measurement_set,
92            &field_rows,
93            &model,
94            &spws,
95            by_channel,
96            image.as_deref().map(|path| (path, scratch)),
97            imager_history,
98        )?;
99        if !scratch_ready {
100            opened = opened
101                .initialize(VisibilityDataColumn::ModelData)
102                .map_err(execution)?;
103            scratch_ready = true;
104        }
105        publish_models(
106            &mut opened,
107            parts,
108            &field_rows,
109            prepared,
110            &spws,
111            scratch,
112            false,
113        )?;
114        record_manual_spectrum(imager_history, &manual_samples);
115        append_values(&mut combined, values);
116    }
117    opened.release().map_err(execution)?;
118    Ok(CasaValue::Dict { entries: combined })
119}
120
121fn append_values(combined: &mut Vec<casa_contracts::CasaEntry>, values: CasaValue) {
122    let CasaValue::Dict { entries } = values else {
123        unreachable!()
124    };
125    for entry in entries {
126        if let Some(existing) = combined
127            .iter_mut()
128            .find(|existing| existing.key == entry.key)
129        {
130            existing.value = entry.value;
131        } else {
132            combined.push(entry);
133        }
134    }
135}

137struct FluxDictionary<'a> {
138    data: &'a Record,
139    windows: Vec<i32>,
140    frequencies: Vec<f64>,
141}
142
143impl<'a> FluxDictionary<'a> {
144    fn new(value: Option<&'a ParameterValue>) -> Result<Self, TaskFailure> {
145        let Some(ParameterValue::Table(data)) = value else {
146            return Err(invalid("fluxdict must be a dictionary"));
147        };
148        if data.is_empty() {
149            return Err(invalid("fluxdict is empty"));
150        }
151        let missing = ["freq", "spwID"]
152            .into_iter()
153            .filter(|&key| !data.contains_key(key))
154            .collect::<Vec<_>>();
155        if !missing.is_empty() {
156            return Err(invalid(format!(
157                "Input fluxdict is missing keywords:{} ",
158                missing.join(" ")
159            )));
160        }
161        let windows = numbers(&data["spwID"])
162            .ok_or_else(|| invalid("fluxdict spwID must be an array of integers"))?
163            .into_iter()
164            .map(|id| {
165                if id.fract() == 0. && (0. ..=f64::from(i32::MAX)).contains(&id) {
166                    Ok(id as i32)
167                } else {
168                    Err(invalid(
169                        "fluxdict spwID must contain nonnegative window IDs",
170                    ))
171                }
172            })
173            .collect::<Result<_, _>>()?;
174        let frequencies = numbers(&data["freq"])
175            .ok_or_else(|| invalid("fluxdict freq must be an array of frequencies"))?;
176        Ok(Self {
177            data,
178            windows,
179            frequencies,
180        })
181    }
182
183    fn fields(
184        &self,
185        selected: &BTreeSet<i32>,
186        order: Option<&str>,
187    ) -> Result<Vec<(i32, &'a ParameterValue)>, TaskFailure> {
188        let supplied = order.and_then(|text| serde_json::from_str::<Vec<String>>(text).ok());
189        let mut visited = BTreeSet::new();
190        let mut fields = Vec::new();
191        for key in supplied.iter().flatten().chain(self.data.keys()) {
192            let Ok(id) = key.parse::<i32>() else { continue };
193            if !selected.contains(&id) || !visited.insert(key) {
194                continue;
195            }
196            let Some(value) = self.data.get(key) else {
197                continue;
198            };
199            fields.push((id, value));
200        }
201        if fields.is_empty() {
202            return Err(invalid(
203                "No field was found in fluxdict for the given data selection",
204            ));
205        }
206        Ok(fields)
207    }
208
209    fn windows(
210        &self,
211        field: &Record,
212        requested: Option<&BTreeSet<i32>>,
213    ) -> Result<Vec<i32>, TaskFailure> {
214        let mut valid = Vec::new();
215        for &window in &self.windows {
216            if requested.is_some_and(|ids| !ids.contains(&window)) {
217                continue;
218            }
219            let flux = window_flux(field, window)?;
220            if flux[0] > -1. {
221                valid.push(window);
222            }
223        }
224        Ok(valid)
225    }
226
227    fn reference(&self, field: &Record, windows: &[i32]) -> Result<([f64; 4], f64), TaskFailure> {
228        let fitted = field
229            .get("fitFluxd")
230            .and_then(number)
231            .ok_or_else(|| invalid("fluxdict fitFluxd must be numeric"))?;
232        if fitted != 0. {
233            let hz = field
234                .get("fitRefFreq")
235                .and_then(number)
236                .ok_or_else(|| invalid("fluxdict fitRefFreq must be numeric"))?;
237            return Ok(([fitted, 0., 0., 0.], hz));
238        }
239        let first = *windows
240            .first()
241            .ok_or_else(|| invalid("list index out of range"))?;
242        let hz = *self
243            .frequencies
244            .get(first as usize)
245            .ok_or_else(|| invalid(format!("fluxdict freq has no entry for window {first}")))?;
246        Ok((window_flux(field, first)?, hz))
247    }
248}
249
250fn spectral_index(field: &Record, id: i32) -> Result<Vec<f64>, TaskFailure> {
251    let coefficients = field
252        .get("spidx")
253        .and_then(numbers)
254        .ok_or_else(|| invalid("fluxdict spidx must be an array of numbers"))?;
255    if coefficients.is_empty() {
256        return Err(invalid("fluxdict spidx must include its intercept"));
257    }
258    let alpha = coefficients.get(1).copied().unwrap_or_default();
259    if alpha.abs() > 10. {
260        let relation = if alpha < -10. {
261            "less than -10"
262        } else {
263            "greater than 10"
264        };
265        return Err(invalid(format!(
266            "Field {id}: the spectral index is {relation}!"
267        )));
268    }
269    Ok(coefficients.into_iter().skip(1).collect())
270}
271
272fn window_flux(field: &Record, window: i32) -> Result<[f64; 4], TaskFailure> {
273    let Some(ParameterValue::Table(record)) = field.get(&window.to_string()) else {
274        return Err(invalid(format!(
275            "fluxdict has no record for window {window}"
276        )));
277    };
278    let values = record
279        .get("fluxd")
280        .and_then(numbers)
281        .filter(|values| !values.is_empty() && values.len() <= 4)
282        .ok_or_else(|| {
283            invalid(format!(
284                "fluxdict window {window} must supply one to four Stokes flux values"
285            ))
286        })?;
287    let mut flux = [0.; 4];
288    flux[..values.len()].copy_from_slice(&values);
289    Ok(flux)
290}

292fn model(
293    task: &TaskInvocation<'_>,
294    flux: [f64; 4],
295    hz: f64,
296    index: Vec<f64>,
297) -> Result<Option<FluxParameters>, TaskFailure> {
298    if flux[0] < 0. {
299        let catalog =
300            FluxCatalog::open(FluxScale::Fixed(FixedScale::PerleyButler2010)).map_err(execution)?;
301        return Ok(Some(FluxParameters::Standard {
302            catalog,
303            interpolation: Curve::Nearest,
304        }));
305    }
306    if !hz.is_finite() || hz <= 0. {
307        return Ok(None);
308    }
309    let model = ManualFlux::new(
310        flux,
311        MFrequency::new(hz, FrequencyRef::LSRK),
312        index,
313        task.numbers("polindex")?,
314        task.numbers("polangle")?,
315        task.number("rotmeas")?,
316    )
317    .map_err(execution)?;
318    Ok(Some(FluxParameters::Manual(model)))
319}

321fn numbers(value: &ParameterValue) -> Option<Vec<f64>> {
322    match value {
323        ParameterValue::Array(values) => values.iter().map(number).collect(),
324        ParameterValue::EmptyArray { .. } => Some(Vec::new()),
325        _ => None,
326    }
327}
328
329fn number(value: &ParameterValue) -> Option<f64> {
330    match value {
331        ParameterValue::Float(value) => Some(*value),
332        ParameterValue::Integer(value) => Some(*value as f64),
333        ParameterValue::NumpyScalar { value, .. } | ParameterValue::ScalarArray(value) => {
334            number(value)
335        }
336        _ => None,
337    }
338}

340fn invalid(message: impl Into<String>) -> TaskFailure {
341    TaskFailure::execution(message).at("fluxdict")
