13/// Clark's major cycles per step (`setMaxNumberMajorCycles(10)`).
14const MAJOR_CYCLES: usize = 10;
15
16/// Clark's point CLEAN on one plane.
17///
18/// Each Clark cycle builds the active list of supported pixels above
19/// `max(threshold, peak × exterior sidelobe × factor)`, cleans it with
20/// subtraction inside the PSF patch only, and then refreshes the whole
21/// residual exactly ([`LinearRefresh`]). A step ends after ten Clark cycles
22/// or when the peak reaches the threshold; its peak residual is the active
23/// list's last maximum, as CASA reports it.
24#[derive(Default)]
25pub struct ClarkPolarized<const N: usize> {
26    refresh: Cell<Option<LinearRefresh>>,
27}
28
29/// The single-polarization Clark solver.
30pub type Clark = ClarkPolarized<1>;
31
32impl<const N: usize> ClarkPolarized<N> {
33    /// Clark that reuses `refresh` when it serves this plane's PSF.
34    #[must_use]
35    pub fn new(refresh: Option<LinearRefresh>) -> Self {
36        Self {
37            refresh: Cell::new(refresh),
38        }
39    }
40
41    /// The refresh, to reuse in later cycles of the same PSF.
42    #[must_use]
43    pub fn into_refresh(self) -> Option<LinearRefresh> {
44        self.refresh.into_inner()
45    }

48/// An active pixel: its `[x, y]`, kept rather than its storage index because
49/// every component's subtraction needs each pixel's offset from it.
50struct Active<const N: usize> {
51    pixel: [u32; 2],
52    value: [f64; N],
53}
54
55/// CASA's `ABSMAXF`, `ABSMAX2F` and `ABSMAX4F` ranking.
56fn magnitude<const N: usize>(value: &[f64; N]) -> f64 {
57    match N {
58        1 => value[0].abs(),
59        2 => (value[0] + value[1]).abs().max((value[0] - value[1]).abs()),
60        4 => (value[0] + (value[1] * value[1] + value[2] * value[2] + value[3] * value[3]).sqrt())
61            .abs(),
62        _ => unreachable!("CASA Clark supports one, two or four polarizations"),
63    }
64}
65
66/// Active pixels per block of [`ClarkState`]'s block maxima.
67const BLOCK: usize = 64;
68
69/// The largest magnitude of each [`BLOCK`] of `active`, NaN ignored.
70fn block_maxima<const N: usize>(active: &[Active<N>], blocks: &mut Vec<f64>) {
71    blocks.clear();
72    blocks.extend(active.chunks(BLOCK).map(largest_magnitude));
73}
74
75/// Recompute the blocks of `blocks` that hold any of `active[range]`.
76fn refresh_blocks<const N: usize>(
77    active: &[Active<N>],
78    blocks: &mut [f64],
79    range: std::ops::Range<usize>,
80) {
81    if range.is_empty() {
82        return;
83    }
84    for block in range.start / BLOCK..=(range.end - 1) / BLOCK {
85        let end = ((block + 1) * BLOCK).min(active.len());
86        blocks[block] = largest_magnitude(&active[block * BLOCK..end]);
87    }
88}
89
90fn largest_magnitude<const N: usize>(pixels: &[Active<N>]) -> f64 {
91    pixels.iter().fold(0.0_f64, |largest, pixel| {
92        largest.max(magnitude(&pixel.value))
93    })
94}
95
96/// ABSMAXF: the first pixel of largest magnitude in list order, as a scan
97/// that keeps the first pixel unless a later one is strictly larger finds
98/// it (so a NaN first pixel is the peak, and a later NaN never is). It runs
99/// once per component, so it reads the blocks' maxima (`blocks`, kept by
100/// [`block_maxima`] and [`refresh_blocks`]) and then only the first block
101/// that holds the largest.
102fn first_largest<'a, const N: usize>(
103    active: &'a [Active<N>],
104    blocks: &[f64],
105) -> Option<&'a Active<N>> {
106    let first = active.first()?;
107    if magnitude(&first.value).is_nan() {
108        return Some(first);
109    }
110    let largest = blocks
111        .iter()
112        .fold(0.0_f64, |largest, &block| largest.max(block));
113    let block = blocks
114        .iter()
115        .position(|&block| block == largest)
116        .unwrap_or(0);
117    active[block * BLOCK..]
118        .iter()
119        .take(BLOCK)
120        .find(|pixel| magnitude(&pixel.value) == largest)
121        .or(Some(first))
122}

124/// Working state of one Clark step.
125pub struct ClarkState<const N: usize = 1> {
126    shape: PlaneShape,
127    psf_peak: [usize; 2],
128    patch: ClarkPatch,
129    threshold: f64,
130    factor: f64,
131    max_residual: f64,
132    flux_limit: f64,
133    iteration_flux_limit: f64,
134    fac: f64,
135    fmn: f64,
136    cycle_iterations: usize,
137    maximum_cycle_iterations: usize,
138    cycles: usize,
139    step_components: usize,
140    active: Vec<Active<N>>,
141    /// Values of the component selected by `next`.
142    selected: [f64; N],
143    /// The largest magnitude of each [`BLOCK`] of `active`.
144    blocks: Vec<f64>,
145    refresh: Vec<LinearRefresh>,
146}

148impl<const N: usize> Solver for ClarkPolarized<N> {
149    type State = ClarkState<N>;
150
151    fn runs_with_empty_support(&self) -> bool {
152        N > 1
153    }
154
155    fn initialize(
156        &self,
157        view: &MinorCycleView<'_>,
158        residual: &[Vec<f64>],
159        threshold: f64,
160    ) -> Result<ClarkState<N>, Error> {
161        let shape = view.shape;
162        // `LatConvEquation` takes the PSF at the plane's centre, whatever its
163        // peak (`ClarkCleanLatModel::solve`).
164        let [x, y] = shape.centre();
165        let peak = shape.index(x, y);
166        assert_eq!(residual.len(), N);
167        assert!(matches!(N, 1 | 2 | 4));
168        let first_refresh = match self.refresh.take() {
169            Some(refresh) if refresh.serves(shape, peak) => refresh,
170            _ => LinearRefresh::new(&view.psf[0], shape, peak)?,
171        };
172        let mut refresh = Vec::with_capacity(N);
173        refresh.push(first_refresh);
174        for _ in 1..N {
175            refresh.push(LinearRefresh::new(&view.psf[0], shape, peak)?);
176        }
177        let patch = view.summary.clark();
178        // A PSF whose sidelobes outside the patch are large gets short Clark
179        // cycles (ClarkCleanLatModel::solve).
180        let maximum_cycle_iterations = if patch.exterior_sidelobe > 0.5 {
181            5
182        } else if patch.exterior_sidelobe > 0.35 {
183            50
184        } else {
185            usize::MAX
186        };
187        let max_residual = (0..shape.len())
188            .filter(|&index| view.support.as_slice()[index])
189            .map(|index| magnitude(&std::array::from_fn::<_, N, _>(|p| residual[p][index])))
190            .fold(0.0_f64, f64::max);
191        let mut state = ClarkState {
192            shape,
193            psf_peak: shape.pixel(peak),
194            patch,
195            threshold,
196            factor: 1.0 / 3.0,
197            max_residual,
198            flux_limit: 0.0,
199            iteration_flux_limit: 0.0,
200            fac: 0.0,
201            fmn: 0.0,
202            cycle_iterations: 0,
203            maximum_cycle_iterations,
204            cycles: 0,
205            step_components: 0,
206            active: Vec::new(),
207            selected: [0.0; N],
208            blocks: Vec::new(),
209            refresh,
210        };
211        state.begin(residual, view);
212        Ok(state)
213    }
214
215    fn next(
216        &self,
217        state: &mut ClarkState<N>,
218        view: &MinorCycleView<'_>,
219        residual: &mut [Vec<f64>],
220    ) -> Result<Next, Error> {
221        loop {
222            let peak = first_largest(&state.active, &state.blocks);
223            debug_assert_eq!(
224                peak.map(|pixel| pixel.pixel),
225                {
226                    let mut fresh = Vec::new();
227                    block_maxima(&state.active, &mut fresh);
228                    first_largest(&state.active, &fresh).map(|pixel| pixel.pixel)
229                },
230                "the block maxima are current"
231            );
232            if state.cycle_iterations < state.maximum_cycle_iterations
233                && let Some(pixel) = peak
234                && magnitude(&pixel.value) > state.iteration_flux_limit
235            {
236                if pixel.value.iter().any(|value| !value.is_finite()) {
237                    return Err(Error::NonFinite);
238                }
239                let index = state
240                    .shape
241                    .index(pixel.pixel[0] as usize, pixel.pixel[1] as usize);
242                state.selected = pixel.value;
243                return Ok(Next::Clean(Candidate::Pixel {
244                    index,
245                    scale: 0,
246                    strength: state.selected[0],
247                }));
248            }
249            if state.cycle_iterations == 0 {
250                return Ok(Next::Stop);
251            }
252            state.refresh_pending(residual, view)?;
253            if state.max_residual <= state.threshold || state.cycles >= MAJOR_CYCLES {
254                return Ok(Next::Stop);
255            }
256            state.begin(residual, view);
257        }
258    }
259
260    fn component_fluxes(
261        &self,
262        state: &Self::State,
263        _: Candidate,
264        gain: f64,
265        fluxes: &mut Vec<f64>,
266    ) {
267        fluxes.extend(state.selected.iter().map(|value| gain * value));
268    }
269
270    fn reported_flux(&self, fluxes: &[f64]) -> f64 {
271        fluxes.iter().sum()
272    }
273
274    fn accept(
275        &self,
276        state: &mut ClarkState<N>,
277        view: &MinorCycleView<'_>,
278        _: &mut [Vec<f64>],
279        candidate: Candidate,
280        gain: f64,
281        delta: &mut Delta,
282    ) -> Result<(), Error> {
283        let Candidate::Pixel { index, .. } = candidate;
284        let fluxes = state.selected.map(|value| gain * value);
285        for (p, &flux) in fluxes.iter().enumerate() {
286            delta.add(p, index, flux);
287            state.refresh[p].add(index, flux);
288        }
289        let shape = state.shape;
290        let [px, py] = shape.pixel(index);
291        let patch = state.patch;
292        let psf = &view.psf[0];
293        // The list runs x outer, so the pixels whose x lies in the patch
294        // are one run of it; only they can change.
295        debug_assert!(
296            state
297                .active
298                .windows(2)
299                .all(|pair| pair[0].pixel[0] <= pair[1].pixel[0]),
300            "the active list runs x outer (`begin`'s order)"
301        );
302        let first_x = px as isize - patch.radius[0] as isize;
303        let end_x = px as isize + (patch.size[0] - patch.radius[0]) as isize;
304        let start = state
305            .active
306            .partition_point(|pixel| (pixel.pixel[0] as isize) < first_x);
307        let end = state
308            .active
309            .partition_point(|pixel| (pixel.pixel[0] as isize) < end_x);
310        for pixel in &mut state.active[start..end] {
311            let [tx, ty] = pixel.pixel;
312            let relative = [tx as isize - px as isize, ty as isize - py as isize];
313            if relative[0] < -(patch.radius[0] as isize)
314                || relative[0] >= (patch.size[0] - patch.radius[0]) as isize
315                || relative[1] < -(patch.radius[1] as isize)
316                || relative[1] >= (patch.size[1] - patch.radius[1]) as isize
317            {
318                continue;
319            }
320            let source = [
321                state.psf_peak[0] as isize + relative[0],
322                state.psf_peak[1] as isize + relative[1],
323            ];
324            if source[0] < 0
325                || source[1] < 0
326                || source[0] >= shape.nx as isize
327                || source[1] >= shape.ny as isize
328            {
329                continue;
330            }
331            let response = psf[shape.index(source[0] as usize, source[1] as usize)];
332            for (value, flux) in pixel.value.iter_mut().zip(fluxes) {
333                *value -= flux * response;
334            }
335        }
336        refresh_blocks(&state.active, &mut state.blocks, start..end);
337        state.cycle_iterations += 1;
338        state.step_components += 1;
339        state.fmn += state.fac / state.step_components as f64;
340        state.iteration_flux_limit = if state.flux_limit > 0.0 {
341            (state.flux_limit * state.fmn).max(state.threshold)
342        } else {
343            // Without an exterior sidelobe the flux limit is 0, CASA's
344            // `Fac = absRes / fluxLimit` (speedup −1) is infinite and its
345            // iteration limit `max(0 · ∞, threshold)` is NaN, so the Clark
346            // cycle ends after one component.
347            f64::INFINITY
348        };
349        Ok(())
350    }
351
352    fn finalize(
353        &self,
354        mut state: ClarkState<N>,
355        view: &MinorCycleView<'_>,
356        residual: &mut [Vec<f64>],
357        _: &Delta,
358    ) -> Result<StepEnd, Error> {
359        if state.cycle_iterations > 0 {
360            state.refresh_pending(residual, view)?;
361        }
362        let end = StepEnd {
363            peak: state.max_residual,
364            refreshes: state.cycles,
365        };
366        self.refresh.set(state.refresh.into_iter().next());
367        Ok(end)
368    }
369
370    /// The refresh ([`LinearRefresh::bytes`]), the active list, which
371    /// grows to at most every pixel (an upper bound, since the list holds
372    /// only the supported pixels above the cycle's flux limit), and its
373    /// block maxima.
374    fn working_bytes(&self, shape: PlaneShape, _terms: usize) -> u64 {
375        let pixels = shape.len().next_power_of_two();
376        let active = pixels * size_of::<Active<N>>();
377        let blocks = pixels.div_ceil(BLOCK).next_power_of_two() * size_of::<f64>();
378        N as u64 * LinearRefresh::bytes(shape)
379            + (active + blocks + N * size_of::<LinearRefresh>()) as u64
380    }
381
382    /// A point component.
383    fn component_cells(&self, _shape: PlaneShape) -> usize {
384        1
385    }
386}

388impl<const N: usize> ClarkState<N> {
389    /// Start a Clark cycle: the flux limit from the exterior sidelobe and
390    /// the active list of supported pixels above it.
391    fn begin(&mut self, residual: &[Vec<f64>], view: &MinorCycleView<'_>) {
392        self.flux_limit = self.max_residual * self.patch.exterior_sidelobe * self.factor;
393        if self.factor > 1.0 {
394            self.flux_limit = self.flux_limit.min(0.95 * self.max_residual);
395        }
396        let cutoff = self.flux_limit.max(self.threshold);
397        // GETBIMF: supported pixels at or above the cutoff, x outer and y
398        // inner, which is storage order. (CASA lists a tiled residual tile
399        // by tile; a plane in one tile is listed in this order.)
400        self.active.clear();
401        for (index, &supported) in view.support.as_slice().iter().enumerate() {
402            let value = std::array::from_fn(|p| residual[p][index]);
403            if supported && magnitude(&value) >= cutoff {
404                let [x, y] = self.shape.pixel(index);
405                self.active.push(Active {
406                    pixel: [x as u32, y as u32],
407                    value,
408                });
409            }
410        }
411        block_maxima(&self.active, &mut self.blocks);
412        let peak = self
413            .blocks
414            .iter()
415            .fold(0.0_f64, |best, &block| best.max(block));
416        // `pow(fluxLimit / absRes, speedup)` with CASA's speedup of −1.
417        self.fac = if self.flux_limit > 0.0 {
418            peak / self.flux_limit
419        } else {
420            0.0
421        };
422        self.fmn = 0.0;
423        self.iteration_flux_limit = cutoff;
424        self.cycle_iterations = 0;
425    }
426
427    /// End a Clark cycle: refresh the residual exactly and widen the next
428    /// cycle's flux limit when the peak grew.
429    fn refresh_pending(
430        &mut self,
431        residual: &mut [Vec<f64>],
432        view: &MinorCycleView<'_>,
433    ) -> Result<(), Error> {
434        let previous = self.max_residual;
435        self.max_residual = self
436            .active
437            .iter()
438            .fold(0.0_f64, |peak, pixel| peak.max(magnitude(&pixel.value)));
439        for (refresh, plane) in self.refresh.iter_mut().zip(residual) {
440            refresh.refresh(plane, &view.psf[0])?;
441        }
442        self.cycles += 1;
443        self.cycle_iterations = 0;
444        if self.max_residual > previous {
445            self.factor *= 3.0;
446            self.maximum_cycle_iterations = 10;
447        }
448        Ok(())
449    }
450}

456    /// The scan `first_largest` replaces: keep the first pixel unless a
457    /// later one is strictly larger.
458    fn scan(active: &[Active<1>]) -> Option<&Active<1>> {
459        active.iter().fold(None::<&Active<1>>, |best, pixel| {
460            if best.is_none_or(|current| pixel.value[0].abs() > current.value[0].abs()) {
461                Some(pixel)
462            } else {
463                best
464            }
465        })
466    }

468    /// `first_largest` over block maxima finds the scan's pixel, on lists
469    /// spanning several blocks with ties, zeros, infinities and NaN, and
470    /// still does after a run of the list changes and only that run's blocks
471    /// are refreshed, as `accept` does.
472    #[test]
473    fn the_first_largest_pixel_is_the_scans() {
474        let mut state = 0x2545_f491_4f6c_dd1d_u64;
475        let mut next = move || {
476            state ^= state << 13;
477            state ^= state >> 7;
478            state ^= state << 17;
479            state
480        };
481        // Few distinct magnitudes, so ties are common, and the edge values.
482        let specials = [
483            0.0,
484            -0.0,
485            f64::INFINITY,
486            f64::NEG_INFINITY,
487            f64::NAN,
488            2.0,
489            -2.0,
490            1.5,
491        ];
492        let value = |draw: u64| specials[(draw % specials.len() as u64) as usize];
493        let mut blocks = Vec::new();
494        let mut multi_block_runs = 0;
495        for length in (0..40).chain([BLOCK - 1, BLOCK, BLOCK + 1, 3 * BLOCK + 7]) {
496            for _ in 0..200 {
497                let mut active = (0..length)
498                    .map(|index| Active {
499                        pixel: [index as u32, 0],
500                        value: [value(next())],
501                    })
502                    .collect::<Vec<_>>();
503                block_maxima(&active, &mut blocks);
504                let found = first_largest(&active, &blocks).map(|pixel| pixel.pixel);
505                assert_eq!(found, scan(&active).map(|pixel| pixel.pixel));
506                if length == 0 {
507                    continue;
508                }
509                // A run of 1 to 2·BLOCK pixels from anywhere, so runs cross
510                // block edges.
511                let start = (next() % length as u64) as usize;
512                let end = (start + 1 + (next() % (2 * BLOCK) as u64) as usize).min(length);
513                if start / BLOCK != (end - 1) / BLOCK {
514                    multi_block_runs += 1;
515                }
516                for pixel in &mut active[start..end] {
517                    pixel.value = [value(next())];
518                }
519                refresh_blocks(&active, &mut blocks, start..end);
520                let found = first_largest(&active, &blocks).map(|pixel| pixel.pixel);
521                assert_eq!(found, scan(&active).map(|pixel| pixel.pixel));
522            }
523        }
524        assert!(
525            multi_block_runs > 100,
526            "only {multi_block_runs} refreshed runs crossed a block edge"
527        );
528    }
