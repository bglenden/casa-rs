22/// Snapshot of an input image's filesystem metadata and table modification generation.
23#[derive(Clone)]
24pub struct InputRevision {
25    /// Filesystem metadata captured when reading the input.
26    pub file: fs::Metadata,
27    /// Table modify counter for paged images, when available.
28    pub generation: Option<u32>,
29}
30
31impl InputRevision {
32    /// Reads the current filesystem snapshot and optional modify counter for `image` at `path`.
33    pub fn read(path: &Path, image: &ImageHandle) -> ImageToolResult<Self> {
34        let file = fs::metadata(path)?;
35        #[cfg(unix)]
36        let generation = with_pixels!(image, storage => match storage {
37            Storage::Paged(image) if !image.uses_advisory_locks() => {
38                Some(image.table().locked_modify_counter()?)
39            }
40            Storage::Paged(_) | Storage::Temp { .. } => None,
41        });
42        #[cfg(not(unix))]
43        let generation = {
44            let _ = image;
45            None
46        };
47        Ok(Self { file, generation })
48    }
49
50    /// Returns whether two snapshots represent the same unmodified input image revision.
51    pub fn matches(&self, other: &Self) -> bool {
52        self.generation == other.generation && same_file_snapshot(&self.file, &other.file)
53    }
54}

56/// Validated metadata and optional open handle for one concatenation input image.
57pub struct ConcatInputEntry<'a> {
58    /// User-supplied path to the input image.
59    pub path: &'a Path,
60    /// Resolved physical/canonical path to the input image.
61    pub resolved_path: PathBuf,
62    /// Pixel shape of the input image.
63    pub shape: Vec<usize>,
64    /// Pixel type of the input image.
65    pub kind: ImagePixelType,
66    /// Parsed coordinate system.
67    pub coordinates: CoordinateSystem,
68    /// Raw coordinate record.
69    pub coordinate_record: RecordValue,
70    /// Parsed image info (beam set and image type).
71    pub info: ImageInfo,
72    /// Raw `imageinfo` record if present.
73    pub info_record: Option<RecordValue>,
74    /// Brightness unit string.
75    pub units: String,
76    /// `miscinfo` record.
77    pub misc: RecordValue,
78    /// History entries copied from this input.
79    pub history: Vec<ImageHistoryRow>,
80    /// Whether the input carries an active pixel mask or FITS blank mask.
81    pub has_mask: bool,
82    /// Whether the input pixels are writable.
83    pub writable: bool,
84    /// Whether the active pixel mask is writable.
85    pub writable_mask: bool,
86    /// Whether the input uses advisory locking (MIRIAD / tempclose).
87    pub advisory_lock: bool,
88    /// Retained handle when not using `tempclose` or when expression operands are held.
89    pub opened: Option<ImageHandle>,
90    /// Bound saved-expression operands when this input is an `.imgexpr`.
91    pub expression: Option<SavedExpressionInputs>,
92    /// All expression operand paths referenced by this input.
93    pub expression_operands: Vec<PathBuf>,
94    /// Nested concatenation operand paths referenced by this input.
95    pub nested_operands: Vec<PathBuf>,
96    /// Revision snapshot captured for `tempclose` validation.
97    pub revision: Option<InputRevision>,
98}
99
100impl<'a> ConcatInputEntry<'a> {
101    /// Opens an input image at `path` for concatenation.
102    pub fn open(path: &'a Path, tempclose: bool) -> ImageToolResult<Self> {
103        Self::open_resolved(path, path, tempclose, &mut HashSet::new(), None)
104    }
105
106    /// Opens an input image at `resolved` (handling `.imgexpr` and staged substitution).
107    pub fn open_resolved(
108        path: &'a Path,
109        resolved: &Path,
110        tempclose: bool,
111        active: &mut HashSet<PathBuf>,
112        output: Option<&StagedOutput<'_>>,
113    ) -> ImageToolResult<Self> {
114        if !casa_images::is_image_concat(resolved)
115            && casa_images::expr_file::is_image_expr(resolved)
116        {
117            let (image, inputs) =
118                saved_expression::open_for_concatenation(resolved, active, output)?;
119            return Self::from_image(path, image, tempclose, Some(inputs));
120        }
121        let image = ImageHandle::open_for_concatenation(resolved)?;
122        Self::from_image(path, image, tempclose, None)
123    }
124
125    /// Wraps an already-opened [`ImageHandle`] into a [`ConcatInputEntry`].
126    pub fn from_image(
127        path: &'a Path,
128        image: ImageHandle,
129        tempclose: bool,
130        mut expression: Option<SavedExpressionInputs>,
131    ) -> ImageToolResult<Self> {
132        let resolved_path = image.path().unwrap_or(path).to_path_buf();
133        let coordinate_record = image.coordinate_record();
134        let coordinates = CoordinateSystem::from_record(&coordinate_record)?;
135        let (units, misc, has_mask) = with_pixels!(&image, storage => with_storage!(storage, image => (
136            image.units().to_string(),
137            image.misc_info(),
138            image.has_pixel_mask(),
139        )));
140        let history = with_pixels!(&image, storage => match storage {
141            Storage::Paged(_) => image_history_rows(path)?,
142            Storage::Temp { image, .. } => image.history_rows()?.into_owned(),
143        });
144        let revision = tempclose
145            .then(|| InputRevision::read(&resolved_path, &image))
146            .transpose()?;
147        let writable = with_pixels!(&image, storage => match storage {
148            Storage::Paged(image) => {
149                image.is_writable()
150                    && casa_tables::publication::check_write_access(&resolved_path.join("table.dat"))
151                        .is_ok()
152            }
153            Storage::Temp { image, fits, .. } => fits.is_none() && image.is_writable(),
154        });
155        let writable_mask = with_pixels!(&image, storage => match storage {
156            Storage::Paged(image) => image
157                .active_pixel_mask_name()
158                .is_some_and(|name| image.named_mask_is_writable(&name).unwrap_or(false)),
159            Storage::Temp { image, .. } => image
160                .concat_source()
161                .is_some_and(|source| source.mask_writable()),
162        }) && writable;
163        let advisory_lock = with_pixels!(&image, storage => match storage {
164            Storage::Paged(_) => false,
165            Storage::Temp { image, .. } => {
166                image.image_type_name() == "MIRIADImage"
167                    || image
168                        .concat_source()
169                        .is_some_and(|source| source.has_advisory_lock())
170            }
171        });
172        let expression_operands = match &mut expression {
173            Some(expression) => std::mem::take(&mut expression.paths),
174            None => image.concat_expression_operands(),
175        };
176        let mut nested_operands = image.concat_operands();
177        if path != resolved_path {
178            nested_operands.push(lexical_absolute(path));
179        }
180        Ok(Self {
181            path,
182            resolved_path,
183            shape: image.shape().to_vec(),
184            kind: image.kind(),
185            coordinates,
186            coordinate_record,
187            info: image_info(&image)?,
188            info_record: image_info_record(&image).cloned(),
189            units,
190            misc,
191            history,
192            has_mask: has_mask || image.fits().is_some_and(|fits| fits.has_mask),
193            writable,
194            writable_mask,
195            advisory_lock,
196            opened: (!tempclose || !expression_operands.is_empty()).then_some(image),
197            expression,
198            expression_operands,
199            nested_operands,
200            revision,
201        })
202    }
203
204    /// Returns a borrowed [`ConcatInput`] descriptor for geometry planning.
205    pub fn geometry(&self) -> ConcatInput<'_> {
206        ConcatInput {
207            path: self.path,
208            shape: &self.shape,
209            coordinates: &self.coordinates,
210            image_info: &self.info,
211            brightness_unit: &self.units,
212        }
213    }
214}

216/// Checks that overwriting `output` cannot clobber any expression operand referenced by `inputs`.
217pub fn protect_expression_sources(
218    output: &Path,
219    inputs: &[ConcatInputEntry<'_>],
220) -> ImageToolResult<()> {
221    saved_expression::check_output_dependencies(
222        output,
223        inputs.iter().flat_map(|input| &input.expression_operands),
224        true,
225    )
226}

228/// Updates `first`'s coordinate record when concatenation converted a coordinate type.
229pub fn coordinate_record(
230    first: &ConcatInputEntry<'_>,
231    plan: &ConcatenationPlan,
232) -> ImageToolResult<RecordValue> {
233    let mut record = first.coordinate_record.clone();
234    if let Some(index) = plan.changed_coordinate {
235        let coordinates = CoordinateSystem::record_coordinates(&record)?;
236        let old = &coordinates[index].field;
237        let old_kind = old.trim_end_matches(|character: char| character.is_ascii_digit());
238        let replacement = plan.coordinates.to_record();
239        let coordinates = CoordinateSystem::record_coordinates(&replacement)?;
240        let new = &coordinates[index].field;
241        let new_kind = new.trim_end_matches(|character: char| character.is_ascii_digit());
242        let value = replacement
243            .get(new)
244            .expect("replacement coordinate")
245            .clone();
246        record.remove(old);
247        record.upsert(format!("{new_kind}{}", &old[old_kind.len()..]), value);
248    }
249    Ok(record)
250}

252/// Merges `source`'s `miscinfo` fields into `output`, disambiguating collisions with `_N`.
253pub fn merge_misc(output: &mut RecordValue, source: &RecordValue) {
254    for field in source.fields() {
255        let mut name = field.name.clone();
256        let mut suffix = 1;
257        while output.get(&name).is_some() {
258            name = format!("{}_{suffix}", field.name);
259            suffix += 1;
260        }
261        output.push(RecordField::new(name, field.value.clone()));
262    }
263}

265/// Builds the combined `imageinfo` record (global or per-plane beams) across `sources`.
266pub fn beam_records(
267    plan: &ConcatenationPlan,
268    sources: &[ConcatInputEntry<'_>],
269) -> ImageToolResult<RecordValue> {
270    let first = &sources[plan.order[0]];
271    let mut record = first
272        .info_record
273        .clone()
274        .unwrap_or_else(|| first.info.to_record());
275    let beams = &plan.image_info.beam_set;
276    if beams == &first.info.beam_set {
277        return Ok(record);
278    }
279    record.remove("restoringbeam");
280    record.remove("perplanebeams");
281    if beams.is_empty() {
282        return Ok(record);
283    }
284    let (spectral, stokes) = plane_axes(&first.coordinate_record)?;
285    let channel_concat = spectral == Some(plan.axis) && beams.n_channels() == plan.shape[plan.axis];
286    let stokes_concat = stokes == Some(plan.axis) && beams.n_stokes() == plan.shape[plan.axis];
287    let origins = if channel_concat || stokes_concat {
288        plan.order
289            .iter()
290            .flat_map(|&index| {
291                (0..sources[index].shape[plan.axis]).map(move |local| (index, local))
292            })
293            .collect::<Vec<_>>()
294    } else {
295        Vec::new()
296    };
297    let equivalent = plan
298        .order
299        .iter()
300        .map(|&index| &sources[index])
301        .find(|input| &input.info.beam_set == beams);
302    let beam = |channel, plane| {
303        let (input, source_channel, source_plane) = if channel_concat {
304            let (index, local) = origins[channel];
305            (Some(&sources[plan.beam_sources[index]]), local, plane)
306        } else if stokes_concat {
307            let (index, local) = origins[plane];
308            (Some(&sources[plan.beam_sources[index]]), channel, local)
309        } else {
310            (equivalent, channel, plane)
311        };
312        Value::Record(input.map_or_else(
313            || beams.beam(channel, plane).to_record(),
314            |input| {
315                let source = &input.info.beam_set;
316                let channel = if source.n_channels() == 1 {
317                    0
318                } else {
319                    source_channel
320                };
321                let plane = if source.n_stokes() == 1 {
322                    0
323                } else {
324                    source_plane
325                };
326                stored_beam_record(input.info_record.as_ref(), source, channel, plane)
327            },
328        ))
329    };
330    if beams.is_single() {
331        record.upsert("restoringbeam", beam(0, 0));
332    } else if beams.is_multi() {
333        let mut grid = beams.to_record();
334        for plane in 0..beams.n_stokes() {
335            for channel in 0..beams.n_channels() {
336                grid.upsert(
337                    format!("*{}", plane * beams.n_channels() + channel),
338                    beam(channel, plane),
339                );
340            }
341        }
342        record.upsert("perplanebeams", Value::Record(grid));
343    }
344    Ok(record)
