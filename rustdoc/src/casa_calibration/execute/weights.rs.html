10/// Calibrater uses partialMedians without takeEvenMean: an even channel count
11/// takes the lower central value, not the average of the two central values.
12pub(super) fn lower_median(mut values: Vec<f32>) -> f32 {
13    assert!(
14        !values.is_empty(),
15        "weight median requires at least one channel"
16    );
17    let middle = (values.len() - 1) / 2;
18    *values.select_nth_unstable_by(middle, f32::total_cmp).1
19}

21pub(super) fn from_sigma(
22    mut value: ArrayValue,
23    column: &str,
24) -> Result<ArrayValue, ApplyExecutionError> {
25    let ArrayValue::Float32(values) = &mut value else {
26        return Err(ApplyExecutionError::UnsupportedParameterShape {
27            path: format!("<measurement-set {column}>"),
28            shape: value.shape().to_vec(),
29        });
30    };
31    values.mapv_inplace(|sigma| if sigma > 0. { 1. / (sigma * sigma) } else { 0. });
32    Ok(value)
33}

35pub(super) fn has_spectrum(
36    table: &Table,
37    row: Option<usize>,
38    column: &str,
39) -> Result<bool, ApplyExecutionError> {
40    if !table
41        .schema()
42        .is_some_and(|schema| schema.contains_column(column))
43    {
44        return Ok(false);
45    }
46    let Some(row) = row else { return Ok(false) };
47    table
48        .cell_accessor(row, column)
49        .and_then(|cell| cell.array_shape())
50        .map(|shape| shape.is_some_and(|shape| shape.iter().all(|length| *length > 0)))
51        .map_err(|source| ApplyExecutionError::MutateMeasurementSet {
52            path: format!("<measurement-set {column}>"),
53            source: MsError::from(source),
54        })
55}
