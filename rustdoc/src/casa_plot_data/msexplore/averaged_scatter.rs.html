72        || (scalar_average && spec.averaging.avgbaseline)

76        && spec.y_axes.len() > 2

78            || antenna_average
79            || ((spec.averaging.avgscan || spec.averaging.avgfield)

86        scalar_average || scalar_spw_time || (extended_derived && scalar_antenna_time);

100        && time_bins.is_some()
101        && !extended_derived
102        && (phase_shift.is_some()

106            || time_reporting_axes

173        selected_i32_values(ms.main_table(), "DATA_DESC_ID", &row_numbers)?;
174    let selected_field_ids = selected_i32_values(ms.main_table(), "FIELD_ID", &row_numbers)?;
175    let selected_scan_numbers = selected_i32_values(ms.main_table(), "SCAN_NUMBER", &row_numbers)?;
176    let selected_observations =
177        (collapse_baselines || scalar_average || antenna_average || spw_average)

184        || spec.x_axis.uses_uvw()
185        || spec.y_axes.iter().copied().any(MsAxis::uses_uvw))
186    .then(|| selected_uvw_values(ms.main_table(), "UVW", &row_numbers))

189        || spec

192            .copied()
193            .any(MsAxis::uses_derived_geometry)
194        || (needs_spectral_coordinates && requested_freqframe.is_some())

196        Some(MsCalEngine::for_calibration(ms).map_err(|error| error.to_string())?)
197    } else {
198        None

201    let spectral_window = ms.spectral_window().map_err(|error| error.to_string())?;
202    let polarization = ms.polarization().map_err(|error| error.to_string())?;
203    let data_description = ms.data_description().map_err(|error| error.to_string())?;
204    let spans_scan_or_field = spec.averaging.avgscan || spec.averaging.avgfield;
205    let averaged_antenna_color = plotms_averaged_colors::axis(spec).filter(|_| time_antenna);
206    let input_color = if averaged_antenna_color.is_some() {
207        MsColorAxis::None
208    } else {
209        spec.color_by
210    };

213            plotms_ordinates::ScalarSpanPlan::new(

218                    scans: &selected_scan_numbers,
219                    fields: &selected_field_ids,
220                    descriptions: &selected_data_desc_ids,
221                    times: &selected_times,

235        .correlation

246            && iteraxis.is_some_and(|axis| {
247                axis == MsIterationAxis::SpectralWindow || axis.supports_averaged_ordinates()

250        spec.x_axis.uses_flag_rows() || spec.y_axes.iter().copied().any(MsAxis::uses_flag_rows);

252        && (needs_visibility_grid || average_statistics)

255        && (!time_spw || needs_calibrated_weights)
256        && (!collapse_baselines || channel_baselines || time_baselines);
257    let needs_weight_spectrum = matches!(spec.x_axis, MsAxis::WeightSpectrum)

266            .iter()
267            .copied()

356                    .or_insert(row_time_value);

362    let mut uv_selections = HashMap::new();
363    let mut clear_uv_channels = std::collections::HashSet::new();

406            && !include_row_flagged_points
407            && time_bins.is_none()
408            && !collapse_baselines
409            && !antenna_cube_flags
410            && !spw_average

413            continue;
414        }

417        if ddid < 0 || (ddid as usize) >= data_description.row_count() {

421        let spw_id = data_description

423            .map_err(|error| error.to_string())?;

443                            scan_number_value,
444                            field_id_value,

465        }

485                    "msexplore requires BOOL flag cells, found {:?}",

488            }

492            .first()
493            .map(|grid| (grid.corr_count, grid.chan_count))
494            .unwrap_or((flags.nrows(), flags.ncols()));
495        if flags.shape() != [corr_count, chan_count] {
496            return Err(format!(
497                "visibility flag shape {:?} does not match data shape [{}, {}]",

503        let weight_values =
504            float_axis_values(selected_weights.get(row_slot)?, corr_count, "WEIGHT")?;

506        let buffer_first = time_bins
507            .and_then(|plan| plan.input_buffer_first_slot(row_slot))
508            .or(channel_buffer_first);
509        let weight_spectrum_grid = selected_weight_spectrum
