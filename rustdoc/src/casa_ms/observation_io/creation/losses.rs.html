2//! What a source holds that a conversion cannot carry, found before
3//! anything is written.
4//!
5//! The inventory is of whole tables and columns: an MSv2's `SIGMA`,
6//! `FLAG_ROW` flags, `FLAG_CATEGORY` cells, `ARRAY_ID`s other than 0,
7//! `FEED` receptors that vary in time (of which the catalogue keeps the
8//! last row, as casacore's `MSIter` does), MAIN columns outside the
9//! boundary's roles and subtables the catalogue does not carry (system
10//! calibration, weather, flag commands, history, Doppler, frequency
11//! offsets, and any further subtable a keyword names), and into an MSv4,
12//! which holds less of them than the catalogue, its `SOURCE` rows,
13//! `PROCESSOR_ID`s other than 0, a `PROCESSOR` of more than one row and
14//! `STATE` rows that say more than their intents; an MSv4's sub-datasets
15//! and main-dataset variables the writer does not write; and in either the
16//! data groups not selected; and, into an MSv4, antennas no partition's
17//! baselines name. What an MSv2 destination's subtables write as MSv2's
18//! default is found by writing them ([`super::msv2_tables::rehearse`]).
19//! What pointing carries is
20//! decided by [`super::pointing_plan`]. Columns of the subtables the
21//! catalogue does carry (an antenna's offset, a field's delay direction, a
22//! spectral window's resolution) are not inventoried, but an ephemeris
23//! table's are ([`ephemeris_tables`]), since the catalogue carries only the
24//! body's directions, distances and velocities. Fields whose direction is a
25//! polynomial in time are refused: the catalogue would hold a constant term
26//! as the phase centre.

41/// The data groups to convert into an MSv4: `requested`, each present in
42/// every partition, or every group of the source with `base` first.
43#[cfg(feature = "msv4")]
44pub(crate) fn selected_groups(
45    catalogue: &Catalogue,
46    requested: Option<&[String]>,
47) -> Result<Vec<String>, ObservationError> {
48    let groups: Vec<String> = match requested {
49        Some(requested) => requested.to_vec(),
50        None => {
51            let mut groups = vec!["base".to_string()];
52            for entry in &catalogue.partitions {
53                for group in &entry.data_groups {
54                    if !groups.contains(&group.name) {
55                        groups.push(group.name.clone());
56                    }
57                }
58            }
59            groups
60        }
61    };
62    if groups.is_empty() {
63        return Err(request_error(
64            "a conversion selects at least one data group",
65        ));
66    }
67    if let Some(group) = groups
68        .iter()
69        .enumerate()
70        .find_map(|(index, group)| groups[..index].contains(group).then_some(group))
71    {
72        return Err(request_error(format!(
73            "data group {group} is selected twice"
74        )));
75    }
76    for entry in &catalogue.partitions {
77        if let Some(missing) = groups
78            .iter()
79            .find(|group| entry.data_group(group).is_none())
80        {
81            return Err(request_error(format!(
82                "partition {} has no data group {missing}",
83                entry.spectral_window
84            )));
85        }
86    }
87    Ok(groups)
88}

90/// The store a conversion creates, which decides what it must refuse.
91#[derive(Clone, Copy, Debug, PartialEq, Eq)]
92pub(crate) enum Store {
93    /// An MSv2, which writes every catalogue field.
94    Msv2,
95    /// An MSv4, which writes the fields its datasets name.
96    Msv4,
97}

99/// Everything `source` holds that converting `groups` into `destination`
100/// would not carry.
101pub(crate) fn inventory(
102    source: &Path,
103    catalogue: &Catalogue,
104    groups: &[String],
105    destination: Store,
106) -> Result<Vec<Loss>, ObservationError> {
107    let mut losses = Vec::new();
108    for entry in &catalogue.partitions {
109        for group in &entry.data_groups {
110            let loss = Loss::DataGroup(group.name.clone());
111            if !groups.contains(&group.name) && !losses.contains(&loss) {
112                losses.push(loss);
113            }
114        }
115    }
116    if source.join("zarr.json").is_file() {
117        msv4(source, &mut losses)?;
118        // An MSv4 holds no sigma: an MSv2 destination states one.
119        if destination == Store::Msv2 {
120            losses.push(Loss::SigmaDerived);
121        }
122    } else {
123        msv2(source, catalogue, destination, &mut losses)?;
124    }
125    losses.sort();
126    losses.dedup();
127    Ok(losses)
128}

130/// The most [`inventory`] holds reading an MSv2 of `catalogue`: a block of
131/// MAIN's row keys, and a block of `FLAG_CATEGORY` cells of three
132/// categories (casacore's `FLAG_CMD`, `ORIGINAL` and `USER`) of its widest
133/// partition.
134pub(super) fn scan_bytes(catalogue: &Catalogue) -> usize {
135    let keys = ROW_CHUNK * (size_of::<usize>() + size_of::<bool>() + 2 * size_of::<i32>());
136    let widest = |each: fn(&casa_observation::PartitionEntry) -> usize| {
137        catalogue.partitions.iter().map(each).max().unwrap_or(0)
138    };
139    let cell = widest(|entry| entry.frequencies_hz.len() * entry.correlations.len());
140    let categories = CATEGORY_CHUNK * (3 * cell + size_of::<Option<casa_types::ArrayValue>>());
141    let correlations = widest(|entry| entry.correlations.len());
142    let sigmas = SIGMA_CHUNK * (size_of::<usize>() + 2 * correlations * size_of::<f32>());
143    keys.max(categories).max(sigmas)
144}

146/// The antennas of `catalogue` that no partition's baselines name, as a
147/// loss when there are any: an MSv4 dataset holds only its baselines'
148/// antennas.
149#[cfg(feature = "msv4")]
150pub(crate) fn unused_antennas(catalogue: &Catalogue) -> Option<Loss> {
151    let mut named = vec![false; catalogue.antennas.len()];
152    for baseline in catalogue
153        .partitions
154        .iter()
155        .flat_map(|entry| &entry.baselines)
156    {
157        for antenna in [baseline.antenna1, baseline.antenna2] {
158            if let Some(named) = named.get_mut(antenna.0 as usize) {
159                *named = true;
160            }
161        }
162    }
163    let unused = named.iter().filter(|named| !**named).count();
164    (unused > 0).then_some(Loss::UnusedAntennas(unused))
165}

167/// That the last samples of `ephemeris` are extrapolated rather than the
168/// source's ([`FieldEphemeris::extrapolated`]), as a loss when they are: the
169/// destination holds them as samples like any other.
170pub(super) fn extrapolated(ephemeris: &FieldEphemeris) -> Option<Loss> {
171    let count = ephemeris.extrapolated();
172    (count > 0).then(|| {
173        Loss::Column(format!(
174            "the ephemeris of {}: that its last {} extrapolated, not the source's (continuing \
175             the line through the two before to reach the last time's TDB)",
176            ephemeris.name(),
177            if count == 1 {
178                "sample is".to_string()
179            } else {
180                format!("{count} samples are")
181            }
182        ))
183    })
184}

186/// Refuse the first of `losses` whose kind is not in `accept`.
187pub(crate) fn refuse_unaccepted(
188    losses: &[Loss],
189    accept: &[LossKind],
190) -> Result<(), ObservationError> {
191    match losses.iter().find(|loss| !accept.contains(&loss.kind())) {
192        Some(loss) => Err(request_error(format!(
193            "the conversion would lose information: {loss}; accept {:?} to convert anyway",
194            loss.kind()
195        ))),
196        None => Ok(()),
197    }
198}

240/// Rows of `FLAG_ROW` read at a time.
241const ROW_CHUNK: usize = 1 << 18;
242
243fn msv2(
244    source: &Path,
245    catalogue: &Catalogue,
246    destination: Store,
247    losses: &mut Vec<Loss>,
248) -> Result<(), ObservationError> {
249    let ms = crate::MeasurementSet::open_retained_read(source).map_err(storage)?;
250    refuse_polynomial_fields(&ms, catalogue, destination)?;
251    ephemeris_tables(source, &ms, catalogue, destination, losses)?;
252    let main = ms.main_table();
253    let columns: Vec<String> = main
254        .schema()
255        .map(|schema| {
256            schema
257                .columns()
258                .iter()
259                .map(|column| column.name().to_string())
260                .collect()
261        })
262        .unwrap_or_default();
263    if columns.iter().any(|name| name == "FLOAT_DATA") && !columns.iter().any(|name| name == "DATA")
264    {
265        return Err(ObservationError::Unsupported {
266            what: "converting single-dish data (FLOAT_DATA without DATA)",
267        });
268    }
269    let spectral = columns.iter().any(|name| name == "WEIGHT_SPECTRUM");
270    for column in &columns {
271        match column.as_str() {
272            "SIGMA" if destination == Store::Msv2 => {
273                // An MSv2 destination derives SIGMA from the weights it
274                // writes: from WEIGHT_SPECTRUM where it is held.
275                if spectral || !sigma_follows_weight(main)? {
276                    losses.extend([Loss::Sigma, Loss::SigmaDerived]);
277                }
278            }
279            "SIGMA" | "SIGMA_SPECTRUM" => losses.push(Loss::Sigma),
280            "FLAG_CATEGORY" => {
281                if flag_categories_held(main)? {
282                    losses.push(Loss::FlagCategories);
283                }
284            }
285            name if !CARRIED_COLUMNS.contains(&name) => losses.push(Loss::Column(name.to_string())),
286            _ => {}
287        }
288    }
289    if destination == Store::Msv2 && !columns.iter().any(|name| name == "SIGMA") {
290        losses.push(Loss::SigmaDerived);
291    }
292    if spectral {
293        losses.push(Loss::Column(
294            "WEIGHT (the row summary beside WEIGHT_SPECTRUM)".to_string(),
295        ));
296    }
297    main_keys(main, destination, losses)?;
298    if crate::observation_io::msv2::catalogue::read_feeds(&ms)?
299        .varying()
300        .is_some()
301    {
302        losses.push(Loss::Column(
303            "FEED receptors' time dependence (the last row of each antenna, feed and window \
304             is kept)"
305                .to_string(),
306        ));
307    }
308    subtables(&ms, destination, losses)?;
309    if destination == Store::Msv2 {
310        super::preservation::inventory(&ms, losses);
311        unnamed_feeds(&ms, catalogue, losses)?;
312    }
313    Ok(())
314}

316/// Rows of `WEIGHT` and `SIGMA` compared at a time.
317const SIGMA_CHUNK: usize = 1 << 16;
318
319/// Whether every stored `SIGMA` is what an MSv2 destination derives from
320/// `WEIGHT` (`1/√WEIGHT`, and -1 where the weight is not positive) to
321/// within one unit in the last place, so that deriving it loses nothing.
322/// A MAIN whose cells cannot be read together (cells of several shapes) is
323/// taken as not following.
324fn sigma_follows_weight(main: &casa_tables::Table) -> Result<bool, ObservationError> {
325    let rows = main.row_count();
326    let weight = main.column_accessor("WEIGHT").map_err(storage)?;
327    let sigma = main.column_accessor("SIGMA").map_err(storage)?;
328    let (mut weights, mut sigmas) = (Vec::new(), Vec::new());
329    for start in (0..rows).step_by(SIGMA_CHUNK) {
330        let chunk: Vec<usize> = (start..rows.min(start + SIGMA_CHUNK)).collect();
331        weights.clear();
332        sigmas.clear();
333        let read = weight
334            .fill_array_cells_1d_typed_uncached(
335                &chunk,
336                SelectedArray1DCellsMut::Float32(&mut weights),
337            )
338            .and_then(|_| {
339                sigma.fill_array_cells_1d_typed_uncached(
340                    &chunk,
341                    SelectedArray1DCellsMut::Float32(&mut sigmas),
342                )
343            });
344        if read.is_err() || weights.len() != sigmas.len() {
345            return Ok(false);
346        }
347        if weights
348            .iter()
349            .zip(&sigmas)
350            .any(|(weight, sigma)| !within_one_ulp(*sigma, derived_sigma(*weight)))
351        {
352            return Ok(false);
353        }
354    }
355    Ok(true)
356}
357
358/// Whether `a` and `b` are equal or adjacent `f32`s.
359fn within_one_ulp(a: f32, b: f32) -> bool {
360    a == b
361        || (a.is_finite()
362            && b.is_finite()
363            && a.is_sign_positive() == b.is_sign_positive()
364            && a.to_bits().abs_diff(b.to_bits()) <= 1)
365}

367/// The `FEED` rows of antennas no partition's baselines name, a table loss:
368/// an MSv2 destination writes the feeds its partitions' receptors need.
369fn unnamed_feeds(
370    ms: &crate::MeasurementSet,
371    catalogue: &Catalogue,
372    losses: &mut Vec<Loss>,
373) -> Result<(), ObservationError> {
374    let mut named = vec![false; catalogue.antennas.len()];
375    for baseline in catalogue
376        .partitions
377        .iter()
378        .flat_map(|entry| &entry.baselines)
379    {
380        for antenna in [baseline.antenna1, baseline.antenna2] {
381            if let Some(named) = named.get_mut(antenna.0 as usize) {
382                *named = true;
383            }
384        }
385    }
386    let feeds = ms.feed().map_err(storage)?;
387    let mut unnamed = 0;
388    for row in 0..feeds.row_count() {
389        let antenna = feeds.i32(row, "ANTENNA_ID").map_err(storage)?;
390        let used = usize::try_from(antenna)
391            .ok()
392            .and_then(|antenna| named.get(antenna).copied())
393            .unwrap_or(false);
394        unnamed += usize::from(!used);
395    }
396    if unnamed > 0 {
397        losses.push(Loss::Table(format!(
398            "FEED rows of antennas no baseline names ({unnamed})"
399        )));
400    }
401    Ok(())
402}

404/// The fields `destination` writes: every `FIELD` row into an MSv2, and
405/// into an MSv4 those `catalogue`'s partitions name (an MSv2 catalogue's
406/// fields are its `FIELD` rows).
407fn written_fields(catalogue: &Catalogue, destination: Store, rows: usize) -> Vec<bool> {
408    let mut named = vec![destination == Store::Msv2; rows];
409    for field in catalogue.partitions.iter().flat_map(|entry| &entry.fields) {
410        if let Some(named) = named.get_mut(field.0 as usize) {
411            *named = true;
412        }
413    }
414    named
415}

417/// Refuse fields whose directions are polynomials in time (`NUM_POLY` above
418/// 0) among those `destination` writes, as XRADIO's converter refuses the
419/// fields it converts: the catalogue holds one constant direction.
420fn refuse_polynomial_fields(
421    ms: &crate::MeasurementSet,
422    catalogue: &Catalogue,
423    destination: Store,
424) -> Result<(), ObservationError> {
425    let fields = ms.field().map_err(storage)?;
426    let named = written_fields(catalogue, destination, fields.row_count());
427    for row in (0..fields.row_count()).filter(|row| named[*row]) {
428        if fields.num_poly(row).map_err(storage)? > 0 {
429            return Err(ObservationError::Unsupported {
430                what: "converting fields whose direction is a polynomial in time (FIELD NUM_POLY)",
431            });
432        }
433    }
434    Ok(())
435}

437/// What the ephemeris tables of the fields `destination` writes hold that
438/// the catalogue does not carry: columns other than `MJD`, `RA`, `DEC`,
439/// `Rho` and `RadVel`, and keywords other than those an MSv2 destination
440/// writes. Into an MSv4, also the tables' `NAME` (an MSv4 ephemeris is
441/// named by its source) and the fields' `TIME`, at which an image without a
442/// phase centre is centred on the body (#987).
443fn ephemeris_tables(
444    source: &Path,
445    ms: &crate::MeasurementSet,
446    catalogue: &Catalogue,
447    destination: Store,
448    losses: &mut Vec<Loss>,
449) -> Result<(), ObservationError> {
450    let ids = crate::ephemeris::field_ephemeris_ids(ms).map_err(storage)?;
451    let named = written_fields(catalogue, destination, ids.len());
452    let mut followed: Vec<i32> = ids
453        .iter()
454        .zip(&named)
455        .filter_map(|(id, named)| id.filter(|_| *named))
456        .collect();
457    followed.sort_unstable();
458    followed.dedup();
459    if !followed.is_empty() && destination == Store::Msv4 {
460        losses.push(Loss::Column(
461            "FIELD TIME of fields that follow an ephemeris (when an image without a phase \
462             centre is centred on the body)"
463                .to_string(),
464        ));
465    }
466    for id in followed {
467        let path = crate::ephemeris::find_field_ephemeris(source, id).map_err(storage)?;
468        let table =
469            casa_tables::Table::open(casa_tables::TableOptions::new(&path)).map_err(storage)?;
470        let name = path
471            .strip_prefix(source)
472            .unwrap_or(&path)
473            .display()
474            .to_string();
475        let columns: Vec<String> = table
476            .schema()
477            .map(|schema| {
478                schema
479                    .columns()
480                    .iter()
481                    .map(|column| column.name().to_string())
482                    .filter(|column| {
483                        !["MJD", "RA", "DEC", "Rho", "RadVel"].contains(&column.as_str())
484                    })
485                    .collect()
486            })
487            .unwrap_or_default();
488        if !columns.is_empty() {
489            losses.push(Loss::Column(format!("{name} {}", columns.join(", "))));
490        }
491        let mut keywords: Vec<String> = table
492            .keywords()
493            .fields()
494            .iter()
495            .map(|field| field.name.clone())
496            .filter(|keyword| {
497                !super::msv2_ephemeris::KEYWORDS.contains(&keyword.as_str())
498                    || (destination == Store::Msv4 && keyword == "NAME")
499            })
500            .collect();
501        keywords.sort();
502        if !keywords.is_empty() {
503            losses.push(Loss::Column(format!(
504                "{name} keywords {}",
505                keywords.join(", ")
506            )));
507        }
508    }
509    Ok(())
510}

512/// MAIN's keys the destination does not hold: a set `FLAG_ROW`, an
513/// `ARRAY_ID` above the 0 an MSv2 destination writes, and into an MSv4,
514/// which has neither, a `PROCESSOR_ID` above 0 (an MSv2 destination writes
515/// each partition's), read in row chunks. A negative id names nothing:
516/// casacore's UVFITS filler writes `PROCESSOR_ID` -1 in every row.
517fn main_keys(
518    main: &casa_tables::Table,
519    destination: Store,
520    losses: &mut Vec<Loss>,
521) -> Result<(), ObservationError> {
522    let rows = main.row_count();
523    let (mut flags, mut arrays, mut processors) = (Vec::new(), Vec::new(), Vec::new());
524    let (mut flagged, mut array, mut processor) = (false, false, false);
525    for start in (0..rows).step_by(ROW_CHUNK) {
526        let chunk: Vec<usize> = (start..rows.min(start + ROW_CHUNK)).collect();
527        flags.clear();
528        arrays.clear();
529        processors.clear();
530        let mut destinations = [
531            RequiredScalarColumnDestination::new(
532                "FLAG_ROW",
533                RequiredScalarColumnValuesMut::Bool(&mut flags),
534            ),
535            RequiredScalarColumnDestination::new(
536                "ARRAY_ID",
537                RequiredScalarColumnValuesMut::Int32(&mut arrays),
538            ),
539            RequiredScalarColumnDestination::new(
540                "PROCESSOR_ID",
541                RequiredScalarColumnValuesMut::Int32(&mut processors),
542            ),
543        ];
544        main.required_scalar_columns_for_rows_into(&chunk, &mut destinations)
545            .map_err(storage)?;
546        flagged |= flags.iter().any(|flag| *flag);
547        array |= arrays.iter().any(|id| *id > 0);
548        processor |= processors.iter().any(|id| *id > 0);
549        if flagged && array && processor {
550            break;
551        }
552    }
553    if flagged {
554        losses.push(Loss::RowFlags);
555    }
556    if array {
557        losses.push(Loss::Column("ARRAY_ID".to_string()));
558    }
559    if processor && destination == Store::Msv4 {
560        losses.push(Loss::Column("PROCESSOR_ID".to_string()));
561    }
562    Ok(())
563}

565/// Subtables, and subtable content, the destination does not hold: the
566/// uncarried subtables with rows, any further table a keyword names, and
567/// into an MSv4 (an MSv2 destination writes them from the catalogue)
568/// `SOURCE` rows, a `PROCESSOR` of more than one row or of another type
569/// than a correlator, and `STATE` rows that say more than their `OBS_MODE`.
570fn subtables(
571    ms: &crate::MeasurementSet,
572    destination: Store,
573    losses: &mut Vec<Loss>,
574) -> Result<(), ObservationError> {
575    for id in UNCARRIED_SUBTABLES {
576        if ms.subtable(*id).is_some_and(|table| table.row_count() > 0) {
577            losses.push(Loss::Table(id.name().to_string()));
578        }
579    }
580    let main = ms.main_table();
581    let standard: Vec<&str> = SubtableId::ALL_REQUIRED
582        .iter()
583        .chain(UNCARRIED_SUBTABLES)
584        .chain(&[SubtableId::Pointing, SubtableId::Source])
585        .map(SubtableId::name)
586        .collect();
587    for field in main.keywords().fields() {
588        if matches!(field.value, Value::TableRef(_)) && !standard.contains(&field.name.as_str()) {
589            losses.push(Loss::Table(field.name.clone()));
590        }
591    }
592    if destination == Store::Msv2 {
593        return Ok(());
594    }
595    if ms
596        .subtable(SubtableId::Source)
597        .is_some_and(|table| table.row_count() > 0)
598    {
599        losses.push(Loss::Table(SubtableId::Source.name().to_string()));
600    }
601    let processors = ms.processor().map_err(storage)?;
602    let correlator = |row| {
603        processors
604            .string(row, "TYPE")
605            .map(|kind| kind == "CORRELATOR")
606            .map_err(storage)
607    };
608    if processors.row_count() > 1 || (processors.row_count() == 1 && !correlator(0)?) {
609        losses.push(Loss::Table("PROCESSOR".to_string()));
610    }
611    let state = ms.state().map_err(storage)?;
612    for row in 0..state.row_count() {
613        let plain = state.i32(row, "SUB_SCAN").map_err(storage)? == 0
614            && state.f64(row, "CAL").map_err(storage)? == 0.0
615            && state.f64(row, "LOAD").map_err(storage)? == 0.0
616            && state.bool(row, "SIG").map_err(storage)?
617            && !state.bool(row, "REF").map_err(storage)?;
618        if !plain {
619            losses.push(Loss::Column(
620                "STATE SUB_SCAN, CAL, LOAD, SIG and REF".to_string(),
621            ));
622            break;
623        }
624    }
625    Ok(())
626}

628/// Rows of `FLAG_CATEGORY` read at once: an undefined or empty cell costs
629/// nothing to read, and a held one ends the search, so this bounds the
630/// cells read with the first held one (256 cells of three categories of
631/// 32,768 channels and 4 correlations are about 100 MB).
632const CATEGORY_CHUNK: usize = 256;
633
634/// Whether MAIN's `FLAG_CATEGORY` holds categories: a defined cell with
635/// elements in any row. The first and last rows are read first, since
636/// writers usually fill the column whole or leave it undefined throughout.
637fn flag_categories_held(main: &casa_tables::Table) -> Result<bool, ObservationError> {
638    let rows = main.row_count();
639    if rows == 0 {
640        return Ok(false);
641    }
642    let column = main.column_accessor("FLAG_CATEGORY").map_err(storage)?;
643    // Uncached: the inventory then reads FLAG_ROW from the lazy table.
644    let held = |chunk: &[usize]| {
645        column
646            .array_cells_owned_uncached(chunk)
647            .map(|cells| cells.iter().flatten().any(|cell| !cell.is_empty()))
648            .map_err(storage)
649    };
650    if held(&[0, rows - 1])? {
651        return Ok(true);
652    }
653    for start in (1..rows.saturating_sub(1)).step_by(CATEGORY_CHUNK) {
654        let chunk: Vec<usize> = (start..(rows - 1).min(start + CATEGORY_CHUNK)).collect();
655        if held(&chunk)? {
656            return Ok(true);
657        }
658    }
659    Ok(false)
660}

662/// The main dataset's variables the writer writes, coordinates included.
663const CARRIED_VARIABLES: &[&str] = &[
664    "time",
665    "baseline_id",
666    "baseline_antenna1_name",
667    "baseline_antenna2_name",
668    "frequency",
669    "polarization",
670    "uvw_label",
671    "field_name",
672    "scan_name",
673    "FLAG",
674    "WEIGHT",
675    "UVW",
676    "TIME_CENTROID",
677    "EFFECTIVE_INTEGRATION_TIME",
678];
679
680fn msv4(source: &Path, losses: &mut Vec<Loss>) -> Result<(), ObservationError> {
681    let datasets = std::fs::read_dir(source).map_err(storage)?;
682    for dataset in datasets {
683        let dataset = dataset.map_err(storage)?.path();
684        if !dataset.join("zarr.json").is_file() {
685            continue;
686        }
687        for child in std::fs::read_dir(&dataset).map_err(storage)? {
688            let child = child.map_err(storage)?.path();
689            let Some(name) = child
690                .file_name()
691                .map(|name| name.to_string_lossy().into_owned())
692            else {
693                continue;
694            };
695            if !child.join("zarr.json").is_file() {
696                continue;
697            }
698            let is_group = std::fs::read_to_string(child.join("zarr.json")).is_ok_and(|metadata| {
699                metadata.contains("\"node_type\": \"group\"")
700                    || metadata.contains("\"node_type\":\"group\"")
701            });
702            if is_group {
703                if name == "pointing_xds" {
704                    pointing_variables(&child, losses)?;
705                } else if name.starts_with("field_and_source_") {
706                    ephemeris_variables(&child, losses)?;
707                } else if name != "antenna_xds" {
708                    losses.push(Loss::Table(name));
709                }
710            } else if !CARRIED_VARIABLES.contains(&name.as_str()) && !name.starts_with("VISIBILITY")
711            {
712                losses.push(Loss::Column(name));
713            }
714        }
715    }
716    Ok(())
717}

719/// The optional variables of XRADIO's `field_and_source_ephemeris`
720/// (from the table's `NP_ang`, `NP_dist`, `DiskLong`, `DiskLat`, `Sl_lon`,
721/// `Sl_lat`, `r`, `rdot` and `phang`), which the catalogue does not carry.
722const EPHEMERIS_EXTRAS: [&str; 7] = [
723    "NORTH_POLE_POSITION_ANGLE",
724    "NORTH_POLE_ANGULAR_DISTANCE",
725    "SUB_OBSERVER_DIRECTION",
726    "SUB_SOLAR_DIRECTION",
727    "SUB_SOLAR_DISTANCE",
728    "HELIOCENTRIC_RADIAL_VELOCITY",
729    "OBSERVER_PHASE_ANGLE",
730];
731
732/// A field dataset's ephemeris variables a conversion does not carry
733/// ([`EPHEMERIS_EXTRAS`]), as one column loss; and an ephemeris in XRADIO's
734/// interpolated layout (no `time_ephemeris`), whose samples start at the
735/// dataset's first time, so that an MSv2 written from them cannot be
736/// evaluated there (`MeasComet::fillMeas` puts a time on a table's first
737/// row at row −1 for some roundings). Their end, the last time, the
738/// reader extends ([`extrapolated`]).
739fn ephemeris_variables(path: &Path, losses: &mut Vec<Loss>) -> Result<(), ObservationError> {
740    let name = |path: &Path| {
741        path.file_name()
742            .map(|name| name.to_string_lossy().into_owned())
743            .unwrap_or_default()
744    };
745    let label = format!(
746        "{}/{}",
747        path.parent().map(name).unwrap_or_default(),
748        name(path)
749    );
750    let moving = std::fs::read_to_string(path.join("zarr.json"))
751        .is_ok_and(|metadata| metadata.contains("field_and_source_ephemeris"));
752    if moving && !path.join("time_ephemeris").join("zarr.json").is_file() {
753        losses.push(Loss::Column(format!(
754            "{label} ephemeris samples (interpolated by XRADIO onto the visibility times; the \
755             table's own samples are not in the store, nor any before the first time, so \
756             casacore's MeasComet cannot evaluate the first time)"
757        )));
758    }
759    let mut lost = Vec::new();
760    for child in std::fs::read_dir(path).map_err(storage)? {
761        let child = child
762            .map_err(storage)?
763            .file_name()
764            .to_string_lossy()
765            .into_owned();
766        if EPHEMERIS_EXTRAS.contains(&child.as_str()) {
767            lost.push(child);
768        }
769    }
770    if !lost.is_empty() {
771        lost.sort();
772        losses.push(Loss::Column(format!("{label} {}", lost.join(", "))));
773    }
774    Ok(())
775}

778/// losses: it carries `POINTING_BEAM` and its coordinates.
779fn pointing_variables(path: &Path, losses: &mut Vec<Loss>) -> Result<(), ObservationError> {

781        "POINTING_BEAM",
782        "time_pointing",
783        "time",
784        "antenna_name",
785        "local_sky_dir_label",
786        "n_polynomial",
787    ];

789        let name = child
790            .map_err(storage)?
791            .file_name()
792            .to_string_lossy()
793            .into_owned();
794        if name != "zarr.json" && !CARRIED.contains(&name.as_str()) {
795            losses.push(Loss::Table(format!("pointing_xds {name}")));
796        }
797    }
798    Ok(())
