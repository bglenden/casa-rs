2//! CASA's terminal and file listings of calibration solution values.
3
4use std::io::Write;
5use std::path::Path;
6
7use casa_ms::MeasurementSet;
8use casa_provider_contracts::CasaValue;
9use casa_task_runtime::{TaskFailure, TaskInvocation, TaskSuccess};
10use serde::Serialize;
11use tempfile::NamedTempFile;
12
13use crate::listing::{CalibrationListing, list_calibration};
14
15#[derive(Serialize)]
16struct Report {
17    listing: Option<CalibrationListing>,
18    listfile: Option<String>,
19}
20
21pub(super) fn run(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
22    let vis = task.input_path("vis")?;
23    let path = task.input_path("caltable")?;
24    let field = task.text("field")?.unwrap_or_default();
25    let antenna = task.text("antenna")?.unwrap_or_default();
26    let spw = task.text("spw")?.unwrap_or_default();
27    // CASA's Python bridge passes a long to Calibrater's signed Int.
28    let page_rows = task.integer("pagerows")? as i32;
29    let ms = MeasurementSet::open(&vis)
30        .map_err(|error| TaskFailure::execution(error.to_string()).at("vis"))?;
31    let listing = list_calibration(&ms, &path, &field, &antenna, &spw, page_rows)
32        .map_err(TaskFailure::execution)?;
33    let listfile = task.text("listfile")?.filter(|path| !path.is_empty());
34    let paged_reply = listfile.is_none()
35        && page_rows > 0
36        && std::env::var_os("CASARS_LISTCAL_PAGED_REPLY").is_some();
37    if let Some(error) = &listing.error
38        && (!paged_reply || listing.parts.is_empty())
39    {
40        return Err(TaskFailure::execution(error.clone()));
41    }
42    let text = listing.lines();
43    if let Some(path) = listfile {
44        publish_listing(Path::new(&path), &text)?;
45        let message = format!("Writing output to file: {path}\n");
46        return Ok(TaskSuccess::new(
47            CasaValue::None,
48            &Report {
49                listing: None,
50                listfile: Some(path),
51            },
52        )?
53        .with_text(message));
54    }
55    Ok(TaskSuccess::new(
56        CasaValue::None,
57        &Report {
58            listing: Some(listing),
59            listfile: None,
60        },
61    )?
62    .with_text(text))
63}
64
65fn publish_listing(path: &Path, text: &str) -> Result<(), TaskFailure> {
66    let fail = |error: std::io::Error| {
67        TaskFailure::execution(format!(
68            "could not create listing file {}: {error}",
69            path.display()
70        ))
71        .at("listfile")
72    };
73    let directory = path
74        .parent()
75        .filter(|parent| !parent.as_os_str().is_empty())
76        .unwrap_or(Path::new("."));
77    let mut stage = NamedTempFile::new_in(directory).map_err(fail)?;
78    stage.write_all(text.as_bytes()).map_err(fail)?;
79    stage
80        .persist_noclobber(path)
81        .map_err(|error| fail(error.error))?;
82    Ok(())
83}
