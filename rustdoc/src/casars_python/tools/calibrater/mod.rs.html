25/// CASA's calibration tool.
26#[pyclass(name = "calibrater", module = "casars.tools", frozen)]
27pub(crate) struct PyCalibrater {
28    lifecycle: Mutex<CalibraterLifecycle>,
29}
30
31impl PyCalibrater {
32    fn with_lifecycle<T: Send>(
33        &self,
34        py: Python<'_>,
35        operation: impl FnOnce(&mut CalibraterLifecycle) -> PyResult<T> + Send,
36    ) -> PyResult<T> {
37        py.detach(|| {
38            let mut lifecycle = self.lifecycle.lock().map_err(runtime_error)?;
39            operation(&mut lifecycle)
40        })
41    }
42
43    fn attached(
44        &self,
45        py: Python<'_>,
46        operation: impl FnOnce(&mut CalibraterLifecycle) -> Result<(), CalibraterLifecycleError> + Send,
47    ) -> PyResult<bool> {
48        let result = self.with_lifecycle(py, |state| {
49            Ok(match operation(state) {
50                Err(CalibraterLifecycleError::Unattached) => false,
51                result => {
52                    result.map_err(runtime_error)?;
53                    true
54                }
55            })
56        })?;
57        if !result {
58            emit(
59                py,
60                LOGGER,
61                &[LogLine::error("Must first open a MeasurementSet.")],
62            )?;
63        }
64        Ok(result)
65    }
66}

77    #[pyo3(signature = (filename, compress = false, addcorr = true, addmodel = true), text_signature = "($self, filename, compress=False, addcorr=True, addmodel=True)")]
78    fn open(
79        &self,
80        py: Python<'_>,
81        filename: &str,
82        compress: bool,
83        addcorr: bool,
84        addmodel: bool,
85    ) -> PyResult<bool> {
86        self.with_lifecycle(py, |state| {
87            state
88                .open_with_scratch(Path::new(filename), compress, addcorr, addmodel)
89                .map_err(runtime_error)?;
90            Ok(true)
91        })
92    }

94    fn close(&self, py: Python<'_>) -> PyResult<bool> {
95        self.with_lifecycle(py, |state| {
96            state.close();
97            Ok(true)
98        })
99    }

101    fn done(&self, py: Python<'_>) -> PyResult<bool> {
102        self.close(py)
103    }

105    #[pyo3(signature = (old = false, quiet = false), text_signature = "($self, old=False, quiet=False)")]
106    fn setvi(&self, py: Python<'_>, old: bool, quiet: bool) -> PyResult<bool> {
107        let (changed, was_old) = self.with_lifecycle(py, |state| {
108            let was_old = state.legacy_iterator();
109            Ok((state.set_legacy_iterator(old), was_old))
110        })?;
111        if !changed {
112            emit(
113                py,
114                LOGGER,
115                &[LogLine::error("Must call setvi _before_ open!")],
116            )?;
117        } else if !quiet {
118            let message = match (old, was_old) {
119                (true, false) => "Forcing use of OLD VisibilityIterator.",
120                (true, true) => "Already using OLD VisibilityIterator.",
121                (false, true) => "Forcing use of NEW VisibilityIterator.",
122                (false, false) => "Already using NEW VisibilityIterator.",
123            };
124            emit(py, LOGGER, &[LogLine::warning(message)])?;
125        }
126        Ok(changed)
127    }

129    #[pyo3(signature = (time = Variant::Empty, spw = Variant::Empty, scan = Variant::Empty, field = Variant::Empty, intent = Variant::Empty, observation = Variant::Empty, baseline = Variant::Empty, uvrange = Variant::Empty, chanmode = "none", nchan = 1, start = 0, step = 1, mstart = Variant::Text("0km/s".into()), mstep = Variant::Text("0km/s".into()), msselect = ""), text_signature = "($self, time=[], spw=[], scan=[], field=[], intent=[], observation=[], baseline=[], uvrange=[], chanmode='none', nchan=1, start=0, step=1, mstart={'value': 0.0, 'unit': 'km/s'}, mstep={'value': 0.0, 'unit': 'km/s'}, msselect='')")]
130    #[allow(clippy::too_many_arguments)]
131    fn selectvis(
132        &self,
133        py: Python<'_>,
134        time: Variant,
135        spw: Variant,
136        scan: Variant,
137        field: Variant,
138        intent: Variant,
139        observation: Variant,
140        baseline: Variant,
141        uvrange: Variant,
142        chanmode: &str,
143        nchan: i64,
144        start: i64,
145        step: i64,
146        mstart: Variant,
147        mstep: Variant,
148        msselect: &str,
149    ) -> PyResult<bool> {
150        let mstart = velocity_argument(&mstart, "mstart")?;
151        let mstep = velocity_argument(&mstep, "mstep")?;
152        let selection = MsSelection {
153            timerange: selector(&time),
154            spw: selector(&spw),
155            scan: selector(&scan),
156            field: selector(&field),
157            intent: selector(&intent),
158            observation: selector(&observation),
159            antenna: selector(&baseline),
160            uvrange: selector(&uvrange),
161            msselect: (!msselect.is_empty()).then(|| msselect.to_owned()),
162            ..MsSelection::default()
163        };
164        let result = self.with_lifecycle(py, |state| {
165            if state.dataset_path().is_none() {
166                return Ok(false);
167            }
168            let velocity = Unit::new("m/s").expect("a velocity unit");
169            for quantity in [&mstart, &mstep] {
170                if !quantity.unit().conformant(&velocity) {
171                    return Err(runtime_error(format!(
172                        "Quantum::assure non-conforming unit type '{}'",
173                        quantity.unit().name(),
174                    )));
175                }
176            }
177            state
178                .select_with_channel_mode(selection, chanmode, [nchan, start, step])
179                .map_err(runtime_error)?;
180            Ok(true)
181        })?;
182        if !result {
183            emit(
184                py,
185                LOGGER,
186                &[LogLine::error("Must first open a MeasurementSet.")],
187            )?;
188        }
189        Ok(result)
190    }

202    #[pyo3(signature = (otf = false, field = Variant::Empty, spw = Variant::Empty, scr = false), text_signature = "($self, otf=False, field=[], spw=[], scr=False)")]
203    fn delmod(
204        &self,
205        py: Python<'_>,
206        otf: bool,
207        field: Variant,
208        spw: Variant,
209        scr: bool,
210    ) -> PyResult<bool> {
211        if !otf && !scr {
212            let lines = self.with_lifecycle(py, |state| match state.model_header_lines() {
213                Ok(lines) => Ok(Some(
214                    lines.into_iter().map(LogLine::info).collect::<Vec<_>>(),
215                )),
216                Err(CalibraterLifecycleError::Unattached) => Ok(None),
217                Err(error) => Err(runtime_error(error)),
218            })?;
219            return if let Some(lines) = lines {
220                emit(py, LOGGER, &lines)?;
221                Ok(true)
222            } else {
223                emit(
224                    py,
225                    LOGGER,
226                    &[LogLine::error("Must first open a MeasurementSet.")],
227                )?;
228                Ok(false)
229            };
230        }
231        self.attached(py, |state| {
232            state.delete_model(
233                selector(&field).as_deref(),
234                selector(&spw).as_deref(),
235                otf,
236                scr,
237            )
238        })
239    }

241    #[pyo3(signature = (stokes = DoubleList::of(&[0., 0., 0., 0.])), text_signature = "($self, stokes=[0.0, 0.0, 0.0, 0.0])")]
242    fn setptmodel(&self, py: Python<'_>, stokes: DoubleList) -> PyResult<bool> {
243        self.with_lifecycle(py, |state| {
244            state
245                .set_point_model(&stokes.0)
246                .map_err(|error| match error {
247                    CalibraterLifecycleError::Unattached => {
248                        runtime_error("Error in Calibrater::setModel: no VisEquation.")
249                    }
250                    error => runtime_error(error),
251                })?;
252            Ok(true)
253        })
254    }

256    #[pyo3(signature = (vary, niter = 0, compshape = "P", par = DoubleList::of(&[1., 0., 0.]), file = ""), text_signature = "($self, vary, niter=0, compshape='P', par=[1.0, 0.0, 0.0], file='')")]
257    fn modelfit<'py>(
258        &self,
259        py: Python<'py>,
260        #[pyo3(from_py_with = bool_vector)] vary: Vec<bool>,
261        niter: i64,
262        compshape: &str,
263        par: DoubleList,
264        file: &str,
265    ) -> PyResult<Bound<'py, PyArray1<f64>>> {
266        let result = self.with_lifecycle(py, |state| {
267            if !state.legacy_iterator() {
268                return Err(runtime_error(
269                    CalibraterLifecycleError::FitNeedsLegacyIterator,
270                ));
271            }
272            let shape = match compshape {
273                "P" => VisibilityFitShape::Point,
274                "G" => VisibilityFitShape::Gaussian,
275                "D" => VisibilityFitShape::Disk,
276                _ => return Err(runtime_error("Error in Calibrater::modelfit.")),
277            };
278            state
279                .fit_model(
280                    VisibilityFitRequest {
281                        shape,
282                        initial: &par.0,
283                        vary: &vary,
284                        iterations: niter,
285                    },
286                    (!file.is_empty()).then(|| Path::new(file)),
287                )
288                .map_err(|_| runtime_error("Error in Calibrater::modelfit."))
289        })?;
290        Ok(PyArray1::from_vec(py, result))
291    }
292}

294fn velocity_argument(value: &Variant, name: &str) -> PyResult<Quantity> {
295    if let Variant::Record(record) = value {
296        match (record.get("value"), record.get("unit")) {
297            (None, None) => return Quantity::new(0., "km/s").map_err(runtime_error),
298            (None, Some(Variant::Text(unit))) => {
299                return Quantity::new(0., unit).map_err(runtime_error);
300            }
301            (Some(Variant::Number(value)), None) => {
302                return Quantity::new(*value, "km/s").map_err(runtime_error);
303            }
304            _ => {}
305        }
306    }
307    if !matches!(value, Variant::Text(_) | Variant::Record(_)) {
308        return Err(PyTypeError::new_err(format!("_{name} is not a dictionary")));
309    }
310    quantity(value).map_err(runtime_error)
311}

313fn bool_vector(value: &Bound<'_, PyAny>) -> PyResult<Vec<bool>> {
314    let flag = |item: &Bound<'_, PyAny>| {
315        item.extract::<bool>()
316            .or_else(|_| item.extract::<i64>().map(|value| value != 0))
317    };
318    if let Ok(flag) = flag(value) {
319        return Ok(vec![flag]);
320    }
321    value.try_iter()?.map(|item| flag(&item?)).collect()
322}
323
324fn selector(value: &Variant) -> Option<String> {
325    match value {
326        Variant::Empty => None,
327        Variant::Text(text) if text.is_empty() => None,
328        Variant::Texts(values) => Some(values.join(",")),
329        Variant::Numbers(values) => Some(
330            values
331                .iter()
332                .map(f64::to_string)
333                .collect::<Vec<_>>()
334                .join(","),
335        ),
336        value => Some(value.describe()),
337    }
338}
