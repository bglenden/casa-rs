4use std::collections::BTreeMap;

6use casa_tables::{Table, TableError};
7use casa_types::ArrayValue;

9use crate::{MeasurementSet, ResolvedMsSelectionRow};

11/// A bounded, read-only source for columns computed from the input
12/// MeasurementSet, for example calibrated visibilities.
13///
14/// Array shapes are the physical input shapes. The transform applies
15/// correlation and channel selection after reading them. An implementation
16/// returns one cell per requested physical row in request order and may
17/// leave untouched column names to the ordinary table reader.
18pub trait TransformInputView: std::fmt::Debug + Send + Sync {
19    /// Prepare the input view for the exact physical rows and channels used
20    /// by one transform product. Called again for each Multi-MS partition.
21    fn prepare<'a>(
22        &self,
23        input: &'a MeasurementSet,
24        rows: &[ResolvedMsSelectionRow],
25        channels: &BTreeMap<i32, &[usize]>,
26    ) -> Result<Box<dyn PreparedTransformInputView + 'a>, TableError>;
27
28    /// Additional transient bytes per input visibility sample while reading.
29    fn additional_bytes_per_sample(&self) -> usize;
30
31    /// Whether the view provides this column for the supplied physical
32    /// input table even when the column does not exist on disk.
33    fn provides(&self, input: &MeasurementSet, column: &str) -> bool;
34}
35
36/// Computed visibility arrays bound to one physical transform selection.
37pub trait PreparedTransformInputView: Send + Sync {
38    /// Materialize requested computed columns for a bounded block.
39    fn read(
40        &self,
41        input: &Table,
42        rows: &[usize],
43        columns: &[&str],
44    ) -> Result<TransformInputColumns, TableError>;
45}
46
47/// Optional input columns read for one transform block.
48pub type TransformInputColumns = BTreeMap<&'static str, Vec<Option<ArrayValue>>>;
