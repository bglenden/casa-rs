23fn has_zero_row(matrix: &Array2<f64>) -> bool {
24    matrix
25        .rows()
26        .into_iter()
27        .any(|row| row.iter().all(|&value| value == 0.0))
28}

30fn singular_linear_transform() -> PyErr {

34pub(in crate::tools) fn validate_restored_linear_coordinates(
35    coordinates: &CoordinateSystem,
36    record: &RecordValue,
37) -> PyResult<()> {
38    if !(0..coordinates.n_coordinates()).any(|index| {
39        matches!(
40            coordinates.coordinate(index),
41            CoordinateModel::Linear(_) | CoordinateModel::System(_)
42        )
43    }) {
44        return Ok(());
45    }
46    for (index, placement) in coordinates
47        .coordinates_in_record(record)
48        .map_err(raised)?
49        .into_iter()
50        .enumerate()
51    {
52        let mapped = coordinates.coordinate(index);
53        let full;
54        let model = if matches!(
55            mapped,
56            CoordinateModel::Linear(_) | CoordinateModel::System(_)
57        ) && (placement.pixel_axes.len() != mapped.n_pixel_axes()
58            || placement.world_axes.len() != mapped.n_world_axes())
59        {
60            // The compact model can omit the invalid axis.
61            full = placement.model(record).map_err(raised)?;
62            &full
63        } else {
64            mapped
65        };
66        match model {
67            CoordinateModel::Linear(linear) => {
68                if linear.increment().contains(&0.0) || has_zero_row(linear.pc_matrix()) {
69                    return Err(singular_linear_transform());
70                }
71            }
72            CoordinateModel::System(nested) => {
73                if let Some(Value::Record(nested_record)) = record.get(&placement.field) {
74                    validate_restored_linear_coordinates(nested, nested_record)?;
75                }
76            }
77            _ => {}
78        }
79    }
80    Ok(())
81}

83pub(super) fn matrix_input(value: Option<&Bound<'_, PyAny>>) -> PyResult<Array2<f64>> {
84    let invalid = || raised("The value array is not a double or integer array");
85    let value = value.filter(|value| !value.is_none()).ok_or_else(invalid)?;
86    let (mut shape, values) = if let Ok(array) = value.cast::<PyUntypedArray>() {
87        if array.ndim() == 0 {
88            return Err(raised("The value array is empty"));
89        }
90        if !matches!(array.dtype().kind(), b'i' | b'u' | b'f') {
91            return Err(invalid());
92        }
93        let native = value.call_method1("astype", ("float64",))?;
94        let native = native.extract::<PyReadonlyArrayDyn<'_, f64>>()?;
95        (
96            native.shape().to_vec(),
97            native.as_array().t().iter().copied().collect(),
98        )
99    } else {
100        match value.extract::<Variant>()? {
101            Variant::Numbers(values) => (vec![values.len()], values),
102            Variant::Array { shape, values } => (shape, values),
103            Variant::Complex(_) => {
104                return Err(raised("The value array is empty"));
105            }
106            _ => return Err(invalid()),
107        }
108    };
109    if shape.len() == 1 {
110        shape.push(1);
111    }
112    let [rows, cols] = shape.as_slice() else {
113        return Err(raised(
114            "The transform matrix must have at most two dimensions",
115        ));
116    };
117    // CASA's displayed transform is the transpose of its native WCS PC.
118    Array2::from_shape_vec((*cols, *rows), values).map_err(raised)
119}

121fn write_fields(target: &mut RecordValue, source: &RecordValue, fields: &[&str]) -> PyResult<()> {
122    for &name in fields {
123        let value = source
124            .get(name)
125            .ok_or_else(|| raised(format!("Missing coordinate field {name}")))?;
126        target.upsert(name, value.clone());
127    }
128    Ok(())
129}

131fn edit_record(original: &RecordValue, changed: &CoordinateModel) -> PyResult<RecordValue> {
132    let mut updated = original.clone();
133    let native = changed.to_record();
134    match changed {
135        CoordinateModel::Spectral(spectral) => {
136            let (key, fields) = if spectral.has_linear_mapping() {
137                ("wcs", &["pc"][..])
138            } else {
139                (
140                    "tabular",
141                    &[
142                        "pc",
143                        "crval",
144                        "cdelt",
145                        "crpix",
146                        "units",
147                        "pixelvalues",
148                        "worldvalues",
149                    ][..],
150                )
151            };
152            let Some(Value::Record(source)) = native.get(key) else {
153                return Err(raised(format!("Missing spectral coordinate field {key}")));
154            };
155            if let Some(Value::Record(destination)) = updated.get(key) {
156                let mut destination = destination.clone();
157                write_fields(&mut destination, source, fields)?;
158                updated.upsert(key, Value::Record(destination));
159            } else {
160                updated.upsert(key, Value::Record(source.clone()));
161            }
162        }
163        CoordinateModel::Direction(_) => write_fields(&mut updated, &native, &["pc", "cdelt"])?,
164        CoordinateModel::Tabular(_) => write_fields(
165            &mut updated,
166            &native,
167            &["pc", "crval", "cdelt", "crpix", "worldvalues"],
168        )?,
169        _ => write_fields(&mut updated, &native, &["pc"])?,
170    }
171    Ok(updated)
172}

174pub(super) fn normalize_restored_matrix(
175    original: &RecordValue,
176    model: &CoordinateModel,
177) -> PyResult<Option<RecordValue>> {
178    let table = match model {
179        CoordinateModel::Tabular(_) => original,
180        CoordinateModel::Spectral(spectral) if !spectral.has_linear_mapping() => {
181            recorded_coordinate(original, "tabular")?
182        }
183        _ => return Ok(None),
184    };
185    let matrix = match table.get("pc") {
186        Some(Value::Array(ArrayValue::Float64(values))) => values.first().copied(),
187        Some(Value::Array(ArrayValue::Float32(values))) => values.first().copied().map(f64::from),
188        Some(Value::Scalar(ScalarValue::Float64(value))) => Some(*value),
189        _ => None,
190    };
191    if matrix.is_some_and(|factor| factor != 1.0) {
192        Ok(Some(edit_record(original, model)?))
193    } else if matches!(model, CoordinateModel::Spectral(_)) {
194        let edited = edit_record(original, model)?;
195        let normalized = recorded_coordinate(&edited, "tabular")?;
196        let changed = ["crpix", "crval", "cdelt"]
197            .iter()
198            .any(|field| table.get(field) != normalized.get(field));
199        Ok(changed.then_some(edited))
200    } else if matches!(model, CoordinateModel::Tabular(table) if !table.pixel_values().is_empty()) {
201        let edited = edit_record(original, model)?;
202        Ok((&edited != original).then_some(edited))
203    } else {
204        Ok(None)
205    }
206}

208fn update_stored_coordinate(
209    record: &mut RecordValue,
210    field: &str,
211    model: &CoordinateModel,
212) -> PyResult<()> {
213    let Some(Value::Record(original)) = record.get(field) else {
214        return Err(raised("Missing coordinate record"));
215    };
216    let stored = edit_record(original, model)?;
217    record.upsert(field, Value::Record(stored));
218    Ok(())
219}

221// Live tabular transforms survive other edits to a CASA tool. Restoration
222// from serialized samples resets the matrix and reference parameters.
223fn live_tabular_factor(model: &CoordinateModel) -> Option<f64> {
224    let factor = match model {
225        CoordinateModel::Tabular(table) if !table.pixel_values().is_empty() => {
226            return Some(table.linear_transform());
227        }
228        CoordinateModel::Tabular(tabular) => tabular.linear_transform(),
229        CoordinateModel::Spectral(spectral) if !spectral.has_linear_mapping() => {
230            spectral.linear_transform()
231        }
232        _ => return None,
233    };
234    (factor != 1.0).then_some(factor)
235}

237fn change_tabular_matrix(model: &CoordinateModel, factor: f64) -> PyResult<CoordinateModel> {
238    match model {
239        CoordinateModel::Tabular(tabular) => Ok(tabular
240            .clone()
241            .with_linear_transform(factor)
242            .map_err(raised)?
243            .into()),
244        CoordinateModel::Spectral(spectral) => Ok(spectral
245            .clone()
246            .with_linear_transform(factor)
247            .map_err(raised)?
248            .into()),
249        _ => Err(raised(
250            "Coordinate no longer supports its tabular transform",
251        )),
252    }
253}

255fn live_tabular_model(model: &CoordinateModel, record: &RecordValue) -> PyResult<CoordinateModel> {
256    match model {
257        CoordinateModel::Tabular(tabular) => Ok(tabular
258            .clone()
259            .with_live_reference(record)
260            .map_err(raised)?
261            .into()),
262        CoordinateModel::Spectral(spectral) => Ok(spectral
263            .clone()
264            .with_live_reference(record)
265            .map_err(raised)?
266            .into()),
267        _ => Err(raised(
268            "Coordinate no longer supports its tabular transform",
269        )),
270    }
271}

273fn recorded_coordinate<'a>(record: &'a RecordValue, field: &str) -> PyResult<&'a RecordValue> {
274    match record.get(field) {
275        Some(Value::Record(value)) => Ok(value),
276        _ => Err(raised("Missing coordinate record")),
277    }

280impl CoordinateState {
281    fn live_tabular_factors(&self) -> impl Iterator<Item = (usize, f64)> + '_ {
282        (0..self.coordinates.n_coordinates()).filter_map(|index| {
283            live_tabular_factor(self.coordinates.coordinate(index)).map(|factor| (index, factor))
284        })
285    }
286
287    pub(super) fn editable_record(&self) -> PyResult<RecordValue> {
288        let mut record = self.record.clone();
289        let mut matrices = self.live_tabular_factors().peekable();
290        if matrices.peek().is_none() {
291            return Ok(record);
292        }
293        let placements = self
294            .coordinates
295            .coordinates_in_record(&record)
296            .map_err(raised)?;
297        for (index, _) in matrices {
298            let original = self.coordinates.coordinate(index);
299            let unscaled = change_tabular_matrix(original, 1.0)?;
300            update_stored_coordinate(&mut record, &placements[index].field, &unscaled)?;
301        }
302        Ok(record)
303    }
304
305    pub(super) fn apply_edited_record(&mut self, record: RecordValue) -> PyResult<()> {
306        let coordinates = self
307            .coordinates
308            .restore_record_with_layout(&record)
309            .map_err(raised)?;
310        self.apply_edited_coordinates(record, coordinates)
311    }

313    pub(super) fn edit_record_axis(
314        &self,
315        record: &mut RecordValue,
316        axis: usize,
317        edit: TabularEdit,
318    ) -> PyResult<()> {
319        if self.live_tabular_factors().next().is_some() {
320            let placements = self
321                .coordinates
322                .coordinates_in_record(record)
323                .map_err(raised)?;
324            let pixel = matches!(edit, TabularEdit::ReferencePixel(_));
325            if let Some((index, placement)) =
326                placements.iter().enumerate().find(|(_, placement)| {
327                    let axes = if pixel {
328                        &placement.pixel_axes
329                    } else {
330                        &placement.world_axes
331                    };
332                    axes.contains(&Some(axis))
333                })
334                && live_tabular_factor(self.coordinates.coordinate(index)).is_some()
335            {
336                let native = recorded_coordinate(record, &placement.field)?;
337                let base = placement.model(record).map_err(raised)?;
338                let mut model = live_tabular_model(&base, native)?;
339                match &mut model {
340                    CoordinateModel::Tabular(table) => {
341                        table.edit_live_reference(edit).map_err(raised)?
342                    }
343                    CoordinateModel::Spectral(spectral) => {
344                        spectral.edit_live_reference(edit).map_err(raised)?
345                    }
346                    _ => {
347                        return Err(raised(
348                            "Coordinate no longer supports its tabular transform",
349                        ));
350                    }
351                }
352                return update_stored_coordinate(record, &placement.field, &model);
353            }
354        }
355        match edit {
356            TabularEdit::ReferencePixel(value) => {
357                CoordinateSystem::set_reference_pixel_in_record(record, axis, value)
358            }
359            TabularEdit::ReferenceValue(value) => {
360                CoordinateSystem::set_world_reference_value_in_record(record, axis, value)
361            }
362            TabularEdit::Increment(value) => {
363                CoordinateSystem::set_world_increment_in_record(record, axis, value)
364            }
365        }
366        .map(|_| ())
367        .map_err(raised)
368    }

370    pub(super) fn apply_edited_coordinates(
371        &mut self,
372        mut record: RecordValue,
373        mut coordinates: CoordinateSystem,
374    ) -> PyResult<()> {
375        let mut matrices = self.live_tabular_factors().peekable();
376        if matrices.peek().is_some() {
377            let previous = self
378                .coordinates
379                .coordinates_in_record(&self.record)
380                .map_err(raised)?;
381            let placements = coordinates.coordinates_in_record(&record).map_err(raised)?;
382            for (index, factor) in &mut matrices {
383                let field = &previous[index].field;
384                let index = placements
385                    .iter()
386                    .position(|placement| &placement.field == field)
387                    .ok_or_else(|| raised("Missing coordinate placement"))?;
388                let native = recorded_coordinate(&record, field)?;
389                let base = live_tabular_model(coordinates.coordinate(index), native)?;
390                let changed = change_tabular_matrix(&base, factor)?;
391                update_stored_coordinate(&mut record, field, &changed)?;
392                coordinates.replace_coordinate(index, changed);
393            }
394        }
395        drop(matrices);
396        coordinates.retain_direction_mix_ranges(&self.coordinates);
397        self.record = record;
398        self.coordinates = coordinates;
399        Ok(())
400    }
401}

403fn apply_matrix(
404    state: &mut CoordinateState,
405    kind: CoordinateType,
406    matrix: Array2<f64>,
407) -> PyResult<bool> {
408    let (model, placement) = coordinate(state, kind)?.ok_or_else(|| missing(kind))?;
409    let placements = state
410        .coordinates
411        .coordinates_in_record(&state.record)
412        .map_err(raised)?;
413    let index = placements
414        .iter()
415        .position(|entry| entry.field == placement.field)
416        .ok_or_else(|| raised("Missing coordinate placement"))?;
417    let sliced = placement.pixel_axes.len() != model.n_pixel_axes()
418        || placement.world_axes.len() != model.n_world_axes();
419    let full = if sliced {
420        placement.model(&state.record).map_err(raised)?
421    } else {
422        model.clone()
423    };
424    let count = full.n_world_axes();
425    let valid_shape = if matches!(full, CoordinateModel::Tabular(_)) {
426        matrix.len() == 1
427    } else {
428        matrix.dim() == (count, count)
429    };
430    // CASA returns true when the coordinate refuses a matrix of the wrong shape.
431    if !valid_shape
432        || matches!(
433            full,
434            CoordinateModel::Stokes(_) | CoordinateModel::Quality(_)
435        )
436    {
437        return Ok(true);
438    }
439    if matrix.iter().any(|value| !value.is_finite()) {
440        return Err(raised(
441            "The linear transform matrix must contain finite values",
442        ));
443    }
444    let changed = match full {
445        CoordinateModel::Direction(direction) => direction
446            .with_normalized_pc_matrix(matrix)
447            .map_err(raised)?
448            .into(),
449        CoordinateModel::Linear(linear) => {
450            if has_zero_row(&matrix) {
451                return Err(singular_linear_transform());
452            }
453            linear.with_pc_matrix(matrix).into()
454        }
455        CoordinateModel::Spectral(spectral) => spectral
456            .with_linear_transform(matrix[[0, 0]])
457            .map_err(raised)?
458            .into(),
459        CoordinateModel::Tabular(tabular) => tabular
460            .with_linear_transform(matrix[[0, 0]])
461            .map_err(raised)?
462            .into(),
463        CoordinateModel::Stokes(_) | CoordinateModel::Quality(_) | CoordinateModel::System(_) => {
464            return Ok(true);
465        }
466    };
467    let mut record = state.record.clone();
468    update_stored_coordinate(&mut record, &placement.field, &changed)?;
469    let changed = if sliced {
470        match state.coordinates.restore_record_with_layout(&record) {
471            Ok(restored) => restored.coordinate(index).clone(),
472            Err(CoordinateError::UnsupportedAxisLayout(_)) => {
473                let mut record = state.editable_record()?;
474                update_stored_coordinate(&mut record, &placement.field, &changed)?;
475                let restored = CoordinateSystem::from_mapped_record(&record).map_err(raised)?;
476                state.apply_edited_coordinates(record, restored)?;
477                return Ok(true);
478            }
479            Err(error) => return Err(raised(error)),
480        }
481    } else {
482        changed
483    };
484    state.coordinates.replace_coordinate(index, changed);
485    state.record = record;
486    Ok(true)
487}

489#[pymethods]
490impl PyCoordsys {
491    #[pyo3(signature = (r#type="", value=None), text_signature = "($self, type='', value=[])")]
492    fn setlineartransform(
493        &self,
494        py: Python<'_>,
495        r#type: &str,
496        value: Option<&Bound<'_, PyAny>>,
497    ) -> PyResult<bool> {
498        if r#type.is_empty() {
499            return Err(raised("You must specify the coordinate type"));
500        }
501        let matrix = matrix_input(value)?;
502        py.detach(|| apply_matrix(&mut self.state(), coordinate_type(r#type), matrix))
503    }
