57/// One partition of the MeasurementSet: its rows on their grid.
58#[derive(Debug)]
59struct Partition {
60    grid: Grid,
61    /// The `STATE::OBS_MODE` its rows share; empty for rows without a
62    /// state.
63    obs_mode: String,
64    /// Its `DATA_DESCRIPTION` row.
65    data_description_id: u32,
66    /// The `SPECTRAL_WINDOW` row of its data description.
67    spectral_window_id: u32,
68    /// Whether its weights are `WEIGHT_SPECTRUM` ([`has_content`] at its
69    /// first row): CASA's iterator asks again at each new spectral window,
70    /// on the first row it meets there.
71    spectral_weights: bool,
72}

74/// An opened MeasurementSet v2 and its catalogue.
75///
76/// Opening reads MAIN's key columns in row chunks and partitions the rows the way
77/// XRADIO's converter does (`DATA_DESC_ID` × `OBS_MODE` × `OBSERVATION_ID`
78/// × `EPHEMERIS_ID` and the opening's
79/// [`PartitionScheme`](super::PartitionScheme)), and further by the feed
80/// pair, which XRADIO does not split by. A row whose `STATE_ID` is -1 has
81/// no intents here, where XRADIO's converter gives it the last `STATE`
82/// row's `OBS_MODE` (NumPy's index -1), so an MS mixing such rows with
83/// `STATE` rows partitions differently. Each partition's rows are
84/// placed on its time × baseline grid (the `index` module). Each partition
85/// has one data group, `base`, associating `DATA`, `FLAG`,
86/// `WEIGHT_SPECTRUM` (when its first row has content, else `WEIGHT`) and
87/// `UVW`. Antenna,
88/// field and observation identities are the subtable rows; scans are the
89/// distinct `SCAN_NUMBER`s in ascending order.
90///
91/// Opening checks every unit and measure reference the batches rely on:
92/// `TIME` and `TIME_CENTROID` share one fixed reference (a variable
93/// reference is refused); the UVW convention is resolved to the phase
94/// centres' frame ([`UvwFrame`]); positions,
95/// diameters, angles and frequencies are in the units their types claim.
96pub struct Msv2Observation {
97    ms: MeasurementSet,
98    catalogue: Arc<Catalogue>,
99    partitions: Vec<Partition>,
100    /// Each `STATE` row's `OBS_MODE`.
101    state_modes: Vec<String>,
102    /// The `STATE_ID`s MAIN's rows hold.
103    main_states: BTreeSet<i32>,
104    /// Whether MAIN's first row has `SIGMA_SPECTRUM` content: CASA's
105    /// iterator asks once per MeasurementSet.
106    spectral_sigmas: bool,
107    /// Each data group's visibility column and its tile shape, read when
108    /// the observation opened: an estimate does not read `table.dat` again,
109    /// which an update session over the same MeasurementSet rewrites as it
110    /// saves each batch.
111    tile_shapes: Vec<(String, Option<Vec<usize>>)>,
112    options: OpenOptions,
113    /// The most its open tables hold: their descriptions and keywords
114    /// ([`held::opening_bytes`]) and the subtables' rows the opening loaded
115    /// ([`held::read_bytes`]).
116    tables_bytes: usize,
117}

120    /// Open the MeasurementSet at `path` and read its catalogue.
121    ///
122    /// The tables stay open with casacore read locks, and their payload
123    /// rows stay on disk: the typed column fills read only the selected
124    /// cells, as the production selected-observation source does.
125    ///
126    /// # Errors
127    ///
128    /// When the MeasurementSet cannot be opened, a subtable the catalogue
129    /// needs is missing or malformed, a unit or measure reference is
130    /// outside the trial profile, or the rows do not fit their grids (two
131    /// rows in one cell, two fields or scans at one time, several
132    /// `PROCESSOR_ID`s in one partition).
133    #[cfg(unix)]
134    pub fn open(path: impl AsRef<Path>) -> Result<Self, ObservationError> {
135        Self::open_with(path, &OpenOptions::default())
136    }

138    /// [`Self::open`] with `options`.
139    ///
140    /// # Errors
141    ///
142    /// As [`Self::open`].
143    #[cfg(unix)]
144    pub fn open_with(
145        path: impl AsRef<Path>,
146        options: &OpenOptions,
147    ) -> Result<Self, ObservationError> {
148        Self::open_granted(path, options, &mut |_| Ok(()))
149    }

151    /// [`Self::open_with`], asking `grant` for the bytes the opening will
152    /// hold before it allocates them: its tables' descriptions and keywords
153    /// (bounded by their files), the rows of the subtables its catalogue is
154    /// read from (a table loads every row on first touch and keeps them)
155    /// and what is read from them, then its index (its chunk of MAIN's key
156    /// columns, each partition's distinct times, baselines and processors as
157    /// the first pass finds them, at most one row's beyond what was granted,
158    /// then the scans and the grids) and its partitions' entries. The
159    /// totals asked for only grow, a few at a time; a caller that admits
160    /// memory admits each increase. An error from `grant` refuses the
161    /// opening with nothing more allocated.
162    /// Temporary payload staging in the spectral-weight content probes is
163    /// not yet separately charged; issue #1041 tracks that inherited gap.
164    ///
165    /// # Errors
166    ///
167    /// As [`Self::open`], and whatever `grant` returns.
168    #[cfg(unix)]
169    pub fn open_granted(
170        path: impl AsRef<Path>,
171        options: &OpenOptions,
172        grant: &mut dyn FnMut(usize) -> Result<(), ObservationError>,
173    ) -> Result<Self, ObservationError> {
174        let described = held::opening_bytes(path.as_ref());
175        grant(described)?;
176        let ms = MeasurementSet::open_retained_read(path).map_err(storage)?;
177        Self::open_described_on(ms, options, described, grant)
178    }

180    /// The entities `ms`'s subtables state, read as opening reads them:
181    /// antennas, fields, windows, sources, states, processors and observing
182    /// sessions, and the frame of every data group's UVW that MAIN's `UVW`
183    /// keywords give. Scans and partitions, which MAIN's rows give, are
184    /// empty. How a creation plan states what reopening its MeasurementSet
185    /// gives.
186    ///
187    /// # Errors
188    ///
189    /// As [`Self::open`], for the subtables.
190    pub(crate) fn subtable_catalogue(
191        ms: &MeasurementSet,
192    ) -> Result<(Catalogue, UvwFrame), ObservationError> {
193        let windows = metadata::read_windows(ms)?;
194        let sources = metadata::read_sources(ms, windows.len())?;
195        let catalogue::Fields {
196            entries: fields,
197            uvw_frame,
198            ..
199        } = catalogue::fields_and_uvw(ms, &sources.ids)?;
200        let subtables = Catalogue {
201            antennas: catalogue::read_antennas(ms)?,
202            fields,
203            scans: Vec::new(),
204            observations: catalogue::read_observations(ms)?,
205            partitions: Vec::new(),
206            windows,
207            sources: sources.entries,
208            states: metadata::read_states(ms)?,
209            processors: metadata::read_processors(ms)?,
210        };
211        Ok((subtables, uvw_frame))
212    }

214    /// [`Self::open_granted`] on `ms`, a MeasurementSet already open for a
215    /// retained read ([`MeasurementSet::open_retained_read`]), so that a
216    /// caller who reads MAIN's row count to admit the opening opens it once.
217    /// The caller must admit [`Self::opening_table_bytes`] before opening
218    /// `ms`; this method includes those bytes in each subsequent grant.
219    ///
220    /// # Errors
221    ///
222    /// As [`Self::open_granted`].
223    pub fn open_granted_on(
224        ms: MeasurementSet,
225        options: &OpenOptions,
226        grant: &mut dyn FnMut(usize) -> Result<(), ObservationError>,
227    ) -> Result<Self, ObservationError> {
228        let described = ms.path().map_or(0, held::opening_bytes);
229        grant(described)?;
230        Self::open_described_on(ms, options, described, grant)
231    }

233    fn open_described_on(
234        ms: MeasurementSet,
235        options: &OpenOptions,
236        described: usize,
237        grant: &mut dyn FnMut(usize) -> Result<(), ObservationError>,
238    ) -> Result<Self, ObservationError> {
239        let mut observation =
240            Self::from_measurement_set(ms, options, &mut |total| grant(described + total))?;
241        observation.tables_bytes += described;
242        Ok(observation)
243    }

245    fn from_measurement_set(
246        ms: MeasurementSet,
247        options: &OpenOptions,
248        grant: &mut dyn FnMut(usize) -> Result<(), ObservationError>,
249    ) -> Result<Self, ObservationError> {
250        // The rows the catalogue's readers load stay loaded while the
251        // observation is open; the entities read from them copy at most as
252        // much again.
253        let tables = held::read_bytes(&ms);
254        grant(2 * tables)?;
255        let antennas = catalogue::read_antennas(&ms)?;
256        let feeds = catalogue::read_feeds(&ms)?;
257        if let Some((antenna, feed, window)) = feeds.varying() {
258            tracing::warn!(
259                "FEED states receptors that vary in time (first for antenna {antenna}, feed \
260                 {feed}, window {window}); the last row of each antenna, feed and window is \
261                 used, as casacore's MSIter uses it"
262            );
263        }
264        let windows = metadata::read_windows(&ms)?;
265        let sources = metadata::read_sources(&ms, windows.len())?;
266        let states = metadata::read_states(&ms)?;
267        let processors = metadata::read_processors(&ms)?;
268        let catalogue::Fields {
269            entries: fields,
270            ephemerides,
271            uvw_frame,
272        } = catalogue::fields_and_uvw(&ms, &sources.ids)?;
273        let observations = catalogue::read_observations(&ms)?;
274        let main = ms.main_table();
275        let time_scale = catalogue::time_scale(main)?;
276        let read = 2 * tables + catalogue::setups_bytes(&ms)?;
277        grant(read)?;
278        let setups = catalogue::spectral_setups(&ms)?;
279        let modes = catalogue::read_state_modes(&ms)?;
280        let spectral_sigmas = has_content(main, "SIGMA_SPECTRUM", 0)?;
281        let mut indexed = 0;
282        let index = Index::build(
283            main,
284            options.partition_scheme,
285            (&modes, &ephemerides),
286            ROWS_PER_CHUNK,
287            &mut |total| {
288                grant(read + total)?;
289                indexed = total;
290                Ok(())
291            },
292        )?;
293        let described = (feeds.largest_receptors_bytes(), antennas.len());
294        // The spectral name is longer than the scalar one, so it bounds
295        // either choice without reading each partition's first cell twice.
296        let groups = data_groups(main, true, uvw_frame);
297        let entries_bytes: usize = index
298            .partitions
299            .iter()
300            .map(|partition| entry_bytes(partition, &setups, &groups, described))
301            .sum();
302        grant(read + indexed + entries_bytes)?;
303        // A shape is a hint for an estimate's natural blocks: a header that
304        // cannot be read leaves its group the hint's fallback, as a column
305        // without tiles has, and refuses nothing until the group is read.
306        let tile_shapes = visibility_columns(main)
307            .map(|(_, column)| {
308                let shape = main.array_column_tile_shape(column);
309                (column.to_string(), shape.ok().flatten())
310            })
311            .collect();
312        let mut partitions = Vec::new();
313        let mut entries = Vec::new();
314        for (key, grid) in index.partitions {
315            let data_description_id = u32::try_from(key.data_description)
316                .ok()
317                .filter(|row| (*row as usize) < setups.len())
318                .ok_or_else(|| {
319                    invalid(format!(
320                        "DATA_DESC_ID {} is not in DATA_DESCRIPTION",
321                        key.data_description
322                    ))
323                })?;
324            let setup = &setups[data_description_id as usize];
325            let observation = u32::try_from(key.observation)
326                .ok()
327                .filter(|id| (*id as usize) < observations.len())
328                .map(ObservationId)
329                .ok_or_else(|| {
330                    invalid(format!(
331                        "OBSERVATION_ID {} is not in OBSERVATION",
332                        key.observation
333                    ))
334                })?;
335            let feed = |feed: i32| {
336                u32::try_from(feed).map_err(|_| invalid(format!("negative feed {feed}")))
337            };
338            let receptors = partition_receptors(
339                &feeds,
340                &grid.baselines,
341                key.feeds,
342                setup.spectral_window_id as i32,
343            )?;
344            let feeds = [feed(key.feeds[0])?, feed(key.feeds[1])?];
345            let spectral_weights = has_content(main, "WEIGHT_SPECTRUM", grid.first_time_row())?;
346            // A state or processor a row names that its table does not hold
347            // (CASA's simulator writes PROCESSOR_ID 0 without a row) has no
348            // entry.
349            let known = |id: i32, rows: usize| usize::try_from(id).ok().filter(|id| *id < rows);
350            let partition_states = grid
351                .states
352                .iter()
353                .filter_map(|state| known(*state, states.len()))
354                .map(|state| StateId(state as u32))
355                .collect();
356            let processor = grid
357                .processor
358                .and_then(|processor| known(processor, processors.len()))
359                .map(|processor| ProcessorId(processor as u32));
360            entries.push(PartitionEntry {
361                spectral_window: setup.spectral_window.clone(),
362                frequencies_hz: setup.frequencies_hz.clone(),
363                frequency_frame: setup.frequency_frame,
364                channel_widths_hz: setup.channel_widths_hz.clone(),
365                correlations: setup.correlations.clone(),
366                intents: catalogue::intents(&key.obs_mode),
367                observation,
368                feeds: Some(feeds),
369                fields: super::distinct(&grid.field_per_time),
370                scans: super::distinct(&grid.scan_per_time),
371                baselines: grid.baselines.clone(),
372                time_scale,
373                time_origin: MjdHighPrec::new(0.0, 0.0),
374                data_groups: data_groups(main, spectral_weights, uvw_frame),
375                receptors,
376                window: WindowId(setup.spectral_window_id),
377                states: partition_states,
378                processor,
379            });
380            partitions.push(Partition {
381                grid,
382                obs_mode: key.obs_mode,
383                data_description_id,
384                spectral_window_id: setup.spectral_window_id,
385                spectral_weights,
386            });
387        }
388        Ok(Self {
389            catalogue: Arc::new(Catalogue {
390                antennas,
391                fields,
392                scans: index.scans,
393                observations,
394                partitions: entries,
395                windows,
396                sources: sources.entries,
397                states,
398                processors,
399            }),
400            partitions,
401            state_modes: modes,
402            main_states: index.states,
403            spectral_sigmas,
404            tile_shapes,
405            options: options.clone(),
406            tables_bytes: tables,
407            ms,
408        })
409    }

470    /// The bytes opening a MeasurementSet of `main_rows` MAIN rows holds
471    /// when its grids are full and its times hold a few rows each: a cell
472    /// and eight bytes of time tables per row, and one chunk of key
473    /// columns. Padding cells, which only opening finds, and times of one
474    /// or two rows hold more, which [`Self::open_granted`] asks for before
475    /// it allocates them.
476    #[must_use]
477    pub const fn opening_bytes(main_rows: usize) -> usize {
478        let chunk = if main_rows < index::ROWS_PER_CHUNK {
479            main_rows
480        } else {
481            index::ROWS_PER_CHUNK
482        };
483        main_rows * (size_of::<u32>() + size_of::<f64>()) + chunk * index::CHUNK_ROW_BYTES
484    }

492    /// The tables, index and catalogue retained while this observation is open.
493    #[must_use]
494    pub fn retained_bytes(&self) -> usize {
495        let partitions = self.partitions.capacity() * size_of::<Partition>()
496            + self
497                .partitions
498                .iter()
499                .map(|partition| {
500                    partition.obs_mode.capacity()
501                        + partition.grid.states.capacity() * size_of::<i32>()
502                })
503                .sum::<usize>();
504        let states = super::heap::strings(&self.state_modes) + self.main_states.len() * 128;
505        let tiles = self.tile_shapes.capacity() * size_of::<(String, Option<Vec<usize>>)>()
506            + self
507                .tile_shapes
508                .iter()
509                .map(|(name, shape)| {
510                    name.capacity()
511                        + shape
512                            .as_ref()
513                            .map_or(0, |shape| shape.capacity() * size_of::<usize>())
514                })
515                .sum::<usize>();
516        self.tables_bytes
517            + self.index_bytes()
518            + super::heap::catalogue_bytes(&self.catalogue)
519            + partitions
520            + states
521            + tiles
522    }

536    /// Bytes the opening's index holds while the observation is open: each
537    /// partition's grid, a MAIN row number per cell with its time and
538    /// baseline tables.
539    #[must_use]
540    pub fn index_bytes(&self) -> usize {
541        self.partitions
542            .iter()
543            .map(|partition| {
544                let grid = &partition.grid;
545                grid.cells.capacity() * size_of::<u32>()
546                    + grid.times_s.capacity() * size_of::<f64>()
547                    + grid.field_per_time.capacity() * size_of::<FieldId>()
548                    + grid.scan_per_time.capacity() * size_of::<ScanId>()
549                    + grid.baselines.capacity() * size_of::<OrderedBaseline>()
550            })
551            .sum()
552    }

561    /// The MAIN row partition `id` holds at `time_s` (MAIN `TIME`, seconds
562    /// from MJD 0, as its batches deliver it) on `baseline`; `None` where
563    /// its grid is padded or the partition has no such time or baseline.
564    ///
565    /// # Panics
566    ///
567    /// If `id` is not a partition of the observation.
568    #[must_use]
569    pub fn main_row(&self, id: PartitionId, time_s: f64, baseline: OrderedBaseline) -> Option<u32> {
570        let grid = &self.partitions[id.0 as usize].grid;
571        let time = grid
572            .times_s
573            .binary_search_by(|stored| stored.total_cmp(&time_s))
574            .ok()?;
575        let baseline = grid.baselines.binary_search(&baseline).ok()?;
576        let row = grid.cells[time * grid.baselines.len() + baseline];
577        (row != index::NO_ROW).then_some(row)
578    }

609    /// The `STATE_ID` of each cell of partition `id`'s grid at times
610    /// `times` (positions on it), in its batches' sample order (time, then
611    /// baseline), into `states`; `rows` is scratch. A padded cell takes the
612    /// state of its time's first row (every time holds one), so a row
613    /// written for it keeps its partition. A batch of those times holds
614    /// `states.len()` samples.
615    ///
616    /// # Errors
617    ///
618    /// Times beyond the grid, or a failed read.
619    pub(crate) fn cell_states_into(
620        &self,
621        id: PartitionId,
622        times: Range<usize>,
623        (rows, states): (&mut Vec<usize>, &mut Vec<i32>),
624    ) -> Result<(), ObservationError> {
625        let grid = &self.partitions[id.0 as usize].grid;
626        if times.end > grid.times_s.len() {
627            return Err(invalid(format!(
628                "times {times:?} of a grid of {} times",
629                grid.times_s.len()
630            )));
631        }
632        let cells = grid.cells(times);
633        rows.clear();
634        rows.extend(
635            cells
636                .iter()
637                .filter(|&&row| row != index::NO_ROW)
638                .map(|&row| row as usize),
639        );
640        states.clear();
641        self.ms
642            .main_table()
643            .required_scalar_columns_for_rows_into(
644                rows.as_slice(),
645                &mut [RequiredScalarColumnDestination::new(
646                    "STATE_ID",
647                    RequiredScalarColumnValuesMut::Int32(states),
648                )],
649            )
650            .map_err(table_error)?;
651        // Spread the rows' states over the cells, last first, so each is
652        // read before its place is written.
653        let per_time = grid.baselines.len();
654        let mut read = states.len();
655        states.resize(cells.len(), -1);
656        for (time, cells) in cells.chunks(per_time.max(1)).enumerate().rev() {
657            let held = cells.iter().filter(|row| **row != index::NO_ROW).count();
658            let first = if held == 0 { -1 } else { states[read - held] };
659            for (baseline, row) in cells.iter().enumerate().rev() {
660                states[time * per_time + baseline] = if *row == index::NO_ROW {
661                    first
662                } else {
663                    read -= 1;
664                    states[read]
665                };
666            }
667        }
668        Ok(())
669    }

701    /// What `request` costs on partition `id`, from the grid and the tile
702    /// shape: every MSv2 role is a required column, so binding refuses
703    /// nothing the request can ask for beyond the request's own
704    /// consistency.
705    fn estimate_partition(
706        &self,
707        request: &AccessRequest,
708        id: PartitionId,
709    ) -> Result<(PartitionEstimate, ResolvedRequest), ObservationError> {
710        let (partition, entry) = self.partition(id)?;
711        let resolved = request.resolve(
712            entry,
713            &partition.grid.times_s,
714            &partition.grid.field_per_time,
715        )?;
716        let roles = request.roles;
717        let weights = weight_source(
718            request.weights,
719            &entry.data_groups[resolved.data_group],
720            partition.spectral_weights,
721            self.spectral_sigmas,
722        );
723        let per_channel = weights.per_channel;
724        let channels = request.channels.len();
725        let (delivered, stored) = (
726            resolved.correlations.products(entry).len(),
727            entry.correlations.len(),
728        );
729        let scalars = [roles.time_centroid, roles.interval, roles.exposure]
730            .into_iter()
731            .filter(|wanted| *wanted)
732            .count();
733        let bytes_per_sample = roles.sample_bytes(channels, delivered, per_channel, scalars);
734        let grid = &partition.grid;
735        // The cursor's scratch: each present row's number, sample, FLAG_ROW
736        // and time auxiliaries; and, when a subset is delivered, every stored
737        // product of the cube roles, read before the delivered ones are
738        // gathered. Padding needs none: rows are spread in place.
739        let gathered = Roles {
740            uvw: false,
741            ..roles
742        };
743        let scratch_bytes_per_sample = 2 * size_of::<usize>()
744            + usize::from(roles.flags)
745            + 8 * scalars
746            + if resolved.correlations.indices().is_some() {
747                gathered.sample_bytes(channels, stored, per_channel, 0)
748            } else {
749                0
750            };
751        let per_sample = |correlations: usize| {
752            usize::from(roles.values) * channels * correlations
753                + usize::from(roles.flags) * channels * correlations
754                + usize::from(roles.weights)
755                    * if per_channel {
756                        channels * correlations
757                    } else {
758                        correlations
759                    }
760                + usize::from(roles.uvw) * 3
761                + scalars
762        };
763        let (times, baselines) = (resolved.times.len(), grid.baselines.len());
764        // The selection's rows: its cells that a row fills, counted at
765        // opening for the whole grid.
766        let cells = times * baselines;
767        let rows = if times == grid.times_s.len() {
768            grid.rows
769        } else {
770            resolved
771                .times
772                .runs()
773                .iter()
774                .map(|run| {
775                    grid.cells(run.clone())
776                        .iter()
777                        .filter(|row| **row != index::NO_ROW)
778                        .count()
779                })
780                .sum()
781        };
782        let bytes_per_time = roles.time_bytes(0);
783        let times_per_batch = times_per_batch(
784            request.batch_bytes,
785            baselines * (bytes_per_sample + scratch_bytes_per_sample) + bytes_per_time,
786            times,
787        )?;
788        let group = &entry.data_groups[resolved.data_group];
789        let tile = self
790            .tile_shapes
791            .iter()
792            .find(|(column, _)| *column == group.visibilities)
793            .and_then(|(_, shape)| shape.clone());
794        let estimate = PartitionEstimate {
795            partition: id,
796            times,
797            time_runs: resolved.times.clone(),
798            weights: roles.weights.then_some(weights),
799            baselines,
800            bytes_per_sample,
801            bytes_per_time,
802            scratch_bytes_per_sample,
803            times_per_batch,
804            batches: if times_per_batch == 0 {
805                0
806            } else {
807                resolved.times.blocks(times_per_batch).count()
808            },
809            decode_units: rows,
810            decoded_elements: rows * per_sample(stored),
811            selected_elements: rows * per_sample(delivered),
812            natural_blocks: match tile.as_deref() {
813                // A cell tile is [correlation, channel, rows].
814                Some([_, channels, rows, ..]) => NaturalBlocks {
815                    times: (*rows / baselines.max(1)).max(1),
816                    channels: *channels,
817                },
818                _ => NaturalBlocks {
819                    times: 1,
820                    channels: entry.frequencies_hz.len(),
821                },
822            },
823            padded_fraction: if cells == 0 {
824                0.0
825            } else {
826                1.0 - rows as f64 / cells as f64
827            },
828        };
829        Ok((estimate, resolved))
830    }

832    /// What `request` costs ([`Observation::estimate`]).
833    ///
834    /// # Errors
835    ///
836    /// When the request names an unknown partition, data group, channel or
837    /// correlation product, or budgets less than one time.
838    pub fn estimate(&self, request: &AccessRequest) -> Result<Estimate, ObservationError> {
839        Ok(Estimate {
840            partitions: request
841                .partitions
842                .iter()
843                .map(|&id| Ok(self.estimate_partition(request, id)?.0))
844                .collect::<Result<_, ObservationError>>()?,
845        })
846    }

848    /// Bind a traversal of `request`: one cursor per partition, whole times
849    /// per batch ([`Observation::bind_read`]).
850    ///
851    /// # Errors
852    ///
853    /// As [`Self::estimate`].
854    pub fn bind_read(&self, request: &AccessRequest) -> Result<Traversal<'_>, ObservationError> {
855        let bound = request
856            .partitions
857            .iter()
858            .map(|&id| self.estimate_partition(request, id))
859            .collect::<Result<Vec<_>, _>>()?;
860        let estimate = Estimate {
861            partitions: bound.iter().map(|(estimate, _)| estimate.clone()).collect(),
862        };
863        let cursors = bound
864            .into_iter()
865            .map(|(estimate, resolved)| {
866                Msv2Cursor::bind(self, request, estimate, resolved)
867                    .map(|cursor| Box::new(cursor) as BoundCursor<'_>)
868            })
869            .collect::<Result<_, _>>()?;
870        Ok(Traversal::new(cursors, estimate))
871    }

873    /// Partition `id`'s grid ([`Observation::grid`]): its times, fields and
874    /// scans from the opening index, and the one `INTERVAL` of its rows,
875    /// read in chunks of at most 262,144 rows. Padding has no
876    /// row, so no interval.
877    ///
878    /// # Errors
879    ///
880    /// As [`Observation::grid`].
881    pub fn grid(
882        &self,
883        id: PartitionId,
884        data_group: &str,
885    ) -> Result<TimeBaselineGrid, ObservationError> {
886        let (partition, entry) = self.partition(id)?;
887        if entry.data_group(data_group).is_none() {
888            return Err(ObservationError::InvalidRequest {
889                reason: format!("no data group {data_group} in {id:?}"),
890            });
891        }
892        let grid = &partition.grid;
893        let mut rows = grid
894            .cells
895            .iter()
896            .filter(|&&row| row != index::NO_ROW)
897            .map(|&row| row as usize);
898        let main = self.ms.main_table();
899        let mut interval = None;
900        let mut chunk = Vec::with_capacity(GRID_READ_ROWS.min(grid.cells.len()));
901        let mut values = Vec::with_capacity(chunk.capacity());
902        loop {
903            chunk.clear();
904            chunk.extend(rows.by_ref().take(GRID_READ_ROWS));
905            if chunk.is_empty() {
906                break;
907            }
908            values.clear();
909            main.required_scalar_columns_for_rows_into(
910                &chunk,
911                &mut [RequiredScalarColumnDestination::new(
912                    "INTERVAL",
913                    RequiredScalarColumnValuesMut::Float64(&mut values),
914                )],
915            )
916            .map_err(table_error)?;
917            for value in &values {
918                if *interval.get_or_insert(*value) != *value {
919                    return Err(ObservationError::Unsupported {
920                        what: "a grid with more than one integration interval",
921                    });
922                }
923            }
924        }
925        Ok(TimeBaselineGrid {
926            baselines: grid.baselines.clone(),
927            time_scale: entry.time_scale,
928            time_origin: entry.time_origin,
929            times_s: grid.times_s.clone(),
930            field_per_time: grid.field_per_time.clone(),
931            scan_per_time: grid.scan_per_time.clone(),
932            interval_s: interval.ok_or_else(|| ObservationError::InvalidRequest {
933                reason: format!("{id:?} has no rows"),
934            })?,
935        })
936    }
937}

939/// The most partition `(key, grid)`'s catalogue entry holds before it is
940/// built: its setup's copy, its intents (the parts of its `OBS_MODE`), its
941/// fields and scans (at most one of each per time), baselines, states and
942/// data groups (`groups`), and the receptors of each antenna its baselines
943/// name, `antennas` at most, each at most `receptors` bytes.
944fn entry_bytes(
945    (key, grid): &(index::PartitionKey, index::Grid),
946    setups: &[catalogue::SpectralSetup],
947    groups: &[DataGroupEntry],
948    (receptors, antennas): (usize, usize),
949) -> usize {
950    let setup = usize::try_from(key.data_description)
951        .ok()
952        .and_then(|row| setups.get(row))
953        .map_or(0, catalogue::SpectralSetup::heap_bytes);
954    let intents =
955        2 * key.obs_mode.len() + (key.obs_mode.matches(',').count() + 1) * size_of::<String>();
956    let groups: usize = groups
957        .iter()
958        .map(|group| size_of::<DataGroupEntry>() + heap::group_bytes(group))
959        .sum();
960    let receptor_count = (2 * grid.baselines.len()).min(antennas);
961    size_of::<PartitionEntry>()
962        + setup
963        + intents
964        + (grid.field_per_time.len() + grid.scan_per_time.len()) * size_of::<u32>()
965        + size_of_val(grid.baselines.as_slice())
966        + grid.states.len() * size_of::<StateId>()
967        + groups
968        + casa_types::memory::btree_node_bytes::<AntennaId, Receptors>(receptor_count)
969        + receptor_count * receptors
970}

972/// The receptors of each antenna `baselines` name, through the feed pair
973/// `feeds` (`FEED1` for the first antenna, `FEED2` for the second) in
974/// spectral window `window`; an antenna named through two feeds must have
975/// the same receptors through both.
976fn partition_receptors(
977    feeds: &catalogue::Feeds,
978    baselines: &[OrderedBaseline],
979    pair: [i32; 2],
980    window: i32,
981) -> Result<BTreeMap<AntennaId, Receptors>, ObservationError> {
982    let mut receptors = BTreeMap::new();
983    for baseline in baselines {
984        for (antenna, feed) in [(baseline.antenna1, pair[0]), (baseline.antenna2, pair[1])] {
985            let id = i32::try_from(antenna.0).map_err(|_| invalid("an antenna beyond i32"))?;
986            let Some(own) = feeds.receptors(id, feed, window) else {
987                continue;
988            };
989            if receptors
990                .insert(antenna, own.clone())
991                .is_some_and(|known| &known != own)
992            {
993                return Err(ObservationError::Unsupported {
994                    what: "an MSv2 antenna with different receptors through its two feeds",
995                });
996            }
997        }
998    }
999    Ok(receptors)
1000}

1052/// What the weights role of data group `group` delivers under `request`,
1053/// on a MAIN with or without `WEIGHT_SPECTRUM` and `SIGMA_SPECTRUM`: the
1054/// group's stored weights, or 1/σ² from `SIGMA_SPECTRUM` when MAIN has both
1055/// it and `WEIGHT_SPECTRUM` (CASA's rule, `VisBufferImpl2`), else from
1056/// `SIGMA`. Every MSv2 has `SIGMA`, so no fallback is taken.
1057fn weight_source(
1058    request: WeightRequest,
1059    group: &DataGroupEntry,
1060    spectral_weights: bool,
1061    spectral_sigmas: bool,
1062) -> WeightSource {
1063    match request {
1064        WeightRequest::Stored => WeightSource {
1065            rule: WeightRule::Stored,
1066            variable: group.weights.clone(),
1067            per_channel: spectral_weights,
1068        },
1069        WeightRequest::InverseSigmaSquared { .. } => {
1070            let per_channel = spectral_weights && spectral_sigmas;
1071            WeightSource {
1072                rule: WeightRule::InverseSigmaSquared,
1073                variable: if per_channel {
1074                    "SIGMA_SPECTRUM"
1075                } else {
1076                    "SIGMA"
1077                }
1078                .to_string(),
1079                per_channel,
1080            }
1081        }
1082    }
1083}

1085/// Whether MAIN's `column` holds content at `row`, as casacore's
1086/// `TableColumn::hasContent` tells, by which CASA's `VisibilityIteratorImpl2`
1087/// picks `WEIGHT_SPECTRUM` and `SIGMA_SPECTRUM` over `WEIGHT` and `SIGMA`
1088/// (`weightSpectrumExists`, `sigmaSpectrumExists`): the column is there,
1089/// the cell is defined, and no axis of it is empty. CASA asks at the first
1090/// row of the rows it iterates: for `WEIGHT_SPECTRUM` again at each new
1091/// spectral window, for `SIGMA_SPECTRUM` once per MeasurementSet. A column
1092/// added and never filled holds none.
1093fn has_content(
1094    main: &casa_tables::Table,
1095    column: &str,
1096    row: usize,
1097) -> Result<bool, ObservationError> {
1098    if row >= main.row_count()
1099        || !main
1100            .schema()
1101            .is_some_and(|schema| schema.contains_column(column))
1102    {
1103        return Ok(false);
1104    }
1105    // One cell, read without the row cache the opening's lazy fills need
1106    // left empty. A cell this reader cannot decode (a storage manager it
1107    // lacks) is not proof of no content: the column is taken as holding
1108    // content, and reading it reports the error, as before.
1109    let Ok(first) = main
1110        .column_accessor(column)
1111        .and_then(|cells| cells.array_cells_owned_uncached(&[row]))
1112    else {
1113        return Ok(true);
1114    };
1115    Ok(first.into_iter().next().flatten().is_some_and(|cell| {
1116        let shape = cell.shape();
1117        !shape.is_empty() && shape.iter().all(|axis| *axis > 0)
1118    }))
1119}
