2//! What a run's setup decides before its first pass: how many workers its
3//! phases use (#737), its imaging weights and its paged cube state.
4//!
5//! The workers are the most, up to the resource policy's, whose memory fits
6//! beside what the run already holds. What grows with them is charged as
7//! admission then charges it:
8//! - the AW kernel sets' cells per holder;
9//! - the imaging-weights pass;
10//! - the paged cube state's cache;
11//! - the region row-work pass, and the gridding work by anchor row it
12//!   keeps (#736);
13//! - the first major-cycle pass at one plane per wave, on the region strips
14//!   that hold the most of any it may cut from that work;
15//! - a minor cycle, or a refresh pass, each beside what the run keeps then:
16//!   the normal state and model, the imaging weights it kept, the masks, a
17//!   cached Clark refresh, the minor cycle's terms and the record of every
18//!   cycle before it, up to `nmajor` (#763 review). A run with no `nmajor`
19//!   is covered to its first refresh: its records grow with each cycle after
20//!   that.
21//! - the pass that writes visibilities, at every plane: the first pass of a
22//!   run that does not clean, or else a refresh pass (#778 review).
23//!
24//! A minor cycle never overlaps the first pass: its masks are admitted, and
25//! its work begins, after the pass's wave is released. So it is charged once,
26//! beside what the run keeps, not on top of the pass
27//! ([`StageCharges::largest`]).
28//!
29//! The count is chosen once, from those charges, and each phase is then
30//! admitted as before. A run that no count fits takes one worker, which
31//! admission refuses. The products, whose scratch grows with their lanes,
32//! choose their lanes when they start, from their own demand
33//! (`native::publish_products`).
34

54/// The workers a run chose and, when fewer than its policy's, why.
55#[derive(Clone, Debug, PartialEq, Eq)]
56pub(crate) struct WorkerChoice {
57    /// The allocation the run uses.
58    pub(crate) cores: CoreAllocation,
59    /// The policy's allocation, the most the run may use.
60    pub(crate) policy: CoreAllocation,
61    /// Why `cores` has fewer workers than `policy`; `None` when it has as
62    /// many.
63    pub(crate) reason: Option<String>,
64}
65
66/// The allocation of the most workers, up to `policy`'s, whose `demand`
67/// fits in `free` bytes, trying `policy`'s first and then each smaller
68/// count; one worker when none fits, which admission then refuses.
69pub(super) fn fit_workers<E>(
70    policy: CoreAllocation,
71    free: u64,
72    mut demand: impl FnMut(CoreAllocation) -> Result<u64, E>,
73) -> Result<WorkerChoice, E> {
74    let gigabytes = |bytes: u64| format!("{:.2} GB", bytes as f64 / 1e9);
75    let needed = demand(policy)?;
76    let choice = |cores, reason| WorkerChoice {
77        cores,
78        policy,
79        reason,
80    };
81    if needed <= free {
82        return Ok(choice(policy, None));
83    }
84    for workers in (1..policy.workers).rev() {
85        let cores = policy.at_most(workers);
86        if demand(cores)? <= free {
87            let reason = format!(
88                "{} workers need {}, {workers} fit {} free",
89                policy.workers,
90                gigabytes(needed),
91                gigabytes(free)
92            );
93            return Ok(choice(cores, Some(reason)));
94        }
95    }
96    let reason = format!(
97        "{} workers need {} and no fewer fit {} free",
98        policy.workers,
99        gigabytes(needed),
100        gigabytes(free)
101    );
102    Ok(choice(
103        policy.at_most(1),
104        (policy.workers > 1).then_some(reason),
105    ))
106}

108/// The charges of a run's setup and first cycles that grow with its
109/// workers, from the operators and source the run has opened.
110pub(super) struct SetupDemand<'r> {
111    pub(super) problem: &'r CompiledProblem,
112    pub(super) domains: &'r [DomainOperator],
113    pub(super) stitch: &'r ChartStitch,
114    /// The source's largest block.
115    pub(super) block: BlockShape,
116    pub(super) native_spacing_hz: f64,
117    pub(super) backend: BackendChoice,
118    /// The initial pass's modes.
119    pub(super) modes: ModeSet,
120    /// What the initial pass also holds: the model.
121    pub(super) model_bytes: u64,
122    /// With a start model, what seeding the model holds beside the pass
123    /// that seeds it (`StartModel::seeding_bytes`, with its conversion when
124    /// the run reseeds); the pass then grids its residual, with the model.
125    pub(super) start_model: Option<u64>,
126    /// Whether the start model is made physical with the initial pass's
127    /// weights and seeds a refresh of its residual (`cycle::seeding`): the
128    /// initial pass then has no model, and the refresh pass seeds.
129    pub(super) reseeds: bool,
130    /// The modes and model of the pass that writes visibilities, and what
131    /// the write holds; it holds every plane.
132    pub(super) write: Option<(ModeSet, bool, u64)>,
133    pub(super) spill_directory: &'r Path,
134    /// Whether the cube state keeps a weight image per plane.
135    pub(super) weight_image: bool,
136    /// What a minor cycle holds on a team of so many workers.
137    pub(super) minor: &'r dyn Fn(usize) -> u64,
138    /// What a refresh pass holds beside its own wave; `None` for a run
139    /// that does not clean.
140    pub(super) refresh: Option<RefreshDemand>,
141    /// The imaging weights the passes keep across passes, when the run
142    /// keeps them (`Run::keep_weights`).
143    pub(super) kept_weights: u64,
144    /// What the products hold on one lane
145    /// (`PlannedContinuumGeneration::planned_demand`): they run after the
146    /// passes, beside the completion they publish.
147    pub(super) products: u64,
148    /// The source's admitted charge, which the run releases before its
149    /// products.
150    pub(super) source: u64,
151    /// The memory the policy leaves free before the setup's charges.
152    pub(super) free: u64,
153}

155/// What the run holds beside a refresh pass's own wave, from the layout
156/// the first pass forms (#763 review): the minor cycle's masks, a cached
157/// Clark refresh, the model terms the minor cycle hands the pass, and the
158/// record of every minor cycle before it, which the run keeps.
159pub(super) struct RefreshDemand {
160    /// The latest masks and the ones an automatic mask evolves from.
161    pub(super) masks: u64,
162    /// The Clark refresh a single-plane Clark run caches.
163    pub(super) cached: u64,
164    /// The minor cycle's model terms, which its solve admits while it still
165    /// holds its own charge and the refresh pass holds until the model
166    /// takes them (#778 review).
167    pub(super) results: u64,
168    /// The records of the cycles up to the last refresh `nmajor` allows,
169    /// or the first when it allows any number.
170    pub(super) records: u64,
171}

173impl SetupDemand<'_> {
174    /// The bytes the run holds at most on `cores`: the kernel sets and the
175    /// cube state throughout, beside the imaging-weights pass or, after it,
176    /// its density grid with the region row-work pass, and then with the
177    /// work and imaging weights it keeps, the first major-cycle pass beside
178    /// the model, and after it a minor cycle or a refresh pass beside the
179    /// normal state and model and [`RefreshDemand`]
180    /// ([`StageCharges::largest`]); or, if more, the products
181    /// ([`Self::products_bytes`]).
182    pub(super) fn bytes(&self, cores: CoreAllocation) -> Result<u64, ImagingError> {
183        let (passes, products) = self.stages(cores)?;
184        Ok(passes.max(products))
185    }
186
187    /// What the products hold beside the run on `cores`, net of the
188    /// source's charge, which the run releases before them: the products on
189    /// one lane, the completion they publish (the normal state and model,
190    /// and a cube's paged state with its cache), and the masks and
191    /// minor-cycle records the run keeps. The kernel sets are released with
192    /// the passes.
193    pub(super) fn products_bytes(&self, cores: CoreAllocation) -> Result<u64, ImagingError> {
194        Ok(self.stages(cores)?.1)
195    }
196
197    /// [`Self::bytes`]' two parts: the setup and cycles, and the products.
198    fn stages(&self, cores: CoreAllocation) -> Result<(u64, u64), ImagingError> {
199        let streaming = cores.streaming_workers;
200        let kernels = self
201            .domains
202            .iter()
203            .map(|domain| domain.pending_kernel_bytes(streaming))
204            .sum::<u64>();
205        let (weighing, weights) =
206            weights_bytes(self.problem, &self.domains[0], self.block, streaming)?;
207        let cube = if matches!(
208            self.problem.reconstruction().basis(),
209            ReconstructionBasis::ChannelLocal { .. }
210        ) {
211            cube_cache_bytes(
212                self.problem,
213                self.spill_directory,
214                (cores.workers, self.weight_image),
215                self.free.saturating_sub(kernels + weights),
216            )? as u64
217        } else {
218            0
219        };
220        // The first pass cuts its strips from the row work measured before
221        // it; no cut holds more than the widest strips (#736).
222        let domains = pass_domains(self.domains, streaming, None)
223            .into_iter()
224            .map(|domain| match domain.partition {
225                Partition::Regions { ref regions, .. } => PassDomain {
226                    partition: Partition::regions_widest(domain.operator, regions.len()),
227                    ..domain
228                },
229                Partition::Planes { .. } => domain,
230            })
231            .collect::<Vec<_>>();
232        let (measuring, measured) = if needs_row_work(&domains) {
233            let rows = domains
234                .iter()
235                .filter(|domain| domain.partition.splits_rows())
236                .map(|domain| domain.operator.geometry().grid_shape()[1])
237                .sum::<usize>();
238            (
239                row_work_pass_bytes(&domains, streaming, self.weight_image),
240                MeasuredRowWork::kept_bytes(rows, domains.len()),
241            )
242        } else {
243            (0, 0)
244        };
245        let wave = |modes: ModeSet, with_model| WaveDemand {
246            domains: &domains,
247            modes,
248            with_model,
249            native_spacing_hz: self.native_spacing_hz,
250            workers: streaming,
251            backend: self.backend,
252            block: self.block,
253            chunk_placements: CHUNK_PLACEMENTS,
254            chunk_sets: 1,
255        };
256        let refreshing = wave(ModeSet::DATA, true);
257        // The pass that seeds holds the seeding beside its wave: the initial
258        // pass, or the refresh after it when the run reseeds.
259        let seeding = self.start_model.unwrap_or(0);
260        let (first_seeding, refresh_seeding) = if self.reseeds {
261            (0, seeding)
262        } else {
263            (seeding, 0)
264        };
265        let mut first = wave(self.modes, self.start_model.is_some() && !self.reseeds).bytes(1)
266            + self.stitch.pass_bytes(true)
267            + first_seeding;
268        let mut refresh = refreshing.bytes(1) + self.stitch.pass_bytes(false) + refresh_seeding;
269        // A run that cleans writes on its last refresh, or on a prediction
270        // after it; one that does not, on its first major cycle
271        // (`write_pass`), beside the seeding (#970 review). The pass that
272        // writes holds every plane (#778 review).
273        if let Some((modes, with_model, written)) = self.write {
274            let demand = wave(modes, with_model);
275            let writing =
276                demand.bytes(demand.planes()) + self.stitch.pass_bytes(modes.psf) + written;
277            let cleans = self
278                .problem
279                .reconstruction()
280                .controls()
281                .max_minor_iterations()
282                > 0;
283            let beside = if cleans { 0 } else { seeding };
284            if self.refresh.is_some() {
285                refresh = refresh.max(writing + if self.reseeds { beside } else { 0 });
286            } else {
287                first = first.max(writing + beside);
288            }
289        }
290        let stages = StageCharges {
291            weighing,
292            weights,
293            measuring,
294            kept: weights + measured + self.kept_weights,
295            first_pass: first + self.model_bytes,
296            minor: (self.minor)(cores.workers),
297            after: self.refresh.as_ref().map(|demand| AfterFirstPass {
298                normal: refreshing.resident_normal_bytes(),
299                model: self.model_bytes,
300                cached: demand.cached,
301                records: demand.records,
302                results: demand.results,
303                refresh_pass: demand.masks + refresh,
304            }),
305        };
306        let kept = self
307            .refresh
308            .as_ref()
309            .map_or(0, |demand| demand.masks + demand.records);
310        let products = (cube + refreshing.resident_normal_bytes() + self.model_bytes + kept)
311            .saturating_add(self.products)
312            .saturating_sub(self.source);
313        Ok((kernels + cube + stages.largest(), products))
314    }
315}

317/// The charges of a run's sequential stages beside its kernels and cube
318/// state ([`Self::largest`]).
319#[derive(Clone, Copy, Debug)]
320struct StageCharges {
321    /// The imaging-weights pass.
322    weighing: u64,
323    /// The density grid the weights keep.
324    weights: u64,
325    /// The region row-work pass.
326    measuring: u64,
327    /// What the run keeps from the first pass on: the density grid, the
328    /// row work and the imaging weights.
329    kept: u64,
330    /// The first major-cycle pass, its stitched images and the model.
331    first_pass: u64,
332    /// A minor cycle: its masks and the larger of its two halves.
333    minor: u64,
334    /// What follows the first pass in a run that cleans; `None` otherwise.
335    after: Option<AfterFirstPass>,
336}
337
338/// What a cleaning run holds after its first pass besides
339/// [`StageCharges::kept`] and either a minor cycle or a refresh pass.
340#[derive(Clone, Copy, Debug)]
341struct AfterFirstPass {
342    /// The resident normal state.
343    normal: u64,
344    /// The model.
345    model: u64,
346    /// A cached Clark refresh.
347    cached: u64,
348    /// The records of the cycles up to the last refresh `nmajor` allows.
349    records: u64,
350    /// The minor cycle's model terms, which its solve and then the refresh
351    /// pass hold.
352    results: u64,
353    /// A refresh pass, or the one that writes visibilities, with the masks.
354    refresh_pass: u64,
355}
356
357impl StageCharges {
358    /// The largest stage, which is what the run holds at most: the
359    /// imaging-weights pass; the density grid beside the row-work pass; the
360    /// first pass beside what it keeps; and after it, a minor cycle or a
361    /// refresh pass beside the kept charges, the normal state and model, a
362    /// cached refresh, the records and the minor cycle's model terms.
363    ///
364    /// A minor cycle is charged once, after the first pass. It never runs
365    /// during one: its masks are admitted and its work begins once the
366    /// pass's wave is released. #763 charged it on top of the first pass
367    /// too, which put B3 on one worker where its phases admit three.
368    fn largest(&self) -> u64 {
369        let after = self.after.map_or(0, |after| {
370            self.kept
371                + after.normal
372                + after.model
373                + after.cached
374                + after.records
375                + after.results
376                + after.refresh_pass.max(self.minor)
377        });
378        self.weighing
379            .max(self.weights + self.measuring)
380            .max(self.kept + self.first_pass)
381            .max(after)
382    }
383}

385/// The imaging weights' density pass on `workers` and the density grid it
386/// keeps, in bytes; none for natural weighting.
387fn weights_bytes(
388    problem: &CompiledProblem,
389    main: &DomainOperator,
390    block: BlockShape,
391    workers: usize,
392) -> Result<(u64, u64), ImagingError> {
393    if matches!(problem.weighting().scheme(), WeightingScheme::Natural) {
394        return Ok((0, 0));
395    }
396    let shape = density_shape(problem, &problem.geometry().domains()[0])?;
397    Ok((
398        density_pass_bytes(&main.resampler, &shape, block, workers),
399        shape.bytes(),
400    ))
401}

403/// CASA's imaging weights for the run: the input weights (natural) or the
404/// density weights of one traversal on the main domain's cells, which every
405/// domain shares (`SynthesisImagerVi2::weight`), with the charge of the
406/// density grid the weights keep. The density pass is admitted under the
407/// run's `(host, policy)` before it allocates.
408pub(super) fn weighting(
409    problem: &CompiledProblem,
410    main: &DomainOperator,
411    source: &mut dyn BoundedSource,
412    team: &WorkerTeam,
413    cancel: &Cancel,
414    (host, policy): (&HostResources, &ResourcePolicy),
415) -> Result<(WeightingGeneration, Reservation), ImagingError> {
416    let contract = problem.weighting();
417    let taper = contract.uv_taper();
418    let (robust, bandwidth) = match contract.scheme() {
419        WeightingScheme::Natural => {
420            return Ok((WeightingGeneration::Natural { taper }, Reservation::none()));
421        }
422        WeightingScheme::Uniform => (None, None),
423        WeightingScheme::Briggs { robust } => (Some(robust), None),
424        WeightingScheme::BriggsBandwidthTaper { robust } => {
425            let spectral = problem.geometry().spectral();
426            let last = spectral.output_channels().saturating_sub(1);
427            let (Some(first_hz), Some(last_hz)) = (
428                spectral.channel_centre_hz(0),
429                spectral.channel_centre_hz(last),
430            ) else {
431                return Err(ImagingError::Unsupported {
432                    reason: "the image spectral axis has no channels",
433                });
434            };
435            (
436                Some(robust),
437                Some(BandwidthTaper::from_frequency_range(first_hz, last_hz)?),
438            )
439        }
440    };
441    let shape = density_shape(problem, &problem.geometry().domains()[0])?;
442    let (pass, grid) = weights_bytes(problem, main, source.maximum_block(), team.workers())?;
443    let mut weights = admit(
444        host,
445        policy,
446        &Demand {
447            phase: "imaging weights",
448            memory: pass,
449        },
450    )?;
451    let grid_values =
452        run_density_pass(&main.operator, &main.resampler, shape, source, team, cancel)?;
453    weights.retain(grid);
454    let weighting = WeightingGeneration::density(grid_values, robust, bandwidth, taper)?;
455    Ok((weighting, weights))
456}

458/// The cache of the paged cube state on `workers`: as large as the whole
459/// cube when a quarter of `free` allows, and at least what the workers'
460/// planes need. With a weight image (a mosaic cube) each plane keeps its own
461/// beside the PSF.
462fn cube_cache_bytes(
463    problem: &CompiledProblem,
464    directory: &Path,
465    (workers, weight_image): (usize, bool),
466    free: u64,
467) -> Result<usize, ImagingError> {
468    let target = problem.model_lifecycle().target();
469    let domains = target
470        .domains()
471        .iter()
472        .map(|domain| domain.pixels())
473        .collect::<Vec<_>>();
474    let planes = target.coefficients() * target.polarizations();
475    let (minimum, full) =
476        CubeState::cache_limits(directory, &domains, planes, weight_image, workers)?;
477    let quarter = usize::try_from(free / 4).unwrap_or(usize::MAX);
478    Ok(full.min(minimum.max(quarter)))
479}

481/// The paged cube state of a channel-local run, its cache admitted as
482/// [`cube_cache_bytes`] sizes it from what the policy leaves free; the state
483/// holds the cache's reservation for as long as its storage lives.
484pub(super) fn cube_state(
485    problem: &CompiledProblem,
486    directory: &Path,
487    (workers, weight_image): (usize, bool),
488    host: &HostResources,
489    policy: &ResourcePolicy,
490) -> Result<CubeState, ImagingError> {
491    let cache = cube_cache_bytes(
492        problem,
493        directory,
494        (workers, weight_image),
495        free_memory(host, policy),
496    )?;
497    let reservation = admit(
498        host,
499        policy,
500        &Demand {
501            phase: "paged cube state",
502            memory: cache as u64,
503        },
504    )?;
505    let target = problem.model_lifecycle().target();
506    let domains = target
507        .domains()
508        .iter()
509        .map(|domain| domain.pixels())
510        .collect::<Vec<_>>();
511    Ok(CubeState::new(
512        directory,
513        &domains,
514        target.coefficients() * target.polarizations(),
515        weight_image,
516        reservation,
517    )?)
518}

578    /// A minor cycle is charged once, after the first pass, beside what the

612        assert!(
613            b3.largest() < 176 + 5_952 + 3_897,
614            "not the pass and the minor cycle"
615        );
616        // Whatever the first pass, the minor cycle and its terms count.
617        let alone = StageCharges {
618            first_pass: 1,
619            ..b3
620        };
621        assert_eq!(alone.largest(), following);
622        let larger = StageCharges {
623            minor: 3_997,
624            ..alone
625        };
626        assert_eq!(larger.largest(), following + 100);
627        let termless = StageCharges {
628            after: Some(AfterFirstPass {
629                results: 0,
630                ..after
631            }),
632            ..alone
633        };
634        assert_eq!(termless.largest(), following - 490);
635        let dirty = StageCharges { after: None, ..b3 };
636        assert_eq!(dirty.largest(), 176 + 5_952);
637        let refreshing = StageCharges {
638            minor: 1,
639            first_pass: 1,
640            ..b3
641        };
642        assert_eq!(refreshing.largest(), 176 + 1_745 + 537 + 17 + 490 + 2_000);
643    }
