25impl PyImageTool {
26    /// Fit Gaussian, Lorentzian and continuum models along an image axis.
27    #[allow(clippy::too_many_arguments)]
28    #[pyo3(signature = (r#box="", region=None, chans="", stokes="", axis=-1, mask=None, ngauss=1, poly=-1, estimates="", minpts=1, multifit=false, model="", residual="", amp="", amperr="", center="", centererr="", fwhm="", fwhmerr="", integral="", integralerr="", stretch=false, logresults=true, pampest=DoubleList::of(&[]), pcenterest=DoubleList::of(&[]), pfwhmest=DoubleList::of(&[]), pfix=StringList::of(&[]), gmncomps=IntList::of(&[]), gmampcon=DoubleList::of(&[]), gmcentercon=DoubleList::of(&[]), gmfwhmcon=DoubleList::of(&[]), gmampest=DoubleList::of(&[0.]), gmcenterest=DoubleList::of(&[0.]), gmfwhmest=DoubleList::of(&[0.]), gmfix=StringList::of(&[]), spxtype="", spxest=DoubleList::of(&[]), spxfix=None, div=None, spxsol="", spxerr="", logfile="", append=true, pfunc=StringList::of(&[]), goodamprange=DoubleList::of(&[0.]), goodcenterrange=DoubleList::of(&[0.]), goodfwhmrange=DoubleList::of(&[0.]), sigma=None, outsigma="", planes=IntList::of(&[])), text_signature = "($self, box='', region='', chans='', stokes='', axis=-1, mask='', ngauss=1, poly=-1, estimates='', minpts=1, multifit=False, model='', residual='', amp='', amperr='', center='', centererr='', fwhm='', fwhmerr='', integral='', integralerr='', stretch=False, logresults=True, pampest=[], pcenterest=[], pfwhmest=[], pfix='', gmncomps=0, gmampcon=[], gmcentercon=[], gmfwhmcon=[], gmampest=[0.0], gmcenterest=[0.0], gmfwhmest=[0.0], gmfix='', spxtype='', spxest=[], spxfix=[], div=[], spxsol='', spxerr='', logfile='', append=True, pfunc='', goodamprange=[0.0], goodcenterrange=[0.0], goodfwhmrange=[0.0], sigma='', outsigma='', planes=[])")]
29    fn fitprofile<'py>(
30        &self,
31        py: Python<'py>,
32        r#box: &str,
33        region: Option<&Bound<'py, PyAny>>,
34        chans: &str,
35        stokes: &str,
36        axis: i64,
37        mask: Option<&Bound<'py, PyAny>>,
38        ngauss: i64,
39        poly: i64,
40        estimates: &str,
41        minpts: i64,
42        #[pyo3(from_py_with = require_bool)] multifit: bool,
43        model: &str,
44        residual: &str,
45        amp: &str,
46        amperr: &str,
47        center: &str,
48        centererr: &str,
49        fwhm: &str,
50        fwhmerr: &str,
51        integral: &str,
52        integralerr: &str,
53        #[pyo3(from_py_with = require_bool)] stretch: bool,
54        #[pyo3(from_py_with = require_bool)] logresults: bool,
55        pampest: DoubleList,
56        pcenterest: DoubleList,
57        pfwhmest: DoubleList,
58        pfix: StringList,
59        gmncomps: IntList,
60        gmampcon: DoubleList,
61        gmcentercon: DoubleList,
62        gmfwhmcon: DoubleList,
63        gmampest: DoubleList,
64        gmcenterest: DoubleList,
65        gmfwhmest: DoubleList,
66        gmfix: StringList,
67        spxtype: &str,
68        spxest: DoubleList,
69        spxfix: Option<Vec<bool>>,
70        div: Option<Variant>,
71        spxsol: &str,
72        spxerr: &str,
73        logfile: &str,
74        #[pyo3(from_py_with = require_bool)] append: bool,
75        pfunc: StringList,
76        goodamprange: DoubleList,
77        goodcenterrange: DoubleList,
78        goodfwhmrange: DoubleList,
79        sigma: Option<Variant>,
80        outsigma: &str,
81        planes: IntList,
82    ) -> PyResult<Py<PyAny>> {
83        let attached = py
84            .detach(|| {
85                self.state()
86                    .image
87                    .as_ref()
88                    .map(check_profile_fit_type)
89                    .transpose()
90            })
91            .map_err(raised)?;
92        if attached.is_none() {
93            return record_to_py(py, RecordValue::default());
94        }
95        let Some(sigma) = sigma_argument(sigma)? else {
96            return record_to_py(py, RecordValue::default());
97        };
98        let region_arg: RegionArgument =
99            region.map_or_else(|| Ok(RegionArgument::default()), |arg| arg.extract())?;
100        let mask = mask_argument(mask)?;
101        let good_ranges = [
102            range("goodamprange", &goodamprange)?,
103            range("goodcenterrange", &goodcenterrange)?,
104            range("goodfwhmrange", &goodfwhmrange)?,
105        ];
106        let divisor = if !spxtype.is_empty() && !spxest.0.is_empty() {
107            divisor_argument(div)?
108        } else {
109            ProfileDivisor::Automatic
110        };
111        let log_details = logresults
112            && logger(py, "casars.tools.image")?
113                .bind(py)
114                .call_method1("isEnabledFor", (20,))?
115                .extract()?;
116        let region_history = region_arg.history();
117        let (record, logs) = py.detach(|| -> PyResult<_> {
118            let state = self.state();
119            let Some(source) = &state.image else {
120                return Ok((RecordValue::default(), Vec::new()));
121            };
122            let request = FitProfileRequest {
123                selection: [r#box, chans, stokes],
124                region: region_arg.selection(),
125                mask: &mask,
126                axis,
127                ngauss,
128                poly,
129                min_points: minpts,
130                estimates_file: estimates,
131                multifit,
132                stretch,
133                estimates: ProfileEstimates {
134                    singlets: peaks(&pampest, &pcenterest, &pfwhmest, &pfix),
135                    functions: list_or_empty(&pfunc),
136                    multiplets: peaks(&gmampest, &gmcenterest, &gmfwhmest, &gmfix),
137                    multiplet_sizes: if gmncomps.0 == [0] { &[] } else { &gmncomps.0 },
138                    multiplet_ties: [&gmampcon.0, &gmcentercon.0, &gmfwhmcon.0],
139                    spectral_kind: spxtype,
140                    spectral_values: &spxest.0,
141                    spectral_fixed: spxfix.as_deref().unwrap_or_default(),
142                },
143                good_ranges,
144                planes: &planes.0,
145                divisor,
146                sigma: match &sigma {
147                    SigmaArgument::None => ProfileSigmaInput::None,
148                    SigmaArgument::Values(v) => ProfileSigmaInput::Values(v),
149                    SigmaArgument::Path(p) => ProfileSigmaInput::Path(p),
150                },
151                outputs: ProfileOutputNames {
152                    spectra: [model, residual],
153                    lines: [
154                        amp,
155                        amperr,
156                        center,
157                        centererr,
158                        fwhm,
159                        fwhmerr,
160                        integral,
161                        integralerr,
162                    ],
163                    coefficients: [spxsol, spxerr],
164                    sigma: outsigma,
165                },
166                logfile,
167                append,
168                log_details,
169                region_history: &region_history,
170                record_history: state.history,
171            };
172            fit_profile(source, request, external_open_check()).map_err(raised)
173        })?;
174        emit_tool_logs(py, &logs)?;
175        profile_record(py, record)
176    }
177}

179fn peaks<'a>(
180    amp: &'a DoubleList,
181    center: &'a DoubleList,
182    width: &'a DoubleList,
183    fixed: &'a StringList,
184) -> ProfilePeaks<'a> {
185    ProfilePeaks {
186        amplitudes: &amp.0,
187        centers: &center.0,
188        widths: &width.0,
189        fixed: list_or_empty(fixed),
190    }
191}
192
193fn list_or_empty(values: &StringList) -> &[String] {
194    if values.0.len() == 1 && values.0[0].is_empty() {
195        &[]
196    } else {
197        &values.0
198    }
199}

201fn range(name: &str, values: &DoubleList) -> PyResult<Option<[f64; 2]>> {
202    match values.0.as_slice() {
203        [first, last] => Ok(Some([*first, *last])),
204        [] | [_] => Ok(None),
205        _ => Err(raised(format!("Too many elements in {name}"))),
206    }
207}
208
209enum SigmaArgument {
210    None,
211    Values(ArrayD<f32>),
212    Path(String),
213}
214
215fn sigma_argument(value: Option<Variant>) -> PyResult<Option<SigmaArgument>> {
216    let (shape, values) = match value.unwrap_or(Variant::Empty) {
217        Variant::Empty => return Ok(Some(SigmaArgument::None)),
218        Variant::Text(path) if path.is_empty() => return Ok(Some(SigmaArgument::None)),
219        Variant::Text(path) => return Ok(Some(SigmaArgument::Path(path))),
220        Variant::Numbers(values) => (vec![values.len()], values),
221        Variant::Array { shape, values } => (shape, values),
222        _ => return Ok(None),
223    };
224    let values = values.into_iter().map(|value| value as f32).collect();
225    let array = ArrayD::from_shape_vec(IxDyn(&shape).f(), values).map_err(raised)?;
226    Ok(Some(SigmaArgument::Values(array)))
227}
228
229fn divisor_argument(value: Option<Variant>) -> PyResult<ProfileDivisor> {
230    match value.unwrap_or(Variant::Empty) {
231        Variant::Empty => Ok(ProfileDivisor::Automatic),
232        Variant::Number(value) => Ok(ProfileDivisor::Number(value)),
233        value @ (Variant::Text(_) | Variant::Record(_)) => {
234            Ok(ProfileDivisor::Quantity(quantity(&value).map_err(raised)?))
235        }
236        value => Err(raised(format!(
237            "Unsupported type for div: {}",
238            value.describe()
239        ))),
240    }
241}

243fn profile_record(py: Python<'_>, record: RecordValue) -> PyResult<Py<PyAny>> {
244    let record = record_to_py(py, record)?;
245    let dict = record.bind(py).cast::<PyDict>()?;
246    for key in ["direction", "type"] {
247        if let Some(value) = dict.get_item(key)? {
248            let kwargs = PyDict::new(py);
249            kwargs.set_item("dtype", "str")?;
250            let array = py
251                .import("numpy")?
252                .call_method("asarray", (value,), Some(&kwargs))?;
253            dict.set_item(key, array)?;
254        }
255    }
256    Ok(record)
