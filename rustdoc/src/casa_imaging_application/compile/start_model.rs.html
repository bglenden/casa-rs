3//! CASA's `startmodel` (#841 Phase 6b): the image the model starts from
4//! (`SIImageStore::setModelImageOne`). As CASA takes it:
5//! - it is in Jy/pixel (`SynthesisImagerVi2::defineImage` refuses others);
6//! - its masked pixels are zero;
7//! - it is regridded linearly onto the main image's grid and scaled by the
8//!   ratio of the pixel areas, so its flux is kept (`regridToModelImage`:
9//!   `ImageRegrid` with `Interpolate2D::LINEAR`). CASA copies a start model
10//!   already on that grid; casa-rs regrids it too, which changes it only by
11//!   rounding;
12//! - its direction frame is the image's: casa-rs does not convert frames;
13//! - its Stokes are the image's, plane for plane, however many there are.
14//!   CASA's regrid leaves the Stokes axis alone, so it seeds a Stokes I
15//!   image from a Stokes Q start model; casa-rs refuses that (`docs/CASA
16//!   (C++) bugs.md`);
17//! - a cube's start model has the image's spectral axis. An MFS image's has
18//!   one plane, at any frequency: CASA's regrid replicates a degenerate
19//!   spectral axis, and an MFS model holds one plane whatever frequency it
20//!   is stated at.
21//!
22//! It is the model's first generation, so the first major cycle subtracts
23//! its prediction and the restored image includes it. Compile reads its
24//! header only; its pixels are read and regridded once, when the initial
25//! pass seeds the model ([`StartModel::main_samples`]).
26

39/// A start model checked against the main image domain.
40#[derive(Clone, Debug)]
41pub(crate) struct StartModel {
42    path: PathBuf,
43    input_shape: Vec<usize>,
44    template: RegridTemplate,
45}
46
47impl StartModel {
48    /// Check the image at `path` as the start model of `main`, an image of
49    /// `polarizations` Stokes planes and `planes` spectral planes.
50    pub(super) fn prepare(
51        path: &Path,
52        main: &PreparedImageDomain,
53        polarizations: usize,
54        planes: usize,
55    ) -> Result<Self, PrepareError> {
56        let [nx, ny] = main.image_size;
57        Self::onto(
58            path,
59            RegridTemplate {
60                coordinates: main.coordinates.clone(),
61                shape: vec![nx, ny, polarizations, planes],
62            },
63        )
64    }
65
66    /// Check the image at `path` as a start model of `template`'s image.
67    pub(crate) fn onto(path: &Path, template: RegridTemplate) -> Result<Self, PrepareError> {
68        let refuse = |reason: String| PrepareError::StartModel {
69            path: path.to_path_buf(),
70            reason,
71        };
72        let opened = AnyPagedImage::open(path).map_err(|error| refuse(error.to_string()))?;
73        let (units, coordinates, shape) = match &opened {
74            AnyPagedImage::Float32(image) => header(image),
75            AnyPagedImage::Float64(image) => header(image),
76            AnyPagedImage::Complex32(_) | AnyPagedImage::Complex64(_) => {
77                return Err(refuse("a start model is a real image".into()));
78            }
79        };
80        if units != "Jy/pixel" {
81            return Err(refuse(format!(
82                "its unit is {units:?}; a start model is in Jy/pixel, as CASA's tclean requires"
83            )));
84        }
85        check_regrid(&coordinates, &shape, &template).map_err(|error| {
86            refuse(format!(
87                "{error}; its Stokes and spectral axes must be the image's"
88            ))
89        })?;
90        // `check_regrid` keeps the input's coordinate on an axis of one
91        // pixel in both images, which a regrid's caller sees; the model
92        // keeps only the pixels, so a single Stokes plane is checked here.
93        let (ours, theirs) = (stokes(&coordinates), stokes(&template.coordinates));
94        if ours != theirs {
95            return Err(refuse(format!(
96                "its Stokes are {ours:?} and the image's {theirs:?}; a start model seeds the \
97                 image's own Stokes planes"
98            )));
99        }
100        Ok(Self {
101            path: path.to_path_buf(),
102            input_shape: shape,
103            template,
104        })
105    }
106
107    /// Bytes [`Self::main_samples`] holds at most besides the model: the
108    /// image read (up to 8 bytes a pixel) with its mask, its copy with the
109    /// masked pixels zero, `regrid`'s own read of that copy, the regridded
110    /// image with its mask and the samples copied out of it, and `regrid`'s
111    /// planes: an input plane of `f64` with its mask, and an output plane of
112    /// `f64`, a flag and an input position each. The direction axes are the
113    /// first two of both images, as `check_regrid` requires.
114    pub(crate) fn seeding_bytes(&self) -> u64 {
115        let input: usize = self.input_shape.iter().product();
116        let output: usize = self.template.shape.iter().product();
117        let input_plane = self.input_shape[0] * self.input_shape[1];
118        let output_plane = self.template.shape[0] * self.template.shape[1];
119        let position = size_of::<Option<[f64; 2]>>();
120        (input * (8 + 1 + 4 + 4)
121            + output * (4 + 1 + 4)
122            + input_plane * (8 + 1)
123            + output_plane * (8 + 1 + position)) as u64
124    }
125
126    /// The image's pixels along x and y.
127    pub(crate) fn image_size(&self) -> [usize; 2] {
128        [self.template.shape[0], self.template.shape[1]]
129    }
130
131    /// The image's polarizations.
132    pub(crate) fn polarizations(&self) -> usize {
133        self.template.shape[2]
134    }
135
136    /// Bytes making the samples physical holds besides
137    /// [`Self::seeding_bytes`]: the physical samples, and one plane's
138    /// sensitivity read in `f64`.
139    pub(crate) fn conversion_bytes(&self) -> u64 {
140        let output: usize = self.template.shape.iter().product();
141        let output_plane = self.template.shape[0] * self.template.shape[1];
142        (output * 4 + output_plane * 8) as u64
143    }
144
145    /// The main domain's model samples in the model's canonical order
146    /// (spectral plane, polarization, y, x, x fastest): the regridded
147    /// image's Fortran order.
148    pub(crate) fn main_samples(&self) -> Result<Vec<f32>, ImageError> {
149        let zeroed = match AnyPagedImage::open(&self.path)? {
150            AnyPagedImage::Float32(image) => zeroed_copy(&image)?,
151            AnyPagedImage::Float64(image) => zeroed_copy(&image)?,
152            AnyPagedImage::Complex32(_) | AnyPagedImage::Complex64(_) => {
153                return Err(ImageError::InvalidMetadata(
154                    "a start model is a real image".into(),
155                ));
156            }
157        };
158        let regridded = regrid(&zeroed, &self.template, RegridMethod::Linear)?;
159        Ok(regridded
160            .data
161            .as_slice_memory_order()
162            .expect("regrid writes a contiguous Fortran-order image")
163            .to_vec())
164    }
165}

167/// The Stokes of `coordinates`' Stokes coordinate, in plane order; `None`
168/// without one.
169fn stokes(coordinates: &CoordinateSystem) -> Option<Vec<StokesType>> {
170    (0..coordinates.n_coordinates()).find_map(|index| match coordinates.coordinate(index) {
171        CoordinateModel::Stokes(stokes) => Some(stokes.stokes().to_vec()),
172        _ => None,
173    })
174}

176/// The unit, coordinates and shape [`StartModel::prepare`] reads.
177fn header<T: ImagePixel>(
178    image: &casa_images::PagedImage<T>,
179) -> (String, CoordinateSystem, Vec<usize>) {
180    (
181        image.units().to_string(),
182        image.coordinates().clone(),
183        image.shape().to_vec(),
184    )
185}

187/// `image` in memory, in single precision, with its masked pixels zero and
188/// no mask, as `setModelImageOne` copies it before regridding.
189fn zeroed_copy<T>(image: &casa_images::PagedImage<T>) -> Result<TempImage<f32>, ImageError>
190where
191    T: ImagePixel + Into<f64> + Copy,
192{
193    let mut values: ArrayD<f32> = image.get()?.mapv(|value| Into::<f64>::into(value) as f32);
194    if let Some(mask) = image.default_mask()? {
195        Zip::from(&mut values).and(&mask).for_each(|value, &valid| {
196            if !valid {
197                *value = 0.0;
198            }
199        });
200    }
201    let mut copy = TempImage::<f32>::new(
202        image.shape().to_vec(),
203        image.coordinates().clone(),
204        TempStoragePolicy::Memory,
205    )?;
206    copy.set_units(image.units())?;
207    copy.put_slice(&values, &vec![0; values.ndim()])?;
208    Ok(copy)
209}
