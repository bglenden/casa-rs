16/// What a call fails with: CASA raises `RuntimeError` for what its table
17/// proxy refuses and `TypeError` for an argument of the wrong type.
18#[derive(Debug)]
19pub(crate) enum Fail {
20    /// A `RuntimeError`.
21    Runtime(String),
22    /// A `TypeError`.
23    Type(String),
24    /// A `NotImplementedError`: casa-rs cannot do what CASA does here yet.
25    Unsupported(String),
26}
27
28impl From<TableError> for Fail {
29    fn from(error: TableError) -> Self {
30        Self::Runtime(error.to_string())
31    }
32}
33
34impl From<Fail> for PyErr {
35    fn from(fail: Fail) -> Self {
36        match fail {
37            Fail::Runtime(message) => PyRuntimeError::new_err(message),
38            Fail::Type(message) => PyTypeError::new_err(message),
39            Fail::Unsupported(message) => PyNotImplementedError::new_err(message),
40        }
41    }
42}

44/// The lock options CASA's `lockoptions` accepts and reports: its name for
45/// the mode, the mode, and the name `lockoptions()` gives back (`default`
46/// is AutoLocking).
47const OPTIONS: [(&str, LockMode, &str); 7] = [
48    ("default", LockMode::DefaultLocking, "auto"),
49    ("auto", LockMode::AutoLocking, "auto"),
50    ("autonoread", LockMode::AutoNoReadLocking, "autonoread"),
51    ("user", LockMode::UserLocking, "user"),
52    ("usernoread", LockMode::UserNoReadLocking, "usernoread"),
53    ("permanent", LockMode::PermanentLocking, "permanent"),
54    (
55        "permanentwait",
56        LockMode::PermanentLockingWait,
57        "permanentwait",
58    ),
59];
60
61/// The `lockoptions` record of `open`, as given.
62#[derive(Debug, Clone, Default)]
63pub(crate) struct LockRequest {
64    pub(crate) option: Option<String>,
65    pub(crate) interval: Option<f64>,
66    pub(crate) maxwait: Option<i64>,
67}

69/// The lock settings a table was opened with, as `lockoptions()` reports
70/// them.
71#[derive(Debug, Clone)]
72pub(super) struct LockSettings {
73    mode: LockMode,
74    option: &'static str,
75    interval: f64,
76    maxwait: i64,
77}
78
79impl LockSettings {
80    /// Whether the table locks itself for a write (and releases the lock
81    /// after), as `default`, `auto` and `autonoread` do.
82    pub(super) fn automatic(&self) -> bool {
83        matches!(
84            self.mode,
85            LockMode::DefaultLocking | LockMode::AutoLocking | LockMode::AutoNoReadLocking
86        )
87    }
88
89    /// Whether the table takes the write lock when it is opened and keeps it
90    /// (the permanent modes).
91    pub(super) fn permanent(&self) -> bool {
92        matches!(
93            self.mode,
94            LockMode::PermanentLocking | LockMode::PermanentLockingWait
95        )
96    }
97
98    /// Whether a read takes the read lock (all but the NoRead modes), as
99    /// casacore's `TableLock::readLocking` says.
100    pub(super) fn reads_lock(&self) -> bool {
101        !matches!(
102            self.mode,
103            LockMode::AutoNoReadLocking | LockMode::UserNoReadLocking
104        )
105    }
106
107    /// The request that opens a table with these settings again.
108    pub(super) fn request(&self) -> LockRequest {
109        LockRequest {
110            option: Some(self.option.to_string()),
111            interval: Some(self.interval),
112            maxwait: Some(self.maxwait),
113        }
114    }
115
116    /// The seconds between a table's looks for a request for its lock.
117    pub(super) fn interval(&self) -> f64 {
118        self.interval
119    }
120
121    /// The seconds a lock may be waited for; `0` waits for it.
122    pub(super) fn maxwait(&self) -> i64 {
123        self.maxwait
124    }
125
126    /// The attempts at a lock `maxwait` allows; `0` waits for it.
127    pub(super) fn attempts(&self) -> u32 {
128        u32::try_from(self.maxwait.max(0)).unwrap_or(u32::MAX)
129    }
130
131    pub(super) fn from_request(request: &LockRequest) -> Result<Self, Fail> {
132        let name = request.option.as_deref().unwrap_or("default");
133        let (_, mode, reported) = OPTIONS
134            .iter()
135            .find(|(option, _, _)| option.eq_ignore_ascii_case(name))
136            .ok_or_else(|| {
137                let valid = OPTIONS.map(|(option, _, _)| option).join(",");
138                Fail::Runtime(format!(
139                    "'{name}' is an unknown lock option; valid are {valid}"
140                ))
141            })?;
142        Ok(Self {
143            mode: *mode,
144            option: reported,
145            interval: request.interval.unwrap_or(5.0),
146            maxwait: request.maxwait.unwrap_or(0),
147        })
148    }
149}

151/// The underlying opened table and lock/metadata state shared between a
152/// root `table` handle and any `selectrows` views created from it.
153pub(super) struct RootOpen {
154    pub(super) table: Table,
155    /// The root table's absolute path.
156    pub(super) name: PathBuf,
157    pub(super) writable: bool,
158    pub(super) lock: LockSettings,
159    /// Memory columns added after creation have a different CASA description.
160    pub(super) memory_added_columns: HashSet<String>,
161    /// The modify counter the last `datachanged()` saw (casacore starts a
162    /// table at 0).
163    pub(super) last_modify_counter: u32,
164    pub(super) keyword_epoch: Arc<AtomicU64>,
165    pub(super) seen_keyword_epoch: u64,
166    /// `True` while this handle is the one that called `create()`, before any
167    /// reopen or `resync()`: casacore leaves `StandardStMan`'s `IndexLength`
168    /// as `0` until the table is opened from disk.
169    pub(super) created_in_this_handle: bool,
170}

172/// A table `open` attached, or a `selectrows` view sharing its root.
173pub(super) struct Open {
174    pub(super) root: Arc<Mutex<RootOpen>>,
175    /// The handle's own path as `name()` reports it (`root.name` for a plain
176    /// open, a scratch name for a transient `selectrows`, or the saved
177    /// `RefTable` path for `selectrows(..., name=...)`).
178    pub(super) name: PathBuf,
179    /// Cached `root.name` so registry scans never take `root`'s mutex.
180    pub(super) root_name: PathBuf,
181    /// Cached plain table paths resolved from disk (`cache_plain_table_paths`),
182    /// kept outside `root`'s mutex so cache/target checks cover opened `RefTable`s.
183    pub(super) plain_paths: Vec<PathBuf>,
184    pub(super) is_memory: bool,
185    /// Present on a `selectrows` view: maps view row indices `0..len` to row
186    /// indices in `root.table`.
187    pub(super) row_map: Option<Vec<usize>>,
188    /// Present on a column-projected `query`/`taql` view: restricts visible
189    /// columns (and column order) relative to `root.table.schema()`.
190    pub(super) columns: Option<Vec<String>>,
191    /// Remains `true` while this `Open` handle is attached to its parent `table`
192    /// tool; dropping or closing `Open` clears it so `tablerow` handles notice.
193    pub(super) alive: Arc<AtomicBool>,
194}
195
196impl Drop for Open {
197    fn drop(&mut self) {
198        self.alive.store(false, Ordering::Release);
199    }
200}

202impl Open {
203    pub(super) fn from_root(root: RootOpen) -> Self {
204        let name = root.name.clone();
205        let plain_paths = root.table.cache_plain_table_paths().unwrap_or_default();
206        let is_memory = root.table.is_memory();
207        Self {
208            root: Arc::new(Mutex::new(root)),
209            name: name.clone(),
210            root_name: name,
211            plain_paths,
212            is_memory,
213            row_map: None,
214            columns: None,
215            alive: Arc::new(AtomicBool::new(true)),
216        }
217    }
218
219    pub(super) fn from_detached_table(
220        table: Table,
221        name: PathBuf,
222        writable: bool,
223        lock: LockSettings,
224    ) -> Self {
225        let root = RootOpen {
226            table,
227            name: name.clone(),
228            writable,
229            lock,
230            memory_added_columns: HashSet::new(),
231            last_modify_counter: 0,
232            keyword_epoch: super::registry::keyword_epoch(&name),
233            seen_keyword_epoch: 0,
234            created_in_this_handle: false,
235        };
236        Self::from_root(root)
237    }
238
239    pub(super) fn view(&self, name: PathBuf, row_map: Vec<usize>) -> Self {
240        self.view_columns(name, row_map, self.columns.clone())
241    }
242
243    pub(super) fn view_columns(
244        &self,
245        name: PathBuf,
246        row_map: Vec<usize>,
247        columns: Option<Vec<String>>,
248    ) -> Self {
249        Self {
250            root: Arc::clone(&self.root),
251            name,
252            root_name: self.root_name.clone(),
253            plain_paths: self.plain_paths.clone(),
254            is_memory: self.is_memory,
255            row_map: Some(row_map),
256            columns,
257            alive: Arc::new(AtomicBool::new(true)),
258        }
259    }
260
261    pub(super) fn root(&self) -> MutexGuard<'_, RootOpen> {
262        self.root.lock().unwrap_or_else(PoisonError::into_inner)
263    }
264
265    pub(super) fn is_view(&self) -> bool {
266        self.row_map.is_some() || self.columns.is_some()
267    }
268
269    pub(super) fn column_allowed(&self, column: &str) -> bool {
270        self.columns
271            .as_ref()
272            .is_none_or(|cols| cols.iter().any(|c| c == column))
273    }
274
275    pub(super) fn visible_columns<'a>(
276        &self,
277        table: &'a Table,
278    ) -> Vec<&'a casa_tables::ColumnSchema> {
279        let Some(schema) = table.schema() else {
280            return Vec::new();
281        };
282        match &self.columns {
283            None => schema.columns().iter().collect(),
284            Some(cols) => cols.iter().filter_map(|c| schema.column(c)).collect(),
285        }
286    }
287
288    pub(super) fn nrows(&self) -> usize {
289        self.row_map
290            .as_ref()
291            .map_or_else(|| self.root().table.row_count(), Vec::len)
292    }
293
294    pub(super) fn translate_row(&self, row: usize) -> usize {
295        self.row_map.as_ref().map_or(row, |map| map[row])
296    }
297
298    pub(super) fn translate_rows(&self, rows: &[usize]) -> Vec<usize> {
299        match &self.row_map {
300            Some(map) => rows.iter().map(|&row| map[row]).collect(),
301            None => rows.to_vec(),
302        }
303    }
304
305    /// The root handle opened at `name` under `request`'s lock options.
306    pub(super) fn attach_root(
307        name: &str,
308        request: &LockRequest,
309        writable: bool,
310    ) -> Result<RootOpen, Fail> {
311        let lock = LockSettings::from_request(request)?;
312        let path = Path::new(name);
313        crate::tools::componentlist::flush_pending_keywords(path)
314            .map_err(|e| Fail::Runtime(e.to_string()))?;
315        if !path.exists() {
316            return Err(Fail::Runtime(format!("Table {name} does not exist")));
317        }
318        // A table opened `nomodify` holds no write lock, in the permanent
319        // modes too.
320        let options = LockOptions::new(lock.mode)
321            .with_inspection_interval(lock.interval)
322            .read_only(!writable);
323        // Register before opening, so a concurrent keyword write during the
324        // open is observed on this handle's next read.
325        let keyword_epoch = super::registry::keyword_epoch(path);
326        let seen_keyword_epoch = keyword_epoch.load(Ordering::Acquire);
327        let table = Table::open_with_lock(TableOptions::new(path), options)?;
328        let name = std::path::absolute(path).unwrap_or_else(|_| path.to_path_buf());
329        Ok(RootOpen {
330            table,
331            name,
332            writable,
333            lock,
334            memory_added_columns: HashSet::new(),
335            last_modify_counter: 0,
336            keyword_epoch,
337            seen_keyword_epoch,
338            created_in_this_handle: false,
339        })
340    }
341
342    /// The table at `name` opened under `request`'s lock options.
343    pub(super) fn attach(name: &str, request: &LockRequest, writable: bool) -> Result<Self, Fail> {
344        Self::attach_root(name, request, writable).map(Self::from_root)
345    }
346}

348/// The tool's state: at most one open table.
349pub(crate) struct State {
350    pub(super) open: Option<Open>,
351    /// Tells the tool apart in the registry of the process's tools
352    /// (`registry.rs`).
353    pub(super) id: u64,
354}
355
356impl Default for State {
357    fn default() -> Self {
358        Self::new(None)
359    }
360}
361
362impl State {
363    /// The state of a tool with `open` attached.
364    pub(super) fn new(open: Option<Open>) -> Self {
365        static NEXT_ID: AtomicU64 = AtomicU64::new(1);
366        Self {
367            open,
368            id: NEXT_ID.fetch_add(1, Ordering::Relaxed),
369        }
370    }
371}

373/// Logs the message CASA logs when a call needs a table and none is open.
374pub(super) fn no_table(log: &mut Vec<LogLine>) {
375    log.push(LogLine::info("No table opened."));
376}
377
378/// Logs the warning CASA gives the lock calls with no table open.
379pub(super) fn no_table_specified(log: &mut Vec<LogLine>) {
380    log.push(LogLine::warning("No table specified, please open first"));
381}
382
383impl State {
384    /// The open table, or `None` with CASA's INFO message logged.
385    pub(super) fn table(&self, log: &mut Vec<LogLine>) -> Option<&Open> {
386        let open = self.open.as_ref();
387        if open.is_none() {
388            no_table(log);
389        }
390        open
391    }
392
393    /// CASA's `open`: closes the table already open, then attaches `name`
394    /// under `request`'s lock options.
395    pub(super) fn open(
396        &mut self,
397        name: &str,
398        request: &LockRequest,
399        writable: bool,
400    ) -> Result<(), Fail> {
401        let _publication = super::registry::lock_attachment_publication();
402        self.open = None;
403        self.open = Some(Open::attach(name, request, writable)?);
404        Ok(())
405    }
406
407    /// CASA's `close` and `done`: releases the table and its locks.
408    pub(super) fn close(&mut self) {
409        self.open = None;
410    }
411
412    /// What every read starts with, as casacore's `TableProxy::syncTable`:
413    /// the read lock, when the lock options read-lock, taken again if it was
414    /// released. Refresh an already-held lock after a component-list keyword
415    /// write in this process; a write lock and its changes are kept.
416    pub(super) fn synchronize(&mut self) -> Result<(), Fail> {
417        if let Some(open) = self.open.as_ref() {
418            let mut root = open.root();
419            synchronize_root(&mut root)?;
420        }
421        Ok(())
422    }
423}
424
425/// Publish only the backing table before a call that actually needs its bytes.
426pub(super) fn publish_component_keywords(path: &Path) -> Result<(), Fail> {
427    crate::tools::componentlist::flush_pending_keywords(path)
428        .map_err(|e| Fail::Runtime(e.to_string()))
429}
430
431/// Synchronizes `root`'s read lock and refreshes any concurrent in-process
432/// component-list keyword updates.
433pub(super) fn synchronize_root(root: &mut RootOpen) -> Result<(), Fail> {
434    publish_component_keywords(&root.name)?;
435    if root.lock.reads_lock() {
436        root.table.lock(LockType::Read, 0)?;
437    }
438    let epoch = root.keyword_epoch.load(Ordering::Acquire);
439    if epoch != root.seen_keyword_epoch {
440        let already_locked = root.table.has_lock(LockType::Read);
441        if !already_locked && !root.table.lock(LockType::Read, 0)? {
442            return Err(Fail::Runtime(
443                "Cannot lock table to refresh keywords".into(),
444            ));
445        }
446        if !root.table.resync_locked()? {
447            // A reference synchronizes its own lock file, while its keywords
448            // come from the parent. A local writer can also retain its lock.
449            root.table.refresh_persisted_keywords()?;
450        }
451        if !already_locked {
452            root.table.unlock_metadata_only()?;
453        }
454        root.seen_keyword_epoch = epoch;
455    }
456    Ok(())
457}

459impl State {
460    /// `lockoptions()`: the settings the table was opened with.
461    pub(super) fn lock_settings(&self) -> Option<(&'static str, f64, i64)> {
462        self.open.as_ref().map(|open| {
463            let root = open.root();
464            (root.lock.option, root.lock.interval, root.lock.maxwait)
465        })
466    }
467
468    /// `haslock(write)`.
469    pub(super) fn has_lock(&self, write: bool) -> bool {
470        self.open.as_ref().is_some_and(|open| {
471            open.root().table.has_lock(if write {
472                LockType::Write
473            } else {
474                LockType::Read
475            })
476        })
477    }
478
479    /// `lock(write, nattempts)`: whether the lock is held afterwards.
480    pub(super) fn lock(&mut self, write: bool, nattempts: u32) -> Result<bool, Fail> {
481        let Some(open) = self.open.as_ref() else {
482            return Ok(false);
483        };
484        let kind = if write {
485            LockType::Write
486        } else {
487            LockType::Read
488        };
489        publish_component_keywords(&open.root_name)?;
490        Ok(open.root().table.lock(kind, nattempts)?)
491    }
492
493    /// `unlock()`: releases the lock. Only table metadata is flushed, so a
494    /// table's storage managers stay as they are.
495    pub(super) fn unlock(&mut self) -> Result<bool, Fail> {
496        let Some(open) = self.open.as_ref() else {
497            return Ok(false);
498        };
499        open.root().table.unlock_metadata_only()?;
500        Ok(true)
501    }
