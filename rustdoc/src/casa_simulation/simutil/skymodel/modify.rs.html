2//! `simutil.modifymodel`: a model image fitted into the four axes
3//! `simobserve` predicts from, `[direction, direction, Stokes, spectral]`,
4//! with its brightness, direction, cell, centre frequency and channel width
5//! overridden as asked.
6//!
7//! The model is read whole, as CASA reads it (`ia.getchunk`), from a CASA
8//! image or a FITS file. Its pixel axes are found by their coordinates: the
9//! direction's two, the spectral and the Stokes axis. An image of two or
10//! three axes gains degenerate ones; a missing spectral or Stokes axis is
11//! the first axis no coordinate claims. The new image is LSRK, in
12//! Jy/pixel, with the model's NaNs as zeros and its pixels scaled to
13//! `inbright` at the peak.
14//!
15//! Its direction is the model's at pixel `(nx/2, ny/2)`, written as
16//! `simutil` writes directions (precision 5) and read back, or
17//! `indirection`, with the reference pixel at `(nx/2, ny/2)` and the
18//! model's projection (SIN for `indirection`). Its cell is the model's
19//! increment through its linear transform, or `incell`; its spectral axis
20//! the model's, or `incenter` at channel `⌊nchan/2⌋` (also the rest
21//! frequency) and `inwidth`.
22

38/// `modifymodel`'s overrides, CASA's strings as `simobserve` passes them.
39/// An empty string keeps the model's own value, as `"unchanged"` does for
40/// `inbright`.
41#[derive(Debug, Clone, Default, PartialEq, Eq)]
42pub struct ModelOverrides {
43    /// The peak brightness the model is scaled to, read as Jy/pixel
44    /// (`inbright`).
45    pub inbright: String,
46    /// The direction of the model's centre (`indirection`, one string or a
47    /// list of one).
48    pub indirection: Vec<String>,
49    /// The cell (`incell`): one cell for both axes, the longitude's made
50    /// negative, or the two axes' cells as given.
51    pub incell: Vec<String>,
52    /// The frequency of the model's centre channel (`incenter`).
53    pub incenter: String,
54    /// The channel width (`inwidth`).
55    pub inwidth: String,
56}

58/// A value `modifymodel` returns either as the override given or as the
59/// model's own quantity.
60#[derive(Debug, Clone, PartialEq)]
61pub enum ModelValue {
62    /// The override string, as CASA returns it.
63    Given(String),
64    /// The model's own value.
65    Model(Quantity),
66}
67
68impl ModelValue {
69    /// The value as a quantity (`qa.quantity` of it).
70    ///
71    /// # Errors
72    ///
73    /// [`SimutilError::Quantity`] for an override casacore cannot read.
74    pub fn quantity(&self) -> Result<Quantity, SimutilError> {
75        match self {
76            Self::Given(text) => qa::read(text),
77            Self::Model(value) => Ok(value.clone()),
78        }
79    }
80}

82/// What `modifymodel` returns, in its order.
83#[derive(Debug, Clone, PartialEq)]
84pub struct ModelSummary {
85    /// The direction of the reference pixel, as `simutil` writes it.
86    pub refdir: String,
87    /// The cell, both axes positive.
88    pub cell: [Quantity; 2],
89    /// The image's size, pixels times cell.
90    pub size: [Quantity; 2],
91    /// The number of channels.
92    pub nchan: usize,
93    /// The frequency of the reference channel.
94    pub specrefval: ModelValue,
95    /// The reference channel.
96    pub specrefpix: f64,
97    /// The channel width.
98    pub width: ModelValue,
99    /// The Stokes planes' names, `"I"` to `"IQUV"`.
100    pub stokes: String,
101}

103/// The new image's role for each of the model's four axes, and the axis
104/// holding each role (`modifymodel`'s `axassigned` and `axmap`): 0 and 1
105/// the direction's, 2 Stokes, 3 spectral.
106#[derive(Debug, Default)]
107struct AxisMap {
108    axis_of: [Option<usize>; 4],
109    role_of: [Option<usize>; 4],
110}
111
112impl AxisMap {
113    fn assign(&mut self, role: usize, axis: usize) {
114        self.axis_of[role] = Some(axis);
115        self.role_of[axis] = Some(role);
116    }
117
118    /// The first axis no coordinate claims, for `role`.
119    fn assign_unused(&mut self, role: usize, what: &str) -> Result<usize, SkyModelError> {
120        let Some(axis) = self.role_of.iter().position(Option::is_none) else {
121            let roles = self
122                .role_of
123                .map(|role| role.map_or(-1, |role| role as i64))
124                .map(|role| role.to_string())
125                .join(" ");
126            return Err(SkyModelError::Model(format!(
127                "I can't find an unused axis to make {what} [{roles}] "
128            )));
129        };
130        self.assign(role, axis);
131        Ok(axis)
132    }
133
134    fn axis(&self, role: usize) -> usize {
135        self.axis_of[role].expect("every role is assigned before the image is built")
136    }
137}

146/// `simutil.modifymodel(inimage, outimage, inbright, indirection, incell,
147/// incenter, inwidth, innchan, flatimage=False)`: write the model at
148/// `input` (a CASA image or a FITS file) as a new image at `output`
149/// (replacing what is there) and return what it holds (see the module
150/// documentation). `innchan` is unused in CASA.
151///
152/// # Errors
153///
154/// [`SkyModelError::Model`] with CASA's message where CASA raises, and
155/// [`SkyModelError::Indirections`] where it returns `False`; a model that
156/// cannot be read or written, or `output` the same as `input` (CASA then
157/// only reports on the model).
158pub fn modify_model(
159    input: &Path,
160    output: &Path,
161    overrides: &ModelOverrides,
162) -> Result<ModelSummary, SkyModelError> {
163    if input == output {
164        return Err(SkyModelError::Model(format!(
165            "modifymodel writes a new image: {} is both the input and the output",
166            input.display()
167        )));
168    }
169    let model = InputImage::read(input)?;
170    let scale = brightness_scale(&overrides.inbright, &model.pixels)?;
171    let pixels = four_axes(model.pixels)?;
172    let shape: [usize; 4] = std::array::from_fn(|axis| pixels.shape()[axis]);
173    let incell = cell_override(&overrides.incell)?;
174    let coordinates = &model.coordinates;
175    let mut axes = AxisMap::default();
176    let (direction, image_cell) = image_direction(coordinates, &shape, &mut axes)?;
177    let direction = direction_override(&overrides.indirection, direction, &shape, &axes)?;
178    let cell = match incell {
179        Some(cell) if qa::conforms(&cell[0], "arcsec") => cell,
180        _ => match image_cell {
181            Some(cell) if qa::conforms(&cell[0], "arcsec") => cell,
182            _ => {
183                return Err(SkyModelError::Model(
184                    "Unable to determine model cell size.  Valid 'incell' parameter must be provided."
185                        .to_string(),
186                ));
187            }
188        },
189    };
190    let spectral = spectral_axis(coordinates, &shape, &mut axes, overrides)?;
191    let (stokes, stokes_types) = stokes_axis(coordinates, &shape, &mut axes)?;
192    let model_shape = [
193        shape[axes.axis(0)],
194        shape[axes.axis(1)],
195        stokes_types.len(),
196        spectral.nchan,
197    ];
198    let new_coordinates = model_coordinates(&direction, &cell, &spectral, stokes_types)?;
199    // The model's axes in the new image's order, NaN as zero, scaled.
200    let order: Vec<usize> = (0..4).map(|role| axes.axis(role)).collect();
201    let arranged = pixels.permuted_axes(IxDyn(&order));
202    let mut written = ArrayD::<f32>::zeros(IxDyn(&model_shape).f());
203    written.zip_mut_with(&arranged, |out, &value| {
204        let value = if value.is_nan() { 0.0 } else { value };
205        *out = (value * scale) as f32;
206    });
207    write_image(output, &written, new_coordinates, "Jy/pixel")?;
208    let cell = cell.map(|cell| Quantity::with_unit(cell.value().abs(), cell.unit().clone()));
209    let size = [0, 1].map(|axis| {
210        qa::mul(&qa::number(model_shape[axis] as f64), &cell[axis]).expect("a number times a cell")
211    });
212    Ok(ModelSummary {
213        refdir: direction.refdir,
214        cell,
215        size,
216        nchan: spectral.nchan,
217        specrefval: spectral.refval,
218        specrefpix: spectral.refpix,
219        width: spectral.width,
220        stokes,
221    })
222}

224/// `scalefactor`: `inbright` (read as Jy/pixel when it has a unit) over
225/// the model's peak, ignoring NaNs; 1 when unset or `"unchanged"`.
226fn brightness_scale(inbright: &str, pixels: &ArrayD<f64>) -> Result<f64, SkyModelError> {
227    if inbright.is_empty() || inbright == "unchanged" {
228        return Ok(1.0);
229    }
230    let brightness = if qa::is_quantity(inbright) {
231        let value = qa::read(inbright)?;
232        if value.unit().name().is_empty() {
233            value.value()
234        } else {
235            // casacore reads `pixel` but not every brightness unit; CASA
236            // then takes the number.
237            qa::value_in(&value, "Jy/pixel").unwrap_or_else(|_| {
238                tracing::warn!(
239                    "assuming inbright={inbright} means {} Jy/pixel",
240                    python_float(value.value())
241                );
242                value.value()
243            })
244        }
245    } else {
246        inbright.trim().parse::<f64>().map_err(|_| {
247            SkyModelError::Model(format!("could not convert string to float: '{inbright}'"))
248        })?
249    };
250    let peak = pixels
251        .iter()
252        .filter(|value| !value.is_nan())
253        .fold(f64::NEG_INFINITY, |peak, value| peak.max(*value));
254    let peak = if peak == f64::NEG_INFINITY {
255        f64::NAN
256    } else {
257        peak
258    };
259    Ok(brightness / peak)
260}

262/// The model with degenerate axes added to make four.
263fn four_axes(pixels: ArrayD<f64>) -> Result<ArrayD<f64>, SkyModelError> {
264    match pixels.ndim() {
265        0 | 1 => Err(SkyModelError::Model(
266            "Your input model has fewer than 2 dimensions.  Can't proceed".to_string(),
267        )),
268        2..=4 => {
269            let mut pixels = pixels;
270            while pixels.ndim() < 4 {
271                pixels.insert_axis_inplace(Axis(pixels.ndim()));
272            }
273            Ok(pixels)
274        }
275        _ => Err(SkyModelError::Model(
276            "model image has more than 4 dimensions.  Not sure how to proceed".to_string(),
277        )),
278    }
279}

281/// `incell` as `modifymodel` makes it: one cell converted to arcsec, made
282/// positive, for `[−cell, cell]`; or two cells converted to arcsec as given.
283/// `None` when unset. A value that is not an angle is kept, and later
284/// passed over, as casacore's conversion keeps it.
285fn cell_override(incell: &[String]) -> Result<Option<[Quantity; 2]>, SkyModelError> {
286    let read = |text: &str| -> Result<Quantity, SkyModelError> {
287        let value = if text.is_empty() {
288            qa::number(0.0)
289        } else {
290            qa::read(text)?
291        };
292        Ok(qa::convert(&value, "arcsec")?)
293    };
294    match incell {
295        [] => Ok(None),
296        [cell] => {
297            let cell = read(cell)?;
298            let cell = Quantity::with_unit(cell.value().abs(), cell.unit().clone());
299            let negative = qa::mul(&cell, &qa::number(-1.0))?;
300            Ok(Some([negative, cell]))
301        }
302        [x, y] => Ok(Some([read(x)?, read(y)?])),
303        _ => Err(SkyModelError::Model(format!(
304            "incell takes one cell or two, got {}",
305            incell.len()
306        ))),
307    }
308}

310/// The model's direction axes, its direction at pixel `(nx/2, ny/2)` and
311/// its cell; the first two axes when it has no direction coordinate.
312fn image_direction(
313    coordinates: &CoordinateSystem,
314    shape: &[usize; 4],
315    axes: &mut AxisMap,
316) -> Result<(Option<ModelDirection>, Option<[Quantity; 2]>), SkyModelError> {
317    let Some((index, pixel_axes)) = pixel_axes(coordinates, CoordinateType::Direction) else {
318        axes.assign(0, 0);
319        axes.assign(1, 1);
320        return Ok((None, None));
321    };
322    let CoordinateModel::Direction(coordinate) = coordinates.coordinate(index) else {
323        unreachable!("the coordinate found is a direction coordinate");
324    };
325    axes.assign(0, pixel_axes[0]);
326    axes.assign(1, pixel_axes[1]);
327    let refpix = [
328        0.5 * shape[pixel_axes[0]] as f64,
329        0.5 * shape[pixel_axes[1]] as f64,
330    ];
331    let world = coordinate.to_world(&refpix)?;
332    let radians = |value: f64| Quantity::with_unit(value, unit("rad"));
333    let refdir = format!(
334        "{} {} {}",
335        coordinate.direction_ref().as_str(),
336        qa::hms(&radians(world[0]))?,
337        qa::dms(&radians(world[1]))?
338    );
339    let pc = coordinate.pc_matrix();
340    if pc[[0, 1]].abs().max(pc[[1, 0]].abs()) > 1e-4 {
341        return Err(SkyModelError::Model(
342            "Your image is rotated with respect to Lat/Lon.  I can't cope with that yet"
343                .to_string(),
344        ));
345    }
346    let increment = coordinate.increment();
347    let cell = [0, 1].map(|axis| {
348        qa::mul(&radians(increment[axis]), &qa::number(pc[[axis, axis]]))
349            .and_then(|cell| qa::convert(&cell, "arcsec"))
350    });
351    let [x, y] = cell;
352    Ok((
353        Some(ModelDirection {
354            refdir,
355            projection: casacore_projection(coordinate.projection()),
356            refpix,
357        }),
358        Some([x?, y?]),
359    ))
360}

362/// The direction `indirection` sets, with a SIN projection, or the model's.
363fn direction_override(
364    indirection: &[String],
365    model: Option<ModelDirection>,
366    shape: &[usize; 4],
367    axes: &AxisMap,
368) -> Result<ModelDirection, SkyModelError> {
369    let indirection = match indirection {
370        [] => "",
371        [one] => one.as_str(),
372        many => {
373            let listed = many
374                .iter()
375                .map(|direction| format!("'{direction}'"))
376                .collect::<Vec<_>>()
377                .join(", ");
378            return Err(SkyModelError::Indirections(format!("[{listed}]")));
379        }
380    };
381    if direction::is_direction(indirection) {
382        let split = direction::split_direction(indirection)?;
383        return Ok(ModelDirection {
384            refdir: format!(
385                "{}{} {}",
386                split.epoch,
387                qa::hms(&split.longitude)?,
388                qa::dms(&split.latitude)?
389            ),
390            projection: Projection::with_parameters(ProjectionType::SIN, vec![0.0, 0.0]),
391            refpix: [
392                0.5 * shape[axes.axis(0)] as f64,
393                0.5 * shape[axes.axis(1)] as f64,
394            ],
395        });
396    }
397    match model {
398        Some(model) if direction::is_direction(&model.refdir) => Ok(model),
399        _ => Err(SkyModelError::Model(
400            "Cannot determine reference direction in model image.  Valid 'indirection' parameter must be provided."
401                .to_string(),
402        )),
403    }
404}

406/// The new image's spectral axis.
407struct ModelSpectrum {
408    nchan: usize,
409    refval: ModelValue,
410    refpix: f64,
411    width: ModelValue,
412    rest_frequency: ModelValue,
413}
414
415/// The model's spectral axis, or the first unclaimed axis, with `incenter`
416/// and `inwidth` applied where they are frequencies and not negative.
417fn spectral_axis(
418    coordinates: &CoordinateSystem,
419    shape: &[usize; 4],
420    axes: &mut AxisMap,
421    overrides: &ModelOverrides,
422) -> Result<ModelSpectrum, SkyModelError> {
423    let mut spectrum = match pixel_axes(coordinates, CoordinateType::Spectral) {
424        Some((index, pixel_axes)) => {
425            let CoordinateModel::Spectral(coordinate) = coordinates.coordinate(index) else {
426                unreachable!("the coordinate found is a spectral coordinate");
427            };
428            axes.assign(3, pixel_axes[0]);
429            let axis_unit = unit(&coordinate.axis_units()[0]);
430            let in_unit = |value: f64| Quantity::with_unit(value, axis_unit.clone());
431            ModelSpectrum {
432                nchan: shape[pixel_axes[0]],
433                refval: ModelValue::Model(in_unit(coordinate.reference_value()[0])),
434                refpix: coordinate.reference_pixel()[0],
435                width: ModelValue::Model(in_unit(coordinate.increment()[0])),
436                rest_frequency: ModelValue::Model(Quantity::with_unit(
437                    coordinate.rest_frequency(),
438                    unit("Hz"),
439                )),
440            }
441        }
442        None => {
443            let axis = axes.assign_unused(3, "Spectral")?;
444            let unset = ModelValue::Given(String::new());
445            ModelSpectrum {
446                nchan: shape[axis],
447                refval: unset.clone(),
448                refpix: 0.0,
449                width: unset.clone(),
450                rest_frequency: unset,
451            }
452        }
453    };
454    let frequency = |text: &str| {
455        qa::is_quantity(text)
456            && qa::read(text).is_ok_and(|value| qa::conforms(&value, "Hz") && value.value() >= 0.0)
457    };
458    let valid = |value: &ModelValue| match value {
459        ModelValue::Model(value) => qa::conforms(value, "Hz"),
460        ModelValue::Given(text) => qa::is_quantity(text) && frequency(text),
461    };
462    if frequency(&overrides.incenter) {
463        spectrum.refval = ModelValue::Given(overrides.incenter.clone());
464        spectrum.refpix = (spectrum.nchan as f64 * 0.5).floor();
465        spectrum.rest_frequency = ModelValue::Given(overrides.incenter.clone());
466    } else if !valid(&spectrum.refval) {
467        return Err(SkyModelError::Model(
468            "Unable to determine model frequency.  Valid 'incenter' parameter must be provided."
469                .to_string(),
470        ));
471    }
472    if frequency(&overrides.inwidth) {
473        spectrum.width = ModelValue::Given(overrides.inwidth.clone());
474    } else if !valid(&spectrum.width) {
475        return Err(SkyModelError::Model(
476            "Unable to determine model channel or bandwidth.  Valid 'inwidth' parameter must be provided."
477                .to_string(),
478        ));
479    }
480    Ok(spectrum)
481}

483/// The model's Stokes axis and its planes, or the first unclaimed axis,
484/// labelled by its length: `IQUV`, `IQV`, `IQ` or `I`.
485fn stokes_axis(
486    coordinates: &CoordinateSystem,
487    shape: &[usize; 4],
488    axes: &mut AxisMap,
489) -> Result<(String, Vec<StokesType>), SkyModelError> {
490    if let Some((index, pixel_axes)) = pixel_axes(coordinates, CoordinateType::Stokes) {
491        let CoordinateModel::Stokes(coordinate) = coordinates.coordinate(index) else {
492            unreachable!("the coordinate found is a Stokes coordinate");
493        };
494        axes.assign(2, pixel_axes[0]);
495        if shape[pixel_axes[0]] > 4 {
496            return Err(SkyModelError::Model(
497                "you appear to have more than 4 Stokes components - please edit your header and/or parameters"
498                    .to_string(),
499            ));
500        }
501        let types = coordinate.stokes().to_vec();
502        let names = types.iter().map(|stokes| stokes.name()).collect();
503        return Ok((names, types));
504    }
505    let axis = axes.assign_unused(2, "Stokes")?;
506    let planes = shape[axis];
507    if planes > 4 {
508        return Err(SkyModelError::Model(format!(
509            "you have {planes} Stokes parameters in your potential Stokes axis {axis}.  something is wrong."
510        )));
511    }
512    let names = match planes {
513        4 => "IQUV",
514        3 => "IQV",
515        2 => "IQ",
516        _ => "I",
517    };
518    if planes == 2 || planes == 3 {
519        tracing::warn!(
520            "setting {names} Stokes parameters from the 4th axis of you model.  If that's not what you want, then edit the header"
521        );
522    }
523    let types = names
524        .chars()
525        .map(|name| StokesType::from_name(&name.to_string()).expect("I, Q, U and V are Stokes"))
526        .collect();
527    Ok((names.to_string(), types))
528}

530/// The new image's coordinate system: the direction read back from its
531/// string, the cell in radians, the Stokes planes, and an LSRK spectral
532/// axis.
533fn model_coordinates(
534    direction: &ModelDirection,
535    cell: &[Quantity; 2],
536    spectrum: &ModelSpectrum,
537    stokes: Vec<StokesType>,
538) -> Result<CoordinateSystem, SkyModelError> {
539    let (refer, crval) = direction::direction_measure(&direction.refdir)?;
540    let cdelt = [
541        qa::value_in(&cell[0], "rad")?,
542        qa::value_in(&cell[1], "rad")?,
543    ];
544    let hz = |value: &ModelValue| -> Result<f64, SkyModelError> {
545        Ok(qa::value_in(&value.quantity()?, "Hz")?)
546    };
547    let width = hz(&spectrum.width)?;
548    check_increments(&[cdelt[0], cdelt[1], width])?;
549    let mut coordinates = CoordinateSystem::new().with_obs_info(shape_obs_info());
550    coordinates.add_coordinate(DirectionCoordinate::new(
551        refer,
552        direction.projection.clone(),
553        crval,
554        cdelt,
555        direction.refpix,
556    ));
557    coordinates.add_coordinate(StokesCoordinate::new(stokes));
558    coordinates.add_coordinate(SpectralCoordinate::new(
559        FrequencyRef::LSRK,
560        hz(&spectrum.refval)?,
561        width,
562        spectrum.refpix,
563        hz(&spectrum.rest_frequency)?,
564    ));
565    Ok(coordinates)
566}

568/// Refuse a zero (or non-finite) increment, as CASA's coordinate setters
569/// do: wcslib cannot invert the axis, and CASA could not open the image.
570pub(super) fn check_increments(increments: &[f64]) -> Result<(), SkyModelError> {
571    if increments
572        .iter()
573        .all(|increment| increment.is_finite() && *increment != 0.0)
574    {
575        Ok(())
576    } else {
577        Err(SkyModelError::Model(
578            "wcs wcsset_error: Linear transformation matrix is singular".to_string(),
579        ))
580    }
581}
