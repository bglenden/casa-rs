37// One admitted wave owns this inline state, independently of its grid payloads.
38#[allow(clippy::large_enum_variant)]
39pub(super) enum WaveConsumer<'a> {
40    Cpu(BulkWave<'a>),
41    Metal(MetalWave<'a>),
42}

44impl BulkConsumer for WaveConsumer<'_> {
45    type Completion = Vec<BandResult>;
46    fn consume(
47        &mut self,
48        block: NativeBlockView<'_>,
49        layout: &NativeLayout,
50        selected: std::ops::Range<usize>,
51        execution: BoundedExecution<'_>,
52    ) -> io::Result<()> {
53        match self {
54            Self::Cpu(wave) => wave.consume(block, layout, selected, execution),
55            Self::Metal(wave) => wave.consume(block, layout, selected, execution),
56        }
57    }
58    fn complete(self, execution: BoundedExecution<'_>) -> io::Result<Vec<BandResult>> {
59        match self {
60            Self::Cpu(wave) => wave.complete(execution),
61            Self::Metal(wave) => wave.complete(execution),
62        }
63    }
64}

66impl MetalWaveMemory {
67    pub(super) fn new(bands: &[BandPlan], workers: usize, rows: usize) -> io::Result<Self> {
68        if bands.first().is_some_and(BandPlan::is_residual) {
69            let wave = BandPlan::residual_wave(bands).map_err(io::Error::other)?;
70            let layout = ResidualLayout::new(&wave, rows)?;
71            let device = wave
72                .spatial_grid_bytes()
73                .map_err(io::Error::other)?
74                .checked_add(BandPlan::spatial_weight_bytes().next_multiple_of(8))
75                .and_then(|bytes| bytes.checked_add(layout.slot_bytes.checked_mul(2)?))
76                .and_then(|bytes| bytes.checked_add(layout.capacities[0].max(1).checked_mul(8)?))
77                .ok_or_else(overflow)?;
78            let scratch = wave.spatial_host_bytes(1).map_err(io::Error::other)?;
79            return Ok(Self {
80                cpu: BulkWave::bytes(std::slice::from_ref(&wave), 1)?,
81                device: device as u64,
82                scratch: scratch as u64,
83                requests: layout.capacities[0],
84            });
85        }
86        let cpu = BulkWave::bytes(bands, workers)?;
87        let grids = bands.iter().try_fold(0_u64, |sum, band| {
88            sum.checked_add(band.spatial_grid_bytes().map_err(io::Error::other)? as u64)
89                .ok_or_else(overflow)
90        })?;
91        let requests = bands
92            .iter()
93            .map(|b| b.spatial_request_capacity(rows))
94            .collect::<Result<Vec<_>, _>>()
95            .map_err(io::Error::other)?
96            .into_iter()
97            .max()
98            .unwrap_or(1);
99        let scratch = bands
100            .iter()
101            .map(|b| {
102                b.spatial_host_bytes(rows)?
103                    .checked_add(
104                        b.spatial_command_capacity()
105                            * (size_of::<DispatchTarget<'_>>() + size_of::<CubeDispatch<'_>>()),
106                    )
107                    .ok_or(SpectralOperatorError::ResidencyOverflow)
108            })
109            .collect::<Result<Vec<_>, _>>()
110            .map_err(io::Error::other)?
111            .into_iter()
112            .max()
113            .unwrap_or(0) as u64;
114        let lanes = workers.min(bands.len()) as u64;
115        let layout = InitialLayout::new(bands, rows)?;
116        let command_bytes = layout
117            .command_capacity
118            .checked_mul(
119                2 * size_of::<CubeDispatch<'_>>()
120                    + 4 * size_of::<MetalBufferRegion<'_>>()
121                    + 8 * size_of::<(AllocationId, usize, usize)>(),
122            )
123            .ok_or_else(overflow)?;
124        let scratch = scratch
125            .checked_mul(lanes)
126            .and_then(|n| n.checked_add(command_bytes as u64))
127            .and_then(|n| n.checked_add((layout.offsets.len() * 2 * size_of::<usize>()) as u64))
128            .ok_or_else(overflow)?;
129        let device = grids
130            .checked_add(BandPlan::spatial_weight_bytes().next_multiple_of(8) as u64)
131            .and_then(|n| {
132                n.checked_add(
133                    (layout.slot_bytes as u64).checked_mul(2)?.checked_add(
134                        (requests as u64)
135                            .checked_mul(size_of::<Complex32>() as u64)?
136                            .checked_mul(lanes)?,
137                    )?,
138                )
139            })
140            .ok_or_else(overflow)?;
141        Ok(Self {
142            cpu,
143            device,
144            scratch,
145            requests,
146        })
147    }
148    pub(super) fn total(&self) -> io::Result<u64> {
149        self.cpu
150            .checked_add(self.device)
151            .and_then(|n| n.checked_add(self.scratch))
152            .ok_or_else(overflow)
153    }
154    pub(super) fn prefix(
155        bands: &[BandPlan],
156        budget: u64,
157        workers: usize,
158        rows: usize,
159    ) -> io::Result<usize> {
160        let mut count = 0;
161        for end in 1..=bands.len() {
162            if Self::new(&bands[..end], workers, rows)?.total()? > budget {
163                break;
164            }
165            count = end;
166        }
167        if count == 0 {
168            return Err(io::Error::new(
169                io::ErrorKind::OutOfMemory,
170                "Metal wave cannot fit one output band",
171            ));
172        }
173        Ok(count)
174    }
175}

202struct InitialLayout {
203    offsets: Vec<usize>,
204    capacities: Vec<usize>,
205    slot_bytes: usize,
206    command_capacity: usize,
207}
208
209impl InitialLayout {
210    fn new(bands: &[BandPlan], rows: usize) -> io::Result<Self> {
211        let mut offsets = Vec::with_capacity(bands.len());
212        let mut capacities = Vec::with_capacity(bands.len());
213        let mut slot_bytes = 0_usize;
214        let mut command_capacity = 0_usize;
215        for band in bands {
216            let capacity = band
217                .spatial_request_capacity(rows)
218                .map_err(io::Error::other)?;
219            offsets.push(slot_bytes);
220            capacities.push(capacity);
221            slot_bytes = slot_bytes
222                .checked_add(
223                    capacity
224                        .checked_mul(size_of::<SpatialTap>())
225                        .ok_or_else(overflow)?,
226                )
227                .ok_or_else(overflow)?;
228            command_capacity = command_capacity
229                .checked_add(band.spatial_command_capacity())
230                .ok_or_else(overflow)?;
231        }
232        Ok(Self {
233            offsets,
234            capacities,
235            slot_bytes,
236            command_capacity,
237        })
238    }
239}

252struct ResidualLayout {
253    capacities: [usize; 3],
254    offsets: [usize; 7],
255    slot_bytes: usize,
256    descriptor_bytes: usize,
257}
258
259impl ResidualLayout {
260    fn new(wave: &BandPlan, rows: usize) -> io::Result<Self> {
261        let capacities = wave.residual_capacities(rows).map_err(io::Error::other)?;
262        let [predictions, native, fine] = capacities;
263        // At most four correlations per Stokes-I input. The source retains its
264        // actual correlation count; unused staging capacity is never read.
265        let sizes = [
266            predictions
267                .max(1)
268                .checked_mul(size_of::<ResidualPrediction>()),
269            native.max(1).checked_mul(size_of::<NativePrediction>()),
270            fine.max(1).checked_mul(size_of::<ResidualSample>()),
271            native.max(1).checked_mul(4 * 8),
272            native.max(1).checked_mul(4 * 4),
273            native.max(1).checked_mul(4),
274            Some(8),
275        ];
276        let mut offsets = [0; 7];
277        let mut bytes = 0_usize;
278        for (i, size) in sizes.iter().enumerate() {
279            offsets[i] = bytes;
280            bytes = bytes
281                .checked_add(
282                    size.ok_or_else(overflow)?
283                        .checked_next_multiple_of(8)
284                        .ok_or_else(overflow)?,
285                )
286                .ok_or_else(overflow)?;
287        }
288        Ok(Self {
289            capacities,
290            offsets,
291            slot_bytes: bytes,
292            descriptor_bytes: offsets[3],
293        })
294    }
295    fn length(&self, index: usize) -> usize {
296        self.offsets
297            .get(index + 1)
298            .copied()
299            .unwrap_or(self.slot_bytes)
300            - self.offsets[index]
301    }
302}

318impl<'a> MetalWave<'a> {
319    #[allow(clippy::too_many_arguments)]
320    pub(super) fn new(
321        mut jobs: Vec<BandPlan>,
322        model: &'a ModelGeneration,
323        output_hz: &'a [f64],
324        polarization: &'a PolarizationOperator,
325        discover: Option<&'a mut [BandPlan]>,
326        access: MetalBatchAccess<'a>,
327        allocation: &'a AllocationId,
328        rows: usize,
329        workers: usize,
330    ) -> io::Result<Self> {
331        let residual = if jobs.first().is_some_and(BandPlan::is_residual) {
332            let wave = BandPlan::residual_wave(&jobs).map_err(io::Error::other)?;
333            let layout = ResidualLayout::new(&wave, rows)?;
334            jobs = vec![wave];
335            Some(ResidualDevice {
336                layout,
337                tickets: [None; 2],
338                next_slot: 0,
339                requested: 0,
340                unique: 0,
341                staging_bytes: 0,
342                refills: 0,
343                profile: std::env::var_os("CASA_RS_PROFILE_CUBE").is_some(),
344                host_seconds: [0.0; 6],
345                staged_bytes: [0; 6],
346                cleared_bytes: 0,
347            })
348        } else {
349            None
350        };
351        let memory = MetalWaveMemory::new(&jobs, workers, rows)?;
352        let initial = if residual.is_none() {
353            let layout = InitialLayout::new(&jobs, rows)?;
354            let chunk = jobs.len().div_ceil(workers);
355            let commands = jobs
356                .chunks(chunk)
357                .map(|bands| {
358                    Vec::with_capacity(bands.iter().map(BandPlan::spatial_command_capacity).sum())
359                })
360                .collect();
361            let batch = Vec::with_capacity(layout.command_capacity);
362            Some(InitialDevice {
363                layout,
364                commands,
365                batch,
366                tickets: [None; 2],
367                next_slot: 0,
368                predictions: jobs.iter().any(|band| !band.is_initial_zero()),
369                refills: 0,
370                grid_samples: 0,
371            })
372        } else {
373            None
374        };
375        let weights = jobs.iter().try_fold(0_usize, |sum, b| {
376            sum.checked_add(b.spatial_grid_bytes().map_err(io::Error::other)?)
377                .ok_or_else(overflow)
378        })?;
379        let packed = weights
380            .checked_add(BandPlan::spatial_weight_bytes().next_multiple_of(8))
381            .ok_or_else(overflow)?;
382        let predicted = packed
383            .checked_add(if let Some(residual) = &residual {
384                residual
385                    .layout
386                    .slot_bytes
387                    .checked_mul(2)
388                    .ok_or_else(overflow)?
389            } else {
390                initial
391                    .as_ref()
392                    .expect("initial wave layout")
393                    .layout
394                    .slot_bytes
395                    .checked_mul(2)
396                    .ok_or_else(overflow)?
397            })
398            .ok_or_else(overflow)?;
399        let mut grids = 0;
400        let jobs = jobs
401            .into_iter()
402            .map(|plan| {
403                let offset = grids;
404                grids += plan.spatial_grid_bytes().expect("checked wave grids");
405                Job {
406                    plan: Some(plan),
407                    band: None,
408                    grids: offset,
409                    fields: [0; 4],
410                    shape: [0; 2],
411                }
412            })
413            .collect();
414        Ok(Self {
415            jobs,
416            model,
417            output_hz,
418            polarization,
419            discover,
420            access,
421            allocation,
422            weights,
423            packed,
424            predicted,
425            capacity: memory.requests,
426            staging_stride: memory
427                .requests
428                .checked_mul(size_of::<Complex32>())
429                .ok_or_else(overflow)?,
430            residual,
431            initial,
432        })
433    }
434}

436struct ResidentBand<'a, 'r, 'm> {
437    access: &'a MetalBatchAccess<'r>,
438    allocation: &'m AllocationId,
439    grids: usize,
440    fields: [usize; 4],
441    shape: [usize; 2],
442    weights: usize,
443    packed: usize,
444    predicted: usize,
445    capacity: usize,
446    queued: Option<&'a mut Vec<CubeDispatch<'m>>>,
447}

449struct DispatchTarget<'a> {
450    field: SpatialField,
451    plane: usize,
452    taps: &'a [SpatialTap],
453}
454
455impl<'m> ResidentBand<'_, '_, 'm> {
456    fn region(&self, offset: usize, bytes: usize) -> MetalBufferRegion<'m> {
457        MetalBufferRegion {
458            allocation: self.allocation,
459            offset,
460            bytes,
461        }
462    }
463    fn field_region(
464        &self,
465        field: SpatialField,
466        plane: usize,
467    ) -> Result<MetalBufferRegion<'m>, SpectralOperatorError> {
468        let lane = match field {
469            SpatialField::Dirty => 0,
470            SpatialField::Residual => 1,
471            SpatialField::Psf => 2,
472            SpatialField::Model => 3,
473        };
474        if plane >= self.fields[lane] {
475            return Err(SpectralOperatorError::InvalidSlab);
476        }
477        let bytes = self.shape[0] * self.shape[1] * size_of::<Complex32>();
478        let index = self.fields[..lane].iter().sum::<usize>() + plane;
479        Ok(self.region(self.grids + index * bytes, bytes))
480    }
481    fn dispatch(
482        &mut self,
483        targets: &[DispatchTarget<'_>],
484        degrid: bool,
485    ) -> Result<(), SpectralOperatorError> {
486        let count = targets
487            .iter()
488            .try_fold(0_usize, |sum, t| sum.checked_add(t.taps.len()))
489            .ok_or(SpectralOperatorError::ResidencyOverflow)?;
490        if count > self.capacity {
491            return Err(SpectralOperatorError::ResidencyOverflow);
492        }
493        if count == 0 {
494            return Ok(());
495        }
496        let samples = self.region(self.packed, count * size_of::<SpatialTap>());
497        self.access
498            .with_bytes(samples, |bytes| {
499                let mut offset = 0;
500                for target in targets {
501                    let input = bytemuck::cast_slice(target.taps);
502                    bytes[offset..offset + input.len()].copy_from_slice(input);
503                    offset += input.len();
504                }
505            })
506            .map_err(spatial_error)?;
507        let mut offset = 0;
508        let mut commands = Vec::with_capacity(targets.len());
509        for target in targets.iter().filter(|t| !t.taps.is_empty()) {
510            commands.push(CubeDispatch {
511                kind: if degrid {
512                    CubeDispatchKind::Degrid {
513                        predicted: self.region(
514                            self.predicted + offset * size_of::<Complex32>(),
515                            target.taps.len() * size_of::<Complex32>(),
516                        ),
517                    }
518                } else {
519                    CubeDispatchKind::Grid
520                },
521                samples: self.region(
522                    self.packed + offset * size_of::<SpatialTap>(),
523                    size_of_val(target.taps),
524                ),
525                weights: self.region(self.weights, BandPlan::spatial_weight_bytes()),
526                grid: self.field_region(target.field, target.plane)?,
527                count: target
528                    .taps
529                    .len()
530                    .try_into()
531                    .map_err(|_| SpectralOperatorError::ResidencyOverflow)?,
532                width: self.shape[0]
533                    .try_into()
534                    .map_err(|_| SpectralOperatorError::ResidencyOverflow)?,
535                height: self.shape[1]
536                    .try_into()
537                    .map_err(|_| SpectralOperatorError::ResidencyOverflow)?,
538            });
539            offset += target.taps.len();
540        }
541        if !degrid && let Some(queued) = self.queued.as_deref_mut() {
542            queued.extend(commands);
543        } else {
544            self.access.execute(&commands).map_err(spatial_error)?;
545        }
546        Ok(())
547    }
548}

554impl CubeSpatialBackend for ResidentBand<'_, '_, '_> {
555    fn initialize(
556        &mut self,
557        shape: [usize; 2],
558        weights: &[[f32; 7]],
559        fields: [usize; 4],
560        model: &[Complex32],
561    ) -> Result<(), SpectralOperatorError> {
562        self.shape = shape;
563        self.fields = fields;
564        self.access
565            .with_bytes(self.region(self.weights, size_of_val(weights)), |bytes| {
566                bytes.copy_from_slice(bytemuck::cast_slice(weights))
567            })
568            .map_err(spatial_error)?;
569        let cells = shape[0] * shape[1];
570        let outputs = fields[..3].iter().sum::<usize>();
571        if outputs > 0 {
572            self.access
573                .with_bytes(self.region(self.grids, outputs * cells * 8), |bytes| {
574                    bytes.fill(0)
575                })
576                .map_err(spatial_error)?;
577        }
578        if !model.is_empty() {
579            self.access
580                .with_bytes(
581                    self.region(self.grids + outputs * cells * 8, size_of_val(model)),
582                    |bytes| bytes.copy_from_slice(bytemuck::cast_slice(model)),
583                )
584                .map_err(spatial_error)?;
585        }
586        Ok(())
587    }
588    fn degrid(
589        &mut self,
590        batches: &mut [SpatialPredictionBatch],
591    ) -> Result<(), SpectralOperatorError> {
592        let targets: Vec<_> = batches
593            .iter()
594            .map(|b| DispatchTarget {
595                field: SpatialField::Model,
596                plane: b.plane,
597                taps: &b.taps,
598            })
599            .collect();
600        self.dispatch(&targets, true)?;
601        let mut offset = 0;
602        for batch in batches.iter_mut().filter(|b| !b.taps.is_empty()) {
603            self.access
604                .with_bytes(
605                    self.region(
606                        self.predicted + offset * size_of::<Complex32>(),
607                        size_of_val(batch.values.as_slice()),
608                    ),
609                    |bytes| {
610                        bytemuck::cast_slice_mut::<Complex32, u8>(&mut batch.values)
611                            .copy_from_slice(bytes);
612                    },
613                )
614                .map_err(spatial_error)?;
615            offset += batch.taps.len();
616        }
617        Ok(())
618    }
619    fn grid(&mut self, batches: &[SpatialGridBatch]) -> Result<(), SpectralOperatorError> {
620        let targets: Vec<_> = batches
621            .iter()
622            .map(|b| DispatchTarget {
623                field: b.field,
624                plane: b.plane,
625                taps: &b.taps,
626            })
627            .collect();
628        self.dispatch(&targets, false)
629    }
630    fn download(
631        &mut self,
632        field: SpatialField,
633        plane: usize,
634        values: &mut [Complex32],
635    ) -> Result<(), SpectralOperatorError> {
636        self.access
637            .with_bytes(self.field_region(field, plane)?, |bytes| {
638                bytemuck::cast_slice_mut::<Complex32, u8>(values).copy_from_slice(bytes)
639            })
640            .map_err(spatial_error)
641    }
642}

644impl BulkConsumer for MetalWave<'_> {
645    type Completion = Vec<BandResult>;
646    fn consume(
647        &mut self,
648        block: NativeBlockView<'_>,
649        layout: &NativeLayout,
650        selected: std::ops::Range<usize>,
651        execution: BoundedExecution<'_>,
652    ) -> io::Result<()> {
653        if let Some(bands) = self.discover.as_deref_mut() {
654            BandPlan::observe_borrowed(bands, block, self.output_hz).map_err(io::Error::other)?;
655        }
656        let model = self.model;
657        execution.for_each_mut(&mut self.jobs, |_, job| {
658            if let Some(plan) = job.plan.take() {
659                job.band = Some(plan.prepare(model, None).map_err(io::Error::other)?);
660            }
661            Ok::<_, io::Error>(())
662        })?;
663        if self.residual.is_some() {
664            return self.consume_residual(block, layout, selected);
665        }
666        let access = &self.access;
667        let allocation = self.allocation;
668        let output_hz = self.output_hz;
669        let polarization = self.polarization;
670        let weights = self.weights;
671        let predicted = self.predicted;
672        let staging_stride = self.staging_stride;
673        let initial = self.initial.as_mut().expect("initial device state");
674        let slot = initial.next_slot;
675        // A nonempty initial model needs synchronous host prediction readback.
676        // Settle its queued grids before that dependency, never reuse stale tickets.
677        if initial.predictions {
678            for ticket in &mut initial.tickets {
679                if let Some(ticket) = ticket.take() {
680                    access.wait(ticket).map_err(io::Error::other)?;
681                }
682            }
683        } else if let Some(ticket) = initial.tickets[slot].take() {
684            access.wait(ticket).map_err(io::Error::other)?;
685        }
686        let packed = self.packed + slot * initial.layout.slot_bytes;
687        let offsets = &initial.layout.offsets;
688        let capacities = &initial.layout.capacities;
689        for commands in &mut initial.commands {
690            commands.clear();
691        }
692        let chunk = self.jobs.len().div_ceil(execution.worker_count());
693        let mut lanes: Vec<_> = self
694            .jobs
695            .chunks_mut(chunk)
696            .zip(initial.commands.iter_mut())
697            .enumerate()
698            .collect();
699        execution.for_each_mut(&mut lanes, |_, (lane, (jobs, commands))| {
700            for (index, job) in jobs.iter_mut().enumerate() {
701                let ordinal = *lane * chunk + index;
702                let mut backend = ResidentBand {
703                    access,
704                    allocation,
705                    grids: job.grids,
706                    fields: job.fields,
707                    shape: job.shape,
708                    weights,
709                    packed: packed + offsets[ordinal],
710                    predicted: predicted + *lane * staging_stride,
711                    capacity: capacities[ordinal],
712                    queued: Some(&mut **commands),
713                };
714                let band = job.band.as_mut().expect("prepared band");
715                if job.shape == [0; 2] {
716                    band.initialize_spatial(&mut backend)
717                        .map_err(io::Error::other)?;
718                    job.shape = backend.shape;
719                    job.fields = backend.fields;
720                }
721                band.consume_source_window_spatial(
722                    block,
723                    layout,
724                    selected.clone(),
725                    output_hz,
726                    polarization,
727                    &mut backend,
728                )
729                .map_err(io::Error::other)?;
730            }
731            Ok::<_, io::Error>(())
732        })?;
733        drop(lanes);
734        initial.batch.clear();
735        for commands in &initial.commands {
736            initial.batch.extend_from_slice(commands);
737        }
738        if !initial.batch.is_empty() {
739            let ticket = access.submit(&initial.batch).map_err(io::Error::other)?;
740            initial.tickets[slot] = Some(ticket);
741            initial.next_slot = (slot + 1) % 2;
742            initial.refills += 1;
743            initial.grid_samples += initial
744                .batch
745                .iter()
746                .map(|dispatch| u64::from(dispatch.count))
747                .sum::<u64>();
748        }
749        Ok(())
750    }
751    fn complete(mut self, execution: BoundedExecution<'_>) -> io::Result<Vec<BandResult>> {
752        let drain_started = self
753            .residual
754            .as_ref()
755            .is_some_and(|state| state.profile)
756            .then(Instant::now);
757        self.access.drain().map_err(io::Error::other)?;
758        let drain_seconds = drain_started.map_or(0.0, |started| started.elapsed().as_secs_f64());
759        if let Some(initial) = &self.initial {
760            eprintln!(
761                "cube device initial: refills={} grid_samples={} grid_staging_bytes={} slots=2",
762                initial.refills,
763                initial.grid_samples,
764                initial.grid_samples * size_of::<SpatialTap>() as u64
765            );
766        }
767        if let Some(residual) = &self.residual {
768            eprintln!(
769                "cube device residual: refills={} requested_predictions={} unique_predictions={} staging_bytes={} host_prediction_bytes=0 slots=2",
770                residual.refills, residual.requested, residual.unique, residual.staging_bytes
771            );
772            if residual.profile {
773                eprintln!(
774                    "cube residual host profile: refills={} slot_wait_seconds={:.6} prepare_seconds={:.6} copy_seconds={:.6} flag_pack_seconds={:.6} clear_seconds={:.6} submit_seconds={:.6} drain_seconds={:.6} prediction_bytes={} native_bytes={} sample_bytes={} value_bytes={} weight_bytes={} flag_bytes={} cleared_bytes={} descriptor_copy_bytes=0",
775                    residual.refills,
776                    residual.host_seconds[0],
777                    residual.host_seconds[1],
778                    residual.host_seconds[2],
779                    residual.host_seconds[3],
780                    residual.host_seconds[4],
781                    residual.host_seconds[5],
782                    drain_seconds,
783                    residual.staged_bytes[0],
784                    residual.staged_bytes[1],
785                    residual.staged_bytes[2],
786                    residual.staged_bytes[3],
787                    residual.staged_bytes[4],
788                    residual.staged_bytes[5],
789                    residual.cleared_bytes,
790                );
791            }
792        }
793        for job in &mut self.jobs {
794            if let Some(plan) = job.plan.take() {
795                job.band = Some(plan.prepare(self.model, None).map_err(io::Error::other)?);
796            }
797            let mut backend = ResidentBand {
798                access: &self.access,
799                allocation: self.allocation,
800                grids: job.grids,
801                fields: job.fields,
802                shape: job.shape,
803                weights: self.weights,
804                packed: self.packed,
805                predicted: self.predicted,
806                capacity: self.capacity,
807                queued: None,
808            };
809            let band = job.band.as_mut().expect("prepared band");
810            if job.shape == [0; 2] {
811                band.initialize_spatial(&mut backend)
812                    .map_err(io::Error::other)?;
813            }
814            band.download_spatial(&mut backend)
815                .map_err(io::Error::other)?;
816        }
817        let mut results: Vec<Option<BandResult>> = (0..self.jobs.len()).map(|_| None).collect();
818        let mut jobs: Vec<_> = std::mem::take(&mut self.jobs)
819            .into_iter()
820            .zip(results.iter_mut())
821            .collect();
822        let model = self.model;
823        execution.for_each_mut(&mut jobs, |_, (job, result)| {
824            **result = Some(
825                job.band
826                    .take()
827                    .expect("active band")
828                    .complete(model)
829                    .map_err(io::Error::other)?
830                    .0,
831            );
832            Ok::<_, io::Error>(())
833        })?;
834        Ok(results.into_iter().map(Option::unwrap).collect())
835    }
836}

838impl MetalWave<'_> {
839    fn consume_residual(
840        &mut self,
841        block: NativeBlockView<'_>,
842        layout: &NativeLayout,
843        selected: std::ops::Range<usize>,
844    ) -> io::Result<()> {
845        let job = &mut self.jobs[0];
846        let band = job.band.as_mut().expect("prepared residual wave");
847        let mut backend = ResidentBand {
848            access: &self.access,
849            allocation: self.allocation,
850            grids: job.grids,
851            fields: job.fields,
852            shape: job.shape,
853            weights: self.weights,
854            packed: self.packed,
855            predicted: self.predicted,
856            capacity: self.capacity,
857            queued: None,
858        };
859        if job.shape == [0; 2] {
860            band.initialize_spatial(&mut backend)
861                .map_err(io::Error::other)?;
862            job.shape = backend.shape;
863            job.fields = backend.fields;
864        }
865        let state = self.residual.as_mut().expect("connected residual state");
866        let slot = state.next_slot;
867        let started = state.profile.then(Instant::now);
868        if let Some(ticket) = state.tickets[slot].take() {
869            self.access.wait(ticket).map_err(io::Error::other)?;
870        }
871        if let Some(started) = started {
872            state.host_seconds[0] += started.elapsed().as_secs_f64();
873        }
874        let base = self.packed + slot * state.layout.slot_bytes;
875        let started = state.profile.then(Instant::now);
876        let (actual, requested) = self
877            .access
878            .with_bytes(
879                MetalBufferRegion {
880                    allocation: self.allocation,
881                    offset: base,
882                    bytes: state.layout.descriptor_bytes,
883                },
884                |bytes| -> io::Result<_> {
885                    let (predictions, tail) = bytes.split_at_mut(state.layout.offsets[1]);
886                    let (native, samples) =
887                        tail.split_at_mut(state.layout.offsets[2] - state.layout.offsets[1]);
888                    let predictions =
889                        bytemuck::try_cast_slice_mut::<u8, ResidualPrediction>(predictions)
890                            .map_err(|error| {
891                                io::Error::other(format!(
892                                    "invalid mapped prediction storage: {error:?}"
893                                ))
894                            })?;
895                    let native = bytemuck::try_cast_slice_mut::<u8, NativePrediction>(native)
896                        .map_err(|error| {
897                            io::Error::other(format!("invalid mapped native storage: {error:?}"))
898                        })?;
899                    let samples = bytemuck::try_cast_slice_mut::<u8, ResidualSample>(samples)
900                        .map_err(|error| {
901                            io::Error::other(format!("invalid mapped sample storage: {error:?}"))
902                        })?;
903                    let mut refill = ResidualRefill {
904                        predictions: &mut predictions[..state.layout.capacities[0]],
905                        native: &mut native[..state.layout.capacities[1]],
906                        samples: &mut samples[..state.layout.capacities[2]],
907                        counts: [0; 3],
908                        requested_predictions: 0,
909                    };
910                    band.prepare_residual_refill(
911                        block,
912                        layout,
913                        selected,
914                        self.output_hz,
915                        &mut refill,
916                    )
917                    .map_err(io::Error::other)?;
918                    Ok((refill.counts, refill.requested_predictions))
919                },
920            )
921            .map_err(io::Error::other)??;
922        if let Some(started) = started {
923            state.host_seconds[1] += started.elapsed().as_secs_f64();
924        }
925        if actual
926            .iter()
927            .zip(state.layout.capacities)
928            .any(|(&n, cap)| n > cap)
929        {
930            return Err(io::Error::other(
931                "connected residual refill exceeds admitted capacity",
932            ));
933        }
934        if actual[2] == 0 {
935            return Ok(());
936        }
937        for (i, bytes) in [
938            actual[0] * size_of::<ResidualPrediction>(),
939            actual[1] * size_of::<NativePrediction>(),
940            actual[2] * size_of::<ResidualSample>(),
941        ]
942        .into_iter()
943        .enumerate()
944        {
945            state.staging_bytes += bytes as u64;
946            if state.profile {
947                state.staged_bytes[i] += bytes as u64;
948            }
949        }
950        let (values, input_weights, flags, weight_flags) = block.sample_arrays();
951        let slot_region = |i: usize| MetalBufferRegion {
952            allocation: self.allocation,
953            offset: base + state.layout.offsets[i],
954            bytes: state.layout.length(i),
955        };
956        let started = state.profile.then(Instant::now);
957        for (i, bytes) in [
958            bytemuck::cast_slice(values),
959            bytemuck::cast_slice(input_weights),
960        ]
961        .into_iter()
962        .enumerate()
963        {
964            let i = i + 3;
965            if bytes.is_empty() {
966                continue;
967            }
968            self.access
969                .with_bytes(slot_region(i), |target| {
970                    target[..bytes.len()].copy_from_slice(bytes)
971                })
972                .map_err(io::Error::other)?;
973            state.staging_bytes += bytes.len() as u64;
974            if state.profile {
975                state.staged_bytes[i] += bytes.len() as u64;
976            }
977        }
978        if let Some(started) = started {
979            state.host_seconds[2] += started.elapsed().as_secs_f64();
980        }
981        let started = state.profile.then(Instant::now);
982        self.access
983            .with_bytes(slot_region(5), |target| {
984                for ((target, &flag), &weight_flag) in
985                    target.iter_mut().zip(flags).zip(weight_flags)
986                {
987                    *target = u8::from(flag) | (u8::from(weight_flag) << 1);
988                }
989            })
990            .map_err(io::Error::other)?;
991        state.staging_bytes += flags.len() as u64;
992        if let Some(started) = started {
993            state.host_seconds[3] += started.elapsed().as_secs_f64();
994            state.staged_bytes[5] += flags.len() as u64;
995        }
996        let started = state.profile.then(Instant::now);
997        self.access
998            .with_bytes(slot_region(6), |target| target.fill(0))
999            .map_err(io::Error::other)?;
1000        if let Some(started) = started {
1001            state.host_seconds[4] += started.elapsed().as_secs_f64();
1002            state.cleared_bytes += slot_region(6).bytes as u64;
1003        }
1004        let models = backend
1005            .field_region(SpatialField::Model, 0)
1006            .map_err(io::Error::other)?;
1007        let residual = backend
1008            .field_region(SpatialField::Residual, 0)
1009            .map_err(io::Error::other)?;
1010        let regions = [
1011            slot_region(0),
1012            slot_region(1),
1013            slot_region(2),
1014            slot_region(3),
1015            slot_region(4),
1016            slot_region(5),
1017            backend.region(self.weights, BandPlan::spatial_weight_bytes()),
1018            MetalBufferRegion {
1019                bytes: models.bytes * job.fields[3],
1020                ..models
1021            },
1022            backend.region(self.predicted, state.layout.capacities[0].max(1) * 8),
1023            MetalBufferRegion {
1024                bytes: residual.bytes * job.fields[1],
1025                ..residual
1026            },
1027            slot_region(6),
1028        ];
1029        let shape = [
1030            actual[0],
1031            actual[2],
1032            actual[1],
1033            job.fields[3],
1034            job.shape[0],
1035            job.shape[1],
1036            job.fields[1],
1037            BandPlan::spatial_weight_bytes() / 28,
1038        ]
1039        .map(|n| u32::try_from(n).map_err(|_| overflow()));
1040        let shape: Vec<_> = shape.into_iter().collect::<io::Result<_>>()?;
1041        let started = state.profile.then(Instant::now);
1042        let ticket = self
1043            .access
1044            .submit_residual(CubeResidualDispatch {
1045                regions,
1046                shape: shape.try_into().expect("eight shape entries"),
1047                correlations: DeviceCorrelations::new(self.polarization)
1048                    .map_err(io::Error::other)?,
1049            })
1050            .map_err(io::Error::other)?;
1051        if let Some(started) = started {
1052            state.host_seconds[5] += started.elapsed().as_secs_f64();
1053        }
1054        state.tickets[slot] = Some(ticket);
1055        state.next_slot = (slot + 1) % 2;
1056        state.requested += requested;
1057        state.unique += actual[0] as u64;
1058        state.refills += 1;
1059        Ok(())
1060    }
1061}

1064    fn drop(&mut self) {
1065        // Source/descriptor failures still drain commands before their arena is released.
1066        let _ = self.access.drain();
1067    }
