29/// Matching tolerances and text used by the tool's HISTORY record.
30pub struct TestConcatenation<'a> {
31    /// Frequency tolerance, in hertz.
32    pub frequency_hz: f64,
33    /// Direction tolerance, in radians.
34    pub direction_rad: f64,
35    /// Merge fields only when their names also agree.
36    pub respect_name: bool,
37    /// Frequency quantity as rendered for CASA's HISTORY command.
38    pub frequency_text: &'a str,
39    /// Direction quantity as rendered for CASA's HISTORY command.
40    pub direction_text: &'a str,
41    /// Donor pathname supplied by the caller.
42    pub donor_name: &'a str,
43    /// The attached path shown by the tool, including any symbolic alias.
44    pub attached_name: &'a Path,
45}

47/// Reconcile the metadata of two MeasurementSets and write the donor history.
48///
49/// The attached table retains its MAIN and POINTING rows. Both inputs are
50/// locked while the subtable changes are prepared.
51///
52/// # Errors
53/// Returns an invalid-input error for unsupported schemas, invalid fields or
54/// tolerances, and reports failures to lock, write, or publish the subtables.
55pub fn test_concatenate(
56    target: &Path,
57    donor: &Path,
58    request: &TestConcatenation<'_>,
59    retained: &mut BTreeMap<PathBuf, TableWriteLock>,
60) -> MsResult<()> {
61    publication::metadata_only(target, donor, request, retained)
62}

64/// Settings for a physical MeasurementSet concatenation.
65pub struct PhysicalConcatenation<'a> {
66    /// Metadata matching tolerances and HISTORY arguments.
67    pub metadata: TestConcatenation<'a>,
68    /// Scale donor weights and apply the reciprocal square root to sigmas.
69    pub weight_scale: f32,
70    /// CASA mode 0 through 3 controlling which of MAIN and POINTING are added.
71    pub handling: i32,
72    /// A separate table for remapped donor MAIN rows, if requested.
73    pub output: Option<&'a Path>,
74}
75
76/// Reconcile metadata and append the donor's physical MAIN and POINTING.
77///
78/// A named output receives donor MAIN rows only, with copies of the
79/// reconciled subtables. The attached MAIN is extended for an in-place call.
80///
81/// # Errors
82/// The inputs cannot be matched, locked, staged, or published.
83pub fn concatenate(
84    target: &Path,
85    donor: &Path,
86    request: &PhysicalConcatenation<'_>,
87    retained: &mut BTreeMap<PathBuf, TableWriteLock>,
88) -> MsResult<()> {
89    if !matches!(request.handling, 0..=3) {
90        return Err(MsError::InvalidInput(
91            "concatenation handling must be in 0..=3".into(),
92        ));
93    }
94    if request.weight_scale < 0.0 {
95        return Err(MsError::InvalidInput(
96            "concatenation weight scale must be nonnegative".into(),
97        ));
98    }
99    publication::physical(target, donor, request, retained)
100}

102/// Settings for preparing a donor MeasurementSet for virtual concatenation.
103pub struct VirtualConcatenation<'a> {
104    /// Metadata matching tolerances and HISTORY arguments.
105    pub metadata: TestConcatenation<'a>,
106    /// Scale the incoming visibility weights and sigmas.
107    pub weight_scale: f32,
108    /// Optional scan bookkeeping shared across successive calls.
109    pub scan_file: Option<&'a Path>,
110    /// Physical row numbers in an attached working selection, if any.
111    pub attached_rows: Option<&'a [usize]>,
112}

114/// Reconcile attached metadata and rewrite the donor IDs and visibilities
115/// for use in a subsequent concatenation of the two MeasurementSets.
116///
117/// # Errors
118/// An input or auxiliary file is invalid, or an edit cannot be published.
119pub fn virtual_concatenate(
120    target: &Path,
121    donor: &Path,
122    request: &VirtualConcatenation<'_>,

127            "concatenation weight scale must be nonnegative".into(),
128        ));
129    }
130    publication::virtual_concat(target, donor, request, retained)
131}

133struct MetadataPlan<'a> {
134    changes: Vec<SubtableChange>,
135    row_map: MainRowMap,
136    calibration: Vec<(&'a Table, TableRowsChange)>,
137    ephemerides: Vec<(PathBuf, String)>,
138    add_ephemeris_column: bool,
139}
140
141struct MainRowMap {
142    antennas: Vec<usize>,
143    descriptions: Vec<spectral::DonorDescription>,
144    fields: Vec<i32>,
145    states: Vec<usize>,
146    attached_has_state: bool,
147    donor_has_state: bool,
148    observations: Vec<usize>,
149    processors: Vec<usize>,
150    attached_observations: Vec<usize>,
151    attached_processors: Vec<usize>,
152}
153
154struct SubtableChange {
155    table: SubtableId,
156    rows: TableRowsChange,
157}
158
159#[derive(Default)]
160struct TableRowsChange {
161    appended: Vec<RecordValue>,
162    updated: Vec<(usize, RecordValue)>,
163    removed: Vec<usize>,
164}
165
166impl<'a> MetadataPlan<'a> {
167    fn reconcile(
168        target: &'a MeasurementSet,
169        donor: &MeasurementSet,
170        calibration_tables: [&'a BTreeMap<&str, Table>; 2],
171        frequency_hz: f64,
172        direction_rad: f64,
173        respect_name: bool,
174        coalesce_attached: bool,
175    ) -> MsResult<Self> {
176        let states = identity::states(
177            required_table(target, SubtableId::State)?,
178            required_table(donor, SubtableId::State)?,
179        )?;
180        let observations = identity::observations(
181            required_table(target, SubtableId::Observation)?,
182            required_table(donor, SubtableId::Observation)?,
183            coalesce_attached,
184        )?;
185        let processors = identity::processors(
186            required_table(target, SubtableId::Processor)?,
187            required_table(donor, SubtableId::Processor)?,
188            coalesce_attached,
189        )?;
190        let spectral = spectral::reconcile(target, donor, frequency_hz)?;
191        let antennas = antenna::reconcile(target, donor, &spectral.window_ids)?;
192        let mut sources = source::reconcile(target, donor, &spectral.window_ids)?;
193        let fields = field::reconcile(target, donor, &sources.ids, direction_rad, respect_name)?;
194        if fields.matched {
195            sources.consolidate(target, &fields.additions)?;
196        }
197        let calibration = calibration::reconcile(
198            target,
199            donor,
200            calibration_tables,
201            &antennas.antenna_ids,
202            &spectral.window_ids,
203            !spectral.windows.is_empty(),
204            [&observations.attached_ids, &observations.donor_ids],
205        )?;
206        let mut plan = Self {
207            changes: Vec::new(),
208            row_map: MainRowMap {
209                antennas: antennas.antenna_ids,
210                descriptions: spectral.donor_descriptions,
211                fields: fields.ids,
212                states: states.donor_ids,
213                attached_has_state: required_table(target, SubtableId::State)?.row_count() > 0,
214                donor_has_state: required_table(donor, SubtableId::State)?.row_count() > 0,
215                observations: observations.donor_ids,
216                processors: processors.donor_ids,
217                attached_observations: observations.attached_ids,
218                attached_processors: processors.attached_ids,
219            },
220            calibration,
221            ephemerides: fields.ephemerides,
222            add_ephemeris_column: fields.add_ephemeris_column,
223        };
224        plan.add(SubtableId::State, states.additions, Vec::new());
225        plan.add(SubtableId::SpectralWindow, spectral.windows, Vec::new());
226        plan.add(SubtableId::Polarization, spectral.polarizations, Vec::new());
227        plan.add(
228            SubtableId::DataDescription,
229            spectral.descriptions,
230            Vec::new(),
231        );
232        plan.add(SubtableId::Antenna, antennas.antennas, Vec::new());
233        plan.add(SubtableId::Feed, antennas.feeds, antennas.feed_updates);
234        if !sources.additions.is_empty()
235            || !sources.updates.is_empty()
236            || !sources.removed.is_empty()
237        {
238            plan.changes.push(SubtableChange {
239                table: SubtableId::Source,
240                rows: TableRowsChange {
241                    appended: sources.additions,
242                    updated: sources.updates,
243                    removed: sources.removed,
244                },
245            });
246        }
247        plan.add(SubtableId::Field, fields.additions, Vec::new());
248        for (table, appended, removed) in [
249            (
250                SubtableId::Observation,
251                observations.additions,
252                observations.removed,
253            ),
254            (
255                SubtableId::Processor,
256                processors.additions,
257                processors.removed,
258            ),
259        ] {
260            if !appended.is_empty() || !removed.is_empty() {
261                plan.changes.push(SubtableChange {
262                    table,
263                    rows: TableRowsChange {
264                        appended,
265                        removed,
266                        ..TableRowsChange::default()
267                    },
268                });
269            }
270        }
271        Ok(plan)
272    }
273
274    fn add(
275        &mut self,
276        table: SubtableId,
277        appended: Vec<RecordValue>,
278        updated: Vec<(usize, RecordValue)>,
279    ) {
280        if !appended.is_empty()
281            || !updated.is_empty()
282            || (table == SubtableId::Field && self.add_ephemeris_column)
283        {
284            self.changes.push(SubtableChange {
285                table,
286                rows: TableRowsChange {
287                    appended,
288                    updated,
289                    ..TableRowsChange::default()
290                },
291            });
292        }
293    }
294}

296fn required_table(ms: &MeasurementSet, table: SubtableId) -> MsResult<&Table> {
297    ms.subtable(table)
298        .ok_or_else(|| MsError::MissingSubtable(table.name().into()))
299}

301pub(super) fn join_with_tracing<A, B, RA, RB>(left: A, right: B) -> (RA, RB)
302where
303    A: FnOnce() -> RA + Send,
304    B: FnOnce() -> RB + Send,
305    RA: Send,
306    RB: Send,
307{
308    let dispatch = tracing::dispatcher::get_default(Clone::clone);
309    rayon::join(
310        || tracing::dispatcher::with_default(&dispatch, left),
311        || tracing::dispatcher::with_default(&dispatch, right),
312    )
