16/// The group of every time and where each group starts and ends, computed
17/// once at binding from the grid's times, fields and scans: the traversal's
18/// group index. It holds a 32-byte key per time and an extent per group;
19/// the trial accepts it in exchange for proven closure at delivery.
20///
21/// Keys are those of the grid's field-selected times, so traversals of
22/// different time selections agree on every time's group, and bins count
23/// from the first selected time of their key ([`GroupKey`]). Each group's
24/// extent is the part of it the
25/// traversal's time selection holds, so fragments are positioned and closed
26/// within it and marked `clipped` when it cuts the group.
27#[derive(Debug)]
28pub(super) struct GroupIndex {
29    /// Each grid time's key; `None` where the field selection excludes it.
30    keys: Vec<Option<GroupKey>>,
31    /// Each group's first time within the selection and the time past its
32    /// last.
33    extent: BTreeMap<GroupKey, Range<usize>>,
34    /// The groups the time selection cuts: empty for the whole grid.
35    clipped: BTreeSet<GroupKey>,
36    closure: Closure,
37    /// Groups whose last time was delivered under deferred closure, still
38    /// to be announced.
39    pending_ends: VecDeque<GroupKey>,
40}
41
42/// The group of each of the grid times `times` (seconds, ascending) with
43/// `fields` and `scans` (when the store names scans) at each, keyed as
44/// production `gaincal` keys solutions ([`GroupKey`]) over the times
45/// `selection` holds; `None` for the times it excludes, and `None` overall
46/// when `grouping` asks for no groups.
47pub(super) fn group_keys(
48    partition: PartitionId,
49    grouping: Grouping,
50    times: &[f64],
51    fields: &[FieldId],
52    scans: Option<&[ScanId]>,
53    selection: &FieldSelection,
54) -> Result<Option<Vec<Option<GroupKey>>>, ObservationError> {
55    let keyed_by_scan = match grouping {
56        Grouping::None => return Ok(None),
57        Grouping::TimeBins { combine_scans, .. } => !combine_scans,
58        Grouping::PerScan { .. } | Grouping::PerIntegration => true,
59    };
60    if fields.len() != times.len() || scans.is_some_and(|scans| scans.len() != times.len()) {
61        return Err(ObservationError::InvalidStore {
62            reason: "the grid has not one field and scan per time".to_string(),
63        });
64    }
65    if keyed_by_scan && scans.is_none() {
66        return Err(ObservationError::InvalidRequest {
67            reason: "the grouping keys by scan, and the store names none".to_string(),
68        });
69    }
70    if let Grouping::TimeBins { width_s, .. } = grouping
71        && width_s.partial_cmp(&0.0) != Some(std::cmp::Ordering::Greater)
72    {
73        return Err(ObservationError::InvalidRequest {
74            reason: format!("time-bin width {width_s} s is not positive"),
75        });
76    }
77    if times.windows(2).any(|pair| pair[1] < pair[0]) {
78        return Err(ObservationError::InvalidStore {
79            reason: "the grid's times are not ascending".to_string(),
80        });
81    }
82    // Production gaincal's rule: each (field, scan) key's bins count from
83    // its first time; each key's integrations count from zero.
84    let mut origins = BTreeMap::<(Option<FieldId>, Option<ScanId>), (f64, u64)>::new();
85    let mut keys = Vec::with_capacity(times.len());
86    for (unit, (&time, &field)) in times.iter().zip(fields).enumerate() {
87        if !selection.holds(field) {
88            keys.push(None);
89            continue;
90        }
91        // Present whenever the grouping keys by it, checked above.
92        let scan = scans.map(|scans| scans[unit]).filter(|_| keyed_by_scan);
93        let field = match grouping {
94            Grouping::None => unreachable!("handled above"),
95            Grouping::TimeBins { combine_fields, .. } | Grouping::PerScan { combine_fields } => {
96                (!combine_fields).then_some(field)
97            }
98            Grouping::PerIntegration => Some(field),
99        };
100        let (origin, count) = origins.entry((field, scan)).or_insert((time, 0));
101        let bin = match grouping {
102            Grouping::TimeBins { width_s, .. } => {
103                ((time - *origin) / width_s).floor().max(0.0) as u64
104            }
105            Grouping::PerScan { .. } => 0,
106            Grouping::PerIntegration => {
107                let bin = *count;
108                *count += 1;
109                bin
110            }
111            Grouping::None => unreachable!("handled above"),
112        };
113        keys.push(Some(GroupKey {
114            partition,
115            field,
116            scan,
117            bin,
118        }));
119    }
120    Ok(Some(keys))
121}

123impl GroupIndex {
124    /// Index the grid times `times` (seconds, ascending) with `fields` and
125    /// `scans` (when the store names scans) at each, for a traversal of
126    /// `request` that delivers the grid times `selection` (the times its
127    /// field selection holds within its time selection); or `None` when
128    /// the request asks for no groups.
129    pub(super) fn new(
130        partition: PartitionId,
131        request: &AccessRequest,
132        closure: Closure,
133        times: &[f64],
134        fields: &[FieldId],
135        scans: Option<&[ScanId]>,
136        selection: &TimeRuns,
137    ) -> Result<Option<Self>, ObservationError> {
138        let Some(keys) = group_keys(
139            partition,
140            request.grouping,
141            times,
142            fields,
143            scans,
144            &request.fields,
145        )?
146        else {
147            return Ok(None);
148        };
149        let mut extent = BTreeMap::<GroupKey, Range<usize>>::new();
150        for run in selection.runs() {
151            for (unit, key) in keys.iter().enumerate().take(run.end).skip(run.start) {
152                let key = key.expect("a selected time has its field selected");
153                extent
154                    .entry(key)
155                    .and_modify(|range| range.end = unit + 1)
156                    .or_insert(unit..unit + 1);
157            }
158        }
159        // The time selection cuts a group it holds that has a field-selected
160        // time outside it.
161        let mut clipped = BTreeSet::new();
162        let mut held = selection.runs().iter().peekable();
163        for (unit, key) in keys.iter().enumerate() {
164            let Some(key) = key else {
165                continue;
166            };
167            while held.next_if(|run| run.end <= unit).is_some() {}
168            let selected = held.peek().is_some_and(|run| run.start <= unit);
169            if !selected && extent.contains_key(key) {
170                clipped.insert(*key);
171            }
172        }
173        Ok(Some(Self {
174            keys,
175            extent,
176            clipped,
177            closure,
178            pending_ends: VecDeque::new(),
179        }))
180    }
181
182    /// The next group announcement owed before further data, if any.
183    pub(super) fn pop_end(&mut self) -> Option<GroupKey> {
184        self.pending_ends.pop_front()
185    }
186
187    /// Append the fragments of times `times` to `fragments`, with sample
188    /// ranges of `samples_per_time` samples per time, counted from the
189    /// batch's first time.
190    pub(super) fn fragments(
191        &mut self,
192        times: Range<usize>,
193        samples_per_time: usize,
194        fragments: &mut Vec<GroupFragment>,
195    ) {
196        let mut start = times.start;
197        while start < times.end {
198            let key = self.keys[start].expect("a delivered time has its field selected");
199            let mut end = start + 1;
200            while end < times.end && self.keys[end] == Some(key) {
201                end += 1;
202            }
203            let extent = &self.extent[&key];
204            let first = start == extent.start;
205            let last = end == extent.end;
206            let position = match (first, last, self.closure) {
207                (true, true, Closure::AtDelivery) => FragmentPosition::Whole,
208                (true, _, _) => FragmentPosition::First,
209                (false, true, Closure::AtDelivery) => FragmentPosition::Last,
210                (false, _, _) => FragmentPosition::Middle,
211            };
212            if last && self.closure == Closure::Deferred {
213                self.pending_ends.push_back(key);
214            }
215            fragments.push(GroupFragment {
216                key,
217                samples: (start - times.start) * samples_per_time
218                    ..(end - times.start) * samples_per_time,
219                position,
220                clipped: self.clipped.contains(&key),
221            });
222            start = end;
223        }
224    }
225}

249    /// Each time's key, which every time has under [`FieldSelection::All`].

254    fn index(grouping: Grouping) -> GroupIndex {
255        // Two fields alternating, one scan then another: field A, A, B, B,
256        // A in scan 1, then B, B in scan 2, ten seconds apart.
257        let times: Vec<f64> = (0..7).map(|t| f64::from(t) * 10.0).collect();
258        let fields = [0, 0, 1, 1, 0, 1, 1].map(FieldId);
259        let scans = [0, 0, 0, 0, 0, 1, 1].map(ScanId);
260        GroupIndex::new(
261            PartitionId(0),
262            &request(grouping, FieldSelection::All),
263            Closure::AtDelivery,
264            &times,
265            &fields,
266            Some(&scans),
267            &TimeRuns::whole(0..7),
268        )
269        .expect("index")
270        .expect("groups")
271    }

273    #[test]
274    fn bins_count_from_each_field_and_scan_runs_first_time() {
275        let index = index(Grouping::TimeBins {
276            width_s: 25.0,
277            combine_fields: false,
278            combine_scans: false,
279        });
280        let bins: Vec<(Option<u32>, Option<u32>, u64)> = keys(&index)
281            .iter()
282            .map(|key| (key.field.map(|f| f.0), key.scan.map(|s| s.0), key.bin))
283            .collect();
284        // Field A's clock starts at 0 s: 0, 10, 40 s -> bins 0, 0, 1.
285        // Field B's starts at 20 s: 20, 30 s -> bin 0; in scan 2 it restarts
286        // at 50 s: 50, 60 s -> bin 0.
287        assert_eq!(
288            bins,
289            [
290                (Some(0), Some(0), 0),
291                (Some(0), Some(0), 0),
292                (Some(1), Some(0), 0),
293                (Some(1), Some(0), 0),
294                (Some(0), Some(0), 1),
295                (Some(1), Some(1), 0),
296                (Some(1), Some(1), 0),
297            ]
298        );
299    }

302    fn per_scan_and_per_integration_match_production_solints() {
303        let per_scan = index(Grouping::PerScan {

307            .iter()

326    /// A time selection keeps the whole grid's keys, positions and closes each
327    /// group within itself, and marks the groups it cuts.
328    #[test]
329    fn a_selection_closes_the_parts_it_holds_and_marks_cut_groups() {
330        let times: Vec<f64> = (0..7).map(|t| f64::from(t) * 10.0).collect();
331        let fields = [0, 0, 1, 1, 0, 1, 1].map(FieldId);
332        let scans = [0, 0, 0, 0, 0, 1, 1].map(ScanId);
333        let grouping = Grouping::TimeBins {
334            width_s: f64::INFINITY,
335            combine_fields: false,
336            combine_scans: true,
337        };
338        let index = |selection| {
339            GroupIndex::new(
340                PartitionId(0),
341                &request(grouping, FieldSelection::All),
342                Closure::AtDelivery,
343                &times,
344                &fields,
345                Some(&scans),
346                &TimeRuns::whole(selection),
347            )
348            .expect("index")
349            .expect("groups")
350        };
351        let mut whole = Vec::new();
352        index(0..7).fragments(0..7, 1, &mut whole);
353        let mut sliced = Vec::new();
354        index(1..4).fragments(1..4, 1, &mut sliced);
355        // Field A's group (times 0, 1, 4) and B's (2, 3, 5, 6) are both
356        // cut by times 1-3: A's part is time 1, B's times 2-3, each whole
357        // within the selection and closed there.
358        let seen: Vec<_> = sliced
359            .iter()
360            .map(|f| (f.key.field, f.samples.clone(), f.position, f.clipped))
361            .collect();
362        assert_eq!(
363            seen,
364            [
365                (Some(FieldId(0)), 0..1, FragmentPosition::Whole, true),
366                (Some(FieldId(1)), 1..3, FragmentPosition::Whole, true),
367            ]
368        );
369        assert!(whole.iter().all(|f| !f.clipped));
370        assert_eq!(sliced[0].key, whole[0].key);
371        assert_eq!(sliced[1].key, whole[1].key);
372    }

397    /// A field selection keys groups over the times it holds: bins count
398    /// from the first selected time, and a group that combines fields is
399    /// the selected fields' times alone, held whole and never clipped.
400    #[test]
401    fn a_field_selection_keys_and_closes_groups_within_its_fields() {
402        let times: Vec<f64> = (0..7).map(|t| f64::from(t) * 10.0).collect();
403        let fields = [0, 0, 1, 1, 0, 1, 1].map(FieldId);
404        let combined = Grouping::TimeBins {
405            width_s: 25.0,
406            combine_fields: true,
407            combine_scans: true,
408        };
409        let bins = |selection: FieldSelection| {
410            group_keys(PartitionId(0), combined, &times, &fields, None, &selection)
411                .expect("keys")
412                .expect("groups")
413                .into_iter()
414                .map(|key| key.map(|key| key.bin))
415                .collect::<Vec<_>>()
416        };
417        // Every field: the clock starts at 0 s. Field B alone (20, 30, 50,
418        // 60 s): it starts at 20 s.
419        assert_eq!(bins(FieldSelection::All), [0, 0, 0, 1, 1, 2, 2].map(Some));
420        assert_eq!(
421            bins(FieldSelection::Only(vec![FieldId(1)])),
422            [None, None, Some(0), Some(0), None, Some(1), Some(1)]
423        );
424
425        let only = FieldSelection::Only(vec![FieldId(0)]);
426        let selection = TimeRuns::selected(0..7, &fields, &only);
427        assert_eq!(selection.runs(), [0..2, 4..5]);
428        let fragments = |combine_fields| {
429            let grouping = Grouping::TimeBins {
430                width_s: f64::INFINITY,
431                combine_fields,
432                combine_scans: true,
433            };
434            let mut index = GroupIndex::new(
435                PartitionId(0),
436                &request(grouping, only.clone()),
437                Closure::AtDelivery,
438                &times,
439                &fields,
440                None,
441                &selection,
442            )
443            .expect("index")
444            .expect("groups");
445            let mut fragments = Vec::new();
446            for run in selection.runs() {
447                index.fragments(run.clone(), 1, &mut fragments);
448            }
449            fragments
450                .into_iter()
451                .map(|f| (f.key.field, f.position, f.clipped))
452                .collect::<Vec<_>>()
453        };
454        assert_eq!(
455            fragments(false),
456            [
457                (Some(FieldId(0)), FragmentPosition::First, false),
458                (Some(FieldId(0)), FragmentPosition::Last, false),
459            ]
460        );
461        assert_eq!(
462            fragments(true),
463            [
464                (None, FragmentPosition::First, false),
465                (None, FragmentPosition::Last, false),
466            ]
467        );
468    }
