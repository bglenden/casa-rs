12/// Invalid or unsupported native EVLA generation inputs.
13#[derive(Clone, Copy, Debug, Error, PartialEq, Eq)]
14pub enum NativeAwRequestError {
15    /// Radial samples are not finite, uniformly spaced, or physically valid.
16    #[error("EVLA surface requires finite, uniformly spaced radius/height/slope samples")]
17    InvalidSurface,
18    /// The requested geometry, sampling, frequency or polarization is unsupported.
19    #[error("invalid or unsupported native EVLA A/W sampling")]
20    InvalidSampling,
21    /// Catalog axes are empty, inconsistent, duplicated, or exceed the declared bound.
22    #[error("native A/W catalog axes are invalid or exceed the declared cell bound")]
23    InvalidCatalog,
24}

26/// Explicit EVLA radial dish samples, in metres and dimensionless surface slope.
27///
28/// Content is validated and identified here, not located by a runtime search or
29/// substituted with an opaque caller-supplied digest. Clones share immutable
30/// samples. Data acquisition and its bounded I/O belong to the application.
31#[derive(Clone, Debug)]
32pub struct EvlaDishSurface {
33    samples: Arc<[[f64; 3]]>,
34    identity: [u8; 32],
35}
36
37impl EvlaDishSurface {
38    /// Validate uniformly spaced samples from the optical axis to the 12.5 m rim.
39    pub fn new(samples: Vec<[f64; 3]>) -> Result<Self, NativeAwRequestError> {
40        if samples.len() < 3
41            || samples.iter().flatten().any(|value| !value.is_finite())
42            || samples[0] != [0.0; 3]
43            || samples.last().is_none_or(|sample| sample[0] != 12.5)
44        {
45            return Err(NativeAwRequestError::InvalidSurface);
46        }
47        let step = 12.5 / (samples.len() - 1) as f64;
48        if samples.iter().enumerate().any(|(index, sample)| {
49            (sample[0] - index as f64 * step).abs() > step * 1e-4
50                || sample[1] < 0.0
51                || sample[2] < 0.0
52        }) {
53            return Err(NativeAwRequestError::InvalidSurface);
54        }
55        let mut identity = Sha256::new();
56        identity.update(b"casa-rs/evla/radial-dish-surface/v1\0");
57        identity.update((samples.len() as u64).to_le_bytes());
58        for value in samples.iter().flatten() {
59            identity.update(value.to_bits().to_le_bytes());
60        }
61        Ok(Self {
62            samples: samples.into(),
63            identity: identity.finalize().into(),
64        })
65    }
66
67    /// Parse a caller-supplied three-column radius/height/slope text input.
68    pub fn from_surface_text(text: &str) -> Result<Self, NativeAwRequestError> {
69        let samples = text
70            .lines()
71            .filter(|line| !line.trim().is_empty())
72            .map(|line| {
73                let mut values = line.split_whitespace();
74                let mut sample = [0.0; 3];
75                for value in &mut sample {
76                    *value = values
77                        .next()
78                        .and_then(|v| v.parse::<f64>().ok())
79                        .ok_or(NativeAwRequestError::InvalidSurface)?;
80                }
81                if values.next().is_some() {
82                    return Err(NativeAwRequestError::InvalidSurface);
83                }
84                Ok(sample)
85            })
86            .collect::<Result<Vec<_>, _>>()?;
87        Self::new(samples)
88    }
89
90    /// Validated immutable physical samples.
91    #[must_use]
92    pub fn samples(&self) -> &[[f64; 3]] {
93        &self.samples
94    }
95
96    /// Identity of the parsed sample content, independent of source path.
97    #[must_use]
98    pub const fn content_identity(&self) -> [u8; 32] {
99        self.identity
100    }
101
102    /// Shared sample payload bytes, charged once to generation residency.
103    #[must_use]
104    pub fn resident_bytes(&self) -> usize {
105        size_of_val(self.samples.as_ref())
106    }
107}

109/// Scientific sampling of one native paired EVLA convolution cell.
110///
111/// Increments describe the working sky grid, not the final cropped UV plane.
112/// The first provider uses centered, even, square SIN grids, circular parallel
113/// hands and f64 geometry / Complex32 field and transform arithmetic.
114#[derive(Clone, Copy, Debug, PartialEq)]
115pub struct EvlaAwCellRequest {
116    /// Full working FFT extent, before support selection.
117    pub size: usize,
118    /// Signed sky increments in radians per pixel.
119    pub sky_increment_rad: [f64; 2],
120    /// Beam frequency in Hz, before CASA-compatible float rounding.
121    pub frequency_hz: f64,
122    /// Explicit beam frequency for the paired weight product, in Hz.
123    pub conjugate_frequency_hz: f64,
124    /// Nonnegative W coordinate in wavelengths at the cell frequency.
125    pub w_wavelengths: f64,
126    /// Physical parallactic angle in radians.
127    pub parallactic_angle_rad: f64,
128    /// Circular parallel-hand Mueller element: 0 or 15.
129    pub mueller: usize,
130    /// Integer CF sampling per normal grid pixel.
131    pub oversampling: usize,
132    /// Whether the prolate-spheroidal anti-aliasing term participates.
133    pub prolate_spheroidal: bool,
134    /// Whether the EVLA aperture response participates.
135    pub aperture: bool,
136}
137
138impl EvlaAwCellRequest {
139    /// Numerical residency ceiling for six Complex32 working planes and the
140    /// shared FFT implementation's conservative plan/scratch reservation.
141    pub fn generation_workspace_bytes(&self) -> Result<usize, NativeAwRequestError> {
142        self.validate()?;
143        let fft_axis_bound = 4 * usize::BITS as usize;
144        self.size
145            .checked_mul(self.size)
146            .and_then(|pixels| pixels.checked_mul(6))
147            .and_then(|planes| {
148                self.size
149                    .checked_mul(3 * fft_axis_bound + 1)
150                    .and_then(|fft| planes.checked_add(fft))
151            })
152            .and_then(|values| values.checked_mul(8))
153            .ok_or(NativeAwRequestError::InvalidSampling)
154    }
155
156    /// Validate a cell before numerical generation or scientific identity minting.
157    pub fn validate(&self) -> Result<(), NativeAwRequestError> {
158        if self.size < 8
159            || self.size % 2 != 0
160            || self
161                .size
162                .checked_mul(self.size)
163                .and_then(|n| n.checked_mul(6 * 8))
164                .is_none()
165            || self
166                .sky_increment_rad
167                .iter()
168                .any(|v| !v.is_finite() || *v == 0.0)
169            || self.sky_increment_rad[0] >= 0.0
170            || self.sky_increment_rad[1] <= 0.0
171            || self.sky_increment_rad[0].abs() != self.sky_increment_rad[1].abs()
172            || !self.frequency_hz.is_finite()
173            || !(0.9e9..=8e9).contains(&self.frequency_hz)
174            || !self.conjugate_frequency_hz.is_finite()
175            || !(0.9e9..=8e9).contains(&self.conjugate_frequency_hz)
176            || !self.w_wavelengths.is_finite()
177            || self.w_wavelengths < 0.0
178            || !self.parallactic_angle_rad.is_finite()
179            || !matches!(self.mueller, 0 | 15)
180            || self.oversampling == 0
181            || self.oversampling > self.size / 4
182        {
183            return Err(NativeAwRequestError::InvalidSampling);
184        }
185        Ok(())
186    }
187
188    pub(crate) fn hash(&self, digest: &mut Sha256) {
189        for value in [self.size, self.mueller, self.oversampling] {
190            digest.update((value as u64).to_le_bytes());
191        }
192        for value in [
193            self.sky_increment_rad[0],
194            self.sky_increment_rad[1],
195            self.frequency_hz,
196            self.conjugate_frequency_hz,
197            self.w_wavelengths,
198            self.parallactic_angle_rad,
199        ] {
200            digest.update(value.to_bits().to_le_bytes());
201        }
202        digest.update([u8::from(self.prolate_spheroidal), u8::from(self.aperture)]);
203    }
204}

228/// Explicit choices for every independently enabled A/W term.
229#[derive(Clone, Copy, Debug, PartialEq, Eq)]
230pub struct NativeAwTerms {
231    /// Apply the EVLA aperture response.
232    pub aperture: bool,
233    /// Apply nonzero W phase screens.
234    pub w_term: bool,
235    /// Apply the prolate-spheroidal anti-aliasing function.
236    pub prolate_spheroidal: bool,

239    /// Use the nearest sampled conjugate frequency in the weight beam product.
240    pub conjugate_beams: bool,

243/// Complete application-resolved science inputs and an explicit metadata bound.
244#[derive(Clone, Debug)]
245pub struct NativeAwRequestInput {
246    /// Explicit content-identified EVLA dish model.
247    pub surface: EvlaDishSurface,
248    /// Antenna diameter in metres; this EVLA model requires exactly 25 m.
249    pub antenna_diameter_m: f64,
250    /// Ordered frequency groups, including selected physical SPW coordinates.
251    pub frequencies: Vec<NativeAwFrequencyGroup>,
252    /// Sorted nonnegative W samples in wavelengths.
253    pub w_values: Vec<f64>,
254    /// W-grid scale used by the shared prepared-cell selector.
255    pub w_increment: f64,
256    /// Sorted physical parallactic angles in radians.
257    pub pa_values: Vec<f64>,
258    /// Sorted required circular parallel-hand Mueller elements, 0 and/or 15.
259    pub mueller_elements: Vec<usize>,
260    /// Image reference frequency defining conjugate-frequency selection.
261    pub reference_frequency_hz: f64,
262    /// Working grid and oversampling; no output crop is assumed.
263    pub grid: NativeAwGrid,
264    /// Explicit independently enabled scientific terms.
265    pub terms: NativeAwTerms,
266    /// Maximum expected cells admitted by the application metadata policy.
267    pub maximum_cells: usize,
268}

270/// Complete immutable request for an exactly covered native EVLA CF catalog.
271///
272/// Every cell identity commits to the full physical request and the versioned
273/// generator, support and normalization contract. The requested metadata bound
274/// is a resource policy, not science. Realized crops are outputs and are bound
275/// separately by the prepared-store owner.
276#[derive(Clone, Debug)]
277pub struct NativeAwRequest {
278    input: Arc<NativeAwRequestInput>,
279    geometry: CompiledGeometryId,
280    identity: [u8; 32],
281    cell_count: usize,
282}
283
284impl NativeAwRequest {
285    /// Validate and freeze the entire catalog request before any generation.
286    pub fn new(
287        geometry: CompiledGeometryId,
288        input: NativeAwRequestInput,
289    ) -> Result<Self, NativeAwRequestError> {
290        let NativeAwRequestInput {
291            frequencies,
292            w_values,
293            pa_values,
294            mueller_elements,
295            grid,
296            terms,
297            ..
298        } = &input;
299        let cell_count = frequencies
300            .len()
301            .checked_mul(w_values.len())
302            .and_then(|n| n.checked_mul(pa_values.len()))
303            .and_then(|n| n.checked_mul(mueller_elements.len()))
304            .filter(|n| *n > 0 && *n <= input.maximum_cells)
305            .ok_or(NativeAwRequestError::InvalidCatalog)?;
306        let strictly_sorted = |values: &[f64]| {
307            values.iter().all(|v| v.is_finite()) && values.windows(2).all(|pair| pair[0] < pair[1])
308        };
309        if input.antenna_diameter_m != 25.0
310            || !strictly_sorted(w_values)
311            || w_values[0] < 0.0
312            || !input.w_increment.is_finite()
313            || input.w_increment <= 0.0
314            || !strictly_sorted(pa_values)
315            || !input.reference_frequency_hz.is_finite()
316            || input.reference_frequency_hz <= 0.0
317            || mueller_elements.iter().any(|m| !matches!(m, 0 | 15))
318            || mueller_elements.windows(2).any(|p| p[0] >= p[1])
319            || frequencies
320                .windows(2)
321                .any(|p| p[0].cf_frequency_hz >= p[1].cf_frequency_hz)
322            || frequencies.iter().any(|group| {
323                !group.cf_frequency_hz.is_finite()
324                    || !(0.9e9..=8e9).contains(&group.cf_frequency_hz)
325                    || group.channel_frequencies_hz.is_empty()
326                    || group
327                        .channel_frequencies_hz
328                        .iter()
329                        .any(|f| !f.is_finite() || *f <= 0.0)
330            })
331            || frequencies.iter().enumerate().any(|(i, g)| {
332                frequencies[..i]
333                    .iter()
334                    .any(|p| p.spectral_window == g.spectral_window)
335            })
336            || (!terms.w_term && w_values.as_slice() != [0.0])
337            || (!terms.wideband
338                && (frequencies.len() != 1
339                    || frequencies[0].cf_frequency_hz != input.reference_frequency_hz))
340        {
341            return Err(NativeAwRequestError::InvalidCatalog);
342        }
343        EvlaAwCellRequest {
344            size: grid.size,
345            sky_increment_rad: grid.sky_increment_rad,
346            frequency_hz: frequencies[0].cf_frequency_hz,
347            conjugate_frequency_hz: frequencies[0].cf_frequency_hz,
348            w_wavelengths: w_values[0],
349            parallactic_angle_rad: pa_values[0],
350            mueller: mueller_elements[0],
351            oversampling: grid.oversampling,
352            prolate_spheroidal: terms.prolate_spheroidal,
353            aperture: terms.aperture,
354        }
355        .validate()?;
356        let mut digest = Sha256::new();
357        digest.update(b"casa-rs/native-EVLA-AW/request/v1\0");
358        // This version fixes receiver selection, squint, three-subpixel
359        // integration, float arithmetic, support search, cropping and area.
360        digest.update(b"EVLA-BeamCalc-61020062/3subpixel/F64-geometry-C32-field/centered-forward-FFT/TM2-support-1e-3-even-buffer2/complex-sampled-area-v1\0");
361        digest.update(geometry.as_bytes());
362        digest.update(input.surface.content_identity());
363        digest.update(input.antenna_diameter_m.to_bits().to_le_bytes());
364        digest.update((grid.size as u64).to_le_bytes());
365        hash_values(&mut digest, &grid.sky_increment_rad);
366        digest.update((grid.oversampling as u64).to_le_bytes());
367        digest.update(input.w_increment.to_bits().to_le_bytes());
368        digest.update(input.reference_frequency_hz.to_bits().to_le_bytes());
369        digest.update([
370            u8::from(terms.aperture),
371            u8::from(terms.w_term),
372            u8::from(terms.prolate_spheroidal),
373            u8::from(terms.wideband),
374            u8::from(terms.conjugate_beams),
375        ]);
376        digest.update((frequencies.len() as u64).to_le_bytes());
377        for group in frequencies {
378            digest.update(group.spectral_window.to_le_bytes());
379            digest.update(group.cf_frequency_hz.to_bits().to_le_bytes());
380            hash_values(&mut digest, &group.channel_frequencies_hz);
381        }
382        hash_values(&mut digest, w_values);
383        hash_values(&mut digest, pa_values);
384        digest.update((mueller_elements.len() as u64).to_le_bytes());
385        for mueller in mueller_elements {
386            digest.update((*mueller as u64).to_le_bytes());
387        }
388        Ok(Self {
389            input: Arc::new(input),
390            geometry,
391            identity: digest.finalize().into(),
392            cell_count,
393        })
394    }

405        self.geometry
406    }

408    /// Exact expected pair count; a prefix never constitutes a usable catalog.
409    #[must_use]

411        self.cell_count
412    }
413
414    /// Retained scientific input allocations, shared by clones of this request.
415    #[must_use]
416    pub fn resident_bytes(&self) -> usize {
417        let i = &self.input;
418        size_of::<NativeAwRequestInput>()
419            + i.surface.resident_bytes()
420            + (i.w_values.capacity() + i.pa_values.capacity()) * size_of::<f64>()
421            + i.mueller_elements.capacity() * size_of::<usize>()
422            + i.frequencies.capacity() * size_of::<NativeAwFrequencyGroup>()
423            + i.frequencies
424                .iter()
425                .map(|g| g.channel_frequencies_hz.capacity() * size_of::<f64>())
426                .sum::<usize>()
427    }

429    /// One reusable numerical cell workspace, independent of catalog size.
430    /// Shared scientific input residency is reported separately.
431    pub fn generation_workspace_bytes(&self) -> Result<usize, NativeAwRequestError> {
432        self.cell(0)
433            .expect("validated nonempty catalog")
434            .0
435            .generation_workspace_bytes()
436    }

438    /// Resolve an expected cell in deterministic frequency/W/PA/Mueller order.
439    #[must_use]
440    pub fn cell(
441        &self,
442        index: usize,
443    ) -> Option<(EvlaAwCellRequest, PreparedArtifactScientificIdentity)> {
444        if index >= self.cell_count {
445            return None;
446        }
447        let i = &self.input;
448        let mueller = i.mueller_elements[index % i.mueller_elements.len()];
449        let rest = index / i.mueller_elements.len();
450        let pa = i.pa_values[rest % i.pa_values.len()];
451        let rest = rest / i.pa_values.len();
452        let w = i.w_values[rest % i.w_values.len()];
453        let frequency = i.frequencies[rest / i.w_values.len()].cf_frequency_hz;
454        let target = (2.0 * i.reference_frequency_hz * i.reference_frequency_hz
455            - frequency * frequency)
456            .sqrt();
457        let mut conjugate_frequency = frequency;
458        if i.terms.conjugate_beams {
459            // CASA nearestValue retains the first frequency for a non-real
460            // conjugate or exact tie; otherwise the smallest distance wins.
461            conjugate_frequency = i.frequencies[0].cf_frequency_hz;
462            let mut distance = (conjugate_frequency - target).abs();
463            for group in i.frequencies.iter().skip(1) {
464                let next = (group.cf_frequency_hz - target).abs();
465                if next < distance {
466                    distance = next;
467                    conjugate_frequency = group.cf_frequency_hz;
468                }
469            }
470        }
471        let cell = EvlaAwCellRequest {
472            size: i.grid.size,
473            sky_increment_rad: i.grid.sky_increment_rad,
474            frequency_hz: frequency,
475            conjugate_frequency_hz: conjugate_frequency,
476            w_wavelengths: w,
477            parallactic_angle_rad: pa,
478            mueller,
479            oversampling: i.grid.oversampling,
480            prolate_spheroidal: i.terms.prolate_spheroidal,
481            aperture: i.terms.aperture,
482        };
483        let identity = PreparedArtifactScientificIdentity::native_evla_cell(self.identity, &cell);
484        Some((cell, identity))
485    }
486}
487
488fn hash_values(digest: &mut Sha256, values: &[f64]) {
489    digest.update((values.len() as u64).to_le_bytes());
490    for value in values {
491        digest.update(value.to_bits().to_le_bytes());
492    }
