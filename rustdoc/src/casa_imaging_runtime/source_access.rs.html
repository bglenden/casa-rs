2//! The block layout of the selected observation's rows, and the budget
3//! that bounds what the source holds before the problem quotes its own.
4
5use casa_imaging_model::{CompiledProblem, SpectralWindowSelection};
6use casa_imaging_operator::RowSpectrum;
7use casa_ms::SelectedObservationContentBudget;
8
9use crate::pass::BlockShape;

11/// Relative allowance for the Doppler factor between the frame of a
12/// spectral window's stored `CHAN_FREQ` and the output frame rows reach the
13/// passes in: 3,000 km/s.
14pub const FRAME_MARGIN: f64 = 0.01;
15
16/// One row of the blocks `problem`'s source delivers: the most channels any
17/// selected spectral window gives a row and their frequencies' extent, the
18/// most correlations any selection routes, and the image domains each row
19/// is projected on; no rows.
20#[must_use]
21pub fn row_layout(problem: &CompiledProblem) -> BlockShape {
22    let sources = problem.observation_transaction().read_set().sources();
23    BlockShape {
24        rows: 0,
25        spectrum: row_spectrum(
26            sources
27                .iter()
28                .flat_map(|source| source.selection().spectral_windows()),
29        ),
30        correlations: sources
31            .iter()
32            .flat_map(|source| source.selection().correlations())
33            .map(|selection| selection.products().len())
34            .max()
35            .unwrap_or(0),
36        // One projection per pass domain: each facet of a faceted image
37        // domain is one (#664).
38        domains: problem
39            .geometry()
40            .domains()
41            .iter()
42            .map(|domain| domain.facets().len())
43            .sum(),
44    }
45}

47/// What bounds the samples a row of `windows` places: its most selected
48/// channels, and from the storage owner's `CHAN_FREQ` the smallest first
49/// separation and the widest span of any window's selected channels,
50/// widened by [`FRAME_MARGIN`]. A window without a catalog leaves its rows
51/// evenly spaced ([`RowSpectrum::evenly_spaced`]); the storage owner binds
52/// one to every MeasurementSet selection.
53fn row_spectrum<'a>(windows: impl Iterator<Item = &'a SpectralWindowSelection>) -> RowSpectrum {
54    let mut spectrum = RowSpectrum::evenly_spaced(0);
55    let mut known = true;
56    for window in windows {
57        let channels = window.channel_indices();
58        spectrum.channels = spectrum.channels.max(channels.len());
59        let frequency = |index: usize| {
60            let catalog = window.coordinate_catalog()?;
61            catalog.channel_frequency_hz(*channels.get(index)? as usize)
62        };
63        if channels.len() < 2 {
64            continue;
65        }
66        let (Some(first), Some(second), Some(last)) =
67            (frequency(0), frequency(1), frequency(channels.len() - 1))
68        else {
69            known = false;
70            continue;
71        };
72        let spacing = (second - first).abs() * (1.0 - FRAME_MARGIN);
73        if spacing > 0.0
74            && (spectrum.first_spacing_hz == 0.0 || spacing < spectrum.first_spacing_hz)
75        {
76            spectrum.first_spacing_hz = spacing;
77        }
78        spectrum.span_hz = spectrum
79            .span_hz
80            .max((last - first).abs() * (1.0 + FRAME_MARGIN));
81    }
82    if known {
83        spectrum
84    } else {
85        RowSpectrum::evenly_spaced(spectrum.channels)
86    }
87}

89/// The budget that bounds source inspection before the compiled problem
90/// can quote its complete initialization and traversal requirements.
91#[must_use]
92pub const fn bootstrap_source_budget() -> SelectedObservationContentBudget {
93    SelectedObservationContentBudget::new(64 << 20, 2, 4)
