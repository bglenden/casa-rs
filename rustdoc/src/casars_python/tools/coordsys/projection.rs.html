17    PyRuntimeError::new_err(message.to_string())
18}

20pub(super) fn direction_projection(name: &str, mut parameters: Vec<f64>) -> PyResult<Projection> {
21    let name = if name == "GLS" { "SFL" } else { name };
22    let (_, count) = PROJECTIONS
23        .iter()
24        .find(|(known, _)| *known == name)
25        .ok_or_else(|| raised("Projection::nParameters() - unknown projection"))?;
26    if parameters.is_empty() && matches!(name, "COP" | "COD" | "COE" | "COO" | "BON") {
27        return Err(raised(
28            "Projection::validate() - there are missingobligatory parameters",
29        ));
30    }
31    let kind = ProjectionType::from_name(name)
32        .ok_or_else(|| raised("Projection::nParameters() - unknown projection"))?;
33    if parameters.len() < *count as usize {
34        // Casacore resizes this vector without copying the supplied prefix,
35        // then fills the missing suffix with the projection's defaults.
36        let supplied = parameters.len();
37        let mut defaults = if name == "SIN" {
38            vec![0.0; 2]
39        } else {
40            kind.fits_parameter_defaults()[..*count as usize].to_vec()
41        };
42        defaults[..supplied].fill(0.0);
43        parameters = defaults;
44    }
45    Ok(Projection::with_parameters(kind, parameters))
46}

48fn apply_projection(
49    state: &mut CoordinateState,
50    name: &str,
51    parameters: Vec<f64>,
52) -> PyResult<bool> {
53    let (index, _, placement) = numbered_coordinates(state)?
54        .into_iter()
55        .find(|(_, model, _)| matches!(model, CoordinateModel::Direction(_)))
56        .ok_or_else(|| missing(CoordinateType::Direction))?;
57    let CoordinateModel::Direction(direction) = placement.model(&state.record).map_err(raised)?
58    else {
59        return Err(missing(CoordinateType::Direction));
60    };
61    let parameters = if parameters == [-1.0] {
62        Vec::new()
63    } else {
64        parameters
65    };
66    let projection = direction_projection(name, parameters)?;
67    let changed = CoordinateModel::from(
68        direction
69            .with_rebuilt_projection(projection)
70            .map_err(raised)?,
71    );
72    let record = Value::Record(changed.to_record());
73    replace_at(state, index as i32, changed, record)?;
74    Ok(true)
75}

77#[pymethods]
78impl PyCoordsys {
79    #[pyo3(signature = (r#type, parameters=DoubleList::of(&[-1.0])), text_signature = "($self, type, parameters=[-1.0])")]
80    fn setprojection(
81        &self,
82        py: Python<'_>,
83        r#type: FrameName,
84        parameters: DoubleList,
85    ) -> PyResult<bool> {
86        let name = r#type.into_name("type")?;
87        py.detach(|| apply_projection(&mut self.state(), &name, parameters.0))
88    }
