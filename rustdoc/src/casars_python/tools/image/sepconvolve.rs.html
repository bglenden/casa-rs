21fn parse_widths(value: Option<&Bound<'_, PyAny>>) -> PyResult<(Option<Vec<Quantity>>, String)> {
22    let variant = value
23        .map(width_variant)
24        .transpose()?
25        .unwrap_or(Variant::Empty);
26    let history = variant.describe();
27    let quantities = match variant {
28        Variant::Numbers(values) => values
29            .into_iter()
30            .map(|value| Quantity::new(value, "pix").map_err(raised))
31            .collect::<PyResult<_>>()?,
32        Variant::Text(text) => {
33            let tokens: Vec<_> = text
34                .split(|byte: char| byte.is_whitespace() || byte == ',')
35                .filter(|token| !token.is_empty())
36                .collect();
37            if tokens.is_empty() {
38                vec![Quantity::dimensionless(0.0)]
39            } else {
40                tokens
41                    .into_iter()
42                    .map(width_text)
43                    .collect::<PyResult<_>>()?
44            }
45        }
46        Variant::Texts(texts) => texts
47            .iter()
48            .map(|text| width_text(text))
49            .collect::<PyResult<_>>()?,
50        Variant::Other(_) => {
51            let Some(value) = value.filter(|value| {
52                value.is_instance_of::<PyList>() || value.is_instance_of::<PyTuple>()
53            }) else {
54                return Ok((None, history));
55            };
56            let items = value.try_iter()?.collect::<PyResult<Vec<_>>>()?;
57            if !items.iter().any(|item| item.is_instance_of::<PyString>()) {
58                return Ok((None, history));
59            }
60            items
61                .iter()
62                .map(|item| width_text(item.str()?.to_str()?))
63                .collect::<PyResult<_>>()?
64        }
65        _ => return Ok((None, history)),
66    };
67    Ok((Some(quantities), history))
68}

70fn width_variant(value: &Bound<'_, PyAny>) -> PyResult<Variant> {
71    let Ok(array) = value.cast::<PyUntypedArray>() else {
72        return value.extract();
73    };
74    let dtype = array.dtype();
75    match (dtype.kind(), dtype.char()) {
76        (b'i' | b'u', _) | (b'f', b'f' | b'd' | b'g') => {
77            let flat = value
78                .call_method1("ravel", ("F",))?
79                .call_method0("tolist")?;
80            let values: Vec<f64> = flat.extract()?;
81            Ok(if values.is_empty() {
82                Variant::Empty
83            } else {
84                Variant::Numbers(values)
85            })
86        }
87        (b'S' | b'U', _) => {
88            if array.ndim() == 0 {
89                return Ok(Variant::Texts(Vec::new()));
90            }
91            if array.shape().iter().filter(|&&length| length > 1).count() > 1 {
92                return Err(raised(
93                    "Cannot read convolution widths from a multidimensional string array",
94                ));
95            }
96            let flat = value
97                .call_method1("ravel", ("F",))?
98                .call_method0("tolist")?;
99            let texts = flat
100                .try_iter()?
101                .map(|item| match item?.extract()? {
102                    Variant::Text(text) => Ok(text),
103                    _ => Err(raised("Convolution width must be a quantity string")),
104                })
105                .collect::<PyResult<_>>()?;
106            Ok(Variant::Texts(texts))
107        }
108        _ => Ok(Variant::Empty),
109    }
110}

112fn width_text(text: &str) -> PyResult<Quantity> {
113    let text = text.trim();
114    if text.is_empty() {
115        return Ok(Quantity::dimensionless(0.0));
116    }
117    let context = " in converting convolution width";
118    if let Some((first, rest)) = text.split_once(|c: char| c.is_whitespace() || c == ',')
119        && rest
120            .trim_start()
121            .starts_with(|c: char| c.is_ascii_digit() || "+-.".contains(c))
122        && let Ok(quantity) = read_quantity(first, context)
123        && !quantity.unit().name().is_empty()
124    {
125        return Ok(quantity);
126    }
127    read_quantity(text, context).map_err(raised)
128}

130#[pymethods]
131impl PyImageTool {
132    /// Convolve a selected Float image with one-dimensional kernels.
133    #[allow(clippy::too_many_arguments)]
134    #[pyo3(signature = (outfile="", axes=IntList::of(&[-1]), types=None, widths=None, scale=-1.0, region=RegionArgument::default(), mask=None, overwrite=false, stretch=false), text_signature = "($self, outfile='', axes=[-1], types=[''], widths=[], scale=-1.0, region={}, mask='', overwrite=False, stretch=False)")]
135    fn sepconvolve(
136        &self,
137        py: Python<'_>,
138        outfile: &str,
139        #[pyo3(from_py_with = numeric_axis_vector)] axes: IntList,
140        types: Option<&Bound<'_, PyAny>>,
141        widths: Option<&Bound<'_, PyAny>>,
142        scale: f64,
143        region: RegionArgument,
144        mask: Option<&Bound<'_, PyAny>>,
145        #[pyo3(from_py_with = require_bool)] overwrite: bool,
146        #[pyo3(from_py_with = require_bool)] stretch: bool,
147    ) -> PyResult<Py<Self>> {
148        let mut types = string_vector(types, "types")?;
149        if types.is_empty() {
150            types.push(String::new());
151        }
152        let widths = parse_widths(widths);
153        let mask = mask_argument(mask);
154        let attachment = py.detach(|| -> PyResult<_> {
155            let state = self.state();
156            let input = state
157                .image
158                .as_ref()
159                .ok_or_else(|| raised("Unable to create image"))?;
160            let mask = mask?;
161            let (widths, widths_history) = widths?;
162            let Some(widths) = widths else {
163                input
164                    .lock()
165                    .check_single_precision("sepconvolve")
166                    .map_err(raised)?;
167                return Ok(None);
168            };
169            let destination = (!outfile.is_empty()).then(|| Path::new(outfile));
170            let options = SepConvolutionOptions {
171                axes: &axes.0,
172                types: &types,
173                widths: &widths,
174                scale,
175                region: region.selection(),
176                mask: (!mask.is_empty()).then_some(mask.as_str()),
177                stretch,
178                destination,
179            };
180            let history_command = state.history.then(|| {
181                let region_hist = region.history();
182                format!(
183                    "ia.sepconvolve(outfile={outfile:?}, axes={:?}, types={:?}, widths={widths_history}, scale={scale}, region={region_hist}, mask={mask:?}, overwrite={overwrite}, stretch={stretch})",
184                    axes.0, types,
185                )
186            });
187            sepconvolve_image(
188                input,
189                outfile,
190                overwrite,
191                options,
192                history_command,
193                external_open_check(),
194            )
195            .map(Some)
196            .map_err(raised)
197        })?;
198        Py::new(py, attachment.map_or_else(Self::new, Self::attached))
199    }
