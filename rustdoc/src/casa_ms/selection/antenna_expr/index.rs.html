9/// The ANTENNA rows an expression resolves against; row `i` is antenna ID `i`.
10pub(super) struct AntennaIndex<'r, 'a> {
11    rows: &'r [AntennaRow<'a>],
12}
13
14impl<'r, 'a> AntennaIndex<'r, 'a> {
15    pub(super) fn new(rows: &'r [AntennaRow<'a>]) -> Self {
16        Self { rows }
17    }
18
19    pub(super) fn len(&self) -> usize {
20        self.rows.len()
21    }
22
23    pub(super) fn rows(&self) -> &'r [AntennaRow<'a>] {
24        self.rows
25    }
26
27    /// `matchAntennaName(const String&)` (`MSAntennaIndex.cc:126-158`): the
28    /// rows whose NAME equals `name`; only when none does, the rows whose
29    /// STATION equals it.
30    pub(super) fn match_antenna_name(&self, name: &str) -> Vec<usize> {
31        let by_name = self.rows_where(|row| row.name == name);
32        if by_name.is_empty() {
33            self.match_station_name(name)
34        } else {
35            by_name
36        }
37    }
38
39    /// The first row [`match_antenna_name`](Self::match_antenna_name) finds,
40    /// without collecting the rest: what the `antidrange` actions keep
41    /// (`tmp[0]`, `MSAntennaGram.yy:354,375`).
42    pub(super) fn first_antenna_name(&self, name: &str) -> Option<usize> {
43        self.rows
44            .iter()
45            .position(|row| row.name == name)
46            .or_else(|| self.rows.iter().position(|row| row.station == name))
47    }
48
49    /// `matchStationName(const String&)` (`MSAntennaIndex.cc:231-263`).
50    pub(super) fn match_station_name(&self, station: &str) -> Vec<usize> {
51        self.rows_where(|row| row.station == station)
52    }
53
54    /// `matchAntennaRegexOrPattern` (`MSAntennaIndex.cc:82-122`): a leading
55    /// `^` negates; an antenna matches when its NAME matches, or else its
56    /// STATION does. Unlike [`match_antenna_name`](Self::match_antenna_name)
57    /// the station fallback is per row.
58    pub(super) fn match_antenna_regex_or_pattern(
59        &self,
60        negated: bool,
61        text: &str,
62        is_regex: bool,
63    ) -> Result<Vec<usize>, String> {
64        let matcher = compile(text, is_regex)?;
65        Ok(self.rows_where(|row| {
66            (matcher.matches(row.name) || matcher.matches(row.station)) != negated
67        }))
68    }
69
70    /// `matchStationRegexOrPattern` (`MSAntennaIndex.cc:187-227`): as the
71    /// antenna form over STATION alone.
72    pub(super) fn match_station_regex_or_pattern(
73        &self,
74        negated: bool,
75        text: &str,
76        is_regex: bool,
77    ) -> Result<Vec<usize>, String> {
78        let matcher = compile(text, is_regex)?;
79        Ok(self.rows_where(|row| matcher.matches(row.station) != negated))
80    }
81
82    /// The antennas of the integer range `start~end`, as the
83    /// `INT DASH INT` action builds them (`MSAntennaGram.yy:358-380`): each
84    /// integer `i` names the first antenna whose NAME (else STATION) is
85    /// `i` written in decimal, and otherwise is antenna ID `i`. Only IDs of
86    /// existing antennas are returned, the ones `matchId` would keep.
87    ///
88    /// The C++ enumerates the whole range; this visits the existing IDs and
89    /// the decimal names instead, so the work and memory do not grow with
90    /// the range width (`0~2147483646` costs what `0~1` does).
91    pub(super) fn integer_range(&self, start: i32, end: i32) -> Vec<usize> {
92        let mut ids = Vec::new();
93        let existing = usize::try_from(start.max(0)).unwrap_or(0)
94            ..usize::try_from(end.saturating_add(1).max(0))
95                .unwrap_or(0)
96                .min(self.rows.len());
97        for id in existing.clone() {
98            let mut digits = DecimalBuffer::default();
99            let decimal = digits.format(i64::try_from(id).unwrap_or(i64::MAX));
100            ids.push(self.first_antenna_name(decimal).unwrap_or(id));
101        }
102        // Integers outside the existing IDs reach an antenna only by name.
103        for row in self.rows {
104            for text in [row.name, row.station] {
105                let Some(value) = canonical_integer(text) else {
106                    continue;
107                };
108                let in_range = (i64::from(start)..=i64::from(end)).contains(&value);
109                let existing_id = usize::try_from(value).is_ok_and(|id| existing.contains(&id));
110                if in_range && !existing_id {
111                    ids.extend(self.first_antenna_name(text));
112                }
113            }
114        }
115        ids
116    }
117
118    fn rows_where(&self, predicate: impl Fn(&AntennaRow<'a>) -> bool) -> Vec<usize> {
119        self.rows
120            .iter()
121            .enumerate()
122            .filter(|(_, row)| predicate(row))
123            .map(|(id, _)| id)
124            .collect()
125    }
126}

128fn compile(text: &str, is_regex: bool) -> Result<FullMatch, String> {
129    if is_regex {
130        FullMatch::compile(text)
131    } else {
132        FullMatch::compile_pattern(text)
133    }
134}

136/// The integer `text` writes as `ostringstream << Int` would (no sign
137/// other than `-`, no leading zeros), if it is one.
138fn canonical_integer(text: &str) -> Option<i64> {
139    let value = text.parse::<i64>().ok()?;
140    let mut digits = DecimalBuffer::default();
141    (digits.format(value) == text).then_some(value)
142}
143
144/// A stack buffer for writing an integer in decimal without allocating.
145#[derive(Default)]
146struct DecimalBuffer {
147    bytes: [u8; 20],
148}
149
150impl DecimalBuffer {
151    fn format(&mut self, value: i64) -> &str {
152        let mut magnitude = value.unsigned_abs();
153        let mut start = self.bytes.len();
154        loop {
155            start -= 1;
156            self.bytes[start] = b'0' + (magnitude % 10) as u8;
157            magnitude /= 10;
158            if magnitude == 0 {
159                break;
160            }
161        }
162        if value < 0 {
163            start -= 1;
164            self.bytes[start] = b'-';
165        }
166        std::str::from_utf8(&self.bytes[start..]).unwrap_or_default()
167    }
168}

170/// glibc's `atoi` on a run of decimal digits: `(int) strtol(text)`, where
171/// `strtol` saturates at `LONG_MAX` (64-bit) and the conversion to `int`
172/// keeps the low 32 bits. `4294967296` is therefore `0`, as in CASA.
173pub(super) fn c_atoi(digits: &str) -> i32 {
174    let value = digits.parse::<i64>().unwrap_or(i64::MAX);
175    value as i32
176}

178/// glibc's `atof` on a `{FLOAT}` lexeme: `strtod` reads the longest valid
179/// prefix, and a Fortran-style `D` exponent is not part of one, so `1d3`
180/// is `1`.
181pub(super) fn c_atof(lexeme: &str) -> f64 {
182    let mantissa = lexeme
183        .find(['D', 'd'])
184        .map_or(lexeme, |exponent| &lexeme[..exponent]);
185    mantissa.parse::<f64>().unwrap_or(0.0)
186}
