13/// How one pass divides an image domain's grid accumulation among owners
14/// (one per worker).
15#[derive(Clone, Debug, PartialEq)]
16pub enum Partition {
17    /// Each owner accumulates a disjoint contiguous range of each wave's
18    /// planes. Nothing is merged, so the result does not depend on the owner
19    /// count.
20    Planes {
21        /// Number of owners.
22        owners: usize,
23    },
24    /// Each owner accumulates the samples whose anchor row lies in its
25    /// region, on a tile of the region plus `halo` rows; tiles are added in
26    /// region order before the transform.
27    Regions {
28        /// The owners' regions, top to bottom.
29        regions: Vec<Region>,
30        /// Rows each tile extends past its region: the kernel set's largest
31        /// half support along `y`.
32        halo: usize,
33    },
34}
35
36/// One owner of a [`Partition::Regions`] pass.
37#[derive(Clone, Debug, PartialEq, Eq)]
38pub struct Region {
39    /// Anchor rows (grid `y`) owned.
40    pub rows: Range<usize>,
41    /// The accumulated window: the owned rows plus the halo, clipped to the grid.
42    pub tile: Tile,
43}
44
45impl Partition {
46    /// `owners` horizontal strips of `operator`'s grid of near-equal height,
47    /// each with the kernel set's largest half support
48    /// ([`casa_imaging_operator::ConvolutionFunctionSet::max_half_support`])
49    /// as a halo on both sides, so every routed sample's support lies in its
50    /// tile.
51    #[must_use]
52    pub fn regions(operator: &MeasurementOperator, owners: usize) -> Self {
53        let [nx, ny] = operator.geometry().grid_shape();
54        let halo = usize::from(operator.cf().max_half_support()[1]);
55        let owners = owners.clamp(1, ny);
56        Self::Regions {
57            regions: (0..owners)
58                .map(|owner| {
59                    let rows = owner * ny / owners..(owner + 1) * ny / owners;
60                    let first = rows.start.saturating_sub(halo);
61                    let last = (rows.end + halo).min(ny);
62                    Region {
63                        tile: Tile {
64                            origin: [0, first],
65                            shape: [nx, last - first],
66                        },
67                        rows,
68                    }
69                })
70                .collect(),
71            halo,
72        }
73    }
74
75    /// Number of owners.
76    #[must_use]
77    pub fn owners(&self) -> usize {
78        match self {
79            Self::Planes { owners } => *owners,
80            Self::Regions { regions, .. } => regions.len(),
81        }
82    }
83}

85/// How many planes a pass holds at once.
86#[derive(Clone, Copy, Debug, PartialEq, Eq)]
87pub enum Residency {
88    /// Every plane in one traversal.
89    All,
90    /// Consecutive waves of at most `planes_per_wave` planes, one source
91    /// traversal each.
92    Waves {
93        /// Planes per wave; positive.
94        planes_per_wave: u32,
95    },

98/// What one wave of a pass holds in memory, for [`Residency::plan`].
99#[derive(Clone, Copy)]
100pub struct WaveDemand<'a> {
101    /// The pass's image domains; they share one plane axis.
102    pub domains: &'a [PassDomain<'a>],
103    /// Modes accumulated.
104    pub modes: ModeSet,
105    /// Whether the pass subtracts a model.
106    pub with_model: bool,
107    /// The widest spacing between adjacent selected native channels, which
108    /// sizes a wave's model halo
109    /// ([`casa_imaging_operator::SpectralResampler::model_halo`]).
110    pub native_spacing_hz: f64,
111    /// Workers transforming planes at once.
112    pub workers: usize,
113    /// Where the owners grid.
114    pub backend: BackendChoice,
115    /// The layout of the source's rows ([`super::BoundedSource::maximum_block`]).
116    pub block: BlockShape,
117}
118
119impl WaveDemand<'_> {
120    /// Planes on each image domain's axis.
121    #[must_use]
122    pub fn planes(&self) -> u32 {
123        self.domains
124            .first()
125            .map_or(0, |domain| domain.operator.basis().planes())
126    }
127
128    /// Bytes a wave of `planes` planes holds: the largest of its stages
129    /// (#738), beside what it holds throughout.
130    ///
131    /// Throughout the wave, each domain's prepared model grids, with a
132    /// model, for the wave's planes and its model halo on each side; on
133    /// Metal, their device copy, which the device degrids, and each owner's
134    /// ring ([`casa_imaging_metal::ring_bytes`]). The stages, summed over
135    /// image domains, since a domain finishes and appends while the others
136    /// hold their tiles or images:
137    /// - gridding: the owners' tiles of the wave's planes, beside the row
138    ///   chunks the wave places [`Self::block`]'s rows with;
139    /// - finishing, beside the chunks: a region pass's tiles and the merged
140    ///   accumulator they are added into, which one owner's whole-grid tile
141    ///   needs none of; then the accumulator beside its `f32` images; and
142    ///   for the transforms, one transform plane and one image per grid
143    ///   polarization for each transform that runs at once, and planning
144    ///   one grid transform ([`grid_planning_bytes`]), which the process
145    ///   does one at a time. Domains finish one after another; a plane
146    ///   pass's owners finish together on up to the workers, a region
147    ///   pass's merged accumulator alone (#745 review);
148    /// - appending: the images, the copy the normal state takes of them and,
149    ///   for a basis whose normal state is resident, that state at its
150    ///   stored size ([`casa_imaging_deconvolution::resident_bytes_per_cell`]);
151    ///   a channel-local state is paged, under its cube cache's charge.
152    ///
153    /// On Metal the accumulators are device memory and cost what a host
154    /// one does. What grows with the source's block is the source's
155    /// ([`super::source_block_bytes`]).
156    #[must_use]
157    pub fn bytes(&self, planes: u32) -> u64 {
158        let mut stages = DomainStages::default();
159        let mut transforming = 0_u64;
160        let mut planning = 0_u64;
161        for domain in self.domains {
162            stages.add(self.domain_stages(domain, planes));
163            let operator = domain.operator;
164            let [width, height] = operator.geometry().image().shape;
165            let image_cells = (width * height) as u64;
166            let grid_cells = operator.geometry().cells() as u64;
167            let gpols = operator.polarization().grid_pols() as u64;
168            let at_once = match &domain.partition {
169                Partition::Planes { owners } => (*owners).min(self.workers),
170                Partition::Regions { .. } => 1,
171            } as u64;
172            transforming = transforming.max(at_once * (grid_cells * 16 + gpols * image_cells * 16));
173            planning = planning.max(grid_planning_bytes(operator.geometry().cells()));
174        }
175        let chunks = super::wave::chunk_bytes(
176            self.domains,
177            self.backend,
178            self.block,
179            self.workers,
180            self.modes.weight,
181            self.with_model,
182        );
183        let transforms = planning + transforming;
184        stages.held
185            + (stages.gridding + chunks)
186                .max(stages.finishing + chunks + transforms)
187                .max(stages.appending)
188    }
189
190    /// What `domain` holds at each stage of a wave of `planes` planes
191    /// ([`Self::bytes`]).
192    fn domain_stages(&self, domain: &PassDomain<'_>, planes: u32) -> DomainStages {
193        let operator = domain.operator;
194        let precision = operator.precision();
195        let one = PlaneRange::single(0);
196        let planes_u64 = u64::from(planes);
197        let layout = |tile: Option<Tile>, modes: ModeSet| {
198            operator
199                .accumulator_layout(one, tile, modes)
200                .bytes(precision) as u64
201        };
202        let accumulator = layout(None, self.modes) * planes_u64;
203        let (tiles, merging) = match &domain.partition {
204            Partition::Planes { .. } => (accumulator, accumulator),
205            Partition::Regions { regions, .. } => {
206                let tiles = regions
207                    .iter()
208                    .map(|region| layout(Some(region.tile), self.modes))
209                    .sum::<u64>()
210                    * planes_u64;
211                // One owner's tile is the whole grid, which the wave
212                // finishes as it is: both key on the owner count.
213                let merged = if regions.len() > 1 { accumulator } else { 0 };
214                (tiles, tiles + merged)
215            }
216        };
217        let [width, height] = operator.geometry().image().shape;
218        let image_cells = (width * height) as u64;
219        let basis = operator.basis();
220        let pols = operator.polarization().requested().len() as u64;
221        // The weight image has the PSF's terms (`AccumulatorLayout`).
222        let terms = (basis.data_terms()
223            + if self.modes.psf { basis.psf_terms() } else { 0 }
224            + if self.modes.weight {
225                basis.psf_terms()
226            } else {
227                0
228            }) as u64;
229        let images = terms * pols * image_cells * size_of::<f32>() as u64 * planes_u64;
230        let normal = if matches!(basis, Basis::ChannelLocal { .. }) {
231            0
232        } else {
233            pols * image_cells
234                * resident_bytes_per_cell(basis.data_terms(), basis.psf_terms(), self.modes.psf)
235                * planes_u64
236        };
237        let mut held = 0;
238        if self.with_model {
239            let model_planes = planes
240                .saturating_add(2 * domain.resampler.model_halo(self.native_spacing_hz))
241                .min(self.planes());
242            // Metal degrids the model on the device, fused into the
243            // residual or in the chunks' prediction batches.
244            let copies = if self.backend == BackendChoice::Metal {
245                2
246            } else {
247                1
248            };
249            held += copies * layout(None, ModeSet::DATA) * u64::from(model_planes);
250        }
251        if self.backend == BackendChoice::Metal {
252            let npol = operator.polarization().correlations().len();
253            held += domain.partition.owners() as u64 * casa_imaging_metal::ring_bytes(npol);
254        }
255        DomainStages {
256            held,
257            gridding: tiles,
258            finishing: merging.max(accumulator + images),
259            appending: 2 * images + normal,
260        }
261    }
262}

264/// What a wave holds at each of its stages ([`WaveDemand::bytes`]), summed
265/// over image domains: a domain finishes and appends while the others
266/// hold their tiles or images, so the sums bound every moment.
267#[derive(Clone, Copy, Default)]
268struct DomainStages {
269    /// Held through every stage.
270    held: u64,
271    /// The owners' tiles.
272    gridding: u64,
273    /// The merge, then the accumulator beside its images.
274    finishing: u64,
275    /// The images, their copy and the normal state.
276    appending: u64,
277}
278
279impl DomainStages {
280    fn add(&mut self, other: Self) {
281        self.held += other.held;
282        self.gridding += other.gridding;
283        self.finishing += other.finishing;
284        self.appending += other.appending;
285    }
286}

288impl Residency {
289    /// The fewest waves whose [`WaveDemand::bytes`] fit `budget`: every
290    /// plane when they fit, otherwise the largest wave that does and at
291    /// least one plane. Whether a wave fits at all is the admission of its
292    /// bytes ([`crate::admit`]).
293    #[must_use]
294    pub fn plan(demand: &WaveDemand<'_>, budget: u64) -> Self {
295        let planes = demand.planes();
296        if planes <= 1 || demand.bytes(planes) <= budget {
297            return Self::All;
298        }
299        if demand.bytes(1) > budget {
300            return Self::Waves { planes_per_wave: 1 };
301        }
302        // `bytes` grows with the wave: the largest fitting wave lies in
303        // `[fits, too_many)`.
304        let (mut fits, mut too_many) = (1, planes);
305        while too_many - fits > 1 {
306            let middle = fits + (too_many - fits) / 2;
307            if demand.bytes(middle) <= budget {
308                fits = middle;
309            } else {
310                too_many = middle;
311            }
312        }
313        Self::Waves {
314            planes_per_wave: fits,
315        }
316    }
317
318    /// The plane ranges of the waves over `planes` planes.
319    #[must_use]
320    pub fn waves(self, planes: u32) -> Vec<PlaneRange> {
321        let step = match self {
322            Self::All => planes.max(1),
323            Self::Waves { planes_per_wave } => {
324                assert!(planes_per_wave > 0, "a wave holds at least one plane");
325                planes_per_wave
326            }
327        };
328        (0..planes)
329            .step_by(step as usize)
330            .map(|start| PlaneRange::new(start, (start + step).min(planes)))
331            .collect()
332    }
333}

335/// The owner of each placement within one wave.
336pub(super) enum Router {
337    /// `starts[k]` is owner `k`'s first plane; owner ranges are contiguous.
338    Planes { starts: Vec<u32> },
339    /// `owner_of_row[y]` owns anchor row `y`; tiles extend `halo` rows.
340    Regions { owner_of_row: Vec<u16>, halo: usize },
341}
342
343impl Router {
344    pub(super) fn new(partition: &Partition, wave: PlaneRange) -> Self {
345        match partition {
346            Partition::Planes { owners } => {
347                let planes = wave.len();
348                let owners = (*owners).clamp(1, planes.max(1));
349                Self::Planes {
350                    starts: (0..owners)
351                        .map(|owner| wave.start + (owner * planes / owners) as u32)
352                        .collect(),
353                }
354            }
355            Partition::Regions { regions, halo } => {
356                let rows = regions.last().map_or(0, |region| region.rows.end);
357                let mut owner_of_row = vec![0; rows];
358                for (owner, region) in regions.iter().enumerate() {
359                    owner_of_row[region.rows.clone()].fill(owner as u16);
360                }
361                Self::Regions {
362                    owner_of_row,
363                    halo: *halo,
364                }
365            }
366        }
367    }
368
369    /// Number of owners in this wave.
370    pub(super) fn owners(&self) -> usize {
371        match self {
372            Self::Planes { starts } => starts.len(),
373            Self::Regions { owner_of_row, .. } => {
374                owner_of_row.last().map_or(1, |owner| *owner as usize + 1)
375            }
376        }
377    }
378
379    /// The one owner that grids the weight image when the grid is split
380    /// into regions: the owner of the centre row, where `Mode::Weight`
381    /// places every sample's `FT[PB²]` taps. `None` when every owner holds
382    /// whole planes and grids its own samples' weights.
383    pub(super) fn weight_owner(&self) -> Option<usize> {
384        match self {
385            Self::Planes { .. } => None,
386            Self::Regions { owner_of_row, .. } => {
387                Some(usize::from(owner_of_row[owner_of_row.len() / 2]))
388            }
389        }
390    }
391
392    /// Plane range and tile owner `owner` accumulates in `wave`.
393    pub(super) fn target(
394        &self,
395        partition: &Partition,
396        wave: PlaneRange,
397        owner: usize,
398    ) -> (PlaneRange, Option<Tile>) {
399        match (self, partition) {
400            (Self::Planes { starts }, _) => {
401                let end = starts.get(owner + 1).copied().unwrap_or(wave.end);
402                (PlaneRange::new(starts[owner], end), None)
403            }
404            (Self::Regions { .. }, Partition::Regions { regions, .. }) => {
405                (wave, Some(regions[owner].tile))
406            }
407            (Self::Regions { .. }, Partition::Planes { .. }) => {
408                unreachable!("router built from this partition")
409            }
410        }
411    }
412
413    /// Owner of one placement; the placement lies on the wave and fits the grid.
414    pub(super) fn owner(&self, operator: &MeasurementOperator, placement: &Placement) -> usize {
415        match self {
416            Self::Planes { starts } => {
417                starts.partition_point(|start| *start <= placement.plane) - 1
418            }
419            Self::Regions { owner_of_row, halo } => {
420                let mut hold = CellHold::new();
421                let taps = operator.cf().taps(placement.cf, &mut hold);
422                debug_assert!(
423                    usize::from(taps.half_support()[1]) <= *halo,
424                    "a kernel cell's support exceeds the regions' halo"
425                );
426                let anchor =
427                    operator
428                        .geometry()
429                        .locate(placement.u, placement.v, taps.oversampling());
430                usize::from(owner_of_row[anchor.y as usize])
431            }
432        }
433    }
434}

436#[cfg(test)]
437mod tests {
438    use super::*;
439    use casa_imaging_model::{CorrelationType, PolarizationCoordinate};
440    use casa_imaging_operator::{
441        Basis, GridGeometry, GridPadding, GridPrecision, ImageExtent, PolarizationRouting,

444
445    #[test]
446    fn regions_cover_the_grid_once_with_the_kernel_halo() {
447        let geometry = GridGeometry::new(
448            ImageExtent {
449                shape: [64, 64],
450                increment_rad: [-1.0e-5, 1.0e-5],
451                reference_pixel: [32, 32],
452            },
453            GridPadding::CasaComposite,
454        )
455        .expect("geometry");
456        let polarization = PolarizationRouting::compile(
457            &[CorrelationType::LinearXx, CorrelationType::LinearYy],
458            &[PolarizationCoordinate::StokesI],
459        )
460        .expect("routing");
461        let cf = Spheroidal::new(&geometry, &polarization);
462        let operator = MeasurementOperator::new(
463            geometry,
464            Basis::Constant,
465            polarization,
466            Box::new(cf),
467            GridPrecision::F64,
468        );
469        let Partition::Regions { regions, halo } = Partition::regions(&operator, 3) else {
470            panic!("regions");
471        };
472        assert_eq!(halo, 3, "the spheroidal kernel's half support");
473        let ny = operator.geometry().grid_shape()[1];
474        assert_eq!(regions.first().unwrap().rows.start, 0);
475        assert_eq!(regions.last().unwrap().rows.end, ny);
476        for pair in regions.windows(2) {
477            assert_eq!(pair[0].rows.end, pair[1].rows.start);
478        }
479        for region in &regions {
480            assert!(region.tile.origin[1] + halo <= region.rows.start.max(halo));
481            assert!(region.tile.origin[1] + region.tile.shape[1] <= ny);
482            assert!(region.tile.origin[1] + region.tile.shape[1] >= region.rows.end.min(ny - halo));
483        }
484    }
485
486    #[test]
487    fn waves_cover_the_planes_in_order() {
488        assert_eq!(
489            Residency::Waves { planes_per_wave: 3 }.waves(7),
490            vec![
491                PlaneRange::new(0, 3),
492                PlaneRange::new(3, 6),
493                PlaneRange::new(6, 7)
494            ]
495        );
496        assert_eq!(Residency::All.waves(5), vec![PlaneRange::new(0, 5)]);
497    }
