3//! Thin PyO3 `ia.crop` wrapper delegating to [`casa_imageanalysis::tool::ops_spatial::crop_image`].
4
5use casa_imageanalysis::analysis::ImageCropOptions;
6use casa_imageanalysis::tool::ops_spatial::crop_image;
7use pyo3::prelude::*;
8use pyo3::types::{PyDict, PyString};
9
10use crate::convert::IntList;
11
12use super::PyImageTool;
13use super::handle::{emit_tool_logs, external_open_check, raised};
14use super::subimage::Region;
15
16#[pymethods]
17impl PyImageTool {
18    #[allow(clippy::too_many_arguments)]
19    #[pyo3(signature = (outfile="", axes=None, overwrite=false, region=None, r#box="", chans="", stokes="", mask="", stretch=false, wantreturn=true), text_signature = "($self, outfile='', axes=[], overwrite=False, region={}, box='', chans='', stokes='', mask='', stretch=False, wantreturn=True)")]
20    fn crop(
21        &self,
22        py: Python<'_>,
23        outfile: &str,
24        axes: Option<IntList>,
25        overwrite: bool,
26        region: Option<&Bound<'_, PyAny>>,
27        r#box: &str,
28        chans: &str,
29        stokes: &str,
30        mask: &str,
31        stretch: bool,
32        wantreturn: bool,
33    ) -> PyResult<Py<Self>> {
34        let record_argument = region.is_none_or(|value| value.is_instance_of::<PyDict>());
35        let omitted_history = match region {
36            _ if record_argument => "{...}",
37            Some(value) if value.is_instance_of::<PyString>() => "\"\"",
38            _ => "[]",
39        };
40        let region = Region::capture(py, region);
41        let (result, logs) = py.detach(|| -> PyResult<_> {
42            let state = self.state();
43            let Some(attached) = state.image.as_ref() else {
44                return Ok((None, Vec::new()));
45            };
46            let region = region?;
47            if record_argument || region.is_some() {
48                if !r#box.is_empty() {
49                    return Err(raised("box and regionPtr cannot be simultaneously specified"));
50                }
51                if !chans.is_empty() || !stokes.is_empty() {
52                    return Err(raised(
53                        "regionPtr and regionName, chans, and/or stokes cannot be simultaneously specified",
54                    ));
55                }
56            }
57            let axes = axes.as_ref().map_or(&[][..], |axes| axes.0.as_slice());
58            let options = ImageCropOptions {
59                axes,
60                box_selection: (!r#box.is_empty()).then_some(r#box),
61                region: region.as_ref().map(Region::as_region),
62                channels: (!chans.is_empty()).then_some(chans),
63                stokes: (!stokes.is_empty()).then_some(stokes),
64                mask: (!mask.is_empty()).then_some(mask),
65                stretch,
66            };
67            let history_command = state.history.then(|| {
68                let axes_str = axes.iter().map(i64::to_string).collect::<Vec<_>>().join(",");
69                let region_hist = region
70                    .as_ref()
71                    .map_or_else(|| omitted_history.into(), Region::history);
72                format!(
73                    "ia.crop(outfile={outfile:?}, axes=[{axes_str}], overwrite={overwrite}, region={region_hist}, box={box:?}, chans={chans:?}, stokes={stokes:?}, mask={mask:?}, stretch={stretch}, wantreturn={wantreturn})",
74                    box = r#box
75                )
76            });
77            crop_image(
78                attached,
79                outfile,
80                overwrite,
81                options,
82                wantreturn,
83                history_command,
84                external_open_check(),
85            )
86            .map_err(raised)
87        })?;
88        emit_tool_logs(py, &logs)?;
89        Py::new(py, result.map_or_else(Self::new, Self::attached))
90    }
91}
