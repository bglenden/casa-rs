25/// Snapshot flags in physical row order for a new, private staging MS.
26/// `shapes` is the same FLAG layout used when writing the source table.
27#[doc(hidden)]
28pub fn save_initial_flag_version(
29    ms: &MeasurementSet,
30    shapes: Vec<MeasurementSetArrayShapePlan>,
31) -> Result<(), FlaggingError> {
32    let path = ms_path(ms)?;
33    ensure_flagversions_dir(path)?;
34    let output = flag_version_table_path(path, "Original");
35    write_flag_snapshot(ms, &output, Some(shapes), true)?;
36    append_flag_version_entry(path, "Original", "Original flags at import into CASA")
37}

39pub(super) fn write_flag_snapshot(
40    ms: &MeasurementSet,
41    output: &Path,
42    shapes: Option<Vec<MeasurementSetArrayShapePlan>>,
43    row_flags: bool,
44) -> Result<(), FlaggingError> {
45    let path = ms_path(ms)?;
46    let error = |reason| FlaggingError::FlagVersion {
47        path: path.display().to_string(),
48        reason,
49    };
50    let source = ms.main_table();
51    let shapes = match shapes {
52        Some(shapes) => shapes,
53        None => snapshot_shapes(source).map_err(error)?,
54    };
55    let options = TableOptions::new(output).with_data_manager(DataManagerKind::StandardStMan);
56    let schema = flag_version_schema(row_flags).map_err(|source| error(source.to_string()))?;
57    Table::with_schema(schema)
58        .save(options.clone())
59        .map_err(|source| error(source.to_string()))?;
60    let row_count = ms.row_count();
61    if row_count == 0 {
62        return Ok(());
63    }
64    let tile_rows = shapes
65        .iter()
66        .filter_map(|shape| shape.tile_shape.last().copied())
67        .max()
68        .unwrap_or(1);
69    let resources = MeasurementSetWriteResources::from_system_memory(2)
70        .map_err(|source| error(source.to_string()))?;
71    let plan = MeasurementSetWritePlan::variable_array_creation(
72        row_count,
73        vec![MeasurementSetArrayColumnPlan {
74            name: FLAG_COLUMN.into(),
75            value_type: StreamedTiledShapeValueType::Bool,
76            shapes,
77            storage_manager: MeasurementSetColumnStorage::TiledShape,
78        }],
79        row_flags
80            .then(|| MeasurementSetScalarColumnPlan {
81                name: FLAG_ROW_COLUMN.into(),
82                value_type: StreamedScalarType::Bool,
83            })
84            .into_iter()
85            .collect(),
86        resources,
87    )
88    .map_err(|source| error(source.to_string()))?;
89    // Keep the owned row views within one row band of the largest tile.
90    let batch_rows = plan.batch_rows().min(tile_rows);
91    let mut writer = MeasurementSetWriteSession::start(output, plan)
92        .map_err(|source| error(source.to_string()))?;
93    let flags = source
94        .column_accessor(FLAG_COLUMN)
95        .map_err(|source| error(source.to_string()))?;
96    let row_flags = row_flags
97        .then(|| source.column_accessor(FLAG_ROW_COLUMN))
98        .transpose()
99        .map_err(|source| error(source.to_string()))?;
100    for start in (0..row_count).step_by(batch_rows) {
101        let rows: Vec<_> = (start..(start + batch_rows).min(row_count)).collect();
102        write_snapshot_flags(&flags, &rows, &mut writer).map_err(error)?;
103        if let Some(row_flags) = &row_flags {
104            let scalars = row_flags
105                .scalar_cells_owned_for_rows(&rows)
106                .map_err(|source| error(source.to_string()))?;
107            for (row, scalar) in rows.into_iter().zip(scalars) {
108                let Some(ScalarValue::Bool(flag)) = scalar else {
109                    return Err(error(format!("missing Boolean FLAG_ROW in row {row}")));
110                };
111                writer
112                    .push_scalar_row(FLAG_ROW_COLUMN, Some(ScalarValue::Bool(flag)))
113                    .map_err(|source| error(source.to_string()))?;
114            }
115        }
116    }
117    let mut table = Table::open_metadata_for_construction(options.clone(), row_count)
118        .map_err(|source| error(source.to_string()))?;
119    let mut overrides = ColumnOverrides::for_row_count(row_count);
120    overrides.insert_deferred(FLAG_COLUMN);
121    let bindings = HashMap::from([(
122        FLAG_COLUMN.into(),
123        ColumnBinding {
124            data_manager: DataManagerKind::TiledShapeStMan,
125            tile_shape: None,
126        },
127    )]);
128    writer
129        .save_table_and_finish(&mut table, options, &bindings, &overrides)
130        .map_err(|source| error(source.to_string()))?;
131    Ok(())
132}

134fn write_snapshot_flags(
135    flags: &TableColumn<'_>,
136    rows: &[usize],
137    writer: &mut MeasurementSetWriteSession,
138) -> Result<(), String> {
139    if let Some(ArrayValue::Bool(block)) = flags
140        .packed_2d_cells_uncached(rows)
141        .map_err(|error| error.to_string())?
142    {
143        // Uniform cells are packed in physical row order without one
144        // allocation per cell. The writer consumes each contiguous row.
145        let shape = &block.shape()[..2];
146        let values = block
147            .as_slice_memory_order()
148            .expect("packed column is contiguous");
149        for row in values.chunks_exact(shape.iter().product()) {
150            writer
151                .push_bool_row(FLAG_COLUMN, shape, row)
152                .map_err(|error| error.to_string())?;
153        }
154        return Ok(());
155    }
156
157    let arrays = flags
158        .array_cells_owned_uncached(rows)
159        .map_err(|error| error.to_string())?;
160    for (&row, array) in rows.iter().zip(arrays) {
161        let Some(ArrayValue::Bool(array)) = array else {
162            return Err(format!("missing Boolean FLAG in row {row}"));
163        };
164        let casa_order = array.t();
165        let values = match casa_order.to_slice() {
166            Some(values) => Cow::Borrowed(values),
167            None => Cow::Owned(casa_order.iter().copied().collect()),
168        };
169        writer
170            .push_bool_row(FLAG_COLUMN, array.shape(), &values)
171            .map_err(|error| error.to_string())?;
172    }
173    Ok(())
174}

176fn snapshot_shapes(table: &Table) -> Result<Vec<MeasurementSetArrayShapePlan>, String> {
177    let mut histogram = BTreeMap::new();
178    for start in (0..table.row_count()).step_by(256) {
179        let rows: Vec<_> = (start..(start + 256).min(table.row_count())).collect();
180        let shapes = read_flag_shapes(table, &rows).map_err(|source| source.to_string())?;
181        for (row, shape) in rows.into_iter().zip(shapes) {
182            let Some([correlations, channels]) = shape.as_deref() else {
183                return Err(format!("missing two-dimensional FLAG in row {row}"));
184            };
185            *histogram.entry((*correlations, *channels)).or_insert(0) += 1;
186        }
187    }
188    Ok(histogram
189        .into_iter()
190        .map(|((correlations, channels), rows)| {
191            MeasurementSetArrayShapePlan::visibility(correlations, channels, rows, "")
192        })
193        .collect())
194}

196pub(super) fn read_flag_shapes(
197    table: &Table,
198    rows: &[usize],
199) -> Result<Vec<Option<Vec<usize>>>, TableError> {
200    let column = table.column_accessor(FLAG_COLUMN)?;
201    column.array_shapes_uncached(rows).or_else(|_| {
202        Ok(column
203            .array_cells_owned_uncached(rows)?
204            .into_iter()
205            .map(|cell| cell.map(|array| array.shape().to_vec()))
206            .collect())
207    })
208}
