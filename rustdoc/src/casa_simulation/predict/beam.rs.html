3//! patterns, as `PBMath` gives them by telescope name.
4
5use casa_numerics::AnnularApertureVoltageTable;
6use ndarray::Array2;
7
8use super::SkyModel;

10/// The primary-beam voltage pattern `setvp(dovp=True)` applies.
11#[derive(Clone, Copy, Debug, PartialEq)]
12pub enum PrimaryBeam {
13    /// CASA `PBMath1DAiry`: an Airy pattern for a dish with a central
14    /// blockage, tabulated out to a maximum radius.
15    Airy {
16        /// Effective dish diameter in metres.
17        dish_diameter_m: f64,
18        /// Blockage diameter in metres.
19        blockage_diameter_m: f64,
20        /// Radius of the tabulated pattern at 1 GHz, in arcmin (scaled as
21        /// 1/frequency). The pattern is zero beyond it, and CASA's 10,000
22        /// table samples span it, so it also sets the lookup quantization.
23        maximum_radius_arcmin_ghz: f64,
24    },
25}
26
27impl PrimaryBeam {
28    /// ALMA 12 m antennas as CASA's default voltage patterns give them
29    /// (`PBMath::initByTelescope`): 10.7 m effective diameter, 0.75 m
30    /// blockage, 1.784° at 1 GHz.
31    pub const ALMA: Self = Self::Airy {
32        dish_diameter_m: 10.7,
33        blockage_diameter_m: 0.75,
34        maximum_radius_arcmin_ghz: 1.784 * 60.0,
35    };
36
37    /// ACA 7 m antennas as CASA's default voltage patterns give them
38    /// (`PBMath::initByTelescope`): 6.25 m effective diameter, 0.75 m
39    /// blockage, 3.568° at 1 GHz.
40    pub const ACA: Self = Self::Airy {
41        dish_diameter_m: 6.25,
42        blockage_diameter_m: 0.75,
43        maximum_radius_arcmin_ghz: 3.568 * 60.0,
44    };
45
46    /// CASA's default voltage pattern for `telescope`, the OBSERVATION
47    /// table's TELESCOPE_NAME, where it is one of these Airy patterns:
48    /// `PBMath::enumerateCommonPB` matches the name's prefix, `ALMA` (which
49    /// also catches `ALMASD`, whose pattern is ALMA's) and `ACA`, and maps
50    /// `ATF` to ALMA. `None` for every other telescope, whose patterns
51    /// (VLA's polynomial among them) casa-rs does not yet have.
52    #[must_use]
53    pub fn for_telescope(telescope: &str) -> Option<Self> {
54        if telescope.starts_with("ALMA") || telescope.starts_with("ATF") {
55            Some(Self::ALMA)
56        } else if telescope.starts_with("ACA") {
57            Some(Self::ACA)
58        } else {
59            None
60        }
61    }
62
63    /// The pattern's lookup table, shared by every plane it is evaluated
64    /// on.
65    pub(crate) fn pattern(self) -> BeamPattern {
66        let Self::Airy {
67            dish_diameter_m,
68            blockage_diameter_m,
69            maximum_radius_arcmin_ghz,
70        } = self;
71        BeamPattern {
72            table: AnnularApertureVoltageTable::new(
73                dish_diameter_m,
74                blockage_diameter_m,
75                maximum_radius_arcmin_ghz,
76            ),
77        }
78    }
79}
80
81/// A primary beam's tabulated voltage pattern.
82#[derive(Clone)]
83pub(crate) struct BeamPattern {
84    table: AnnularApertureVoltageTable,
85}
86
87impl BeamPattern {
88    /// Bytes the pattern holds.
89    pub(crate) const BYTES: u64 = AnnularApertureVoltageTable::BYTES as u64;
90
91    /// The power pattern on model plane `plane`, `[y][x]`, centred on the
92    /// pixel `centre`, at the plane's frequency (`BeamSkyJones::apply`
93    /// squares the voltage pattern).
94    pub(super) fn power_plane(
95        &self,
96        model: &SkyModel,
97        plane: usize,
98        centre: [f64; 2],
99    ) -> Array2<f32> {
100        let [nx, ny] = model.shape();
101        let ghz = model.plane_frequency_hz(plane) / 1.0e9;
102        let increment = model.increment_rad;
103        Array2::from_shape_fn((ny, nx), |(y, x)| {
104            // PBMath1D works in `Float` degrees: keep its rounding of the
105            // radius at the lookup boundaries.
106            let l_deg = ((x as f64 - centre[0]) * increment[0]).to_degrees();
107            let m_deg = ((y as f64 - centre[1]) * increment[1]).to_degrees();
108            let [l_deg, m_deg] = [l_deg as f32, m_deg as f32];
109            let radius_deg = (l_deg * l_deg + m_deg * m_deg).sqrt();
110            let radius = (f64::from(radius_deg) * 60.0 * ghz) as f32;
111            let voltage = self.table.evaluate(f64::from(radius));
112            voltage * voltage
113        })
114    }
115}

117#[cfg(test)]
118mod tests {
119    use super::*;
120
121    #[test]
122    fn telescopes_match_casas_common_beams_by_prefix() {
123        assert_eq!(PrimaryBeam::for_telescope("ALMA"), Some(PrimaryBeam::ALMA));
124        assert_eq!(
125            PrimaryBeam::for_telescope("ALMASD"),
126            Some(PrimaryBeam::ALMA)
127        );
128        assert_eq!(PrimaryBeam::for_telescope("ATF"), Some(PrimaryBeam::ALMA));
129        assert_eq!(PrimaryBeam::for_telescope("ACA"), Some(PrimaryBeam::ACA));
130        assert_eq!(PrimaryBeam::for_telescope("ACA-7m"), Some(PrimaryBeam::ACA));
131        for other in ["VLA", "EVLA", "alma", "Unknown", "", "SMA", "AL"] {
132            assert_eq!(PrimaryBeam::for_telescope(other), None, "{other}");
133        }
134    }
