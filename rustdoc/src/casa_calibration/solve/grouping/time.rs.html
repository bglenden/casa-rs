11/// Each selected row of `rows`, in solve order, with the group and
12/// interval it is solved in.
13///
14/// CASA's iteration sorts by the axes `combine` keeps apart and then by
15/// time, so the rows of a group come in time order whatever their fields
16/// and scans, and a `solint` interval starts at its group's earliest time.
17pub(super) fn bucketed_rows(
18    rows: &[SelectedSolveRow],
19    solve_interval: GainSolveInterval,
20    combine: GainSolveCombine,
21    include_autocorrelations: bool,
22) -> Vec<BucketedRow<'_>> {
23    let mut keyed = rows
24        .iter()
25        .filter(|row| include_autocorrelations || row.antenna1 != row.antenna2)
26        .map(|row| {
27            let base_key = SolveBaseKey {
28                field_id: if combine.fields { 0 } else { row.field_id },
29                spw_id: row.data_spw_id,
30                observation_id: row.observation_id,
31                scan_number: if combine.scans { 0 } else { row.scan_number },
32            };
33            (row, base_key)
34        })
35        .collect::<Vec<_>>();
36    keyed.sort_by(|(a, a_key), (b, b_key)| {
37        a_key
38            .cmp(b_key)
39            .then(a.time_seconds.total_cmp(&b.time_seconds))
40            .then(a.interval_seconds.total_cmp(&b.interval_seconds))
41            .then(a.row_index.cmp(&b.row_index))
42    });
43
44    let mut bucket_starts = BTreeMap::<SolveBaseKey, f64>::new();
45    keyed
46        .into_iter()
47        .map(|(row, base_key)| {
48            let bucket_key = match solve_interval {
49                GainSolveInterval::Infinite => SolveBucketKey::Infinite,
50                GainSolveInterval::Integration => SolveBucketKey::Integration {
51                    time_bits: row.time_seconds.to_bits(),
52                    interval_bits: row.interval_seconds.to_bits(),
53                },
54                GainSolveInterval::Seconds(seconds) => {
55                    // MSIter2 starts its time grid 10 ms before the first
56                    // timestamp and receives solint as a Float.
57                    let start = *bucket_starts
58                        .entry(base_key)
59                        .or_insert(row.time_seconds - 0.01);
60                    let index = ((row.time_seconds - start) / f64::from(seconds as f32))
61                        .floor()
62                        .max(0.0);
63                    SolveBucketKey::Seconds(index as u64)
64                }
65            };
66            (row, base_key, bucket_key)
67        })
68        .collect()
69}

72pub(super) struct FirstTBuffer {

75    pub(super) max_span: Option<f64>,

79/// First buffer seen by `TJones::guessPar`. The CASA
80/// initializer starts at row 1. The skipped row can have zero weight or be
81/// an autocorrelation, so decide this using row metadata before reading DATA.
82pub(super) fn first_t_buffers(
83    rows: &[BucketedRow<'_>],
84    max_span: Option<f64>,
85) -> BTreeMap<(SolveBaseKey, SolveBucketKey), FirstTBuffer> {
86    let mut first = BTreeMap::new();
87    for &(row, base, bucket) in rows {
88        let seed = first.entry((base, bucket)).or_insert(row);
89        // BaselineIndex::configure uses the row order of the first input
90        // buffer, including rows later ignored by the solver.
91        if (row.field_id, row.time_seconds, row.row_index)
92            < (seed.field_id, seed.time_seconds, seed.row_index)
93        {
94            *seed = row;
95        }
96    }
97    first
98        .into_iter()
99        .map(|(key, row)| {
100            (
101                key,
102                FirstTBuffer {
103                    field: row.field_id,
104                    time: row.time_seconds,
105                    max_span,
106                    skipped_baseline: (row.antenna1, row.antenna2),
107                },
108            )
109        })
110        .collect()
111}
