2//! The engine of an observation's catalogue, whatever its store
3//! ([`MsCalEngine::from_catalogue`]).
4
5use std::collections::HashMap;
6use std::sync::{Arc, RwLock};
7
8use casa_observation::Catalogue;
9use casa_types::measures::MeasuresProvider;
10use casa_types::measures::direction::DirectionRef;
11use casa_types::measures::epoch::EpochRef;
12use casa_types::measures::frame::MeasFrame;
13use casa_types::measures::position::MPosition;
14
15use super::{FieldMotion, MsCalEngine};
16use crate::error::MsResult;
17
18impl MsCalEngine {
19    /// An engine over the observation `catalogue` describes, whatever its
20    /// store, evaluating every frame conversion with `measures`: what
21    /// [`Self::with_measures`] and [`Self::with_field_ephemerides`] read from
22    /// a MeasurementSet's subtables, read from the catalogue. Its antennas
23    /// and their mounts, the observatory its first observation's telescope
24    /// names (else its first antenna, as CASA's array reference), its
25    /// fields' phase centres, the epochs they are stated at and the
26    /// ephemerides they follow, and the time scale of its partitions' times
27    /// (`UTC` without a partition).
28    ///
29    /// For a MeasurementSet's catalogue this is the engine
30    /// [`Self::with_measures`] builds, field ephemerides attached: the MSv2
31    /// adapter reads the same subtables the same way (law
32    /// `the_catalogue_engine_is_the_measurement_set_engine`).
33    ///
34    /// # Errors
35    ///
36    /// A field's phase centre that cannot be converted to J2000.
37    pub fn from_catalogue(
38        catalogue: &Catalogue,
39        measures: Arc<dyn MeasuresProvider>,
40    ) -> MsResult<Self> {
41        let antenna_positions = catalogue
42            .antennas
43            .iter()
44            .map(|antenna| {
45                let [x, y, z] = antenna.position.as_itrf();
46                MPosition::new_itrf(x, y, z)
47            })
48            .collect::<Vec<_>>();
49        let antenna_mount_alt_az = catalogue
50            .antennas
51            .iter()
52            .map(casa_observation::AntennaEntry::alt_az)
53            .collect::<Vec<_>>();
54        let observatory_position = MPosition::casa_array_reference(
55            catalogue
56                .observations
57                .first()
58                .map(|observation| observation.telescope.as_str()),
59            antenna_positions.first(),
60            measures.as_ref(),
61        );
62        let mut field_directions = Vec::with_capacity(catalogue.fields.len());
63        let mut field_phase_offsets = Vec::with_capacity(catalogue.fields.len());
64        let mut field_reference_epochs = Vec::with_capacity(catalogue.fields.len());
65        let mut field_motions = Vec::with_capacity(catalogue.fields.len());
66        for field in &catalogue.fields {
67            let (longitude, latitude) = field.phase_centre.as_angles();
68            field_phase_offsets.push([longitude, latitude]);
69            field_directions.push(if field.phase_centre.refer() == DirectionRef::J2000 {
70                field.phase_centre.clone()
71            } else {
72                let frame = MeasFrame::new()
73                    .with_position(observatory_position.clone())
74                    .with_measures(Arc::clone(&measures));
75                let frame = match &field.reference_epoch {
76                    Some(epoch) => frame.with_epoch(epoch.clone()),
77                    None => frame,
78                };
79                field.phase_centre.convert_to(DirectionRef::J2000, &frame)?
80            });
81            field_reference_epochs.push(field.reference_epoch.clone());
82            field_motions.push(match &field.ephemeris {
83                Some(ephemeris) => FieldMotion::Ephemeris(Arc::clone(ephemeris)),
84                None => FieldMotion::Fixed,
85            });
86        }
87        let time_reference = catalogue
88            .partitions
89            .first()
90            .map_or(EpochRef::UTC, |partition| partition.time_scale);
91        Ok(Self {
92            antenna_positions: antenna_positions.into_boxed_slice(),
93            antenna_mount_alt_az: antenna_mount_alt_az.into_boxed_slice(),
94            field_directions: field_directions.into_boxed_slice(),
95            field_phase_offsets: field_phase_offsets.into_boxed_slice(),
96            field_motions: field_motions.into_boxed_slice(),
97            field_reference_epochs: field_reference_epochs.into_boxed_slice(),
98            observatory_position,
99            time_reference,
100            measures: Some(measures),
101            azel_cache: RwLock::new(HashMap::new()),
102            hadec_cache: RwLock::new(HashMap::new()),
103            parallactic_angle_cache: RwLock::new(HashMap::new()),
104        })
105    }
106}
