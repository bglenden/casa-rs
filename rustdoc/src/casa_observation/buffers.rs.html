17/// The storage behind one [`VisibilityBatch`].
18///
19/// Owned by the caller (or a [`BufferPool`](crate::BufferPool)), filled in
20/// place by a cursor and lent to the batch. Its arrays are private: a
21/// cursor fills them through [`Self::begin`], which checks the batch's
22/// [`BatchHeader`] against the observation's [`Catalogue`] and reserves
23/// exactly the arrays it describes, and [`BatchFill::finish`] checks the
24/// filled arrays once; the buffers then keep the header and a shared
25/// reference to that catalogue. The batch is built from the header, the
26/// arrays and the catalogue's partition entry and nothing else, so no
27/// second description of the same data can drift from the arrays: the
28/// per-partition constants (frequencies, correlations, baselines, time
29/// system, data group) are never copied into a batch, and no other
30/// catalogue can be paired with the arrays.
31///
32/// Clearing keeps capacity, so after the first fill of the largest batch a
33/// traversal allocates nothing. Buffers made [`Self::with_envelope`] never
34/// hold more than their envelope, whatever sequence of batch shapes and
35/// roles they are filled with. The buffers are `Send`: they can be handed
36/// between work units whole, which is how a batch's lifetime is separated
37/// from a traversal's.
38#[derive(Debug, Default)]
39pub struct BatchBuffers {
40    arrays: BatchArrays,
41    header: Option<BatchHeader>,
42    catalogue: Option<Arc<Catalogue>>,
43    envelope: Option<usize>,
44}
45
46/// The arrays of one batch, as a cursor fills them.
47#[derive(Clone, Debug, Default)]
48pub struct BatchArrays {
49    /// Visibilities, `[sample][channel][correlation]`.
50    pub values: Vec<Complex32>,
51    /// Flags, the same shape.
52    pub flags: Vec<bool>,
53    /// Weights, `[sample][channel][correlation]`, or `[sample][correlation]`
54    /// when the header says they are not per channel.
55    pub weights: Vec<f32>,
56    /// UVW, `[sample][3]`.
57    pub uvw: Vec<f64>,
58    /// Time offsets in seconds from the partition's origin, per time.
59    pub time_offsets_s: Vec<f64>,
60    /// Time-centroid offsets in seconds, per sample.
61    pub centroid_offsets_s: Vec<f64>,
62    /// Intervals in seconds, in the header's sharing.
63    pub interval_s: Vec<f64>,
64    /// Exposures in seconds, in the header's sharing.
65    pub exposure_s: Vec<f64>,
66    /// Field per time.
67    pub field: Vec<FieldId>,
68    /// Scan per time, when the roles include it.
69    pub scan: Vec<ScanId>,
70    /// Scientific-group fragments.
71    pub fragments: Vec<GroupFragment>,
72}

74/// How a per-sample quantity is stored in a batch ([`Shared`]).
75#[derive(Clone, Copy, Debug, PartialEq)]
76pub enum Sharing<T> {
77    /// One value per sample.
78    PerSample,
79    /// One value per time.
80    PerTime,
81    /// One value for every sample; the arrays hold none.
82    Constant(T),
83}

85/// Everything about a batch except its arrays' contents and what the
86/// catalogue already states: what a cursor decides before it fills, and
87/// what [`BatchFill::finish`] checks the arrays against.
88#[derive(Clone, Debug, PartialEq)]
89pub struct BatchHeader {
90    /// The partition.
91    pub partition: PartitionId,
92    /// The data group: an index into the partition's
93    /// [`data_groups`](PartitionEntry::data_groups).
94    pub data_group: usize,
95    /// Whole times in the batch; the samples are these times by the
96    /// partition's baselines.
97    pub times: usize,
98    /// The partition's native channel of the first channel.
99    pub first_channel: usize,
100    /// Channels.
101    pub channels: usize,
102    /// The delivered correlation products: the traversal's resolved axis,
103    /// shared, never a copy of the list.
104    pub correlations: CorrelationAxis,
105    /// The roles filled.
106    pub roles: Roles,
107    /// Whether the weights are per channel (`[sample][channel][correlation]`)
108    /// or one per correlation shared by every channel.
109    pub weights_per_channel: bool,
110    /// What the weights are: stored, or made from the sigmas.
111    pub weight_rule: WeightRule,
112    /// How intervals are stored, when the roles include them.
113    pub interval: Sharing<f64>,
114    /// How exposures are stored, when the roles include them.
115    pub exposure: Sharing<f64>,
116}

129/// The length of every array of the batch a header describes.
130#[derive(Clone, Copy, Debug, PartialEq, Eq)]
131struct Lengths {
132    values: usize,
133    flags: usize,
134    weights: usize,
135    uvw: usize,
136    times: usize,
137    centroid: usize,
138    interval: usize,
139    exposure: usize,
140    scan: usize,
141}
142
143impl Lengths {
144    /// The lengths for `header` over `baselines`, with overflow-checked
145    /// sizes; the fragments have room for one per time.
146    fn of(
147        header: &BatchHeader,
148        baselines: usize,
149        correlations: usize,
150    ) -> Result<Self, ObservationError> {
151        let times = header.times;
152        let samples = product("samples", times, baselines)?;
153        let cell = product(
154            "cube",
155            samples,
156            product("cube", header.channels, correlations)?,
157        )?;
158        let roles = header.roles;
159        let wanted = |role: bool, length: usize| if role { length } else { 0 };
160        let shared = |sharing: Sharing<f64>| match sharing {
161            Sharing::PerSample => samples,
162            Sharing::PerTime => times,
163            Sharing::Constant(_) => 0,
164        };
165        let weights = if header.weights_per_channel {
166            cell
167        } else {
168            product("weights", samples, correlations)?
169        };
170        Ok(Self {
171            values: wanted(roles.values, cell),
172            flags: wanted(roles.flags, cell),
173            weights: wanted(roles.weights, weights),
174            uvw: wanted(roles.uvw, product("uvw", samples, 3)?),
175            times,
176            centroid: wanted(roles.time_centroid, samples),
177            interval: wanted(roles.interval, shared(header.interval)),
178            exposure: wanted(roles.exposure, shared(header.exposure)),
179            scan: wanted(roles.scan, times),
180        })
181    }
182
183    /// Bytes the arrays hold at these lengths
184    /// ([`PartitionEstimate::buffer_bytes`](crate::PartitionEstimate::buffer_bytes)
185    /// for a full batch).
186    fn bytes(&self) -> usize {
187        // Saturating: a header too large to allocate exceeds any envelope.
188        [
189            (self.values, size_of::<Complex32>()),
190            (self.flags, size_of::<bool>()),
191            (self.weights, size_of::<f32>()),
192            (self.uvw, size_of::<f64>()),
193            (self.times, size_of::<f64>()),
194            (self.centroid, size_of::<f64>()),
195            (self.interval, size_of::<f64>()),
196            (self.exposure, size_of::<f64>()),
197            (
198                self.times,
199                size_of::<FieldId>() + size_of::<GroupFragment>(),
200            ),
201            (self.scan, size_of::<ScanId>()),
202        ]
203        .into_iter()
204        .fold(0, |total: usize, (length, size)| {
205            total.saturating_add(length.saturating_mul(size))
206        })
207    }
208
209    /// Bytes `arrays` would hold once reserved to these lengths, keeping
210    /// any larger capacity they have.
211    fn retained(&self, arrays: &BatchArrays) -> usize {
212        fn held<T>(array: &Vec<T>, length: usize) -> usize {
213            array.capacity().max(length).saturating_mul(size_of::<T>())
214        }
215        [
216            held(&arrays.values, self.values),
217            held(&arrays.flags, self.flags),
218            held(&arrays.weights, self.weights),
219            held(&arrays.uvw, self.uvw),
220            held(&arrays.time_offsets_s, self.times),
221            held(&arrays.centroid_offsets_s, self.centroid),
222            held(&arrays.interval_s, self.interval),
223            held(&arrays.exposure_s, self.exposure),
224            held(&arrays.field, self.times),
225            held(&arrays.scan, self.scan),
226            held(&arrays.fragments, self.times),
227        ]
228        .into_iter()
229        .fold(0, usize::saturating_add)
230    }
231
232    /// Reserve exactly these lengths in the cleared `arrays`; with `trim`,
233    /// first free every array whose capacity exceeds its length.
234    fn reserve(&self, arrays: &mut BatchArrays, trim: bool) {
235        fn exact<T>(array: &mut Vec<T>, length: usize, trim: bool) {
236            if trim && array.capacity() > length {
237                *array = Vec::new();
238            }
239            array.reserve_exact(length);
240        }
241        exact(&mut arrays.values, self.values, trim);
242        exact(&mut arrays.flags, self.flags, trim);
243        exact(&mut arrays.weights, self.weights, trim);
244        exact(&mut arrays.uvw, self.uvw, trim);
245        exact(&mut arrays.time_offsets_s, self.times, trim);
246        exact(&mut arrays.centroid_offsets_s, self.centroid, trim);
247        exact(&mut arrays.interval_s, self.interval, trim);
248        exact(&mut arrays.exposure_s, self.exposure, trim);
249        exact(&mut arrays.field, self.times, trim);
250        exact(&mut arrays.scan, self.scan, trim);
251        exact(&mut arrays.fragments, self.times, trim);
252    }
253}

255impl BatchBuffers {
256    /// Buffers that never hold more than `envelope` bytes: a fill whose
257    /// batch needs more is refused before anything is allocated, and arrays
258    /// kept from earlier fills are freed when keeping them would exceed it.
259    #[must_use]
260    pub fn with_envelope(envelope: usize) -> Self {
261        Self {
262            envelope: Some(envelope),
263            ..Self::default()
264        }
265    }
266
267    /// The bytes these buffers may hold, if bounded.
268    #[must_use]
269    pub fn envelope(&self) -> Option<usize> {
270        self.envelope
271    }
272
273    /// Empty every array, keeping its capacity, and forget the batch.
274    pub fn clear(&mut self) {
275        self.header = None;
276        self.catalogue = None;
277        let arrays = &mut self.arrays;
278        arrays.values.clear();
279        arrays.flags.clear();
280        arrays.weights.clear();
281        arrays.uvw.clear();
282        arrays.time_offsets_s.clear();
283        arrays.centroid_offsets_s.clear();
284        arrays.interval_s.clear();
285        arrays.exposure_s.clear();
286        arrays.field.clear();
287        arrays.scan.clear();
288        arrays.fragments.clear();
289    }
290
291    /// Start filling a batch described by `header`, for an observation
292    /// with `catalogue`: the header is checked against the catalogue (the
293    /// partition, data group, channel window and correlation axis exist),
294    /// and the arrays are cleared and reserved to exactly the batch's
295    /// lengths, then lent through the returned fill.
296    ///
297    /// # Errors
298    ///
299    /// [`ObservationError::InvalidBatch`] when the header does not fit the
300    /// catalogue; [`ObservationError::InvalidRequest`] when the batch needs
301    /// more than the buffers' envelope. Nothing has been allocated, and the
302    /// buffers hold no batch.
303    pub fn begin(
304        &mut self,
305        header: BatchHeader,
306        catalogue: &Arc<Catalogue>,
307    ) -> Result<BatchFill<'_>, ObservationError> {
308        self.clear();
309        let entry = context(&header, catalogue)?;
310        let lengths = Lengths::of(
311            &header,
312            entry.baselines.len(),
313            header.correlations.products(entry).len(),
314        )?;
315        let trim = match self.envelope {
316            Some(envelope) if lengths.bytes() > envelope => {
317                return Err(ObservationError::InvalidRequest {
318                    reason: format!(
319                        "a batch of {} B exceeds its buffer's {envelope} B envelope",
320                        lengths.bytes()
321                    ),
322                });
323            }
324            Some(envelope) => lengths.retained(&self.arrays) > envelope,
325            None => false,
326        };
327        lengths.reserve(&mut self.arrays, trim);
328        Ok(BatchFill {
329            buffers: self,
330            header,
331            catalogue: Arc::clone(catalogue),
332            lengths,
333        })
334    }

336    /// The arrays, read-only: what the batch's views borrow.
337    #[must_use]
338    pub fn arrays(&self) -> &BatchArrays {
339        &self.arrays
340    }
341
342    /// The header of the batch the buffers hold, if they hold one.
343    #[must_use]
344    pub fn header(&self) -> Option<&BatchHeader> {
345        self.header.as_ref()
346    }
347
348    /// Bytes the arrays hold now, from their capacities.
349    #[must_use]
350    pub fn capacity_bytes(&self) -> usize {
351        let arrays = &self.arrays;
352        arrays.values.capacity() * size_of::<Complex32>()
353            + arrays.flags.capacity() * size_of::<bool>()
354            + arrays.weights.capacity() * size_of::<f32>()
355            + (arrays.uvw.capacity()
356                + arrays.time_offsets_s.capacity()
357                + arrays.centroid_offsets_s.capacity()
358                + arrays.interval_s.capacity()
359                + arrays.exposure_s.capacity())
360                * size_of::<f64>()
361            + arrays.field.capacity() * size_of::<FieldId>()
362            + arrays.scan.capacity() * size_of::<ScanId>()
363            + arrays.fragments.capacity() * size_of::<GroupFragment>()
364    }
365
366    /// The batch the buffers hold, if a fill finished: views of the arrays
367    /// and of the entry for the batch's partition in the catalogue the fill
368    /// was checked against.
369    #[must_use]
370    pub fn batch<'a>(&'a self) -> Option<VisibilityBatch<'a>> {
371        let (header, catalogue) = (self.header.as_ref()?, self.catalogue.as_ref()?);
372        let entry = catalogue.partition(header.partition);
373        let arrays = &self.arrays;
374        let baselines = entry.baselines.len();
375        let samples = header.times * baselines;
376        let channels = header.channels;
377        let products = header.correlations.products(entry);
378        let correlations = products.len();
379        let cube = [samples, channels, correlations];
380        let roles = header.roles;
381        let epochs = |offsets: &'a [f64]| {
382            EpochView::new(
383                entry.time_scale,
384                entry.time_origin,
385                ArrayView1::from(offsets),
386            )
387        };
388        let group = &entry.data_groups[header.data_group];
389        Some(VisibilityBatch {
390            catalogue,
391            partition: header.partition,
392            time: epochs(&arrays.time_offsets_s),
393            baselines: &entry.baselines,
394            field: ArrayView1::from(&arrays.field[..]),
395            scan: roles.scan.then(|| ArrayView1::from(&arrays.scan[..])),
396            timing: SampleTiming {
397                centroid: roles
398                    .time_centroid
399                    .then(|| epochs(&arrays.centroid_offsets_s)),
400                interval_s: roles
401                    .interval
402                    .then(|| shared(header.interval, &arrays.interval_s)),
403                exposure_s: roles
404                    .exposure
405                    .then(|| shared(header.exposure, &arrays.exposure_s)),
406            },
407            first_channel: header.first_channel,
408            frequency: FrequencyView::new(
409                entry.frequency_frame,
410                ArrayView1::from(
411                    &entry.frequencies_hz[header.first_channel..header.first_channel + channels],
412                ),
413            ),
414            correlations: products,
415            axis: &header.correlations,
416            group: DataGroupView {
417                entry: group,
418                values: roles.values.then(|| cube_view(cube, &arrays.values)),
419                flags: roles.flags.then(|| cube_view(cube, &arrays.flags)),
420                weights: roles.weights.then(|| {
421                    if header.weights_per_channel {
422                        cube_view(cube, &arrays.weights)
423                    } else {
424                        // One weight per correlation, shared by every
425                        // channel: the same view with a zero channel stride.
426                        ArrayView3::from_shape(
427                            cube.strides([correlations, 0, 1]),
428                            &arrays.weights[..],
429                        )
430                        .expect("length checked at finish")
431                    }
432                }),
433                weight_rule: header.weight_rule,
434                uvw: roles.uvw.then(|| {
435                    UvwView::new(
436                        group.uvw_frame.effective,
437                        ArrayView2::from_shape([samples, 3], &arrays.uvw[..]).expect("checked uvw"),
438                    )
439                }),
440            },
441            fragments: &arrays.fragments,
442        })
443    }
444}

446/// `values` in `sharing`.
447fn shared<T: Copy>(sharing: Sharing<T>, values: &[T]) -> Shared<'_, T> {
448    match sharing {
449        Sharing::PerSample => Shared::PerSample(ArrayView1::from(values)),
450        Sharing::PerTime => Shared::PerTime(ArrayView1::from(values)),
451        Sharing::Constant(value) => Shared::Constant(value),
452    }
453}

455/// A `[sample, channel, correlation]` view of `data`, whose length
456/// [`BatchFill::finish`] checked.
457fn cube_view<T>(cube: [usize; 3], data: &[T]) -> ArrayView3<'_, T> {
458    ArrayView3::from_shape(cube, data).expect("length checked at finish")
459}

461impl BatchFill<'_> {
462    /// The arrays to fill.
463    pub fn arrays(&mut self) -> &mut BatchArrays {
464        &mut self.buffers.arrays
465    }
466
467    /// The header the arrays must match.
468    #[must_use]
469    pub fn header(&self) -> &BatchHeader {
470        &self.header
471    }
472
473    /// Check every array against the header, once, and keep the header and
474    /// the catalogue: the lengths [`BatchBuffers::begin`] reserved; field
475    /// and scan identities against the catalogue's counts; and the
476    /// fragments, which must tile the samples in order by whole times.
477    ///
478    /// # Errors
479    ///
480    /// [`ObservationError::Shape`] or [`ObservationError::InvalidBatch`]: a
481    /// cursor bug or a store the adapter should have refused, never a
482    /// property a consumer must handle. The buffers then hold no batch.
483    pub fn finish(self) -> Result<(), ObservationError> {
484        let Self {
485            buffers,
486            header,
487            catalogue,
488            lengths,
489        } = self;
490        check(&buffers.arrays, &header, &catalogue, lengths)?;
491        buffers.header = Some(header);
492        buffers.catalogue = Some(catalogue);
493        Ok(())
494    }
495}

497fn invalid(reason: impl Into<String>) -> ObservationError {
498    ObservationError::InvalidBatch {
499        reason: reason.into(),
500    }

503/// `a × b`, or a shape error naming `array`.
504fn product(array: &'static str, a: usize, b: usize) -> Result<usize, ObservationError> {
505    a.checked_mul(b)
506        .ok_or_else(|| invalid(format!("{array} size overflows")))
507}
508
509fn length(array: &'static str, actual: usize, expected: usize) -> Result<(), ObservationError> {
510    if actual == expected {
511        Ok(())
512    } else {
513        Err(ObservationError::Shape {
514            array,
515            actual,
516            expected,
517        })
518    }
519}
520
521fn identities(
522    what: &'static str,
523    ids: impl IntoIterator<Item = u32>,
524    limit: usize,
525) -> Result<(), ObservationError> {
526    match ids.into_iter().find(|id| *id as usize >= limit) {
527        Some(id) => Err(invalid(format!(
528            "{what} {id} is not in the catalogue's {limit}"
529        ))),
530        None => Ok(()),
531    }
532}

534/// The header's partition, data group, channel window and correlations, as
535/// the catalogue describes them.
536fn context<'c>(
537    header: &BatchHeader,
538    catalogue: &'c Catalogue,
539) -> Result<&'c PartitionEntry, ObservationError> {
540    let entry = catalogue
541        .partitions
542        .get(header.partition.0 as usize)
543        .ok_or_else(|| invalid(format!("no partition {:?}", header.partition)))?;
544    if header.data_group >= entry.data_groups.len() {
545        return Err(invalid(format!("no data group {}", header.data_group)));
546    }
547    if header.channels == 0
548        || header
549            .first_channel
550            .checked_add(header.channels)
551            .is_none_or(|end| end > entry.frequencies_hz.len())
552    {
553        return Err(invalid(format!(
554            "channels {}..{} outside the partition's {}",
555            header.first_channel,
556            header.first_channel.saturating_add(header.channels),
557            entry.frequencies_hz.len()
558        )));
559    }
560    if !header.correlations.fits(entry) {
561        return Err(invalid(format!(
562            "correlations {:?} are not the partition's {:?} in its order",
563            header.correlations, entry.correlations
564        )));
565    }
566    if header.times == 0 || entry.baselines.is_empty() {
567        return Err(invalid(
568            "a batch has at least one time, one baseline and one correlation",
569        ));
570    }
571    Ok(entry)
572}

574fn check(
575    arrays: &BatchArrays,
576    header: &BatchHeader,
577    catalogue: &Catalogue,
578    lengths: Lengths,
579) -> Result<(), ObservationError> {
580    for (array, actual, expected) in [
581        ("time", arrays.time_offsets_s.len(), lengths.times),
582        ("field", arrays.field.len(), lengths.times),
583        ("scan", arrays.scan.len(), lengths.scan),
584        ("values", arrays.values.len(), lengths.values),
585        ("flags", arrays.flags.len(), lengths.flags),
586        ("weights", arrays.weights.len(), lengths.weights),
587        ("uvw", arrays.uvw.len(), lengths.uvw),
588        (
589            "time centroid",
590            arrays.centroid_offsets_s.len(),
591            lengths.centroid,
592        ),
593        ("interval", arrays.interval_s.len(), lengths.interval),
594        ("exposure", arrays.exposure_s.len(), lengths.exposure),
595    ] {
596        length(array, actual, expected)?;
597    }
598    identities(
599        "field",
600        arrays.field.iter().map(|field| field.0),
601        catalogue.fields.len(),
602    )?;
603    identities(
604        "scan",
605        arrays.scan.iter().map(|scan| scan.0),
606        catalogue.scans.len(),
607    )?;
608    let per_time = catalogue.partition(header.partition).baselines.len();
609    fragments(arrays, header, per_time, lengths.times * per_time)
610}
611
612/// The fragments tile `0..samples` in order, each non-empty, split only
613/// between times, all of this partition.
614fn fragments(
615    arrays: &BatchArrays,
616    header: &BatchHeader,
617    per_time: usize,
618    samples: usize,
619) -> Result<(), ObservationError> {
620    if arrays.fragments.is_empty() {
621        return Ok(());
622    }
623    let mut covered = 0;
624    for fragment in &arrays.fragments {
625        let range = &fragment.samples;
626        if range.start != covered
627            || range.end <= range.start
628            || !range.end.is_multiple_of(per_time)
629            || fragment.key.partition != header.partition
630        {
631            return Err(invalid(format!(
632                "fragment {fragment:?} does not continue the batch at {covered}"
633            )));
634        }
635        covered = range.end;
636    }
637    if covered == samples {
638        Ok(())
639    } else {
640        Err(invalid(format!(
641            "fragments cover {covered} of {samples} samples"
642        )))
643    }
644}

646impl BatchBuffers {
647    /// Copy `batch` (typically a compute window of a larger batch) into
648    /// these buffers, replacing what they held, through the same checked
649    /// fill a cursor uses, and return the payload elements copied: the
650    /// explicit, accounted packing that gives a window storage of its own.
651    /// Strided views are gathered into standard order; a weight shared by
652    /// every channel stays one per correlation; shared quantities keep
653    /// their sharing; the piece refers to the batch's catalogue.
654    ///
655    /// # Errors
656    ///
657    /// When the batch's data group is not in the catalogue's partition, or
658    /// the piece needs more than the buffers' envelope.
659    pub fn pack(&mut self, batch: &VisibilityBatch<'_>) -> Result<usize, ObservationError> {
660        let catalogue = batch.catalogue;
661        let group = &batch.group;
662        let data_group = catalogue
663            .partition(batch.partition)
664            .data_group(&group.entry.name)
665            .map(|(index, _)| index)
666            .ok_or_else(|| invalid(format!("no data group {}", group.entry.name)))?;
667        let sharing = |shared: Option<Shared<'_, f64>>| match shared {
668            Some(Shared::PerTime(_)) => Sharing::PerTime,
669            Some(Shared::Constant(value)) => Sharing::Constant(value),
670            Some(Shared::PerSample(_)) | None => Sharing::PerSample,
671        };
672        let weights_per_channel = group
673            .weights
674            .is_none_or(|weights| weights.strides()[1] != 0);
675        let header = BatchHeader {
676            partition: batch.partition,
677            data_group,
678            times: batch.times(),
679            first_channel: batch.first_channel,
680            channels: batch.channel_count(),
681            correlations: batch.axis.clone(),
682            roles: Roles {
683                values: group.values.is_some(),
684                flags: group.flags.is_some(),
685                weights: group.weights.is_some(),
686                uvw: group.uvw.is_some(),
687                time_centroid: batch.timing.centroid.is_some(),
688                interval: batch.timing.interval_s.is_some(),
689                exposure: batch.timing.exposure_s.is_some(),
690                scan: batch.scan.is_some(),
691            },
692            weights_per_channel,
693            weight_rule: group.weight_rule,
694            interval: sharing(batch.timing.interval_s),
695            exposure: sharing(batch.timing.exposure_s),
696        };
697        let mut fill = self.begin(header, catalogue)?;
698        let arrays = fill.arrays();
699        arrays.fragments.extend_from_slice(batch.fragments);
700        arrays.time_offsets_s.extend(batch.time.offsets_s().iter());
701        arrays.field.extend(batch.field.iter());
702        if let Some(scan) = batch.scan {
703            arrays.scan.extend(scan.iter());
704        }
705        if let Some(centroid) = batch.timing.centroid {
706            arrays
707                .centroid_offsets_s
708                .extend(centroid.offsets_s().iter());
709        }
710        for (shared, target) in [
711            (batch.timing.interval_s, &mut arrays.interval_s),
712            (batch.timing.exposure_s, &mut arrays.exposure_s),
713        ] {
714            if let Some(Shared::PerSample(values) | Shared::PerTime(values)) = shared {
715                target.extend(values.iter());
716            }
717        }
718        let mut copied = 0;
719        if let Some(values) = group.values {
720            arrays.values.extend(values.iter());
721            copied += values.len();
722        }
723        if let Some(flags) = group.flags {
724            arrays.flags.extend(flags.iter());
725            copied += flags.len();
726        }
727        if let Some(weights) = group.weights {
728            if weights_per_channel {
729                arrays.weights.extend(weights.iter());
730                copied += weights.len();
731            } else {
732                let per_correlation = weights.index_axis(Axis(1), 0);
733                arrays.weights.extend(per_correlation.iter());
734                copied += per_correlation.len();
735            }
736        }
737        if let Some(uvw) = group.uvw {
738            arrays.uvw.extend(uvw.metres().iter());
739            copied += uvw.metres().len();
740        }
741        fill.finish()?;
742        Ok(copied)
743    }
744}

894    fn a_checked_fill_builds_the_batch_its_header_describes() {
895        let mut buffers = BatchBuffers::default();
896        fill(&mut buffers, |_, _| {}).expect("fill");
897        let batch = buffers.batch().expect("batch");
898        assert_eq!(batch.samples(), 4);
899        assert_eq!(batch.shape(), [4, 2, 1]);
900        assert_eq!(batch.interval_s(3), Some(10.0));
901        assert_eq!(batch.scan(3), Some(ScanId(0)));
902        assert_eq!(batch.timing.centroid.expect("centroid").len(), 4);
903        assert_eq!(batch.fragments.len(), 1);
904        assert_eq!(batch.frequency.hz()[1], 1.1e9, "from the catalogue");
905        assert_eq!(batch.baselines.len(), 2, "from the catalogue");
906        let weights = batch.group.weights.expect("weights");
907        assert_eq!(
908            weights.strides(),
909            [1, 0, 1],
910            "one weight per correlation, every channel"
911        );
912        assert_eq!(weights[[2, 1, 0]], 3.0);
913        assert!(weights.as_slice().is_none());
914        let window = batch.channel_window(1..2).expect("window");
915        assert!(window.fragments.is_empty(), "a window never closes a group");
916        assert_eq!(window.scan(3), Some(ScanId(0)));
917        assert_eq!(window.frequency.hz()[0], 1.1e9);
918        let times = batch.sample_window(2..4).expect("one time");
919        assert!(times.fragments.is_empty());
920        assert_eq!(
921            times.timing.centroid.expect("centroid").offsets_s()[0],
922            10.0
923        );
924        assert!(batch.sample_window(1..3).is_err(), "splits a time");
925    }

928    fn a_fill_that_disagrees_with_its_header_or_catalogue_is_refused() {
929        let mut buffers = BatchBuffers::default();
930        type Tweak = fn(&mut BatchHeader, &mut BatchArrays);
931        let cases: [(&str, Tweak); 9] = [
932            ("short flags", |_, a| {
933                a.flags.pop();
934            }),
935            ("scan out of the catalogue", |_, a| a.scan[1] = ScanId(1)),
936            ("field out of the catalogue", |_, a| a.field[0] = FieldId(1)),
937            ("a gap before the fragment", |_, a| {
938                a.fragments[0].samples = 2..4
939            }),
940            ("a fragment inside a time", |_, a| {
941                a.fragments[0].samples = 0..3
942            }),
943            ("a short exposure", |_, a| {
944                a.exposure_s.pop();
945            }),
946            ("a channel window past the partition", |h, _| {
947                h.first_channel = 1
948            }),
949            ("a correlation the partition lacks", |h, _| {
950                h.correlations = CorrelationAxis::Subset {
951                    products: [Correlation::RR].into(),
952                    indices: [0].into(),
953                }
954            }),
955            ("products out of the partition's order", |h, _| {
956                h.correlations = CorrelationAxis::Subset {
957                    products: [Correlation::YY, Correlation::XX].into(),
958                    indices: [1, 0].into(),
959                }
960            }),
961        ];
962        for (label, tweak) in cases {
963            assert!(fill(&mut buffers, tweak).is_err(), "{label}");
964            assert!(
965                buffers.batch().is_none(),
966                "{label}: no batch after a refused fill"
967            );
968        }
