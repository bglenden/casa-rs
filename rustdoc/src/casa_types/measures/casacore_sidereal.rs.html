2//! casacore's sidereal epochs (`MCEpoch`): UT1 to and from GMST1 and GAST,
3//! and GMST1/GAST to and from local sidereal time, in casacore's sidereal day
4//! count.
5//!
6//! casacore counts a sidereal epoch from UT1 plus the IAU 1982 GMST at 0h, in
7//! days, plus 6713 days: UT1 MJD 51544.5 is GMST1 58257.779. The fraction is
8//! the sidereal time of day; the whole days count sidereal days (#850).
9//!
10//! Both IAU models use casacore's IAU 1976/1980 formulas here.
11//! `MCEpoch::UT1_GAST` takes `Nutation::STANDARD` under either model, which
12//! the apparent sidereal time of every HADEC, AZEL and ITRF conversion
13//! inherits. casacore's IAU 2000 `UT1_GMST1` adds the TT day fraction to its
14//! GMST, so its GMST1 and LMST are up to half a day off and its inverse does
15//! not invert; casa-rs does not follow it (`docs/CASA (C++) bugs.md`).
16
17use super::casacore_nutation::standard_equation_of_equinoxes_rad;
18use super::epoch::MjdHighPrec;
19
20const MJD2000: f64 = 51_544.5;
21const JULIAN_CENTURY_DAYS: f64 = 36_525.0;
22const SECONDS_PER_DAY: f64 = 86_400.0;
23
24/// The sidereal day count's offset in days (`MCEpoch`'s `mve6713`).
25const SIDEREAL_DAY_OFFSET: f64 = 6_713.0;
26
27/// `MeasTable::GMST0`: the IAU 1982 GMST at 0h UT1, in seconds, as a
28/// polynomial evaluated at the UT1 epoch itself.
29fn gmst0_seconds(ut1_mjd: f64) -> f64 {
30    let t = (ut1_mjd - MJD2000) / JULIAN_CENTURY_DAYS;
31    24_110.548_41 + t * (8_640_184.812_866 + t * (0.093_104 + t * -6.2e-6))
32}
33
34/// `MeasTable::GMUT0`: the inverse polynomial, UT1 − GMST1 in units of
35/// `SECinDAY / JDCEN` days, evaluated at the GMST1 epoch.
36fn gmut0(gmst1_mjd: f64) -> f64 {
37    let t = (gmst1_mjd - MJD2000 - SIDEREAL_DAY_OFFSET) / JULIAN_CENTURY_DAYS;
38    -0.658_308_450_562_548_7
39        + t * (-235.909_469_167_107_52 + t * (-0.000_002_528_225_535_979_72 + t * 1.679e-10))
40}
41
42/// Add `days` as casacore's `MVEpoch` does, its whole and fractional parts
43/// separately, so a 6713-day offset costs no precision.
44fn add_days(epoch: MjdHighPrec, days: f64) -> MjdHighPrec {
45    let whole = days.floor();
46    MjdHighPrec::new(epoch.day() + whole, epoch.frac() + (days - whole))
47}
48
49/// UT1 to GMST1 (`MCEpoch::UT1_GMST1`, IAU 1976/1980).
50pub(super) fn ut1_to_gmst1(ut1: MjdHighPrec) -> MjdHighPrec {
51    let gmst = add_days(ut1, gmst0_seconds(ut1.as_mjd()) / SECONDS_PER_DAY);
52    add_days(gmst, SIDEREAL_DAY_OFFSET)
53}
54
55/// GMST1 to UT1 (`MCEpoch::GMST1_UT1`, IAU 1976/1980): casacore's inverse
56/// polynomial, which leaves up to about 1 µs of the round trip.
57pub(super) fn gmst1_to_ut1(gmst1: MjdHighPrec) -> MjdHighPrec {
58    let ut1 = add_days(
59        gmst1,
60        gmut0(gmst1.as_mjd()) * JULIAN_CENTURY_DAYS / SECONDS_PER_DAY,
61    );
62    add_days(ut1, -SIDEREAL_DAY_OFFSET)
63}
64
65/// UT1 to GAST (`MCEpoch::UT1_GAST`): GMST1 plus casacore's IAU 1980
66/// equation of the equinoxes at the UT1 epoch.
67pub(super) fn ut1_to_gast(ut1: MjdHighPrec) -> MjdHighPrec {
68    let equation_of_equinoxes = standard_equation_of_equinoxes_rad(ut1.as_mjd());
69    add_days(
70        ut1_to_gmst1(ut1),
71        equation_of_equinoxes / std::f64::consts::TAU,
72    )
73}
74
75/// GAST to UT1 (`MCEpoch::GAST_UT1`): remove the equation of the equinoxes
76/// at a first UT1 guess, then invert GMST1.
77pub(super) fn gast_to_ut1(gast: MjdHighPrec) -> MjdHighPrec {
78    let gast_mjd = gast.as_mjd();
79    let ut1_guess =
80        gast_mjd + gmut0(gast_mjd) * JULIAN_CENTURY_DAYS / SECONDS_PER_DAY - SIDEREAL_DAY_OFFSET;
81    let equation_of_equinoxes = standard_equation_of_equinoxes_rad(ut1_guess);
82    gmst1_to_ut1(add_days(
83        gast,
84        -equation_of_equinoxes / std::f64::consts::TAU,
85    ))
86}
87
88/// Greenwich to local sidereal time (`MCEpoch::GMST1_LMST`,
89/// `MCEpoch::GAST_LAST`), or back for a negated longitude.
90pub(super) fn add_longitude(sidereal: MjdHighPrec, longitude_rad: f64) -> MjdHighPrec {
91    add_days(sidereal, longitude_rad / std::f64::consts::TAU)
92}
93
94/// The Greenwich apparent sidereal time of a UT1 epoch as an angle in
95/// [0, 2π), as `MCFrame::getLASTr` takes it before adding the longitude.
96pub(super) fn gast_rad(ut1: MjdHighPrec) -> f64 {
97    ut1_to_gast(ut1).frac() * std::f64::consts::TAU
98}

100#[cfg(test)]
101mod tests {
102    use super::*;
103
104    #[test]
105    fn j2000_is_casacore_s_sidereal_day_58257() {
106        let gmst1 = ut1_to_gmst1(MjdHighPrec::from_mjd(MJD2000));
107        assert_eq!(gmst1.day(), 58_257.0);
108        // 18h 41m 50.54841s, the IAU 1982 GMST at J2000.0.
109        let expected = 0.5 + 24_110.548_41 / SECONDS_PER_DAY;
110        assert!((gmst1.frac() - expected).abs() < 1e-15, "{}", gmst1.frac());
111    }
112
113    #[test]
114    fn inverses_round_trip_to_casacore_s_microsecond() {
115        for mjd in [51_544.5, 58_849.906_25, 61_181.25] {
116            let ut1 = MjdHighPrec::from_mjd(mjd);
117            for back in [
118                gmst1_to_ut1(ut1_to_gmst1(ut1)),
119                gast_to_ut1(ut1_to_gast(ut1)),
120            ] {
121                let error_s = (back - ut1).abs() * SECONDS_PER_DAY;
122                assert!(error_s < 2e-6, "MJD {mjd}: {error_s} s");
123            }
124        }
125    }
126
127    #[test]
128    fn longitude_moves_only_the_sidereal_time() {
129        let gast = ut1_to_gast(MjdHighPrec::from_mjd(MJD2000));
130        let last = add_longitude(gast, -1.878_283_2);
131        let back = add_longitude(last, 1.878_283_2);
132        assert!((back - gast).abs() < 1e-15);
133    }
134}
