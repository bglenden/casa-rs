39/// One owner's accumulator and backend.
40struct Owner<'w> {
41    acc: Option<GridAccumulator>,
42    backend: OwnerBackend<'w>,
43    images: Option<NormalImages>,
44    /// Time the owner spent accumulating in the wave, which a region pass
45    /// reports beside its strips (#736).
46    busy: std::time::Duration,
47    /// The weight image's contributions, summed before they are gridded
48    /// ([`WeightSums`], #787), when this owner grids the weight image: the
49    /// weight owner of a region pass, every owner of a plane pass.
50    weights: Option<WeightSums>,
51}

53impl Owner<'_> {
54    /// Wait for the owner's grid work still on the device, reporting
55    /// whether it failed.
56    fn complete(&mut self) -> Result<(), PassError> {
57        match &mut self.backend {
58            OwnerBackend::Cpu(_) => Ok(()),
59            OwnerBackend::Metal(backend) => Ok(backend.complete()?),
60        }
61    }
62
63    /// Grid owner `index`'s placements in `chunks`, of `target`'s domain,
64    /// in every mode `pass` holds, less what `subtract` names; and sum the
65    /// weight image's contributions over every placement when this owner
66    /// holds it (`weight_owner`; otherwise each owner sums its own
67    /// placements'). The sums are gridded when the table fills and at
68    /// [`Self::flush_weights`].
69    fn accumulate(
70        &mut self,
71        pass: &MajorCyclePass<'_>,
72        target: &PassDomain<'_>,
73        (index, weight_owner): (usize, Option<usize>),
74        subtract: Subtract<'_>,
75        chunks: &[Chunk],
76    ) -> Result<(), PassError> {
77        let started = std::time::Instant::now();
78        let Self {
79            acc,
80            backend,
81            weights,
82            busy,
83            ..
84        } = self;
85        let acc = acc.as_mut().expect("owners accumulate until finished");
86        let cf = target.operator.cf();
87        let own_weights = if weight_owner.is_none() {
88            weights.as_mut()
89        } else {
90            None
91        };
92        match &mut *backend {
93            OwnerBackend::Cpu(cpu) => {
94                for chunk in chunks {
95                    let block = chunk.owned[index].block();
96                    if !block.is_empty() {
97                        let mut apply = |work: Work<'_>| cpu.apply(&block, cf, work);
98                        accumulate(pass, &mut apply, subtract, acc)?;
99                    }
100                }
101            }
102            OwnerBackend::Metal(metal) => {
103                let parts = chunks
104                    .iter()
105                    .map(|chunk| chunk.owned[index].block())
106                    .collect::<Vec<_>>();
107                if parts.iter().any(|part| !part.is_empty()) {
108                    let mut apply = |work: Work<'_>| metal.apply_parts(&parts, cf, work);
109                    accumulate(pass, &mut apply, subtract, acc)?;
110                }
111            }
112        }
113        if let Some(sums) = own_weights {
114            for chunk in chunks {
115                let mut grid = weight_grid(backend, cf, acc);
116                sums.add(&chunk.owned[index].block(), &mut grid)?;
117            }
118        }
119        if weight_owner == Some(index)
120            && let Some(sums) = weights
121        {
122            for chunk in chunks {
123                let mut grid = weight_grid(backend, cf, acc);
124                sums.add(&chunk.weight_owned.block(), &mut grid)?;
125            }
126        }
127        *busy += started.elapsed();
128        Ok(())
129    }
130
131    /// Grid the weight image's summed contributions still in the owner's
132    /// table, of `target`'s domain ([`WeightSums::flush`]), and drop the
133    /// table: the wave's last placement is accumulated.
134    ///
135    /// The flush's time is the owner's ([`Self::busy`]). Its placements,
136    /// group grids and grids are returned for the wave's log. A panic while
137    /// gridding fails the pass with [`PassError::OwnerPanicked`], as an
138    /// owner's panic does in a pipelined pass.
139    fn flush_weights(&mut self, target: &PassDomain<'_>) -> Result<[u64; 3], PassError> {
140        let Some(mut sums) = self.weights.take() else {
141            return Ok([0; 3]);
142        };
143        let started = std::time::Instant::now();
144        let acc = self.acc.as_mut().expect("owners flush before they finish");
145        let backend = &mut self.backend;
146        let flushed = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
147            let mut grid = weight_grid(backend, target.operator.cf(), acc);
148            sums.flush(&mut grid)
149        }))
150        .unwrap_or(Err(PassError::OwnerPanicked));
151        self.busy += started.elapsed();
152        flushed?;
153        Ok(sums.totals())
154    }
155}

157/// Grid a block of summed weight-image contributions on `backend` into
158/// `acc` ([`WeightSums`]).
159fn weight_grid<'a>(
160    backend: &'a mut OwnerBackend<'_>,
161    cf: &'a dyn ConvolutionFunctionSet,
162    acc: &'a mut GridAccumulator,
163) -> impl FnMut(&SampleBlock<'_>) -> Result<(), PassError> + 'a {
164    move |groups: &SampleBlock<'_>| {
165        let work = Work::Grid {
166            mode: Mode::Weight,
167            acc: &mut *acc,
168        };
169        match backend {
170            OwnerBackend::Cpu(cpu) => Ok(cpu.apply(groups, cf, work)?),
171            OwnerBackend::Metal(metal) => Ok(metal.apply(groups, cf, work)?),
172        }
173    }
174}

176/// The backend an owner dispatches to. The CPU grids each placement chunk
177/// as it comes; a Metal owner dispatches its share of every chunk of a slice
178/// at once, read from the chunks' buffers ([`MetalBackend::apply_parts`]),
179/// so each device dispatch is a slice's worth of samples rather than one
180/// chunk's, and leaves it running while the workers place the next slice
181/// ([`MetalBackend::with_deferred_grid_completion`]).
182enum OwnerBackend<'w> {
183    Cpu(CpuBackend),
184    Metal(Box<MetalBackend<'w>>),
185}

198impl Domain<'_> {
199    /// Add the gridding work by anchor row `chunks` placed.
200    fn add_row_work(&mut self, chunks: &[Chunk]) {
201        if let Some(total) = &mut self.row_work {
202            for chunk in chunks {
203                for (total, work) in total.iter_mut().zip(&chunk.row_work) {
204                    *total += work;
205                }
206            }
207        }
208    }

211/// What placing one block's rows on one domain reads ([`Wave::placer`]).
212struct Placer<'b, 'w, 'p> {
213    pass: &'w MajorCyclePass<'p>,
214    target: &'p PassDomain<'p>,
215    router: &'b Router,
216    owners: usize,
217    planes: PlaneRange,
218    native_residuals: bool,
219    shape: ChunkShape,
220    /// The pass's kept weights, for the first domain.
221    weights: Option<(&'w WeightCache, bool)>,
222    /// The slot of the block's first sample.
223    weight_base: usize,
224    /// Rows of the domain's work histogram, when it is region-split.
225    grid_rows: Option<usize>,
226    block: &'b NativeBlock,
227    index: usize,
228    ordered: bool,
229}

231impl Placer<'_, '_, '_> {
232    /// Place the rows `chunks` hold on `team`: each placement into its
233    /// owner's buffer, and every placement into the weight owner's when the
234    /// pass grids the weight image on one owner; each chunk's gridding work
235    /// by anchor row beside them.
236    fn place(&self, team: &WorkerTeam, chunks: &mut [Chunk]) -> Result<(), PassError> {
237        let Self {
238            pass,
239            target,
240            router,
241            owners,
242            planes,
243            native_residuals,
244            shape,
245            weights,
246            weight_base,
247            grid_rows,
248            block,
249            index,
250            ordered,
251        } = *self;
252        let channels = block.channels();
253        // With region owners the weight image has one owner, which must see
254        // every placement; plane owners grid their own samples' weights.
255        let route_weight = pass.modes.weight && router.weight_owner().is_some();
256        team.for_each_mut(chunks, |_, chunk| {
257            chunk.placed = 0;
258            chunk.owned[..owners]
259                .iter_mut()
260                .for_each(SampleBuffer::clear);
261            chunk.weight_owned.clear();
262            chunk.row_work.clear();
263            if let Some(rows) = grid_rows {
264                chunk.row_work.resize(rows, 0);
265            }
266            for position in chunk.rows.clone() {
267                let row_index = row_at(block, ordered, position);
268                let native = block.row(index, row_index);
269                let slots = weights.and_then(|(cache, filled)| {
270                    cache.row(weight_base + row_index * channels, channels, filled)
271                });
272                let mut row_samples = 0;
273                let (owned, weight_owned, placed, row_work) = (
274                    &mut chunk.owned,
275                    &mut chunk.weight_owned,
276                    &mut chunk.placed,
277                    &mut chunk.row_work,
278                );
279                let mut keep = |placement: &Placement, values: &[Complex32], weights: &[f32]| {
280                    row_samples += 1;
281                    if !planes.contains(placement.plane) {
282                        return None;
283                    }
284                    if route_weight {
285                        weight_owned.push(*placement, values, weights);
286                    }
287                    *placed += 1;
288                    let (owner, work) = router.owner_and_work(target.operator, placement);
289                    if let Some((row, work)) = work
290                        && let Some(total) = row_work.get_mut(row)
291                    {
292                        *total += work;
293                    }
294                    Some(owner)
295                };
296                if native_residuals {
297                    target.resampler.place_each_sourced(
298                        target.operator,
299                        pass.weighting,
300                        &native,
301                        cell_of(position, channels),
302                        slots,
303                        |placement, values, weights, source| {
304                            if let Some(owner) = keep(&placement, values, weights) {
305                                owned[owner].push_sourced(placement, values, weights, source);
306                            }
307                        },
308                    )?;
309                } else {
310                    target.resampler.place_each(
311                        target.operator,
312                        pass.weighting,
313                        &native,
314                        slots,
315                        |placement, values, weights| {
316                            if let Some(owner) = keep(&placement, values, weights) {
317                                owned[owner].push(placement, values, weights);
318                            }
319                        },
320                    )?;
321                }
322                debug_assert!(
323                    row_samples <= shape.row,
324                    "a row places within the chunk's per-row bound"
325                );
326            }
327            debug_assert!(
328                chunk.placed as usize <= shape.placements
329                    && chunk.capacity_bytes() == shape.bytes(),
330                "a chunk places within the placements it was allocated for"
331            );
332            Ok::<_, PassError>(())
333        })
334    }
335}

337/// One row chunk's placements on the wave's planes, by owner, and its
338/// prediction batch.
339///
340/// Every buffer is allocated once, for a chunk's rows of the source's widest
341/// row ([`ChunkShape`]), and each owner's can take every placement of the
342/// chunk, so a chunk holds [`ChunkShape::bytes`] however its placements fall
343/// among the owners.
344struct Chunk {
345    rows: Range<usize>,
346    /// Each owner's placements, in row order, with their native endpoints
347    /// when the wave forms residuals at native channels.
348    owned: Vec<SampleBuffer>,
349    /// Every placement of the chunk, in row order, when the weight image has
350    /// one owner ([`Router::weight_owner`]): that owner grids `Mode::Weight`
351    /// over it.
352    weight_owned: SampleBuffer,
353    placed: u64,
354    prediction: PredictionScratch,
355    /// The gridding work of the chunk's placements by anchor row, on a
356    /// region-partitioned domain ([`Router::owner_and_work`]); merged into
357    /// the domain's after each slice.
358    row_work: Vec<u64>,

361/// The largest chunk of a wave over blocks of `block`: at most
362/// [`chunk_placements`] placements, whatever rows the source puts in a
363/// block.
364#[derive(Clone, Copy)]
365struct ChunkShape {
366    /// Placements one native row places on any domain.
367    row: usize,
368    /// Rows a chunk places at once.
369    rows: usize,
370    /// Placements the chunk's rows place.
371    placements: usize,
372    /// Correlations per placement.
373    npol: usize,
374    /// Channels of one row.
375    channels: usize,
376    /// Correlations and channels of one row.
377    cells: usize,
378    /// Owners of the domain with the most.
379    owners: usize,
380    /// Whether the pass grids the weight image.
381    weight: bool,
382    /// Whether placements carry native endpoints (a residual at native
383    /// channels).
384    sourced: bool,
385    /// Samples the prediction batch of the chunk's rows degrids at most.
386    predicted: usize,
387    /// Bytes the prediction batch of the chunk's rows holds
388    /// ([`SpectralResampler::prediction_scratch`]) with each row's span in
389    /// it.
390    prediction: u64,
391    /// Grid rows of the region-partitioned domain with the most: the
392    /// length of the chunk's work histogram ([`Chunk::row_work`]).
393    grid_rows: usize,
394}
395
396impl ChunkShape {
397    /// The chunk of `chunks` chunks over blocks of `block`, each taking at
398    /// most `placements` placements before the owner scaling of
399    /// [`chunk_placements`].
400    fn new(
401        domains: &[PassDomain<'_>],
402        block: BlockShape,
403        weight: bool,
404        sourced: bool,
405        chunks: usize,
406        placements: usize,
407    ) -> Self {
408        let row = domains
409            .iter()
410            .map(|domain| domain.resampler.samples_per_row(block.spectrum))
411            .max()
412            .unwrap_or(0);
413        let owners = domains
414            .iter()
415            .map(|domain| domain.partition.owners())
416            .max()
417            .unwrap_or(1);
418        let rows = chunk_rows(
419            row,
420            block.rows,
421            chunks,
422            chunk_placements(placements, owners),
423        );
424        Self {
425            row,
426            rows,
427            placements: rows * row,
428            npol: block.correlations,
429            channels: block.channels(),
430            cells: block.channels() * block.correlations,
431            owners,
432            weight,
433            sourced,
434            predicted: rows
435                * domains
436                    .iter()
437                    .map(|domain| {
438                        domain
439                            .resampler
440                            .prediction_samples_per_row(block.channels())
441                    })
442                    .max()
443                    .unwrap_or(0),
444            prediction: SpectralResampler::prediction_scratch_bytes(
445                domains.iter().map(|domain| domain.resampler),
446                block.correlations,
447                block.channels(),
448                rows,
449            ),
450            grid_rows: domains
451                .iter()
452                .filter(|domain| domain.partition.splits_rows())
453                .map(|domain| domain.operator.geometry().grid_shape()[1])
454                .max()
455                .unwrap_or(0),
456        }
457    }
458
459    /// An empty buffer for `placements` placements of this shape, with room
460    /// for their native endpoints when they carry them.
461    fn buffer(self, placements: usize) -> SampleBuffer {
462        if self.sourced {
463            SampleBuffer::with_source_capacity(self.npol, placements)
464        } else {
465            SampleBuffer::with_capacity(self.npol, placements)
466        }
467    }
468
469    /// Bytes [`Self::buffer`] holds.
470    const fn buffer_bytes(self, placements: usize) -> u64 {
471        if self.sourced {
472            SampleBuffer::sourced_bytes(self.npol, placements)
473        } else {
474            SampleBuffer::bytes(self.npol, placements)
475        }
476    }
477
478    /// Placements the weight-owner buffer takes.
479    const fn weight_placements(self) -> usize {
480        if self.weight { self.placements } else { 0 }
481    }
482
483    /// Bytes a chunk of this shape holds: each owner's and the weight
484    /// owner's placements, its prediction batch and its work histogram.
485    const fn bytes(self) -> u64 {
486        self.owners as u64 * self.buffer_bytes(self.placements)
487            + SampleBuffer::bytes(self.npol, self.weight_placements())
488            + self.prediction
489            + (self.grid_rows * size_of::<u64>()) as u64
490    }
491}

493/// The backend that predicts one domain's batch for one chunk and forms the
494/// model at its rows' native cells ([`Work::PredictNative`]): the pass's,
495/// so that on Metal the model never leaves the device and the device works
496/// on one chunk's batch while the host places the next chunk's.
497enum Predictor<'w> {
498    Cpu(CpuBackend),
499    Metal(Box<MetalBackend<'w>>),
500}
501
502/// One predictor per domain on `backend`, for one chunk.
503fn predictors<'w>(
504    domains: &[PassDomain<'w>],
505    backend: BackendChoice,
506    stage_timing: bool,
507) -> Result<Vec<Predictor<'w>>, PassError> {
508    domains
509        .iter()
510        .map(|domain| {
511            Ok(match backend {
512                BackendChoice::Cpu => Predictor::Cpu(CpuBackend::new()),
513                BackendChoice::Metal => Predictor::Metal(Box::new(MetalBackend::new(
514                    domain.operator.cf(),
515                    stage_timing,
516                )?)),
517            })
518        })
519        .collect()
520}

522impl Chunk {
523    fn new(shape: ChunkShape, domains: &[PassDomain<'_>]) -> Self {
524        Self {
525            rows: 0..0,
526            owned: (0..shape.owners)
527                .map(|_| shape.buffer(shape.placements))
528                .collect(),
529            weight_owned: SampleBuffer::with_capacity(shape.npol, shape.weight_placements()),
530            placed: 0,
531            prediction: SpectralResampler::prediction_scratch(
532                domains.iter().map(|domain| domain.resampler),
533                shape.npol,
534                shape.channels,
535                shape.rows,
536            ),
537            row_work: Vec::with_capacity(shape.grid_rows),
538        }
539    }
540
541    /// Bytes the chunk holds now, from its buffers' capacities: at most
542    /// [`ChunkShape::bytes`] while every buffer is within the shape.
543    fn capacity_bytes(&self) -> u64 {
544        self.owned
545            .iter()
546            .map(SampleBuffer::capacity_bytes)
547            .sum::<u64>()
548            + self.weight_owned.capacity_bytes()
549            + self.prediction.capacity_bytes()
550            + (self.row_work.capacity() * size_of::<u64>()) as u64
551    }
552}

559/// Bytes a wave places with on `workers` workers over blocks of `block`,
560/// gridding the weight image when `weight`: its row chunks and, with a
561/// model, its backends' batches. On Metal each chunk's predictors keep a
562/// ring and a prediction batch ([`casa_imaging_metal::ring_bytes`],
563/// [`casa_imaging_metal::batch_bytes`]); on the CPU each chunk's predictors
564/// keep their batch's predictions, and each owner of a pass forming
565/// residuals at native channels the residual values of a chunk's
566/// placements ([`CpuBackend::batch_bytes`]). They grow with the source's
567/// block only up to `placements` a chunk.
568pub(super) fn chunk_bytes(
569    domains: &[PassDomain<'_>],
570    backend: BackendChoice,
571    block: BlockShape,
572    workers: usize,
573    weight: bool,
574    with_model: bool,
575    placements: usize,
576) -> u64 {
577    let count = chunks(workers);
578    let sourced = super::native_residuals(domains, with_model);
579    let shape = ChunkShape::new(domains, block, weight, sourced, count, placements);
580    let device = match backend {
581        BackendChoice::Cpu if with_model => {
582            let owners = if sourced {
583                domains
584                    .iter()
585                    .map(|domain| domain.partition.owners())
586                    .sum::<usize>()
587            } else {
588                0
589            };
590            (count * domains.len()) as u64 * CpuBackend::batch_bytes(shape.npol, shape.predicted)
591                + owners as u64 * CpuBackend::batch_bytes(shape.npol, shape.placements)
592        }
593        BackendChoice::Cpu => 0,
594        BackendChoice::Metal if with_model => {
595            (count * domains.len()) as u64
596                * (casa_imaging_metal::ring_bytes(shape.npol)
597                    + casa_imaging_metal::batch_bytes(shape.npol, shape.predicted))
598        }
599        BackendChoice::Metal => 0,
600    };
601    count as u64 * shape.bytes() + device
602}

605/// wave with a model forms for every row of the block: in host memory, or
606/// on Metal in the device's.
607pub(super) const fn prediction_bytes(block: BlockShape) -> u64 {
608    (block.rows * block.channels() * block.correlations * size_of::<Complex32>()) as u64
609}

642/// Wall time of a wave's stages ([`Wave::stages`]), as the placing thread
643/// sees them. With one set of chunks, `placing` is the placement and
644/// `accumulating` the owners' accumulation, slice by slice. When the owners
645/// are pipelined (`pipelined`), they accumulate while the placer places
646/// and reads. Then `placing` also counts waits for a free team thread,
647/// including drains a waiting thread steals. `accumulating` is only the
648/// placer's wait for a free set and for the last sets. The owners' own
649/// accumulation times are in the region owners' log line.
650#[derive(Clone, Copy, Default)]
651struct WaveStages {
652    predicting: std::time::Duration,
653    placing: std::time::Duration,
654    accumulating: std::time::Duration,
655    pipelined: bool,
656}

658impl<'w, 'p> Wave<'w, 'p> {
659    /// A wave over `planes` with each domain's prepared `models`, forming
660    /// residuals at native channels when `native_residuals`, on `workers`
661    /// workers over blocks of at most `block`. It places with
662    /// [`chunk_bytes`] and predicts into [`prediction_bytes`].
663    pub(super) fn new(
664        pass: &'w MajorCyclePass<'p>,
665        planes: PlaneRange,
666        models: Option<&'w [PreparedModelGrids]>,
667        native_residuals: bool,
668        block: BlockShape,
669        workers: usize,
670    ) -> Result<Self, PassError> {
671        let count = chunks(workers);
672        let shape = ChunkShape::new(
673            pass.domains,
674            block,
675            pass.modes.weight,
676            native_residuals,
677            count,
678            pass.chunk_placements,
679        );
680        let domains = pass
681            .domains
682            .iter()
683            .enumerate()
684            .map(|(index, domain)| {
685                let router = Router::new(&domain.partition, planes);
686                let operator = domain.operator;
687                let weight_owner = router.weight_owner();
688                let owners = (0..router.owners())
689                    .map(|owner| {
690                        let (range, tile) = router.target(&domain.partition, planes, owner);
691                        let weights = (pass.modes.weight
692                            && weight_owner.is_none_or(|holder| holder == owner))
693                        .then(|| WeightSums::new(block.correlations, WEIGHT_GROUPS));
694                        Ok(match pass.backend {
695                            BackendChoice::Cpu => Owner {
696                                acc: Some(
697                                    pass.summing(operator.accumulator(range, tile, pass.modes)),
698                                ),
699                                backend: OwnerBackend::Cpu(CpuBackend::new()),
700                                images: None,
701                                busy: std::time::Duration::ZERO,
702                                weights,
703                            },
704                            BackendChoice::Metal => {
705                                assert_eq!(
706                                    operator.precision(),
707                                    GridPrecision::F32,
708                                    "Metal grids are f32 (D2)"
709                                );
710                                Owner {
711                                    acc: Some(pass.summing(MetalBackend::accumulator(
712                                        operator.accumulator_layout(range, tile, pass.modes),
713                                    )?)),
714                                    backend: OwnerBackend::Metal(Box::new(
715                                        MetalBackend::new(operator.cf(), pass.metal_stage_timing)?
716                                            .with_deferred_grid_completion(),
717                                    )),
718                                    images: None,
719                                    busy: std::time::Duration::ZERO,
720                                    weights,
721                                }
722                            }
723                        })
724                    })
725                    .collect::<Result<_, PassError>>()?;
726                let row_work = domain
727                    .partition
728                    .splits_rows()
729                    .then(|| vec![0; operator.geometry().grid_shape()[1]]);
730                Ok(Domain {
731                    router,
732                    owners,
733                    model: models.map(|models| &models[index]),
734                    row_work,
735                })
736            })
737            .collect::<Result<_, PassError>>()?;
738        // A pass with a model predicts each block, for its native residuals or
739        // for a visibility writer.
740        let predictions = if pass.model.is_some() {
741            block.rows * shape.cells
742        } else {
743            0
744        };
745        let predictions = match pass.backend {
746            BackendChoice::Metal if predictions > 0 => {
747                NativeStore::Device(MetalBackend::native_cells(predictions)?)
748            }
749            BackendChoice::Cpu | BackendChoice::Metal => {
750                NativeStore::Host(Vec::with_capacity(predictions))
751            }
752        };
753        Ok(Self {
754            pass,
755            planes,
756            domains,
757            native_residuals,
758            check_spacing: false,
759            shape,
760            chunks: (0..count)
761                .map(|_| Chunk::new(shape, pass.domains))
762                .collect(),
763            predictors: (0..count)
764                .map(|_| predictors(pass.domains, pass.backend, pass.metal_stage_timing))
765                .collect::<Result<_, _>>()?,
766            predictions,
767            samples: 0,
768            weights: pass.weight_cache().map(|cache| (cache, cache.filled())),
769            weight_base: 0,
770            stages: WaveStages::default(),
771        })
772    }

786    /// Place every row of `block` on every domain and accumulate the
787    /// placements; with a visibility sink, first hand it the block, with
788    /// every selected sample's prediction when it asks for them.
789    ///
790    /// When the wave forms residuals at native channels, the predictors
791    /// first form `Σ_d A_d·m_d` at every row's native cells, and each placed
792    /// sample's observed value is gridded less that model at its native
793    /// endpoints ([`Work::NativeResidualGrid`]); otherwise the domain's model
794    /// is subtracted at each placed sample.
795    pub(super) fn consume(
796        &mut self,
797        block: &NativeBlock,
798        team: &WorkerTeam,
799        visibilities: Option<&mut VisibilitySink<'_>>,
800    ) -> Result<(), PassError> {
801        let rows = block.len();
802        if self.check_spacing {
803            let bound_hz = self.pass.native_spacing_hz;
804            for row in 0..rows {
805                let observed_hz = block
806                    .row(0, row)
807                    .frequencies_hz
808                    .windows(2)
809                    .fold(0.0_f64, |widest, pair| {
810                        widest.max((pair[1] - pair[0]).abs())
811                    });
812                if observed_hz > bound_hz {
813                    return Err(PassError::NativeSpacing {
814                        observed_hz,
815                        bound_hz,
816                    });
817                }
818            }
819        }
820        let sink_predictions = visibilities.as_ref().is_some_and(|sink| sink.predictions);
821        // Chunks take rows by position in the block's locality order, and
822        // predictions are stored by position, unless the sink reads them in
823        // the block's row order.
824        let ordered = !sink_predictions;
825        let cells = block.channels() * block.correlations();
826        if self.native_residuals || sink_predictions {
827            if let NativeStore::Host(predictions) = &mut self.predictions {
828                // The first domain with a model writes every cell of the
829                // rows it predicts, so stale cells never reach a sample;
830                // without a model the cells stay zero.
831                if predictions.len() < rows * cells
832                    || self.domains.iter().all(|domain| domain.model.is_none())
833                {
834                    predictions.clear();
835                    predictions.resize(rows * cells, Complex32::default());
836                }
837            }
838            // The same slices as placement, so no chunk predicts more rows
839            // than its batch was allocated for.
840            let chunks = self.chunks.len();
841            for slice in slices(rows, chunks, self.shape.rows) {
842                let mut count = 0;
843                for (chunk, rows) in self
844                    .chunks
845                    .iter_mut()
846                    .zip(chunk_ranges(slice.clone(), chunks))
847                {
848                    chunk.rows = rows;
849                    count += 1;
850                }
851                let started = std::time::Instant::now();
852                self.predict(block, team, count, slice, ordered)?;
853                self.stages.predicting += started.elapsed();
854            }
855        }
856        if let Some(sink) = visibilities {
857            let predictions: &[Complex32] = if sink.predictions {
858                self.host_predictions(rows * cells)?
859            } else {
860                &[]
861            };
862            (sink.write)(block, predictions).map_err(PassError::VisibilityWrite)?;
863        }
864        // Slices of the block, each owner's in the block's order as a whole
865        // block's would be, so the accumulation is the same.
866        let chunks = self.chunks.len();
867        for slice in slices(rows, chunks, self.shape.rows) {
868            let mut count = 0;
869            for (chunk, rows) in self.chunks.iter_mut().zip(chunk_ranges(slice, chunks)) {
870                chunk.rows = rows;
871                count += 1;
872            }
873            for index in 0..self.domains.len() {
874                let placed = self.place(block, team, count, index, ordered)?;
875                if index == 0 {
876                    self.samples += placed;
877                }
878            }
879        }
880        self.weight_base += rows * block.channels();
881        Ok(())
882    }

884    /// What placing `block`'s rows on domain `index` reads: the wave's
885    /// state at this block. Chunks hold positions in the block's locality
886    /// order when `ordered`, else row indices ([`row_at`]).
887    fn placer<'b>(
888        &'b self,
889        block: &'b NativeBlock,
890        index: usize,
891        ordered: bool,
892    ) -> Placer<'b, 'w, 'p> {
893        let domain = &self.domains[index];
894        Placer {
895            pass: self.pass,
896            target: &self.pass.domains[index],
897            router: &domain.router,
898            owners: domain.router.owners(),
899            planes: self.planes,
900            native_residuals: self.native_residuals,
901            shape: self.shape,
902            weights: self.weights.filter(|_| index == 0),
903            weight_base: self.weight_base,
904            grid_rows: domain.row_work.as_ref().map(Vec::len),
905            block,
906            index,
907            ordered,
908        }

911    /// Place the rows of `block` on domain `index` and accumulate them;
912    /// returns the samples placed. Chunks hold positions in the block's
913    /// locality order when `ordered`, else row indices ([`row_at`]).
914    fn place(
915        &mut self,
916        block: &NativeBlock,
917        team: &WorkerTeam,
918        count: usize,
919        index: usize,
920        ordered: bool,
921    ) -> Result<u64, PassError> {
922        let placing = std::time::Instant::now();
923        let mut chunks = std::mem::take(&mut self.chunks);
924        let placed = self
925            .placer(block, index, ordered)
926            .place(team, &mut chunks[..count]);
927        self.chunks = chunks;
928        placed?;
929        let pass = self.pass;
930        let target = &pass.domains[index];
931        let domain = &mut self.domains[index];
932        let chunks = &self.chunks[..count];
933        domain.add_row_work(chunks);
934        self.stages.placing += placing.elapsed();
935        let accumulating = std::time::Instant::now();
936        let model = domain.model.filter(|_| !self.native_residuals);
937        let native = self.native_residuals.then(|| match &self.predictions {
938            NativeStore::Host(cells) => NativeModel::Host { cells, first: 0 },
939            NativeStore::Device(cells) => NativeModel::Device(cells),
940        });
941        let weight_owner = domain.router.weight_owner();
942        team.for_each_mut(&mut domain.owners, |owner_index, owner| {
943            let subtract = Subtract { model, native };
944            owner.accumulate(pass, target, (owner_index, weight_owner), subtract, chunks)
945        })?;
946        self.stages.accumulating += accumulating.elapsed();
947        Ok(chunks.iter().map(|chunk| chunk.placed).sum())
948    }

950    /// The model at native cells of every selected sample of the rows
951    /// `slice` of `block`, which the first `count` chunks hold, summed over
952    /// every domain's model (`SIMapperCollection::degrid`), `[row][channel]
953    /// [correlation]`, into their rows of the wave's model store, which stay
954    /// zero without a model. Rows are by position in the block's locality
955    /// order when `ordered` ([`row_at`]).
956    ///
957    /// Domain by domain, every chunk places its rows' samples in one batch
958    /// with its rows' native terms and hands it to its predictor
959    /// ([`Work::PredictNative`]): the first domain writes the cells and
960    /// later ones add to them. A Metal predictor leaves the work on the
961    /// device, which runs it before any later device work that reads the
962    /// cells, while the host places the next chunk's batch.
963    fn predict(
964        &mut self,
965        block: &NativeBlock,
966        team: &WorkerTeam,
967        count: usize,
968        slice: Range<usize>,
969        ordered: bool,
970    ) -> Result<(), PassError> {
971        if self.domains.iter().all(|domain| domain.model.is_none()) {
972            return Ok(());
973        }
974        let channels = block.channels();
975        let cells = channels * block.correlations();
976        let pass = self.pass;
977        let shape = self.shape;
978        let models = self
979            .domains
980            .iter()
981            .map(|domain| domain.model)
982            .collect::<Vec<_>>();
983        let mut pieces = Vec::with_capacity(count);
984        let chunks = self.chunks[..count].iter_mut().zip(&mut self.predictors);
985        match &mut self.predictions {
986            NativeStore::Host(predictions) => {
987                let mut remaining = &mut predictions[slice.start * cells..slice.end * cells];
988                for (chunk, predictors) in chunks {
989                    let (head, tail) = remaining.split_at_mut(chunk.rows.len() * cells);
990                    remaining = tail;
991                    pieces.push((chunk, predictors, ChunkCells::Host(head)));
992                }
993            }
994            NativeStore::Device(cells) => {
995                for (chunk, predictors) in chunks {
996                    pieces.push((chunk, predictors, ChunkCells::Device(cells)));
997                }
998            }
999        }
1000        let mut add = false;
1001        for (index, (target, model)) in pass.domains.iter().zip(&models).enumerate() {
1002            let Some(model) = *model else {
1003                continue;
1004            };
1005            let npol = target.operator.polarization().correlations().len();
1006            team.for_each_mut(&mut pieces, |_, (chunk, predictors, cells)| {
1007                chunk.prediction.begin(npol);
1008                for position in chunk.rows.clone() {
1009                    let row = block.row(index, row_at(block, ordered, position));
1010                    let placed = target.resampler.place_prediction(
1011                        target.operator,
1012                        model,
1013                        &row,
1014                        &mut chunk.prediction,
1015                    )?;
1016                    target.resampler.native_terms(
1017                        &row,
1018                        npol,
1019                        placed,
1020                        &mut chunk.prediction,
1021                        cell_of(position, channels),
1022                    );
1023                }
1024                let cf = target.operator.cf();
1025                match (&mut predictors[index], cells) {
1026                    (Predictor::Cpu(backend), ChunkCells::Host(cells)) => {
1027                        let first = cell_of(chunk.rows.start, channels);
1028                        let target = NativeTarget::Host { cells, first };
1029                        chunk
1030                            .prediction
1031                            .apply_native(backend, cf, model, target, add)?;
1032                    }
1033                    (Predictor::Metal(backend), ChunkCells::Device(cells)) => {
1034                        let backend: &mut MetalBackend<'_> = backend;
1035                        let target = NativeTarget::Device(cells);
1036                        chunk
1037                            .prediction
1038                            .apply_native(backend, cf, model, target, add)?;
1039                    }
1040                    _ => unreachable!("a wave's predictors and model store share its backend"),
1041                }
1042                Ok::<_, PassError>(())
1043            })?;
1044            add = true;
1045        }
1046        debug_assert!(
1047            pieces
1048                .iter()
1049                .all(|(chunk, ..)| chunk.capacity_bytes() == shape.bytes()),
1050            "a chunk predicts within the buffers it was allocated"
1051        );
1052        Ok(())
1053    }

1055    /// The model at the first `len` native cells of the block for the host:
1056    /// on Metal once every predictor's work, which alone writes the cells,
1057    /// has completed.
1058    fn host_predictions(&mut self, len: usize) -> Result<&[Complex32], PassError> {
1059        match &self.predictions {
1060            NativeStore::Host(cells) => Ok(&cells[..len]),
1061            NativeStore::Device(cells) => {
1062                for predictor in self.predictors.iter_mut().flatten() {
1063                    if let Predictor::Metal(backend) = predictor {
1064                        backend.complete()?;
1065                    }
1066                }
1067                // SAFETY: only the predictors' work writes the cells, every
1068                // command of it has completed, and none is issued while the
1069                // slice, which borrows the wave, lives.
1070                Ok(&unsafe { cells.host_cells() }[..len])
1071            }
1072        }
1073    }

1084    /// Transform the accumulated grids into this wave's normal images, one
1085    /// per domain.
1086    pub(super) fn finish(mut self, team: &WorkerTeam) -> Result<Vec<NormalImages>, PassError> {
1087        // Metal owners leave their last grid work running on the device, and
1088        // Metal predictors their native models: complete both, and learn
1089        // whether any failed, before any accumulator is read.
1090        for predictor in self.predictors.iter_mut().flatten() {
1091            if let Predictor::Metal(backend) = predictor {
1092                backend.complete()?;
1093            }
1094        }
1095        // The weight image's last summed contributions go to the grid first:
1096        // accumulation, which the stages count as such.
1097        let flushing = std::time::Instant::now();
1098        for (domain, target) in self.domains.iter_mut().zip(self.pass.domains) {
1099            let summed = std::sync::Mutex::new([0_u64; 3]);
1100            team.for_each_mut(&mut domain.owners, |_, owner| {
1101                let totals = owner.flush_weights(target)?;
1102                let mut summed = summed.lock().unwrap_or_else(PoisonError::into_inner);
1103                for (sum, total) in summed.iter_mut().zip(totals) {
1104                    *sum += total;
1105                }
1106                drop(summed);
1107                owner.complete()
1108            })?;
1109            let [placements, groups, grids] =
1110                summed.into_inner().unwrap_or_else(PoisonError::into_inner);
1111            if placements > 0 {
1112                tracing::info!(
1113                    "weight image summed: {placements} placements as {groups} group grids in \
1114                     {grids} grids"
1115                );
1116            }
1117        }
1118        self.stages.accumulating += flushing.elapsed();
1119        let pass = self.pass;
1120        let planes = self.planes;
1121        let stages = self.stages;
1122        let transforming = std::time::Instant::now();
1123        let images = self
1124            .domains
1125            .into_iter()
1126            .zip(pass.domains)
1127            .map(|(mut domain, target)| {
1128                let operator = target.operator;
1129                match &target.partition {
1130                    Partition::Planes { .. } => {
1131                        team.for_each_mut(&mut domain.owners, |_, owner| {
1132                            let acc = owner.acc.take().expect("each owner finishes once");
1133                            owner.images = Some(operator.finish(acc)?);
1134                            Ok::<_, PassError>(())
1135                        })?;
1136                        Ok(concatenate(
1137                            domain
1138                                .owners
1139                                .into_iter()
1140                                .map(|owner| owner.images.expect("every owner finished")),
1141                        ))
1142                    }
1143                    Partition::Regions { regions, .. } => {
1144                        // One owner's tile is the whole grid (`Partition::regions`
1145                        // gives a lone region every row): it is the sum, and
1146                        // `finish` refuses a tile. The charge keys on the same
1147                        // count (`DomainStages`).
1148                        if let [owner] = domain.owners.as_mut_slice() {
1149                            let acc = owner.acc.take().expect("each owner finishes once");
1150                            drop(domain.owners);
1151                            return Ok(operator.finish(acc)?);
1152                        }
1153                        // How evenly the strips shared the gridding (#736).
1154                        tracing::info!(
1155                            "region owners: rows {:?}, accumulating {:?} s",
1156                            regions
1157                                .iter()
1158                                .map(|region| region.rows.clone())
1159                                .collect::<Vec<_>>(),
1160                            domain
1161                                .owners
1162                                .iter()
1163                                .map(|owner| (owner.busy.as_secs_f64() * 10.0).round() / 10.0)
1164                                .collect::<Vec<_>>(),
1165                        );
1166                        let mut merged =
1167                            pass.summing(operator.accumulator(planes, None, pass.modes));
1168                        for owner in &domain.owners {
1169                            merged
1170                                .merge_from(owner.acc.as_ref().expect("owners hold their tiles"))?;
1171                        }
1172                        drop(domain.owners);
1173                        Ok(operator.finish(merged)?)
1174                    }
1175                }
1176            })
1177            .collect::<Result<Vec<_>, PassError>>()?;
1178        if stages.pipelined {
1179            tracing::info!(
1180                "wave stages (owners pipelined): predicting {:.1} s, placing with thread \
1181                 waits {:.1} s, waiting for owners {:.1} s, transforming {:.1} s",
1182                stages.predicting.as_secs_f64(),
1183                stages.placing.as_secs_f64(),
1184                stages.accumulating.as_secs_f64(),
1185                transforming.elapsed().as_secs_f64(),
1186            );
1187        } else {
1188            tracing::info!(
1189                "wave stages: predicting {:.1} s, placing {:.1} s, accumulating {:.1} s, \
1190                 transforming {:.1} s",
1191                stages.predicting.as_secs_f64(),
1192                stages.placing.as_secs_f64(),
1193                stages.accumulating.as_secs_f64(),
1194                transforming.elapsed().as_secs_f64(),
1195            );
1196        }
1197        Ok(images)
1198    }
1199}

1201/// The row at `position`: of the block's locality order when `ordered`
1202/// ([`NativeBlock::ordered_row`]), else the row itself.

1204    if ordered {
1205        block.ordered_row(position)
1206    } else {
1207        position
1208    }

1211/// One chunk's share of the wave's model store while it predicts.
1212enum ChunkCells<'c> {
1213    /// The chunk's rows' host cells.
1214    Host(&'c mut [Complex32]),
1215    /// The device cells of the whole block.
1216    Device(&'c DeviceNativeCells),
1217}

1219/// The first native cell of the row at `position` of a block of `channels`
1220/// channels.
1221fn cell_of(position: usize, channels: usize) -> u32 {
1222    u32::try_from(position * channels).expect("a block's native cells fit u32 indices")
1223}

1225/// What an owner subtracts from its data terms: the domain's model at each
1226/// placed sample, the model at the samples' native endpoints, or nothing.
1227#[derive(Clone, Copy)]
1228struct Subtract<'a> {
1229    model: Option<&'a PreparedModelGrids>,
1230    native: Option<NativeModel<'a>>,
1231}
1232
1233/// Grid one owner's share of a slice, through `apply`, in the data and PSF
1234/// modes the pass holds; the data terms receive the residual of what
1235/// `subtract` names. The weight image is summed apart ([`WeightSums`]).
1236fn accumulate(
1237    pass: &MajorCyclePass<'_>,
1238    apply: &mut dyn FnMut(Work<'_>) -> Result<(), casa_imaging_operator::OperatorError>,
1239    subtract: Subtract<'_>,
1240    acc: &mut GridAccumulator,
1241) -> Result<(), PassError> {
1242    if pass.modes.data {
1243        let work = match subtract {
1244            Subtract {
1245                native: Some(model),
1246                ..
1247            } => Work::NativeResidualGrid { model, acc },
1248            Subtract {
1249                model: Some(model), ..
1250            } => Work::ResidualGrid {
1251                model,
1252                acc,
1253                residual_out: None,
1254            },
1255            Subtract { .. } => Work::Grid {
1256                mode: Mode::Data,
1257                acc,
1258            },
1259        };
1260        apply(work)?;
1261    }
1262    if pass.modes.psf {
1263        apply(Work::Grid {
1264            mode: Mode::Psf,
1265            acc,
1266        })?;
1267    }
1268    Ok(())
1269}

1271/// Join per-owner images of consecutive plane ranges.
1272pub(super) fn concatenate(parts: impl IntoIterator<Item = NormalImages>) -> NormalImages {
1273    let mut parts = parts.into_iter();
1274    let mut joined = parts.next().expect("a wave has at least one owner");
1275    for part in parts {
1276        assert!(
1277            part.pols == joined.pols
1278                && part.data_terms == joined.data_terms
1279                && part.psf_terms == joined.psf_terms
