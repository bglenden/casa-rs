15/// Conservative numerical workspace bound, including FFT plans and scratch.
16fn evla_aw_workspace_bytes(request: EvlaAwCellRequest) -> Result<usize, NativeAwGenerationError> {
17    request
18        .generation_workspace_bytes()
19        .map_err(|_| NativeAwGenerationError::InvalidGrid)
20}

22/// One generated, independently cropped plane in x-contiguous order.
23#[derive(Debug)]
24pub struct NativeAwPlane<'a> {
25    /// Square pixel extent after support-buffer cropping.
26    pub size: usize,
27    /// Symmetric support in non-oversampled grid pixels.
28    pub support: usize,
29    /// Complex32 samples with the CASA sampled-area normalization applied.
30    pub values: &'a [Complex32],
31}
32
33/// Imaging and weight cells produced from the same scientific request.
34#[derive(Debug)]
35pub struct NativeAwPair<'a> {
36    /// A times W times the optional anti-aliasing term.
37    pub imaging: NativeAwPlane<'a>,
38    /// Frequency/conjugate-frequency beam product, without the W phase.
39    pub weight: NativeAwPlane<'a>,
40}

42/// Reusable, one-cell numerical allocation. Construct only after admission.
43pub struct EvlaAwWorkspace {
44    size: usize,
45    fft: PreparedFft<f32>,
46    jones: Vec<Complex32>,
47    imaging: Vec<Complex32>,
48    weight: Vec<Complex32>,
49    realized: Option<[(usize, usize); 2]>,
50    reserved_bytes: usize,
51}
52
53impl EvlaAwWorkspace {
54    /// Allocate exactly one admitted working grid; catalog size does not
55    /// multiply its residency. The same allocation is reused for every cell.
56    pub fn new(
57        request: EvlaAwCellRequest,
58        admitted_bytes: usize,
59    ) -> Result<Self, NativeAwGenerationError> {
60        let reserved_bytes = evla_aw_workspace_bytes(request)?;
61        if reserved_bytes > admitted_bytes {
62            return Err(NativeAwGenerationError::WorkspaceMismatch);
63        }
64        let reserve = fft_resident_complex_values_for_shape([request.size; 2])
65            .map_err(|_| NativeAwGenerationError::WorkspaceMismatch)?;
66        let n = request.size * request.size;
67        if (6 * n + reserve) * size_of::<Complex32>() != reserved_bytes {
68            return Err(NativeAwGenerationError::WorkspaceMismatch);
69        }
70        Ok(Self {
71            size: request.size,
72            fft: PreparedFft::<f32>::new([request.size; 2], reserve)
73                .map_err(|_| NativeAwGenerationError::WorkspaceMismatch)?,
74            jones: vec![Complex32::default(); 4 * n],
75            imaging: vec![Complex32::default(); n],
76            weight: vec![Complex32::default(); n],
77            realized: None,
78            reserved_bytes,
79        })
80    }
81
82    /// Charged allocation, including the shared FFT's opaque-plan reservation.
83    #[must_use]
84    pub const fn resident_bytes(&self) -> usize {
85        self.reserved_bytes
86    }
87
88    /// Borrow the last successfully realized pair, with no pixel copies.
89    #[must_use]
90    pub fn pair(&self) -> Option<NativeAwPair<'_>> {
91        let [
92            (imaging_size, imaging_support),
93            (weight_size, weight_support),
94        ] = self.realized?;
95        Some(NativeAwPair {
96            imaging: NativeAwPlane {
97                size: imaging_size,
98                support: imaging_support,
99                values: &self.imaging[..imaging_size * imaging_size],
100            },
101            weight: NativeAwPlane {
102                size: weight_size,
103                support: weight_support,
104                values: &self.weight[..weight_size * weight_size],
105            },
106        })
107    }
108
109    /// Generate a genuine paired EVLA A/W cell without CASA or cache access.
110    ///
111    /// Admission must precede this call using the request's workspace bound plus
112    /// the model's retained bytes. The computation keeps six square Complex32
113    /// planes, independent of catalog size, and compacts both outputs in place.
114    pub fn generate(
115        &mut self,
116        model: &EvlaApertureModel,
117        request: EvlaAwCellRequest,
118    ) -> Result<NativeAwPair<'_>, NativeAwGenerationError> {
119        self.realized = None;
120        evla_aw_workspace_bytes(request)?;
121        if self.size != request.size {
122            return Err(NativeAwGenerationError::WorkspaceMismatch);
123        }
124        let Self {
125            fft,
126            jones,
127            imaging,
128            weight,
129            ..
130        } = self;
131        fill_sky_mueller(model, request, request.frequency_hz, fft, jones, imaging)?;
132        fill_sky_mueller(
133            model,
134            request,
135            request.conjugate_frequency_hz,
136            fft,
137            jones,
138            weight,
139        )?;
140        let origin = (request.size / 2) as isize;
141        let inner_half = (request.size / request.oversampling / 2) as isize;
142        for y in 0..request.size {
143            let iy = y as isize - origin;
144            let m = request.sky_increment_rad[1] * iy as f64;
145            for x in 0..request.size {
146                let ix = x as isize - origin;
147                let p = if request.prolate_spheroidal {
148                    if (-inner_half..inner_half).contains(&ix)
149                        && (-inner_half..inner_half).contains(&iy)
150                    {
151                        (grdsf(ix as f64 / inner_half as f64) as f32)
152                            * (grdsf(iy as f64 / inner_half as f64) as f32)
153                    } else {
154                        0.0
155                    }
156                } else {
157                    1.0
158                };
159                let pixel = x + request.size * y;
160                weight[pixel] =
161                    Complex32::new(p * p, 0.0) * (imaging[pixel] * weight[pixel].conj());
162                let l = request.sky_increment_rad[0] * ix as f64;
163                let rsq = l * l + m * m;
164                let mut screen = Complex32::new(p, 0.0);
165                if request.w_wavelengths > 0.0 && rsq < 1.0 {
166                    let phase =
167                        std::f64::consts::TAU * request.w_wavelengths * ((1.0 - rsq).sqrt() - 1.0);
168                    let (sin, cos) = phase.sin_cos();
169                    screen *= Complex32::new(cos as f32, sin as f32);
170                }
171                imaging[pixel] *= screen;
172            }
173        }
174        // CASA's copy back from the FFT image excludes the final row and column.
175        // Retain these pre-transform edges in the no-longer-used Jones workspace.
176        let edges = &mut jones[..4 * request.size];
177        save_edges(imaging, request.size, &mut edges[..2 * request.size]);
178        save_edges(weight, request.size, &mut edges[2 * request.size..]);
179        transform(fft, request.size, imaging);
180        transform(fft, request.size, weight);
181        restore_edges(imaging, request.size, &edges[..2 * request.size]);
182        restore_edges(weight, request.size, &edges[2 * request.size..]);
183        let mut peak = imaging[0];
184        for value in &imaging[1..] {
185            if value.norm_sqr() > peak.norm_sqr() {
186                peak = *value;
187            }
188        }
189        if peak.norm() == 0.0 || !peak.norm().is_finite() {
190            return Err(NativeAwGenerationError::InvalidNumerics);
191        }
192        for value in imaging.iter_mut() {
193            *value /= peak;
194        }
195        self.realized = Some([
196            crop_normalize(imaging, request)?,
197            crop_normalize(weight, request)?,
198        ]);
199        Ok(self.pair().expect("both crops succeeded"))
200    }
201}

203fn fill_sky_mueller(
204    model: &EvlaApertureModel,
205    request: EvlaAwCellRequest,
206    frequency: f64,
207    fft: &mut PreparedFft<f32>,
208    jones: &mut [Complex32],
209    output: &mut [Complex32],
210) -> Result<(), NativeAwGenerationError> {
211    if !request.aperture {
212        output.fill(Complex32::new(1.0, 0.0));
213        return Ok(());
214    }
215    let wavelength = SPEED_OF_LIGHT_M_PER_S / f64::from(frequency as f32);
216    let cell_m = wavelength / (request.size as f64 * request.sky_increment_rad[0].abs());
217    model.fill_aperture(
218        EvlaApertureGrid::new(
219            request.size,
220            cell_m,
221            3,
222            frequency,
223            request.parallactic_angle_rad,
224        )?,
225        jones,
226    )?;
227    let n = request.size * request.size;
228    for plane in jones.chunks_exact_mut(n) {
229        transform(fft, request.size, plane);
230    }
231    let center = request.size / 2 * (request.size + 1);
232    let mut norm_squared = 0.0_f32;
233    for plane in jones.chunks_exact(n) {
234        norm_squared = (f64::from(norm_squared)
235            + f64::from((plane[center] * plane[center]).norm()) / 2.0)
236            as f32;
237    }
238    if !norm_squared.is_finite() || norm_squared <= 0.0 {
239        return Err(NativeAwGenerationError::InvalidNumerics);
240    }
241    let normalization = norm_squared.sqrt();
242    let selected = if request.mueller == 0 { 0 } else { 3 };
243    for (output, value) in output
244        .iter_mut()
245        .zip(&jones[selected * n..(selected + 1) * n])
246    {
247        let value = (*value / normalization).conj();
248        *output = value * value.conj();
249    }
250    Ok(())
251}

253fn transform(fft: &mut PreparedFft<f32>, size: usize, values: &mut [Complex32]) {
254    let mut plane =
255        ArrayViewMut2::from_shape((size, size).f(), values).expect("validated square workspace");
256    fft.transform(&mut plane, false);
257}
258
259fn save_edges(values: &[Complex32], size: usize, edges: &mut [Complex32]) {
260    edges[..size].copy_from_slice(&values[(size - 1) * size..]);
261    for y in 0..size {
262        edges[size + y] = values[y * size + size - 1];
263    }
264}
265
266fn restore_edges(values: &mut [Complex32], size: usize, edges: &[Complex32]) {
267    values[(size - 1) * size..].copy_from_slice(&edges[..size]);
268    for y in 0..size {
269        values[y * size + size - 1] = edges[size + y];
270    }
271}

273fn crop_normalize(
274    values: &mut [Complex32],
275    request: EvlaAwCellRequest,
276) -> Result<(usize, usize), NativeAwGenerationError> {
277    let origin = request.size / 2;
278    let threshold = (f64::from(values[origin * (request.size + 1)].norm()) * 1e-3) as f32;
279    let mut radius = None;
280    'radii: for r in (2..=origin - 2).rev() {
281        for pixel in 0..90 * r {
282            let angle = std::f64::consts::TAU * pixel as f64 / r as f64;
283            let x = (origin as f64 + r as f64 * angle.sin()) as usize;
284            let y = (origin as f64 + r as f64 * angle.cos()) as usize;
285            if values[x + request.size * y].norm() > threshold {
286                radius = Some(r);
287                break 'radii;
288            }
289        }
290    }
291    let radius = radius.ok_or(NativeAwGenerationError::InvalidNumerics)?;
292    let mut support = (0.5_f32 + radius as f32 / request.oversampling as f32) as usize + 1;
293    if support * request.oversampling + (request.oversampling as f32 / 2.0 + 0.5) as usize > origin
294    {
295        support = origin / request.oversampling - 1;
296    }
297    if support == 0 {
298        return Err(NativeAwGenerationError::InvalidNumerics);
299    }
300    let extent = (request.oversampling * (support + 2)) as isize;
301    let bottom = (((origin as isize - extent) / 2) * 2).max(0) as usize;
302    let top = ((((origin as isize + extent) / 2) * 2 - 1) as usize).min(request.size - 1);
303    let size = top - bottom + 1;
304    let mut area = Complex32::default();
305    for ix in -(support as isize)..support as isize {
306        for iy in -(support as isize)..support as isize {
307            let x = (origin as isize + ix * request.oversampling as isize) as usize;
308            let y = (origin as isize + iy * request.oversampling as isize) as usize;
309            area += values[x + request.size * y];
310        }
311    }
312    if area.norm() == 0.0 || !area.norm().is_finite() {
313        return Err(NativeAwGenerationError::InvalidNumerics);
314    }
315    for y in 0..size {
316        for x in 0..size {
317            values[x + size * y] = values[x + bottom + request.size * (y + bottom)] / area;
318        }
319    }
320    if values[..size * size]
321        .iter()
322        .any(|v| !v.re.is_finite() || !v.im.is_finite())
323    {
324        return Err(NativeAwGenerationError::InvalidNumerics);
325    }
326    Ok((size, support))
327}

335        let surface = casa_imaging_model::EvlaDishSurface::new(
336            (0..=125)

571    fn assert_frozen_plane(role: &str, plane: &NativeAwPlane<'_>, bytes: &[u8]) {
572        eprintln!("T52 {role} size={} support={}", plane.size, plane.support);
573        assert_eq!(bytes.len(), plane.values.len() * 8);
574        let mut error_squared = 0.0_f64;
575        let mut reference_squared = 0.0_f64;
576        let mut native_squared = 0.0_f64;
577        let mut native_sum = num_complex::Complex64::default();
578        let mut reference_sum = num_complex::Complex64::default();
579        let mut maximum_error = 0.0_f64;
580        let mut reference_peak = 0.0_f64;
581        // The T51 exporter writes its canonical last-axis-contiguous
582        // representation; native numerical workspaces are x-contiguous.
583        for (index, bytes) in bytes.chunks_exact(8).enumerate() {
584            let value = plane.values[index / plane.size + plane.size * (index % plane.size)];
585            let reference = Complex32::new(
586                f32::from_le_bytes(bytes[..4].try_into().unwrap()),
587                f32::from_le_bytes(bytes[4..].try_into().unwrap()),
588            );
589            let error = f64::from((value - reference).norm());
590            let magnitude = f64::from(reference.norm());
591            error_squared += error * error;
592            reference_squared += magnitude * magnitude;
593            native_squared += f64::from(value.norm_sqr());
594            native_sum += num_complex::Complex64::new(f64::from(value.re), f64::from(value.im));
595            reference_sum +=
596                num_complex::Complex64::new(f64::from(reference.re), f64::from(reference.im));
597            maximum_error = maximum_error.max(error);
598            reference_peak = reference_peak.max(magnitude);
599        }
600        let relative_l2 = (error_squared / reference_squared).sqrt();
601        let peak_scaled_max = maximum_error / reference_peak;
602        let relative_energy = (native_squared / reference_squared - 1.0).abs();
603        let relative_normalization = (native_sum - reference_sum).norm() / reference_sum.norm();
604        eprintln!(
605            "T52 {role} relative_l2={relative_l2:.12e} peak_scaled_max={peak_scaled_max:.12e} relative_energy={relative_energy:.12e} relative_normalization={relative_normalization:.12e}"
606        );
607        // Independent f32 FFT implementations may differ in roundoff, but
608        // no fitted scale, phase, support, or science tolerance is allowed.
609        assert!(relative_l2 <= 1e-5, "{role} relative L2 {relative_l2}");
610        assert!(
611            relative_energy <= 1e-5,
612            "{role} relative energy {relative_energy}"
613        );
614        assert!(
615            relative_normalization <= 1e-5,
616            "{role} normalization {relative_normalization}"
617        );
618        assert!(
619            peak_scaled_max <= 1e-5,
620            "{role} element error {peak_scaled_max}"
621        );
622    }
