23fn no_coordinates() -> PyErr {
24    raised("Exception: Images in expression have no coordinates.")
25}

27fn storage_from_expression<T: ImageExprValue>(
28    expression: ImageExpression<T>,
29    path: Option<&Path>,
30) -> PyResult<Storage<T>> {
31    if let Some(path) = path {
32        let image = materialize_image_tool_expression(&expression, path).map_err(raised)?;
33        return Ok(Storage::Paged(Box::new(image)));
34    }
35    let record = expression.coordinate_record();
36    let image = TempImage::from_expression(expression).map_err(raised)?;
37    Ok(Storage::Temp {
38        image: Box::new(image),
39        record,
40        fits: None,
41    })
42}

44struct Metadata {
45    coordinates: Option<RecordValue>,
46    image_info: RecordValue,
47    misc_info: RecordValue,
48    units: String,
49}
50
51impl Metadata {
52    fn read(name: &str) -> PyResult<Self> {
53        let path = Path::new(name);
54        if !path.exists() {
55            return Err(raised(format!(
56                "Exception: Cannot access {name} so cannot copy its metadata to output image."
57            )));
58        }
59        if casa_images::expr_file::is_image_expr(path) {
60            let source = super::saved_expression::open(path)?;
61            return Self::from_image(&source.lock());
62        }
63        Self::from_image(&ImageHandle::open(path).map_err(raised)?)
64    }
65
66    fn read_staged(name: &Path, output: &StagedOutput<'_>) -> PyResult<Self> {
67        let entry = casa_tables::lexical_absolute(name);
68        let referent = name.canonicalize().map_err(raised)?;
69        saved_expression::check_output_dependencies(output.target, &[entry, referent], false)?;
70        let image = if casa_images::concat_file::is_image_concat(name) {
71            super::concatenate::virtual_image::open_metadata(name, output)?
72        } else {
73            saved_expression::open_metadata(name, output)?
74        };
75        Self::from_image(&image)
76    }
77
78    fn from_image(image: &ImageHandle) -> PyResult<Self> {
79        if !matches!(
80            image.kind(),
81            ImagePixelType::Float32 | ImagePixelType::Complex32
82        ) {
83            return Err(raised("Exception: Unsupported image pixel data type."));
84        }
85        let info = match image_info_record(image) {
86            Some(record) => record.clone(),
87            None => image_info(image).map_err(raised)?.to_record(),
88        };
89        let (misc_info, units) = with_pixels!(image, storage => with_storage!(storage, image =>
90            (image.misc_info(), image.units().to_owned())
91        ));
92        Ok(Self {
93            coordinates: Some(image.coordinate_record()),
94            image_info: info,
95            misc_info,
96            units,
97        })
98    }
99
100    fn new_output() -> Self {
101        Self {
102            coordinates: None,
103            image_info: ImageInfo::default().to_record(),
104            misc_info: RecordValue::default(),
105            units: String::new(),
106        }
107    }
108
109    fn apply(self, image: &mut ImageHandle) -> PyResult<()> {
110        if let Some(coordinates) = self.coordinates {
111            let system = CoordinateSystem::from_record(&coordinates).map_err(raised)?;
112            if system.n_pixel_axes() == image.shape().len() {
113                image.set_coordinate_record(coordinates)?;
114            }
115        }
116        with_pixels!(image, storage => with_storage!(storage, image => {
117            image.set_units(self.units).map_err(raised)?;
118            image.set_misc_info(self.misc_info).map_err(raised)?;
119            image.set_image_info_record(self.image_info).map_err(raised)
120        }))
121    }
122}

124fn calls_function(expression: &str, function: &str) -> bool {
125    let mut chars = expression.char_indices().peekable();
126    while let Some((start, ch)) = chars.next() {
127        if matches!(ch, '\'' | '"') {
128            // Quoted paths are single image-reference tokens. The expression
129            // parser treats the opposite quote literally and has no escapes.
130            for (_, next) in chars.by_ref() {
131                if next == ch {
132                    break;
133                }
134            }
135        } else if ch.is_ascii_alphabetic() || ch == '_' {
136            while chars
137                .peek()
138                .is_some_and(|(_, ch)| ch.is_alphanumeric() || matches!(ch, '_' | '.' | '/' | '~'))
139            {
140                chars.next();
141            }
142            let end = chars.peek().map_or(expression.len(), |(index, _)| *index);
143            if expression[start..end].eq_ignore_ascii_case(function)
144                && expression[end..].trim_start().starts_with('(')
145            {
146                return true;
147            }
148        }
149    }
150    false
151}

153fn apply_expression_metadata(image: &mut ImageHandle, expression: &str) -> PyResult<()> {
154    let units = if calls_function(expression, "spectralindex") {
155        ""
156    } else if calls_function(expression, "pa") {
157        let mut system =
158            CoordinateSystem::from_record(&image.coordinate_record()).map_err(raised)?;
159        let index = system.find_coordinate(CoordinateType::Stokes).ok_or_else(|| raised(
160            "Failed AlwaysAssert which < nCoordinates() && newCoordinate.nPixelAxes() == coordinates_p[which]->nPixelAxes() && newCoordinate.nWorldAxes() == coordinates_p[which]->nWorldAxes()"
161        ))?;
162        system.replace_coordinate(index, StokesCoordinate::new(vec![StokesType::Pangle]));
163        image.set_coordinate_record(system.to_record())?;
164        "deg"
165    } else {
166        return Ok(());
167    };
168    with_pixels!(image, storage => with_storage!(storage, image => image.set_units(units)))
169        .map_err(raised)
170}

172fn same_file_path(first: &Path, second: &Path) -> bool {
173    first == second
174        || path_identity(first)
175            .zip(path_identity(second))
176            .is_some_and(|(first, second)| first == second)
177}
178
179fn path_identity(path: &Path) -> Option<PathBuf> {
180    path.canonicalize().ok().or_else(|| {
181        let parent = path.parent()?;
182        let parent = if parent.as_os_str().is_empty() {
183            Path::new(".")
184        } else {
185            parent
186        };
187        Some(parent.canonicalize().ok()?.join(path.file_name()?))
188    })
189}

191pub(super) fn check_geometry(
192    expression: &ImageToolPixelExpression,
193    has_sources: bool,
194) -> PyResult<()> {
195    let (shape, has_coordinates) = match expression {
196        ImageToolPixelExpression::Real(expression) => (
197            expression.shape(),
198            expression.coordinates().n_pixel_axes() > 0,
199        ),
200        ImageToolPixelExpression::Complex(expression) => (
201            expression.shape(),
202            expression.coordinates().n_pixel_axes() > 0,
203        ),
204        ImageToolPixelExpression::Boolean(shape) => {
205            return Err(if shape.is_empty() || !has_sources {
206                no_coordinates()
207            } else {
208                raised(
209                    "LatticeExpr::constructor - Bool expression cannot be converted to a numeric type",
210                )
211            });
212        }
213    };
214    if shape.is_empty() || !has_coordinates {
215        return Err(no_coordinates());
216    }
217    Ok(())
218}

220#[pymethods]
221impl PyImageTool {
222    #[pyo3(signature = (outfile="", pixels="", overwrite=false, imagemd="", prec="float"), text_signature = "($self, outfile='', pixels='', overwrite=False, imagemd='', prec='float')")]
223    fn imagecalc(
224        &self,
225        py: Python<'_>,
226        outfile: &str,
227        pixels: &str,
228        overwrite: bool,
229        imagemd: &str,
230        prec: &str,
231    ) -> PyResult<Py<PyImageTool>> {
232        let attached = py.detach(|| {
233            if pixels.is_empty() {
234                return Err(raised("Exception: You must provide an expression using the pixels parameter."));
235            }
236            let as_float = match prec.as_bytes().first().map(u8::to_ascii_lowercase) {
237                Some(b'f') => true,
238                Some(b'd') => false,
239                _ => return Err(raised("Exception: Unsupported value for type, it must be 'float' or 'double'.")),
240            };
241            let inputs = ImageToolExpressionInputs::resolve(pixels).map_err(raised)?;
242            let has_sources = inputs.paths().next().is_some();
243            let uses_bound_sources = saved_expression::has_manifest_operands(&inputs);
244            let (expression, retained, paths) = if uses_bound_sources {
245                saved_expression::compile_inline(&inputs, outfile.is_empty())?
246            } else {
247                let retained = AttachedImage::open_expression_inputs(inputs.paths())?;
248                let expression = if outfile.is_empty() { inputs.compile_lazy() } else { inputs.compile() }
249                    .map_err(raised)?;
250                (expression, retained, inputs.paths().map(Path::to_path_buf).collect())
251            };
252            if !outfile.is_empty() && !overwrite && Path::new(outfile).symlink_metadata().is_ok() {
253                return Err(raised(format!("Exception: File {outfile} exists, and the user does not want to remove it..")));
254            }
255            check_geometry(&expression, has_sources)?;
256            if !outfile.is_empty() {
257                let output = Path::new(outfile);
258                let overlap_error = || raised(
259                    "Exception: Caught TableError. This often means the table you are trying to overwrite has been opened by another method and so cannot be overwritten at this time. Try closing it and rerunning."
260                );
261                if uses_bound_sources {
262                    saved_expression::check_output_dependencies(output, &paths, false)
263                        .map_err(|_| overlap_error())?;
264                } else {
265                    for input in &paths {
266                        ensure_outfile_does_not_alias_input(output, input).map_err(|_| overlap_error())?;
267                    }
268                }
269            }
270            let metadata_path = Path::new(imagemd);
271            let output_path = Path::new(outfile);
272            let same_output = !outfile.is_empty() && same_file_path(output_path, metadata_path);
273            let staged_metadata = !outfile.is_empty() && !same_output
274                && (casa_images::expr_file::is_image_expr(metadata_path)
275                    || casa_images::concat_file::is_image_concat(metadata_path));
276            let metadata = if imagemd.is_empty() || staged_metadata {
277                None
278            } else if same_output {
279                if !as_float {
280                    return Err(raised("Exception: Unsupported image pixel data type."));
281                }
282                Some(Metadata::new_output())
283            } else {
284                Some(Metadata::read(imagemd)?)
285            };
286            let history = self.state().history;
287            let mut result = publish_image(outfile, overwrite, |path| {
288                let mut output = match expression {
289                    ImageToolPixelExpression::Real(expression) if as_float => ImageHandle::Float(
290                        storage_from_expression(expression.convert_pixels(|value| value as f32).map_err(raised)?, path)?,
291                    ),
292                    ImageToolPixelExpression::Real(expression) => ImageHandle::Double(storage_from_expression(expression, path)?),
293                    ImageToolPixelExpression::Complex(expression) if as_float => ImageHandle::Complex(
294                        storage_from_expression(expression.convert_pixels(|value| Complex32::new(value.re as f32, value.im as f32)).map_err(raised)?, path)?,
295                    ),
296                    ImageToolPixelExpression::Complex(expression) => ImageHandle::DComplex(storage_from_expression(expression, path)?),
297                    ImageToolPixelExpression::Boolean(_) => unreachable!("numeric expression checked"),
298                };
299                if staged_metadata {
300                    let staged = StagedOutput::new(output_path, path.expect("named staged image"))?;
301                    Metadata::read_staged(metadata_path, &staged)?.apply(&mut output)?;
302                } else if let Some(metadata) = metadata {
303                    metadata.apply(&mut output)?;
304                }
305                apply_expression_metadata(&mut output, pixels)?;
306                if history {
307                    let [outfile, pixels, imagemd] = [outfile, pixels, imagemd].map(|value| if value.len() > 300 { "..." } else { value });
308                    output.record_history("imagecalc", format!(
309                        "ia.imagecalc(outfile=\"{outfile}\", pixels=\"{pixels}\", overwrite={overwrite}, imagemd=\"{imagemd}\")"
310                    ))?;
311                }
312                output.save().map_err(raised)?;
313                Ok(output)
314            })?;
315            if outfile.is_empty() {
316                result.retain_expression_inputs(retained, paths);
317            }
318            Ok(result)
319        })?;
320        Py::new(py, Self::attached(attached))
321    }
