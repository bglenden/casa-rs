4//! O2 line refractivities (ATM's opacity sources 1 to 4).
5//!
6//! The (16)O(16)O ground-state lines (the 60 GHz band and 118.75 GHz) are
7//! broadened by dry air and water vapour and include line interference.
8//! ATM adds nine weak mesospheric O2 (¹Δ) lines to them. The vibrational
9//! state and the isotopologues use a fixed broadening.
10
11use super::tables::{o2, o2_16o18o, o2_vib};
12use super::{
13    H_OVER_K, PI, PICUBE8DIV3HCESU, ZERO, by_pressure, frequency_bin, interference_o2,
14    line_broadening, line_broadening_o2, line_shape, window_bounds, window_range,
15};
16use num_complex::Complex64;
17
18/// O2's dipole moment as ATM writes it (the magnetic dipole, in its units).
19const MU: f64 = 0.0186;
20
21/// (16)O(16)O from the ground vibrational state (source 1), up to
22/// 1599.9 GHz.
23///
24/// The nine mesospheric lines are multiplied by 3e13 in layers below
25/// 1.7 hPa, and they count only when the main catalogue's window is not
26/// empty. Between 118.750345 and 124 GHz the window starts at line 37
27/// (the 118.75 GHz line). A negative absorption is set to zero, keeping
28/// the delay.
29pub(super) fn o2(tt: f64, pp: f64, eh2o: f64, nu: f64) -> Complex64 {
30    if nu > 1599.9 {
31        return ZERO;
32    }
33    let mut sum = ZERO;
34    for i in 0..o2::FREEXTRA.len() {
35        let fre = o2::FREEXTRA[i];
36        let width = line_broadening(
37            fre,
38            tt,
39            pp,
40            32.0,
41            o2::BRDO2AIREXTRA[i] * 0.001,
42            o2::TEXPO2EXTRA[i],
43        );
44        let mut shape = line_shape(nu, fre, width, 0.0)
45            * o2::FLINEXTRA[i]
46            * (-(o2::EUEXTRA[i] - 0.0479924207 * fre) / tt).exp()
47            * fre;
48        if pp < 1.7 {
49            shape *= 30000000000000.0;
50        }
51        sum += shape;
52    }
53    let tables = by_pressure(
54        pp,
55        (&o2::INI1[..], &o2::IFIN1[..]),
56        (&o2::INI1[..], &o2::IFIN1[..]),
57        (&o2::INI2[..], &o2::IFIN2[..]),
58    );
59    let (mut start, end) = window_bounds(frequency_bin(nu), 2, tables, 0);
60    if (118.750345..=124.0).contains(&nu) {
61        start = 37;
62    }
63    let Some(range) = window_range(start, end) else {
64        return ZERO;
65    };
66    for i in range {
67        let fre = o2::FRE[i];
68        let [b1, b2, b3, b4] = o2::BROADENING[i];
69        let width = line_broadening_o2(fre, tt, pp, eh2o, 32.0, b1, b2);
70        sum += line_shape(nu, fre, width, interference_o2(tt, pp, b3, b4))
71            * o2::FLIN[i]
72            * (-o2::EL[i] / tt).exp()
73            * (1.0 - (-H_OVER_K * fre / tt).exp());
74    }
75    let mut sum = sum * (nu / PI) * (PICUBE8DIV3HCESU * MU.powf(2.0) / (0.72923 * tt));
76    if sum.im < 0.0 {
77        sum = Complex64::new(sum.re, 0.0);
78    }
79    sum * 1e-4
80}
81
82/// A fixed-broadening O2 catalogue summed over all its lines, up to
83/// 999.9 GHz.
84fn fixed_broadening(
85    tt: f64,
86    pp: f64,
87    eh2o: f64,
88    nu: f64,
89    mmol: f64,
90    q: f64,
91    lines: [&[f64]; 3],
92) -> Complex64 {
93    if nu > 999.9 {
94        return ZERO;
95    }
96    let [frequency, intensity, energy] = lines;
97    let mut sum = ZERO;
98    for i in 0..frequency.len() {
99        let fre = frequency[i];
100        let width = line_broadening_o2(fre, tt, pp, eh2o, mmol, 2.0, 0.2);
101        sum += line_shape(nu, fre, width, 0.0)
102            * intensity[i]
103            * (-energy[i] / tt).exp()
104            * (1.0 - (-H_OVER_K * fre / tt).exp());
105    }
106    sum * (nu / PI) * (PICUBE8DIV3HCESU * MU.powf(2.0) / q) * 1e-4
107}

109/// (16)O(16)O from the first vibrational state (source 2).
110pub(super) fn o2_vibrational(tt: f64, pp: f64, eh2o: f64, nu: f64) -> Complex64 {
111    fixed_broadening(
112        tt,
113        pp,
114        eh2o,
115        nu,
116        32.0,
117        0.72923 * tt,
118        [&o2_vib::FRE, &o2_vib::FLIN, &o2_vib::EL],
119    )
120}

122/// (16)O(18)O (source 3).
123pub(super) fn o2_isotope18(tt: f64, pp: f64, eh2o: f64, nu: f64) -> Complex64 {
124    fixed_broadening(
125        tt,
126        pp,
127        eh2o,
128        nu,
129        34.0,
130        1.536568889 * tt,
131        [&o2_16o18o::FRE, &o2_16o18o::FLIN, &o2_16o18o::EL],
132    )
133}
