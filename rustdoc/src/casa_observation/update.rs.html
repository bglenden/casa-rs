2//! Writing back: what an update session may write, and the values one
3//! delivered batch's samples receive (`docs/measurement-set/contracts.md`,
4//! section 7, "Updates").
5//!
6//! An update addresses the samples of a batch the same session delivered:
7//! the store maps them to its own native units (MAIN rows, Zarr chunks),
8//! and a sample with no native address (MSv2 padding) receives nothing. The
9//! targets are declared when the session opens, so the structural preflight
10//! (the variable exists, the data groups sharing it are accepted) runs
11//! before any payload. Values are checked against the batch before anything
12//! is written; a refused write writes nothing of its batch.
13//!
14//! **Trial status.** This is the write-back trial (`docs/measurement-set/
15//! trial/README.md`, "Write-back"): one updater per traversal, written batch
16//! for batch in delivery order, boolean flags only; the slices of a
17//! [`TimeSplit`](crate::TimeSplit) each have their own updater, live at
18//! once.
19

25/// A role a session writes, of one data group.
26#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
27pub enum WritableRole {
28    /// The group's visibilities (MSv2 `DATA`, `CORRECTED_DATA` or
29    /// `MODEL_DATA`; MSv4 the group's correlated-data variable).
30    Visibilities,
31    /// The group's element flags (`FLAG`). Row flags (`FLAG_ROW`) and flag
32    /// categories are never written: a flag change cannot clear them.
33    Flags,
34    /// The group's weights (`WEIGHT`, `WEIGHT_SPECTRUM` or MSv4 `WEIGHT`).
35    Weights,
36}
37
38/// Whether a target may write a variable other data groups share: data
39/// groups commonly share their flags and weights, so writing one group's
40/// flags changes every group's.
41#[derive(Clone, Copy, Debug, PartialEq, Eq, Default)]
42pub enum AliasScope {
43    /// Refuse the target when another group shares its variable.
44    #[default]
45    Refuse,
46    /// Write the shared variable, for every group that shares it.
47    Accept,
48}

50/// One thing an update session may write, declared when it opens.
51#[derive(Clone, Debug, PartialEq, Eq)]
52pub struct UpdateTarget {
53    /// The data group, by name.
54    pub data_group: String,
55    /// The role.
56    pub role: WritableRole,
57    /// Whether the variable may be shared with other groups.
58    pub aliases: AliasScope,
59    /// Create the group's visibility variable when the store lacks it,
60    /// initialized as CASA initializes it: `corrected` from the `base`
61    /// group's visibilities (`CORRECTED_DATA` from `DATA`), so what a pass
62    /// skips reads as uncorrected; `model` with a unit point source at the
63    /// phase centre (1, and 0 in the cross hands), so what a pass skips
64    /// reads as CASA's default model. Stores that cannot add a variable
65    /// refuse.
66    pub create: bool,
67}
68
69impl UpdateTarget {
70    /// A target that writes `role` of `data_group`, refusing aliases and
71    /// creating nothing.
72    #[must_use]
73    pub fn new(data_group: impl Into<String>, role: WritableRole) -> Self {
74        Self {
75            data_group: data_group.into(),
76            role,
77            aliases: AliasScope::Refuse,
78            create: false,
79        }
80    }
81}

83/// A target as the store resolved it: the native variable it writes and
84/// the other groups that share that variable.
85#[derive(Clone, Debug, PartialEq, Eq)]
86pub struct ResolvedTarget {
87    /// The declared target.
88    pub target: UpdateTarget,
89    /// The native variable or column written.
90    pub variable: String,
91    /// The other data groups that read the same variable, which the write
92    /// changes too.
93    pub aliases: Vec<String>,
94    /// Whether the session creates the variable.
95    pub created: bool,
96}

98/// What a store's updates guarantee on failure.
99#[derive(Clone, Copy, Debug, PartialEq, Eq)]
100pub struct UpdateGuarantees {
101    /// Each physical unit (an MSv4 chunk object) is replaced whole: a
102    /// failure or a killed process leaves the old unit or the new one,
103    /// never a mixture. Units already replaced stay replaced: there is no
104    /// transaction across units.
105    pub atomic_units: bool,
106    /// Another process is excluded while any of the session's updaters is
107    /// live (casacore's write lock on MAIN for MSv2). MSv4 has no
108    /// cross-process lock protocol.
109    pub cross_process_lock: bool,
110}

112/// How a weight patch treats channels the native weight shares: an MSv2
113/// `WEIGHT` is one value per row and correlation that every channel reads.
114#[derive(Clone, Copy, Debug, PartialEq, Eq)]
115pub enum WeightChannels {
116    /// Change the delivered channels only. Where the native weight is
117    /// shared by channels, this is refused unless the batch delivers every
118    /// channel and gives each sample and product one value: the patch must
119    /// not change a channel it does not address.
120    Delivered,
121    /// Change every channel of the partition: the patch gives each sample
122    /// and product one value, which every channel takes.
123    All,
124}

126/// The new values of one target for the samples, channels and products of
127/// the batch they are written against, each `[sample, channel, product]`.
128#[derive(Clone, Copy, Debug)]
129pub enum PatchValues<'a> {
130    /// Replace the visibilities.
131    Visibilities(ArrayView3<'a, Complex32>),
132    /// Change element flags: `set` sets and `clear` clears where true;
133    /// elements neither names keep their stored flag. A sample may not be
134    /// both set and cleared.
135    Flags {
136        /// Flags to set.
137        set: Option<ArrayView3<'a, bool>>,
138        /// Flags to clear.
139        clear: Option<ArrayView3<'a, bool>>,
140    },
141    /// Replace the weights.
142    Weights {
143        /// The weights.
144        values: ArrayView3<'a, f32>,
145        /// Which channels they change.
146        channels: WeightChannels,
147    },
148}
149
150impl PatchValues<'_> {
151    /// The role these values write.
152    #[must_use]
153    pub fn role(&self) -> WritableRole {
154        match self {
155            Self::Visibilities(_) => WritableRole::Visibilities,
156            Self::Flags { .. } => WritableRole::Flags,
157            Self::Weights { .. } => WritableRole::Weights,
158        }
159    }
160}

162/// What an updater wrote, for the evidence.
163#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
164pub struct UpdateReport {
165    /// Batches written.
166    pub batches: usize,
167    /// Samples given a native address and written.
168    pub samples: usize,
169    /// Samples with no native address (MSv2 padding): nothing was written
170    /// for them.
171    pub unaddressed: usize,
172    /// Native units written: MSv2 cells, MSv4 chunk objects.
173    pub units_written: usize,
174    /// Native units read back to merge a change into what they hold: an
175    /// MSv2 cell a patch covers only partly, an MSv4 chunk a flush covers
176    /// only partly. A unit a patch supplies whole is never read.
177    pub units_read: usize,
178    /// The most bytes the updater holds between batches: MSv4, one
179    /// chunk-row of times of each target variable over the channel chunks
180    /// the traversal reaches (every channel for weights); MSv2, none.
181    pub held_bytes: usize,
182}

184/// Writes one traversal's batches back to its store.
185///
186/// An updater is bound to a traversal and to some of its session's
187/// targets (`casa_ms::observation_io::UpdateSession::updater`); each call
188/// gives the values of those targets, in that order, for one delivered
189/// batch.
190///
191/// Only [`Self::finish`] completes an update: an updater dropped without
192/// it leaves what it wrote and discards what it held (an MSv4 updater's
193/// window of up to one chunk-row of times), and reports neither.
194pub trait Updater: Send {
195    /// Write `values`, one per bound target, for `batch`'s samples.
196    ///
197    /// # Errors
198    ///
199    /// When a value disagrees with its target or the batch, the batch is
200    /// not the traversal's next, or a change is refused (a weight shared by
201    /// channels the patch does not address, a flag cleared or a finite
202    /// weight written where the store holds padding). Nothing of `batch` is
203    /// then written. A storage failure
204    /// leaves what the store's guarantees state.
205    fn write(
206        &mut self,
207        batch: &VisibilityBatch<'_>,
208        values: &[PatchValues<'_>],
209    ) -> Result<(), ObservationError>;
210
211    /// Finish: write what is still held and report.
212    ///
213    /// # Errors
214    ///
215    /// When a batch of the traversal was never written (a store that plans
216    /// its writes up front needs every one), or storage fails.
217    fn finish(self: Box<Self>) -> Result<UpdateReport, ObservationError>;
218}

220/// Check `values` against the bound `roles` and `batch` before anything is
221/// written: one value per role, in order, each of the batch's shape; no
222/// sample both set and cleared; and a weight patch for every channel giving
223/// each sample and product one value.
224///
225/// # Errors
226///
227/// [`ObservationError::InvalidRequest`] naming the first disagreement.
228pub fn check_values(
229    batch: &VisibilityBatch<'_>,
230    roles: &[WritableRole],
231    values: &[PatchValues<'_>],
232) -> Result<(), ObservationError> {
233    let refuse = |reason: String| Err(ObservationError::InvalidRequest { reason });
234    if values.len() != roles.len() {
235        return refuse(format!(
236            "{} patch values for {} bound targets",
237            values.len(),
238            roles.len()
239        ));
240    }
241    let shape = batch.shape();
242    for (index, (value, role)) in values.iter().zip(roles).enumerate() {
243        if value.role() != *role {
244            return refuse(format!(
245                "patch value {index} writes {:?}; its target is {role:?}",
246                value.role()
247            ));
248        }
249        let shapes: Vec<&[usize]> = match value {
250            PatchValues::Visibilities(values) => vec![values.shape()],
251            PatchValues::Flags { set, clear } => {
252                set.iter().chain(clear).map(|mask| mask.shape()).collect()
253            }
254            PatchValues::Weights { values, .. } => vec![values.shape()],
255        };
256        if let Some(wrong) = shapes.iter().find(|actual| **actual != shape.as_slice()) {
257            return refuse(format!(
258                "patch value {index} has shape {wrong:?}; the batch's is {shape:?}"
259            ));
260        }
261        match value {
262            PatchValues::Flags {
263                set: Some(set),
264                clear: Some(clear),
265            } if set.iter().zip(clear).any(|(set, clear)| *set && *clear) => {
266                return refuse(format!("patch value {index} both sets and clears a flag"));
267            }
268            // Weights patched from weights made from the sigmas would write
269            // sigmas back as weights, unannounced.
270            PatchValues::Weights { .. } if batch.group.weight_rule != WeightRule::Stored => {
271                return refuse(format!(
272                    "patch value {index} writes weights for a batch whose weights were made \
273                     from the sigmas ({:?}); patch weights from a traversal of the stored \
274                     weights",
275                    batch.group.weight_rule
276                ));
277            }
278            PatchValues::Weights {
279                values,
280                channels: WeightChannels::All,
281            } if !uniform_over_channels(values) => {
282                return refuse(format!(
283                    "patch value {index} changes every channel's weight but gives a sample's \
284                     channels different values"
285                ));
286            }
287            _ => {}
288        }
289    }
290    Ok(())
291}

293/// Whether every sample and product has one value across channels.
294#[must_use]
295pub fn uniform_over_channels(values: &ArrayView3<'_, f32>) -> bool {
296    values.axis_iter(Axis(0)).all(|sample| {
297        let first = sample.index_axis(Axis(0), 0);
298        sample.axis_iter(Axis(0)).all(|channel| {
299            channel
300                .iter()
301                .zip(&first)
302                .all(|(a, b)| a.to_bits() == b.to_bits())
303        })
304    })
