2//! Regrid TP, multiply by the synthesis PB, feather, and divide by PB.
3use super::{failure, plan::Plan};
4use crate::{SessionResources, simobserve::SimobserveError, simutil::analysis};
5use casa_imageanalysis::{
6    ImpbcorRequest, RegridTemplate,
7    analysis::{FeatherRequest, feather},
8    impbcor, regrid,
9};
10use casa_images::{PagedImage, RegridMethod};
11use casa_imaging_runtime::{Demand, admit};
12use std::path::{Path, PathBuf};
13
14pub(super) fn execute(
15    plan: &Plan,
16    high: &Path,
17    resources: SessionResources,
18) -> Result<Vec<PathBuf>, SimobserveError> {
19    let low = plan.directory.join(format!("{}.sd.image", plan.project));
20    let image = PagedImage::<f32>::open(high).map_err(failure)?;
21    let tp = PagedImage::<f32>::open(&low).map_err(failure)?;
22    let cells = |shape: &[usize]| {
23        shape
24            .iter()
25            .try_fold(1u64, |count, &n| count.checked_mul(n as u64))
26    };
27    let memory = cells(image.shape())
28        .and_then(|n| n.checked_mul(160))
29        .and_then(|high| cells(tp.shape())?.checked_mul(48)?.checked_add(high))
30        .and_then(|bytes| bytes.checked_add(128 << 20))
31        .ok_or_else(|| failure("combined image dimensions overflow"))?;
32    let _memory = admit(
33        resources.host(),
34        resources.policy(),
35        &Demand {
36            phase: "ALMA PB-weighted feather",
37            memory,
38        },
39    )
40    .map_err(failure)?;
41    let template = RegridTemplate::of(&image);
42    let interpolated = regrid(&tp, &template, RegridMethod::Cubic).map_err(failure)?;
43    let staging = tempfile::tempdir_in(&plan.directory).map_err(failure)?;
44    let regridded = staging.path().join("tp.regrid");
45    analysis::write_product(
46        &regridded,
47        &interpolated.data,
48        interpolated.coordinates,
49        tp.units(),
50        &tp.image_info().map_err(failure)?,
51        Some(&interpolated.mask),
52    )
53    .map_err(failure)?;
54    drop(interpolated.data);
55    let pb = high.with_extension("pb");
56    let scaled = plan
57        .directory
58        .join(format!("{}.sd.image.pbscaled", plan.project));
59    pbcor(&regridded, &pb, &scaled, "multiply")?;
60    let raw = plan
61        .directory
62        .join(format!("{}.feather.image0", plan.project));
63    feather(&FeatherRequest {
64        imagename: raw.clone(),
65        highres: high.to_path_buf(),
66        lowres: scaled.clone(),
67        sdfactor: 1.,
68        effdishdiam: -1.0,
69        lowpassfiltersd: false,
70        overwrite: false,
71    })
72    .map_err(failure)?;
73    let output = plan
74        .directory
75        .join(format!("{}.feather.image", plan.project));
76    pbcor(&raw, &pb, &output, "divide")?;
77    Ok(vec![scaled, raw, output])
78}
79
80fn pbcor(image: &Path, pb: &Path, output: &Path, mode: &str) -> Result<(), SimobserveError> {
81    impbcor(&ImpbcorRequest {
82        imagename: image.to_path_buf(),
83        pbimage: pb.to_path_buf().into(),
84        outfile: output.to_path_buf(),
85        mode: mode.to_owned(),
86        cutoff: -1.,
87        box_selection: None,
88        region: None,
89        chans: None,
90        stokes: None,
91        mask: None,
92        stretch: false,
93        overwrite: false,
94    })
95    .map_err(failure)?;
96    Ok(())
97}
