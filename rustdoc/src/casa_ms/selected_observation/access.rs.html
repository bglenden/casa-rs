60fn aw_pointing_plan_retained_byte_ceiling(
61    problem: &CompiledProblem,
62    source: &ObservationSource,
63) -> Result<usize, BoundObservationSourceError> {
64    let enabled = problem
65        .science()
66        .measurement_equation()
67        .aw_projection()
68        .is_some_and(|contract| contract.use_pointing());
69    if !enabled {
70        return Ok(0);
71    }
72    usize::try_from(source.selection().rows().selected_row_count())
73        .ok()
74        .and_then(|rows| rows.checked_mul(AW_POINTING_PLAN_BYTES_PER_SELECTED_ROW))
75        .and_then(|bytes| bytes.checked_add(size_of::<AwPointingEpochPlan>()))
76        .ok_or(BoundObservationSourceError::MeasurementOverflow)
77}

366    #[allow(clippy::too_many_arguments)]
367    fn from_locked_measurement_set(
368        problem: &CompiledProblem,
369        source: &ObservationSource,
370        selected_read_state: ObservationSourceState,
371        measures: &SelectedObservationMeasures,
372        shared_bytes: SelectedObservationSharedBytes,
373        content_budget: SelectedObservationContentBudget,
374        measurement_set: MeasurementSet,
375        ephemeris: Option<&Arc<crate::SelectedObservationEphemeris>>,
376        pointing_query_domain: Option<&SelectedPointingQueryDomain>,
377    ) -> Result<Self, BoundObservationSourceError> {
378        let aw_pointing_plan_ceiling = aw_pointing_plan_retained_byte_ceiling(problem, source)?;
379        let shared_bytes = shared_bytes.with_source_plan_retained_bytes(aw_pointing_plan_ceiling);
380        let preliminary_content_plan = Self::requirements_for_locked_source(
381            &measurement_set,
382            problem,
383            source,
384            shared_bytes,
385            content_budget.maximum_pointing_polynomial_terms(),
386            pointing_query_domain,
387        )?
388        .plan(content_budget)?;
389        let pointing_catalog = if let PointingCentreLaw::Observation(law) =
390            problem.geometry().centres().pointing()
391        {
392            let domain = pointing_query_domain
393                .ok_or(BoundObservationSourceError::MissingPointingQueryDomain)?;
394            let column = match law.direction_column() {
395                PointingDirectionColumn::Direction => StoredPointingDirectionColumn::Direction,
396                PointingDirectionColumn::Target => StoredPointingDirectionColumn::Target,
397            };
398            let catalog_budget = selected_pointing_catalog_budget(
399                &measurement_set,
400                problem,
401                source,
402                shared_bytes,
403                content_budget,
404            )?;
405            let catalog = measurement_set.prepare_selected_pointing_catalog(
406                column,
407                domain,
408                law.time_sampling(),
409                PointingReadPlan::new(
410                    preliminary_content_plan.rows_per_block(),
411                    preliminary_content_plan.maximum_pointing_polynomial_terms(),
412                    catalog_budget,
413                )?,
414            )?;
415            let catalog_measurements = catalog.measurements();
416            tracing::info!(
417                target: "casa_ms::selected_pointing",
418                "prepared selected POINTING catalog source_rows_scanned={} retained_rows={} retained_bytes={} construction_peak_bytes={} build_nanos={}",
419                catalog_measurements.source_rows_scanned(),
420                catalog_measurements.retained_rows(),
421                catalog_measurements.retained_bytes(),
422                catalog_measurements.construction_peak_bytes(),
423                catalog_measurements.build_nanos(),
424            );
425            Some(catalog)
426        } else {
427            None
428        };
429        let content_plan = selected_content_plan_with_pointing_catalog(
430            &measurement_set,
431            problem,
432            source,
433            shared_bytes,
434            content_budget,
435            pointing_catalog
436                .as_ref()
437                .map(|catalog| catalog.measurements()),
438        )?;
439        let mut bound = Self::from_planned_locked_measurement_set(
440            source,
441            selected_read_state,
442            measures,
443            content_plan,
444            measurement_set,
445            ephemeris,
446            pointing_catalog,
447        )?;
448        if aw_pointing_plan_ceiling != 0 {
449            let plan = build_aw_pointing_epoch_plan(problem, source, &bound)?;
450            if plan.retained_byte_ceiling > aw_pointing_plan_ceiling {
451                return Err(BoundObservationSourceError::MeasurementOverflow);
452            }
453            tracing::info!(
454                target: "casa_ms::selected_pointing",
455                "prepared CASA AW pointing epoch plan epochs={} retained_byte_ceiling={}",
456                plan.chart_pixels_by_epoch.len(),
457                plan.retained_byte_ceiling,
458            );
459            bound.aw_pointing_plan = Some(plan);
460        }
461        Ok(bound)
462    }

464    fn from_planned_locked_measurement_set(
465        source: &ObservationSource,
466        selected_read_state: ObservationSourceState,
467        measures: &SelectedObservationMeasures,
468        content_plan: SelectedObservationContentPlan,
469        measurement_set: MeasurementSet,
470        ephemeris: Option<&Arc<crate::SelectedObservationEphemeris>>,
471        pointing_catalog: Option<PreparedSelectedPointingCatalog>,
472    ) -> Result<Self, BoundObservationSourceError> {
473        let row_predicate = selected_row_predicate(&measurement_set, source)?;
474        let coordinates = Arc::from(selected_coordinates(&measurement_set, source.selection())?);
475        let data_description_count = measurement_set.data_description()?.row_count();
476        if row_predicate.requires_every_source_row(data_description_count)
477            && source.selection().rows().selected_row_count()
478                != u64::try_from(measurement_set.row_count())
479                    .map_err(|_| BoundObservationSourceError::PhysicalRowIndexOverflow)?
480        {
481            return Err(BoundObservationSourceError::IncompleteUnconditionalRowManifest);
482        }
483        let geometry_engine = Arc::new(MsCalEngine::new_selected_observation(
484            &measurement_set,
485            measures.provider(),
486            measures.provider_state(),
487            ephemeris.cloned(),
488        )?);
489        geometry_engine.verify_selected_observation_measures()?;
490        Ok(Self {
491            source_identity: source.identity(),
492            selected_read_state,
493            source_row_count_matches: usize::try_from(source.selection().rows().source_row_count())
494                .ok()
495                == Some(measurement_set.row_count()),
496            measurement_set,
497            geometry_engine,
498            row_predicate,
499            coordinates,
500            pointing_catalog,
501            aw_pointing_plan: None,
502            content_plan,
503        })
504    }

610    pub(super) fn fill_next_selected_block(
611        &self,
612        problem: &CompiledProblem,
613        logical_source: &MeasurementSetReadAccess,
614        replay: &mut SelectedRowReplay,
615        block: &mut SelectedObservationBlock,
616        measurements: &mut SelectedObservationTraversalMeasurementsBuilder,
617    ) -> Result<bool, BoundObservationSourceError> {
618        self.geometry_engine
619            .verify_selected_observation_measures()?;
620        let Some(coordinate_index) =
621            self.fill_selected_row_group(logical_source, replay, &mut block.request_rows)?
622        else {
623            return Ok(false);
624        };
625        let coordinates = &self.coordinates[coordinate_index];
626        let fill_report = self.measurement_set.fill_selected_observation_buffer(
627            &SelectedObservationBufferRequest::new(
628                selected_visibility(logical_source.selected_columns().visibility()),
629                selected_weight(logical_source.selected_columns().weights()),
630                &block.request_rows,
631                VisibilityChannelReadRange::new(
632                    coordinates.channel_start,
633                    coordinates.channel_count,
634                ),
635            ),
636            &mut block.buffer,
637        )?;
638        block.logical_bytes = fill_report.logical_output_bytes;
639        block.source_read_operations = fill_report.read_operation_count;
640        measurements
641            .record_selected_channel_runs(fill_report.row_count, coordinates.channels.len())?;
642        measurements.record_fill(fill_report)?;
643        let arrangement_started = Instant::now();
644        for row in 0..block.buffer.row_count() {
645            let stored = block
646                .buffer
647                .sample(0, row, 0)
648                .ok_or(BoundObservationSourceError::StoredSampleShapeMismatch)?;
649            if u32::try_from(stored.data_description_id()).ok()
650                != Some(coordinates.data_description.data_description_id())
651            {
652                return Err(
653                    BoundObservationSourceError::DataDescriptionCoordinateMismatch {
654                        data_description_id: coordinates.data_description.data_description_id(),
655                    },
656                );
657            }
658            if !self.row_predicate.matches(StoredMainRow::from(stored)) {
659                return Err(BoundObservationSourceError::SelectedRowPredicateMismatch {
660                    physical_row: u64::try_from(stored.physical_row())
661                        .map_err(|_| BoundObservationSourceError::PhysicalRowIndexOverflow)?,
662                });
663            }
664        }
665        let observation_pointings = evaluate_observation_pointings(self, problem, &block.buffer)?;
666        block.row_geometry.clear();
667        for row in 0..block.buffer.row_count() {
668            let stored = block
669                .buffer
670                .sample(0, row, 0)
671                .ok_or(BoundObservationSourceError::StoredSampleShapeMismatch)?;
672            block.row_geometry.push(evaluate_row_geometry(
673                self,
674                problem,
675                stored,
676                observation_pointings
677                    .as_ref()
678                    .map(|pointings| pointings[row]),
679            )?);
680        }
681        block.coordinate_index = coordinate_index;
682        block.bind_source(self, logical_source);
683        measurements.record_arrangement_nanos(arrangement_started.elapsed().as_nanos())?;
684        Ok(true)
685    }

687    pub(super) fn selected_row_replay(
688        &self,
689    ) -> Result<SelectedRowReplay, BoundObservationSourceError> {
690        let rows_per_block = self.content_plan.rows_per_block();
691        let available_bytes = rows_per_block
692            .checked_mul(crate::SelectedObservationRow::STORAGE_BYTES_PER_ROW)
693            .ok_or(BoundObservationSourceError::MeasurementOverflow)?;
694        let plan = MsReadPlan::new(
695            self.measurement_set.row_count(),
696            MsSelectionIoBudget {
697                available_bytes,
698                maximum_live_blocks: 1,
699                requested_bytes_per_row: crate::SelectedObservationRow::STORAGE_BYTES_PER_ROW,
700                storage_alignment_rows: Some(rows_per_block),
701            },
702        )
703        .map_err(|error| MsError::InvalidInput(error.to_string()))?;
704        Ok(SelectedRowReplay {
705            cursor: self.measurement_set.main_row_selection_cursor(plan)?,
706            pending: None,
707            manifest: Some(SelectedRowsBuilder::with_data_description_capacity(
708                u64::try_from(self.measurement_set.row_count())
709                    .map_err(|_| BoundObservationSourceError::PhysicalRowIndexOverflow)?,
710                self.coordinates.len(),
711            )),
712            terminal_checked: false,
713        })
714    }

716    fn fill_selected_row_group(
717        &self,
718        logical_source: &MeasurementSetReadAccess,
719        replay: &mut SelectedRowReplay,
720        physical_rows: &mut Vec<usize>,
721    ) -> Result<Option<usize>, BoundObservationSourceError> {
722        let Some(first) = replay.next_selected(self, logical_source)? else {
723            return Ok(None);
724        };
725        let coordinate_index = self
726            .coordinates
727            .iter()
728            .position(|coordinates| {
729                coordinates.data_description.data_description_id() == first.data_description_id()
730            })
731            .ok_or(
732                BoundObservationSourceError::DataDescriptionCoordinateMismatch {
733                    data_description_id: first.data_description_id(),
734                },
735            )?;
736        physical_rows.clear();
737        physical_rows.push(
738            usize::try_from(first.physical_row())
739                .map_err(|_| BoundObservationSourceError::PhysicalRowIndexOverflow)?,
740        );
741        while physical_rows.len() < self.content_plan.rows_per_block() {
742            let Some(row) = replay.next_selected(self, logical_source)? else {
743                break;
744            };
745            if row.data_description_id() != first.data_description_id() {
746                replay.pending = Some(row);
747                break;
748            }
749            physical_rows.push(
750                usize::try_from(row.physical_row())
751                    .map_err(|_| BoundObservationSourceError::PhysicalRowIndexOverflow)?,
752            );
753        }
754        Ok(Some(coordinate_index))
755    }

758pub(super) struct SelectedRowReplay {
759    cursor: MainRowSelectionCursor,
760    pending: Option<SelectedMainRow>,
761    manifest: Option<SelectedRowsBuilder>,
762    terminal_checked: bool,

765impl SelectedRowReplay {
766    fn next_selected(
767        &mut self,
768        source: &BoundObservationSource,
769        logical_source: &MeasurementSetReadAccess,
770    ) -> Result<Option<SelectedMainRow>, BoundObservationSourceError> {
771        if let Some(row) = self.pending.take() {
772            return Ok(Some(row));
773        }
774        loop {
775            let Some(fact) = self.cursor.next(&source.measurement_set)? else {
776                if self.terminal_checked {
777                    return Ok(None);
778                }
779                self.terminal_checked = true;
780                if !source.source_row_count_matches {
781                    return Err(BoundObservationSourceError::SourceRowCountMismatch);
782                }
783                let observed = self
784                    .manifest
785                    .take()
786                    .expect("selected-row manifest finishes once")
787                    .finish();
788                if &observed != logical_source.selection().rows() {
789                    return Err(BoundObservationSourceError::StaleSelectedRows);
790                }
791                return Ok(None);
792            };
793            if !source.row_predicate.matches(StoredMainRow::from(fact)) {
794                continue;
795            }
796            let row = SelectedMainRow::new(
797                u64::try_from(fact.physical_row())
798                    .map_err(|_| BoundObservationSourceError::PhysicalRowIndexOverflow)?,
799                u32::try_from(fact.data_description_id()).map_err(|_| {
800                    BoundObservationSourceError::DataDescriptionCoordinateMismatch {
801                        data_description_id: u32::MAX,
802                    }
803                })?,
804            );
805            self.manifest
806                .as_mut()
807                .expect("selected-row manifest remains active before terminal")
808                .push(row)
809                .map_err(|_| BoundObservationSourceError::StaleSelectedRows)?;
810            return Ok(Some(row));
811        }
812    }

848impl BoundObservationSamples<'_> {
849    pub(super) fn next_projected(
850        &mut self,
851    ) -> Option<Result<ProjectedSelectedObservationSample, BoundObservationSourceError>> {
852        if self.finished {
853            return None;
854        }
855        loop {
856            if let Some(block) = &self.active_block {
857                let coordinate_index = block.coordinate_index;
858                let coordinates = &self.source.coordinates[coordinate_index];
859                if self.row_offset < block.buffer.row_count() {
860                    let channel_index = coordinates.channels[self.channel_ordinal].channel_index;
861                    let product = coordinates.products[self.correlation_ordinal];
862                    let channel_offset = usize::try_from(channel_index)
863                        .ok()
864                        .and_then(|channel| channel.checked_sub(coordinates.channel_start));
865                    let correlation_offset = usize::try_from(product.correlation_index()).ok();
866                    let stored = match (channel_offset, correlation_offset) {
867                        (Some(channel), Some(correlation)) => {
868                            block.buffer.sample(channel, self.row_offset, correlation)
869                        }
870                        _ => None,
871                    };
872                    let Some(stored) = stored else {
873                        self.finished = true;
874                        return Some(Err(BoundObservationSourceError::StoredSampleShapeMismatch));
875                    };
876                    let input_weight_group = match selected_input_weight_group(
877                        &block.buffer,
878                        coordinates,
879                        channel_offset,
880                        self.row_offset,
881                    ) {
882                        Some(group) => group
883                            .with_density_owner(self.correlation_ordinal == 0)
884                            .with_terminal_member(
885                                self.correlation_ordinal + 1 == coordinates.products.len(),
886                            ),
887                        None => {
888                            self.finished = true;
889                            return Some(Err(
890                                BoundObservationSourceError::StoredSampleShapeMismatch,
891                            ));
892                        }
893                    };
894                    let parallel_hand_group_flag =
895                        if product.correlation_type().contributes_to_stokes_i() {
896                            let mut flagged = false;
897                            for peer in
898                                coordinates.products.iter().copied().filter(|peer| {
899                                    peer.correlation_type().contributes_to_stokes_i()
900                                })
901                            {
902                                let peer_offset = usize::try_from(peer.correlation_index()).ok();
903                                let peer_stored = match (channel_offset, peer_offset) {
904                                    (Some(channel), Some(correlation)) => {
905                                        block.buffer.sample(channel, self.row_offset, correlation)
906                                    }
907                                    _ => None,
908                                };
909                                let Some(peer_stored) = peer_stored else {
910                                    self.finished = true;
911                                    return Some(Err(
912                                        BoundObservationSourceError::StoredSampleShapeMismatch,
913                                    ));
914                                };
915                                flagged |= peer_stored.channel_flag();
916                            }
917                            flagged
918                        } else {
919                            stored.channel_flag()
920                        };
921                    let sample = self.project_sample(
922                        coordinates,
923                        coordinates.channels[self.channel_ordinal],
924                        product,
925                        stored,
926                        parallel_hand_group_flag,
927                        &block.row_geometry[self.row_offset],
928                    );
929                    self.advance(coordinates.channels.len(), coordinates.products.len());
930                    if sample.is_err() {
931                        self.finished = true;
932                    }
933                    return Some(sample.map(|selected| ProjectedSelectedObservationSample {
934                        selected,
935                        input_weight_group,
936                    }));
937                }
938            }
939            self.recycle_active_block();
940            self.prefetch_to_capacity();
941            if let Some(block) = self.ready_blocks.pop_front() {
942                self.active_block = Some(block);
943                self.row_offset = 0;
944                self.channel_ordinal = 0;
945                self.correlation_ordinal = 0;
946                if let Err(error) = self.record_live_block_high_water() {
947                    self.finished = true;
948                    return Some(Err(error));
949                }
950                continue;
951            }
952            if let Some(error) = self.pending_error.take() {
953                self.finished = true;
954                return Some(Err(error));
955            }
956            if let Err(error) = self
957                .source
958                .geometry_engine
959                .verify_selected_observation_measures()
960            {
961                self.finished = true;
962                return Some(Err(error.into()));
963            }
964            self.finished = true;
965            return None;
966        }
967    }

969    fn advance(&mut self, channel_count: usize, product_count: usize) {
970        self.correlation_ordinal += 1;
971        if self.correlation_ordinal == product_count {
972            self.correlation_ordinal = 0;
973            self.channel_ordinal += 1;
974            if self.channel_ordinal == channel_count {
975                self.channel_ordinal = 0;
976                self.row_offset += 1;
977            }
978        }
979    }
980
981    fn recycle_active_block(&mut self) {
982        let Some(mut block) = self.active_block.take() else {
983            return;
984        };
985        if self.source_exhausted || self.pending_error.is_some() {
986            return;
987        }
988        debug_assert!(block.slot < self.source.content_plan.maximum_live_blocks());
989        match self.fill_next_block(&mut block) {
990            Ok(true) => {
991                self.ready_blocks.push_back(block);
992                if let Err(error) = self.record_live_block_high_water() {
993                    self.source_exhausted = true;
994                    self.pending_error = Some(error);
995                }
996            }
997            Ok(false) => self.source_exhausted = true,
998            Err(error) => {
999                self.source_exhausted = true;
1000                self.pending_error = Some(error);
1001            }
1002        }
1003    }
1004
1005    fn prefetch_to_capacity(&mut self) {
1006        let maximum_live_blocks = self.source.content_plan.maximum_live_blocks();
1007        while !self.source_exhausted
1008            && self.pending_error.is_none()
1009            && self.active_block.is_some() as usize + self.ready_blocks.len() < maximum_live_blocks
1010        {
1011            debug_assert!(self.next_buffer_slot < maximum_live_blocks);
1012            let mut block = BufferedObservationBlock::new(
1013                self.next_buffer_slot,
1014                self.source.content_plan.rows_per_block(),
1015            );
1016            self.next_buffer_slot += 1;
1017            match self.fill_next_block(&mut block) {
1018                Ok(true) => {
1019                    self.ready_blocks.push_back(block);
1020                    if let Err(error) = self.record_live_block_high_water() {
1021                        self.source_exhausted = true;
1022                        self.pending_error = Some(error);
1023                    }
1024                }
1025                Ok(false) => self.source_exhausted = true,
1026                Err(error) => {
1027                    self.source_exhausted = true;
1028                    self.pending_error = Some(error);
1029                }
1030            }
1031        }
1032    }

1034    fn record_live_block_high_water(&mut self) -> Result<(), BoundObservationSourceError> {
1035        self.live_block_high_water = self
1036            .live_block_high_water
1037            .max(self.active_block.is_some() as usize + self.ready_blocks.len());
1038        let mut current_bytes = 0_u64;
1039        let mut capacity_bytes = 0_u64;
1040        for block in self.active_block.iter().chain(self.ready_blocks.iter()) {
1041            current_bytes = current_bytes
1042                .checked_add(block.current_bytes()?)
1043                .ok_or(BoundObservationSourceError::MeasurementOverflow)?;
1044            capacity_bytes = capacity_bytes
1045                .checked_add(block.capacity_bytes()?)
1046                .ok_or(BoundObservationSourceError::MeasurementOverflow)?;
1047        }
1048        self.measurements.borrow_mut().record_live_blocks(
1049            u64::try_from(self.active_block.is_some() as usize + self.ready_blocks.len())
1050                .map_err(|_| BoundObservationSourceError::MeasurementOverflow)?,
1051            current_bytes,
1052            capacity_bytes,
1053        );
1054        Ok(())
1055    }

1448    pub(super) fn visit_selected_samples<E>(
1449        &self,
1450        problem: &CompiledProblem,
1451        correlations: &mut Vec<SelectedObservationRunCorrelation>,
1452        mut consume: impl FnMut(
1453            &SelectedObservationRunRow,
1454            SelectedObservationRunChannel,
1455            &[SelectedObservationRunCorrelation],
1456            &MsCalEngine,
1457        ) -> Result<(), E>,
1458    ) -> Result<(), BlockVisitError<E>> {
1459        let logical_source = self.logical_source.as_ref().ok_or(BlockVisitError::Source(
1460            BoundObservationSourceError::StoredSampleShapeMismatch,
1461        ))?;
1462        let coordinates = self
1463            .coordinates
1464            .as_ref()
1465            .and_then(|coordinates| coordinates.get(self.coordinate_index))
1466            .ok_or(BlockVisitError::Source(
1467                BoundObservationSourceError::StoredSampleShapeMismatch,
1468            ))?;
1469        let geometry_engine = self
1470            .geometry_engine
1471            .as_deref()
1472            .ok_or(BlockVisitError::Source(
1473                BoundObservationSourceError::StoredSampleShapeMismatch,
1474            ))?;
1475        for row in 0..self.buffer.row_count() {
1476            let stored_row = self
1477                .buffer
1478                .sample(0, row, 0)
1479                .ok_or(BlockVisitError::Source(
1480                    BoundObservationSourceError::StoredSampleShapeMismatch,
1481                ))?;
1482            let run_row = project_stored_run_row(
1483                logical_source,
1484                problem,
1485                geometry_engine,
1486                coordinates,
1487                stored_row,
1488                &self.row_geometry[row],
1489            )
1490            .map_err(BlockVisitError::Source)?;
1491            for channel in coordinates.channels.iter().copied() {
1492                let run_channel = project_run_channel(channel);
1493                correlations.clear();
1494                if correlations.capacity() < coordinates.products.len() {
1495                    return Err(BlockVisitError::Source(
1496                        BoundObservationSourceError::StoredSampleShapeMismatch,
1497                    ));
1498                }
1499                let channel_offset = usize::try_from(channel.channel_index)
1500                    .ok()
1501                    .and_then(|channel| channel.checked_sub(coordinates.channel_start));
1502                let stokes_i_group_flag = coordinates
1503                    .products
1504                    .iter()
1505                    .copied()
1506                    .filter(|peer| peer.correlation_type().contributes_to_stokes_i())
1507                    .try_fold(false, |flagged, peer| {
1508                        let correlation = usize::try_from(peer.correlation_index()).ok();
1509                        let peer = match (channel_offset, correlation) {
1510                            (Some(channel), Some(correlation)) => {
1511                                self.buffer.sample(channel, row, correlation)
1512                            }
1513                            _ => None,
1514                        }
1515                        .ok_or(BlockVisitError::Source(
1516                            BoundObservationSourceError::StoredSampleShapeMismatch,
1517                        ))?;
1518                        Ok::<_, BlockVisitError<E>>(flagged || peer.channel_flag())
1519                    })?;
1520                for product in coordinates.products.iter().copied() {
1521                    let correlation_offset = usize::try_from(product.correlation_index()).ok();
1522                    let stored = match (channel_offset, correlation_offset) {
1523                        (Some(channel), Some(correlation)) => {
1524                            self.buffer.sample(channel, row, correlation)
1525                        }
1526                        _ => None,
1527                    }
1528                    .ok_or(BlockVisitError::Source(
1529                        BoundObservationSourceError::StoredSampleShapeMismatch,
1530                    ))?;
1531                    let parallel_hand_group_flag =
1532                        if product.correlation_type().contributes_to_stokes_i() {
1533                            stokes_i_group_flag
1534                        } else {
1535                            stored.channel_flag()
1536                        };
1537                    correlations.push(project_stored_run_correlation(
1538                        product,
1539                        stored,
1540                        parallel_hand_group_flag,
1541                    ));
1542                }
1543                consume(
1544                    &run_row,
1545                    run_channel,
1546                    correlations.as_slice(),
1547                    geometry_engine,
1548                )
1549                .map_err(BlockVisitError::Consumer)?;
1550            }
1551        }
1552        Ok(())
1553    }

1555    fn current_bytes(&self) -> Result<u64, BoundObservationSourceError> {
1556        let buffer = self
1557            .buffer
1558            .retained_current_bytes()
1559            .ok_or(BoundObservationSourceError::MeasurementOverflow)?;
1560        let geometry = self
1561            .row_geometry
1562            .len()
1563            .checked_mul(size_of::<EvaluatedRowGeometry>())
1564            .ok_or(BoundObservationSourceError::MeasurementOverflow)?;
1565        let projection_payload = self
1566            .row_geometry
1567            .iter()
1568            .try_fold(0_usize, |bytes, geometry| {
1569                geometry
1570                    .domain_projections
1571                    .retained_heap_bytes()
1572                    .and_then(|payload| bytes.checked_add(payload))
1573            })
1574            .ok_or(BoundObservationSourceError::MeasurementOverflow)?;
1575        let request = self
1576            .request_rows
1577            .len()
1578            .checked_mul(size_of::<usize>())
1579            .ok_or(BoundObservationSourceError::MeasurementOverflow)?;
1580        u64::try_from(
1581            buffer
1582                .checked_add(geometry)
1583                .and_then(|bytes| bytes.checked_add(projection_payload))
1584                .and_then(|bytes| bytes.checked_add(request))
1585                .ok_or(BoundObservationSourceError::MeasurementOverflow)?,
1586        )
1587        .map_err(|_| BoundObservationSourceError::MeasurementOverflow)
1588    }

1600        let projection_payload = self
1601            .row_geometry
1602            .iter()
1603            .try_fold(0_usize, |bytes, geometry| {
1604                geometry
1605                    .domain_projections
1606                    .retained_heap_bytes()
1607                    .and_then(|payload| bytes.checked_add(payload))

1617                .checked_add(geometry)

1622        .map_err(|_| BoundObservationSourceError::MeasurementOverflow)

1803fn selected_coordinates(
1804    measurement_set: &MeasurementSet,
1805    selection: &ObservationSelection,
1806) -> Result<Box<[SelectedCoordinates]>, BoundObservationSourceError> {
1807    let spectral_windows = measurement_set.spectral_window()?;
1808    let polarizations = measurement_set.polarization()?;
1809    let mut coordinates = Vec::with_capacity(selection.data_descriptions().len());
1810    for data_description in selection.data_descriptions().iter().copied() {
1811        let spectral_window = selection
1812            .spectral_windows()
1813            .iter()
1814            .find(|candidate| {
1815                candidate.spectral_window_id() == data_description.spectral_window_id()
1816            })
1817            .expect("compiled DATA_DESCRIPTION has one spectral-window selection");
1818        let spw_row = usize::try_from(data_description.spectral_window_id()).map_err(|_| {
1819            BoundObservationSourceError::SpectralCoordinateMismatch {
1820                spectral_window_id: data_description.spectral_window_id(),
1821            }
1822        })?;
1823        let Some(casa_types::ArrayValue::Float64(centres)) = spectral_windows
1824            .table()
1825            .column_accessor("CHAN_FREQ")
1826            .map_err(MsError::from)?
1827            .array_cells_owned_uncached(&[spw_row])
1828            .map_err(MsError::from)?
1829            .pop()
1830            .flatten()
1831        else {
1832            return Err(BoundObservationSourceError::SpectralCoordinateMismatch {
1833                spectral_window_id: data_description.spectral_window_id(),
1834            });
1835        };
1836        let Some(casa_types::ArrayValue::Float64(widths)) = spectral_windows
1837            .table()
1838            .column_accessor("CHAN_WIDTH")
1839            .map_err(MsError::from)?
1840            .array_cells_owned_uncached(&[spw_row])
1841            .map_err(MsError::from)?
1842            .pop()
1843            .flatten()
1844        else {
1845            return Err(BoundObservationSourceError::SpectralCoordinateMismatch {
1846                spectral_window_id: data_description.spectral_window_id(),
1847            });
1848        };
1849        if let Some(catalog) = spectral_window.coordinate_catalog()
1850            && (catalog.channel_count() != centres.len()
1851                || widths.first().is_none_or(|width| {
1852                    width.to_bits() != catalog.first_channel_width_hz().to_bits()
1853                })
1854                || catalog
1855                    .channel_frequencies_hz()
1856                    .iter()
1857                    .copied()
1858                    .zip(centres.iter().copied())
1859                    .any(|(expected, actual)| expected.to_bits() != actual.to_bits()))
1860        {
1861            return Err(BoundObservationSourceError::SpectralCoordinateMismatch {
1862                spectral_window_id: data_description.spectral_window_id(),
1863            });
1864        }
1865        let frame = frequency_frame(selected_i32_scalar(
1866            spectral_windows.table(),
1867            "MEAS_FREQ_REF",
1868            spw_row,
1869        )?)?;
1870        let mut channels = Vec::with_capacity(spectral_window.channel_indices().len());
1871        for &channel_index in spectral_window.channel_indices() {
1872            let channel = usize::try_from(channel_index).map_err(|_| {
1873                BoundObservationSourceError::SpectralCoordinateMismatch {
1874                    spectral_window_id: data_description.spectral_window_id(),
1875                }
1876            })?;
1877            let centre_hz = *centres.get(channel).ok_or(
1878                BoundObservationSourceError::SpectralCoordinateMismatch {
1879                    spectral_window_id: data_description.spectral_window_id(),
1880                },
1881            )?;
1882            let width_hz = *widths.get(channel).ok_or(
1883                BoundObservationSourceError::SpectralCoordinateMismatch {
1884                    spectral_window_id: data_description.spectral_window_id(),
1885                },
1886            )?;
1887            let first_edge = centre_hz - 0.5 * width_hz;
1888            let second_edge = centre_hz + 0.5 * width_hz;
1889            if ![centre_hz, width_hz, first_edge, second_edge]
1890                .into_iter()
1891                .all(f64::is_finite)
1892            {
1893                return Err(BoundObservationSourceError::SpectralCoordinateMismatch {
1894                    spectral_window_id: data_description.spectral_window_id(),
1895                });
1896            }
1897            channels.push(SelectedChannel {
1898                channel_index,
1899                centre_hz,
1900                lower_hz: first_edge.min(second_edge),
1901                upper_hz: first_edge.max(second_edge),
1902                width_hz,
1903                frame,
1904            });
1905        }
1906        let channel_start = channels
1907            .first()
1908            .and_then(|channel| usize::try_from(channel.channel_index).ok())
1909            .expect("compiled spectral-window selection is nonempty");
1910        let channel_end = channels
1911            .last()
1912            .and_then(|channel| usize::try_from(channel.channel_index).ok())
1913            .and_then(|channel| channel.checked_add(1))
1914            .ok_or(BoundObservationSourceError::SpectralCoordinateMismatch {
1915                spectral_window_id: data_description.spectral_window_id(),
1916            })?;
1917        drop(centres);
1918        drop(widths);
1919        let polarization = selection
1920            .correlations()
1921            .iter()
1922            .find(|candidate| candidate.polarization_id() == data_description.polarization_id())
1923            .expect("compiled DATA_DESCRIPTION has one polarization selection");
1924        let polarization_row =
1925            usize::try_from(data_description.polarization_id()).map_err(|_| {
1926                BoundObservationSourceError::CorrelationCoordinateMismatch {
1927                    polarization_id: data_description.polarization_id(),
1928                }
1929            })?;
1930        let Some(casa_types::ArrayValue::Int32(stored_products)) = polarizations
1931            .table()
1932            .column_accessor("CORR_TYPE")
1933            .map_err(MsError::from)?
1934            .array_cells_owned_uncached(&[polarization_row])
1935            .map_err(MsError::from)?
1936            .pop()
1937            .flatten()
1938        else {
1939            return Err(BoundObservationSourceError::CorrelationCoordinateMismatch {
1940                polarization_id: data_description.polarization_id(),
1941            });
1942        };
1943        for product in polarization.products() {
1944            let product_index = usize::try_from(product.correlation_index()).map_err(|_| {
1945                BoundObservationSourceError::CorrelationCoordinateMismatch {
1946                    polarization_id: data_description.polarization_id(),
1947                }
1948            })?;
1949            if stored_products
1950                .get(product_index)
1951                .and_then(|code| correlation_type(*code))
1952                != Some(product.correlation_type())
1953            {
1954                return Err(BoundObservationSourceError::CorrelationCoordinateMismatch {
1955                    polarization_id: data_description.polarization_id(),
1956                });
1957            }
1958        }
1959        validate_input_weight_group(polarization.products(), data_description.polarization_id())?;
1960        coordinates.push(SelectedCoordinates {
1961            data_description,
1962            channels: channels.into_boxed_slice(),
1963            products: polarization.products().into(),
1964            channel_start,
1965            channel_count: channel_end - channel_start,
1966        });
1967    }
1968    Ok(coordinates.into_boxed_slice())
1969}

1971pub(super) fn validate_input_weight_group(
1972    products: &[CorrelationProduct],
1973    polarization_id: u32,
1974) -> Result<(), BoundObservationSourceError> {
1975    if products.len() == 1 {
1976        return Ok(());
1977    }
1978    let valid = match (products.first(), products.last()) {
1979        (Some(first), Some(last))
1980            if first.correlation_type() == CorrelationType::CircularRr
1981                && last.correlation_type() == CorrelationType::CircularLl =>
1982        {
1983            matches_canonical_correlation_order(
1984                products,
1985                &[
1986                    CorrelationType::CircularRr,
1987                    CorrelationType::CircularRl,
1988                    CorrelationType::CircularLr,
1989                    CorrelationType::CircularLl,
1990                ],
1991            )
1992        }
1993        (Some(first), Some(last))
1994            if first.correlation_type() == CorrelationType::LinearXx
1995                && last.correlation_type() == CorrelationType::LinearYy =>
1996        {
1997            matches_canonical_correlation_order(
1998                products,
1999                &[
2000                    CorrelationType::LinearXx,
2001                    CorrelationType::LinearXy,
2002                    CorrelationType::LinearYx,
2003                    CorrelationType::LinearYy,
2004                ],
2005            )
2006        }
2007        _ => false,
2008    };
2009    if valid {
2010        Ok(())
2011    } else {
2012        Err(
2013            BoundObservationSourceError::UnsupportedImagingWeightCorrelationGroup {
2014                polarization_id,
2015            },
2016        )
2017    }
2018}

2022    canonical: &[CorrelationType; 4],
2023) -> bool {

2025    products.iter().all(|product| {
2026        remaining
2027            .by_ref()
2028            .any(|candidate| *candidate == product.correlation_type())
2029    })

2161                    field_id,

2165                    uvw_law: problem.geometry().uvw(),

2218                    .get(&stored.antenna2())

2255    stored: SelectedStoredSample,
2256    observation_direction: SkyDirection,
2257    request: PhaseCentreProjectionRequest,
2258) -> Result<SelectedPhaseCentreProjection, BoundObservationSourceError> {

2260        stored.time_mjd_seconds(),
2261        [
2262            request.target_direction.longitude_rad(),
2263            request.target_direction.latitude_rad(),
2264        ],
2265        direction_ref(request.target_direction.frame()),
2266    )?;
2267    if let Some([longitude, latitude]) = request.moving_direction_shift {
2268        target_angles[0] += longitude;
2269        target_angles[1] += latitude;
2270    }
2271    let target_j2000 = SkyDirection::new(DirectionFrame::J2000, target_angles[0], target_angles[1]);
2272    let (transformed_uvw_m, phase_shift_m) = if matches!(
2273        request.uvw_law,
2274        casa_imaging_model::UvwCoordinateLaw::MosaicPhaseTrackingCentre
2275    ) {
2276        if request.project_to_observation_plane {
2277            return Err(BoundObservationSourceError::InvalidRowGeometry);
2278        }

2285            .geometry_engine
2286            .reproject_raw_uvw_for_mosaic_to_direction(stored.uvw_m(), request.field_id, &target)?;

2288    } else if target_j2000 == observation_direction && !request.project_to_observation_plane {
2289        (stored.uvw_m(), 0.0)
2290    } else if request.project_to_observation_plane {
2291        source
2292            .geometry_engine
2293            .reproject_raw_uvw_for_faceted_gridft_between_j2000_directions(
2294                stored.uvw_m(),
2295                [
2296                    observation_direction.longitude_rad(),
2297                    observation_direction.latitude_rad(),
2298                ],
2299                target_angles,
2300            )?
2301    } else {
2302        source
2303            .geometry_engine
2304            .reproject_raw_uvw_for_gridft_between_j2000_directions(
2305                stored.uvw_m(),
2306                [
2307                    observation_direction.longitude_rad(),
2308                    observation_direction.latitude_rad(),
2309                ],
2310                target_angles,
2311            )?
2312    };
2313    SelectedPhaseCentreProjection::new(transformed_uvw_m, phase_shift_m)
2314        .ok_or(BoundObservationSourceError::InvalidRowGeometry)
2315}

2353    }

2355        .pointing_catalog
2356        .as_ref()
2357        .ok_or(BoundObservationSourceError::MissingPointingQueryDomain)?

2371                antenna_brackets[1],
2372                queries[2 * row + 1],
2373                *law,
2374                fallback,

2376        });
2377    }

2379}

2455        .map(aw_direction_coordinate)

2457    let [grouping_arcsec, refresh_arcsec] = contract.pointing_offset_sigdev_arcsec();
2458    let grouping_arcsec = if grouping_arcsec > 0.0 {
2459        grouping_arcsec
2460    } else {
2461        1.0e-3
2462    };
2463    let mut chart_pixels_by_epoch = BTreeMap::new();
2464    let mut cached_field = None;
2465    let mut cached_row_count = 0_usize;
2466    let mut cached_antenna_pixels = BTreeMap::<i32, [f64; 2]>::new();
2467    let mut cached_grouped = AwPointingChartPixels::from([]);
2468    let primary_increment = problem
2469        .geometry()
2470        .domains()
2471        .first()
2472        .ok_or(BoundObservationSourceError::InvalidRowGeometry)?
2473        .direction()
2474        .increment_rad();
2475    let primary_increment_norm = primary_increment[0].hypot(primary_increment[1]);
2476    let refresh_threshold_rad = refresh_arcsec.to_radians() / 3600.0;
2477
2478    for epoch in epochs {
2479        let time = f64::from_bits(epoch.key.time_bits);
2480        let fallback = pointing_epoch_phase_direction(source, problem, epoch.key.field_id, time)?;
2481        let queries = epoch
2482            .antennas
2483            .iter()
2484            .copied()
2485            .map(|antenna| PointingDirectionQuery::new(antenna, time))
2486            .collect::<Result<Vec<_>, _>>()?;
2487        let brackets = catalog.direction_brackets(&source.geometry_engine, &queries)?;
2488        let directions = epoch
2489            .antennas
2490            .iter()
2491            .copied()
2492            .zip(brackets)
2493            .zip(queries.iter().copied())
2494            .map(|((antenna, bracket), query)| {
2495                resolve_pointing_direction(bracket, query, law, fallback)
2496                    .map(|direction| (antenna, direction))
2497            })
2498            .collect::<Result<BTreeMap<_, _>, _>>()?;
2499        let pixels_by_chart = charts
2500            .iter()
2501            .map(|coordinate| {
2502                directions
2503                    .iter()
2504                    .map(|(&antenna, direction)| {
2505                        coordinate
2506                            .to_pixel(&[direction.longitude_rad(), direction.latitude_rad()])
2507                            .map_err(|_| BoundObservationSourceError::InvalidRowGeometry)
2508                            .and_then(|pixel| {
2509                                pixel
2510                                    .get(0..2)
2511                                    .filter(|pixel| pixel.iter().all(|value| value.is_finite()))
2512                                    .map(|pixel| (antenna, [pixel[0], pixel[1]]))
2513                                    .ok_or(BoundObservationSourceError::InvalidRowGeometry)
2514                            })
2515                    })
2516                    .collect::<Result<BTreeMap<_, _>, _>>()
2517            })
2518            .collect::<Result<Vec<_>, _>>()?;
2519        let current_primary = pixels_by_chart
2520            .first()
2521            .ok_or(BoundObservationSourceError::InvalidRowGeometry)?;
2522        let same_antennas = cached_antenna_pixels.len() == current_primary.len()
2523            && cached_antenna_pixels.keys().eq(current_primary.keys());
2524        let refresh = if cached_field != Some(epoch.key.field_id)
2525            || cached_row_count != epoch.row_count
2526            || !same_antennas
2527        {
2528            true
2529        } else {
2530            let residual = cached_antenna_pixels.iter().zip(current_primary).fold(
2531                [0.0, 0.0],
2532                |[sum_x, sum_y], ((_, cached), (_, current))| {
2533                    [
2534                        sum_x + cached[0] - current[0],
2535                        sum_y + cached[1] - current[1],
2536                    ]
2537                },
2538            );
2539            residual[0].hypot(residual[1]) / current_primary.len().max(1) as f64
2540                * primary_increment_norm
2541                >= refresh_threshold_rad
2542        };
2543        if refresh {
2544            let grouped =
2545                pixels_by_chart
2546                    .iter()
2547                    .zip(
2548                        problem.geometry().domains().iter().flat_map(|domain| {
2549                            domain.facets().iter().map(|facet| facet.direction())
2550                        }),
2551                    )
2552                    .map(|(pixels, direction)| {
2553                        casa_aw_grouped_pixels(pixels, grouping_arcsec, direction.increment_rad())
2554                    })
2555                    .collect::<Result<Vec<_>, _>>()?;
2556            cached_grouped = Arc::from(grouped);
2557        }
2558        cached_field = Some(epoch.key.field_id);
2559        cached_row_count = epoch.row_count;
2560        cached_antenna_pixels = current_primary.clone();
2561        if chart_pixels_by_epoch
2562            .insert(epoch.key, Arc::clone(&cached_grouped))
2563            .is_some()
2564        {
2565            return Err(BoundObservationSourceError::InvalidRowGeometry);
2566        }
2567    }
2568    let retained_byte_ceiling = size_of::<AwPointingEpochPlan>()
2569        .checked_add(
2570            chart_pixels_by_epoch
2571                .values()
2572                .try_fold(0_usize, |bytes, charts| {
2573                    charts.iter().try_fold(bytes, |bytes, chart| {
2574                        chart
2575                            .len()
2576                            .checked_mul(usize::BITS as usize)
2577                            .and_then(|entries| bytes.checked_add(entries))
2578                    })
2579                })
2580                .ok_or(BoundObservationSourceError::MeasurementOverflow)?,
2581        )
2582        .ok_or(BoundObservationSourceError::MeasurementOverflow)?;
2583    Ok(AwPointingEpochPlan {
2584        chart_pixels_by_epoch,
2585        retained_byte_ceiling,
2586    })
2587}

2609fn casa_aw_grouped_pixels(
2610    pixels: &BTreeMap<i32, [f64; 2]>,
2611    grouping_arcsec: f64,
2612    increment_rad: [f64; 2],
2613) -> Result<BTreeMap<i32, [f64; 2]>, BoundObservationSourceError> {
2614    let bin_width =
2615        grouping_arcsec.to_radians() / 3600.0 / increment_rad[0].hypot(increment_rad[1]);
2616    if !(bin_width.is_finite() && bin_width > 0.0) || pixels.is_empty() {
2617        return Err(BoundObservationSourceError::InvalidRowGeometry);
2618    }
2619    let min_x = pixels
2620        .values()
2621        .map(|pixel| pixel[0])
2622        .fold(f64::INFINITY, f64::min);
2623    let min_y = pixels
2624        .values()
2625        .map(|pixel| pixel[1])
2626        .fold(f64::INFINITY, f64::min);
2627    let mut bins = BTreeMap::<i32, (i64, i64)>::new();
2628    let mut sums = BTreeMap::<(i64, i64), (f32, f32, usize)>::new();
2629    for (&antenna, &pixel) in pixels {
2630        let bin = (
2631            ((pixel[0] - min_x) / bin_width).floor() as i64,
2632            ((pixel[1] - min_y) / bin_width).floor() as i64,
2633        );
2634        bins.insert(antenna, bin);
2635        let sum = sums.entry(bin).or_insert((0.0, 0.0, 0));
2636        sum.0 = (f64::from(sum.0) + pixel[0]) as f32;
2637        sum.1 = (f64::from(sum.1) + pixel[1]) as f32;
2638        sum.2 = sum
2639            .2
2640            .checked_add(1)
2641            .ok_or(BoundObservationSourceError::MeasurementOverflow)?;
2642    }
2643    let means = sums
2644        .into_iter()
2645        .map(|(bin, (x, y, count))| (bin, [(x / count as f32) as f64, (y / count as f32) as f64]))
2646        .collect::<BTreeMap<_, _>>();
2647    Ok(bins
2648        .into_iter()
2649        .map(|(antenna, bin)| (antenna, means[&bin]))
2650        .collect())
2651}

2715fn resolve_pointing_direction(
2716    bracket: PointingDirectionBracket,
2717    query: PointingDirectionQuery,
2718    law: casa_imaging_model::ObservationPointingLaw,
2719    fallback: SkyDirection,
2720) -> Result<SkyDirection, BoundObservationSourceError> {
2721    if let Some(covering) = bracket.covering() {
2722        return Ok(j2000_direction(covering.direction_j2000_rad()));
2723    }
2724    let direction = match (bracket.before(), bracket.after()) {
2725        (Some(before), Some(after)) if before.row_index() == after.row_index() => {
2726            before.direction_j2000_rad()
2727        }
2728        (Some(before), Some(after)) => match law.interpolation() {
2729            PointingInterpolation::Nearest => {
2730                let before_distance = query.time_mjd_seconds() - before.row_time_mjd_seconds();
2731                let after_distance = after.row_time_mjd_seconds() - query.time_mjd_seconds();
2732                if before_distance <= after_distance {
2733                    before.direction_j2000_rad()
2734                } else {
2735                    after.direction_j2000_rad()
2736                }
2737            }
2738            PointingInterpolation::GreatCircleShortestArc => interpolate_direction(
2739                before.direction_j2000_rad(),
2740                after.direction_j2000_rad(),
2741                (query.time_mjd_seconds() - before.row_time_mjd_seconds())
2742                    / (after.row_time_mjd_seconds() - before.row_time_mjd_seconds()),
2743            )?,
2744        },
2745        (Some(endpoint), None) | (None, Some(endpoint))
2746            if matches!(law.extrapolation(), PointingExtrapolation::HoldNearest) =>
2747        {
2748            endpoint.direction_j2000_rad()
2749        }
2750        (Some(_), None) | (None, Some(_)) => {
2751            return Err(BoundObservationSourceError::PointingOutsideCoverage {
2752                antenna_id: query.antenna_id(),
2753                time_mjd_seconds: query.time_mjd_seconds(),
2754            });
2755        }
2756        (None, None) if matches!(law.missing(), MissingPointingPolicy::UsePhaseTrackingCentre) => {
2757            return Ok(fallback);
2758        }
2759        (None, None) => {
2760            return Err(BoundObservationSourceError::MissingPointingDirection {
2761                antenna_id: query.antenna_id(),
2762                time_mjd_seconds: query.time_mjd_seconds(),
2763            });
2764        }
2765    };
2766    Ok(j2000_direction(direction))
2767}

2773fn interpolate_direction(
2774    before: [f64; 2],
2775    after: [f64; 2],
2776    fraction: f64,
2777) -> Result<[f64; 2], BoundObservationSourceError> {
2778    if !(fraction.is_finite() && (0.0..=1.0).contains(&fraction)) {
2779        return Err(BoundObservationSourceError::InvalidPointingInterpolation);
2780    }
2781    let left = unit_direction(before);
2782    let right = unit_direction(after);
2783    let cosine = left
2784        .iter()
2785        .zip(right)
2786        .map(|(left, right)| left * right)
2787        .sum::<f64>()
2788        .clamp(-1.0, 1.0);
2789    let angle = cosine.acos();
2790    let vector = if angle.abs() < 1.0e-15 {
2791        left
2792    } else {
2793        let denominator = angle.sin();
2794        let before_weight = ((1.0 - fraction) * angle).sin() / denominator;
2795        let after_weight = (fraction * angle).sin() / denominator;
2796        [
2797            before_weight * left[0] + after_weight * right[0],
2798            before_weight * left[1] + after_weight * right[1],
2799            before_weight * left[2] + after_weight * right[2],
2800        ]
2801    };
2802    let norm = vector.iter().map(|value| value * value).sum::<f64>().sqrt();
2803    if !(norm.is_finite() && norm > 0.0) {
2804        return Err(BoundObservationSourceError::InvalidPointingInterpolation);
2805    }
2806    let [x, y, z] = vector.map(|value| value / norm);
2807    Ok([y.atan2(x).rem_euclid(std::f64::consts::TAU), z.asin()])
2808}

2939fn validate_selected_read_state(
2940    expected_identity: casa_imaging_model::MeasurementSetIdentity,
2941    expected_rows: &casa_imaging_model::SelectedRows,
2942    expected_generations: &casa_imaging_model::SourceGenerations,
2943    current: &ObservationSourceState,
2944) -> Result<(), BoundObservationSourceError> {

2946        return Err(BoundObservationSourceError::CurrentSourceIdentityMismatch);

2949        return Err(BoundObservationSourceError::StaleSelectedRows);
2950    }

2953    let corrected_data_changed =
2954        current_generations.corrected_data_column() != expected_generations.corrected_data_column();
2955    let consistency_changed =
2956        current_generations.consistency_token() != expected_generations.consistency_token();

2958    // selected-observation reads. Their owner update may complete between
2959    // scientific traversal and publication without invalidating the read set.
2960    if current_generations.columns() != expected_generations.columns()
2961        || current_generations.metadata_generations() != expected_generations.metadata_generations()
2962        || (model_changed || corrected_data_changed) != consistency_changed
2963    {
2964        return Err(BoundObservationSourceError::StaleSourceGenerations);
2965    }
2966    Ok(())
2967}
