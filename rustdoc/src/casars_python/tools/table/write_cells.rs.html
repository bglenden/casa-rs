2//! The `table` tool's cell and column writes (`putcell`, `putcol`,
3//! `putvarcol`), as CASA 6.7.6.14's `tb` does them
4//! (`tools/casa-compat/table_tool_oracle.py`, `cells_write` and
5//! `ms_cells_write`).
6//!
7//! A value is coerced to the column's declared type ([`super::cell_value`]),
8//! checked against the column's shape, set in memory, and the cells written
9//! in place in the column's storage manager with
10//! `TableWritePlan::save_selected_rows`; no other cell and no other storage
11//! manager is touched. The lock handling is [`State::write_with`]'s.
12

27fn runtime<T>(message: String) -> Result<T, Fail> {
28    Err(Fail::Runtime(message))

31pub(super) fn put_column_error(what: &str) -> Fail {
32    Fail::Runtime(format!("TableProxy::TableProxy::putColumn: {what}"))
33}

35/// What a column holds: its type (`None`: records) and whether its cells are
36/// arrays.
37fn describe(table: &Table, column: &str) -> Result<(Option<PrimitiveType>, bool), Fail> {
38    let schema: &ColumnSchema = table
39        .schema()
40        .and_then(|schema| schema.column(column))
41        .ok_or_else(|| put_column_error(&format!("column {column} does not exist")))?;
42    Ok((
43        schema.data_type(),
44        matches!(schema.column_type(), ColumnType::Array(_)),
45    ))
46}

48fn shape_text(shape: &[usize]) -> String {
49    let parts: Vec<String> = shape.iter().map(ToString::to_string).collect();
50    format!("[{}]", parts.join(", "))
51}

53/// The refusal of an array whose shape the column's cell cannot take.
54pub(super) fn conformance(row: usize, column: &str, shape: &[usize], expected: &[usize]) -> Fail {
55    Fail::Runtime(format!(
56        "ArrayColumn::put for row {row} in column {column}: Table array conformance error \
57         (shape={}, expected {})",
58        shape_text(shape),
59        shape_text(expected)
60    ))
61}

63/// Whether the column's storage manager holds one shape per cell once set
64/// (the tiled managers do; the indirect ones let a cell be reshaped).
65fn keeps_cell_shapes(table: &Table, column: &str) -> bool {
66    table.data_manager_info().iter().any(|manager| {
67        manager.dm_type.contains("Tiled") && manager.columns.iter().any(|c| c == column)
68    })
69}

71/// What a column asks of the shape of the arrays written to it, worked out
72/// once for all the cells of a call.
73pub(super) struct ShapeRule {
74    /// The one shape every cell has.
75    fixed: Option<Vec<usize>>,
76    /// Whether a cell keeps the shape it has once set (a tiled manager).
77    kept: bool,
78}
79
80impl ShapeRule {
81    pub(super) fn of(table: &Table, column: &str) -> Self {
82        let contract = table
83            .schema()
84            .and_then(|schema| schema.column(column))
85            .map(ColumnSchema::column_type);
86        Self {
87            fixed: match contract {
88                Some(ColumnType::Array(ArrayShapeContract::Fixed { shape })) => Some(shape.clone()),
89                _ => None,
90            },
91            kept: keeps_cell_shapes(table, column),
92        }
93    }
94
95    /// Checks an array cell's shape against its column's.
96    pub(super) fn check(
97        &self,
98        table: &Table,
99        column: &str,
100        row: usize,
101        cell: &Value,
102    ) -> Result<(), Fail> {
103        let Value::Array(array) = cell else {
104            return Ok(());
105        };
106        let shape = array.shape();
107        if let Some(fixed) = &self.fixed
108            && shape != fixed.as_slice()
109        {
110            return Err(conformance(row, column, shape, fixed));
111        }
112        if self.kept
113            && let Some(existing) = table.array_shape(row, column)?
114            && existing != shape
115        {
116            return Err(conformance(row, column, shape, &existing));
117        }
118        Ok(())
119    }
120}

122/// Sets cells of one column in memory: each cell is checked, and none is
123/// set before every one has been (a refused call changes nothing). A cell
124/// of a scalar or array column is set where it is, without loading the
125/// rest of the table (a table is read lazily, and a MeasurementSet's visibilities
126/// would not fit in memory); a record cell goes through the row store.
127pub(super) fn set_cells(
128    table: &mut Table,
129    column: &str,
130    cells: Vec<(usize, Value)>,
131) -> Result<(), Fail> {
132    let record_column = matches!(
133        table
134            .schema()
135            .and_then(|schema| schema.column(column))
136            .map(ColumnSchema::column_type),
137        Some(ColumnType::Record)
138    );
139    let rule = ShapeRule::of(table, column);
140    for (row, cell) in &cells {
141        rule.check(table, column, *row, cell)?;
142    }
143    if record_column
144        || cells
145            .iter()
146            .any(|(_, cell)| matches!(cell, Value::Record(_)))
147    {
148        let mut accessor = table.column_accessor_mut(column)?;
149        for (row, cell) in cells {
150            match cell {
151                Value::Record(record) => accessor.set_record(row, record)?,
152                other => accessor.set(row, other)?,
153            }
154        }
155        return Ok(());
156    }
157    let mut writer = table.row_accessor_mut().prepare(&[column])?;
158    for (row, cell) in cells {
159        writer.seek(row)?;
160        writer.set_value_at(0, cell)?;
161    }
162    writer.flush()?;
163    Ok(())
164}

166/// The distinct rows of `rows`, in order, as a save of selected rows wants
167/// them.
168fn distinct(rows: &[usize]) -> Vec<usize> {
169    let mut seen = std::collections::HashSet::with_capacity(rows.len());
170    rows.iter()
171        .copied()
172        .filter(|row| seen.insert(*row))
173        .collect()
174}
175
176/// The elements of `data` at `indices`.
177fn take(data: &Data, indices: impl Iterator<Item = usize>) -> Data {
178    match data {
179        Data::Bools(values) => Data::Bools(indices.map(|i| values[i]).collect()),
180        Data::Ints(values) => Data::Ints(indices.map(|i| values[i]).collect()),
181        Data::Floats(values) => Data::Floats(indices.map(|i| values[i]).collect()),
182        Data::Complexes(values) => Data::Complexes(indices.map(|i| values[i]).collect()),
183        Data::Texts(values) => Data::Texts(indices.map(|i| values[i].clone()).collect()),
184    }
185}

187/// A column's values split into one value per row: the rows are the last
188/// axis of an array column's values (NumPy's, in C order), the only axis of
189/// a scalar column's.
190pub(super) fn split_rows(
191    wide: Wide,
192    array: bool,
193    column: &str,
194    wanted: usize,
195) -> Result<Vec<Wide>, Fail> {
196    let shape = wide.shape.clone().unwrap_or_else(|| vec![1]);
197    let (cell_shape, nrows) = if array {
198        match shape.split_last() {
199            Some((rows, cell)) if !cell.is_empty() => (Some(cell.to_vec()), *rows),
200            _ => (None, usize::MAX),
201        }
202    } else {
203        (
204            None,
205            if shape.len() == 1 {
206                shape[0]
207            } else {
208                usize::MAX
209            },
210        )
211    };
212    if nrows != wanted {
213        return runtime(if array {
214            format!(
215                "ArrayColumn::putColumn - column {column} has {wanted}, array has {} rows: \
216                 Table array conformance error",
217                if nrows == usize::MAX { 0 } else { nrows }
218            )
219        } else {
220            "ScalarColumn::putColumn(Vector&): Table conformance error (#rows mismatch)".into()
221        });
222    }
223    let per_row: usize = cell_shape.as_ref().map_or(1, |cell| cell.iter().product());
224    Ok((0..nrows)
225        .map(|row| {
226            let indices = (0..per_row).map(|element| element * nrows + row);
227            Wide {
228                shape: cell_shape.clone(),
229                data: take(&wide.data, indices),
230            }
231        })
232        .collect())
233}

235pub(super) enum ColumnInput {
236    Normal(Incoming),
237    PackedComplex {
238        shape: [usize; 3],
239        values: Vec<Complex32>,
240    },
241}
242
243impl ColumnInput {
244    pub(super) fn from_python(py: Python<'_>, value: Option<&Bound<'_, PyAny>>) -> PyResult<Self> {
245        if let Some(value) = value
246            && let Ok(array) = value.cast::<PyArray3<Complex32>>()
247            && value.get_type().is(py.get_type::<PyArray3<Complex32>>())
248            && array.dtype().is_native_byteorder() == Some(true)
249            && array.is_fortran_contiguous()
250            && let Ok(array) = array.try_readonly()
251            && let Ok(values) = array.as_slice()
252        {
253            return Ok(Self::PackedComplex {
254                shape: array.shape().try_into().expect("rank checked"),
255                // Detach from NumPy storage before releasing the GIL.
256                values: values.to_vec(),
257            });
258        }
259        super::keyword_value(py, value).map(Self::Normal)
260    }
261
262    fn into_incoming(self) -> Incoming {
263        match self {
264            Self::Normal(value) => value,
265            Self::PackedComplex {
266                shape: [width, channels, rows],
267                values,
268            } => {
269                let reversed = ndarray::ArrayView3::from_shape((rows, channels, width), &values)
270                    .expect("shape of the original NumPy array");
271                Incoming::Wide(Wide {
272                    shape: Some(vec![width, channels, rows]),
273                    data: Data::Complexes(
274                        reversed
275                            .reversed_axes()
276                            .iter()
277                            .map(|value| Complex64::new(f64::from(value.re), f64::from(value.im)))
278                            .collect(),
279                    ),
280                })
281            }
282        }
283    }
284}

286impl State {
287    /// `putcell(columnname, rownr, thevalue)`: one row number, or several,
288    /// each given the value. A row that does not exist refuses the call
289    /// whole (CASA writes the rows before it, then fails).
290    pub(super) fn putcell(
291        &mut self,
292        column: String,
293        rows: Vec<i64>,
294        value: Incoming,
295        log: &mut Vec<LogLine>,
296    ) -> Reply {
297        if let Some(open) = self.open.as_ref()
298            && !open.column_allowed(&column)
299        {
300            return Err(put_column_error(&format!("column {column} does not exist")));
301        }
302        self.write_with(Target::Cells, log, move |table, row_map| {
303            let (ty, array) = describe(table, &column)?;
304            let nrows = row_map.map_or_else(|| table.row_count(), <[usize]>::len);
305            let rows = rows
306                .into_iter()
307                .map(|row| {
308                    usize::try_from(row)
309                        .ok()
310                        .filter(|row| *row < nrows)
311                        .map(|row| row_map.map_or(row, |map| map[row]))
312                        .ok_or_else(|| put_column_error("no such row"))
313                })
314                .collect::<Result<Vec<_>, _>>()?;
315            if rows.is_empty() {
316                return Ok(Persist::Nothing);
317            }
318            let cell = value.into_cell(ty, array)?;
319            let rows = distinct(&rows);
320            set_cells(
321                table,
322                &column,
323                rows.iter().map(|row| (*row, cell.clone())).collect(),
324            )?;
325            Ok(Persist::Rows {
326                columns: vec![column],
327                rows,
328            })
329        })
330    }

332    /// `putcol(columnname, value, startrow, nrow, rowincr)`.
333    pub(super) fn putcol(
334        &mut self,
335        column: String,
336        value: ColumnInput,
337        selection: (i64, i64, i64),
338        log: &mut Vec<LogLine>,
339    ) -> Reply {
340        if let Some(open) = self.open.as_ref()
341            && !open.column_allowed(&column)
342        {
343            return Err(put_column_error(&format!("column {column} does not exist")));
344        }
345        self.write_with(Target::Cells, log, move |table, row_map| {
346            let (ty, array) = describe(table, &column)?;
347            let nrows = row_map.map_or_else(|| table.row_count(), <[usize]>::len);
348            let selected = rows(
349                nrows,
350                selection.0,
351                selection.1,
352                selection.2,
353                "TableProxy::TableProxy::putColumn: no such row",
354            )?;
355            let selected: Vec<usize> = selected
356                .into_iter()
357                .map(|row| row_map.map_or(row, |map| map[row]))
358                .collect();
359            if let ColumnInput::PackedComplex { shape, values } = &value
360                && array
361                && ty == Some(PrimitiveType::Complex32)
362                && row_map.is_none()
363                && selection.2 == 1
364                && shape[2] == selected.len()
365                && let (Some(first), Some(last)) = (selected.first(), selected.last())
366                && table.prepare_write().try_save_packed_complex_rows(
367                    &column,
368                    *first..*last + 1,
369                    [shape[0], shape[1]],
370                    values,
371                )?
372            {
373                return Ok(Persist::Done);
374            }
375            let cells = match (value.into_incoming(), ty) {
376                (Incoming::Wide(wide), Some(_)) => {
377                    split_rows(wide, array, &column, selected.len())?
378                }
379                (value, _) => {
380                    // A record column is written a cell at a time with
381                    // `putcell`; `putcol` of one is not possible.
382                    let _ = value;
383                    return runtime(
384                        "TableProxy::putColumn not possible for a column containing records".into(),
385                    );
386                }
387            };
388            let cells = selected
389                .iter()
390                .zip(cells)
391                .map(|(row, cell)| Ok((*row, cell.into_cell(ty.expect("typed column"), array)?)))
392                .collect::<Result<Vec<_>, Fail>>()?;
393            if cells.is_empty() {
394                return Ok(Persist::Nothing);
395            }
396            set_cells(table, &column, cells)?;
397            Ok(Persist::Rows {
398                columns: vec![column],
399                rows: selected,
400            })
401        })
402    }
403
404    /// `putvarcol(columnname, value, startrow, nrow, rowincr)`: a record
405    /// whose fields, in the order they have, are the values of the selected
406    /// rows, each with a trailing axis of one, as `getvarcol` returns them.
407    /// The field names are labels (`getvarcol` names them `r1`, `r2`, ...),
408    /// not row numbers.
409    pub(super) fn putvarcol(
410        &mut self,
411        column: String,
412        fields: Vec<(String, Incoming)>,
413        selection: (i64, i64, i64),
414        log: &mut Vec<LogLine>,
415    ) -> Reply {
416        if let Some(open) = self.open.as_ref()
417            && !open.column_allowed(&column)
418        {
419            return Err(put_column_error(&format!("column {column} does not exist")));
420        }
421        self.write_with(Target::Cells, log, move |table, row_map| {
422            let (ty, array) = describe(table, &column)?;
423            let nrows = row_map.map_or_else(|| table.row_count(), <[usize]>::len);
424            let selected = rows_stepping(
425                nrows,
426                selection.0,
427                selection.1,
428                selection.2,
429                "TableProxy::TableProxy::putVarColumn: no such row",
430                "TableProxy::putVarColumn: rowincr<=0",
431            )?;
432            let selected: Vec<usize> = selected
433                .into_iter()
434                .map(|row| row_map.map_or(row, |map| map[row]))
435                .collect();
436            if fields.len() != selected.len() {
437                return runtime("TableProxy::putVarColumn: #rows mismatches #elem in value".into());
438            }
439            let Some(ty) = ty else {
440                return runtime(
441                    "TableProxy::putVarColumn not possible for a column containing records".into(),
442                );
443            };
444            let mut cells = Vec::with_capacity(fields.len());
445            for ((key, value), row) in fields.into_iter().zip(&selected) {
446                let Incoming::Wide(wide) = value else {
447                    return runtime(format!("a value of {key} is not an array"));
448                };
449                let mut shape = wide.shape.clone().unwrap_or_else(|| vec![1]);
450                if array && shape.len() < 2 || shape.pop() != Some(1) {
451                    return runtime(format!(
452                        "Invalid Table operation: ArrayColumn::setShape: mismatch in #dim of array of column {column}"
453                    ));
454                }
455                let cell = Wide {
456                    shape: array.then_some(shape),
457                    data: wide.data,
458                };
459                cells.push((*row, cell.into_cell(ty, array)?));
460            }
461            if cells.is_empty() {
462                return Ok(Persist::Nothing);
463            }
464            set_cells(table, &column, cells)?;
465            Ok(Persist::Rows {
466                columns: vec![column],
467                rows: selected,
468            })
469        })
470    }
