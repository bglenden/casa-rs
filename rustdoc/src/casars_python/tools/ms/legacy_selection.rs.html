31#[derive(Default)]
32pub(super) struct LegacySelection {
33    // Indices are relative to the working MS, which can differ from the
34    // currently selected view after an exact-time record selection.
35    rows: Option<Arc<[usize]>>,
36    initial: Option<InitialSelection>,
37    pub(super) output: Option<Box<AttachedMs>>,
38    pub(super) inactive_iterator: Option<Box<AttachedMs>>,
39    pub(super) axes: LegacyAxes,
40    pub(super) baselines: Option<Vec<(i32, i32)>>,
41    pub(super) iteration: Option<LegacyIteration>,
42    query_parent: LegacyQueryParent,
43    positioned: bool,
44}
45
46#[derive(Default)]
47enum LegacyQueryParent {
48    #[default]
49    Attachment,
50    Named(Arc<Vec<PathBuf>>),
51    Transient,
52}
53
54enum PreparedLegacyQuery {
55    Disk(Box<StagedLegacyQuery>),
56    Memory(Box<AttachedMs>),
57}

59impl LegacySelection {
60    pub(super) fn rows(&self) -> Option<&[usize]> {
61        self.rows.as_deref()
62    }
63
64    pub(super) fn shared_rows(&self) -> Option<Arc<[usize]>> {
65        self.rows.clone()
66    }
67
68    pub(super) fn positioned_iteration(&self) -> Option<&LegacyIteration> {
69        self.iteration.as_ref().filter(|_| self.positioned)
70    }
71
72    pub(super) fn set_iterator_rows(&mut self, rows: Option<Arc<[usize]>>, positioned: bool) {
73        if self.rows != rows || self.positioned != positioned {
74            self.axes.baseline_read = None;
75        }
76        self.rows = rows;
77        self.positioned = positioned;
78    }
79
80    pub(super) fn channel_count(&self) -> usize {
81        self.initial.as_ref().map_or(0, |initial| initial.channels)
82    }
83
84    pub(super) fn descriptions(&self) -> &[i32] {
85        self.initial
86            .as_ref()
87            .map_or(&[], |initial| &initial.descriptions)
88    }
89
90    pub(super) fn query_parent<'a>(&'a self, attachment: &'a Path) -> Option<&'a Path> {
91        match &self.query_parent {
92            LegacyQueryParent::Attachment => Some(attachment),
93            LegacyQueryParent::Named(paths) => paths.first().map(PathBuf::as_path),
94            LegacyQueryParent::Transient => None,
95        }
96    }
97}

108    #[pyo3(signature = (selected = false), text_signature = "($self, selected=False)")]
109    fn nrowold(
110        &self,
111        py: Python<'_>,
112        #[pyo3(from_py_with = strict_bool)] selected: bool,
113    ) -> PyResult<usize> {
114        deprecation(py, "nrowold", "nrow")?;
115        self.attachment_info(py, |attached| {
116            let Some(ms) = attached else { return Ok(0) };
117            if selected {
118                let view = ms.legacy_view();
119                if let Some(rows) = &view.legacy_selection.rows {
120                    return Ok(rows.len());
121                }
122                return view
123                    .working_metadata_link()
124                    .row_count()
125                    .map_err(runtime_error);
126            }
127            ms.working_metadata_link()
128                .row_count()
129                .map_err(runtime_error)
130        })
131    }

133    #[pyo3(signature = (datadescid = 0, reset = false), text_signature = "($self, datadescid=0, reset=False)")]
134    fn selectinitold(
135        &self,
136        py: Python<'_>,
137        #[pyo3(from_py_with = selectinit_id)] datadescid: i64,
138        #[pyo3(from_py_with = strict_bool)] reset: bool,
139    ) -> PyResult<bool> {
140        deprecation(py, "selectinitold", "selectinit")?;
141        py.detach(|| {
142            let mut attached = self.attached.lock().map_err(runtime_error)?;
143            let Some(ms) = attached.as_mut() else {
144                return Ok(false);
145            };
146            if reset {
147                ms.replace_legacy_output(None);
148                ms.legacy_selection = LegacySelection {
149                    iteration: ms.legacy_selection.iteration.take(),
150                    inactive_iterator: ms.legacy_selection.inactive_iterator.take(),
151                    ..LegacySelection::default()
152                };
153                return Ok(true);
154            }
155            let id = datadescid as i32;
156            if id < 0 {
157                return Err(runtime_error(
158                    "The data description id must be a list of positive integers",
159                ));
160            }
161            ms.replace_legacy_output(None);
162            ms.initialize_legacy_selection(Some(id))
163        })
164    }

200impl AttachedMs {
201    pub(super) fn legacy_view(&self) -> &Self {
202        self.legacy_selection.output.as_deref().unwrap_or(self)
203    }
204
205    pub(super) fn legacy_view_mut(&mut self) -> &mut Self {
206        if self.legacy_selection.output.is_some() {
207            return self.legacy_selection.output.as_deref_mut().unwrap();
208        }
209        self
210    }
211
212    pub(super) fn legacy_view_for_filter(&mut self) -> &mut Self {
213        if self
214            .legacy_selection
215            .output
216            .as_ref()
217            .is_some_and(|view| view.legacy_selection.initial.is_none())
218        {
219            self.replace_legacy_output(None);
220        }
221        self.legacy_view_mut()
222    }
223
224    pub(super) fn replace_legacy_output(&mut self, output: Option<Box<Self>>) {
225        let previous = std::mem::replace(&mut self.legacy_selection.output, output);
226        if self.legacy_selection.output.is_none()
227            && let Some(view) = previous.as_ref()
228        {
229            self.legacy_selection.axes = view.legacy_selection.axes.without_baseline_read();
230            self.legacy_selection.baselines = view.legacy_selection.baselines.clone();
231        }
232        if previous
233            .as_ref()
234            .is_some_and(|view| view.legacy_selection.iteration.is_some())
235        {
236            self.legacy_selection.inactive_iterator = previous;
237        }
238    }
239
240    pub(super) fn restore_legacy_iterator_view(&mut self, mut source: Box<Self>) {
241        let current = &self.legacy_view().legacy_selection;
242        source.legacy_selection.axes = current.axes.without_baseline_read();
243        source.legacy_selection.baselines = current.baselines.clone();
244        if self.legacy_selection.initial.is_none() {
245            source.legacy_selection.initial = None;
246        }
247        self.replace_legacy_output(Some(source));
248    }

250    pub(super) fn initialize_legacy_selection(&mut self, requested: Option<i32>) -> PyResult<bool> {
251        if let Some(id) = requested
252            && let Some(initial) = &self.legacy_selection.initial
253            && initial.descriptions == [id]
254        {
255            self.legacy_selection.query_parent = if initial.rows.is_some() {
256                LegacyQueryParent::Transient
257            } else {
258                LegacyQueryParent::Attachment
259            };
260            self.legacy_selection
261                .set_iterator_rows(initial.rows.clone(), false);
262            return Ok(true);
263        }
264        let working = self.working_metadata_link();
265        let working_rows = working.selected_rows().map_err(runtime_error)?;
266        let refreshed = self.refresh_legacy_main()?;
267        let table = refreshed.as_ref().unwrap_or_else(|| self.legacy_main());
268        let uncached = super::subtable_queries::Subtables::default();
269        let subtables = if refreshed.is_some() {
270            &uncached
271        } else {
272            &self.subtables
273        };
274        let links = [
275            SubtableId::DataDescription,
276            SubtableId::SpectralWindow,
277            SubtableId::Polarization,
278        ];
279        let subtables = LegacySubtables::read(subtables, self.memory_ms.as_ref(), table, &links)
280            .map_err(runtime_error)?;
281        let descriptions = subtables
282            .tables(SubtableId::DataDescription)
283            .map_err(runtime_error)?;
284        let description_count = descriptions.iter().map(Table::row_count).sum::<usize>();
285        let requested = requested.filter(|_| description_count > 1);
286        let columns = match working_rows {
287            Some(rows) => table.required_scalar_columns_owned_for_rows(&["DATA_DESC_ID"], &rows),
288            None => table.required_scalar_columns_owned(&["DATA_DESC_ID"]),
289        }
290        .map_err(runtime_error)?;
291        let RequiredScalarColumnValues::Int32(description_by_row) = &columns["DATA_DESC_ID"] else {
292            return Err(runtime_error("DATA_DESC_ID must be an integer column"));
293        };
294        let count = description_by_row.len();
295        let mut ids = BTreeSet::new();
296        let mut rows = Vec::with_capacity(count);
297        for (relative, &id) in description_by_row.iter().enumerate() {
298            if requested.is_none_or(|requested| requested == id) {
299                ids.insert(id);
300                rows.push((relative, id));
301            }
302        }
303        let (same_shape, channels) = common_visibility_shape(
304            descriptions,
305            subtables
306                .tables(SubtableId::SpectralWindow)
307                .map_err(runtime_error)?,
308            subtables
309                .tables(SubtableId::Polarization)
310                .map_err(runtime_error)?,
311            &ids,
312        )?;
313        if !same_shape && let Some(first) = ids.first().copied() {
314            rows.retain(|&(_, id)| id == first);
315            ids.retain(|&id| id == first);
316        }
317        let initialized = !rows.is_empty();
318        let rows = (rows.len() != count)
319            .then(|| Arc::<[usize]>::from_iter(rows.into_iter().map(|(row, _)| row)));
320        self.legacy_selection.initial = initialized.then(|| InitialSelection {
321            descriptions: ids.into_iter().collect(),
322            channels,
323            rows: rows.clone(),
324        });
325        self.legacy_selection.query_parent = if rows.is_some() {
326            LegacyQueryParent::Transient
327        } else {
328            LegacyQueryParent::Attachment
329        };
330        self.legacy_selection.set_iterator_rows(rows, false);
331        self.legacy_selection.axes = LegacyAxes::default();
332        self.legacy_selection.baselines = None;
333        Ok(same_shape)
334    }

336    fn refresh_legacy_main(&self) -> PyResult<Option<Table>> {
337        if self.memory_ms.is_some() {
338            return Ok(None);
339        }
340        self.working_metadata_link()
341            .verify_directory()
342            .map_err(runtime_error)?;
343        if self.main_stamp.is_some() && self.main_files.same_descriptor(self.main_stamp) {
344            return Ok(None);
345        }
346        Table::open(TableOptions::new(&self.path))
347            .map(Some)
348            .map_err(runtime_error)
349    }

351    pub(super) fn legacy_correlation_layout(&self) -> PyResult<(usize, Vec<i32>)> {
352        let refreshed = self.refresh_legacy_main()?;
353        let table = refreshed.as_ref().unwrap_or_else(|| self.legacy_main());
354        let uncached = super::subtable_queries::Subtables::default();
355        let cache = if refreshed.is_some() {
356            &uncached
357        } else {
358            &self.subtables
359        };
360        let ids = [SubtableId::DataDescription, SubtableId::Polarization];
361        let linked = LegacySubtables::read(cache, self.memory_ms.as_ref(), table, &ids)
362            .map_err(runtime_error)?;
363        let descriptions = linked
364            .tables(SubtableId::DataDescription)
365            .map_err(runtime_error)?;
366        let polarization = linked
367            .tables(SubtableId::Polarization)
368            .map_err(runtime_error)?;
369        let count = descriptions.iter().map(Table::row_count).sum::<usize>();
370        let id = self
371            .legacy_selection
372            .descriptions()
373            .first()
374            .copied()
375            .ok_or_else(|| runtime_error("no initialized legacy data description"))?;
376        let (table, row) =
377            linked_subtable_row(descriptions, id as usize, SubtableId::DataDescription)
378                .map_err(runtime_error)?;
379        let pol = MsDataDescription::new(table)
380            .polarization_id(row)
381            .map_err(runtime_error)?;
382        let (table, row) = linked_subtable_row(
383            polarization,
384            usize::try_from(pol).map_err(runtime_error)?,
385            SubtableId::Polarization,
386        )
387        .map_err(runtime_error)?;
388        let codes = MsPolarization::new(table)
389            .corr_type(row)
390            .map_err(runtime_error)?;
391        Ok((count, codes))
392    }

394    pub(super) fn prepare_legacy_filter(&mut self) -> PyResult<bool> {
395        if self.legacy_selection.initial.is_none() {
396            self.initialize_legacy_selection(None)?;
397        }
398        Ok(self.legacy_selection.initial.is_some()
399            && self
400                .legacy_selection
401                .rows
402                .as_ref()
403                .is_none_or(|rows| !rows.is_empty()))
404    }

406    fn select_legacy_taql(&mut self, text: &str) -> PyResult<bool> {
407        let original_path = self.path.clone();
408        let view = self.legacy_view_for_filter();
409        if !view.prepare_legacy_filter()? || text.bytes().all(|character| character == b' ') {
410            return Ok(false);
411        }
412        let (selected, staged) = view.prepare_legacy_taql(text, &original_path)?;
413        let nonempty = !selected.is_empty();
414        let output = match staged {
415            Some(PreparedLegacyQuery::Disk(stage)) => Some(stage.publish(&mut self.write_locks)?),
416            Some(PreparedLegacyQuery::Memory(view)) => {
417                Some(PublishedLegacyQuery::Independent(view))
418            }
419            None => None,
420        };
421        let view = self.legacy_view_mut();
422        if let Some(PublishedLegacyQuery::Independent(mut output)) = output {
423            output.legacy_selection.initial =
424                view.legacy_selection.initial.clone().map(|mut initial| {
425                    initial.rows = None;
426                    initial
427                });
428            output.legacy_selection.axes = view.legacy_selection.axes.without_baseline_read();
429            output.legacy_selection.baselines = view.legacy_selection.baselines.clone();
430            if !nonempty {
431                output
432                    .legacy_selection
433                    .set_iterator_rows(Some(Arc::from([])), false);
434            }
435            self.replace_legacy_output(Some(output));
436        } else {
437            view.legacy_selection.query_parent = match output {
438                Some(PublishedLegacyQuery::Reference(paths)) => LegacyQueryParent::Named(paths),
439                _ => LegacyQueryParent::Transient,
440            };
441            view.legacy_selection
442                .set_iterator_rows(Some(selected), false);
443        }
444        Ok(nonempty)
445    }

447    fn prepare_legacy_taql(
448        &self,
449        text: &str,
450        original_path: &std::path::Path,
451    ) -> PyResult<(Arc<[usize]>, Option<PreparedLegacyQuery>)> {
452        let previous = self.legacy_selection.rows();
453        let working = self.working_metadata_link();
454        let outer = working.selected_rows().map_err(runtime_error)?;
455        let physical_rows = compose_rows(outer.as_deref(), previous);
456        let refreshed = self.refresh_legacy_main()?;
457        let table = refreshed.as_ref().unwrap_or_else(|| self.legacy_main());
458        let fast = select_projected_scalar_rows(table, text, physical_rows.as_deref())
459            .map_err(runtime_error)?;
460        let selection = if let Some(selection) = fast {
461            selection
462        } else if let Some(memory) = &self.memory_ms {
463            if let Some(rows) = previous {
464                let view = memory.to_memory_rows(rows).map_err(runtime_error)?;
465                select_legacy_rows(&view, text).map_err(runtime_error)?
466            } else {
467                select_legacy_rows(memory, text).map_err(runtime_error)?
468            }
469        } else {
470            let snapshot = working
471                .snapshot_relative_rows(previous)
472                .map_err(runtime_error)?;
473            select_legacy_rows(&snapshot.ms, text).map_err(runtime_error)?
474        };
475        let selected = selection
476            .rows
477            .into_iter()
478            .map(|row| previous.map_or(row, |previous| previous[row]))
479            .collect::<Arc<[usize]>>();
480        let staged = if let Some(giving) = selection.giving {
481            let options = QueryOutputOptions::parse(&giving)?;
482            let output = compose_rows(outer.as_deref(), Some(&selected)).expect("row projection");
483            working.verify_directory().map_err(runtime_error)?;
484            if options.memory {
485                QuerySubtable::validate_memory_output(
486                    &giving.table_name,
487                    self.legacy_selection.query_parent(&self.path),
488                    &self.path,
489                    self.memory_ms.as_ref(),
490                )?;
491                self.memory_ms
492                    .is_none()
493                    .then(|| {
494                        let mut memory = self.new_legacy_memory(&output, original_path)?;
495                        memory.legacy_selection.query_parent = LegacyQueryParent::Transient;
496                        Ok::<_, PyErr>(PreparedLegacyQuery::Memory(Box::new(memory)))
497                    })
498                    .transpose()?
499            } else {
500                self.stage_legacy_query(table, &output, &giving.table_name, options, original_path)?
501                    .map(|stage| PreparedLegacyQuery::Disk(Box::new(stage)))
502            }
503        } else {
504            None
505        };
506        Ok((selected, staged))
507    }

509    fn select_legacy_record(&mut self, record: RecordValue) -> PyResult<bool> {
510        if !self.legacy_view_for_filter().prepare_legacy_filter()? {
511            return Ok(false);
512        }
513        let mut fields = record.into_fields();
514        fields.sort_unstable_by(|left, right| left.name.cmp(&right.name));
515        for field in fields {
516            if field.name == "rows" {
517                self.select_explicit_legacy_rows(&field)?;
518            } else {
519                self.legacy_view_mut().apply_legacy_record_field(field)?;
520            }
521        }
522        Ok(self
523            .legacy_view()
524            .legacy_selection
525            .rows
526            .as_ref()
527            .is_none_or(|rows| !rows.is_empty()))
528    }

530    fn select_explicit_legacy_rows(&mut self, field: &RecordField) -> PyResult<()> {
531        let rows = integer_items_as(field, |value| value)?;
532        if rows.is_empty() {
533            return Ok(());
534        }
535        let available = self
536            .working_metadata_link()
537            .row_count()
538            .map_err(runtime_error)?;
539        let selected = rows
540            .into_iter()
541            .map(|row| {
542                usize::try_from(row)
543                    .ok()
544                    .filter(|&row| row < available)
545                    .ok_or_else(|| {
546                        runtime_error(format!("row {row} is outside the working MeasurementSet"))
547                    })
548            })
549            .collect::<PyResult<Arc<[usize]>>>()?;
550        self.replace_legacy_output(None);
551        self.legacy_selection.query_parent = LegacyQueryParent::Transient;
552        self.legacy_selection
553            .set_iterator_rows(Some(selected), false);
554        Ok(())
555    }

557    fn apply_legacy_record_field(&mut self, field: RecordField) -> PyResult<()> {
558        // MSSelectionKeywords looks up the legacy spelling before the
559        // C++ selector uppercases it to find the physical column.
560        if field.name.bytes().any(|byte| byte.is_ascii_uppercase()) {
561            return Ok(());
562        }
563        let label = field.name.to_ascii_uppercase();
564        let selector = RecordSelector::from_label(&label).expect("converted selection field");
565        let predicate = match selector {
566            RecordSelector::Rows => return Ok(()),
567            RecordSelector::ScalarIds => {
568                let ids = integer_items(&field)?;
569                (!ids.is_empty()).then(|| format!("{label} IN [{ids}]", ids = int_list(&ids)))
570            }
571            RecordSelector::IfrNumber => {
572                let mut ids = integer_items(&field)?;
573                if ids.is_empty() {
574                    self.legacy_selection.baselines = None;
575                    return Ok(());
576                }
577                let snapshot;
578                let source = if let Some(memory) = &self.memory_ms {
579                    memory
580                } else {
581                    (snapshot, _) = self
582                        .working_metadata_link()
583                        .source_and_rows()
584                        .map_err(runtime_error)?;
585                    &snapshot.ms
586                };
587                let antennas = source.antenna().map_err(runtime_error)?.row_count() as i32;
588                ids.retain(|&id| {
589                    (0..antennas).contains(&(id / 1000)) && (0..antennas).contains(&(id % 1000))
590                });
591                self.legacy_selection.baselines = (!ids.is_empty())
592                    .then(|| ids.iter().map(|id| (id / 1000, id % 1000)).collect());
593                Some(if ids.is_empty() {
594                    "False".into()
595                } else {
596                    format!(
597                        "(ANTENNA1 * 1000 + ANTENNA2) IN [{ids}]",
598                        ids = int_list(&ids)
599                    )
600                })
601            }
602            RecordSelector::Times => {
603                let values = real_items(&field)?;
604                if values.is_empty() {
605                    return Ok(());
606                }
607                let values = values
608                    .into_iter()
609                    .filter(|value| !value.is_nan())
610                    .map(real_literal)
611                    .collect::<Vec<_>>()
612                    .join(",");
613                Some(if values.is_empty() {
614                    "False".into()
615                } else {
616                    format!("TIME IN [{values}]")
617                })
618            }
619            range => legacy_range_predicate(range, &field)?,
620        };
621        if let Some(predicate) = predicate {
622            let (selected, _) = self.prepare_legacy_taql(&predicate, &self.path)?;
623            self.legacy_selection.query_parent = LegacyQueryParent::Transient;
624            self.legacy_selection
625                .set_iterator_rows(Some(selected), false);
626        }
627        Ok(())
628    }
629}

635fn real_literal(value: f64) -> String {
636    if value.is_infinite() {
637        if value.is_sign_positive() {
638            "1e309"
639        } else {
640            "-1e309"
641        }
642        .into()
643    } else {
644        format!("{value:?}")
645    }
646}
647
648fn legacy_range_predicate(
649    selector: RecordSelector,
650    field: &RecordField,
651) -> PyResult<Option<String>> {
652    let values = real_items(field)?;
653    let &[lower, upper] = values.as_slice() else {
654        return Ok(None);
655    };
656    if lower.is_nan() || upper.is_nan() {
657        return Ok(Some("False".into()));
658    }
659    let predicate = match selector {
660        RecordSelector::Time => format!(
661            "TIME >= {} && TIME <= {}",
662            real_literal(lower),
663            real_literal(upper)
664        ),
665        RecordSelector::Uvw(axis) => {
666            let column = format!("UVW[{axis}]");
667            format!(
668                "{column} >= {} && {column} <= {}",
669                real_literal(lower),
670                real_literal(upper)
671            )
672        }
673        RecordSelector::UvDistance => {
674            let lo = real_literal(lower * lower);
675            let hi = real_literal(upper * upper);
676            let distance = "(sqr(UVW[1]) + sqr(UVW[2]))";
677            format!(
678                "({distance} >= {lo} || near({distance}, {lo})) && ({distance} <= {hi} || near({distance}, {hi}))"
679            )
680        }
681        _ => unreachable!("record range field checked before dispatch"),
682    };
683    Ok(Some(predicate))
684}

686fn common_visibility_shape(
687    descriptions: &[Table],
688    spectral_windows: &[Table],
689    polarizations: &[Table],
690    ids: &BTreeSet<i32>,
691) -> PyResult<(bool, usize)> {
692    let mut first: Option<(usize, i32)> = None;
693    for &id in ids {
694        let (table, row) =
695            linked_subtable_row(descriptions, id as usize, SubtableId::DataDescription)
696                .map_err(runtime_error)?;
697        let description = MsDataDescription::new(table);
698        let spw_id = description.spectral_window_id(row).map_err(runtime_error)?;
699        let pol_id = description.polarization_id(row).map_err(runtime_error)?;
700        let (spw, spw_row) = linked_subtable_row(
701            spectral_windows,
702            spw_id as usize,
703            SubtableId::SpectralWindow,
704        )
705        .map_err(runtime_error)?;
706        let (pol, pol_row) =
707            linked_subtable_row(polarizations, pol_id as usize, SubtableId::Polarization)
708                .map_err(runtime_error)?;
709        let shape = (
710            usize::try_from(
711                MsSpectralWindow::new(spw)
712                    .num_chan(spw_row)
713                    .map_err(runtime_error)?,
714            )
715            .map_err(runtime_error)?,
716            MsPolarization::new(pol)
717                .num_corr(pol_row)
718                .map_err(runtime_error)?,
719        );
720        if let Some(prior) = first
721            && prior != shape
722        {
723            return Ok((false, prior.0));
724        }
725        first = Some(shape);
726    }
727    Ok((true, first.map_or(0, |shape| shape.0)))
728}
