20/// Checked owned-array and metadata demand for one planned continuum generation.
21///
22/// This is deliberately non-persistent. It describes the current algorithm
23/// catalog's owned payload, validity, and working arrays so the runtime can
24/// acquire one bounded lease before production starts. The generated summary's
25/// owned metadata and temporary beam arrays are included; allocator bookkeeping,
26/// the borrowed plan and borrowed reconstruction inputs are not charged here.
27#[derive(Debug, Clone, Copy, PartialEq, Eq)]
28pub struct ContinuumGenerationDemand {
29    storage_plan: ProductStoragePlan,
30    algorithm_scratch_bytes: u64,
31    retained_metadata_bytes: u64,
32    beam_scratch_bytes: u64,
33    maximum_member_payload_bytes: u64,
34    maximum_member_validity_bytes: u64,
35    maximum_window_payload_bytes: u64,
36    maximum_window_validity_bytes: u64,
37    peak_residency_bytes: u64,
38}
39
40impl ContinuumGenerationDemand {
41    /// Explicit selected window bound used by this generation.

44        self.storage_plan

46

51    }
52
53    /// Largest resident stored-mask window.
54    #[must_use]
55    pub const fn maximum_window_validity_bytes(self) -> u64 {

57    }
58
59    /// Maximum product-algorithm working-array bytes retained with production.

93    }

96impl PlannedContinuumGeneration {
97    /// Derive current-catalog array and metadata demand for these inputs.
98    ///
99    /// # Errors
100    ///
101    /// Fails closed when the inputs do not name this plan's problem and source
102    /// generations, or when any byte calculation overflows.
103    pub fn demand(
104        &self,
105        inputs: &ContinuumProductInputs<'_>,
106        storage_plan: ProductStoragePlan,
107    ) -> Result<ContinuumGenerationDemand, ProductsError> {
108        if inputs.problem().problem_id() != self.problem_id()
109            || inputs.problem().product_graph().graph_id() != self.graph_id()
110            || inputs.major_cycle_completion() != self.major_cycle_completion()
111            || inputs.normal_state_completion() != self.normal_state_completion()
112            || inputs.final_model().generation_id() != self.final_model_generation()
113            || inputs.reconstruction_mask_generation() != self.reconstruction_mask_generation()
114            || inputs
115                .coupled_reconstruction_masks()
116                .map(|masks| masks.line().generation_id())
117                != self.line_reconstruction_mask_generation()
118        {
119            return Err(ProductsError::SourceLineageMismatch);
120        }
121
122        let mut maximum_member_payload_bytes = 0_u64;
123        let mut maximum_member_validity_bytes = 0_u64;
124        let mut maximum_window_payload_bytes = 0_u64;
125        let mut maximum_window_validity_bytes = 0_u64;
126        let mut maximum_windows = 1;
127        for member in self.members() {
128            let values = checked_shape_values(member.shape())?;
129            if values != member.payload_values() {
130                return Err(ProductsError::PayloadLengthMismatch {
131                    expected: values,
132                    actual: member.payload_values(),
133                });
134            }
135            let payload = bytes_for::<f32>(values, "member payload")?;
136            let validity = bytes_for::<bool>(values, "member validity")?;
137            maximum_member_payload_bytes = maximum_member_payload_bytes.max(payload);
138            maximum_member_validity_bytes = maximum_member_validity_bytes.max(validity);
139            let layout = storage_plan.layout(member.axes())?;
140            maximum_windows = maximum_windows
141                .max(layout.shape()[layout.spectral_axis()].div_ceil(layout.maximum_channels()));
142            let window_values = layout.maximum_values();
143            maximum_window_payload_bytes = maximum_window_payload_bytes
144                .max(bytes_for::<f32>(window_values, "product window payload")?);
145            maximum_window_validity_bytes = maximum_window_validity_bytes
146                .max(bytes_for::<bool>(window_values, "product window validity")?);
147        }
148
149        let requires_beam = self
150            .members()
151            .iter()
152            .any(|member| member.beam_rule() != ProductBeamRule::None);
153        let beam_jobs = if requires_beam {
154            inputs
155                .normal_state()
156                .domain_count()
157                .checked_mul(inputs.normal_state().channel_count())
158                .and_then(|count| count.checked_mul(inputs.normal_state().polarization_count()))
159                .ok_or(ProductsError::ResourceDemandOverflow(
160                    "generation beam count",
161                ))?
162        } else {
163            0
164        };
165        let workers = match inputs.normal_state().catalog() {
166            NormalStateCatalog::UnnormalizedTaylorBlockV1
167            | NormalStateCatalog::UnnormalizedJointBlockV1 => 1,
168            _ => storage_plan
169                .maximum_workers()
170                .min(maximum_windows.max(beam_jobs)),
171        };
172        let storage_plan = ProductStoragePlan::new(storage_plan.maximum_channels(), workers)?;
173
174        let mut algorithm_scratch_bytes = match inputs.normal_state().catalog() {
175            NormalStateCatalog::UnnormalizedTaylorBlockV1 => taylor_scratch_bytes(inputs)?,
176            NormalStateCatalog::UnnormalizedJointBlockV1 => generic_scratch_bytes(self, inputs)?,
177            NormalStateCatalog::UnnormalizedPlaneV1
178            | NormalStateCatalog::UnnormalizedChannelSlabV1 => generic_scratch_bytes(self, inputs)?,
179        };
180        if matches!(
181            inputs.normal_state().catalog(),
182            NormalStateCatalog::UnnormalizedTaylorBlockV1
183                | NormalStateCatalog::UnnormalizedJointBlockV1
184        ) {
185            algorithm_scratch_bytes = checked_add(
186                algorithm_scratch_bytes,
187                checked_add(
188                    maximum_member_payload_bytes,
189                    maximum_member_validity_bytes,
190                    "coupled member arrays",
191                )?,
192                "coupled member production scratch",
193            )?;
194        }
195        algorithm_scratch_bytes = checked_add(
196            algorithm_scratch_bytes,
197            checked_add(
198                maximum_window_payload_bytes,
199                maximum_window_validity_bytes,
200                "active product window",
201            )?,
202            "windowed production scratch",
203        )?;
204        // Each lane can retain a completed window while another lane still owns
205        // its plane workspace. Beam fitting joins before these windows begin.
206        algorithm_scratch_bytes = algorithm_scratch_bytes.checked_mul(workers as u64).ok_or(
207            ProductsError::ResourceDemandOverflow("parallel product windows"),
208        )?;
209        if !matches!(
210            inputs.normal_state().catalog(),
211            NormalStateCatalog::UnnormalizedTaylorBlockV1
212                | NormalStateCatalog::UnnormalizedJointBlockV1
213        ) {
214            algorithm_scratch_bytes = checked_add(
215                algorithm_scratch_bytes,
216                bytes_for::<Option<crate::ProductWindow>>(workers, "product window slots")?,
217                "parallel product slots",
218            )?;
219        }
220        let (retained_metadata_bytes, member_beam_bytes, beam_scratch_bytes) =
221            self.metadata_demand(inputs, workers)?;
222        let transient_bytes = if matches!(
223            inputs.normal_state().catalog(),
224            NormalStateCatalog::UnnormalizedTaylorBlockV1
225                | NormalStateCatalog::UnnormalizedJointBlockV1
226        ) {
227            checked_add(
228                algorithm_scratch_bytes,
229                beam_scratch_bytes,
230                "coupled beam workspace",
231            )?
232        } else {
233            checked_add(
234                algorithm_scratch_bytes,
235                member_beam_bytes,
236                "active member preparation",
237            )?
238            .max(beam_scratch_bytes)
239        };
240        let peak_residency_bytes = checked_add(
241            retained_metadata_bytes,
242            transient_bytes,
243            "product generation residency",
244        )?;
245        Ok(ContinuumGenerationDemand {
246            storage_plan,
247            algorithm_scratch_bytes,
248            retained_metadata_bytes,
249            beam_scratch_bytes,
250            maximum_member_payload_bytes,
251            maximum_member_validity_bytes,
252            maximum_window_payload_bytes,
253            maximum_window_validity_bytes,
254            peak_residency_bytes,
255        })
256    }

258    fn metadata_demand(
259        &self,
260        inputs: &ContinuumProductInputs<'_>,
261        workers: usize,
262    ) -> Result<(u64, u64, u64), ProductsError> {
263        let state = inputs.normal_state();
264        let domains = state.domain_count();
265        if domains == 0 {
266            return Err(ProductsError::SourceLineageMismatch);
267        }
268        let requires_beam = self
269            .members()
270            .iter()
271            .any(|member| member.beam_rule() != ProductBeamRule::None);
272        let fitted = if !requires_beam {
273            0
274        } else {
275            match state.catalog() {
276                NormalStateCatalog::UnnormalizedTaylorBlockV1 => 1,
277                NormalStateCatalog::UnnormalizedJointBlockV1 => state.channel_count(),
278                _ => domains
279                    .checked_mul(state.channel_count())
280                    .and_then(|count| count.checked_mul(state.polarization_count()))
281                    .ok_or(ProductsError::ResourceDemandOverflow(
282                        "generation beam count",
283                    ))?,
284            }
285        };
286        let restoring_policy = inputs.problem().products().restoring_beam();
287        let restoring = match (restoring_policy, state.catalog()) {
288            (RestoringBeamPolicy::None, _) => 0,
289            (_, NormalStateCatalog::UnnormalizedTaylorBlockV1) => 1,
290            _ => fitted,
291        };
292        let mut retained = checked_add(
293            size_of::<PublishedContinuumGeneration>() as u64,
294            bytes_for::<PublishedMember>(self.members().len(), "published members")?,
295            "generation summary",
296        )?;
297        for count in [fitted, restoring] {
298            retained = checked_add(
299                retained,
300                bytes_for::<Option<RestoringBeam>>(count, "generation beams")?,
301                "generation metadata",
302            )?;
303        }
304        let mut active_member_beams = 0;
305        for member in self.members() {
306            let domain_name = match member.axes().domain() {
307                ImageDomainRole::Main => 0,
308                ImageDomainRole::Outlier(name) => name.len(),
309            };
310            let spectral_values = match member.axes().spectral().wcs() {
311                SpectralWcs::Linear { .. } => 0,
312                SpectralWcs::Tabular {
313                    channel_centres_hz,
314                    channel_boundaries_hz,
315                } => channel_centres_hz
316                    .len()
317                    .checked_add(channel_boundaries_hz.len())
318                    .ok_or(ProductsError::ResourceDemandOverflow("member WCS"))?,
319            };
320            let beams = match member.beam_rule() {
321                ProductBeamRule::None
322                | ProductBeamRule::Restoring(RestoringBeamPolicy::None)
323                | ProductBeamRule::Metadata(RestoringBeamPolicy::None) => 0,
324                ProductBeamRule::Fitted => fitted / domains,
325                ProductBeamRule::Restoring(RestoringBeamPolicy::Common)
326                | ProductBeamRule::Metadata(RestoringBeamPolicy::Common) => 1,
327                _ => restoring / domains,
328            };
329            let beam_bytes = bytes_for::<Option<RestoringBeam>>(beams, "member beams")?;
330            active_member_beams = active_member_beams.max(beam_bytes);
331            for bytes in [
332                bytes_for::<u8>(member.name().len(), "member name")?,
333                bytes_for::<u8>(domain_name, "member domain")?,
334                bytes_for::<f64>(spectral_values, "member WCS")?,
335                std::mem::size_of_val(member.axes().polarization()) as u64,
336                std::mem::size_of_val(member.dependencies()) as u64,
337                beam_bytes,
338            ] {
339                retained = checked_add(retained, bytes, "member metadata")?;
340            }
341        }
342        let common_scratch = if restoring_policy == RestoringBeamPolicy::Common
343            && matches!(
344                state.catalog(),
345                NormalStateCatalog::UnnormalizedPlaneV1
346                    | NormalStateCatalog::UnnormalizedChannelSlabV1
347            ) {
348            checked_add(
349                bytes_for::<RestoringBeam>(fitted / domains, "common valid beams")?,
350                checked_mul(
351                    bytes_for::<casa_numerics::EllipticalGaussian>(
352                        fitted / domains,
353                        "common numeric beams",
354                    )?,
355                    2,
356                    "common converted and mutable numeric beams",
357                )?,
358                "common beam scratch",
359            )?
360        } else {
361            0
362        };
363        let mut fitting_scratch = 0;
364        if requires_beam {
365            for ordinal in 0..domains {
366                let shape = state
367                    .domain_shape(ordinal)
368                    .ok_or(ProductsError::SourceLineageMismatch)?;
369                let cells = checked_shape_values([shape[0], shape[1], 1, 1])?;
370                let input = bytes_for::<f32>(cells, "beam real PSF")?;
371                // Paged PSFs own the decoded complex plane while converting to
372                // f32; fitting then retains only f32 and the fit workspace.
373                let conversion = bytes_for::<Complex64>(cells, "beam complex PSF")?;
374                fitting_scratch = fitting_scratch.max(
375                    checked_add(conversion, input, "beam input conversion")?.max(checked_add(
376                        input,
377                        casa_imaging_reconstruction::psf_fit_workspace_bytes(shape),
378                        "beam fitting workspace",
379                    )?),
380                );
381            }
382            fitting_scratch = checked_mul(fitting_scratch, workers as u64, "parallel beam fits")?;
383            fitting_scratch = checked_add(
384                fitting_scratch,
385                bytes_for::<Option<Option<RestoringBeam>>>(workers, "beam summary slots")?,
386                "parallel beam summaries",
387            )?;
388        }
389        Ok((
390            retained,
391            active_member_beams,
392            active_member_beams.max(common_scratch).max(fitting_scratch),
393        ))
394    }
395}

397fn generic_scratch_bytes(
398    planned: &PlannedContinuumGeneration,
399    inputs: &ContinuumProductInputs<'_>,
400) -> Result<u64, ProductsError> {
401    let requires_validity = planned
402        .members()
403        .iter()
404        .any(|member| member.validity() != ProductValidityRule::All);
405    let requires_restoration = planned
406        .members()
407        .iter()
408        .any(|member| matches!(member.role(), ProductRole::RestoredImage(_)));
409    let mut maximum = 0;
410    let mut normal_window = 0;
411    for ordinal in 0..inputs.normal_state().domain_count() {
412        let shape = inputs
413            .normal_state()
414            .domain_shape(ordinal)
415            .ok_or(ProductsError::SourceLineageMismatch)?;
416        let cells = checked_shape_values([shape[0], shape[1], 1, 1])?;
417        let plane = bytes_for::<f32>(cells, "generic plane")?;
418        let validity = bytes_for::<bool>(cells, "generic validity plane")?;
419        // The full-window allowance conservatively bounds field-selective reads,
420        // including the scalar-to-complex conversion overlap.
421        normal_window = checked_add(
422            normal_window,
423            casa_imaging_reconstruction::normal_state_window_residency_bytes(
424                shape,
425                inputs.normal_state().polarization_count(),
426                inputs.normal_state().channel_count(),
427                1,
428            )?,
429            "generic normal-state input window",
430        )?;
431        let mut scratch = if requires_validity { validity } else { 0 };
432
433        // Normalization overlaps one converted source plane with its result.
434        scratch = scratch.max(checked_mul(plane, 2, "generic converted plane pair")?);
435        if requires_restoration {
436            // Restoration retains its restored result and normalized residual while
437            // a Gaussian kernel and one exact rustfft convolution workspace live.
438            scratch = scratch.max(checked_add(
439                checked_mul(plane, 3, "generic restoration planes")?,
440                fft_convolution_workspace_bytes(shape)?,
441                "generic restoration scratch",
442            )?);
443        }
444        maximum = maximum.max(checked_add(
445            scratch,
446            bytes_for::<casa_imaging_model::ModelSample>(cells, "generic model input plane")?,
447            "generic loaded model and algorithm scratch",
448        )?);
449    }
450    checked_add(
451        maximum,
452        normal_window,
453        "generic loaded normal state and algorithm scratch",
454    )
455}

457fn taylor_scratch_bytes(inputs: &ContinuumProductInputs<'_>) -> Result<u64, ProductsError> {
458    let state = inputs.normal_state();
459    let shape = state.shape();
460    let cells = checked_shape_values([shape[0], shape[1], 1, 1])?;
461    let terms = u64::try_from(state.coefficient_term_count())
462        .map_err(|_| ProductsError::ResourceDemandOverflow("Taylor term count"))?;
463    let moments = u64::try_from(state.normal_moment_count())
464        .map_err(|_| ProductsError::ResourceDemandOverflow("Taylor moment count"))?;
465    if terms < 2 || moments != terms.saturating_mul(2).saturating_sub(1) {
466        return Err(ProductsError::SourceLineageMismatch);
467    }
468    let cells = u64::try_from(cells)
469        .map_err(|_| ProductsError::ResourceDemandOverflow("Taylor cell count"))?;
470
471    // TaylorProducts retains PSF and weight moments; residual, model,
472    // restored, PB and PB-corrected term families; alpha, PB-corrected alpha,
473    // alpha-error, sensitivity and mask;
474    // two validity planes; and scalar sum weights until every planned member
475    // has been copied into the produced generation.
476    let retained_f32_planes = checked_add(
477        checked_mul(moments, 2, "Taylor PSF and weight planes")?,
478        checked_add(
479            checked_mul(terms, 5, "Taylor term families")?,
480            5,
481            "Taylor scalar product planes",
482        )?,
483        "Taylor retained plane count",
484    )?;
485    let retained = checked_add(
486        checked_mul(
487            checked_mul(retained_f32_planes, cells, "Taylor retained values")?,
488            size_of::<f32>() as u64,
489            "Taylor retained payload bytes",
490        )?,
491        checked_add(
492            checked_mul(
493                checked_mul(2, cells, "Taylor validity values")?,
494                size_of::<bool>() as u64,
495                "Taylor validity bytes",
496            )?,
497            checked_mul(moments, size_of::<f32>() as u64, "Taylor sum weights")?,
498            "Taylor validity and sum-weight bytes",
499        )?,
500        "Taylor retained scratch",
501    )?;
502
503    // Construction additionally retains the principal residual family and PB0
504    // source plane. The normal solve owns three dense term matrices plus two
505    // term vectors at its peak. This intentionally follows the product owner's
506    // concrete current data structures, rather than a runtime approximation.
507    let transient_planes = checked_mul(
508        checked_add(terms, 1, "Taylor transient plane count")?,
509        checked_mul(
510            cells,
511            size_of::<f32>() as u64,
512            "Taylor transient plane bytes",
513        )?,
514        "Taylor transient planes",
515    )?;
516    let term_square = checked_mul(terms, terms, "Taylor normal matrix values")?;
517    let solve = checked_mul(
518        checked_add(
519            checked_mul(term_square, 3, "Taylor normal matrices")?,
520            checked_mul(terms, 2, "Taylor solve vectors")?,
521            "Taylor solve values",
522        )?,
523        size_of::<f64>() as u64,
524        "Taylor solve bytes",
525    )?;
526    let mut total = checked_add(
527        retained,
528        checked_add(transient_planes, solve, "Taylor transient scratch")?,
529        "Taylor algorithm scratch",
530    )?;
531    if inputs.problem().products().restoring_beam() != RestoringBeamPolicy::None {
532        total = checked_add(
533            total,
534            checked_add(
535                checked_mul(cells, size_of::<f32>() as u64, "Taylor restoring kernel")?,
536                fft_convolution_workspace_bytes(shape)?,
537                "Taylor convolution scratch",
538            )?,
539            "Taylor restoration scratch",
540        )?;
541    }
542    Ok(total)
543}

545fn fft_convolution_workspace_bytes(shape: [usize; 2]) -> Result<u64, ProductsError> {
546    let cells = checked_shape_values([shape[0], shape[1], 1, 1])?;
547    let complex_planes = bytes_for::<Complex64>(
548        cells
549            .checked_mul(2)
550            .ok_or(ProductsError::ResourceDemandOverflow("FFT complex planes"))?,
551        "FFT complex planes",
552    )?;
553    let output = bytes_for::<f32>(cells, "FFT output plane")?;
554    let mut planner = FftPlanner::<f64>::new();
555    let mut lane_values = 0usize;
556    for length in shape {
557        for plan in [
558            planner.plan_fft_forward(length),
559            planner.plan_fft_inverse(length),
560        ] {
561            lane_values = lane_values.max(
562                length
563                    .checked_add(plan.get_inplace_scratch_len())
564                    .ok_or(ProductsError::ResourceDemandOverflow("FFT lane scratch"))?,
565            );
566        }
567    }
568    checked_add(
569        checked_add(complex_planes, output, "FFT planes and output")?,
570        bytes_for::<Complex64>(lane_values, "FFT lane and scratch")?,
571        "FFT convolution workspace",
572    )
573}
574
575fn checked_shape_values(shape: [usize; 4]) -> Result<usize, ProductsError> {
576    shape.into_iter().try_fold(1usize, |values, extent| {
577        values
578            .checked_mul(extent)
579            .ok_or(ProductsError::ResourceDemandOverflow("member shape"))
580    })
581}
582
583fn bytes_for<T>(count: usize, what: &'static str) -> Result<u64, ProductsError> {
584    let count = u64::try_from(count).map_err(|_| ProductsError::ResourceDemandOverflow(what))?;
585    checked_mul(count, size_of::<T>() as u64, what)
586}
587
588fn checked_add(left: u64, right: u64, what: &'static str) -> Result<u64, ProductsError> {
589    left.checked_add(right)
590        .ok_or(ProductsError::ResourceDemandOverflow(what))
591}
592
593fn checked_mul(left: u64, right: u64, what: &'static str) -> Result<u64, ProductsError> {
594    left.checked_mul(right)
595        .ok_or(ProductsError::ResourceDemandOverflow(what))
596}
