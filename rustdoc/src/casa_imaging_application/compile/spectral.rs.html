110/// The request's spectral axis over the surveyed selection of `stores`,
111/// laid on store `reference`'s metadata (CASA's `msobj`): a continuum
112/// plane over `continuum`, the band of every store, or a cube on the
113/// reference store's window that reads each store's window where it reaches
114/// the cube.
115pub(super) fn prepare_axis(
116    request: &ImagingRequest,
117    (stores, reference): (&[SurveyedStore], usize),
118    continuum: Option<&ContinuumBand>,
119    frame: &FrameContext<'_>,
120    moving_rest_frame: Option<&MeasFrame>,
121) -> Result<PreparedSpectralAxis, PrepareError> {
122    let store = &stores[reference];
123    let (msv2, catalogue) = (
124        store
125            .msv2()
126            .map(|observation| observation.measurement_set()),
127        store.catalogue(),
128    );
129    let survey = &store.survey;
130    let source_reference = survey.source_frequency_reference;
131    let law = match request.specmode {
132        SpecMode::Mfs => continuum_law(
133            request,
134            (msv2, catalogue),
135            survey,
136            continuum.expect("prepare spans the band of every continuum run"),
137        )?,
138        SpecMode::Cube | SpecMode::Cubedata => cube_law(request, (stores, reference), frame)?,
139        SpecMode::Cubesource => {
140            super::stores::one_store(stores, "a source-frame cube (specmode cubesource)")?;
141            let frame_with_velocity = moving_rest_frame
142                .expect("prepare resolves the moving rest frame of every source-frame cube");
143            source_frame_cube_law(request, store, frame, frame_with_velocity)?
144        }
145    };
146    let source_frame = imaging_frequency_frame(source_reference)?;
147    let output_frame = imaging_frequency_frame(law.output_frequency_reference)?;
148    Ok(PreparedSpectralAxis {
149        selected_source_channels: law.selected_source_channels,
150        frequency_bands: law.frequency_bands,
151        source_frame,
152        output_frequency_reference: law.output_frequency_reference,
153        output_frame,
154        anchor: spectral_frame_anchor(source_frame, output_frame, request.specmode, frame)?,
155        wcs: SpectralWcs::Linear {
156            channels: law.output_channels,
157            reference_pixel: 0.0,
158            reference_frequency_hz: law.reference_frequency_hz,
159            increment_hz: law.increment_hz,
160        },
161        rest_frequency: law.rest_frequency,
162        // CASA lists tclean's further rest frequencies after the image's.
163        image: ImageSpectralCoordinate {
164            additional_rest_frequencies_hz: request.restfreq.iter().skip(1).copied().collect(),
165            ..law.image
166        },
167        doppler: law.doppler,
168        sampling: law.sampling,
169        basis: law.basis,
170        output_channels: law.output_channels,
171        reference_frequency_hz: law.reference_frequency_hz,
172        increment_hz: law.increment_hz,
173    })
174}

176/// One plane over the selected band's envelope over every store (`band`),
177/// gridded in LSRK as CASA's `FTMachine::matchChannel` grids it whatever
178/// the image's `outframe`. Its header records the envelope in `outframe`
179/// (`buildCoordinateSystem`'s `MSUtil::getFreqRangeInSpw`) and `restfreq`'s
180/// first rest frequency, else the reference store's (`survey`'s) first
181/// visibility buffer's source line ([`continuum_rest_frequency_hz`] on an
182/// MSv2, whose `SOURCE` and `DOPPLER` tables CASA reads; the catalogue's
183/// first line on an MSv4), else the recorded band's midpoint (`getImFreq`).
184fn continuum_law(
185    request: &ImagingRequest,
186    (msv2, catalogue): (Option<&MeasurementSet>, &Catalogue),
187    survey: &Survey,
188    band: &ContinuumBand,
189) -> Result<AxisLaw, PrepareError> {
190    let (channels, envelope, recorded) = (&band.channels, &band.gridded, &band.recorded);
191    let [lower_hz, upper_hz] = envelope.edges_hz();
192    let [recorded_lower_hz, recorded_upper_hz] = recorded.edges_hz();
193    // `getImFreq`: a given reffreq is the reference at pixel 0, with the
194    // band widened by twice its offset from the band's centre (CAS-9518).
195    let reference = |midpoint_hz: f64, width_hz: f64| match request.reffreq {
196        Some(reference_hz) => (
197            reference_hz,
198            width_hz + 2.0 * (reference_hz - midpoint_hz).abs(),
199        ),
200        None => (midpoint_hz, width_hz),
201    };
202    if request.reffreq.is_some() && request.outframe != FrequencyRef::LSRK {
203        return Err(PrepareError::ReferenceFrequencyOutsideLsrk {
204            outframe: request.outframe,
205        });
206    }
207    let (reference_frequency_hz, increment_hz) =
208        reference(envelope.midpoint_hz(), upper_hz - lower_hz);
209    let (image_reference_hz, image_increment_hz) = reference(
210        recorded.midpoint_hz(),
211        recorded_upper_hz - recorded_lower_hz,
212    );
213    let field_id = survey.first_visibility_field_id;
214    let spw_id = survey.spectral_windows[0].spw_id;
215    let image_rest_frequency_hz = match (request.restfreq.first(), msv2) {
216        (Some(&hertz), _) => hertz,
217        (None, Some(ms)) => {
218            continuum_rest_frequency_hz(ms, field_id, spw_id, recorded.midpoint_hz())?
219        }
220        (None, None) => catalogue
221            .source_records(
222                FieldId(u32::try_from(field_id).expect("a FIELD row is an u32 id")),
223                WindowId(u32::try_from(spw_id).expect("a window id fits u32")),
224            )
225            .first()
226            .and_then(|record| record.rest_frequencies.as_ref()?.first())
227            .map(casa_types::measures::frequency::MFrequency::hz)
228            .filter(|value| *value != 0.0)
229            .unwrap_or(recorded.midpoint_hz()),
230    };
231    Ok(AxisLaw {
232        selected_source_channels: channels.clone(),
233        frequency_bands: vec![WindowBands::new(); channels.len()],
234        output_frequency_reference: FrequencyRef::LSRK,
235        reference_frequency_hz,
236        increment_hz,
237        output_channels: 1,
238        rest_frequency: RestFrequency::NotApplicable,
239        image: ImageSpectralCoordinate {
240            frequency_reference: request.outframe,
241            reference_frequency_hz: image_reference_hz,
242            increment_hz: image_increment_hz,
243            rest_frequency_hz: image_rest_frequency_hz,
244            additional_rest_frequencies_hz: Vec::new(),
245            undefined_frame: false,
246        },
247        doppler: DopplerConvention::NotApplicable,
248        sampling: SpectralSamplingLaw::IDENTITY,
249        basis: ReconstructionBasis::Constant,
250    })
251}

253/// CASA `SynthesisParamsImage::buildCoordinateSystemCore` asks
254/// `MSDopplerUtil::dopplerInfo` for the first selected SPW and field. Its
255/// first rest frequency wins; an absent or zero value uses the MFS band's
256/// midpoint. This header label does not change the continuum/Taylor basis.
257fn continuum_rest_frequency_hz(
258    ms: &MeasurementSet,
259    field_id: usize,
260    spw_id: usize,
261    band_midpoint_hz: f64,
262) -> Result<f64, PrepareError> {
263    let source_id = ms.field()?.source_id(field_id)?;
264    let window = ms
265        .subtable(SubtableId::SpectralWindow)
266        .expect("required SPW table");
267    let doppler_id = if window
268        .schema()
269        .is_some_and(|schema| schema.contains_column("DOPPLER_ID"))
270    {
271        let scalar = window
272            .cell_accessor(spw_id, "DOPPLER_ID")
273            .map_err(casa_ms::MsError::from)?
274            .scalar()
275            .map_err(casa_ms::MsError::from)?;
276        match scalar {
277            casa_types::ScalarValue::Int32(value) => *value,
278            _ => {
279                return Err(
280                    casa_ms::MsError::InvalidInput("DOPPLER_ID must be Int32".into()).into(),
281                );
282            }
283        }
284    } else {
285        -1
286    };
287    if doppler_id >= 0 && ms.subtable(SubtableId::Doppler).is_some() {
288        return Err(casa_ms::MsError::InvalidInput(
289            "MFS rest-frequency metadata with a DOPPLER table is not supported".into(),
290        )
291        .into());
292    }
293    if ms.subtable(SubtableId::Source).is_none() {
294        return Ok(band_midpoint_hz);
295    }
296    let resolved = continuum_source_line(ms, source_id, spw_id)?;
297    Ok(resolved
298        .filter(|value| *value != 0.0)
299        .unwrap_or(band_midpoint_hz))
300}

302/// The first SOURCE line within a matching key group; -1 is a wildcard.
303/// CASA binary-searches sorted unique keys using its wildcard comparator.
304/// With multiple key groups, wildcards can select or hide a matching group;
305/// refuse that unresolved case instead of merging physical row order.
306fn continuum_source_line(
307    ms: &MeasurementSet,
308    source_id: i32,
309    spw_id: usize,
310) -> Result<Option<f64>, PrepareError> {
311    let source = ms.source()?;
312    let spw_id = i32::try_from(spw_id).expect("SPW ids are stored i32 values");
313    let mut first_key = None;
314    let mut multiple_keys = false;
315    let mut wildcard = false;
316    let mut matching_key = None;
317    let mut resolved = None;
318    for row in 0..source.row_count() {
319        let row_source = source.i32(row, "SOURCE_ID")?;
320        let row_spw = source.i32(row, "SPECTRAL_WINDOW_ID")?;
321        let key = (row_source, row_spw);
322        multiple_keys |= first_key.is_some_and(|prior| prior != key);
323        first_key.get_or_insert(key);
324        wildcard |= row_source == -1 || row_spw == -1;
325        if (row_source != source_id && row_source != -1) || (row_spw != spw_id && row_spw != -1) {
326            continue;
327        }
328        if matching_key.is_some_and(|prior| prior != key) {
329            return Err(casa_ms::MsError::InvalidInput(
330                "overlapping SOURCE key groups for MFS rest-frequency metadata are not supported"
331                    .into(),
332            )
333            .into());
334        }
335        matching_key = Some(key);
336        if resolved.is_some() {
337            continue;
338        }
339        let Some(ArrayValue::Float64(values)) = source.optional_array(row, "REST_FREQUENCY")?
340        else {
341            continue;
342        };
343        if values.is_empty() {
344            continue;
345        }
346        resolved = values.first().copied();
347    }
348    if wildcard && multiple_keys {
349        return Err(casa_ms::MsError::InvalidInput(
350            "multiple SOURCE key groups with wildcard ids for MFS rest-frequency metadata are not supported".into(),
351        ).into());
352    }
353    Ok(resolved)
354}

356/// Whether CASA tunes a `cube`'s frequency selection to the image's band
357/// (`SynthesisImagerVi2::tuneSelectData`): when the window of its first
358/// visibility buffer has 30 or more channels, or when the model is saved to
359/// `MODEL_DATA`, `SynthesisImager::tuneSelectData` narrows each window's
360/// bands to the LSRK frequencies from pixel −1.5 to `nchan + 0.5` of the
361/// image axis ([`frequency_bands::cube_bands`]); a `cubedata` or
362/// `cubesource` cube keeps the selection. The tuned band keeps every
363/// channel a plane is interpolated from, since CASA refuses planes
364/// narrower than a channel (`calcChanFreqs`), so tuning changes which
365/// channels are read but not the image: a window the axis misses is still
366/// read (its bands are kept where the image's overlaps none of them), and
367/// the density `VisImagingWeight` forms is the untuned selection's.
368/// casa-rs's visibility-domain continuum fit (`fitspw`, which CASA's
369/// tclean lacks) keeps the named channels as its output channels.
370fn tunes_selection(request: &ImagingRequest, first_window: &SourceSpectralWindow) -> bool {
371    request.specmode == SpecMode::Cube
372        && request.fitspw.is_none()
373        && (first_window.frequencies_hz.len() >= 30 || request.savemodel == SaveModel::ModelColumn)
374}

376/// CASA's cube axis over the reference store's selected windows
377/// (`buildCoordinateSystemCore` on `msobj`'s windows): one window's
378/// channels, or several windows' merged into one axis
379/// ([`casa_ms::combine_spectral_windows`]), with `axis`'s grid laid over
380/// them; and the band their `spw` selection spans in the data's frame
381/// ([`casa_ms::casa_selected_band_hz`]). Every window's samples are
382/// gridded onto the one axis (`FTMachine::matchChannel`).
383fn cube_setup(
384    request: &ImagingRequest,
385    survey: &Survey,
386    frame: &FrameContext<'_>,
387    mut axis: CubeAxisConfig,
388) -> Result<(CubeSpectralSetup, [f64; 2]), PrepareError> {
389    fn selected<'w>(
390        survey: &'w Survey,
391        window: &'w SourceSpectralWindow,
392    ) -> (&'w [f64], &'w [f64], Option<&'w [usize]>) {
393        (
394            window.frequencies_hz.as_slice(),
395            window.channel_widths_hz.as_slice(),
396            survey.spw_channels.get(&window.spw_id).map(Vec::as_slice),
397        )
398    }
399    let windows = &survey.spectral_windows;
400    // Name the window whose selection spans no band.
401    if let Some(window) = windows
402        .iter()
403        .find(|window| casa_ms::casa_selected_band_hz([selected(survey, window)]).is_none())
404    {
405        return Err(PrepareError::SpectralWindowCoordinates {
406            spw_id: window.spw_id,
407        });
408    }
409    let band_hz =
410        casa_ms::casa_selected_band_hz(windows.iter().map(|window| selected(survey, window)))
411            .ok_or(PrepareError::NoSelectedSamples)?;
412    axis.selected_band_hz = Some(band_hz);
413    let (frequencies_hz, widths_hz) = match windows.as_slice() {
414        [window] => (
415            window.frequencies_hz.clone(),
416            window.channel_widths_hz.clone(),
417        ),
418        _ => casa_ms::combine_spectral_windows(
419            &windows
420                .iter()
421                .map(|window| {
422                    (
423                        window.frequencies_hz.as_slice(),
424                        window.channel_widths_hz.as_slice(),
425                    )
426                })
427                .collect::<Vec<_>>(),
428        )?,
429    };
430    let setup = CubeSpectralSetup::for_casa_cube_axis(
431        survey.source_frequency_reference,
432        &frequencies_hz,
433        &widths_hz,
434        request.channel_count,
435        &axis,
436        frame.anchor_time_mjd_seconds,
437        frame.field_id,
438        Some(frame.phase.clone()),
439        frame.engine,
440    )?;
441    Ok((setup, band_hz))
442}

444/// What a cube on `setup` reads of each store, in store order: the
445/// channels of each window, and the bands of the windows whose times read
446/// only some of them ([`store_reads`]). Whether CASA tunes the selection
447/// to the image's band is decided once, from the iterator's first buffer,
448/// the first store's ([`tunes_selection`]), and applies to every store
449/// (`SynthesisImager::tuneSelectData`).
450///
451/// The reference store's windows are those the axis is laid on; another
452/// store's one window is read where it reaches the axis, as CASA's cube
453/// gridding selects each MeasurementSet's channels in the image's band.
454///
455/// # Errors
456///
457/// An axis none of the reference store's windows reaches; another store
458/// of several windows, or of another source frame than the reference
459/// store's; a frame conversion that fails.
460fn cube_store_reads(
461    request: &ImagingRequest,
462    (stores, reference): (&[SurveyedStore], usize),
463    frame: &FrameContext<'_>,
464    setup: &CubeSpectralSetup,
465) -> Result<(Vec<WindowChannels>, Vec<WindowBands>), PrepareError> {
466    let first = &stores[0].survey;
467    let first_window = first
468        .spectral_windows
469        .iter()
470        .find(|window| window.spw_id == first.first_visibility_spw_id)
471        .expect("the first visibility is in a surveyed window");
472    let tuned = tunes_selection(request, first_window);
473    let image_band = image_band(frame, setup, tuned)?;
474    let mut channels = Vec::with_capacity(stores.len());
475    let mut bands = Vec::with_capacity(stores.len());
476    for (index, store) in stores.iter().enumerate() {
477        let survey = &store.survey;
478        let field = if index == reference {
479            frame.field_id
480        } else {
481            one_window(&survey.spectral_windows, "native cube imaging")?;
482            if survey.source_frequency_reference != setup.source_freq_ref {
483                return Err(PrepareError::MixedSourceFrames);
484            }
485            usize::try_from(*survey.fields.first().expect("a surveyed store's field"))
486                .expect("a surveyed field is a FIELD row")
487        };
488        let reads = store_reads(
489            request,
490            ((store.catalogue(), &store.engine), survey, field),
491            setup,
492            (image_band, tuned),
493        )?;
494        if index == reference && !reads.reaches_axis {
495            return Err(PrepareError::NoCommonCubeChannels);
496        }
497        channels.push(reads.channels);
498        bands.push(reads.bands);
499    }
500    Ok((channels, bands))
501}

511/// The channels a cube on `setup` reads of each window of one store's
512/// `survey`, its samples' frequencies converted towards `field`: those
513/// whose samples reach the axis
514/// ([`CubeSpectralSetup::source_channel_support`]) and that the window's
515/// frequency bands, tuned to the image's band `image_band` where CASA
516/// tunes the selection ([`tunes_selection`]), select at some epoch of its
517/// samples ([`cube_bands`]); a `fitspw` fit, which has no bands, reads
518/// those `spw` names. Where the bands do not
519/// select every channel read at every epoch, the window carries them and
520/// each time reads only the channels they select then, as CASA's iterator
521/// does.
522///
523/// A window none of whose channels reaches the axis is read as `spw`
524/// selects it, tuned or not, as CASA does: its samples fall outside every
525/// plane, but `VisImagingWeight`'s density counts them, since CASA forms it
526/// over the untuned selection of the imager that defines the cube, and a
527/// final pass writes the cube model's prediction there, zero, into its
528/// `MODEL_DATA` (test_modelvis_29).
529fn store_reads(
530    request: &ImagingRequest,
531    ((catalogue, engine), survey, field): ((&Catalogue, &MsCalEngine), &Survey, usize),
532    setup: &CubeSpectralSetup,
533    (image_band, tuned): (Option<[f64; 2]>, bool),
534) -> Result<StoreReads, PrepareError> {
535    let bands = cube_bands(request, (catalogue, engine), survey, image_band)?;
536    let mut reads = StoreReads {
537        channels: WindowChannels::new(),
538        bands: WindowBands::new(),
539        reaches_axis: false,
540    };
541    for window in &survey.spectral_windows {
542        let mut channels = setup
543            .source_channel_support(
544                &window.frequencies_hz,
545                &window.channel_widths_hz,
546                survey.time_bounds_mjd_seconds,
547                field,
548                engine,
549            )?
550            .indices;
551        let selection = match &bands {
552            Some(bands) => bands.epoch_selection((catalogue, engine), survey, window)?,
553            None => None,
554        };
555        let explicit = survey.spw_channels.get(&window.spw_id);
556        match (&selection, explicit) {
557            (Some(selection), _) => channels.retain(|channel| selection.ever(*channel)),
558            (None, Some(explicit)) if !tuned => {
559                channels.retain(|channel| explicit.contains(channel));
560            }
561            _ => {}
562        }
563        if channels.is_empty() {
564            // `spw`'s channels, or the whole window when it names none.
565            channels = explicit
566                .cloned()
567                .unwrap_or_else(|| (0..window.frequencies_hz.len()).collect());
568        } else {
569            reads.reaches_axis = true;
570            if let (Some(bands), Some(selection)) = (&bands, &selection)
571                && channels.iter().any(|channel| !selection.always(*channel))
572                && let Some(window_bands) = bands.frequency_bands(window.spw_id)
573            {
574                reads.bands.insert(window.spw_id, window_bands.clone());
575            }
576        }
577        reads.channels.insert(window.spw_id, channels);
578    }
579    Ok(reads)
580}

582/// The image's LSRK band a `cube` tunes its selection to
583/// ([`frequency_bands::image_band_lsrk`]), where CASA tunes it.
584fn image_band(
585    frame: &FrameContext<'_>,
586    setup: &CubeSpectralSetup,
587    tuned: bool,
588) -> Result<Option<[f64; 2]>, PrepareError> {
589    if !tuned {
590        return Ok(None);
591    }
592    let channels = setup.output_channel_frequencies_hz.len();
593    let (first_hz, increment_hz) = first_and_increment(setup, channels, 1.0)?;
594    Ok(Some(frequency_bands::image_band_lsrk(
595        frame,
596        setup.output_freq_ref,
597        (first_hz, increment_hz, channels),
598    )?))
599}

601/// The bands a cube's gridding selects one store's windows' channels by at
602/// each time ([`frequency_bands::cube_bands`]): a `cube`'s in `outframe`,
603/// or tuned to the image's band `image_band` where CASA tunes the
604/// selection; in the data's frame where tclean's selection frame is
605/// `Undefined` (`cubedata`) or `REST` (`cubesource`, or a `cube` whose
606/// `outframe` is `REST`), which select each named range's channels at
607/// every time, a range with a stride whole. None for a `fitspw` fit,
608/// which reads its channels as named.
609fn cube_bands(
610    request: &ImagingRequest,
611    (catalogue, engine): (&Catalogue, &MsCalEngine),
612    survey: &Survey,
613    image_band: Option<[f64; 2]>,
614) -> Result<Option<frequency_bands::CubeBands>, PrepareError> {
615    use frequency_bands::SelectionFrame;
616    if request.fitspw.is_some() {
617        return Ok(None);
618    }
619    let frame = match request.specmode {
620        SpecMode::Cube if image_band.is_some() => SelectionFrame::Converted(FrequencyRef::LSRK),
621        SpecMode::Cube if request.outframe != FrequencyRef::REST => {
622            SelectionFrame::Converted(request.outframe)
623        }
624        SpecMode::Mfs | SpecMode::Cube | SpecMode::Cubedata | SpecMode::Cubesource => {
625            SelectionFrame::Data
626        }
627    };
628    frequency_bands::cube_bands((catalogue, engine), survey, frame, image_band).map(Some)
629}

631/// A cube in its output frame (`cube`) or the data's (`cubedata`), laid
632/// on the windows of store `reference` (CASA's `msobj`), reading each
633/// store's channels where they reach it ([`cube_store_reads`]).
634fn cube_law(
635    request: &ImagingRequest,
636    (stores, reference): (&[SurveyedStore], usize),
637    frame: &FrameContext<'_>,
638) -> Result<AxisLaw, PrepareError> {
639    let store = &stores[reference];
640    let (catalogue, survey) = (store.catalogue(), &store.survey);
641    let (setup, band_hz) = cube_setup(request, survey, frame, cube_axis(request)?)?;
642    let (selected_source_channels, frequency_bands) =
643        cube_store_reads(request, (stores, reference), frame, &setup)?;
644    let output_channels = setup.output_channel_frequencies_hz.len();
645    let (reference_frequency_hz, increment_hz) = first_and_increment(&setup, output_channels, 1.0)?;
646    let source_rest_hz =
647        source_rest_frequency(catalogue, frame.field_id, &survey.spectral_windows)?;
648    let (resolved_rest_hz, image_rest_frequency_hz) =
649        cube_rest_frequency_hz(request.restfreq.first().copied(), source_rest_hz, band_hz);
650    let (rest_frequency, doppler) = match resolved_rest_hz {
651        None => (
652            RestFrequency::NotApplicable,
653            DopplerConvention::NotApplicable,
654        ),
655        Some(hertz) => (
656            RestFrequency::Line { hertz },
657            doppler_convention(request.veltype)?,
658        ),
659    };
660    Ok(AxisLaw {
661        selected_source_channels,
662        frequency_bands,
663        output_frequency_reference: setup.output_freq_ref,
664        reference_frequency_hz,
665        increment_hz,
666        output_channels,
667        rest_frequency,
668        image: ImageSpectralCoordinate {
669            frequency_reference: setup.output_freq_ref,
670            reference_frequency_hz,
671            increment_hz,
672            rest_frequency_hz: image_rest_frequency_hz,
673            additional_rest_frequencies_hz: Vec::new(),
674            undefined_frame: request.specmode == SpecMode::Cubedata
675                && setup.output_freq_ref != FrequencyRef::REST,
676        },
677        doppler,
678        sampling: sampling_law(setup.interpolation),
679        basis: ReconstructionBasis::ChannelLocal {
680            channels: output_channels,
681        },
682    })
683}

685/// A cube in the rest frame of a moving source: the data-frame axis
686/// scaled into the source's rest frame (`cubesource`), on `store`'s one
687/// window, reading its channels as [`store_reads`] reads a `cubedata`
688/// cube's.
689///
690/// # Errors
691///
692/// Several selected windows: CASA's source-frame cube over several
693/// windows is not yet ported (#1059).
694fn source_frame_cube_law(
695    request: &ImagingRequest,
696    store: &SurveyedStore,
697    frame: &FrameContext<'_>,
698    moving_rest_frame: &MeasFrame,
699) -> Result<AxisLaw, PrepareError> {
700    let (catalogue, survey) = (store.catalogue(), &store.survey);
701    if survey.spectral_windows.len() > 1 {
702        return Err(PrepareError::SourceFrameCubeWindows {
703            selected: survey.spectral_windows.len(),
704        });
705    }
706    let source_reference = survey.source_frequency_reference;
707    let axis = CubeAxisConfig {
708        specmode: CubeSpecMode::Cubedata,
709        outframe: source_reference,
710        ..cube_axis(request)?
711    };
712    let (setup, _) = cube_setup(request, survey, frame, axis)?;
713    let reads = store_reads(
714        request,
715        ((catalogue, &store.engine), survey, frame.field_id),
716        &setup,
717        (None, false),
718    )?;
719    if !reads.reaches_axis {
720        return Err(PrepareError::NoCommonCubeChannels);
721    }
722    let output_channels = setup.output_channel_frequencies_hz.len();
723    let factor = casa_ms::convert_frequency_to_frame_with_frame(
724        source_reference,
725        FrequencyRef::REST,
726        1.0,
727        Some(moving_rest_frame),
728    )?;
729    let (reference_frequency_hz, increment_hz) =
730        first_and_increment(&setup, output_channels, factor)?;
731    let rest_frequency_hz = request
732        .restfreq
733        .first()
734        .copied()
735        .or(source_rest_frequency(
736            catalogue,
737            frame.field_id,
738            &survey.spectral_windows,
739        )?)
740        .ok_or(PrepareError::RestFrequencyMissing)?;
741    Ok(AxisLaw {
742        selected_source_channels: vec![reads.channels],
743        frequency_bands: vec![reads.bands],
744        output_frequency_reference: FrequencyRef::REST,
745        reference_frequency_hz,
746        increment_hz,
747        output_channels,
748        rest_frequency: RestFrequency::Line {
749            hertz: rest_frequency_hz,
750        },
751        image: ImageSpectralCoordinate {
752            frequency_reference: FrequencyRef::REST,
753            reference_frequency_hz,
754            increment_hz,
755            rest_frequency_hz,
756            additional_rest_frequencies_hz: Vec::new(),
757            undefined_frame: false,
758        },
759        doppler: doppler_convention(request.veltype)?,
760        sampling: sampling_law(request.interpolation),
761        basis: ReconstructionBasis::ChannelLocal {
762            channels: output_channels,
763        },
764    })
765}

767/// CASA's cube axis of the request: `start` (else `channel_start`) and
768/// `width` in the request's velocity convention; [`cube_setup`] sets the
769/// band `spw` selects.
770fn cube_axis(request: &ImagingRequest) -> Result<CubeAxisConfig, PrepareError> {
771    let start = match (&request.start, request.channel_start) {
772        (Some(start), _) => Some(CubeAxisValue::parse(start, request.veltype)?),
773        (None, Some(channel)) => Some(CubeAxisValue::Channel(
774            i32::try_from(channel).map_err(|_| PrepareError::CubeStartChannel { channel })?,
775        )),
776        (None, None) => None,
777    };
778    let width = request
779        .width
780        .as_deref()
781        .map(|width| CubeAxisValue::parse(width, request.veltype))
782        .transpose()?;
783    Ok(CubeAxisConfig {
784        specmode: if request.specmode == SpecMode::Cubedata {
785            CubeSpecMode::Cubedata
786        } else {
787            CubeSpecMode::Cube
788        },
789        outframe: request.outframe,
790        veltype: request.veltype,
791        interpolation: request.interpolation,
792        rest_frequency_hz: request.restfreq.first().copied(),
793        start,
794        width,
795        selected_band_hz: None,
796    })
797}

799/// The first output channel's centre and the channel increment, scaled by
800/// `factor`; a one-channel cube's increment is its width.
801fn first_and_increment(
802    setup: &CubeSpectralSetup,
803    output_channels: usize,
804    factor: f64,
805) -> Result<(f64, f64), PrepareError> {
806    let first_hz = setup.output_channel_frequencies_hz[0];
807    let increment_hz = if output_channels > 1 {
808        setup.output_channel_frequencies_hz[1] - first_hz
809    } else {
810        setup.output_channel_widths_hz[0]
811    } * factor;
812    if !increment_hz.is_finite() || increment_hz == 0.0 {
813        return Err(PrepareError::CubeIncrement { increment_hz });
814    }
815    Ok((first_hz * factor, increment_hz))
816}

818/// The model's Doppler convention of CASA `veltype`.
819fn doppler_convention(veltype: DopplerRef) -> Result<DopplerConvention, PrepareError> {
820    match veltype {
821        DopplerRef::RADIO => Ok(DopplerConvention::Radio),
822        DopplerRef::Z => Ok(DopplerConvention::Optical),
823        DopplerRef::BETA => Ok(DopplerConvention::Relativistic),
824        DopplerRef::RATIO | DopplerRef::GAMMA => Err(PrepareError::UnsupportedDoppler {
825            convention: veltype,
826        }),
827    }
828}

838/// The image's epoch, reference direction and observatory, where the
839/// rows' frame and the output frame differ, or where a `cube` is gridded
840/// through LSRK ([`FrequencyFrame::gridded_through_lsrk`]): CASA grids a
841/// `cube` (not `cubedata`, whose rows keep their frequencies) in LSRK
842/// whatever its `outframe`, so a TOPO cube of TOPO rows is fixed in LSRK at
843/// the first time and each row is Doppler tracked from it.
844fn spectral_frame_anchor(
845    source_frame: FrequencyFrame,
846    output_frame: FrequencyFrame,
847    specmode: SpecMode,
848    frame: &FrameContext<'_>,
849) -> Result<SpectralFrameAnchor, PrepareError> {
850    let tracked = specmode == SpecMode::Cube && output_frame.gridded_through_lsrk();
851    if source_frame == output_frame && !tracked {
852        return Ok(SpectralFrameAnchor::NotApplicable);
853    }
854    let [x_metres, y_metres, z_metres] = frame.engine.observatory_position().as_itrf();
855    Ok(SpectralFrameAnchor::Conversion {
856        // As `MEpoch::from_mjd_seconds` reads a row's time, so a row at the
857        // first time is at the anchor's epoch exactly.
858        epoch: Epoch::new(
859            frame.anchor_time_mjd_seconds * (1.0 / 86_400.0),
860            imaging_time_scale(frame.engine.time_reference())?,
861        ),
862        direction: frame.direction.reference_direction(),
863        observatory_position: ItrfPosition::new(x_metres, y_metres, z_metres),
864    })
865}

889/// The rest frequency the field's source states over the selected windows
890/// (MSv2 `SOURCE::REST_FREQUENCY`, MSv4 `LINE_REST_FREQUENCY`), if any: the
891/// first line of each record for a selected window or for every window
892/// ([`Catalogue::source_records`]), which must agree.
893fn source_rest_frequency(
894    catalogue: &Catalogue,
895    field_id: usize,
896    spectral_windows: &[SourceSpectralWindow],
897) -> Result<Option<f64>, PrepareError> {
898    let field = FieldId(u32::try_from(field_id).expect("a FIELD row is an u32 id"));
899    let mut resolved = None;
900    for window in spectral_windows {
901        let window = WindowId(u32::try_from(window.spw_id).expect("a window id fits u32"));
902        for record in catalogue.source_records(field, window) {
903            let Some(value) = record
904                .rest_frequencies
905                .as_ref()
906                .and_then(|lines| lines.first())
907                .map(casa_types::measures::frequency::MFrequency::hz)
908                .filter(|value| *value > 0.0)
909            else {
910                continue;
911            };
912            if resolved.is_some_and(|prior: f64| prior.to_bits() != value.to_bits()) {
913                return Err(PrepareError::RestFrequencyConflict);
914            }
915            resolved = Some(value);
916        }
917    }
918    Ok(resolved)
919}

921/// CASA's precedence for a cube's rest frequency: the request's, else the
922/// source's; the header records the centre of the selected band `band_hz`
923/// when neither exists (`buildCoordinateSystemCore`'s provisional rest
924/// frequency, `(datafend + datafstart)/2`).
925fn cube_rest_frequency_hz(
926    explicit_hz: Option<f64>,
927    source_hz: Option<f64>,
928    band_hz: [f64; 2],
929) -> (Option<f64>, f64) {
930    let resolved = explicit_hz.or(source_hz);
931    (
932        resolved,
933        resolved.unwrap_or((band_hz[1] + band_hz[0]) / 2.0),
934    )
935}
