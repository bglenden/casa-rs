40impl LegacyEpConvolution {
41    /// Working memory for the Jones aperture and convolution FFT screens.
42    pub const SCREEN_WORKSPACE_BYTES: u64 = 256 << 20;
43
44    /// Form the nPBWProjectFT cells for the original VLA L-band feed.
45    ///
46    /// `frequency_hz` is the highest channel in the first visibility buffer.
47    /// `angle_rad` is that buffer's feed angle for antenna zero.
48    ///
49    /// # Errors
50    /// The input geometry is invalid, the aperture cannot be evaluated or
51    /// the convolution has no finite positive integral.
52    pub fn vla_l_band(
53        aperture: &EvlaApertureModel,
54        image_side: usize,
55        cell_rad: f64,
56        frequency_hz: f64,
57        angle_rad: f64,
58    ) -> Result<Self, OperatorError> {
59        let frequency_hz = f64::from(frequency_hz as f32);
60        let beam =
61            |side, sky_cell| aperture_beam(aperture, side, sky_cell, frequency_hz, angle_rad);
62        let mut average = beam(image_side, cell_rad)?;
63        let average_beam = average
64            .power(false)?
65            .into_iter()
66            .map(|value| value.re)
67            .collect();
68        drop(average);
69        let mut far = beam(
70            SCREEN,
71            cell_rad * (SAMPLING * image_side) as f64 / SCREEN as f64,
72        )?;
73        let mut kernels = [far.power(true)?, far.power(false)?];
74        drop(far);
75        let mut fft = PlaneFft::<f32>::new([SCREEN, SCREEN], false)?;
76        let origin = SCREEN / 2;
77        for kernel in &mut kernels {
78            fft.transform(kernel, false)?;
79            let peak = kernel[origin * SCREEN + origin];
80            if !peak.re.is_finite() || peak.re <= 0. {
81                return Err(invalid("legacy EP convolution has no positive peak"));
82            }
83            for value in kernel.iter_mut() {
84                *value /= peak;
85            }
86        }
87        let support = kernel_support(&kernels[0], 1e-4)?;
88        let radius = (support + 1) * SAMPLING;
89        let kernel_side = radius * 2 + 1;
90        let integral = sampled_integral(&kernels[0], support);
91        if !integral.is_finite() || integral <= 0. {
92            return Err(invalid("legacy EP convolution integral is not positive"));
93        }
94        let kernels = kernels.map(|kernel| crop(&kernel, radius));
95        let [mut imaging, mut weights] = kernels;
96        for value in &mut imaging {
97            *value /= integral;
98        }
99        for value in &mut weights {
100            *value /= integral * integral;
101        }
102        Ok(Self {
103            sampling: SAMPLING,
104            support: vec![support],
105            kernel_side,
106            kernels: vec![[imaging, weights]],
107            average_beam,
108        })
109    }

111    /// Prepare the beam-corrected RR, RL, LR and LL Fourier planes from an
112    /// IQUV sky image as the legacy EP solver's prediction grid.
113    /// # Errors
114    /// The image and average beam must have matching square IQUV planes.
115    pub fn prepare_model(
116        image: &ndarray::ArrayD<f32>,
117        beam: &ndarray::ArrayD<f32>,
118        sampling: usize,
119    ) -> Result<[Vec<Complex32>; 4], OperatorError> {
120        Self::prepare_pbw_model(image, beam, sampling, Some(0.05))
121    }

123    /// Prepare the four circular Fourier planes, optionally dividing by
124    /// the average beam above the requested legacy PB cutoff.
125    ///
126    /// # Errors
127    /// The image, average beam, sampling or cutoff is invalid.
128    pub fn prepare_pbw_model(
129        image: &ndarray::ArrayD<f32>,
130        beam: &ndarray::ArrayD<f32>,
131        sampling: usize,
132        cutoff: Option<f32>,
133    ) -> Result<[Vec<Complex32>; 4], OperatorError> {
134        if cutoff.is_some_and(|limit| !limit.is_finite() || limit < 0.) {
135            return Err(invalid(
136                "the PB correction cutoff must be finite and nonnegative",
137            ));
138        }
139        let [nx, ny, 4, 1] = *image.shape() else {
140            return Err(invalid("legacy EP needs a single-channel IQUV image"));
141        };
142        if nx != ny || !nx.is_multiple_of(2) || sampling == 0 || image.shape() != beam.shape() {
143            return Err(invalid("invalid model or beam shape in legacy EP"));
144        }
145        let mut fft = PlaneFft::new([nx, ny], false)?;
146        let mut result = std::array::from_fn(|_| Vec::with_capacity(nx * ny));
147        let sinc = (0..nx)
148            .map(|pixel| {
149                if pixel == nx / 2 {
150                    1.
151                } else {
152                    let value = std::f32::consts::PI * (pixel as isize - (nx / 2) as isize) as f32
153                        / (nx * sampling) as f32;
154                    value.sin() / value
155                }
156            })
157            .collect::<Vec<_>>();
158        let peak = beam.iter().copied().fold(0_f32, f32::max);
159        if !peak.is_finite() || peak < 1e-4 {
160            return Err(invalid("invalid average beam in legacy EP"));
161        }
162        let normalize = (peak - 1.).abs() > 1e-3;
163        let primary_beam = |x, y, pol| {
164            if pol < 2 {
165                0.5 * (beam[[x, y, 0, 0]] + beam[[x, y, 1, 0]])
166            } else {
167                beam[[x, y, pol, 0]]
168            }
169        };
170        for (pol, grid) in result.iter_mut().enumerate() {
171            let peak = if normalize {
172                (0..ny)
173                    .flat_map(|y| (0..nx).map(move |x| primary_beam(x, y, pol)))
174                    .fold(0_f32, f32::max)
175            } else {
176                1.
177            };
178            for y in 0..ny {
179                for x in 0..nx {
180                    let [i, q, u, v] = std::array::from_fn(|index| image[[x, y, index, 0]]);
181                    let circular = [
182                        Complex32::from(i + v),
183                        Complex32::new(q, u),
184                        Complex32::new(q, -u),
185                        Complex32::from(i - v),
186                    ][pol];
187                    let taper = sinc[x] * sinc[y];
188                    let correction = primary_beam(x, y, pol) / peak * taper;
189                    grid.push(
190                        if cutoff.is_some_and(|limit| correction != 0. && correction.abs() >= limit)
191                        {
192                            circular / correction
193                        } else {
194                            circular * taper
195                        },
196                    );
197                }
198            }
199            fft.transform(grid, false)?;
200        }
201        Ok(result)
202    }
203}

205impl LegacyEpWPlanePlan {
206    /// Measure the support of the cold legacy PB/W cells. The caller admits
207    /// the aperture, FFT scratch and the materialized output independently.
208    /// # Errors
209    /// The aperture or image geometry is invalid, or a cell has no support.
210    pub fn new(
211        aperture: &EvlaApertureModel,
212        image_side: usize,
213        cell_rad: [f64; 2],
214        frequency_hz: f64,
215        angle_rad: f64,
216        plane_count: usize,
217    ) -> Result<Self, OperatorError> {
218        if !(2..=usize::from(u16::MAX)).contains(&plane_count) {
219            return Err(invalid("legacy cold PB/W requires at least two W planes"));
220        }
221        let frequency_hz = f64::from(frequency_hz as f32);
222        let average_beam =
223            aperture_beam(aperture, image_side, cell_rad[0], frequency_hz, angle_rad)?
224                .power(false)?
225                .into_iter()
226                .map(|value| value.re)
227                .collect();
228        let sky_cell_rad =
229            cell_rad.map(|cell| cell * (SAMPLING * image_side) as f64 / SCREEN as f64);
230        let mut aperture =
231            aperture_beam(aperture, SCREEN, sky_cell_rad[0], frequency_hz, angle_rad)?;
232        let screens = [aperture.power(true)?, aperture.power(false)?];
233        drop(aperture);
234        let sky_peaks = screens.each_ref().map(|screen| peak_index(screen));
235        let mut fft = PlaneFft::new([SCREEN, SCREEN], false)?;
236        let mut first_imaging = screens[0].clone();
237        fft.transform(&mut first_imaging, false)?;
238        let width = (plane_count - 1) as f64;
239        let w_scale = f64::from((width * width) as f32) / (0.25 / cell_rad[0].abs());
240        if !w_scale.is_finite() || w_scale <= 0. {
241            return Err(invalid("invalid legacy EP cold W scale"));
242        }
243        let mut support = Vec::with_capacity(plane_count);
244        let first_peak = centre(&first_imaging);
245        for plane in 0..plane_count {
246            let peak = if plane == 0 {
247                first_peak
248            } else {
249                // The FFT centre is the sum of the projected screen. Only
250                // the inner window differs from the W=0 screen.
251                first_peak
252                    + w_phases(sky_cell_rad, w_scale, plane)
253                        .map(|(index, phase)| screens[0][index] * (phase - 1.))
254                        .sum::<Complex32>()
255            };
256            // CASA searches plane zero even for the threshold from plane W.
257            support.push(kernel_support(&first_imaging, peak.norm() * 1e-4)?);
258        }
259        Ok(Self {
260            average_beam,
261            screens,
262            sky_peaks,
263            first_imaging,
264            support,
265            fft,
266            sky_cell_rad,
267            w_scale,
268        })
269    }

271    /// Additional storage required for all cropped imaging and weight cells.
272    pub fn retained_kernel_bytes(&self) -> u64 {
273        let side = (2 * self.crop_radius() + 1) as u64;
274        side.saturating_pow(2)
275            .saturating_mul(self.support.len() as u64)
276            .saturating_mul((2 * size_of::<Complex32>()) as u64)
277            .saturating_add((self.support.len() * 128) as u64)
278    }

280    /// The W coordinate used during the tool call that generated the cache.
281    pub fn w_scale(&self) -> f64 {
282        self.w_scale
283    }

285    fn crop_radius(&self) -> usize {
286        (self.support.iter().copied().max().unwrap_or(0) + 1) * SAMPLING
287    }

289    /// Transform and crop the cells using their common canvas.
290    /// # Errors
291    /// A Fourier transform failed, or the plane's normalization is invalid.
292    pub fn build(mut self) -> Result<LegacyEpConvolution, OperatorError> {
293        let radius = self.crop_radius();
294        let mut first_weights = self.screens[1].clone();
295        self.fft.transform(&mut first_weights, false)?;
296        let peaks = [centre(&self.first_imaging), centre(&first_weights)];
297        if peaks
298            .iter()
299            .any(|peak| !peak.re.is_finite() || peak.re <= 0.)
300        {
301            return Err(invalid("legacy EP convolution has no positive peak"));
302        }
303        let mut kernels = Vec::with_capacity(self.support.len());
304        let mut imaging = std::mem::take(&mut self.first_imaging);
305        let mut weight = first_weights;
306        for (plane, &support) in self.support.iter().enumerate() {
307            let mut normalization = peaks;
308            if plane > 0 {
309                imaging.copy_from_slice(&self.screens[0]);
310                weight.copy_from_slice(&self.screens[1]);
311                for (index, phase) in w_phases(self.sky_cell_rad, self.w_scale, plane) {
312                    imaging[index] *= phase;
313                    weight[index] *= phase;
314                }
315                // Dividing by the projected complex sky peak before FFT
316                // is equivalent to removing its phase from the result.
317                for (cell, grid) in [&imaging, &weight].into_iter().enumerate() {
318                    let peak = self.sky_peaks[cell];
319                    normalization[cell] *= grid[peak] / self.screens[cell][peak];
320                }
321                self.fft.transform(&mut imaging, false)?;
322                self.fft.transform(&mut weight, false)?;
323            }
324            let [mut imaging, mut weights] = [&imaging, &weight].map(|grid| crop(grid, radius));
325            for value in &mut imaging {
326                *value /= normalization[0];
327            }
328            let integral = sample_cropped(&imaging, 2 * radius + 1, support)?;
329            if !integral.is_finite() || integral <= 0. {
330                return Err(invalid("legacy EP convolution integral is not positive"));
331            }
332            for value in &mut imaging {
333                *value /= integral;
334            }
335            for value in &mut weights {
336                *value /= normalization[1] * integral * integral;
337            }
338            kernels.push([imaging, weights]);
339        }
340        Ok(LegacyEpConvolution {
341            sampling: SAMPLING,
342            support: self.support,
343            kernel_side: 2 * radius + 1,
344            kernels,
345            average_beam: self.average_beam,
346        })
347    }
348}

350fn centre(screen: &[Complex32]) -> Complex32 {
351    screen[(SCREEN / 2) * SCREEN + SCREEN / 2]
352}
353
354fn w_phases(
355    [dx, dy]: [f64; 2],
356    scale: f64,
357    plane: usize,
358) -> impl Iterator<Item = (usize, Complex32)> {
359    let half_window = (SCREEN / SAMPLING / 2) as isize;
360    let origin = (SCREEN / 2) as isize;
361    let index = plane as f64;
362    let phase_factor = std::f64::consts::TAU * index * index / scale;
363    (-half_window..half_window).flat_map(move |y| {
364        let m_squared = (y as f64 * dy).powi(2);
365        (-half_window..half_window).filter_map(move |x| {
366            let radius_squared = (x as f64 * dx).powi(2) + m_squared;
367            (radius_squared < 1.).then(|| {
368                let phase = phase_factor * ((1. - radius_squared).sqrt() - 1.);
369                let (sin, cos) = phase.sin_cos();
370                let index = (origin + y) as usize * SCREEN + (origin + x) as usize;
371                (index, Complex32::new(cos as f32, sin as f32))
372            })
373        })
374    })
375}

377struct ApertureBeam {
378    size: usize,
379    jones: Vec<Complex32>,
380    fft: PlaneFft<f32>,
381}
382
383fn aperture_beam(
384    aperture: &EvlaApertureModel,
385    size: usize,
386    sky_cell: f64,
387    frequency_hz: f64,
388    angle_rad: f64,
389) -> Result<ApertureBeam, OperatorError> {
390    let cell_m = SPEED_OF_LIGHT_M_PER_S / (frequency_hz * size as f64 * sky_cell.abs());
391    let grid = EvlaApertureGrid::new(size, cell_m, 3, frequency_hz, angle_rad)
392        .map_err(|_| invalid("invalid legacy EP aperture geometry"))?;
393    let mut jones = vec![Complex32::default(); grid.workspace_values()];
394    aperture
395        .fill_vla_l_aperture(grid, &mut jones)
396        .map_err(|_| invalid("cannot evaluate the legacy VLA aperture"))?;
397    let fft = PlaneFft::new([size, size], false)?;
398    Ok(ApertureBeam { size, jones, fft })
399}

401impl ApertureBeam {
402    fn power(&mut self, squint: bool) -> Result<Vec<Complex32>, OperatorError> {
403        let plane_size = self.size * self.size;
404        let center = self.size / 2;
405        let index = center * self.size + center;
406        let mut normalization = 0.;
407        let mut rr = Vec::new();
408        for (pol, source) in self.jones.chunks_exact(plane_size).enumerate() {
409            let mut plane = source.to_vec();
410            if !squint && pol == 0 {
411                // The legacy loop removes the R phase in place starting at
412                // R itself. Later planes see R with its phase already removed.
413                for value in &mut plane {
414                    *value = Complex32::new(value.norm(), 0.);
415                }
416            }
417            self.fft.transform(&mut plane, false)?;
418            normalization += 0.5 * plane[index].norm_sqr();
419            if pol == 0 {
420                rr = plane;
421            }
422        }
423        if !normalization.is_finite() || normalization <= 0. {
424            return Err(invalid("legacy EP beam normalization is zero"));
425        }
426        for value in &mut rr {
427            *value = Complex32::new(value.norm_sqr() / normalization, 0.);
428        }
429        Ok(rr)
430    }
431}

433fn kernel_support(kernel: &[Complex32], threshold: f32) -> Result<usize, OperatorError> {
434    let origin = SCREEN as f64 * 0.5;
435    for radius in (2..=SCREEN / 4).rev() {
436        // The original repeatedly sweeps the same circle at angles 2πi/R.
437        for step in 0..radius {
438            let (sin, cos) = (std::f64::consts::TAU * step as f64 / radius as f64).sin_cos();
439            let x = (origin + radius as f64 * sin) as usize;
440            let y = (origin + radius as f64 * cos) as usize;
441            if kernel[y * SCREEN + x].norm() > threshold {
442                return Ok((radius + SAMPLING / 2) / SAMPLING + 1);
443            }
444        }
445    }
446    Err(invalid("legacy EP convolution has no support"))
447}

449fn crop(kernel: &[Complex32], radius: usize) -> Vec<Complex32> {
450    let start = SCREEN / 2 - radius;
451    let end = SCREEN / 2 + radius;
452    let mut cropped = Vec::with_capacity((2 * radius + 1).pow(2));
453    for y in start..=end {
454        cropped.extend_from_slice(&kernel[y * SCREEN + start..=y * SCREEN + end]);
455    }
456    cropped
457}
458
459fn sampled_integral(kernel: &[Complex32], support: usize) -> f32 {
460    let origin = SCREEN / 2;
461    let start = origin - support * SAMPLING;
462    let end = origin + support * SAMPLING;
463    let mut sum = 0.;
464    for y in (start..end).step_by(SAMPLING) {
465        for x in (start..end).step_by(SAMPLING) {
466            sum += kernel[y * SCREEN + x].re;
467        }
468    }
469    sum
470}
471
472fn sample_cropped(kernel: &[Complex32], side: usize, support: usize) -> Result<f32, OperatorError> {
473    // CASA uses the peak's X coordinate for both sampling axes.
474    let origin = peak_index(kernel) % side;
475    let radius = support * SAMPLING;
476    debug_assert!(radius >= SAMPLING);
477    if origin < radius || origin + radius - SAMPLING >= side {
478        return Err(invalid(
479            "legacy EP convolution peak is outside its sampling area",
480        ));
481    }
482    let bounds = origin - radius..origin + radius;
483    let mut integral = 0.;
484    for y in bounds.clone().step_by(SAMPLING) {
485        for x in bounds.clone().step_by(SAMPLING) {
486            integral += kernel[y * side + x].re;
487        }
488    }
489    Ok(integral)
490}
491
492fn peak_index(samples: &[Complex32]) -> usize {
493    let mut peak = 0.;
494    let mut best = 0;
495    for (index, sample) in samples.iter().enumerate() {
496        let magnitude = sample.norm_sqr();
497        if magnitude > peak {
498            peak = magnitude;
499            best = index;
500        }
501    }
502    best
503}

535        let surface =
536            std::fs::read_to_string(std::env::var_os("CASARS_EP_VLA_SURFACE").unwrap()).unwrap();

577        eprintln!("avgPB normalized RMS {difference:.9e}");

579    }

581    fn nrms(actual: &[Complex32], expected: &[Complex32]) -> f64 {
582        assert_eq!(actual.len(), expected.len());
583        let squared = actual
584            .iter()
585            .zip(expected)
586            .map(|(&left, &right)| f64::from((left - right).norm_sqr()))
587            .sum::<f64>();
588        (squared
589            / expected
590                .iter()
591                .map(|value| f64::from(value.norm_sqr()))
592                .sum::<f64>())
593        .sqrt()
594    }
