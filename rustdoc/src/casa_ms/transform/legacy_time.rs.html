2//! Chunk geometry and baseline identity used by the legacy visibility iterator.
3
4use std::collections::{BTreeMap, HashSet};
5use std::path::Path;
6
7use casa_tables::{RequiredScalarColumnValues, TableError};
8
9use crate::MeasurementSet;
10use crate::selection::ResolvedMsSelectionRow;
11
12use super::time_average::{TimeChunk, TimeGroup, TimeInputRow, TimeSample};
13use super::{LegacyTimeOptions, MsTransformError, TimeAverageOptions};
14
15struct Sample {
16    time: TimeSample,
17    interval: f64,
18    baseline: [i32; 5],
19}
20
21pub(super) fn plan(
22    input: &MeasurementSet,
23    rows: &[ResolvedMsSelectionRow],
24    valid: Vec<usize>,
25    time: TimeAverageOptions,
26    legacy: LegacyTimeOptions,
27) -> Result<Vec<TimeGroup>, MsTransformError> {
28    let path = input.path().unwrap_or_else(|| Path::new("<memory>"));
29    let error = |source| MsTransformError::MutateMeasurementSet {
30        path: path.display().to_string(),
31        source: Box::new(source),
32    };
33    let physical = rows.iter().map(|row| row.row_index).collect::<Vec<_>>();
34    let values = input
35        .main_table()
36        .required_scalar_columns_owned_for_rows(
37            &[
38                "ARRAY_ID",
39                "SCAN_NUMBER",
40                "STATE_ID",
41                "OBSERVATION_ID",
42                "PROCESSOR_ID",
43                "FEED1",
44                "FEED2",
45                "INTERVAL",
46            ],
47            &physical,
48        )
49        .map_err(error)?;
50    let integer = |name| match &values[name] {
51        RequiredScalarColumnValues::Int32(values) => Ok(values.as_slice()),
52        _ => Err(error(TableError::Schema(format!("{name} must hold Int32")))),
53    };
54    let RequiredScalarColumnValues::Float64(interval) = &values["INTERVAL"] else {
55        return Err(error(TableError::Schema(
56            "INTERVAL must hold Float64".into(),
57        )));
58    };
59    let arrays = integer("ARRAY_ID")?;
60    let scans = integer("SCAN_NUMBER")?;
61    let states = integer("STATE_ID")?;
62    let observations = integer("OBSERVATION_ID")?;
63    let proc = integer("PROCESSOR_ID")?;
64    let feed1 = integer("FEED1")?;
65    let feed2 = integer("FEED2")?;
66    let mut samples = rows
67        .iter()
68        .enumerate()
69        .map(|(index, row)| Sample {
70            time: TimeSample {
71                row: row.row_index,
72                description: row.data_desc_id,
73                time: row.time_mjd_seconds,
74                chunk: TimeChunk([
75                    arrays[index],
76                    if time.span_scans { 0 } else { scans[index] },
77                    if time.span_states { 0 } else { states[index] },
78                    row.field_id,
79                    row.data_desc_id,
80                    if legacy.combine_observations {
81                        0
82                    } else {
83                        observations[index]
84                    },
85                    0,
86                    0,
87                ]),
88                antenna_pair: [row.antenna1_id, row.antenna2_id],
89                uvw: [0.0; 3],
90            },
91            interval: interval[index],
92            baseline: [
93                proc[index],
94                row.antenna1_id,
95                row.antenna2_id,
96                feed1[index],
97                feed2[index],
98            ],
99        })
100        .collect::<Vec<_>>();
101    samples.sort_unstable_by(|left, right| {
102        left.time
103            .chunk
104            .cmp(&right.time.chunk)
105            .then_with(|| left.time.time.total_cmp(&right.time.time))
106            .then_with(|| left.time.row.cmp(&right.time.row))
107    });
108    let valid = valid.into_iter().collect::<HashSet<_>>();
109    let mut pending = samples.as_slice();
110    let mut output = Vec::new();
111    while let Some(first) = pending.first() {
112        let offset = first.time.time - 0.01;
113        let end = pending
114            .partition_point(|sample| {
115                sample.time.chunk == first.time.chunk
116                    && (sample.time.time == first.time.time
117                        || sample.time.time - offset < time.interval_seconds)
118            })
119            .max(1);
120        append_chunk(&pending[..end], &valid, &mut output);
121        pending = &pending[end..];
122    }
123    Ok(output)
124}
125
126fn append_chunk(samples: &[Sample], valid: &HashSet<usize>, output: &mut Vec<TimeGroup>) {
127    let Some(first) = samples
128        .iter()
129        .find(|sample| valid.contains(&sample.time.row))
130    else {
131        return;
132    };
133    let last = samples
134        .iter()
135        .rev()
136        .find(|sample| valid.contains(&sample.time.row))
137        .expect("at least one valid integration");
138    let elapsed = last.time.time - first.time.time;
139    let geometry = (
140        first.time.time + 0.5 * elapsed,
141        elapsed + 0.5 * (first.interval + last.interval),
142    );
143    let mut baselines = BTreeMap::new();
144    let mut previous = None;
145    let mut buffer_first = first.time.row;
146    for sample in samples {
147        if previous != Some(sample.time.time) {
148            buffer_first = sample.time.row;
149            previous = Some(sample.time.time);
150        }
151        if !valid.contains(&sample.time.row) {
152            continue;
153        }
154        let (_, rows) = baselines
155            .entry(sample.baseline)
156            .or_insert_with(|| (sample.time, Vec::new()));
157        rows.push(TimeInputRow {
158            physical: sample.time.row,
159            buffer_first,
160        });
161    }
162    output.extend(
163        baselines
164            .into_values()
165            .enumerate()
166            .map(|(index, (first, rows))| TimeGroup::legacy(first, rows, index, geometry)),
167    );
168}
