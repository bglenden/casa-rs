2//! MAIN's rows at `close`: each write batch's geometry, prediction and noise
3//! computed on the session's worker team, then handed to the writer's
4//! thread.
5//!
6//! Per batch of the write plan's rows (whole bands of DATA tiles):
7//! 1. **Geometry.** The integrations the batch reaches beyond the rows the
8//!    last batch left are computed, one job per integration:
9//!    [`ArrayGeometry::integration_rows`] depends only on the scan and the
10//!    time.
11//! 2. **Prediction.** DATA is zeroed in the writer's recycled batch, and
12//!    the rows of scans a `predict` covers are predicted into it, in blocks
13//!    on the team ([`super::prediction`]).
14//! 3. **Cells and columns.** Then, on the team at once:
15//!    - each scan's noise is added to its rows, a run of rows per job: a
16//!      draw depends only on the term, the MAIN row, the channel and the
17//!      correlation ([`crate::corrupt`]);
18//!    - one job copies FLAG_ROW, UVW, WEIGHT and SIGMA from the geometry and
19//!      pushes the scalar columns of the integrations computed for the
20//!      batch, in MAIN order.
21//! 4. **Send.** The writer's thread writes the batch while the next one is
22//!    computed.
23//!
24//! On Metal the device adds the noise of the rows it predicts as it
25//! predicts them, before their model is copied out
26//! ([`super::prediction`]), and step 3 adds noise only to the rows no
27//! `predict` covers. (Predicting a batch into cells of its own while its
28//! noise was added, then adding the two, was measured on the CPU and
29//! deleted: no faster on either test host, and 8% more CPU, #855.)
30//!
31//! The draws and, on the CPU, the predictions depend on no worker, batch or
32//! block, so on the CPU the MeasurementSet is bitwise the same for any
33//! worker count or memory budget. Metal's predictions agree with the CPU's
34//! to the device's `f32` arithmetic, and how rows are split among its
35//! command buffers changes their rounding, so on Metal it is the same to
36//! that rounding (about 1e-7 relative). An integration that straddles two
37//! batches is computed once: its unwritten rows open the next batch.

53/// Jobs per worker a batch's noise is split into, so that workers
54/// finishing early take more.
55const NOISE_JOBS_PER_WORKER: usize = 4;

57/// The fewest rows of one noise job.
58const MINIMUM_NOISE_JOB_ROWS: usize = 16;

69/// Where `close`'s time went, for the pass's summary.

71pub(super) struct StageTimes {

84/// The rows of the batch in hand and what fills their cells and columns.
85struct Rows<'s> {
86    scans: &'s [Scan],
87    geometry: &'s ArrayGeometry<'s>,
88    team: &'s WorkerTeam,
89    /// The session's DATA steps, which give each scan's noise.
90    plan: &'s DataPlan,
91    correlations: usize,
92    samples: usize,
93    /// Rows of the integrations in hand: those the last batch left
94    /// unwritten first, then those computed for this one.
95    rows: Vec<RowGeometry>,
96    /// The scan of the rows the last batch left.
97    carried_scan: usize,
98    /// The integrations computed for this batch.
99    integrations: Vec<Integration>,
100}

102/// The pipeline's state across batches.
103pub(super) struct Pipeline<'s> {
104    rows: Rows<'s>,
105    /// The model's prediction, when a `predict` covers any scan.
106    prediction: Option<PredictionStage<'s>>,
107    pub(super) times: StageTimes,
108}

110impl<'s> Pipeline<'s> {
111    /// The pipeline for `scans`, predicting with `prediction` and adding
112    /// the noise of `plan`, on cells of `(correlations, channels)`.
113    pub(super) fn new(
114        scans: &'s [Scan],
115        geometry: &'s ArrayGeometry<'s>,
116        team: &'s WorkerTeam,
117        (prediction, plan): (Option<PredictionStage<'s>>, &'s DataPlan),
118        (correlations, channels): (usize, usize),
119    ) -> Self {
120        Self {
121            rows: Rows {
122                scans,
123                geometry,
124                team,
125                plan,
126                correlations,
127                samples: correlations * channels,
128                rows: Vec::new(),
129                carried_scan: 0,
130                integrations: Vec::new(),
131            },
132            prediction,
133            times: StageTimes::default(),
134        }
135    }
136
137    /// Time the prediction spent filling blocks of native rows.
138    pub(super) fn block_fill_time(&self) -> Duration {
139        self.prediction
140            .as_ref()
141            .map_or(Duration::ZERO, PredictionStage::fill_time)
142    }
143
144    /// Write every row of the scans: `next` yields each integration with
145    /// rows, as (scan, stored TIME), in MAIN order.
146    pub(super) fn write_rows(
147        &mut self,
148        writer: &mut MainWriter,
149        mut next: impl Iterator<Item = (usize, f64)>,
150        row_count: usize,
151        batch_rows: usize,
152    ) -> SimulatorResult<()> {
153        let mut first_row = 0;
154        while first_row < row_count {
155            let rows = batch_rows.min(row_count - first_row);
156            let started = Instant::now();
157            self.rows.compute_geometry(&mut next, rows)?;
158            self.times.geometry += started.elapsed();
159            let started = Instant::now();
160            let mut batch = writer.visibility_columns();
161            batch
162                .data
163                .resize(rows * self.rows.samples, Complex32::new(0.0, 0.0));
164            self.times.zero += started.elapsed();
165            let runs = self.rows.segments(rows);
166            let first = first_row as u64;
167            let started = Instant::now();
168            if let Some(stage) = &mut self.prediction {
169                let rows = &self.rows;
170                stage.predict(
171                    rows.team,
172                    (rows.scans, &rows.rows),
173                    &runs,
174                    first,
175                    &mut batch.data,
176                    |scan| rows.plan.noise_for_scan(scan),
177                )?;
178            }
179            if self.rows.plan.predicts_single_dish() {
180                let rows = &self.rows;
181                super::single_dish::add_single_dish(
182                    (rows.scans, &rows.rows),
183                    rows.plan,
184                    &runs,
185                    (&mut batch.data, rows.samples),
186                );
187            }
188            self.times.predict += started.elapsed();
189            let started = Instant::now();
190            // A run the stage predicted on the device has its noise.
191            let (prediction, plan) = (&self.prediction, self.rows.plan);
192            let noised = |scan| {
193                prediction.as_ref().is_some_and(|stage| {
194                    stage.predicts(scan) && stage.noises_on_device(plan.noise_for_scan(scan))
195                })
196            };
197            self.rows.fill(writer, &mut batch, (first, rows), &noised)?;
198            self.times.cells += started.elapsed();
199            let started = Instant::now();
200            writer.send(batch)?;
201            self.times.send += started.elapsed();
202            self.times.batches += 1;
203            self.rows.carry(rows);
204            first_row += rows;
205        }
206        Ok(())
207    }
208}

210impl Rows<'_> {
211    /// Compute the integrations the next `rows` rows reach beyond those
212    /// carried.
213    fn compute_geometry(
214        &mut self,
215        next: &mut impl Iterator<Item = (usize, f64)>,
216        rows: usize,
217    ) -> SimulatorResult<()> {
218        self.integrations.clear();
219        let mut covered = self.rows.len();
220        while covered < rows {
221            let (scan, time_s) = next
222                .next()
223                .ok_or_else(|| SimulatorError::raised("the scans have fewer rows than planned"))?;
224            self.integrations.push(Integration {
225                scan,
226                time_s,
227                first: covered,
228            });
229            covered += self.scans[scan].baseline_count();
230        }
231        let carried = self.rows.len();
232        self.rows.resize(covered, RowGeometry::default());
233        let mut jobs = Vec::with_capacity(self.integrations.len());
234        let mut rest = &mut self.rows[carried..];
235        for integration in &self.integrations {
236            let (rows, after) = rest.split_at_mut(self.scans[integration.scan].baseline_count());
237            jobs.push((*integration, rows));
238            rest = after;
239        }
240        let (scans, geometry) = (self.scans, self.geometry);
241        self.team.for_each_mut(&mut jobs, |_, (integration, rows)| {
242            geometry.integration_rows(&scans[integration.scan], integration.time_s, rows)
243        })
244    }
245
246    /// Fill the batch's `rows` rows, the first of which is MAIN row
247    /// `first_row`: the noise of DATA but for the scans `noised` already
248    /// has, the per-row columns, and the scalar columns of the integrations
249    /// computed for the batch.
250    fn fill(
251        &self,
252        writer: &mut MainWriter,
253        batch: &mut VisibilityColumns,
254        (first_row, rows): (u64, usize),
255        noised: &dyn Fn(usize) -> bool,
256    ) -> SimulatorResult<()> {
257        let VisibilityColumns {
258            data,
259            flag_row,
260            uvw,
261            weight,
262            sigma,
263        } = batch;
264        let mut jobs = vec![BatchJob::Columns(ColumnsJob {
265            writer,
266            flag_row,
267            uvw,
268            weight,
269            sigma,
270        })];
271        self.noise_jobs((first_row, rows), data, noised, &mut jobs);
272        let (scans, integrations, geometry) = (self.scans, &self.integrations, &self.rows);
273        let (correlations, samples) = (self.correlations, self.samples);
274        self.team.for_each_mut(&mut jobs, |_, job| match job {
275            BatchJob::Columns(job) => job.run(&geometry[..rows], scans, integrations, geometry),
276            BatchJob::Noise(job) => {
277                job.run(correlations, samples);
278                Ok(())
279            }
280        })
281    }
282
283    /// Rows of one noise job for a batch of `rows` rows.
284    fn job_rows(&self, rows: usize) -> usize {
285        rows.div_ceil(self.team.workers() * NOISE_JOBS_PER_WORKER)
286            .max(MINIMUM_NOISE_JOB_ROWS)
287    }
288
289    /// Split the batch's `rows` rows of `data` into noise jobs, runs of one
290    /// scan with noise that `noised` does not already have.
291    fn noise_jobs<'a>(
292        &'a self,
293        (first_row, rows): (u64, usize),
294        data: &'a mut [Complex32],
295        noised: &dyn Fn(usize) -> bool,
296        jobs: &mut Vec<BatchJob<'a>>,
297    ) {
298        let job_rows = self.job_rows(rows);
299        let mut rest = data;
300        for (scan, start, count) in self.segments(rows) {
301            let (segment, after) = rest.split_at_mut(count * self.samples);
302            rest = after;
303            let noise = self.plan.noise_for_scan(scan);
304            if noise.is_empty() || noised(scan) {
305                continue;
306            }
307            for (index, cells) in segment.chunks_mut(job_rows * self.samples).enumerate() {
308                let start = start + index * job_rows;
309                jobs.push(BatchJob::Noise(NoiseJob {
310                    first_row: first_row + start as u64,
311                    cells,
312                    rows: &self.rows[start..],
313                    noise,
314                }));
315            }
316        }
317    }
318
319    /// The batch's first `rows` rows as runs of one scan: (scan, first row,
320    /// rows).
321    fn segments(&self, rows: usize) -> Vec<(usize, usize, usize)> {
322        let carried = self
323            .integrations
324            .first()
325            .map_or(self.rows.len(), |integration| integration.first);
326        let mut segments: Vec<(usize, usize, usize)> = Vec::new();
327        let mut push = |scan: usize, start: usize, end: usize| {
328            let end = end.min(rows);
329            if start >= end {
330                return;
331            }
332            match segments.last_mut() {
333                Some((last, _, count)) if *last == scan => *count += end - start,
334                _ => segments.push((scan, start, end - start)),
335            }
336        };
337        push(self.carried_scan, 0, carried);
338        for integration in &self.integrations {
339            let end = integration.first + self.scans[integration.scan].baseline_count();
340            push(integration.scan, integration.first, end);
341        }
342        segments
343    }
344
345    /// Keep the rows after the batch's first `rows` for the next batch.
346    fn carry(&mut self, rows: usize) {
347        if let Some(last) = self.integrations.last() {
348            self.carried_scan = last.scan;
349        }
350        self.rows.drain(..rows);
351    }
352}
353
354/// A job of a batch's fill.
355enum BatchJob<'a> {
356    Columns(ColumnsJob<'a>),
357    Noise(NoiseJob<'a>),
358}
359
360/// The batch's per-row columns and the scalar columns of its integrations.
361struct ColumnsJob<'a> {
362    writer: &'a mut MainWriter,
363    flag_row: &'a mut Vec<bool>,
364    uvw: &'a mut Vec<[f64; 3]>,
365    weight: &'a mut Vec<f32>,
366    sigma: &'a mut Vec<f32>,
367}
368
369impl ColumnsJob<'_> {
370    /// Copy the columns of `rows`, the batch's, and push the scalar columns
371    /// of `integrations`, whose rows are in `geometry`.
372    fn run(
373        &mut self,
374        rows: &[RowGeometry],
375        scans: &[Scan],
376        integrations: &[Integration],
377        geometry: &[RowGeometry],
378    ) -> SimulatorResult<()> {
379        for row in rows {
380            self.flag_row.push(row.flagged);
381            self.uvw.push(row.uvw_m);
382            self.weight.push(row.weight);
383            self.sigma.push(row.sigma);
384        }
385        for integration in integrations {
386            let scan = &scans[integration.scan];
387            let rows = &geometry[integration.first..integration.first + scan.baseline_count()];
388            self.writer.push_scalars(scan, integration.time_s, rows)?;
389        }
390        Ok(())
391    }
392}

394/// One noise job: consecutive rows of one scan.
395struct NoiseJob<'a> {
396    /// The MAIN row of the first cell.
397    first_row: u64,
398    cells: &'a mut [Complex32],
399    /// The geometry of the rows, from the first.
400    rows: &'a [RowGeometry],
401    noise: &'a [NoiseApplication],
402}
403
404impl NoiseJob<'_> {
405    /// Add the noise to every cell of `samples` samples of `correlations`
406    /// correlations.
407    fn run(&mut self, correlations: usize, samples: usize) {
408        for (offset, (cell, row)) in self
409            .cells
410            .chunks_exact_mut(samples)
411            .zip(self.rows)
412            .enumerate()
413        {
414            for application in self.noise {
415                application.add_to_row(
416                    self.first_row + offset as u64,
417                    row.antenna1 == row.antenna2,
418                    correlations,
419                    cell,
420                );
421            }
422        }
423    }
