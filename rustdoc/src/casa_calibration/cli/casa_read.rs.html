2//! Reads of the CASA-named calibration surfaces, `applycal`, `gaincal`,
3//! `bandpass`, `fluxscale` and `gencal`, into their requests (#931).
4//!
5//! These surfaces take CASA 6.7.6's parameters and values, read as CASA's
6//! task scripts read them:
7//!
8//! - `field`, `spw` and `intent` always select; the other selectors only
9//!   while `selectdata` is set, as CASA resets them otherwise.
10//! - The chain is `callib` when `docallib` is set, else `gaintable`. Its
11//!   per-table lists are positional: `gaintable[i]` takes `gainfield[i]`,
12//!   `interp[i]` and `spwmap[i]`, and a list shorter than `gaintable` leaves
13//!   the later tables CASA's defaults (every field, `'linear'`, no window
14//!   map). An empty `gaintable` entry is skipped. `calwt` with one value
15//!   holds for every table; a solve applies each table with `calwt=True`.
16//!
17//! A value CASA takes that casa-rs does not implement yet is refused naming
18//! #979; the session already refuses the parameters casa-rs does not
19//! implement at all unless they hold CASA's default.
20

37/// The request a CASA-named surface makes.
38pub(super) fn task_request(
39    task: &TaskInvocation<'_>,
40) -> Result<CalibrationTaskRequest, TaskFailure> {
41    Ok(match task.surface() {
42        "applycal" => CalibrationTaskRequest::ExecuteApply(applycal(task)?),
43        "gaincal" => CalibrationTaskRequest::SolveGain(gaincal(task)?),
44        "bandpass" => CalibrationTaskRequest::SolveBandpass(bandpass(task)?),
45        "fluxscale" => CalibrationTaskRequest::FluxScale(fluxscale(task)?),
46        "gencal" => CalibrationTaskRequest::Gencal(gencal(task)?),
47        other => {
48            return Err(TaskFailure::validation(format!(
49                "calibrate runs no surface {other:?}"
50            )));
51        }
52    })
53}

55fn applycal(task: &TaskInvocation<'_>) -> Result<ExecuteApplyTaskRequest, TaskFailure> {
56    let (calibration_tables, callibrary_error, argument_error) = if task.flag("docallib")? {
57        let source = task.text("callib")?.unwrap_or_default();
58        match load_apply_specs_from_callib(Path::new(&source)) {
59            Ok(tables) => (tables, None, None),
60            Err(error) => (Vec::new(), Some(error.to_string()), None),
61        }
62    } else {
63        let (tables, error) = traditional_chain(task, Chain::Apply)?;
64        (tables, None, error)
65    };
66    Ok(ExecuteApplyTaskRequest {
67        measurement_set: task.path("vis")?,
68        selection: selection(task)?,
69        calibration_tables,
70        callibrary_error,
71        argument_error,
72        apply_mode: applymode(task)?,
73        parang: task.flag("parang")?,
74        flagbackup: task.flag("flagbackup")?,
75    })
76}

78fn gaincal(task: &TaskInvocation<'_>) -> Result<SolveGainTaskRequest, TaskFailure> {
79    let measurement_set = task.path("vis")?;
80    let min_snr = min_snr(task)?;
81    let (model_source, smodel) = model(task, &measurement_set)?;
82    let reference = task.required_text("refant")?;
83    let strict = match task.required_text("refantmode")?.as_str() {
84        "flex" => false,
85        "strict" => true,
86        other => return Err(unknown("refantmode", other, "'flex' or 'strict'")),
87    };
88    Ok(SolveGainTaskRequest {
89        append: task.flag("append")?,
90        selection: selection(task)?,
91        output_table: PathBuf::from(task.required_text("caltable")?),
92        gain_type: gaintype(task)?,
93        solve_mode: calmode(task)?,
94        solve_interval: solint(task)?,
95        preaverage_seconds: task.number("preavg")?,
96        correlation_dependent_flags: task.flag("corrdepflags")?,
97        combine: combine(task)?,
98        refant: if reference.is_empty() {
99            RefAntSelector::Unspecified
100        } else {
101            RefAntSelector::Expression {
102                selection: reference,
103                strict,
104            }
105        },
106        prior_calibration_tables: chain(task, Chain::Solve)?,
107        parang: task.flag("parang")?,
108        model_source,
109        normalize_average_amplitude: task.flag("solnorm")?,
110        normalization_type: match task.required_text("normtype")?.as_str() {
111            "mean" => GainNormalization::Mean,
112            "median" => GainNormalization::Median,
113            other => return Err(unknown("normtype", other, "'mean' or 'median'")),
114        },
115        robustness: robust_gain_solve(task)?,
116        min_snr,
117        min_baselines_per_antenna: count(task, "minblperant")?,
118        smodel,
119        measurement_set,
120    })
121}

123fn bandpass(task: &TaskInvocation<'_>) -> Result<SolveBandpassTaskRequest, TaskFailure> {
124    let measurement_set = task.path("vis")?;
125    let (model_source, smodel) = model(task, &measurement_set)?;
126    if matches!(model_source, GainSolveModelSource::ModelColumn) {
127        return Err(not_yet("smodel", "[] with a MODEL_DATA column"));
128    }
129    let band_type = match task.required_text("bandtype")?.as_str() {
130        "B" => BandpassType::B,
131        "BPOLY" => BandpassType::BPoly,
132        other => return Err(unknown("bandtype", other, "'B' or 'BPOLY'")),
133    };
134    let combine = combine(task)?;
135    // CASA reads maskedge only for BPOLY.
136    let mask_edge_percent = match band_type {
137        BandpassType::B => BandpassSolveCuts::CASA_DEFAULTS.mask_edge_percent,
138        BandpassType::BPoly => {
139            let percent = task.number("maskedge")?;
140            if !(0.0..50.0).contains(&percent) {
141                return Err(TaskFailure::validation(
142                    "maskedge must be a percentage of the channels from 0 to below 50",
143                )
144                .at("maskedge"));
145            }
146            percent as f32
147        }
148    };
149    Ok(SolveBandpassTaskRequest {
150        append: task.flag("append")?,
151        selection: selection(task)?,
152        output_table: PathBuf::from(task.required_text("caltable")?),
153        refant: refant(&task.required_text("refant")?)?,
154        prior_calibration_tables: chain(task, Chain::Solve)?,
155        parang: task.flag("parang")?,
156        combine: BandpassSolveCombine {
157            scans: combine.scans,
158            fields: combine.fields,
159        },
160        band_type,
161        normalize_average_amplitude: task.flag("solnorm")?,
162        amplitude_degree: count(task, "degamp")?,
163        phase_degree: count(task, "degphase")?,
164        smodel,
165        cuts: Some(BandpassSolveCuts {
166            min_baselines_per_antenna: count(task, "minblperant")?,
167            min_snr: min_snr(task)?,
168            mask_edge_percent,
169        }),
170        measurement_set,
171    })
172}

174/// `minsnr`, which CASA takes as any number; casa-rs takes it finite and
175/// not negative.
176fn min_snr(task: &TaskInvocation<'_>) -> Result<f32, TaskFailure> {
177    let min_snr = task.number("minsnr")? as f32;
178    if !min_snr.is_finite() || min_snr < 0.0 {
179        return Err(
180            TaskFailure::validation("minsnr must be a finite non-negative number").at("minsnr"),
181        );
182    }
183    Ok(min_snr)
184}

186fn fluxscale(task: &TaskInvocation<'_>) -> Result<FluxScaleRequest, TaskFailure> {
187    // CASA opens `vis` for its field and window names; casa-rs reads them
188    // from the gain table, but a MeasurementSet that is not there is still
189    // an error, as it is to CASA.
190    let vis = task.path("vis")?;
191    if !vis.exists() {
192        return Err(TaskFailure::validation(format!(
193            "MeasurementSet {} does not exist",
194            vis.display()
195        ))
196        .at("vis"));
197    }
198    let reference_fields = fields(task, "reference")?;
199    if reference_fields.is_empty() {
200        return Err(
201            TaskFailure::validation("fluxscale requires at least one reference field")
202                .at("reference"),
203        );
204    }
205    let refspwmap = task
206        .integers("refspwmap")?
207        .into_iter()
208        .map(|id| i32::try_from(id).map_err(|_| out_of_range("refspwmap", id)))
209        .collect::<Result<Vec<_>, _>>()?;
210    let gainthreshold = task.number("gainthreshold")?;
211    Ok(FluxScaleRequest {
212        options: crate::FluxScaleOptions {
213            measurement_set: Some(vis),
214            antenna: task.text("antenna")?.unwrap_or_default(),
215            timerange: task.text("timerange")?.unwrap_or_default(),
216            scan: task.text("scan")?.unwrap_or_default(),
217            fitorder: i32::try_from(task.integer("fitorder")?).map_err(|_| {
218                TaskFailure::validation("fitorder exceeds CASA's integer range").at("fitorder")
219            })?,
220            listfile: task
221                .text("listfile")?
222                .filter(|text| !text.is_empty())
223                .map(PathBuf::from),
224            display: task.flag("display")?,
225        },
226        append: task.flag("append")?,
227        input_table: PathBuf::from(task.required_text("caltable")?),
228        output_table: PathBuf::from(task.required_text("fluxtable")?),
229        reference_fields,
230        transfer_fields: fields(task, "transfer")?,
231        // CASA's [-1] maps no window.
232        refspwmap: if refspwmap == [-1] {
233            Vec::new()
234        } else {
235            refspwmap
236        },
237        // CASA's -1 excludes no gain.
238        gainthreshold: (gainthreshold >= 0.0).then_some(gainthreshold),
239        incremental: task.flag("incremental")?,
240    })
241}

243fn gencal(task: &TaskInvocation<'_>) -> Result<GencalTaskRequest, TaskFailure> {
244    let caltype = match task.required_text("caltype")?.as_str() {
245        "" => {
246            return Err(TaskFailure::validation("gencal requires a caltype").at("caltype"));
247        }
248        "antpos" => GencalType::Antpos,
249        "gceff" => GencalType::Gceff,
250        "opac" => GencalType::Opac,
251        other => return Err(not_yet("caltype", other)),
252    };
253    let antenna = task.text("antenna")?.unwrap_or_default();
254    match caltype {
255        // CASA looks the corrections up (the VLA's position database, the
256        // ALMA tables) when no antenna is named.
257        GencalType::Antpos if antenna.trim().is_empty() => {
258            return Err(not_yet("antenna", &antenna));
259        }
260        // With no window named, CASA takes one opacity for all of them.
261        GencalType::Opac
262            if task.text("spw")?.is_none_or(|spw| spw.trim().is_empty())
263                && task.numbers("parameter")?.len() > 1 =>
264        {
265            return Err(TaskFailure::validation(
266                "opac with spw='' takes one opacity for every window; name the windows to give \
267                 one for each",
268            )
269            .at("parameter"));
270        }
271        _ => {}
272    }
273    Ok(GencalTaskRequest {
274        measurement_set: task.path("vis")?,
275        output_table: PathBuf::from(task.required_text("caltable")?),
276        caltype,
277        antenna,
278        spw: task.text("spw")?.unwrap_or_default(),
279        parameter: task.numbers("parameter")?,
280        gaincurve_table: task.text("gaincurve_table")?.map(PathBuf::from),
281    })
282}

284/// CASA's selection: `field`, `spw` and `intent` always; the others are inactive,
285/// and so unset, while `selectdata` is not.
286fn selection(task: &TaskInvocation<'_>) -> Result<MsSelection, TaskFailure> {
287    let full = task.surface() != "bandpass";
288    let text = |name: &str| {
289        task.text(name).map(|value| {
290            if full {
291                value.filter(|text| !text.is_empty())
292            } else {
293                value
294            }
295        })
296    };
297    let mut selection = MsSelection {
298        selectdata: true,
299        field: text("field")?,
300        spw: text("spw")?,
301        antenna: text("antenna")?,
302        scan: text("scan")?,
303        observation: if full {
304            selector_text(task.value("observation"))
305        } else {
306            text("observation")?
307        },
308        array: text("array")?,
309        timerange: text("timerange")?,
310        msselect: text("msselect")?,
311        ..MsSelection::default()
312    };
313    if !full {
314        return Ok(selection);
315    }
316    selection.intent = task.text("intent")?.filter(|text| !text.is_empty());
317    selection.uvrange = uv_selector_text(task.value("uvrange"));
318    // calibrater.selectvis converts these variants with toCasaString, which
319    // removes one outer bracket pair even when the input was a string.
320    for selector in [
321        &mut selection.field,
322        &mut selection.spw,
323        &mut selection.intent,
324        &mut selection.timerange,
325        &mut selection.uvrange,
326        &mut selection.antenna,
327        &mut selection.scan,
328        &mut selection.observation,
329    ] {
330        if let Some(text) = selector
331            && let Some(inner) = text
332                .strip_prefix('[')
333                .and_then(|text| text.strip_suffix(']'))
334        {
335            *selector = (!inner.is_empty()).then(|| inner.to_string());
336        }
337    }
338    // CASA accepts these parameters even when disabled, retaining their input
339    // values for HISTORY while passing empty selectors to calibrater.selectvis.
340    if !task.flag("selectdata")? {
341        selection.timerange = None;
342        selection.uvrange = None;
343        selection.antenna = None;
344        selection.scan = None;
345        selection.observation = None;
346        selection.msselect = None;
347    }
348    if task.surface() == "gaincal" {
349        // The public task always adds this predicate, including when the
350        // optional selectors are disabled.
351        selection.msselect = Some(match selection.msselect {
352            Some(predicate) => format!("({predicate}) && ANTENNA1!=ANTENNA2"),
353            None => "ANTENNA1!=ANTENNA2".into(),
354        });
355    }
356    Ok(selection)
357}

359fn selector_text(value: Option<&ParameterValue>) -> Option<String> {
360    let text = match value? {
361        ParameterValue::String(text) => text.clone(),
362        ParameterValue::NumpyScalar { value, .. } | ParameterValue::ScalarArray(value) => {
363            return selector_text(Some(value));
364        }
365        other => python_repr(other),
366    };
367    (!text.is_empty()).then_some(text)
368}
369
370fn uv_selector_text(value: Option<&ParameterValue>) -> Option<String> {
371    match value? {
372        ParameterValue::Array(values) => {
373            let text = values
374                .iter()
375                .filter_map(|value| selector_text(Some(value)))
376                .collect::<Vec<_>>()
377                .join(",");
378            (!text.is_empty()).then_some(text)
379        }
380        ParameterValue::EmptyArray { .. } => None,
381        value => selector_text(Some(value)),
382    }
383}

385/// Whether a chain is applied, each table with its own `calwt`, or
386/// pre-applied while solving, each with `calwt=True`.
387#[derive(Debug, Clone, Copy, PartialEq, Eq)]
388enum Chain {
389    Apply,
390    Solve,
391}
392
393/// The chain `callib` names when `docallib` is set, else `gaintable` with
394/// its positional lists.
395fn chain(
396    task: &TaskInvocation<'_>,
397    chain: Chain,
398) -> Result<Vec<ApplyCalibrationTableSpec>, TaskFailure> {
399    if task.flag("docallib")? {
400        let callib = task.text("callib")?.ok_or_else(|| {
401            TaskFailure::validation("docallib requires callib, a calibration library file")
402                .at("callib")
403        })?;
404        return load_apply_specs_from_callib(Path::new(&callib))
405            .map_err(|error| TaskFailure::validation(error.to_string()).at("callib"));
406    }
407    traditional_chain(task, chain).map(|(tables, _)| tables)
408}
409
410fn traditional_chain(
411    task: &TaskInvocation<'_>,
412    chain: Chain,
413) -> Result<(Vec<ApplyCalibrationTableSpec>, Option<ApplyArgumentError>), TaskFailure> {
414    let tables = task.texts("gaintable")?;
415    let gainfield = task.texts("gainfield")?;
416    let interp = task.texts("interp")?;
417    let spwmap = match apply_spw_maps(task.value("spwmap")) {
418        Ok(maps) => maps,
419        Err(error) if chain == Chain::Apply => return Ok((Vec::new(), Some(error))),
420        Err(error) => return Err(TaskFailure::validation(error.to_string()).at("spwmap")),
421    };
422    let calwt = match chain {
423        Chain::Apply => task.flags("calwt")?,
424        Chain::Solve => vec![true],
425    };
426    let mut specs = Vec::new();
427    for (index, table) in tables.iter().enumerate() {
428        if table.is_empty() {
429            continue;
430        }
431        let weight = match calwt.as_slice() {
432            [one] => *one,
433            values => match values.get(index) {
434                Some(weight) => *weight,
435                // Python indexes calwt immediately before setapply for an active
436                // table. All earlier table setup still takes place first.
437                None => return Ok((specs, Some(ApplyArgumentError::MissingCalwt))),
438            },
439        };
440        let mut spec = ApplyCalibrationTableSpec::new(table);
441        spec.gainfield = gainfield
442            .get(index)
443            .and_then(|field| parse_gainfield_value(field));
444        spec.interp = match interp.get(index) {
445            Some(text) => {
446                parse_interp_value(text).unwrap_or(crate::ApplyInterpolationMode::Invalid)
447            }
448            None => crate::ApplyInterpolationMode::Linear,
449        };
450        if let Some(map) = spwmap.get(index) {
451            match apply_spw_map(map) {
452                Ok(map) => spec.spwmap = map,
453                Err(error) if chain == Chain::Apply => return Ok((specs, Some(error))),
454                Err(error) => return Err(TaskFailure::validation(error.to_string()).at("spwmap")),
455            }
456        }
457        spec.calwt = weight;
458        specs.push(spec);
459    }
460    Ok((specs, None))
461}
462
463fn apply_spw_maps(
464    value: Option<&ParameterValue>,
465) -> Result<Vec<ParameterValue>, ApplyArgumentError> {
466    Ok(match value {
467        Some(ParameterValue::Array(items))
468            if !items.is_empty() && !matches!(items[0], ParameterValue::Array(_)) =>
469        {
470            vec![ParameterValue::Array(items.clone())]
471        }
472        Some(ParameterValue::Array(items)) => items.clone(),
473        Some(ParameterValue::String(text)) if text.is_empty() => Vec::new(),
474        Some(ParameterValue::Table(entries)) if entries.is_empty() => Vec::new(),
475        Some(ParameterValue::Table(_)) => return Err(ApplyArgumentError::SpwMap("0".into())),
476        Some(ParameterValue::EmptyArray { .. }) | None => Vec::new(),
477        Some(value) => {
478            let ty = match value {
479                ParameterValue::Integer(_) => "int",
480                ParameterValue::Float(_) => "float",
481                ParameterValue::Bool(_) => "bool",
482                _ => {
483                    return Err(ApplyArgumentError::SpwMap(
484                        "expected an integer window map".into(),
485                    ));
486                }
487            };
488            return Err(ApplyArgumentError::SpwMap(format!(
489                "object of type '{ty}' has no len()"
490            )));
491        }
492    })
493}
494
495fn apply_spw_map(value: &ParameterValue) -> Result<Vec<i32>, ApplyArgumentError> {
496    let ParameterValue::Array(items) = value else {
497        return Err(ApplyArgumentError::SpwMap(
498            "expected an integer window map".into(),
499        ));
500    };
501    let ids = items
502        .iter()
503        .map(|item| match item {
504            ParameterValue::Integer(id) => Ok(*id),
505            ParameterValue::Bool(id) => Ok(i64::from(*id)),
506            ParameterValue::NumpyScalar { value, .. }
507                if matches!(value.as_ref(), ParameterValue::Integer(_)) =>
508            {
509                let ParameterValue::Integer(id) = value.as_ref() else {
510                    unreachable!()
511                };
512                Ok(*id)
513            }
514            _ => Err(ApplyArgumentError::SpwMap(
515                "expected an integer window map".into(),
516            )),
517        })
518        .collect::<Result<Vec<_>, _>>()?;
519    window_map(&ids).map_err(|failure| ApplyArgumentError::SpwMap(failure.message))
520}
521
522fn window_map(map: &[i64]) -> Result<Vec<i32>, TaskFailure> {
523    if map == [-1] {
524        return Ok(Vec::new());
525    }
526    map.iter()
527        .map(|&id| i32::try_from(id).map_err(|_| out_of_range("spwmap", id)))
528        .collect()
529}

531/// Parse the point-source Stokes vector, with omitted Stokes set to zero.
532/// Gaincal resolves an empty vector from the selected fields of the MS.
533fn model(
534    task: &TaskInvocation<'_>,
535    measurement_set: &Path,
536) -> Result<(GainSolveModelSource, [f32; 4]), TaskFailure> {
537    let smodel = task.numbers("smodel")?;
538    match smodel.as_slice() {
539        [] if task.surface() == "gaincal" => {
540            std::fs::metadata(measurement_set).map_err(|error| {
541                TaskFailure::validation(format!(
542                    "open MeasurementSet {}: {error}",
543                    measurement_set.display()
544                ))
545                .at("vis")
546            })?;
547            Ok((
548                GainSolveModelSource::FromMeasurementSet,
549                [1.0, 0.0, 0.0, 0.0],
550            ))
551        }
552        [] => {
553            let ms = MeasurementSet::open(measurement_set).map_err(|error| {
554                TaskFailure::validation(format!(
555                    "open MeasurementSet {}: {error}",
556                    measurement_set.display()
557                ))
558                .at("vis")
559            })?;
560            let virtual_fields = casa_ms::scratch_columns::virtual_model_fields(&ms);
561            if !virtual_fields.is_empty() {
562                return Err(TaskFailure::validation(format!(
563                    "smodel=[] reads the virtual model of field(s) {virtual_fields:?} (setjy with \
564                     usescratch=False), which casa-rs does not implement yet (#979); give smodel, \
565                     or write MODEL_DATA"
566                ))
567                .at("smodel"));
568            }
569            let source = if has_model_column(&ms) {
570                GainSolveModelSource::ModelColumn
571            } else {
572                GainSolveModelSource::PointSource
573            };
574            Ok((source, [1.0, 0.0, 0.0, 0.0]))
575        }
576        &[i, ref polarized @ ..] if polarized.len() < 4 => {
577            if !(i > 0.0 && i.is_finite()) {
578                return Err(TaskFailure::validation(
579                    "smodel's Stokes I must be a positive finite flux density",
580                )
581                .at("smodel"));
582            }
583            if task.surface() == "bandpass" && polarized.iter().any(|stokes| *stokes != 0.0) {
584                return Err(not_yet("smodel", &format!("{smodel:?}")));
585            }
586            let mut stokes = [0.0; 4];
587            for (value, input) in stokes.iter_mut().zip(smodel) {
588                *value = input as f32;
589            }
590            Ok((GainSolveModelSource::PointSource, stokes))
591        }
592        _ => Err(TaskFailure::validation(format!(
593            "smodel takes [I, Q, U, V], or fewer with the rest zero, or [] for the MODEL_DATA \
594             column; found {} values",
595            smodel.len()
596        ))
597        .at("smodel")),
598    }
599}

601/// Whether `ms` has a `MODEL_DATA` column.
602fn has_model_column(ms: &MeasurementSet) -> bool {
603    ms.main_table()
604        .schema()

637fn robust_gain_solve(task: &TaskInvocation<'_>) -> Result<crate::GainSolveRobustness, TaskFailure> {
638    let mode = task.required_text("solmode")?;
639    let (l1, reject) = match mode.as_str() {
640        "" => (false, false),
641        "L1" => (true, false),
642        "R" => (false, true),
643        "L1R" => (true, true),
644        other => return Err(unknown("solmode", other, "'', 'L1', 'R' or 'L1R'")),
645    };
646    let mut rms_thresholds = Vec::new();
647    if reject {
648        rms_thresholds = task
649            .numbers("rmsthresh")?
650            .into_iter()
651            .map(|value| value as f32)
652            .collect();
653        // GJones::setSolve and TJones::setSolve supply these for an empty list.
654        if rms_thresholds.is_empty() {
655            rms_thresholds.extend([7.0, 5.0, 4.0, 3.5, 3.0, 2.8, 2.6, 2.4, 2.2, 2.5]);
656        }
657    }
658    Ok(crate::GainSolveRobustness { l1, rms_thresholds })
659}

661/// CASA's `solint`, read as `SolvableVisCal::setSolve` reads it: empty
662/// text, or text that contains `inf`, in any case, is one solution per
663/// scan; text that contains `int`, one per integration; anything else is a
664/// time, in seconds when it has no unit. A time of zero is CASA's `inf`,
665/// and a negative one its `int`. G and T have a single gain per spectral
666/// window, so CASA ignores the frequency interval after a comma.
667fn solint(task: &TaskInvocation<'_>) -> Result<GainSolveInterval, TaskFailure> {
668    let supplied = selector_text(task.value("solint")).unwrap_or_default();
669    let text = match supplied.split_once(',') {
670        Some((time, _)) if task.surface() == "gaincal" => time,
671        Some(_) => return Err(not_yet("solint", &supplied)),
672        None => &supplied,
673    };
674    let upper = text.to_ascii_uppercase();
675    if upper.is_empty() || upper.contains("INF") {
676        return Ok(GainSolveInterval::Infinite);
677    }
678    if upper.contains("INT") {
679        return Ok(GainSolveInterval::Integration);
680    }
681    let invalid = || unknown("solint", text, "'inf', 'int' or a time such as '60s'");
682    let quantity = text.parse::<Quantity>().map_err(|_| invalid())?;
683    let seconds = if quantity.unit().name().is_empty() {
684        quantity.value()
685    } else {
686        let second = Unit::new("s").map_err(|_| invalid())?;
687        quantity.get_value_in(&second).map_err(|_| invalid())?
688    };
689    if !seconds.is_finite() {
690        return Err(invalid());
691    }
692    Ok(if seconds > 0.0 {
693        GainSolveInterval::Seconds(seconds)
694    } else if seconds == 0.0 {
695        GainSolveInterval::Infinite
696    } else {
697        GainSolveInterval::Integration
698    })
699}

701/// The axes CASA's `combine` names, found as `SolvableVisCal` finds them:
702/// by the axis name anywhere in the text, in any case (`'Scan,Field'` and
703/// `'scan field'` both combine scans and fields).
704fn combine(task: &TaskInvocation<'_>) -> Result<GainSolveCombine, TaskFailure> {
705    let text = task.required_text("combine")?;
706    let upper = text.to_ascii_uppercase();
707    let spectral_windows = upper.contains("SPW");
708    let observations = upper.contains("OBS");
709    if task.surface() == "bandpass" && (spectral_windows || observations) {
710        return Err(not_yet("combine", &text));
711    }
712    Ok(GainSolveCombine {
713        scans: upper.contains("SCAN"),
714        fields: upper.contains("FIELD"),
715        observations,
716        spectral_windows,
717    })
718}
