2//! The primary beam `sm.setvp(dovp=True)` applies: CASA's default voltage
3//! patterns, as `PBMath` gives them by telescope name and frequency
4//! ([`PrimaryBeam`], from `casa-numerics`), unsquinted, which is what CASA
5//! applies to the simulator's one-plane Stokes I model.
6
7pub use casa_numerics::{BeamPattern, PrimaryBeam};
8use ndarray::Array2;
9
10use super::SkyModel;
11
12/// `pattern`'s power on model plane `plane`, `[y][x]`, centred on the pixel
13/// `centre`, at the plane's frequency (`BeamSkyJones::apply` squares the
14/// voltage pattern).
15///
16/// The radius is `PBMath1D::apply`'s: each axis's squared offset in degrees
17/// is formed in `Double` and kept as `Float`, their sum and its root are
18/// `Float`, and the root times 60 · ν/GHz is kept as `Float` before the
19/// lookup, so a radius at a table bin's edge falls as CASA's does.
20pub(super) fn power_plane(
21    pattern: &BeamPattern,
22    model: &SkyModel,
23    plane: usize,
24    centre: [f64; 2],
25) -> Array2<f32> {
26    let [nx, ny] = model.shape();
27    let frequency_hz = model.plane_frequency_hz(plane);
28    let factor = 60.0 * frequency_hz / 1.0e9;
29    let increment_deg = model.increment_rad.map(f64::to_degrees);
30    let squared = |pixel: usize, axis: usize| {
31        ((pixel as f64 - centre[axis]) * increment_deg[axis]).powi(2) as f32
32    };
33    let lookup = pattern.at_frequency(frequency_hz);
34    Array2::from_shape_fn((ny, nx), |(y, x)| {
35        let radius = (f64::from((squared(x, 0) + squared(y, 1)).sqrt()) * factor) as f32;
36        lookup.power(f64::from(radius))
37    })
38}
