13/// Intensity and mask sample returned alongside `ia.coordmeasures`.
14#[derive(Debug, Clone)]
15pub struct CoordMeasureIntensity {
16    /// Pixel intensity scalar (`0.0` when `pixel` is out of bounds).
17    pub value: PixelScalar,
18    /// Brightness unit (`""` when `pixel` is out of bounds).
19    pub unit: String,
20    /// Optional pixel validity mask (`None` when `pixel` is out of bounds).
21    pub mask: Option<bool>,
22}
23
24/// Samples single-precision intensity and mask for `ia.coordmeasures`.
25pub fn coord_measures_intensity(
26    attached: &AttachedImage,
27    pixel: &[f64],
28) -> ImageToolResult<CoordMeasureIntensity> {
29    let image = attached.lock();
30    image.check_single_precision("coordmeasures")?;
31    let requested: Vec<i64> = pixel.iter().map(|&axis| axis as i64).collect();
32    match &*image {
33        ImageHandle::Float(storage) => {
34            storage.with_float_view(|img| sample_intensity(img, requested, PixelScalar::Float))
35        }
36        ImageHandle::Complex(storage) => with_storage!(storage, img => {
37            sample_intensity(img.as_ref(), requested, |v| {
38                PixelScalar::DComplex(Complex64::new(f64::from(v.re), f64::from(v.im)))
39            })
40        }),
41        _ => unreachable!("double precision was rejected"),
42    }
43}
44
45fn sample_intensity<T: ImagePixel>(
46    image: &dyn ImageInterface<T>,
47    mut requested: Vec<i64>,
48    wrap: impl Fn(T) -> PixelScalar,
49) -> ImageToolResult<CoordMeasureIntensity> {
50    if requested.is_empty() {
51        requested = image
52            .coordinates()
53            .reference_pixel()
54            .into_iter()
55            .map(|pixel| pixel as i64)
56            .collect();
57    }
58    match pixel_position(image.coordinates(), image.shape(), &requested) {
59        Some(position) => {
60            let (sample, mask) = pixel_sample(image, &position)?;
61            Ok(CoordMeasureIntensity {
62                value: wrap(sample),
63                unit: image.units().to_owned(),
64                mask: Some(mask),
65            })
66        }
67        None => Ok(CoordMeasureIntensity {
68            value: wrap(T::default_value()),
69            unit: String::new(),
70            mask: None,
71        }),
72    }
73}

75/// Converts pixel coordinates to world coordinates using the attached image's [`CoordinateSystem`].
76pub fn to_world(
77    attached: &AttachedImage,
78    value: &[f64],
79    _format: &str,
80) -> ImageToolResult<Vec<f64>> {
81    let mut image = attached.lock();
82    attached.require_pixel_read_lock(&mut image)?;
83    let coords = image.coordinates();
84    let n_pix = coords.n_pixel_axes();
85    let mut pix = if value.is_empty() {
86        let mut refpix = vec![0.0; n_pix];
87        for idx in 0..coords.n_coordinates() {
88            let c = coords.coordinate(idx);
89            let rp = c.reference_pixel();
90            for (li, ax_opt) in coords
91                .coordinate_pixel_axes(idx)
92                .unwrap_or_default()
93                .into_iter()
94                .enumerate()
95            {
96                if let Some(ax) = ax_opt
97                    && ax < n_pix
98                {
99                    refpix[ax] = rp.get(li).copied().unwrap_or(0.0);
100                }
101            }
102        }
103        refpix
104    } else {
105        if value.len() > n_pix {
106            return Err(ImageToolError::new(format!(
107                "Pixel vector length ({}) exceeds pixel axes ({n_pix}).",
108                value.len()
109            )));
110        }
111        let mut v = value.to_vec();
112        v.resize(n_pix, 0.0);
113        v
114    };
115    let _ = &mut pix;
116    Ok(coords.to_world(&pix)?)
117}

119/// Converts world coordinates to pixel coordinates using the attached image's [`CoordinateSystem`].
120pub fn to_pixel(attached: &AttachedImage, value: &[f64]) -> ImageToolResult<Vec<f64>> {
121    let mut image = attached.lock();
122    attached.require_pixel_read_lock(&mut image)?;
123    let coords = image.coordinates();
124    let n_world = coords.n_world_axes();
125    if value.len() > n_world {
126        return Err(ImageToolError::new(format!(
127            "World vector length ({}) exceeds world axes ({n_world}).",
128            value.len()
129        )));
130    }
131    let mut world = value.to_vec();
132    if world.len() < n_world {
133        world.resize(n_world, 0.0);
134    }
135    Ok(coords.to_pixel(&world)?)
136}
