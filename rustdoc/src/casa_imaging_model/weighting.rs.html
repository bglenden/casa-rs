3//! The visibility-weighting contract: CASA's imaging-weight formulas
4//! (`SynthesisImagerVi2::weight`), the scope and cells of the visibility
5//! density the density formulas divide by, and the Gaussian uv taper
6//! (`VisImagingWeight::setFilter`).
7//!
8//! CASA's density formulas are one law, `w / (D·f² + d²)` over the gridded
9//! weight density `D` (`VisImagingWeight::weightUniform`,
10//! `BriggsCubeWeightor::getWeightUniform`): uniform is `f² = 1`, `d² = 0`,
11//! and Briggs's two robustness modes choose `f²` and `d²`. What CASA's
12//! `npixels` changes is the grid `D` lives on ([`DensityCells`]).
13
14use std::num::{NonZeroU32, NonZeroUsize};
15
16/// Visibility-weighting formula.
17#[derive(Debug, Clone, Copy, PartialEq)]
18pub enum WeightingScheme {
19    /// Natural weighting.
20    Natural,
21    /// CASA `radial`: each sample's weight times its uv distance in
22    /// wavelengths (`VisImagingWeight::weightRadial`); no density.
23    Radial,
24    /// Uniform density weighting, `w / D`: CASA `uniform`, and CASA
25    /// `superuniform` with a [`DensityCells::Box`] density.
26    Uniform,
27    /// Briggs robust weighting (CASA `rmode = "norm"`):
28    /// `f² = (5·10^{−robust})² / (ΣD² / ΣD)`, `d² = 1`.
29    Briggs {
30        /// Robustness in the conventional interval `[-2, 2]`.
31        robust: f64,
32    },
33    /// CASA `briggsabs` (`rmode = "abs"`): `f² = robust²` and
34    /// `d² = 2·noise²`, the expected noise of one visibility of unit weight.
35    BriggsAbsolute {
36        /// Robustness in the conventional interval `[-2, 2]`.
37        robust: f64,
38        /// The noise in Jy; it and `robust` are not both zero.
39        noise_jy: f64,
40    },
41    /// Briggs bandwidth-taper weighting.
42    BriggsBandwidthTaper {
43        /// Robustness in the conventional interval `[-2, 2]`.
44        robust: f64,
45    },
46}
47
48impl WeightingScheme {
49    /// Whether the formula divides by a gridded visibility density.
50    #[must_use]
51    pub const fn uses_density(self) -> bool {
52        !matches!(self, Self::Natural | Self::Radial)
53    }
54}
55
56/// Domain over which visibility-density weights are derived.
57#[derive(Debug, Clone, Copy, PartialEq, Eq)]
58pub enum WeightDensityScope {
59    /// The weighting formula does not use a density generation.
60    NotApplicable,
61    /// All selected data contributing to the logical product.
62    GlobalSelection,
63    /// An explicit global density generation per output channel.
64    PerOutputChannel,
65    /// One density per field over all its selected data, each with its own
66    /// Briggs factors: tclean's `mosweight` (`VisImagingWeight`'s
67    /// `multiField`, which keys a density by MeasurementSet and FIELD_ID).
68    /// The densities are those of FIELD_IDs `first_field` to
69    /// `first_field + fields − 1`; a field in the span with no data keeps an
70    /// empty density.
71    PerField {
72        /// The first FIELD_ID with a density.
73        first_field: u32,
74        /// The FIELD_IDs the densities span.
75        fields: NonZeroU32,
76    },
77}
78
79impl WeightDensityScope {
80    /// Whether one density covers all the data of what it weighs, a field or
81    /// the whole selection, rather than one output channel: the continuum
82    /// density of CASA's `VisImagingWeight`.
83    #[must_use]
84    pub const fn is_continuum(self) -> bool {
85        matches!(self, Self::GlobalSelection | Self::PerField { .. })
86    }
87}

89/// The cells a visibility density is gridded on: CASA's `npixels`, whose
90/// meaning `SynthesisImagerVi2::weight` takes from the weighting and the
91/// density scope.
92#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
93pub enum DensityCells {
94    /// The image's own uv cells; a sample adds to the cell it falls in.
95    #[default]
96    Image,
97    /// An `n × n` grid spanning the image's uv extent, so each cell is
98    /// `N / n` image uv cells wide along an image axis of `N` pixels: CASA's
99    /// continuum (`VisImagingWeight`) uniform and Briggs density with
100    /// `npixels = n`, its sidelobes suppressed over the inner `n` pixels.
101    Square(NonZeroUsize),
102    /// The image's uv cells, each sample spread over the `(2b + 1)²` cells
103    /// about its own: CASA `superuniform`'s box (`VisImagingWeight`'s
104    /// `uBox = vBox = b`, with `b = npixels`, or 3 when `npixels ≤ 0`), and a
105    /// cube's per-channel density with `npixels = b`
106    /// (`BriggsCubeWeightor`'s `superUniformBox`).
107    Box(NonZeroUsize),
108}

110/// Gaussian taper in the UV plane, expressed as baseline HWHM wavelengths.
111#[derive(Debug, Clone, Copy, PartialEq)]
112pub struct UvTaper {
113    major_lambda: f64,
114    minor_lambda: f64,
115    position_angle_rad: f64,
116}

122        Self {
123            major_lambda,
124            minor_lambda,
125            position_angle_rad,
126        }
127    }

129    /// Return the major-axis baseline HWHM in wavelengths.

133    }

148/// Complete logical visibility-weighting requirements.
149#[derive(Debug, Clone, Copy, PartialEq)]
150pub struct WeightingContract {
151    scheme: WeightingScheme,
152    density_scope: WeightDensityScope,
153    density_cells: DensityCells,
154    uv_taper: Option<UvTaper>,
155    casa_cube_density_padding: Option<usize>,
156}
157
158impl WeightingContract {
159    /// Construct weighting requirements on the image's own density cells.
160    #[must_use]
161    pub const fn new(scheme: WeightingScheme, density_scope: WeightDensityScope) -> Self {
162        Self {
163            scheme,
164            density_scope,
165            density_cells: DensityCells::Image,
166            uv_taper: None,
167            casa_cube_density_padding: None,
168        }
169    }
170
171    /// Grid the density on `cells` (CASA's `npixels`).
172    #[must_use]
173    pub const fn with_density_cells(mut self, cells: DensityCells) -> Self {
174        self.density_cells = cells;
175        self
176    }
177
178    /// Add a Gaussian UV taper to the weighting metric.
179    #[must_use]
180    pub const fn with_uv_taper(mut self, uv_taper: UvTaper) -> Self {
181        self.uv_taper = Some(uv_taper);
182        self
183    }
184
185    /// Bind CASA's cube density and native-weight transfer law, with a
186    /// metadata-derived number of density channels on each side of the image.
187    /// Zero padding still selects the cube law. The published axis is unchanged.
188    #[must_use]
189    pub const fn with_casa_cube_density_padding(mut self, channels_per_side: usize) -> Self {
190        self.casa_cube_density_padding = Some(channels_per_side);
191        self
192    }
193
194    /// Return the bound CASA cube law's symmetric density-axis padding.
195    #[must_use]
196    pub const fn casa_cube_density_padding(self) -> Option<usize> {
197        self.casa_cube_density_padding
198    }
199
200    /// Return the weighting formula.
201    #[must_use]
202    pub const fn scheme(self) -> WeightingScheme {
203        self.scheme
204    }
205
206    /// Return the density-generation scope.
207    #[must_use]
208    pub const fn density_scope(self) -> WeightDensityScope {
209        self.density_scope
210    }
211
212    /// Return the cells the density is gridded on.
213    #[must_use]
214    pub const fn density_cells(self) -> DensityCells {
215        self.density_cells
216    }
217
218    /// Return the optional Gaussian UV taper.
219    #[must_use]
220    pub const fn uv_taper(self) -> Option<UvTaper> {
221        self.uv_taper
222    }
223
224    /// Why the requirements are outside the logical domain, if they are.
225    pub(crate) fn validate(self) -> Result<(), &'static str> {
226        if self.casa_cube_density_padding.is_some()
227            && self.density_scope != WeightDensityScope::PerOutputChannel
228        {
229            return Err("CASA cube density requires per-output-channel weighting");
230        }
231        let density = self.scheme.uses_density();
232        if !density && self.density_scope != WeightDensityScope::NotApplicable {
233            return Err("natural and radial weighting have no density generation");
234        }
235        if density && self.density_scope == WeightDensityScope::NotApplicable {
236            return Err("density weighting requires an explicit global density scope");
237        }
238        match self.density_cells {
239            DensityCells::Image => {}
240            _ if !density => return Err("a weighting without a density has no density cells"),
241            DensityCells::Square(_) if !self.density_scope.is_continuum() => {
242                return Err("a square density grid is CASA's continuum density");
243            }
244            _ if matches!(self.scheme, WeightingScheme::BriggsBandwidthTaper { .. }) => {
245                return Err("CASA's bandwidth-tapered Briggs density has the image's cells");
246            }
247            DensityCells::Square(_) | DensityCells::Box(_) => {}
248        }
249        let robust = match self.scheme {
250            WeightingScheme::Briggs { robust }
251            | WeightingScheme::BriggsAbsolute { robust, .. }
252            | WeightingScheme::BriggsBandwidthTaper { robust } => Some(robust),
253            WeightingScheme::Natural | WeightingScheme::Radial | WeightingScheme::Uniform => None,
254        };
255        if robust.is_some_and(|value| !(value.is_finite() && (-2.0..=2.0).contains(&value))) {
256            return Err("Briggs robustness must be finite and in [-2, 2]");
257        }
258        if let WeightingScheme::BriggsAbsolute { robust, noise_jy } = self.scheme
259            && (!noise_jy.is_finite() || (robust == 0.0 && noise_jy == 0.0))
260        {
261            // CASA's denominator robust²·D + 2·noise² would vanish
262            // (`SynthesisImagerVi2::weight` refuses it for cubes).
263            return Err("absolute Briggs noise must be finite, and not zero with robust zero");
264        }
265        if self.uv_taper.is_some_and(|taper| {
266            !(taper.major_lambda.is_finite()
267                && taper.major_lambda > 0.0
268                && taper.minor_lambda.is_finite()
269                && taper.minor_lambda > 0.0
270                && taper.position_angle_rad.is_finite())
271        }) {
272            return Err("UV taper axes must be finite and positive and its angle must be finite");
273        }
274        Ok(())
275    }
276}

278#[cfg(test)]
279mod tests {
280    use super::*;
281
282    fn cells(count: usize) -> NonZeroUsize {
283        NonZeroUsize::new(count).expect("non-zero")
284    }
285
286    fn contract(scheme: WeightingScheme, scope: WeightDensityScope) -> WeightingContract {
287        WeightingContract::new(scheme, scope)
288    }
289
290    #[test]
291    fn density_cells_follow_casas_npixels_readings() {
292        use WeightDensityScope::{GlobalSelection, NotApplicable, PerOutputChannel};
293        let briggs = WeightingScheme::Briggs { robust: 0.5 };
294        let absolute = WeightingScheme::BriggsAbsolute {
295            robust: 1.0,
296            noise_jy: 0.1,
297        };
298        let per_field = WeightDensityScope::PerField {
299            first_field: 2,
300            fields: NonZeroU32::new(3).expect("non-zero"),
301        };
302        for valid in [
303            contract(WeightingScheme::Radial, NotApplicable),
304            // `mosweight`'s densities are continuum densities.
305            contract(briggs, per_field).with_density_cells(DensityCells::Square(cells(7))),
306            contract(WeightingScheme::Uniform, per_field)
307                .with_density_cells(DensityCells::Box(cells(3))),
308            contract(WeightingScheme::Uniform, GlobalSelection)
309                .with_density_cells(DensityCells::Square(cells(32))),
310            contract(briggs, GlobalSelection).with_density_cells(DensityCells::Square(cells(7))),
311            contract(absolute, GlobalSelection),
312            // Superuniform is uniform over a box.
313            contract(WeightingScheme::Uniform, GlobalSelection)
314                .with_density_cells(DensityCells::Box(cells(3))),
315            contract(absolute, PerOutputChannel)
316                .with_density_cells(DensityCells::Box(cells(2)))
317                .with_casa_cube_density_padding(4),
318        ] {
319            assert_eq!(valid.validate(), Ok(()), "{valid:?}");
320        }
321        for invalid in [
322            contract(WeightingScheme::Radial, GlobalSelection),
323            contract(WeightingScheme::Natural, per_field),
324            contract(briggs, per_field).with_casa_cube_density_padding(1),
325            contract(WeightingScheme::Natural, NotApplicable)
326                .with_density_cells(DensityCells::Box(cells(3))),
327            contract(briggs, PerOutputChannel).with_density_cells(DensityCells::Square(cells(8))),
328            contract(
329                WeightingScheme::BriggsBandwidthTaper { robust: 0.5 },
330                PerOutputChannel,
331            )
332            .with_density_cells(DensityCells::Box(cells(1))),
333            contract(
334                WeightingScheme::BriggsAbsolute {
335                    robust: 0.0,
336                    noise_jy: 0.0,
337                },
338                GlobalSelection,
339            ),
340            contract(
341                WeightingScheme::BriggsAbsolute {
342                    robust: 2.5,
343                    noise_jy: 1.0,
344                },
345                GlobalSelection,
346            ),
347            contract(
348                WeightingScheme::BriggsAbsolute {
349                    robust: 0.5,
350                    noise_jy: f64::NAN,
351                },
352                GlobalSelection,
353            ),
354        ] {
355            assert!(invalid.validate().is_err(), "{invalid:?}");
356        }
357        // Either term alone keeps the absolute denominator positive.
358        for (robust, noise_jy) in [(0.0, 1.0), (-1.0, 0.0)] {
359            let scheme = WeightingScheme::BriggsAbsolute { robust, noise_jy };
360            assert_eq!(contract(scheme, GlobalSelection).validate(), Ok(()));
361        }
362    }
