21pub(super) fn run(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
22    let selection = selection(task)?;
23    let scratch = task.flag("usescratch")?;
24    if !scratch
25        && [
26            &selection.timerange,
27            &selection.uvrange,
28            &selection.antenna,
29            &selection.scan,
30            &selection.observation,
31            &selection.intent,
32        ]
33        .into_iter()
34        .any(Option::is_some)
35    {
36        return Err(TaskFailure::validation(
37            "If any of the selection parameters timerange, uvrange, antenna, scan, observation or intent are not the default ones then usescratch must be set to True",
38        )
39        .at("usescratch"));
40    }
41    let path = task.input_path("vis")?;
42    let mut ms = OpenedMs::open(&path).map_err(execution)?;
43    let mut parts = super::multi_ms::partition_paths(&path)?
44        .unwrap_or_default()
45        .iter()
46        .map(|path| {
47            let opened = OpenedMs::open(path).map_err(execution)?;
48            if scratch {
49                opened.initialize(MODEL).map_err(execution)
50            } else {
51                Ok(opened)
52            }
53        })
54        .collect::<Result<Vec<_>, _>>()?;
55    if scratch && parts.is_empty() {
56        ms = ms.initialize(MODEL).map_err(execution)?;
57    }
58    let selected = ms.resolve(&selection).map_err(execution)?;
59    let request = read_model(task, scratch)?;
60    if !selected.selected_rows.is_empty() && request.images_are_tables(task) {
61        if parts.is_empty() || !scratch {
62            let models = read_models(task, &ms.measurement_set, &selected, &request)?;
63            if scratch {
64                write_scratch(&mut ms, &selected, &request, models)?;
65            } else {
66                write_virtual(&mut ms, &mut parts, &request, models)?;
67            }
68        } else {
69            write_partition_scratch(task, parts.as_mut_slice(), selected, &request)?;
70        }
71    }
72    for part in parts {
73        part.release().map_err(execution)?;
74    }
75    ms.release().map_err(execution)?;
76    super::history::record_task(task, &path);
77    TaskSuccess::new(CasaValue::None, &())
78}

80struct ModelRequest {
81    images: Vec<String>,
82    component: String,
83    scratch: bool,
84    incremental: bool,
85    terms: i64,
86}
87
88impl ModelRequest {
89    fn images_are_tables(&self, task: &TaskInvocation<'_>) -> bool {
90        if self
91            .images
92            .iter()
93            .filter(|name| !name.is_empty())
94            .any(|name| !task.workspace().join(name).join("table.dat").is_file())
95        {
96            // The legacy imager's ft returns false for an existing FITS
97            // file or another path that cannot name a CASA image table.
98            tracing::warn!("ft requires a CASA image table");
99            return false;
100        }
101        true
102    }
103}
104
105fn read_model(task: &TaskInvocation<'_>, scratch: bool) -> Result<ModelRequest, TaskFailure> {
106    let models = task.texts("model").map_err(|_| {
107        TaskFailure::validation(
108            "The model image must be a string or a list of strings (or '' or [])",
109        )
110        .at("model")
111    })?;
112    let component = task.text("complist")?.unwrap_or_default();
113    if (models.is_empty() || models == [""]) && component.is_empty() {
114        return Err(TaskFailure::validation(
115            "Please specify a model image or component list to ft",
116        )
117        .at("model"));
118    }
119    for model in &models {
120        if !model.is_empty() && !task.workspace().join(model).exists() {
121            return Err(
122                TaskFailure::validation(format!("Model image {model} cannot be found")).at("model"),
123            );
124        }
125    }
126    if !component.is_empty() && !task.workspace().join(&component).exists() {
127        return Err(
128            TaskFailure::validation(format!("Componentlist {component} cannot be found"))
129                .at("complist"),
130        );
131    }
132    let terms = task.integer("nterms")?;
133    validate_terms(terms, models.len())?;
134    Ok(ModelRequest {
135        images: models,
136        component,
137        scratch,
138        incremental: task.flag("incremental")?,
139        terms,
140    })
141}

143type Models = (Vec<VirtualModelEntry>, Vec<ImageModel>);
144
145fn read_models(
146    task: &TaskInvocation<'_>,
147    ms: &MeasurementSet,
148    selected: &ResolvedMsSelection,
149    request: &ModelRequest,
150) -> Result<Models, TaskFailure> {
151    let mut entries = Vec::new();
152    if !request.component.is_empty() && !request.incremental {
153        let list = ComponentList::read_table(task.workspace().join(&request.component))
154            .map_err(execution)?;
155        let container = list.to_record().map_err(execution)?;
156        entries.push(
157            model_entry(ms, selected, VirtualModelKind::ComponentList, container)
158                .map_err(execution)?,
159        );
160    }
161    let images = image_models(task, ms, selected, request, entries.is_empty())?;
162    Ok((entries, images))
163}

165fn write_scratch(
166    ms: &mut OpenedMs,
167    selected: &ResolvedMsSelection,
168    request: &ModelRequest,
169    (mut entries, images): Models,
170) -> Result<(), TaskFailure> {
171    entries.extend(images.iter().map(|image| image.entry.clone()));
172    write_model_scratch(
173        &mut ms.measurement_set,
174        selected,
175        &entries,
176        request.incremental,
177        true,
178        &mut ms.lock,
179    )
180    .map_err(execution)
181}
182
183fn write_partition_scratch(
184    task: &TaskInvocation<'_>,
185    parts: &mut [OpenedMs],
186    mut selected: ResolvedMsSelection,
187    request: &ModelRequest,
188) -> Result<(), TaskFailure> {
189    let rows = std::mem::take(&mut selected.selected_rows);
190    let mut rows = rows.into_iter().peekable();
191    let mut start = 0;
192    for part in parts {
193        let end = start + part.measurement_set.main_table().row_count();
194        while let Some(row) = rows.next_if(|row| row.row_index < end) {
195            selected
196                .selected_rows
197                .push(casa_ms::ResolvedMsSelectionRow {
198                    row_index: row.row_index - start,
199                    ..row
200                });
201        }
202        if !selected.selected_rows.is_empty() {
203            let models = read_models(task, &part.measurement_set, &selected, request)?;
204            write_scratch(part, &selected, request, models)?;
205            selected.selected_rows.clear();
206        }
207        start = end;
208    }
209    if rows.next().is_some() {
210        return Err(TaskFailure::execution(
211            "selected Multi-MS row is outside its partitions",
212        ));
213    }
214    Ok(())
215}

217fn write_virtual(
218    root: &mut OpenedMs,
219    parts: &mut [OpenedMs],
220    request: &ModelRequest,
221    (entries, images): Models,
222) -> Result<(), TaskFailure> {
223    let tables = std::iter::once(&root.measurement_set)
224        .chain(parts.iter().map(|part| &part.measurement_set));
225    let mut publication = VirtualFtPublication::new(
226        tables,
227        entries,
228        images,
229        request.incremental,
230        !parts.is_empty(),
231    )
232    .map_err(execution)?;
233    publication
234        .publish(
235            std::iter::once(root)
236                .chain(parts)
237                .map(|opened| (&mut opened.measurement_set, &mut opened.lock)),
238        )
239        .map_err(execution)
240}

242fn image_models(
243    task: &TaskInvocation<'_>,
244    ms: &MeasurementSet,
245    selected: &ResolvedMsSelection,
246    request: &ModelRequest,
247    no_components: bool,
248) -> Result<Vec<ImageModel>, TaskFailure> {
249    let models = &request.images;
250    let terms = request.terms;
251    let zero = request.scratch && !request.incremental && no_components;
252    let reference = if terms > 1 {
253        let quantity = task.text("reffreq")?.unwrap_or_default();
254        let explicit = if quantity.is_empty() {
255            0.0
256        } else {
257            quantity
258                .parse::<Quantity>()
259                .map_err(|_| {
260                    TaskFailure::execution(format!(
261                        "In conversion of reffreq={quantity:?} to a numerical value"
262                    ))
263                    .at("reffreq")
264                })?
265                .get_si_value()
266        };
267        Some(if explicit == 0.0 {
268            reference_frequency(&task.workspace().join(&models[0])).map_err(execution)?
269        } else {
270            explicit
271        })
272    } else {
273        None
274    };
275    let mut images = Vec::new();
276    for (index, path) in models
277        .iter()
278        .enumerate()
279        .filter(|(_, path)| !path.is_empty())
280    {
281        let taylor = reference.map(|reference_hz| TaylorTerm {
282            count: terms as i32,
283            index: index as i32,
284            reference_hz,
285        });
286        if let Some(model) = prepare_image(
287            ms,
288            selected,
289            &task.workspace().join(path),
290            taylor,
291            zero && index == 0,
292            request.scratch,
293        )
294        .map_err(execution)?
295        {
296            images.push(model);
297        }
298    }
299    Ok(images)
300}

302fn selection(task: &TaskInvocation<'_>) -> Result<MsSelection, TaskFailure> {
303    let text = |name| {
304        task.text(name)
305            .map(|value| value.filter(|value| !value.is_empty()))
306    };
307    Ok(MsSelection {
308        field: text("field")?,
309        spw: text("spw")?,
310        timerange: text("timerange")?,
311        uvrange: text("uvrange")?,
312        antenna: text("antenna")?,
313        scan: text("scan")?,
314        observation: text("observation")?,
315        intent: text("intent")?,
316        ..MsSelection::default()
317    })
318}
319
320fn validate_terms(terms: i64, images: usize) -> Result<(), TaskFailure> {
321    if terms < 1 {
322        return Err(
323            TaskFailure::validation("nterms must be greater than or equal to 1").at("nterms"),
324        );
325    }
326    if terms > 1 && images != terms as usize {
327        return Err(TaskFailure::validation(
328            "For nterms>1, please provide a list of nterms model-image names",
329        )
330        .at("model"));
331    }
332    Ok(())
333}
