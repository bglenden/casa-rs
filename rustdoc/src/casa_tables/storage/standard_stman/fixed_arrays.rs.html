4use super::array_rows::ReadCell;
5use super::*;
6use crate::storage::stman_array_file::CanonicalArrayElement;
7use crate::table::{SelectedArray2D, SelectedArray2DCells};
8use casa_core::{Complex32, Complex64};

10struct FixedCells<T> {
11    shape: [usize; 2],
12    values: Vec<T>,
13    order: Vec<usize>,
14    ordered: bool,
15    stored_shape: Vec<usize>,
16}
17
18impl<T: CanonicalArrayElement> ReadCell for FixedCells<T> {
19    type Cell = ();
20    fn indirect(
21        &mut self,
22        reader: Option<&mut StManArrayFileReader>,
23        offset: i64,
24        _data_type: CasacoreDataType,
25        desc: &ColumnDescContents,
26        (row, slot): (usize, usize),
27    ) -> Result<Option<()>, StorageError> {
28        let Some(reader) = reader else {
29            return if offset == 0 {
30                Ok(None)
31            } else {
32                Err(StorageError::FormatMismatch(format!(
33                    "SSM column {} row {row} has no array file",
34                    desc.col_name
35                )))
36            };
37        };
38        if !reader.append_fixed_shape(
39            offset,
40            &self.shape,
41            &mut self.stored_shape,
42            &mut self.values,
43        )? {
44            return Ok(None);
45        }
46        self.order.try_reserve(1).map_err(|error| {
47            StorageError::FormatMismatch(format!("cannot allocate fixed array row order: {error}"))
48        })?;
49        self.ordered &= slot == self.order.len();
50        self.order.push(slot);
51        Ok(Some(()))
52    }
53    fn direct(
54        &mut self,
55        _raw: &ColumnRawData,
56        _desc: &ColumnDescContents,
57        _position: (usize, usize),
58        _rows: usize,
59    ) -> Result<Option<()>, StorageError> {
60        Ok(None)
61    }
62}
63
64fn read<T: CanonicalArrayElement>(
65    path: &Path,
66    blob: &[u8],
67    descs: &[&ColumnDescContents],
68    column: usize,
69    rows: &[usize],
70    shape: [usize; 2],
71) -> Result<Option<SelectedArray2D<T>>, StorageError> {
72    let count = shape[0]
73        .checked_mul(shape[1])
74        .filter(|&count| count > 0)
75        .ok_or_else(|| {
76            StorageError::FormatMismatch("fixed array shape is empty or overflows".into())
77        })?;
78    let total = count
79        .checked_mul(rows.len())
80        .filter(|&total| {
81            total
82                .checked_mul(std::mem::size_of::<T>())
83                .is_some_and(|bytes| bytes <= isize::MAX as usize)
84        })
85        .ok_or_else(|| {
86            StorageError::FormatMismatch("fixed selected array size overflows".into())
87        })?;
88    let mut cells = FixedCells {
89        shape,
90        values: Vec::new(),
91        order: Vec::new(),
92        ordered: true,
93        stored_shape: Vec::with_capacity(2),
94    };
95    let Some(defined) = array_rows::read(path, blob, descs, column, rows, &mut cells)? else {
96        return Ok(None);
97    };
98    if defined.iter().any(Option::is_none) {
99        return Ok(None);
100    }
101    let values = if shape[1] == 1 && cells.ordered {
102        cells.values
103    } else {
104        // Allocate the transposed result only after all stored shapes match.
105        let mut values = Vec::new();
106        values.try_reserve_exact(total).map_err(|error| {
107            StorageError::FormatMismatch(format!("cannot allocate packed fixed arrays: {error}"))
108        })?;
109        values.resize(total, T::default());
110        for (source_row, slot) in cells.order.into_iter().enumerate() {
111            for channel in 0..shape[1] {
112                let source = (source_row * shape[1] + channel) * shape[0];
113                let target = (channel * rows.len() + slot) * shape[0];
114                values[target..target + shape[0]]
115                    .copy_from_slice(&cells.values[source..source + shape[0]]);
116            }
117        }
118        values
119    };
120    Ok(Some(SelectedArray2D::new(
121        rows.len(),
122        shape[0],
123        shape[1],
124        values,
125    )))
126}

128/// Whole indirect cells, all of exactly `shape`, packed channel/row/axis0.
129/// Undefined or differently-shaped cells return None; all I/O errors propagate.
130pub(crate) fn read_fixed_2d_cells(
131    path: &Path,
132    blob: &[u8],
133    descs: &[&ColumnDescContents],
134    column: usize,
135    rows: &[usize],
136    shape: [usize; 2],
137) -> Result<Option<SelectedArray2DCells>, StorageError> {
138    let Some(desc) = descs.get(column) else {
139        return Err(StorageError::FormatMismatch(
140            "SSM fixed column index is out of range".into(),
141        ));
142    };
143    if !is_ssm_array_file_indirect(desc) {
144        return Ok(None);
145    }
146    macro_rules! load {
147        ($type:ty, $variant:ident) => {
148            read::<$type>(path, blob, descs, column, rows, shape)
149                .map(|cells| cells.map(SelectedArray2DCells::$variant))
150        };
151    }
152    match desc.require_primitive_type()? {
153        casa_core::PrimitiveType::Bool => load!(bool, Bool),
154        casa_core::PrimitiveType::Float32 => load!(f32, Float32),
155        casa_core::PrimitiveType::Float64 => load!(f64, Float64),
156        casa_core::PrimitiveType::Complex32 => load!(Complex32, Complex32),
157        casa_core::PrimitiveType::Complex64 => load!(Complex64, Complex64),
158        _ => Ok(None),
159    }
160}
