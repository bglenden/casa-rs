3//! The CASA image writer.
4//!
5//! Each member is written to a private staging directory beside its target.
6//! Publishing then replaces each target on its own: the staged image is
7//! renamed into place, or exchanged with an existing image in one call
8//! (`renamex_np` with `RENAME_SWAP` on macOS, `renameat2` with
9//! `RENAME_EXCHANGE` on Linux), and both directories are synced. An open
10//! after the replacement sees the whole new image, and one before it the
11//! whole old image. A member already replaced stays valid if a later one
12//! fails; the set is then incomplete and the run must be repeated.
13//!
14//! Limits:
15//! - **A held reader keeps only what it has opened.** casacore opens some
16//!   storage, such as a pixel mask, by path when it is first read. A reader
17//!   that holds the old image across a replacement and first reads such
18//!   storage afterwards reads the new image's, or fails if the new image has
19//!   none. Reopen an image after a run replaces it. CASA's own in-place
20//!   overwrite gives a held reader no coherence at all.
21//! - **Exchange needs local filesystem support.** A filesystem without an
22//!   atomic exchange (SMB, some NFS) refuses a run into an existing image,
23//!   with [`ProductsError::Io`].
24//! - **A killed run leaves its staging.** A failed generation discards what
25//!   it staged; a killed process leaves `.<name>.casa-rs-stage-*`
26//!   directories beside the targets, which are safe to delete.
27

91/// Keep the `logtable` of the image at `target` in the image about to replace
92/// it: a run on an existing image appends to its history, as CASA's does,
93/// which updates the image in place.
94fn carry_history(target: &Path, staging: &Path) -> std::io::Result<()> {
95    let earlier = target.join("logtable");
96    if !earlier.is_dir() {
97        return Ok(());
98    }
99    let staged = staging.join("logtable");
100    fs::remove_dir_all(&staged)?;
101    copy_directory(&earlier, &staged)
102}
103
104fn copy_directory(from: &Path, to: &Path) -> std::io::Result<()> {
105    fs::create_dir(to)?;
106    for entry in fs::read_dir(from)? {
107        let entry = entry?;
108        let target = to.join(entry.file_name());
109        if entry.file_type()?.is_dir() {
110            copy_directory(&entry.path(), &target)?;
111        } else {
112            fs::copy(entry.path(), target)?;
113        }
114    }
115    Ok(())
116}

118impl CasaImageWriter {
119    /// Write the main domain's members under `base` with `coordinates`.
120    #[must_use]
121    pub fn new(base: PathBuf, coordinates: CoordinateSystem) -> Self {
122        Self::for_domains([CasaImageDomainOutput::new(
123            ImageDomainRole::Main,
124            base,
125            coordinates,
126        )])
127    }
128
129    /// Write every compiled image domain's members under its own prefix.
130    ///
131    /// # Panics
132    ///
133    /// If two outputs share a role or a prefix, or none is the main domain.
134    /// Compiling a request refuses domains that name one output, and always
135    /// compiles the main domain.
136    #[must_use]
137    pub fn for_domains(domains: impl IntoIterator<Item = CasaImageDomainOutput>) -> Self {
138        let mut outputs = BTreeMap::new();
139        let mut roots = BTreeSet::new();
140        for output in domains {
141            let new_root = roots.insert(output.base.clone());
142            let new_role = outputs.insert(output.role.clone(), output).is_none();
143            assert!(
144                new_root && new_role,
145                "compiled image domains have distinct roles and output roots"
146            );
147        }
148        assert!(
149            outputs.contains_key(&ImageDomainRole::Main),
150            "compiled image domains include the main domain"
151        );
152        Self {
153            domains: outputs,
154            staged: Mutex::new(Vec::new()),
155        }
156    }
157
158    fn output(&self, role: &ImageDomainRole) -> &CasaImageDomainOutput {
159        self.domains
160            .get(role)
161            .expect("the writer binds every compiled image domain")
162    }

164    /// Bytes the writer holds while it writes `planned` a window at a time:
165    /// the image's tile cache and serialization buffers, the pixel mask's
166    /// cache (opened with the image's budget, so up to four windows of
167    /// validity), its coordinates and beam metadata, and the record of each
168    /// staged member kept until [`Self::publish`].
169    pub(crate) fn residency_bytes(
170        &self,
171        planned: &PlannedContinuumGeneration,
172        window_payload_bytes: u64,
173        window_validity_bytes: u64,
174    ) -> Result<u64, ProductsError> {
175        // The open image's table, data-manager and record metadata. An 8×8
176        // PB-only member with empty coordinates measured 46 kB live against
177        // 18 kB charged with a 4 kB envelope (review of #701).
178        const IMAGE_ENVELOPE_BYTES: u64 = 64 * 1024;
179        const STAGED_MEMBER_RECORD_BYTES: u64 = 512;
180        let overflow = || ProductsError::ResourceDemandOverflow("CASA image writer");
181        let coordinate_bytes = self.domains.values().try_fold(0_u64, |maximum, output| {
182            let bytes = output
183                .coordinates
184                .to_record()
185                .retained_heap_bytes()
186                .and_then(|bytes| bytes.checked_mul(4))
187                .and_then(|bytes| u64::try_from(bytes).ok())
188                .ok_or_else(overflow)?;
189            Ok::<_, ProductsError>(maximum.max(bytes))
190        })?;
191        let beam_bytes = planned
192            .members()
193            .iter()
194            .try_fold(0_u64, |maximum, member| {
195                let count = if member.storage().attach_beam() {
196                    member
197                        .axes()
198                        .spectral()
199                        .output_channels()
200                        .checked_mul(member.axes().polarization().len())
201                        .ok_or_else(overflow)?
202                } else {
203                    0
204                };
205                Ok::<_, ProductsError>(maximum.max(beam_metadata_residency_bytes(count)?))
206            })?;
207        let registry_bytes = planned.members().iter().try_fold(0_u64, |total, member| {
208            let base = self.output(member.axes().domain()).base.as_os_str().len();
209            let path_bytes = base
210                .checked_add(member.name().len())
211                .and_then(|bytes| bytes.checked_mul(2))
212                .and_then(|bytes| u64::try_from(bytes).ok())
213                .ok_or_else(overflow)?;
214            total
215                .checked_add(path_bytes)
216                .and_then(|bytes| bytes.checked_add(STAGED_MEMBER_RECORD_BYTES))
217                .ok_or_else(overflow)
218        })?;
219        window_payload_bytes
220            .checked_mul(4)
221            // The validity pack and its copy, and the mask's cache.
222            .and_then(|bytes| bytes.checked_add(window_validity_bytes.checked_mul(6)?))
223            .and_then(|bytes| bytes.checked_add(IMAGE_ENVELOPE_BYTES))
224            .and_then(|bytes| bytes.checked_add(coordinate_bytes))
225            .and_then(|bytes| bytes.checked_add(beam_bytes))
226            .and_then(|bytes| bytes.checked_add(registry_bytes))
227            .ok_or_else(overflow)
228    }

230    /// The number of members staged and not yet published.
231    pub(crate) fn staged_count(&self) -> usize {
232        self.staged
233            .lock()
234            .unwrap_or_else(PoisonError::into_inner)
235            .len()
236    }

238    /// Discard the members staged after the first `kept`, deleting their
239    /// staging: a failed generation publishes none of its members.
240    pub(crate) fn discard_staged_after(&self, kept: usize) {
241        self.staged
242            .lock()
243            .unwrap_or_else(PoisonError::into_inner)
244            .truncate(kept);
245    }
246
247    /// Replace every staged member's target, in generation order, telling
248    /// `published` each target once it is replaced.
249    ///
250    /// # Errors
251    ///
252    /// The first replacement that fails. Members replaced before it stay
253    /// replaced, and `published` has heard of each; the run must be
254    /// repeated.
255    pub fn publish(&self, mut published: impl FnMut(&Path)) -> Result<(), ProductsError> {
256        let staged =
257            std::mem::take(&mut *self.staged.lock().unwrap_or_else(PoisonError::into_inner));
258        for image in staged {
259            let failed = |source| ProductsError::Io {
260                operation: "replacing",
261                path: image.target.clone(),
262                source,
263            };
264            carry_history(&image.target, &image.staging).map_err(failed)?;
265            promote_atomically(&image.staging, &image.target).map_err(failed)?;
266            published(&image.target);
267        }
268        Ok(())
269    }

271    /// Create one member's image in private staging beside its target, with
272    /// its units, beams and metadata.
273    pub(crate) fn begin(
274        &self,
275        member: &PlannedMember,
276        layout: ProductWindowLayout,
277        beams: &[Option<RestoringBeam>],
278        uses_weight_image: bool,
279    ) -> Result<MemberImage<'_>, ProductsError> {
280        let started = Instant::now();
281        let output = self.output(member.axes().domain());
282        let mut target = output.base.clone().into_os_string();
283        target.push(member.name());
284        let target = PathBuf::from(target);
285        let io = |operation, path: &Path| {
286            let path = path.to_path_buf();
287            move |source| ProductsError::Io {
288                operation,
289                path,
290                source,
291            }
292        };
293        let parent = target.parent().unwrap_or_else(|| Path::new("."));
294        fs::create_dir_all(parent).map_err(io("creating", parent))?;
295        let mut prefix = OsString::from(".");
296        prefix.push(target.file_name().expect("a product target names a file"));
297        prefix.push(".casa-rs-stage-");
298        let directory = tempfile::Builder::new()
299            .prefix(&prefix)
300            .tempdir_in(parent)
301            .map_err(io("staging", parent))?;
302        let staging = directory.path().join("image");
303        let image_error = |source| ProductsError::Image {
304            path: target.clone(),
305            source,
306        };
307        let mut tile = layout.shape();
308        tile[layout.spectral_axis()] = 1;
309        let mut image = PagedImage::<f32>::create_with_tile_shape_and_cache(
310            layout.shape().to_vec(),
311            tile.to_vec(),
312            output.coordinates.clone(),
313            &staging,
314            layout.maximum_values() * size_of::<f32>(),
315        )
316        .map_err(image_error)?;
317        let storage = member.storage();
318        image
319            .set_units(storage.unit().map_or("", unit_label))
320            .map_err(image_error)?;
321        let role = role_label(member.role());
322        image
323            .set_image_info(&ImageInfo {
324                beam_set: if storage.attach_beam() {
325                    persisted_beam_set(beams, layout.shape()[layout.spectral_axis()])?
326                } else {
327                    ImageBeamSet::default()
328                },
329                // CASA's tclean leaves every product the default type.
330                image_type: ImageType::Intensity,
331                object_name: output.object_name.clone(),
332            })
333            .map_err(image_error)?;
334        // `SynthesisImager::createIMStore`'s, but its `memreq` and `memavail`,
335        // which describe CASA's run: one process, one chunk.
336        let mut miscinfo = vec![
337            RecordField::new(
338                "casars_imager_role",
339                Value::Scalar(ScalarValue::String(role.to_string())),
340            ),
341            RecordField::new(
342                "INSTRUME",
343                Value::Scalar(ScalarValue::String(output.telescope.clone())),
344            ),
345            RecordField::new("distance", Value::Scalar(ScalarValue::Float64(0.0))),
346            RecordField::new("mpiprocs", Value::Scalar(ScalarValue::Int32(1))),
347            RecordField::new("chnchnks", Value::Scalar(ScalarValue::Int32(1))),
348        ];
349        // CASA tools read `.sumwt`'s `useweightimage` to learn whether the
350        // residual was divided by a weight image (`SIImageStore`
351        // `getUseWeightImage`).
352        if matches!(member.role(), ProductRole::SumWeights(_)) {
353            miscinfo.push(RecordField::new(
354                "useweightimage",
355                Value::Scalar(ScalarValue::Bool(uses_weight_image)),
356            ));
357        }
358        image
359            .set_misc_info(RecordValue::new(miscinfo))
360            .map_err(image_error)?;
361        Ok(MemberImage {
362            image,
363            explicit_mask: matches!(storage.pixel_mask(), ProductPixelMask::Explicit(_)),
364            writer: self,
365            staged: StagedImage {
366                _directory: directory,
367                staging,
368                target,
369            },
370            begin_nanos: started.elapsed().as_nanos(),
371            window_nanos: 0,
372        })
373    }
374}

376/// One member's image in staging, written a window at a time.
377pub(crate) struct MemberImage<'a> {
378    image: PagedImage<f32>,
379    explicit_mask: bool,
380    writer: &'a CasaImageWriter,
381    staged: StagedImage,
382    begin_nanos: u128,
383    window_nanos: u128,
384}
385
386impl MemberImage<'_> {
387    /// Write one generated window's values and, where the member stores
388    /// one, its pixel mask.
389    pub(crate) fn write(&mut self, window: ProductWindow) -> Result<(), ProductsError> {
390        let started = Instant::now();
391        let (start, shape, payload, validity) = window.into_parts();
392        let image_error = |source| ProductsError::Image {
393            path: self.staged.target.clone(),
394            source,
395        };
396        let data = ArrayD::from_shape_vec(IxDyn(&shape), payload)
397            .expect("a product window's payload fills its shape");
398        self.image
399            .put_slice_view(data.view(), &start)
400            .map_err(image_error)?;
401        if self.explicit_mask {
402            let mask = ArrayD::from_shape_vec(IxDyn(&shape), validity)
403                .expect("a product window's validity fills its shape");
404            self.image
405                .put_mask_slice("mask0", &mask, &start)
406                .map_err(image_error)?;
407        }
408        self.window_nanos += started.elapsed().as_nanos();
409        Ok(())
410    }
411
412    /// Save the image in staging and register it for [`CasaImageWriter::publish`].
413    pub(crate) fn finish(mut self) -> Result<(), ProductsError> {
414        let started = Instant::now();
415        let target = self.staged.target.clone();
416        let image_error = |source| ProductsError::Image {
417            path: target.clone(),
418            source,
419        };
420        if self.explicit_mask {
421            self.image.set_default_mask("mask0").map_err(image_error)?;
422        }
423        self.image.save().map_err(image_error)?;
424        if let Some(stats) = self.image.tiled_io_stats() {
425            tracing::debug!(
426                target = %self.staged.target.display(),
427                pack_nanos = stats.direct_tile_pack_ns,
428                swap_nanos = stats.direct_tile_swap_ns,
429                write_nanos = stats.direct_tile_write_ns,
430                direct_bytes = stats.direct_tile_write_bytes,
431                direct_calls = stats.direct_tile_write_calls,
432                "product image pixels written"
433            );
434        }
435        let Self {
436            image,
437            writer,
438            staged,
439            begin_nanos,
440            window_nanos,
441            ..
442        } = self;
443        drop(image);
444        tracing::debug!(
445            target = %staged.target.display(),
446            begin_nanos,
447            window_nanos,
448            finish_nanos = started.elapsed().as_nanos(),
449            "product image staged"
450        );
451        writer
452            .staged
453            .lock()
454            .unwrap_or_else(PoisonError::into_inner)
455            .push(staged);
456        Ok(())
457    }
458}

460fn beam_metadata_residency_bytes(count: usize) -> Result<u64, ProductsError> {
461    if count == 0 {
462        return Ok(0);
463    }
464    let overflow = || ProductsError::ResourceDemandOverflow("beam metadata");
465    let beam_record = GaussianBeam::new(1.0, 1.0, 0.0)
466        .to_record()
467        .retained_heap_bytes()
468        .ok_or_else(overflow)?;
469    let key_bytes = count.ilog10() as usize + 2;
470    // The growable outer keyword record has at most twice its live field count.
471    // Save overlaps owned keywords, the storage snapshot, table.dat keywords,
472    // and the encoded control buffer; the record heap bounds its encoded form.
473    let record = count
474        .checked_add(2)
475        .and_then(|fields| fields.checked_mul(2 * size_of::<RecordField>()))
476        .and_then(|bytes| {
477            bytes.checked_add(count.checked_mul(beam_record.checked_add(key_bytes)?)?)
478        })
479        .and_then(|bytes| bytes.checked_mul(4))
480        .ok_or_else(overflow)?;
481    let arrays = count
482        .checked_mul(
483            size_of::<RestoringBeam>() + size_of::<Vec<GaussianBeam>>() + size_of::<GaussianBeam>(),
484        )
485        .ok_or_else(overflow)?;
486    u64::try_from(record.checked_add(arrays).ok_or_else(overflow)?).map_err(|_| overflow())
487}

489/// The beam set casacore keeps for a member's beams, which are listed channel
490/// by channel with polarization fastest. As `SIImageStore::makeImageBeamSet`
491/// and `restore` store them, one beam is the image's beam and more are a
492/// channel × Stokes grid, equal or not: the shape `ImageInfo::checkBeamSet`
493/// requires of an image with `channels` channels before it opens it. A
494/// plane without a beam takes the largest one, as CASA's do. Beams that do
495/// not fill whole channels are refused.
496fn persisted_beam_set(
497    beams: &[Option<RestoringBeam>],
498    channels: usize,
499) -> Result<ImageBeamSet, ProductsError> {
500    if beams.is_empty() {
501        return Ok(ImageBeamSet::default());
502    }
503    if let [Some(only)] = beams {
504        return Ok(ImageBeamSet::new(gaussian_beam(*only)));
505    }
506    if channels == 0 || !beams.len().is_multiple_of(channels) {
507        return Err(ProductsError::BeamGrid {
508            beams: beams.len(),
509            channels,
510        });
511    }
512    let valid = beams.iter().flatten().copied().collect::<Vec<_>>();
513    let filler = valid
514        .iter()
515        .copied()
516        .max_by(|left, right| {
517            beam_area(*left)
518                .partial_cmp(&beam_area(*right))
519                .expect("validated beam areas are finite")
520        })
521        .map(gaussian_beam)
522        .unwrap_or_else(|| {
523            let one_microarcsecond_rad = std::f64::consts::PI / (180.0 * 3_600_000_000.0);
524            GaussianBeam::new(one_microarcsecond_rad, one_microarcsecond_rad, 0.0)
525        });
526    let polarizations = beams.len() / channels;
527    Ok(ImageBeamSet::from_grid(
528        beams
529            .chunks(polarizations)
530            .map(|channel| {
531                channel
532                    .iter()
533                    .map(|beam| beam.map_or(filler, gaussian_beam))
534                    .collect()
535            })
536            .collect(),
537    ))
538}

540fn gaussian_beam(beam: RestoringBeam) -> GaussianBeam {
541    GaussianBeam::new(
542        beam.major_fwhm_rad(),
543        beam.minor_fwhm_rad(),
544        beam.position_angle_rad(),
545    )
546}
547
548fn beam_area(beam: RestoringBeam) -> f64 {
549    beam.major_fwhm_rad() * beam.minor_fwhm_rad()
550}

552const fn unit_label(unit: ProductUnit) -> &'static str {
553    match unit {
554        ProductUnit::NotApplicable | ProductUnit::Dimensionless | ProductUnit::VisibilityWeight => {
555            ""
556        }
557        ProductUnit::JyPerBeam => "Jy/beam",
558        ProductUnit::JyPerPixel => "Jy/pixel",
559    }

562const fn role_label(role: ProductRole) -> &'static str {
563    match role {
564        ProductRole::Psf(_) => "psf",
565        ProductRole::Residual(_) => "residual",
566        ProductRole::Model(_) => "model",
567        ProductRole::RestoredImage(_) => "image",
568        ProductRole::SumWeights(_) => "sumwt",
569        ProductRole::CleanMask => "mask",
570        ProductRole::Weight(_) => "weight",
571        ProductRole::PrimaryBeam(_) => "pb",
572        ProductRole::Sensitivity => "sensitivity",
573        ProductRole::PbCorrectedImage(_) => "pbcor.image",
574        ProductRole::SpectralIndex => "alpha",
575        ProductRole::SpectralIndexError => "alpha.error",
576        ProductRole::PbCorrectedSpectralIndex => "alpha.pbcor",
577        ProductRole::BeamMetadata => "beam",
578        _ => "product",
579    }

582fn promote_atomically(staging: &Path, target: &Path) -> std::io::Result<()> {
583    let parent = target.parent().ok_or_else(|| {
584        std::io::Error::new(
585            std::io::ErrorKind::InvalidInput,
586            "product target has no parent directory",
587        )
588    })?;
589    if staging.parent().and_then(Path::parent) != Some(parent) {
590        return Err(std::io::Error::new(
591            std::io::ErrorKind::InvalidInput,
592            "product staging must be private to the output directory",
593        ));
594    }
595    if target.exists() {
596        exchange_directories(staging, target)?;
597    } else {
598        fs::rename(staging, target)?;
599    }
600    File::open(parent).and_then(|directory| directory.sync_all())?;
601    File::open(
602        staging
603            .parent()
604            .expect("validated private output directory"),
605    )
606    .and_then(|directory| directory.sync_all())?;
607    Ok(())
608}

610#[cfg(target_os = "macos")]
611fn exchange_directories(staging: &Path, target: &Path) -> std::io::Result<()> {
612    use std::os::unix::ffi::OsStrExt;
613    let staging = CString::new(staging.as_os_str().as_bytes())
614        .map_err(|_| std::io::Error::other("staging path contains NUL"))?;
615    let target = CString::new(target.as_os_str().as_bytes())
616        .map_err(|_| std::io::Error::other("target path contains NUL"))?;
617    // SAFETY: both C strings remain valid for the duration of this atomic call.
618    let status = unsafe { libc::renamex_np(staging.as_ptr(), target.as_ptr(), libc::RENAME_SWAP) };
619    (status == 0)
620        .then_some(())
621        .ok_or_else(std::io::Error::last_os_error)
622}

625fn exchange_directories(staging: &Path, target: &Path) -> std::io::Result<()> {
626    use std::os::unix::ffi::OsStrExt;
627    let staging = CString::new(staging.as_os_str().as_bytes())
628        .map_err(|_| std::io::Error::other("staging path contains NUL"))?;
629    let target = CString::new(target.as_os_str().as_bytes())
630        .map_err(|_| std::io::Error::other("target path contains NUL"))?;
631    // SAFETY: both C strings remain valid for the duration of this atomic call.
632    let status = unsafe {
633        libc::syscall(
634            libc::SYS_renameat2,
635            libc::AT_FDCWD,
636            staging.as_ptr(),
637            libc::AT_FDCWD,
638            target.as_ptr(),
639            libc::RENAME_EXCHANGE,
640        )
641    };
642    (status == 0)
643        .then_some(())
644        .ok_or_else(std::io::Error::last_os_error)
645}
646
647#[cfg(not(any(target_os = "macos", target_os = "linux")))]
648fn exchange_directories(_: &Path, _: &Path) -> std::io::Result<()> {
649    Err(std::io::Error::new(
650        std::io::ErrorKind::Unsupported,
651        "atomic non-empty directory exchange is unavailable",
652    ))
653}

781    #[test]
782    fn individual_outputs_replace_atomically_and_failed_sets_require_rerun() {
783        let root = tempfile::tempdir().unwrap();
784        let target = root.path().join("image");
785        std::fs::create_dir(&target).unwrap();
786        std::fs::write(target.join("pixels"), b"old").unwrap();
787        let private = tempfile::tempdir_in(root.path()).unwrap();
788        let staged = private.path().join("image");
789        std::fs::create_dir(&staged).unwrap();
790        std::fs::write(staged.join("pixels"), b"new").unwrap();
791        super::promote_atomically(&staged, &target).unwrap();
792        assert_eq!(std::fs::read(target.join("pixels")).unwrap(), b"new");
793        assert_eq!(std::fs::read(staged.join("pixels")).unwrap(), b"old");
794        let missing = private.path().join("missing");
795        assert!(super::promote_atomically(&missing, &target).is_err());
796        assert_eq!(std::fs::read(target.join("pixels")).unwrap(), b"new");
797    }
