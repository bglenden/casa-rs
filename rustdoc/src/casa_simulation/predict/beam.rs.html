2//! The primary beam `sm.setvp(dovp=True)` applies: CASA's default voltage
3//! patterns, as `PBMath` gives them by telescope name and frequency.
4//!
5//! CASA gives the VLA's and the EVLA's patterns a beam squint, which moves
6//! the RR and LL beams apart. It reaches only an image with RR and LL
7//! planes: a Stokes I model, the only one the simulator predicts, is one
8//! plane, and CASA applies the unsquinted pattern to it (measured on CASA
9//! 6.7.6: RR and LL agree, whatever the offset's direction). casa-rs's
10//! patterns are unsquinted.
11
12use casa_numerics::{AnnularApertureVoltageTable, EvlaPowerPattern, VlaBand};
13use ndarray::Array2;
14
15use super::SkyModel;
16
17/// The primary-beam voltage pattern `setvp(dovp=True)` applies.
18#[derive(Clone, Copy, Debug, PartialEq)]
19pub enum PrimaryBeam {
20    /// CASA `PBMath1DAiry`: an Airy pattern for a dish with a central
21    /// blockage, tabulated out to a maximum radius.
22    Airy {
23        /// Effective dish diameter in metres.
24        dish_diameter_m: f64,
25        /// Blockage diameter in metres.
26        blockage_diameter_m: f64,
27        /// Radius of the tabulated pattern at 1 GHz, in arcmin (scaled as
28        /// 1/frequency). The pattern is zero beyond it, and CASA's 10,000
29        /// table samples span it, so it also sets the lookup quantization.
30        maximum_radius_arcmin_ghz: f64,
31    },
32    /// CASA `PBMath1DEVLA`: the EVLA's polynomial power patterns, each
33    /// model plane taking the fit CASA picks for its frequency
34    /// ([`EvlaPowerPattern::at`]), tabulated to 58′ at 1 GHz.
35    Evla,
36}
37
38impl PrimaryBeam {
39    /// ALMA 12 m antennas as CASA's default voltage patterns give them
40    /// (`PBMath::initByTelescope`): 10.7 m effective diameter, 0.75 m
41    /// blockage, 1.784° at 1 GHz.
42    pub const ALMA: Self = Self::Airy {
43        dish_diameter_m: 10.7,
44        blockage_diameter_m: 0.75,
45        maximum_radius_arcmin_ghz: 1.784 * 60.0,
46    };
47
48    /// ACA 7 m antennas as CASA's default voltage patterns give them
49    /// (`PBMath::initByTelescope`): 6.25 m effective diameter, 0.75 m
50    /// blockage, 3.568° at 1 GHz.
51    pub const ACA: Self = Self::Airy {
52        dish_diameter_m: 6.25,
53        blockage_diameter_m: 0.75,
54        maximum_radius_arcmin_ghz: 3.568 * 60.0,
55    };
56
57    /// VLA antennas in a recognised band ([`VlaBand`]; `PBMath`'s `VLA_4`,
58    /// `VLA_P`, `VLA_L`, `VLA_C`, `VLA_X`, `VLA_U`, `VLA_K` and `VLA_Q`):
59    /// 25 m with a 2.36 m blockage, 0.8564° at 1 GHz.
60    pub const VLA: Self = Self::Airy {
61        dish_diameter_m: 25.0,
62        blockage_diameter_m: 2.36,
63        maximum_radius_arcmin_ghz: 0.8564 * 60.0,
64    };
65
66    /// VLA antennas outside the recognised bands (`VLA_NVSS`): 24.5 m
67    /// without blockage, 0.8564° at 1 GHz.
68    pub const VLA_NVSS: Self = Self::Airy {
69        dish_diameter_m: 24.5,
70        blockage_diameter_m: 0.0,
71        maximum_radius_arcmin_ghz: 0.8564 * 60.0,
72    };
73
74    /// CASA's default voltage pattern for `telescope`, the OBSERVATION
75    /// table's TELESCOPE_NAME, at `reference_frequency_hz`, spectral window
76    /// 0's REF_FREQUENCY (`VPSkyJones`), where it is one of these Airy
77    /// patterns. `PBMath::enumerateCommonPB` matches the name's prefix:
78    /// - `ALMA` (which also catches `ALMASD`, whose pattern is ALMA's), and
79    ///   `ATF`, which maps to ALMA;
80    /// - `ACA`;
81    /// - `VLA`, whose pattern follows the band of the reference frequency
82    ///   ([`VlaBand::of`], `PBMath::whichCommonPBtoUse`): [`Self::VLA`] in
83    ///   the bands CASA recognises and [`Self::VLA_NVSS`] between them;
84    /// - `EVLA`: [`Self::Evla`], which picks its fit per model plane.
85    ///
86    /// `None` for every other telescope, whose patterns casa-rs does not yet
87    /// have.
88    #[must_use]
89    pub fn for_telescope(telescope: &str, reference_frequency_hz: f64) -> Option<Self> {
90        if telescope.starts_with("ALMA") || telescope.starts_with("ATF") {
91            Some(Self::ALMA)
92        } else if telescope.starts_with("ACA") {
93            Some(Self::ACA)
94        } else if telescope.starts_with("VLA") {
95            Some(match VlaBand::of(reference_frequency_hz) {
96                Some(_) => Self::VLA,
97                None => Self::VLA_NVSS,
98            })
99        } else if telescope.starts_with("EVLA") {
100            Some(Self::Evla)
101        } else {
102            None
103        }
104    }
105
106    /// Whether CASA gives this pattern a beam squint: the VLA's and the
107    /// EVLA's (`PBMath1D` with `squint_p` set), which moves the RR and LL
108    /// beams apart for a component.
109    #[must_use]
110    pub fn squints(self) -> bool {
111        self == Self::VLA || self == Self::VLA_NVSS || self == Self::Evla
112    }
113
114    /// The pattern's lookup: an Airy pattern's table, shared by every plane
115    /// it is evaluated on, or the EVLA's fits, which hold none.
116    pub(crate) fn pattern(self) -> BeamPattern {
117        BeamPattern(match self {
118            Self::Airy {
119                dish_diameter_m,
120                blockage_diameter_m,
121                maximum_radius_arcmin_ghz,
122            } => Lookup::Airy(AnnularApertureVoltageTable::new(
123                dish_diameter_m,
124                blockage_diameter_m,
125                maximum_radius_arcmin_ghz,
126            )),
127            Self::Evla => Lookup::Evla,
128        })
129    }
130}
131
132/// A primary beam's pattern, as it is looked up by radius.
133#[derive(Clone)]
134pub(crate) struct BeamPattern(Lookup);
135
136#[derive(Clone)]
137enum Lookup {
138    /// An Airy pattern's voltage table.
139    Airy(AnnularApertureVoltageTable),
140    /// The EVLA's fits, picked per plane.
141    Evla,
142}
143
144impl BeamPattern {
145    /// The most bytes a pattern holds: an Airy pattern's table.
146    pub(crate) const BYTES: u64 = AnnularApertureVoltageTable::BYTES as u64;
147
148    /// The power pattern on model plane `plane`, `[y][x]`, centred on the
149    /// pixel `centre`, at the plane's frequency (`BeamSkyJones::apply`
150    /// squares the voltage pattern).
151    ///
152    /// The radius is `PBMath1D::apply`'s: each axis's squared offset in
153    /// degrees is formed in `Double` and kept as `Float`, their sum and its
154    /// root are `Float`, and the root times 60 · ν/GHz is kept as `Float`
155    /// before the lookup, so a radius at a table bin's edge falls as CASA's
156    /// does.
157    pub(super) fn power_plane(
158        &self,
159        model: &SkyModel,
160        plane: usize,
161        centre: [f64; 2],
162    ) -> Array2<f32> {
163        let [nx, ny] = model.shape();
164        let frequency_hz = model.plane_frequency_hz(plane);
165        let factor = 60.0 * frequency_hz / 1.0e9;
166        let increment_deg = model.increment_rad.map(f64::to_degrees);
167        let squared = |pixel: usize, axis: usize| {
168            ((pixel as f64 - centre[axis]) * increment_deg[axis]).powi(2) as f32
169        };
170        let lookup = self.lookup_at(frequency_hz);
171        Array2::from_shape_fn((ny, nx), |(y, x)| {
172            let radius = (f64::from((squared(x, 0) + squared(y, 1)).sqrt()) * factor) as f32;
173            lookup.power(f64::from(radius))
174        })
175    }
176
177    /// The power pattern at `separation_rad` from the pointing, at
178    /// `frequency_hz`, as `PBMath1D::apply` evaluates it for a component:
179    /// the separation in arcmin times the frequency in GHz, in `Double`,
180    /// looked up in the table (the EVLA's fit of that frequency, as
181    /// `PBMath1DEVLA` takes it per channel), and zero beyond it.
182    pub(crate) fn power_at(&self, separation_rad: f64, frequency_hz: f64) -> f32 {
183        let radius = separation_rad.to_degrees() * 60.0 * frequency_hz / 1.0e9;
184        self.lookup_at(frequency_hz).power(radius)
185    }
186
187    /// The power pattern `offset_rad` along an image's x axis from its
188    /// centre, at `frequency_hz`, with [`Self::power_plane`]'s arithmetic
189    /// (`SDGrid::findPBAsConvFunction` applies the beam to a one-row
190    /// image).
191    pub(super) fn power_on_axis(&self, offset_rad: f64, frequency_hz: f64) -> f32 {
192        let ghz = frequency_hz / 1.0e9;
193        let l_deg = offset_rad.to_degrees() as f32;
194        let radius_deg = (l_deg * l_deg).sqrt();
195        let radius = (f64::from(radius_deg) * 60.0 * ghz) as f32;
196        self.lookup_at(frequency_hz).power(f64::from(radius))
197    }
198
199    /// `PBMath1D::support`: the pattern's diameter in pixels of
200    /// `increment_x_rad` at `frequency_hz`, rounded down.
201    pub(super) fn support_pixels(&self, increment_x_rad: f64, frequency_hz: f64) -> usize {
202        let maximum_arcmin_ghz = match &self.0 {
203            Lookup::Airy(table) => table.maximum_radius(),
204            Lookup::Evla => EvlaPowerPattern::MAXIMUM_RADIUS_ARCMIN_GHZ,
205        };
206        let radius_rad = (maximum_arcmin_ghz / 60.0).to_radians();
207        (radius_rad / increment_x_rad.abs() * 2.0 * 1.0e9 / frequency_hz).floor() as usize
208    }
209
210    /// The lookup at `frequency_hz`: the Airy table, or the EVLA's fit of
211    /// that frequency.
212    fn lookup_at(&self, frequency_hz: f64) -> PlaneLookup<'_> {
213        match &self.0 {
214            Lookup::Airy(table) => PlaneLookup::Airy(table),
215            Lookup::Evla => PlaneLookup::Evla(EvlaPowerPattern::at(frequency_hz)),
216        }
217    }
218}
219
220/// A pattern's lookup on one plane: the Airy table, or the EVLA fit of the
221/// plane's frequency.
222enum PlaneLookup<'a> {
223    Airy(&'a AnnularApertureVoltageTable),
224    Evla(EvlaPowerPattern),
225}
226
227impl PlaneLookup<'_> {
228    /// The power at `radius_arcmin_ghz`.
229    fn power(&self, radius_arcmin_ghz: f64) -> f32 {
230        match self {
231            Self::Airy(table) => {
232                let voltage = table.evaluate(radius_arcmin_ghz);
233                voltage * voltage
234            }
235            Self::Evla(fit) => fit.power(radius_arcmin_ghz),
236        }
237    }
238}

240#[cfg(test)]
241mod tests {
242    use super::*;
243
244    #[test]
245    fn telescopes_match_casas_common_beams_by_prefix_and_band() {
246        let beam = |telescope: &str, ghz: f64| PrimaryBeam::for_telescope(telescope, ghz * 1.0e9);
247        assert_eq!(beam("ALMA", 230.0), Some(PrimaryBeam::ALMA));
248        assert_eq!(beam("ALMASD", 100.0), Some(PrimaryBeam::ALMA));
249        assert_eq!(beam("ATF", 230.0), Some(PrimaryBeam::ALMA));
250        assert_eq!(beam("ACA", 230.0), Some(PrimaryBeam::ACA));
251        assert_eq!(beam("ACA-7m", 230.0), Some(PrimaryBeam::ACA));
252        // The VLA's pattern follows spectral window 0's band: between CASA's
253        // bands (3 GHz) it is the NVSS pattern.
254        assert_eq!(beam("VLA", 1.4), Some(PrimaryBeam::VLA));
255        assert_eq!(beam("VLA", 45.0), Some(PrimaryBeam::VLA));
256        assert_eq!(beam("VLA", 3.0), Some(PrimaryBeam::VLA_NVSS));
257        assert_eq!(beam("EVLA", 1.4), Some(PrimaryBeam::Evla));
258        assert_eq!(beam("EVLA", 18.5), Some(PrimaryBeam::Evla));
259        for other in ["alma", "Unknown", "", "SMA", "AL", "NGVLA"] {
260            assert_eq!(beam(other, 1.4), None, "{other}");
261        }
262    }
