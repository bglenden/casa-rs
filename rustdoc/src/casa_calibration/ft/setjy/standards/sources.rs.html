218pub(super) fn identify(
219    field: &str,
220    direction: &MDirection,
221    cone_table: Option<&Path>,
222) -> MsResult<Option<&'static str>> {
223    if let Some(name) = named(field) {
224        return Ok(Some(name));
225    }
226    let Some(path) = cone_table else {
227        return Ok(None);
228    };
229    let table = Table::open(TableOptions::new(path))?;
230    let name = table.column_accessor("Name")?;
231    let ra = table.column_accessor("RA_J2000")?;
232    let dec = table.column_accessor("Dec_J2000")?;
233    let lon = direction.longitude_rad();
234    let lat = direction.latitude_rad();
235    let cone = (0.5_f64.to_radians() / 3600.).sin().powi(2);
236    for row in 0..table.row_count() {
237        let (ScalarValue::Float64(ra), ScalarValue::Float64(dec)) =
238            (ra.scalar_cell(row)?, dec.scalar_cell(row)?)
239        else {
240            return Err(invalid(
241                "flux calibrator direction must be in double-precision radians",
242            ));
243        };
244        let haversine = ((dec - lat) * 0.5).sin().powi(2)
245            + dec.cos() * lat.cos() * ((ra - lon) * 0.5).sin().powi(2);
246        if haversine > cone {
247            continue;
248        }
249        let ScalarValue::String(name) = name.scalar_cell(row)? else {
250            return Err(invalid("flux calibrator Name must be a string"));
251        };
252        return Ok(CALIBRATORS
253            .iter()
254            .map(|(known, _)| *known)
255            .find(|known| name == known));
256    }
257    Ok(None)
258}
259
260fn named(field: &str) -> Option<&'static str> {
261    let field = field.to_ascii_uppercase();
262    CALIBRATORS.iter().find_map(|&(name, aliases)| {
263        aliases
264            .iter()
265            .any(|alias| field.contains(*alias))
266            .then_some(name)
267    })
268}
