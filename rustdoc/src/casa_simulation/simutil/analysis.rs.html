2//! The `simutil` analysis that `simanalyze` and `simalma` run on a simulated
3//! image (simulation Phase 6f, #841): CASA 6.7.6's `simutil.py` and the
4//! analysis section of `task_simanalyze.py`, ported as their specification
5//! (ADR-0018).
6//!
7//! | CASA | casa-rs |
8//! |---|---|
9//! | `cellsize` | [`cell_size`] |
10//! | `statim(plot=False)` | [`statim`] |
11//! | `ismstp` | [`is_total_power`] |
12//! | `convimage` | [`convimage()`] |
13//! | `simanalyze`'s difference, fidelity and scalar fidelity | [`fidelity()`] |
14//! | `flatimage` | [`flat_image`](super::skymodel::flat_image) |
15//!
16//! `simanalyze` compares the image it made with the sky model it simulated:
17//! - both are flattened to one Stokes I plane (`.flat`, the channel mean of a
18//!   cube);
19//! - [`convimage()`] regrids the model's flat onto the image's flat and
20//!   convolves it with the image's restoring beam, giving the model as the
21//!   telescope would see it at full sensitivity;
22//! - [`fidelity()`] subtracts the image from that and divides the convolved
23//!   model by the difference, floored at the difference's median absolute
24//!   deviation over √2.
25//!
26//! **Where casa-rs differs**, each recorded in `docs/CASA (C++) bugs.md`:
27//! - `simanalyze` writes the fidelity's floor into its image expression with
28//!   `%f`, six decimal places: a floor under 5e-7 becomes 0, and one of
29//!   7.07e-6 becomes 7e-6. casa-rs uses the floor itself.
30//! - CASA's fidelity image carries the convolved model's Jy/beam; casa-rs
31//!   gives the ratio no unit.
32//! - CASA computes the difference and fidelity with image expressions,
33//!   whose evaluator labels any expression containing "pa" as a position
34//!   angle (unit deg, Stokes axis `Pangle`), so a project whose path
35//!   contains "pa" gets them so labelled. casa-rs computes them directly and
36//!   keeps the Stokes axis.
37//! - `statim` multiplies the statistics of a Jy/beam image without a
38//!   restoring beam by the pixel area in square arcseconds, and still reports
39//!   them as Jy/beam. casa-rs reports the image's own values.
40//! - `ismstp` tests `uvw.all() == 0`, which is true when any UVW value is
41//!   zero; casa-rs tests what was meant, that all are.
42//!
43//! The plotting parts of `statim` (the display range from the cumulative
44//! histogram, and the figure) go with the graphics (Phase 6i).

64/// What the analysis functions refuse.
65#[derive(Debug, Error)]
66pub enum AnalysisError {
67    /// A refusal CASA makes with its own message (`simutil.msg` at error
68    /// priority, which raises), or one casa-rs makes where CASA fails
69    /// without a message of its own.
70    #[error("{0}")]
71    Analysis(String),
72    /// An image that cannot be read or written.
73    #[error(transparent)]
74    Image(#[from] ImageError),
75    /// Statistics that cannot be computed.
76    #[error(transparent)]
77    Lattice(#[from] LatticeError),
78    /// A MeasurementSet table that cannot be read.
79    #[error("{}: {reason}", table.display())]
80    Table {
81        /// The table.
82        table: PathBuf,
83        /// The failure.
84        reason: String,
85    },
86    /// A file-system operation that failed.
87    #[error("{}: {source}", path.display())]
88    Io {
89        /// The path.
90        path: PathBuf,
91        /// The failure.
92        source: std::io::Error,
93    },
94}

96/// `simutil.cellsize`: the image's cell on the sky in radians, each
97/// direction increment's magnitude scaled by the magnitude of its diagonal
98/// element of the coordinate's linear transform.
99///
100/// # Errors
101///
102/// No direction coordinate, or one rotated by more than an off-diagonal
103/// element of 1e-4, which CASA refuses.
104pub fn cell_size(coordinates: &CoordinateSystem) -> Result<[f64; 2], AnalysisError> {
105    let (index, _) = pixel_axes(coordinates, CoordinateType::Direction)
106        .ok_or_else(|| AnalysisError::Analysis("the image has no direction coordinate".into()))?;
107    let CoordinateModel::Direction(direction) = coordinates.coordinate(index) else {
108        unreachable!("the coordinate found is a direction coordinate");
109    };
110    let pc = direction.pc_matrix();
111    if pc[[0, 1]].abs().max(pc[[1, 0]].abs()) > 1e-4 {
112        return Err(AnalysisError::Analysis(
113            "Your image is rotated with respect to Lat/Lon.  I can't cope with that yet".into(),
114        ));
115    }
116    let increment = direction.increment();
117    Ok([
118        increment[0].abs() * pc[[0, 0]].abs(),
119        increment[1].abs() * pc[[1, 1]].abs(),
120    ])
121}

123/// What `simutil.statim` returns: the image's extremes, the rms of its
124/// lower-left corner, and its brightness unit.
125#[derive(Debug, Clone, PartialEq)]
126pub struct ImageStatistics {
127    /// The smallest unmasked pixel; 0 when every pixel is masked.
128    pub min: f64,
129    /// The largest unmasked pixel; 1 when every pixel is masked.
130    pub max: f64,
131    /// The rms of the unmasked pixels of the first quarter of each direction
132    /// axis, through every plane; 0 when all of those are masked.
133    pub rms: f64,
134    /// The brightness unit.
135    pub unit: String,
136}
137
138/// `simutil.statim(image, plot=False)`.
139///
140/// Like CASA, it first takes the image's [`cell_size`], so it refuses an
141/// image whose direction axes are rotated. The values are the image's own.
142/// CASA scales them by a flux-per-pixel factor, which is 1 for Jy/pixel, for
143/// Jy/beam with a restoring beam and for the units it does not know. For
144/// Jy/beam without a beam the factor is the pixel area in square arcseconds,
145/// while CASA still reports the unit as Jy/beam; casa-rs does not scale them
146/// (`docs/CASA (C++) bugs.md`). The corner is CASA's `rg.box([0, 0],
147/// [nx/4, ny/4])`: pixels 0 to `nx/4` rounded half up, inclusive, on the
148/// image's first two axes.
149///
150/// # Errors
151///
152/// An image that cannot be read, or that [`cell_size`] refuses.
153pub fn statim(image: &Path) -> Result<ImageStatistics, AnalysisError> {
154    let image = PagedImage::<f32>::open(image)?;
155    cell_size(image.coordinates())?;
156    let shape = image.shape().to_vec();
157    let pixels = image.get()?;
158    let mask = image.default_mask()?;
159    let corner = [0, 1].map(|axis| {
160        let top = (shape[axis] as f64 * 0.25 + 0.5).floor() as usize;
161        top.min(shape[axis] - 1)
162    });
163    let (mut min, mut max) = (f64::INFINITY, f64::NEG_INFINITY);
164    let (mut sum_squares, mut count) = (0.0, 0_usize);
165    for (position, value) in pixels.indexed_iter() {
166        if mask.as_ref().is_some_and(|mask| !mask[&position]) {
167            continue;
168        }
169        let value = f64::from(*value);
170        min = min.min(value);
171        max = max.max(value);
172        if position[0] <= corner[0] && position[1] <= corner[1] {
173            sum_squares += value * value;
174            count += 1;
175        }
176    }
177    let any = min <= max;
178    Ok(ImageStatistics {
179        min: if any { min } else { 0.0 },
180        max: if any { max } else { 1.0 },
181        rms: if count == 0 {
182            0.0
183        } else {
184            (sum_squares / count as f64).sqrt()
185        },
186        unit: image.units().to_string(),
187    })
188}

190/// `simutil.ismstp`: the MeasurementSet is total power when its first
191/// antenna is named `…TP…`, when it has one antenna, or when every UVW
192/// value is zero. CASA tests `uvw.all() == 0`, true when any value is zero;
193/// casa-rs tests what was meant.
194///
195/// # Errors
196///
197/// A MeasurementSet whose ANTENNA table or UVW column cannot be read.
198pub fn is_total_power(ms: &Path) -> Result<bool, AnalysisError> {
199    let failed = |reason: &dyn std::fmt::Display| AnalysisError::Table {
200        table: ms.to_path_buf(),
201        reason: reason.to_string(),
202    };
203    let set = MeasurementSet::open(ms).map_err(|error| failed(&error))?;
204    let antennas = set.antenna().map_err(|error| failed(&error))?;
205    let names = (0..antennas.row_count())
206        .map(|row| antennas.name(row))
207        .collect::<Result<Vec<_>, _>>()
208        .map_err(|error| failed(&error))?;
209    if names.first().is_some_and(|name| name.contains("TP")) || names.len() == 1 {
210        return Ok(true);
211    }
212    let main = set.main_table();
213    let column = main
214        .column_accessor("UVW")
215        .map_err(|error| failed(&error))?;
216    let rows: Vec<usize> = (0..main.row_count()).collect();
217    for chunk in rows.chunks(1 << 16) {
218        for cell in column
219            .array_cells_owned(chunk)
220            .map_err(|error| failed(&error))?
221        {
222            if let Some(ArrayValue::Float64(uvw)) = cell
223                && uvw.iter().any(|&value| value != 0.)
224            {
225                return Ok(false);
226            }
227        }
228    }
229    Ok(true)
230}

232/// The one restoring beam of an image's information, which `convimage`
233/// reads as `ia.restoringbeam()`.
234fn single_beam(info: &ImageInfo, image: &Path) -> Result<GaussianBeam, AnalysisError> {
235    info.beam_set
236        .single_beam()
237        .filter(|beam| !beam.is_null())
238        .ok_or_else(|| {
239            AnalysisError::Analysis(format!(
240                "{}: convimage needs an image with one restoring beam",
241                image.display()
242            ))
243        })
244}

246/// The element-wise AND of two optional masks: what an image expression of
247/// two images keeps.
248fn both(a: Option<ArrayD<bool>>, b: Option<ArrayD<bool>>) -> Option<ArrayD<bool>> {
249    match (a, b) {
250        (Some(mut a), Some(b)) => {
251            a.zip_mut_with(&b, |a, b| *a = *a && *b);
252            Some(a)
253        }
254        (a, b) => a.or(b),
255    }
256}

258/// Write `pixels` as a new single-precision image at `path`, replacing what
259/// is there, with `info` and, when given, `mask` as the default `mask0`.
260fn write_product(
261    path: &Path,
262    pixels: &ArrayD<f32>,
263    coordinates: CoordinateSystem,
264    units: &str,
265    info: &ImageInfo,
266    mask: Option<&ArrayD<bool>>,
267) -> Result<PagedImage<f32>, AnalysisError> {
268    let removed = if path.is_dir() {
269        std::fs::remove_dir_all(path)
270    } else if path.exists() {
271        std::fs::remove_file(path)
272    } else {
273        Ok(())
274    };
275    removed.map_err(|source| AnalysisError::Io {
276        path: path.to_path_buf(),
277        source,
278    })?;
279    let shape = pixels.shape().to_vec();
280    let mut image = PagedImage::<f32>::create(shape.clone(), coordinates, path)?;
281    image.put_slice(pixels, &vec![0; shape.len()])?;
282    image.set_units(units)?;
283    image.set_image_info(info)?;
284    if let Some(mask) = mask {
285        image.put_mask("mask0", mask)?;
286        image.set_default_mask("mask0")?;
287    }
288    image.save()?;
289    Ok(image)
290}
