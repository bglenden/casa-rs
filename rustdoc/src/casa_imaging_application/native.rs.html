71/// Why a native run did not complete.
72#[derive(Debug, thiserror::Error)]
73pub(crate) enum NativeError {
74    /// A phase's memory did not fit the resource policy.
75    #[error(transparent)]
76    Admission(#[from] Admission),
77    /// The run was cancelled.
78    #[error(transparent)]
79    Cancelled(#[from] Cancelled),
80    /// Any other failure.
81    #[error("{0}")]
82    Other(ApplicationError),
83}

85impl From<NativeError> for ApplicationDispatchError {
86    fn from(error: NativeError) -> Self {
87        match error {
88            NativeError::Admission(admission) => Self::Admission(admission),
89            NativeError::Cancelled(_) => Self::Cancelled,
90            NativeError::Other(error) => Self::Native(error),
91        }
92    }
93}

107impl From<ImagingError> for NativeError {
108    fn from(error: ImagingError) -> Self {
109        match error {
110            ImagingError::Admission(admission)
111            | ImagingError::Minor(casa_imaging_runtime::MinorCycleRunError::Admission(admission)) => {
112                Self::Admission(admission)

114            ImagingError::Cancelled(cancelled) => Self::Cancelled(cancelled),

167/// The batch route of `problem` on `observation`, the observation compile
168/// opened: each store's selection resolved to its partitions and its
169/// geometry compiled, and the batches of every store admitted
170/// ([`imaging::admit_batches`]). A run that writes the model (`model_data`)
171/// or the corrected data (`corrected_data`) reads each store through the
172/// update session that writes them ([`imaging::WriteBack`]);
173/// `predicts_final_model` says which channels of each time the model's
174/// write takes.
175pub(crate) fn batch_route(
176    problem: &CompiledProblem,
177    observation: OpenedObservation,
178    (model_data, corrected_data, predicts_final_model): (bool, bool, bool),
179    runtime: &ApplicationRuntime,
180) -> Result<Admitted<imaging::BatchRoute>, NativeError> {
181    let other = |error: Box<dyn std::error::Error + Send + Sync>| NativeError::Other(error);
182    let OpenedObservation {
183        stores,
184        measures,
185        phase_table,
186    } = observation;
187    let mut opening = Reservation::none();
188    let mut routes = Vec::with_capacity(stores.len());
189    for (source, Admitted { value, reservation }) in stores.into_iter().enumerate() {
190        opening.join(reservation);
191        let selection = resolve_partitions(problem, source, value.observation())
192            .map_err(|error| other(Box::new(error)))?;
193        let plan = GeometryPlan::compile(
194            problem,
195            value.catalogue(),
196            Arc::clone(&measures),
197            phase_table.clone(),
198        )
199        .map_err(|error| other(Box::new(error)))?;
200        let pointing_budget = if plan.reads_pointing() {
201            value
202                .observation()
203                .pointing_bytes()
204                .map_err(|error| other(Box::new(error)))?
205        } else {
206            0
207        };
208        routes.push((value, selection, plan, pointing_budget));
209    }
210    let Admitted {
211        value: batching,
212        reservation,
213    } = imaging::admit_batches(
214        problem,
215        &routes
216            .iter()
217            .map(
218                |(store, selection, plan, pointing_budget)| imaging::StoreBatches {
219                    observation: store.observation(),
220                    selection,
221                    plan,
222                    pointing_budget: *pointing_budget,
223                },
224            )
225            .collect::<Vec<_>>(),
226        imaging::row_samples(problem)?,
227        (&runtime.host, &runtime.resource_policy),
228    )?;
229    opening.join(reservation);
230    let write_back = (model_data || corrected_data).then(|| {
231        imaging::WriteBack::new(problem, model_data, corrected_data, !predicts_final_model)
232    });
233    let stores = routes
234        .into_iter()
235        .map(|(opened, selection, plan, pointing_budget)| {
236            let store = match &write_back {
237                Some(write_back) => imaging::RouteStore::Write(Box::new(
238                    write_back
239                        .session(opened)
240                        .map_err(|error| other(Box::new(error)))?,
241                )),
242                None => imaging::RouteStore::Read(opened.into_observation()),
243            };
244            Ok(imaging::StoreRoute {
245                store,
246                selection,
247                plan,
248                pointing_budget,
249            })
250        })
251        .collect::<Result<Vec<_>, NativeError>>()?;
252    Ok(Admitted {
253        value: imaging::BatchRoute {
254            stores,
255            batching,
256            write_back,
257        },
258        reservation: opening,
259    })
260}

262/// The end of a run's preparation: check the products against `problem`
263/// and admit the source, its pass storage sized for the policy's streaming
264/// workers ([`casa_imaging_runtime::ResourcePolicy::cores`]), the most the run's setup then
265/// chooses.
266pub(crate) fn prepare(
267    problem: &CompiledProblem,
268    input: NativeInput,
269    native: ApplicationNative,
270) -> Result<NativeRun, NativeError> {
271    let mvc_taylor = input
272        .mvc_taylor
273        .map(|view| view.compile_from_cube(problem))
274        .transpose()
275        .map_err(|error| NativeError::Other(Box::new(error)))?;
276    let ApplicationNative {
277        runtime,
278        mut publication,
279        aw_catalog,
280        awp2,
281    } = native;
282    if let Some(existing) = &input.existing
283        && let Some(beam) = existing.mosaic_beam()
284    {
285        publication.controls = publication
286            .controls
287            .preserving_mosaic_products(beam.reading_bytes(), existing.has_higher_primary_beam());
288    }
289    publication.controls.validate_for_problem(problem)?;
290    let saved_controls = input
291        .existing
292        .as_ref()
293        .and_then(crate::restart::ExistingProducts::taylor_mask_read_bytes)
294        .map(|bytes| publication.controls.preserving_taylor_residuals(bytes));
295    let products = if let Some(saved) = saved_controls {
296        let saved_bytes = products_floor(problem, &publication, saved)?;
297        if problem.reconstruction().controls().max_minor_iterations() == 0 {
298            saved_bytes
299        } else {
300            saved_bytes.max(products_floor(problem, &publication, publication.controls)?)
301        }
302    } else {
303        products_floor(problem, &publication, publication.controls)?
304    };
305    let measures = input
306        .psf_view
307        .as_ref()
308        .map(|_| Arc::clone(&input.observation.measures));
309    let Admitted {
310        value: route,
311        reservation: source,
312    } = batch_route(
313        problem,
314        input.observation,
315        (
316            input.write_model_column,
317            input.write_corrected_data,
318            input.predicts_final_model,
319        ),
320        &runtime,
321    )?;
322    let psf_view = input
323        .psf_view
324        .zip(measures)
325        .map(|(view, measures)| psf::PsfRoute::new(view, measures, &route, &runtime))
326        .transpose()?;
327    Ok(NativeRun {
328        mvc_taylor,
329        psf_view,
330        runtime,
331        publication,
332        aw_catalog,
333        awp2,
334        route: Box::new(route),
335        source,
336        products,
337        masks: input.masks,
338        start_model: input.start_model,
339        existing: input.existing,
340        forms: input.forms,
341        image_response: input.minor_cycle_image_response,
342        virtual_model: input.virtual_model,
343    })
344}

346/// Run the major-cycle passes, the minor cycles between them and any
347/// visibility write, then publish the products. `summary` collects the
348/// phases and totals as they end; a completed run's outcome takes it once
349/// the caller has finished it.
350pub(crate) fn run(
351    problem: &CompiledProblem,
352    run: NativeRun,
353    summary: &mut RunSummary,
354) -> Result<impl FnOnce(RunSummary) -> ImagingOutcome + use<>, NativeError> {
355    let NativeRun {
356        mvc_taylor,
357        psf_view,
358        runtime,
359        publication,
360        aw_catalog,
361        awp2,
362        mut route,
363        source,
364        products,
365        masks,
366        start_model,
367        existing,
368        forms,
369        image_response,
370        virtual_model,
371    } = run;
372    let saved_mosaic_beam = existing
373        .as_ref()
374        .and_then(crate::restart::ExistingProducts::mosaic_beam)
375        .cloned();
376    let saved_psf_beams = existing
377        .as_ref()
378        .and_then(crate::restart::ExistingProducts::psf_beams);
379    let psf_beams = existing
380        .as_ref()
381        .and_then(crate::restart::ExistingProducts::psf_pixel_beams);
382    let mask_beams = existing
383        .as_ref()
384        .and_then(crate::restart::ExistingProducts::mask_beams);
385    // An unlabelled saved PSF is fitted by getBeamSet's default cutoff at
386    // minor-cycle initialization. Publication must reuse that fit's cutoff.
387    let mut publication = publication;
388    if !forms.psf
389        && psf_beams.is_none()
390        && problem.reconstruction().controls().max_minor_iterations() > 0
391    {
392        publication.controls = publication
393            .controls
394            .with_psf_cutoff(casa_imaging_deconvolution::DEFAULT_PSF_FIT_CUTOFF)?;
395    }
396    if !forms.psf {
397        publication.controls = publication.controls.with_reused_single_psf();
398    }
399    let vla_selection_hz = publication
400        .vla_selection_store
401        .map(|store| vla_selection_frequency(problem, &route, store));
402    let teams = OnceCell::new();
403    let primary_beam = match publication.controls.primary_beam_model() {
404        Some(model) => {
405            AnalyticPrimaryBeam::new(problem, model, publication.primary_beam_centres.as_ref())?
406                .map(|beam| {
407                    vla_selection_hz.map_or(beam, |frequency| {
408                        beam.with_vla_selection_frequency(frequency)
409                    })
410                })
411        }
412        None => None,
413    };
414    let separate_psf = psf_view
415        .map(|view| view.form(&mut route, &runtime, summary))
416        .transpose()?;
417    // The run's source borrows the batch route, which is released with the
418    // source's charge once the run has finished with it.
419    let visibility_write = route
420        .write_back()
421        .map(|_| imaging::VisibilityWriteTarget(&route));
422    let outcome = imaging::run(
423        imaging::ImagingInputs {
424            problem,
425            mvc: mvc_taylor.as_ref().and_then(|view| {
426                (view
427                    .problem
428                    .reconstruction()
429                    .controls()
430                    .max_minor_iterations()
431                    > 0)
432                .then_some(imaging::MvcCycleSpec {
433                    taylor_problem: &view.problem,
434                    start: view.start.as_ref(),
435                    expansion_reference_hz: view.expansion_reference_hz,
436                    controls: publication.controls,
437                    centres: publication.primary_beam_centres.as_ref(),
438                })
439            }),
440            psf_cutoff: if !forms.psf && psf_beams.is_none() {
441                casa_imaging_deconvolution::DEFAULT_PSF_FIT_CUTOFF
442            } else {
443                publication.controls.psf_cutoff()
444            },
445            psf_beams,
446            mask_beams,
447            source: &route,
448            separate_psf,
449            masks,
450            start_model,
451            existing: existing.clone(),
452            forms,
453            image_response,
454            primary_beam,
455            visibility_write,
456            products: (products, source.memory()),
457            host: runtime.host,
458            policy: runtime.resource_policy,
459            teams: &teams,
460            cancel: runtime.cancel.clone(),
461            spill_directory: &runtime.spill_directory,
462            aw_catalog,
463            awp2,
464            backend: runtime.backend,
465            grid_precision: runtime.grid_precision,
466            ft_padding: runtime.ft_padding,
467            diagnostics: runtime.diagnostics,
468        },
469        summary,
470    )?;
471    drop(route);
472    drop(source);
473    let team = teams
474        .get()
475        .expect("a run that completed formed its teams")
476        .compute();
477    let mut published = Vec::new();
478    let mut lanes = (0, None);
479    let mvc_run = mvc_taylor.is_some();
480    // CASA creates the cube model when it invokes its first minor cycle.
481    let cube_model_created = outcome.mvc_taylor.is_none()
482        || !outcome.minor_cycles.is_empty()
483        || mvc_taylor.as_ref().is_some_and(|view| view.start.is_some());
484    let products: Result<_, NativeError> = run_phase(
485        "products",
486        &runtime.cancel,
487        runtime.diagnostics,
488        summary,
489        || {
490            publish_run_products(
491                RunProductInputs {
492                    problem,
493                    source: ProductSource {
494                        scientific: outcome.scientific,
495                        masks: outcome.masks,
496                        facet_sum_weights: outcome.facet_sum_weights,
497                        mem_alpha: outcome.mem_alpha,
498                        saved_psf_beams,
499                        vla_selection_hz,
500                        primary_beam: saved_mosaic_beam
501                            .as_ref()
502                            .map(ProductPrimaryBeam::SavedMosaic),
503                    },
504                    taylor_view: mvc_taylor,
505                    taylor_completion: outcome.mvc_taylor,
506                    cube_model_created,
507                },
508                &runtime,
509                publication_after_cycles(publication, existing.as_ref(), outcome.major_cycle_count),
510                team,
511                &mut published,
512                &mut lanes,
513            )
514        },
515    );
516    summary.products = published;
517    (summary.product_lanes, summary.product_lanes_reason) = lanes;
518    let (products, taylor_names, names_reservation) = products?;
519    let mut retained = outcome.retained;
520    retained.join(names_reservation);
521    for plan in &virtual_model {
522        run_phase(
523            "virtual model",
524            &runtime.cancel,
525            runtime.diagnostics,
526            summary,
527            || {
528                crate::virtual_model::save(
529                    problem,
530                    products.scientific.final_model(),
531                    plan,
532                    &runtime,
533                )
534            },
535        )?;
536    }
537    let workers = team.workers();
538    let problem = problem.clone();
539    Ok(move |summary| ImagingOutcome {
540        problem,
541        minor_cycles: outcome.minor_cycles,
542        plane_rows: outcome.plane_rows,
543        final_residual_statistics: outcome.final_residual_statistics,
544        stop: outcome.stop,
545        major_cycle_count: outcome.major_cycle_count,
546        total_minor_iterations: outcome.total_minor_iterations,
547        total_actual_minor_iterations: outcome.total_actual_minor_iterations,
548        workers,
549        planes_per_wave: outcome.planes_per_wave,
550        visibility_products: outcome.visibility_products,
551        summary,
552        scientific: products.scientific,
553        planned_products: products.planned,
554        products: products.published,
555        mvc_taylor_names: taylor_names,
556        mvc_run,
557        _retained: retained,
558    })
559}

561/// CASA selects the cube's VLA aperture through its LSRK conversion layer,
562/// but evaluates it at each output plane's own frequency.
563fn vla_selection_frequency(
564    problem: &CompiledProblem,
565    route: &imaging::BatchRoute,
566    store: usize,
567) -> f64 {
568    let factor = route.stores[store]
569        .plan
570        .output_lsrk_factor()
571        .expect("a tracked cube has an LSRK conversion layer");
572    problem
573        .geometry()
574        .spectral()
575        .channel_centre_hz(0)
576        .expect("a cube has a first channel")
577        * factor
578}

580/// One run's cube or continuum products and, for MVC, its Taylor view.
581struct RunProductInputs<'a> {
582    problem: &'a CompiledProblem,
583    source: ProductSource<'a>,
584    taylor_view: Option<crate::compile::CompiledTaylorView>,
585    taylor_completion: Option<Admitted<crate::mvc::TaylorCompletion>>,
586    cube_model_created: bool,
587}

589/// Reduce MVC's cube before staging both families, or publish an ordinary run.
590fn publish_run_products(
591    inputs: RunProductInputs<'_>,
592    runtime: &ApplicationRuntime,
593    publication: ApplicationPublication,
594    team: &WorkerTeam,
595    published: &mut Vec<String>,
596    lanes: &mut (usize, Option<String>),
597) -> Result<(Products, Vec<String>, Reservation), NativeError> {
598    let RunProductInputs {
599        problem,
600        source,
601        taylor_view,
602        taylor_completion,
603        cube_model_created,
604    } = inputs;
605    if let Some(view) = taylor_view {
606        let crate::compile::CompiledTaylorView {
607            problem: taylor_problem,
608            coordinates,
609            expansion_reference_hz,
610            reservation: _view_reservation,
611            start,
612        } = view;
613        let saved = start.as_ref().and_then(|start| start.saved_psf.as_ref());
614        let taylor = if let Some(taylor) = taylor_completion {
615            taylor
616        } else {
617            crate::mvc::assemble_taylor_completion(
618                crate::mvc::TaylorAssemblyInputs {
619                    cube_problem: problem,
620                    cube: &source.scientific,
621                    taylor_problem: &taylor_problem,
622                    expansion_reference_hz,
623                    controls: publication.controls,
624                    centres: publication.primary_beam_centres.as_ref(),
625                },
626                (&runtime.host, &runtime.resource_policy, &runtime.cancel),
627            )?
628        };
629        let products = publish_mvc_products(
630            MvcProductSource {
631                cube_problem: problem,
632                cube_model_created,
633                cube: source,
634                taylor_problem: &taylor_problem,
635                taylor,
636                taylor_coordinates: coordinates,
637                saved,
638            },
639            runtime,
640            publication,
641            team,
642            published,
643            lanes,
644        )?;
645        Ok((
646            products.cube,
647            products.taylor_names,
648            products.names_reservation,
649        ))
650    } else {
651        if taylor_completion.is_some() {
652            return Err(NativeError::Other(
653                std::io::Error::other("MVC Taylor completion has no compiled view").into(),
654            ));
655        }
656        let products = publish_products(
657            problem,
658            source,
659            runtime,
660            publication,
661            team,
662            published,
663            lanes,
664        )?;
665        Ok((products, Vec::new(), Reservation::none()))
666    }
667}

699/// A run that stopped before its first major cycle still holds the saved
700/// Taylor residual and its model in image units.
701fn publication_after_cycles(
702    mut publication: ApplicationPublication,
703    existing: Option<&crate::restart::ExistingProducts>,
704    major_cycles: usize,
705) -> ApplicationPublication {
706    if major_cycles == 0
707        && let Some(bytes) =
708            existing.and_then(crate::restart::ExistingProducts::taylor_mask_read_bytes)
709    {
710        publication.controls = publication.controls.preserving_taylor_residuals(bytes);
711    }
712    publication
713}

715/// What `problem`'s products hold on one lane, before the run
716/// (`PlannedContinuumGeneration::planned_demand`): the least they can run
717/// on, which the run's setup fits beside it.
718fn products_floor(
719    problem: &CompiledProblem,
720    publication: &ApplicationPublication,
721    controls: casa_imaging_products::ContinuumProductControls,
722) -> Result<u64, NativeError> {
723    Ok(PlannedContinuumGeneration::for_problem(problem, &controls)?
724        .planned_demand(problem, ProductStoragePlan::new(1, 1)?, &publication.writer)?
725        .peak_residency_bytes())
726}

728/// Generate every planned product member into the writer's private staging,
729/// one output channel per window on the run's team, and publish them,
730/// adding each image published to `published`; a cancelled run publishes
731/// nothing. `lanes` receives the lanes the products ran on and, when fewer
732/// than the team's workers, why.
733fn publish_products(
734    problem: &CompiledProblem,
735    source: ProductSource<'_>,
736    runtime: &ApplicationRuntime,
737    mut publication: ApplicationPublication,
738    team: &WorkerTeam,
739    published: &mut Vec<String>,
740    lanes: &mut (usize, Option<String>),
741) -> Result<Products, NativeError> {
742    if let Some(alpha) = source.mem_alpha {
743        publication.writer.set_mem_alpha(alpha)?;
744    }
745    let (products, held) =
746        stage_products(problem, source, runtime, &publication, team, lanes, false)?;
747    if runtime.cancel.is_cancelled() {
748        // The staged members are removed when the writer drops.
749        return Err(Cancelled.into());
750    }
751    publication
752        .writer
753        .publish(|image: &Path| published.push(image.to_string_lossy().into_owned()))?;
754    // The charge covers the staged records until they are published.
755    drop(held);
756    Ok(products)
757}

783struct MvcProducts {
784    cube: Products,
785    taylor_names: Vec<String>,
786    names_reservation: Reservation,

789/// Stage both MVC families before publishing either. A failure or cancellation
790/// drops the shared writer and discards its staged members.
791#[cfg_attr(
792    not(test),
793    allow(dead_code, reason = "MVC publication is being wired in #1069")
794)]
795fn publish_mvc_products(
796    source: MvcProductSource<'_>,
797    runtime: &ApplicationRuntime,
798    publication: ApplicationPublication,
799    team: &WorkerTeam,
800    published: &mut Vec<String>,
801    lanes: &mut (usize, Option<String>),
802) -> Result<MvcProducts, NativeError> {
803    if runtime.cancel.is_cancelled() {
804        return Err(Cancelled.into());
805    }
806    let MvcProductSource {
807        cube_problem,
808        cube_model_created,
809        cube,
810        taylor_problem,
811        taylor,
812        taylor_coordinates,
813        saved,
814    } = source;
815    let Admitted {
816        value:
817            crate::mvc::TaylorCompletion {
818                scientific: taylor_science,
819                primary_beam,
820            },
821        reservation: taylor_reservation,
822    } = taylor;
823    let ProductSource {
824        scientific,
825        masks,
826        facet_sum_weights,
827        mem_alpha: _,
828        saved_psf_beams,
829        vla_selection_hz,
830        primary_beam: _,
831    } = cube;
832    let taylor_publication = ApplicationPublication {
833        controls: publication.controls,
834        primary_beam_centres: publication.primary_beam_centres.clone(),
835        vla_selection_store: None,
836        writer: publication.writer.with_coordinates(taylor_coordinates),
837    };
838    let (cube, cube_charge) = stage_products(
839        cube_problem,
840        ProductSource {
841            scientific,
842            masks: None,
843            facet_sum_weights,
844            mem_alpha: None,
845            saved_psf_beams,
846            vla_selection_hz,
847            primary_beam: Some(ProductPrimaryBeam::MvcCube(
848                &primary_beam,
849                saved.map(|saved| &saved.cube_pb),
850            )),
851        },
852        runtime,
853        &publication,
854        team,
855        lanes,
856        !cube_model_created,
857    )?;
858    let (taylor, taylor_charge) = stage_products(
859        taylor_problem,
860        ProductSource {
861            scientific: taylor_science,
862            masks,
863            facet_sum_weights: Default::default(),
864            mem_alpha: None,
865            saved_psf_beams: saved.and_then(|saved| saved.taylor.psf_beams()),
866            vla_selection_hz: None,
867            primary_beam: Some(ProductPrimaryBeam::MvcTaylor(&primary_beam, saved)),
868        },
869        runtime,
870        &taylor_publication,
871        team,
872        lanes,
873        false,
874    )?;
875    let count = taylor.planned.members().len();
876    let overflow = || ProductsError::ResourceDemandOverflow("MVC product names");
877    let bytes = taylor.planned.members().iter().try_fold(
878        count
879            .checked_mul(size_of::<String>())
880            .ok_or_else(overflow)?,
881        |sum, member| sum.checked_add(member.name().len()).ok_or_else(overflow),
882    )?;
883    let names_reservation = admit(
884        &runtime.host,
885        &runtime.resource_policy,
886        &Demand {
887            phase: "MVC product inventory",
888            memory: u64::try_from(bytes).map_err(|_| overflow())?,
889        },
890    )?;
891    let mut taylor_names = Vec::with_capacity(count);
892    taylor_names.extend(
893        taylor
894            .planned
895            .members()
896            .iter()
897            .map(|member| member.name().to_owned()),
898    );
899    if runtime.cancel.is_cancelled() {
900        return Err(Cancelled.into());
901    }
902    publication
903        .writer
904        .publish(|image: &Path| published.push(image.to_string_lossy().into_owned()))?;
905    drop(taylor);
906    drop(taylor_charge);
907    drop(cube_charge);
908    drop(taylor_reservation);
909    Ok(MvcProducts {
910        cube,
911        taylor_names,
912        names_reservation,
913    })
914}

916/// Stage one product family without publishing it. The caller holds its
917/// admission until every staged family has been published or discarded.
918fn stage_products(
919    problem: &CompiledProblem,
920    source: ProductSource<'_>,
921    runtime: &ApplicationRuntime,
922    publication: &ApplicationPublication,
923    team: &WorkerTeam,
924    lanes: &mut (usize, Option<String>),
925    omit_uncreated_mvc_cube_model: bool,
926) -> Result<(Products, casa_imaging_runtime::Reservation), NativeError> {
927    let ProductSource {
928        scientific,
929        masks: reconstruction_masks,
930        facet_sum_weights,
931        mem_alpha: _,
932        saved_psf_beams,
933        vla_selection_hz,
934        primary_beam,
935    } = source;
936    let mut controls = publication.controls;
937    let mut inputs = ContinuumProductInputs::from_major_cycle(problem, &scientific);
938    if let Some(beam) = primary_beam {
939        let saved = match beam {
940            ProductPrimaryBeam::MvcCube(beam, saved) => {
941                inputs = inputs.with_cube_taylor_primary_beam(beam)?;
942                saved.map(|beam| (beam, false))
943            }
944            ProductPrimaryBeam::MvcTaylor(beam, saved) => {
945                inputs = inputs
946                    .with_taylor_primary_beam(beam)?
947                    .with_taylor_residual_sum_weight(1.0)?;
948                saved.map(|saved| (&saved.taylor_pb, saved.taylor.has_higher_primary_beam()))
949            }
950            ProductPrimaryBeam::SavedMosaic(beam) => {
951                inputs = inputs.with_saved_mosaic_beam(beam)?;
952                None
953            }
954        };
955        if let Some((beam, higher)) = saved {
956            inputs = inputs.with_saved_mosaic_beam(beam)?;
957            controls = controls.preserving_mosaic_products(beam.reading_bytes(), higher);
958        }
959    }
960    if let Some(frequency) = vla_selection_hz {
961        inputs = inputs.with_vla_selection_frequency(frequency);
962    }
963    if let Some(beams) = saved_psf_beams.as_deref() {
964        inputs = inputs.with_saved_psf_beams(beams)?;
965    }
966    if let Some(masks) = reconstruction_masks.as_ref() {
967        inputs = match masks {
968            ReconstructionMaskSet::Shared(mask) => inputs.with_reconstruction_mask(mask),
969            ReconstructionMaskSet::Domains(masks) => inputs.with_domain_reconstruction_masks(masks),
970        };
971    }
972    if let Some(centres) = publication.primary_beam_centres.as_ref() {
973        inputs = inputs.with_primary_beam_centres(centres);
974    }
975    if !facet_sum_weights.is_empty() {
976        inputs = inputs.with_facet_sum_weights(&facet_sum_weights);
977    }
978    let mut planned = PlannedContinuumGeneration::new(&inputs, &controls)?;
979    if omit_uncreated_mvc_cube_model {
980        planned = planned.without_uncreated_mvc_cube_model();
981    }
982    // The products' scratch grows with their lanes, and the gridding's
983    // workers were chosen without it: the products take the most lanes, up
984    // to the team's, whose demand fits what the policy leaves free (#763
985    // review), and one when none does, which admission then refuses.
986    let free = free_memory(&runtime.host, &runtime.resource_policy);
987    let demand_on = |count: usize| -> Result<_, NativeError> {
988        let window = ProductStoragePlan::new(1, count)?;
989        Ok((
990            window,
991            planned.demand(&inputs, window, &publication.writer)?,
992        ))
993    };
994    let workers = team.workers();
995    let (mut window, mut demand) = demand_on(workers)?;
996    debug_assert_eq!(
997        planned.planned_demand(problem, window, &publication.writer)?,
998        demand,
999        "the products' demand before the run is the run's"
1000    );
1001    let needed = demand.peak_residency_bytes();
1002    if needed > free {
1003        for count in (1..workers).rev() {
1004            (window, demand) = demand_on(count)?;
1005            if demand.peak_residency_bytes() <= free {
1006                break;
1007            }
1008        }
1009        let count = window.maximum_workers();
1010        let gigabytes = |bytes: u64| format!("{:.2} GB", bytes as f64 / 1e9);
1011        let reason = if demand.peak_residency_bytes() <= free {
1012            format!(
1013                "{workers} lanes need {}, {count} fit {} free",
1014                gigabytes(needed),
1015                gigabytes(free)
1016            )
1017        } else {
1018            format!(
1019                "{workers} lanes need {} and no fewer fit {} free",
1020                gigabytes(needed),
1021                gigabytes(free)
1022            )
1023        };
1024        if count < workers {
1025            tracing::warn!("products on {count} lanes, not the team's {workers}: {reason}");
1026            lanes.1 = Some(reason);
1027        }
1028    }
1029    lanes.0 = window.maximum_workers();
1030    let held = admit(
1031        &runtime.host,
1032        &runtime.resource_policy,
1033        &Demand {
1034            phase: "products",
1035            memory: demand.peak_residency_bytes(),
1036        },
1037    )?;
1038    let generation =
1039        produce_continuum_members(&planned, &inputs, window, team, &publication.writer)?;
1040    Ok((
1041        Products {
1042            planned,
1043            scientific,
1044            published: generation,
1045        },
1046        held,
1047    ))
1048}
