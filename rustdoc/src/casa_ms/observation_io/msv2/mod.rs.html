49/// An opened MeasurementSet v2 and its catalogue.
50///
51/// Opening reads MAIN's key columns in row chunks and partitions the rows the way
52/// XRADIO's converter does (`DATA_DESC_ID` × `OBS_MODE` × `OBSERVATION_ID`
53/// × `EPHEMERIS_ID`, the feed pair, and the opening's
54/// [`PartitionScheme`](super::PartitionScheme)); each partition's rows are
55/// placed on its time × baseline grid (the `index` module). Each partition
56/// has one data group, `base`, associating `DATA`, `FLAG`,
57/// `WEIGHT_SPECTRUM` (when MAIN has it, else `WEIGHT`) and `UVW`. Antenna,
58/// field and observation identities are the subtable rows; scans are the
59/// distinct `SCAN_NUMBER`s in ascending order.
60///
61/// Opening checks every unit and measure reference the batches rely on:
62/// `TIME` and `TIME_CENTROID` share one fixed reference (a variable
63/// reference is refused); the UVW convention is resolved to the phase
64/// centres' frame ([`UvwFrame`](casa_observation::UvwFrame)); positions,
65/// diameters, angles and frequencies are in the units their types claim.
66pub struct Msv2Observation {
67    ms: MeasurementSet,
68    catalogue: Arc<Catalogue>,
69    partitions: Vec<Partition>,
70    spectral_weights: bool,
71    options: OpenOptions,
72}
73
74impl Msv2Observation {
75    /// Open the MeasurementSet at `path` and read its catalogue.
76    ///
77    /// The tables stay open with casacore read locks, and their payload
78    /// rows stay on disk: the typed column fills read only the selected
79    /// cells, as the production selected-observation source does.
80    ///
81    /// # Errors
82    ///
83    /// When the MeasurementSet cannot be opened, a subtable the catalogue
84    /// needs is missing or malformed, a unit or measure reference is
85    /// outside the trial profile, or the rows do not fit their grids (two
86    /// rows in one cell, two fields or scans at one time, several
87    /// `PROCESSOR_ID`s in one partition).
88    #[cfg(unix)]
89    pub fn open(path: impl AsRef<Path>) -> Result<Self, ObservationError> {
90        Self::open_with(path, &OpenOptions::default())
91    }

93    /// [`Self::open`] with `options`.
94    ///
95    /// # Errors
96    ///
97    /// As [`Self::open`].
98    #[cfg(unix)]
99    pub fn open_with(
100        path: impl AsRef<Path>,
101        options: &OpenOptions,
102    ) -> Result<Self, ObservationError> {
103        let ms = MeasurementSet::open_retained_read(path).map_err(storage)?;
104        Self::from_measurement_set(ms, options)
105    }

107    fn from_measurement_set(
108        ms: MeasurementSet,
109        options: &OpenOptions,
110    ) -> Result<Self, ObservationError> {
111        let antennas = catalogue::read_antennas(&ms)?;
112        let catalogue::Fields {
113            entries: fields,
114            ephemerides,
115            uvw_frame,
116        } = catalogue::fields_and_uvw(&ms)?;
117        let observations = catalogue::read_observations(&ms)?;
118        let main = ms.main_table();
119        let time_scale = catalogue::time_scale(main)?;
120        let units = catalogue::data_units(main);
121        let setups = catalogue::spectral_setups(&ms)?;
122        let modes = catalogue::read_state_modes(&ms)?;
123        let spectral_weights = main
124            .schema()
125            .is_some_and(|schema| schema.contains_column("WEIGHT_SPECTRUM"));
126        let index = Index::build(
127            main,
128            options.partition_scheme,
129            &modes,
130            &ephemerides,
131            ROWS_PER_CHUNK,
132        )?;
133        let group = DataGroupEntry {
134            name: "base".to_string(),
135            visibilities: "DATA".to_string(),
136            flags: "FLAG".to_string(),
137            weights: if spectral_weights {
138                "WEIGHT_SPECTRUM"
139            } else {
140                "WEIGHT"
141            }
142            .to_string(),
143            uvw: "UVW".to_string(),
144            units,
145            uvw_frame,
146        };
147        let mut partitions = Vec::new();
148        let mut entries = Vec::new();
149        for (key, grid) in index.partitions {
150            let setup = usize::try_from(key.data_description)
151                .ok()
152                .and_then(|row| setups.get(row))
153                .ok_or_else(|| {
154                    invalid(format!(
155                        "DATA_DESC_ID {} is not in DATA_DESCRIPTION",
156                        key.data_description
157                    ))
158                })?;
159            let observation = u32::try_from(key.observation)
160                .ok()
161                .filter(|id| (*id as usize) < observations.len())
162                .map(ObservationId)
163                .ok_or_else(|| {
164                    invalid(format!(
165                        "OBSERVATION_ID {} is not in OBSERVATION",
166                        key.observation
167                    ))
168                })?;
169            let feed = |feed: i32| {
170                u32::try_from(feed).map_err(|_| invalid(format!("negative feed {feed}")))
171            };
172            let feeds = [feed(key.feeds[0])?, feed(key.feeds[1])?];
173            entries.push(PartitionEntry {
174                spectral_window: setup.spectral_window.clone(),
175                frequencies_hz: setup.frequencies_hz.clone(),
176                frequency_frame: setup.frequency_frame,
177                channel_widths_hz: setup.channel_widths_hz.clone(),
178                correlations: setup.correlations.clone(),
179                intents: catalogue::intents(&key.obs_mode),
180                observation,
181                feeds: Some(feeds),
182                fields: super::distinct(&grid.field_per_time),
183                scans: super::distinct(&grid.scan_per_time),
184                baselines: grid.baselines.clone(),
185                time_scale,
186                time_origin: MjdHighPrec::new(0.0, 0.0),
187                data_groups: vec![group.clone()],
188            });
189            partitions.push(Partition { grid });
190        }
191        Ok(Self {
192            catalogue: Arc::new(Catalogue {
193                antennas,
194                fields,
195                scans: index.scans,
196                observations,
197                partitions: entries,
198            }),
199            partitions,
200            spectral_weights,
201            options: options.clone(),
202            ms,
203        })
204    }

227    fn partition(
228        &self,
229        id: PartitionId,
230    ) -> Result<(&Partition, &PartitionEntry), ObservationError> {
231        match (
232            self.partitions.get(id.0 as usize),
233            self.catalogue.partitions.get(id.0 as usize),
234        ) {
235            (Some(partition), Some(entry)) => Ok((partition, entry)),
236            _ => Err(ObservationError::InvalidRequest {
237                reason: format!("no partition {id:?}"),
238            }),
239        }
240    }

242    /// What `request` costs on partition `id`, from the grid and the tile
243    /// shape: every MSv2 role is a required column, so binding refuses
244    /// nothing the request can ask for beyond the request's own
245    /// consistency.
246    fn estimate_partition(
247        &self,
248        request: &AccessRequest,
249        id: PartitionId,
250    ) -> Result<(PartitionEstimate, ResolvedRequest), ObservationError> {
251        let (partition, entry) = self.partition(id)?;
252        let resolved = request.resolve(entry)?;
253        let roles = request.roles;
254        let channels = request.channels.len();
255        let (delivered, stored) = (
256            resolved.correlations.products(entry).len(),
257            entry.correlations.len(),
258        );
259        let scalars = [roles.time_centroid, roles.interval, roles.exposure]
260            .into_iter()
261            .filter(|wanted| *wanted)
262            .count();
263        let bytes_per_sample =
264            roles.sample_bytes(channels, delivered, self.spectral_weights, scalars);
265        let grid = &partition.grid;
266        // The cursor's scratch: each present row's number, sample, FLAG_ROW
267        // and time auxiliaries; and, when a subset is delivered, every stored
268        // product of the cube roles, read before the delivered ones are
269        // gathered. Padding needs none: rows are spread in place.
270        let gathered = Roles {
271            uvw: false,
272            ..roles
273        };
274        let scratch_bytes_per_sample = 2 * size_of::<usize>()
275            + usize::from(roles.flags)
276            + 8 * scalars
277            + if resolved.correlations.indices().is_some() {
278                gathered.sample_bytes(channels, stored, self.spectral_weights, 0)
279            } else {
280                0
281            };
282        let per_sample = |correlations: usize| {
283            usize::from(roles.values) * channels * correlations
284                + usize::from(roles.flags) * channels * correlations
285                + usize::from(roles.weights)
286                    * if self.spectral_weights {
287                        channels * correlations
288                    } else {
289                        correlations
290                    }
291                + usize::from(roles.uvw) * 3
292                + scalars
293        };
294        let (times, baselines) = (grid.times(), grid.baselines.len());
295        let bytes_per_time = roles.time_bytes(0);
296        let times_per_batch = times_per_batch(
297            request.batch_bytes,
298            baselines * (bytes_per_sample + scratch_bytes_per_sample) + bytes_per_time,
299            times,
300        )?;
301        let group = &entry.data_groups[resolved.data_group];
302        let tile = self
303            .ms
304            .main_table()
305            .array_column_tile_shape(&group.visibilities)
306            .map_err(table_error)?;
307        let estimate = PartitionEstimate {
308            partition: id,
309            times,
310            baselines,
311            bytes_per_sample,
312            bytes_per_time,
313            scratch_bytes_per_sample,
314            times_per_batch,
315            batches: if times_per_batch == 0 {
316                0
317            } else {
318                times.div_ceil(times_per_batch)
319            },
320            decode_units: grid.rows,
321            decoded_elements: grid.rows * per_sample(stored),
322            selected_elements: grid.rows * per_sample(delivered),
323            natural_blocks: match tile.as_deref() {
324                // A cell tile is [correlation, channel, rows].
325                Some([_, channels, rows, ..]) => NaturalBlocks {
326                    times: (*rows / baselines.max(1)).max(1),
327                    channels: *channels,
328                },
329                _ => NaturalBlocks {
330                    times: 1,
331                    channels: entry.frequencies_hz.len(),
332                },
333            },
334            padded_fraction: grid.padded_fraction(),
335        };
336        Ok((estimate, resolved))
337    }

339    /// What `request` costs ([`Observation::estimate`]).
340    ///
341    /// # Errors
342    ///
343    /// When the request names an unknown partition, data group, channel or
344    /// correlation product, or budgets less than one time.
345    pub fn estimate(&self, request: &AccessRequest) -> Result<Estimate, ObservationError> {
346        Ok(Estimate {
347            partitions: request
348                .partitions
349                .iter()
350                .map(|&id| Ok(self.estimate_partition(request, id)?.0))
351                .collect::<Result<_, ObservationError>>()?,
352        })
353    }

355    /// Bind a traversal of `request`: one cursor per partition, whole times
356    /// per batch ([`Observation::bind_read`]).
357    ///
358    /// # Errors
359    ///
360    /// As [`Self::estimate`].
361    pub fn bind_read(&self, request: &AccessRequest) -> Result<Traversal<'_>, ObservationError> {
362        let bound = request
363            .partitions
364            .iter()
365            .map(|&id| self.estimate_partition(request, id))
366            .collect::<Result<Vec<_>, _>>()?;
367        let estimate = Estimate {
368            partitions: bound.iter().map(|(estimate, _)| estimate.clone()).collect(),
369        };
370        let cursors = bound
371            .into_iter()
372            .map(|(estimate, resolved)| {
373                Msv2Cursor::bind(self, request, estimate, resolved)
374                    .map(|cursor| Box::new(cursor) as BoundCursor<'_>)
375            })
376            .collect::<Result<_, _>>()?;
377        Ok(Traversal::new(cursors, estimate))
378    }

380    /// Partition `id`'s grid ([`Observation::grid`]): its times, fields and
381    /// scans from the opening index, and the one `INTERVAL` of its rows,
382    /// read in bounded row chunks. Padding has no row, so no interval.
383    ///
384    /// # Errors
385    ///
386    /// As [`Observation::grid`].
387    pub fn grid(
388        &self,
389        id: PartitionId,
390        data_group: &str,
391    ) -> Result<TimeBaselineGrid, ObservationError> {
392        let (partition, entry) = self.partition(id)?;
393        if entry.data_group(data_group).is_none() {
394            return Err(ObservationError::InvalidRequest {
395                reason: format!("no data group {data_group} in {id:?}"),
396            });
397        }
398        let grid = &partition.grid;
399        let rows: Vec<usize> = grid
400            .cells
401            .iter()
402            .filter(|&&row| row != index::NO_ROW)
403            .map(|&row| row as usize)
404            .collect();
405        let main = self.ms.main_table();
406        let mut interval = None;
407        let mut values = Vec::new();
408        for chunk in rows.chunks(ROWS_PER_CHUNK) {
409            values.clear();
410            main.required_scalar_columns_for_rows_into(
411                chunk,
412                &mut [RequiredScalarColumnDestination::new(
413                    "INTERVAL",
414                    RequiredScalarColumnValuesMut::Float64(&mut values),
415                )],
416            )
417            .map_err(table_error)?;
418            for value in &values {
419                if *interval.get_or_insert(*value) != *value {
420                    return Err(ObservationError::Unsupported {
421                        what: "a grid with more than one integration interval",
422                    });
423                }
424            }
425        }
426        Ok(TimeBaselineGrid {
427            baselines: grid.baselines.clone(),
428            time_scale: entry.time_scale,
429            time_origin: entry.time_origin,
430            times_s: grid.times_s.clone(),
431            field_per_time: grid.field_per_time.clone(),
432            scan_per_time: grid.scan_per_time.clone(),
433            interval_s: interval.ok_or_else(|| ObservationError::InvalidRequest {
434                reason: format!("{id:?} has no rows"),
435            })?,
436        })
437    }
