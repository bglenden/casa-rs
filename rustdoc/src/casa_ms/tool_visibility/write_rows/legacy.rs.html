13/// The sampled row order and baseline layout returned by the last legacy read.
14#[derive(Clone, Copy)]
15pub struct LegacyVisibilityWriteGrid<'a> {
16    /// Logical MAIN rows represented in the returned record, in read order.
17    pub rows: &'a [usize],
18    /// The explicitly requested antenna pairs, including their order and duplicates.
19    pub baseline_pairs: Option<&'a [(i32, i32)]>,
20    /// Gap slots between ascending first-antenna groups.
21    pub gap: i32,
22}
23
24/// The current channel and polarization write selection and last baseline read.
25#[derive(Clone, Copy)]
26pub struct LegacyVisibilityWriteOptions<'a, 'grid> {
27    /// The baseline-axis mapping used only by input fields with both baseline and time axes.
28    pub baseline: Option<LegacyVisibilityWriteGrid<'grid>>,
29    /// Selected physical channels, or the bins to expand for FLAG.
30    pub channels: Option<VisibilityChannels>,
31    /// The most recent stored correlation selection or Stokes conversion.
32    pub polarization: Option<&'a VisibilityPolarization>,
33    /// Whether complex data needs an unsupported correlation permutation.
34    pub noncanonical_complex_polarization: bool,
35}
36
37impl WriteGrid {
38    fn from_legacy_read(
39        main: &Table,
40        source: LegacyVisibilityWriteGrid<'_>,
41    ) -> MsResult<(Vec<usize>, Self)> {
42        let columns = main.required_scalar_columns_owned_for_rows(
43            &["TIME", "ANTENNA1", "ANTENNA2", "DATA_DESC_ID"],
44            source.rows,
45        )?;
46        let grid = BaselineGrid::legacy(&columns, source.baseline_pairs, source.gap)?;
47        let baselines = grid.baseline_count();
48        let (rows, positions) = grid
49            .occupied_slots()
50            .iter()
51            .map(|&(slot, source_row)| {
52                (
53                    source.rows[source_row],
54                    [slot % baselines, slot / baselines],
55                )
56            })
57            .unzip();
58        Ok((
59            rows,
60            Self {
61                shape: [baselines, grid.time_count()],
62                positions,
63                collapsed: false,
64            },
65        ))
66    }
67}

69impl<'a> VisibilityWriteRecord<'a> {
70    /// Resolve legacy row-axis and baseline-axis fields against the rows they represent.
71    ///
72    /// # Errors
73    /// A recognized field cannot be converted, or its layout cannot be mapped safely.
74    pub fn new_legacy(
75        main: &Table,
76        rows: Vec<usize>,
77        items: RecordValue,
78        options: LegacyVisibilityWriteOptions<'a, '_>,
79    ) -> MsResult<Option<Vec<Self>>> {
80        let binned = options
81            .channels
82            .is_some_and(|selection| selection.width() > 1);
83        let derived = options
84            .polarization
85            .is_some_and(VisibilityPolarization::is_conversion);
86        let mut fields = Vec::with_capacity(items.fields().len());
87        for field in items.into_fields() {
88            let Some(column) = Column::from_legacy_name(&field.name) else {
89                continue;
90            };
91            if binned && column.is_visibility_data() {
92                continue;
93            }
94            if (derived && column.is_visibility_data())
95                || (options.noncanonical_complex_polarization && column.is_complex_data())
96            {
97                return Ok(None);
98            }
99            fields.push(field);
100        }
101        let mut row_columns = Vec::new();
102        let mut baseline_columns = Vec::new();
103        for mut field in write_columns(main, RecordValue::new(fields), true)? {
104            let rank = field.column.cell_rank();
105            field.select_polarization = rank == 2;
106            field.channels = options.channels.filter(|_| rank == 2);
107            match field.values.ndim().checked_sub(rank) {
108                Some(1) => row_columns.push(field),
109                Some(2) => baseline_columns.push(field),
110                _ => {}
111            }
112        }
113        let mut records = Vec::with_capacity(2);
114        if !row_columns.is_empty() {
115            validate_shape(&row_columns, &[rows.len()])?;
116            records.push(Self {
117                rows,
118                columns: row_columns,
119                grid: None,
120                polarization: options.polarization,
121            });
122        }
123        if !baseline_columns.is_empty() {
124            let saved = options.baseline.ok_or_else(|| {
125                input("call getdataold with ifraxis=True before writing a baseline-axis record")
126            })?;
127            let (rows, grid) = WriteGrid::from_legacy_read(main, saved)?;
128            // MSSelector ignores a visibility or FLAG cube with mismatched
129            // trailing grid dimensions. Valid fields in its record still write.
130            baseline_columns.retain(|field| {
131                field.column.cell_rank() != 2 || field.values.shape()[2..] == grid.shape
132            });
133            validate_shape(&baseline_columns, &grid.shape)?;
134            if !rows.is_empty() && !baseline_columns.is_empty() {
135                records.push(Self {
136                    rows,
137                    columns: baseline_columns,
138                    grid: Some(grid),
139                    polarization: options.polarization,
140                });
141            }
142        }
143        Ok(Some(records))
144    }
145}
146
147fn validate_shape(columns: &[WriteColumn], expected: &[usize]) -> MsResult<()> {
148    for field in columns {
149        let shape = field.values.shape();
150        if shape.get(field.column.cell_rank()..) != Some(expected) {
151            return Err(input(format!(
152                "putdataold {} shape {shape:?} does not match the selected axes {expected:?}",
153                field.column.name()
154            )));
155        }
156    }
157    Ok(())
158}
