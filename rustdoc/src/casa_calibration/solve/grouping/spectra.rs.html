2//! Per-correlation visibility spectra after averaging to each solve interval.
3
4use super::{Complex32, SolveAccumulator, SolveEdgeAccumulation};
5
6/// One time-averaged baseline spectrum before the solver collapses it.
7pub(in crate::solve) struct DelaySpectrum<'a> {
8    pub(in crate::solve) antennas: (i32, i32),
9    pub(in crate::solve) buffer: (i32, i32, u64),
10    pub(in crate::solve) correlation: usize,
11    pub(in crate::solve) parallel_hand: bool,
12    pub(in crate::solve) receptor: usize,
13    pub(in crate::solve) time_seconds: f64,
14    values: &'a [SolveEdgeAccumulation],
15}
16
17impl DelaySpectrum<'_> {
18    pub(in crate::solve) fn channels(&self) -> impl Iterator<Item = (Complex32, f32)> + '_ {
19        self.values
20            .iter()
21            .map(|value| (value.weighted_sum, value.weight))
22    }
23}
24
25impl SolveAccumulator {
26    pub(in crate::solve) fn delay_spectra(&self) -> impl Iterator<Item = DelaySpectrum<'_>> {
27        self.receptor_accumulations
28            .iter()
29            .enumerate()
30            .flat_map(move |(receptor, baselines)| {
31                baselines.iter().map(move |(key, values)| DelaySpectrum {
32                    antennas: key.baseline.antennas,
33                    buffer: (
34                        key.baseline.data_desc_id,
35                        key.baseline.field_id,
36                        key.baseline.start_bits,
37                    ),
38                    correlation: key.correlation_index,
39                    parallel_hand: key.parallel_hand,
40                    receptor,
41                    time_seconds: self
42                        .baseline_centroids
43                        .get(&key.baseline)
44                        .or_else(|| {
45                            self.baseline_centroids
46                                .get(&key.baseline.time_bucket_key(true))
47                        })
48                        .and_then(|centroid| centroid.mean())
49                        .unwrap_or(self.min_time),
50                    values,
51                })
52            })
53    }
54}
