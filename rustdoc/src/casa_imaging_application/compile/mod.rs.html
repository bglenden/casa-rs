85/// The MeasurementSet as [`prepare`] surveyed it: the table, its geometry
86/// engine, the source budget, the selection survey and the spectral axis.
87struct Surveyed<'a> {
88    ms: &'a MeasurementSet,
89    engine: &'a MsCalEngine,
90    budget: SelectedObservationContentBudget,
91    survey: &'a Survey,
92    spectral: &'a PreparedSpectralAxis,

95/// Compile `request` against its MeasurementSet.
96///
97/// What compile reads is admitted, not compile itself: the walks of the
98/// MeasurementSet's rows are admitted once it is open and its rows are
99/// known, before the first walk; opening it reads only its tables'
100/// descriptions, under the run's fixed charges.
101pub(crate) fn prepare(
102    request: &ImagingRequest,
103    context: &RunContext,
104) -> Result<Prepared, PrepareError> {
105    let ms = MeasurementSet::open(&request.vis)?;
106    let budget = casa_imaging_runtime::bootstrap_source_budget();
107    // The survey here and the observation's resolution after it each walk
108    // the MAIN rows under `budget`, one after the other: one walk's charge
109    // covers both.
110    let walks = casa_imaging_runtime::admit(
111        &context.host,
112        &context.policy,
113        &casa_imaging_runtime::Demand {
114            phase: "MeasurementSet walks",
115            memory: ms.row_walk_bytes(budget.row_io_budget())? as u64,
116        },
117    )?;
118    // The one Measures provider of the run: compile's geometry and the bound
119    // observation's both evaluate with it.
120    let measures = casa_ms::open_measures_runtime()?;
121    let engine =
122        MsCalEngine::with_measures(&ms, Arc::clone(&measures))?.with_field_ephemerides(&ms)?;
123    // An MFS run's spectral envelope is held beside the survey walk and
124    // admitted on its own, against the run's memory.
125    let envelope_bytes = if request.specmode == SpecMode::Mfs {
126        selection::envelope_bytes(&ms, context.policy.memory(&context.host))?
127    } else {
128        0
129    };
130    let envelope = casa_imaging_runtime::admit(
131        &context.host,
132        &context.policy,
133        &casa_imaging_runtime::Demand {
134            phase: "spectral envelope",
135            memory: envelope_bytes as u64,
136        },
137    )?;
138    let survey = selection::survey(request, &ms, budget, envelope_bytes, &engine)?;
139    drop(envelope);
140    let centre = direction::resolve_centre(request, &ms, &survey, &engine, budget)?;
141    let frame = FrameContext {
142        anchor_time_mjd_seconds: survey.first_time_mjd_seconds,
143        time_bounds_mjd_seconds: survey.time_bounds_mjd_seconds,
144        field_id: centre.field_id,
145        phase: centre.phase.clone(),
146        direction: centre.direction,
147        engine: &engine,
148    };
149    let moving_rest_frame = moving_rest_frame(request, &centre, &frame)?;
150    let spectral =
151        spectral::prepare_axis(request, &ms, &survey, &frame, moving_rest_frame.as_ref())?;
152    let surveyed = Surveyed {
153        ms: &ms,
154        engine: &engine,
155        budget,
156        survey: &survey,
157        spectral: &spectral,
158    };
159    let (continuum_transform, channels) = continuum_selection(request, &surveyed, &centre)?;
160    let observation_info = direction::observation_info(&ms, &survey, &centre, &engine)?;
161    let domains = domains::prepare_domains(
162        request,
163        &centre,
164        ImageSpectralCoordinate {
165            frequency_reference: spectral.output_frequency_reference,
166            reference_frequency_hz: spectral.reference_frequency_hz,
167            increment_hz: spectral.increment_hz,
168            rest_frequency_hz: spectral.image_rest_frequency_hz,
169        },
170        &observation_info,
171    )?;
172    let primary_beam = primary_beam_model(request, &ms)?;
173    let primary_beam_centres = match primary_beam {
174        None | Some(AnalyticPrimaryBeamModel::MosaicSensitivity) => None,
175        Some(_) => Some(direction::primary_beam_centres(
176            &domains, &ms, &survey, &centre, &engine,
177        )?),
178    };
179    let deployment = deployment(
180        request,
181        context,
182        &surveyed,
183        &domains,
184        primary_beam,
185        primary_beam_centres,
186    )?;
187    let instrument = instrument::scientific_instrument_model(&request.gridder, &ms)?;
188    let specification = specification::specification(
189        request,
190        &spectral,
191        specification_inputs(
192            request,
193            &surveyed,
194            instrument,
195            primary_beam.is_some(),
196            continuum_transform,
197        )?,
198    )?;
199    let reconstruction_planes = match request.deconvolver {
200        Deconvolver::Mtmfs => request.nterms,
201        _ => spectral.output_channels,
202    };
203    let model_samples = domains::model_samples(&domains, reconstruction_planes, &request.stokes)?;
204    let start_model = request
205        .startmodel
206        .as_deref()
207        .map(|path| {
208            StartModel::prepare(
209                path,
210                &domains[0],
211                request.stokes.len(),
212                reconstruction_planes,
213            )
214        })
215        .transpose()?;
216    let geometry = geometry(request, &domains, &centre, &spectral);
217    let observation = observation_request(
218        request,
219        &ms,
220        survey,
221        &channels,
222        measures,
223        centre.ephemeris,
224        budget,
225    )?;
226    Ok(Prepared {
227        specification,
228        geometry,
229        model_lifecycle: ModelLifecycleRequirements::new(
230            ModelBounds::new(model_samples, model_samples, f64::MAX, f64::MAX)?,
231            NumericPrecision::F64,
232        ),
233        masks: domains::mask_plans(&domains)?,
234        start_model,
235        minor_cycle_image_response: match &request.gridder {
236            Gridder::Mosaic { .. } | Gridder::Awproject(_) => Some(MinorCycleImageResponse::new(
237                specification::normalization(request),
238                specification::primary_beam_validity(request)?,
239            )?),
240            Gridder::Standard { .. } | Gridder::Wproject { .. } => None,
241        },
242        observation,
243        write_model_column: request.savemodel,
244        write_corrected_data: request.save_continuum_residual,
245        deployment,
246        walks,
247    })
248}

250/// The rest frame of a moving source (`cubesource`): the observatory frame
251/// at the anchor time with the ephemeris's radial velocity.
252fn moving_rest_frame(
253    request: &ImagingRequest,
254    centre: &Centre,
255    frame: &FrameContext<'_>,
256) -> Result<Option<MeasFrame>, PrepareError> {
257    if request.specmode != SpecMode::Cubesource {
258        return Ok(None);
259    }
260    let ephemeris = centre
261        .ephemeris
262        .as_ref()
263        .ok_or(PrepareError::MovingPhaseCentre)?;
264    let velocity = frame.engine.ephemeris_radial_velocity(
265        frame.anchor_time_mjd_seconds,
266        frame.field_id,
267        ephemeris,
268    )?;
269    Ok(Some(
270        frame
271            .engine
272            .spectral_frame_observatory_direction(
273                frame.anchor_time_mjd_seconds,
274                centre.phase.clone(),
275            )?
276            .with_radial_velocity(velocity),
277    ))
278}

280/// The continuum transform of a `fitspw` request and the source channels
281/// the observation reads: the fit channels with the output channels, or
282/// the output channels alone.
283fn continuum_selection(
284    request: &ImagingRequest,
285    surveyed: &Surveyed<'_>,
286    centre: &Centre,
287) -> Result<(Option<SequentialContinuumTransform>, WindowChannels), PrepareError> {
288    let Some(fitspw) = &request.fitspw else {
289        return Ok((None, surveyed.spectral.selected_source_channels.clone()));
290    };
291    let window = selection::one_window(&surveyed.survey.spectral_windows, "continuum subtraction")?;
292    let (transform, selected) = specification::continuum_transform(
293        fitspw,
294        request.fitorder,
295        i32::try_from(centre.field_id).expect("the centre's field id is a stored i32 FIELD_ID"),
296        window,
297        surveyed.spectral.window_channels(window.spw_id),
298    )?;
299    Ok((Some(transform), BTreeMap::from([(window.spw_id, selected)])))
300}

302/// The specification's inputs beyond the request: the instrument, the
303/// W and AW projections over the observed W range, the cube's density
304/// padding and the continuum transform.
305fn specification_inputs(
306    request: &ImagingRequest,
307    surveyed: &Surveyed<'_>,
308    instrument: Option<casa_imaging_model::InstrumentModel>,
309    primary_beam: bool,
310    continuum_transform: Option<SequentialContinuumTransform>,
311) -> Result<SpecificationInputs, PrepareError> {
312    let windows = &surveyed.survey.spectral_windows;
313    Ok(SpecificationInputs {
314        instrument,
315        primary_beam,
316        w_projection: match &request.gridder {
317            Gridder::Wproject { wprojplanes } => Some(specification::w_projection(
318                *wprojplanes,
319                windows,
320                surveyed.survey.w,
321            )?),
322            _ => None,
323        },
324        aw_projection: match &request.gridder {
325            Gridder::Awproject(aw) => Some(specification::aw_projection(
326                aw,
327                windows,
328                surveyed.survey.w,
329            )?),
330            _ => None,
331        },
332        cube_density_padding: cube_density_padding(request, surveyed)?,
333        continuum_transform,
334    })
335}

337/// The selected-observation request: the survey's rows and the channels
338/// read, the visibility and weight columns, and the Measures provider and
339/// centre ephemeris compile evaluated with.
340fn observation_request(
341    request: &ImagingRequest,
342    ms: &MeasurementSet,
343    survey: Survey,
344    channels: &WindowChannels,
345    measures: Arc<dyn MeasuresProvider>,
346    ephemeris: Option<SelectedObservationEphemeris>,
347    budget: SelectedObservationContentBudget,
348) -> Result<SelectedObservationResolutionRequest, PrepareError> {
349    let weight_column = if survey.weight_spectrum_complete {
350        OwnerWeightColumn::WeightSpectrum
351    } else {
352        OwnerWeightColumn::Weight
353    };
354    Ok(SelectedObservationResolutionRequest::new(
355        request.vis.display().to_string(),
356        selection::observation_selection(ms, survey, channels)?,
357        selection::visibility_column(ms, request.datacolumn)?,
358        weight_column,
359        budget,
360        measures,
361    )
362    .with_ephemeris(ephemeris))
363}

365/// The image geometry: the domains, the centre laws and the spectral axis.
366fn geometry(
367    request: &ImagingRequest,
368    domains: &[PreparedImageDomain],
369    centre: &Centre,
370    spectral: &PreparedSpectralAxis,
371) -> GeometryInput {
372    let direction_dependent = matches!(
373        request.gridder,
374        Gridder::Mosaic { .. } | Gridder::Awproject(_)
375    );
376    GeometryInput::new(
377        domains.iter().map(PreparedImageDomain::spec).collect(),
378        CentreLaws::new(
379            centre.law.clone(),
380            instrument::pointing_centre_law(&request.gridder),
381        ),
382        if direction_dependent {
383            UvwCoordinateLaw::MosaicPhaseTrackingCentre
384        } else {
385            UvwCoordinateLaw::PhaseTrackingCentre
386        },
387        spectral.coordinate(),
388    )
389}

391/// The primary beam the products are written with: the mosaic
392/// sensitivity of a direction-dependent gridder, else the telescope's
393/// analytic beam when `.pb` or `.image.pbcor` is asked for.
394fn primary_beam_model(
395    request: &ImagingRequest,
396    ms: &MeasurementSet,
397) -> Result<Option<AnalyticPrimaryBeamModel>, PrepareError> {
398    Ok(match request.gridder {
399        Gridder::Mosaic { .. } | Gridder::Awproject(_) => {
400            Some(AnalyticPrimaryBeamModel::MosaicSensitivity)
401        }
402        Gridder::Standard { .. } | Gridder::Wproject { .. }
403            if request.write_pb || request.pbcor =>
404        {
405            Some(instrument::standard_primary_beam_model(ms)?)
406        }
407        Gridder::Standard { .. } | Gridder::Wproject { .. } => None,
408    })
409}

411/// CASA's padded per-channel density grid of a linearly interpolated
412/// cube with per-channel density.
413fn cube_density_padding(
414    request: &ImagingRequest,
415    surveyed: &Surveyed<'_>,
416) -> Result<Option<usize>, PrepareError> {
417    let (survey, spectral) = (surveyed.survey, surveyed.spectral);
418    if !specification::cube_density_padded(request, spectral) {
419        return Ok(None);
420    }
421    let window = selection::one_window(&survey.spectral_windows, "cube density")?;
422    Ok(Some(
423        surveyed.ms.selected_observation_cube_density_padding(
424            &survey.row_selection,
425            SelectedObservationSpectralWindow::borrow_selected(
426                u32::try_from(window.spw_id).expect("SPW ids are nonnegative stored i32 values"),
427                window.frequency_reference,
428                &window.frequencies_hz,
429                &window.channel_widths_hz,
430                spectral.window_channels(window.spw_id),
431            ),
432            survey.fields.iter().copied(),
433            spectral.output_frequency_reference,
434            [
435                spectral.reference_frequency_hz,
436                spectral.reference_frequency_hz
437                    + (spectral.output_channels - 1) as f64 * spectral.increment_hz,
438            ],
439            spectral.output_channels,
440            surveyed.engine,
441            surveyed.budget.row_io_budget(),
442        )?,
443    ))
444}

446/// The native run's runtime, product publication and AW catalog, still to
447/// be deployed: no directory exists and no AW cell is generated until
448/// [`Deployment::deploy`].
449pub(crate) struct Deployment {
450    runtime: PendingRuntime,
451    publication: ApplicationPublication,
452    /// Every domain's output directory.
453    output_directories: Vec<PathBuf>,
454    aw: Option<AwPlan>,
455}
456
457/// The run's runtime but for its device, which the compiled problem resolves
458/// ([`Deployment::deploy`]); its spill directory is made canonical on
459/// deployment.
460struct PendingRuntime {
461    host: crate::HostResources,
462    resource_policy: crate::ResourcePolicy,
463    grid_precision: Option<crate::GridPrecision>,
464    cancel: crate::Cancel,
465    spill_directory: PathBuf,
466    diagnostics: crate::Diagnostics,
467}

469/// The AW catalog a run opens and, for a native cache, the cells to bring
470/// to the request's policy.
471struct AwPlan {
472    root: PathBuf,
473    native: Option<(NativeAwCachePolicy, NativeAwRequestInput)>,
474    indexing: AwIndexing,
475    resident_bytes: usize,
476    /// Whether the run raises the bound into the memory its workers leave
477    /// (`cf_resident_mb = auto`).
478    automatic: bool,
479}
480
481/// The cell cache's bound for `cf_resident_mb = auto` before the run raises
482/// it into the memory its workers leave (#449).
483const AUTOMATIC_CELL_CACHE_FLOOR: usize = 256 << 20;
484
485impl Deployment {
486    /// Create every domain's output directory, settle the spill directory
487    /// beside the main image, bring a native AW cache to the request's
488    /// policy and grid on `backend`, the device the request resolved to on
489    /// the compiled problem ([`crate::availability::resolve_backend`]).
490    pub(crate) fn deploy(self, backend: BackendChoice) -> Result<ApplicationNative, PrepareError> {
491        for directory in &self.output_directories {
492            std::fs::create_dir_all(directory).map_err(|source| PrepareError::OutputDirectory {
493                path: directory.clone(),
494                source,
495            })?;
496        }
497        let PendingRuntime {
498            host,
499            resource_policy,
500            grid_precision,
501            cancel,
502            spill_directory,
503            diagnostics,
504        } = self.runtime;
505        let spill_directory =
506            spill_directory
507                .canonicalize()
508                .map_err(|source| PrepareError::OutputDirectory {
509                    path: spill_directory.clone(),
510                    source,
511                })?;
512        Ok(ApplicationNative {
513            runtime: ApplicationRuntime {
514                host,
515                resource_policy,
516                backend,
517                grid_precision,
518                cancel,
519                spill_directory,
520                diagnostics,
521            },
522            publication: self.publication,
523            aw_catalog: self
524                .aw
525                .map(|aw| aw.deploy((&host, &resource_policy)))
526                .transpose()?,
527        })
528    }
529}

531impl AwPlan {
532    /// Under every policy the directory is compared with the request: its
533    /// request manifest, the cells the request names, and any it does not
534    /// (#735: a cell's conjugate beam and its lookup depend on the whole
535    /// frequency list). Reuse needs a directory generated for this request,
536    /// complete and holding nothing else. Generation fills in the absent
537    /// cells of such a directory, and clears and regenerates any other.
538    /// Regeneration clears the cache first. The loader then checks each
539    /// cell's sky increment against the image (plan section 5.6).
540    ///
541    /// The run reads a native cache under its shared [`NativeCacheLock`],
542    /// held in the deployment for as long as the run lives, since cells load
543    /// lazily. It writes one only under the exclusive lock, which it cannot
544    /// take while another run reads the cache. After writing, it takes the
545    /// shared lock again and checks the directory once more before it keeps
546    /// it (#739 review). Generation is admitted on `admission`'s host and
547    /// policy before it starts and released when it ends (#738).
548    fn deploy(
549        self,
550        admission: (&crate::HostResources, &crate::ResourcePolicy),
551    ) -> Result<AwCatalogDeployment, PrepareError> {
552        let (lock, generation_workers) = match &self.native {
553            Some((policy, input)) => {
554                let (lock, workers) = self.bring_native(*policy, input, admission)?;
555                (Some(Arc::new(lock)), workers)
556            }
557            None => (None, None),
558        };
559        Ok(AwCatalogDeployment {
560            root: self.root,
561            indexing: self.indexing,
562            resident_bytes: self.resident_bytes,
563            automatic: self.automatic,
564            _lock: lock,
565            generation_workers,
566        })
567    }
568
569    /// Bring the native cache to `policy` for `input` and return the shared
570    /// lock the run keeps while it reads it ([`Self::deploy`]), and the
571    /// workers that generated cells, if any did.
572    fn bring_native(
573        &self,
574        policy: NativeAwCachePolicy,
575        input: &NativeAwRequestInput,
576        (host, resources): (&crate::HostResources, &crate::ResourcePolicy),
577    ) -> Result<(NativeCacheLock, Option<usize>), PrepareError> {
578        let root = &self.root;
579        let shared = NativeCacheLock::shared(root)?;
580        let state = AwCatalog::native_cache_state(root, input)?;
581        let fill = match policy {
582            NativeAwCachePolicy::ReuseOnly if state.reusable() => return Ok((shared, None)),
583            NativeAwCachePolicy::ReuseOnly => {
584                return Err(PrepareError::AwCacheNotReusable { state });
585            }
586            NativeAwCachePolicy::GenerateMissing if state.reusable() => return Ok((shared, None)),
587            NativeAwCachePolicy::GenerateMissing if state.fillable() => true,
588            NativeAwCachePolicy::GenerateMissing => {
589                tracing::warn!(
590                    "native AW cache {} does not belong to this request ({state}); \
591                     regenerating it",
592                    root.display()
593                );
594                false
595            }
596            NativeAwCachePolicy::Regenerate => false,
597        };
598        drop(shared);
599        let exclusive = NativeCacheLock::exclusive(root)?;
600        // The most workers, up to the policy's and one per cell to generate,
601        // whose workspaces fit what the policy leaves free (#708); one when
602        // none does, which admission then refuses. Generation gives each of
603        // them a unit (#747 review).
604        let cells = if fill {
605            state.expected - state.present
606        } else {
607            state.expected
608        };
609        let free = casa_imaging_runtime::free_memory(host, resources);
610        let workers = (1..=resources.workers(host).min(cells))
611            .rev()
612            .find(|&workers| AwCatalog::generation_bytes(input, workers) <= free)
613            .unwrap_or(1);
614        let generating = casa_imaging_runtime::admit(
615            host,
616            resources,
617            &casa_imaging_runtime::Demand {
618                phase: "native AW generation",
619                memory: AwCatalog::generation_bytes(input, workers),
620            },
621        )?;
622        tracing::info!(
623            "generating {cells} native AW cells into {} on {workers} workers",
624            root.display()
625        );
626        AwCatalog::generate_native(root, input, fill, workers)?;
627        drop(generating);
628        drop(exclusive);
629        let shared = NativeCacheLock::shared(root)?;
630        let state = AwCatalog::native_cache_state(root, input)?;
631        if !state.reusable() {
632            // Another run replaced the cache between our write and our read.
633            return Err(PrepareError::AwCacheNotReusable { state });
634        }
635        Ok((shared, Some(workers)))
636    }
637}

639/// The native run's deployment: the runtime with the paged cube state
640/// beside the main image, the product controls and image writer, and the AW
641/// plan.
642fn deployment(
643    request: &ImagingRequest,
644    context: &RunContext,
645    surveyed: &Surveyed<'_>,
646    domains: &[PreparedImageDomain],
647    primary_beam: Option<AnalyticPrimaryBeamModel>,
648    primary_beam_centres: Option<PrimaryBeamCentres>,
649) -> Result<Deployment, PrepareError> {
650    let parent = |path: &Path| {
651        path.parent()
652            .unwrap_or_else(|| Path::new("."))
653            .to_path_buf()
654    };
655    let mut controls = ContinuumProductControls::new(request.psfcutoff as f32).map_err(|_| {
656        PrepareError::PsfCutoff {
657            psfcutoff: request.psfcutoff,
658        }
659    })?;
660    if let Some(model) = primary_beam {
661        controls = controls.with_primary_beam_model(model);
662    }
663    Ok(Deployment {
664        runtime: PendingRuntime {
665            host: context.host,
666            resource_policy: context.policy,
667            grid_precision: request.gridprecision,
668            cancel: context.cancel.clone(),
669            spill_directory: parent(&request.imagename),
670            diagnostics: request.diagnostics,
671        },
672        publication: ApplicationPublication {
673            controls,
674            primary_beam_centres,
675            writer: CasaImageWriter::for_domains(domains.iter().map(PreparedImageDomain::output)),
676        },
677        output_directories: domains
678            .iter()
679            .map(|domain| parent(&domain.output))
680            .collect(),
681        aw: match &request.gridder {
682            Gridder::Awproject(aw) => Some(aw_plan(request, aw, surveyed)?),
683            _ => None,
684        },
685    })
686}

688/// The AW catalog to open: a CASA cache read as it is, or a native cache
689/// in CASA's format with the input its cells are generated from.
690fn aw_plan(
691    request: &ImagingRequest,
692    aw: &AwProjection,
693    surveyed: &Surveyed<'_>,
694) -> Result<AwPlan, PrepareError> {
695    let (root, native) = match &aw.cf_source {
696        AwCfSource::CasaImport { cfcache } => (cfcache.clone(), None),
697        AwCfSource::NativeEvla {
698            native_cf_cache,
699            native_cf_policy,
700            ..
701        } => {
702            let first_row = surveyed
703                .survey
704                .first_cross_row
705                .ok_or(PrepareError::NoCrossCorrelationRow)?;
706            let input = native_aw::resolve(request, aw, surveyed, first_row)?;
707            input.validate()?;
708            (native_cf_cache.clone(), Some((*native_cf_policy, input)))
709        }
710    };
711    Ok(AwPlan {
712        root,
713        native,
714        indexing: AwIndexing {
715            conjugate_beams: true,
716            image_reference_hz: surveyed.spectral.reference_frequency_hz,
717        },
718        resident_bytes: match aw.cf_resident_mb {
719            Some(megabytes) => megabytes
720                .checked_mul(1 << 20)
721                .ok_or(PrepareError::CfResidentSize { megabytes })?,
722            None => AUTOMATIC_CELL_CACHE_FLOOR,
723        },
724        automatic: aw.cf_resident_mb.is_none(),
725    })
726}
