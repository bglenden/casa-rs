2//! The raster sampling of a total-power map, as CASA estimates it from the
3//! pointings (`SingleDishBeamUtil::getPointingSamplingRaster`, behind
4//! `im.pointingsampling(pattern='raster')`), and the sampling `tsdimaging`
5//! then takes for the theoretical beam (`set_beam_size`).
6//!
7//! The pointings are one antenna's, or several antennas' one after
8//! another, each in time order. A pointing whose time equals the one
9//! before it (to 10⁻¹⁴ relative) is dropped. Then:
10//! - along the scan, the interval is the median distance between
11//!   consecutive pointings, the longitude difference scaled by the cosine
12//!   of the map's central latitude (floored at 10⁻⁸ rad);
13//! - the scan's position angle is the arctangent of the median slope;
14//! - the scan rows end where the time step exceeds five times the median
15//!   positive step, or goes back (a new antenna), and at the end;
16//! - across the scan, the interval is the median distance between
17//!   consecutive rows' median positions, projected on the direction
18//!   perpendicular to the scan.
19//!
20//! The medians are casacore's: an even count of at most 100 takes the
21//! mean of the middle two, and a larger even count the lower one.
22//!
23//! One departure (`docs/CASA (C++) bugs.md`): CASA projects the rows'
24//! longitude differences without the cosine of the latitude, so a map
25//! scanned in latitude has its row spacing overestimated by `1/cos δ`;
26//! casa-rs scales them as it scales the along-scan steps.
27
28/// A pointing direction at a time.
29#[derive(Debug, Clone, Copy, PartialEq)]
30pub struct Pointing {
31    /// The time, seconds.
32    pub time_s: f64,
33    /// Longitude and latitude in the image's frame, rad.
34    pub direction_rad: [f64; 2],
35}
36
37/// The sampling CASA estimates for a raster scan.
38#[derive(Debug, Clone, Copy, PartialEq)]
39pub struct RasterSampling {
40    /// The interval along the scan and between the scan's rows, rad; 0
41    /// where there is none to measure (a single pointing, or no row gap).
42    pub interval_rad: [f64; 2],
43    /// The scan's position angle, rad.
44    pub position_angle_rad: f64,
45}
46
47/// `getPointingSamplingRaster` over `pointings`, each antenna's in time
48/// order; `None` for no pointings.
49#[must_use]
50pub fn raster_sampling(pointings: &[Pointing]) -> Option<RasterSampling> {
51    let first = pointings.first()?;
52    let mut longitudes: Vec<f64> = pointings.iter().map(|p| p.direction_rad[0]).collect();
53    unwrap_longitudes(&mut longitudes);
54    // Unique time stamps: a pointing differing from the previous one.
55    let mut unique = vec![0];
56    let mut current = first.time_s;
57    for (index, pointing) in pointings.iter().enumerate().skip(1) {
58        if (pointing.time_s - current).abs() > 1.0e-14 * current {
59            unique.push(index);
60        }
61        current = pointing.time_s;
62    }
63    if unique.len() == 1 {
64        return Some(RasterSampling {
65            interval_rad: [0.0; 2],
66            position_angle_rad: 0.0,
67        });
68    }
69    let times: Vec<f64> = unique.iter().map(|&i| pointings[i].time_s).collect();
70    let longitude: Vec<f64> = unique.iter().map(|&i| longitudes[i]).collect();
71    let latitude: Vec<f64> = unique
72        .iter()
73        .map(|&i| pointings[i].direction_rad[1])
74        .collect();
75    let (minimum, maximum) = latitude
76        .iter()
77        .fold((f64::INFINITY, f64::NEG_INFINITY), |(lo, hi), &v| {
78            (lo.min(v), hi.max(v))
79        });
80    let scale = (0.5 * (minimum + maximum)).cos();
81    let steps: Vec<[f64; 2]> = (0..longitude.len() - 1)
82        .map(|i| {
83            let along = (longitude[i + 1] - longitude[i]) * scale;
84            [
85                if along.abs() > 1.0e-8 { along } else { 1.0e-8 },
86                latitude[i + 1] - latitude[i],
87            ]
88        })
89        .collect();
90    let along = median(steps.iter().map(|[x, y]| x * x + y * y).collect()).sqrt();
91    let slope = median(steps.iter().map(|[x, y]| y / x).collect()).atan();
92    let position_angle_rad = if slope.is_finite() {
93        slope
94    } else {
95        0.5 * std::f64::consts::PI
96    };
97    let across = row_spacing(&times, (&longitude, &latitude), scale, position_angle_rad);
98    Some(RasterSampling {
99        interval_rad: [along, across],
100        position_angle_rad,
101    })
102}

104/// The interval between the scan's rows, 0 when no row gap is found.
105fn row_spacing(
106    times: &[f64],
107    (longitude, latitude): (&[f64], &[f64]),
108    scale: f64,
109    position_angle_rad: f64,
110) -> f64 {
111    let steps: Vec<f64> = times.windows(2).map(|t| t[1] - t[0]).collect();
112    let positive: Vec<f64> = steps.iter().copied().filter(|&step| step > 0.0).collect();
113    if positive.is_empty() {
114        return 0.0;
115    }
116    let threshold = 5.0 * median(positive);
117    let mut ends = Vec::new();
118    let mut antenna_ends = 0;
119    for (index, &step) in steps.iter().enumerate() {
120        if step > threshold {
121            ends.push(index);
122        } else if step < 0.0 || index == steps.len() - 1 {
123            ends.push(index);
124            antenna_ends += 1;
125        }
126    }
127    if ends.len() == antenna_ends {
128        return 0.0;
129    }
130    let normal = [
131        (position_angle_rad + 0.5 * std::f64::consts::PI).cos(),
132        (position_angle_rad + 0.5 * std::f64::consts::PI).sin(),
133    ];
134    let mut start = 0;
135    let rows: Vec<[f64; 2]> = ends
136        .iter()
137        .map(|&end| {
138            let row = [
139                median(longitude[start..=end].to_vec()),
140                median(latitude[start..=end].to_vec()),
141            ];
142            start = end + 1;
143            row
144        })
145        .collect();
146    median(
147        rows.windows(2)
148            .map(|pair| {
149                let longitude = (pair[0][0] - pair[1][0]) * scale;
150                let latitude = pair[0][1] - pair[1][1];
151                (longitude * normal[0] + latitude * normal[1]).abs()
152            })
153            .collect(),
154    )
155}

157/// `getMapPointings`: longitudes split by the periodic boundary (their
158/// sample standard deviation above 2π/3) are made continuous, across ±π
159/// when their mean is within π/2 of 0, or across 0 when it is within π/2
160/// of π.
161fn unwrap_longitudes(longitudes: &mut [f64]) {
162    if longitudes.len() < 2 {
163        return;
164    }
165    let count = longitudes.len() as f64;
166    let mean = longitudes.iter().sum::<f64>() / count;
167    let variance = longitudes.iter().map(|l| (l - mean).powi(2)).sum::<f64>() / (count - 1.0);
168    if variance.sqrt() <= 2.0 / 3.0 * std::f64::consts::PI {
169        return;
170    }
171    let pi = std::f64::consts::PI;
172    if mean.abs() < 0.5 * pi {
173        longitudes
174            .iter_mut()
175            .filter(|l| **l < 0.0)
176            .for_each(|l| *l += 2.0 * pi);
177    } else if (mean - pi).abs() < 0.5 * pi {
178        longitudes
179            .iter_mut()
180            .filter(|l| **l < pi)
181            .for_each(|l| *l += 2.0 * pi);
182    }
183}

185/// casacore's `median`: the middle value, and for an even count the mean
186/// of the middle two when there are at most 100, else the lower one. An
187/// empty list has median 0.
188pub(crate) fn median(mut values: Vec<f64>) -> f64 {
189    if values.is_empty() {
190        return 0.0;
191    }
192    let count = values.len();
193    let middle = (count - 1) / 2;
194    let (_, &mut lower, above) = values.select_nth_unstable_by(middle, f64::total_cmp);
195    if count.is_multiple_of(2) && count <= 100 {
196        let upper = above.iter().copied().fold(f64::INFINITY, f64::min);
197        0.5 * (lower + upper)
198    } else {
199        lower
200    }
201}

203/// The sampling `tsdimaging` gives the theoretical beam (`set_beam_size`),
204/// from a raster's intervals in arcsec: both when they differ by more than
205/// 1%, else one; the along-scan one alone when the across-scan one is
206/// invalid (below 1 mas, or not finite). `None`, and CASA sets no beam,
207/// when the along-scan interval is invalid (below 2.2 mas, or not finite),
208/// whatever the other: CASA logs that it uses the across-scan interval, but
209/// marks the sampling invalid.
210#[must_use]
211pub fn beam_sampling(interval_arcsec: [f64; 2]) -> Option<Vec<f64>> {
212    let [along, across] = interval_arcsec;
213    let along_valid = along.is_finite() && along.abs() >= 2.2e-3;
214    let across_valid = across.is_finite() && across.abs() >= 1.0e-3;
215    match (along_valid, across_valid) {
216        (false, _) => None,
217        (true, false) => Some(vec![along]),
218        (true, true) if (along - across).abs() <= 0.01 * along.abs() => Some(vec![along]),
219        (true, true) => Some(vec![along, across]),
220    }
221}
