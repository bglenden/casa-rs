2//! Measure UDF functions for TaQL (`meas.*`).
3//!
4//! Provides measure-aware conversion functions that can be called from TaQL
5//! expressions, mirroring the C++ `meas.*` UDFs registered by
6//! `register_meas()` in `meas/MeasUDF/Register.cc`.
7//!
8//! # Supported functions
9//!
10//! | Function | Description |
11//! |----------|-------------|
12//! | `meas.epoch` | Epoch (time scale) conversion |
13//! | `meas.last` / `meas.lst` | Local apparent sidereal time (seconds-of-day) |
14//! | `meas.help` | Help/introspection text for the `meas.*` TaQL surface |
15//! | `meas.dir` / `meas.direction` | Sky direction conversion |
16//! | `meas.dircos` / `meas.directioncosine` | Direction conversion returning cosines |
17//! | `meas.pos` / `meas.position` | Position conversion |
18//! | `meas.itrfxyz` / `meas.itrfll` / `meas.itrfh` / `meas.itrfllh` | ITRF position extractors |
19//! | `meas.wgs` / `meas.wgsxyz` | WGS84 raw-value extractor |
20//! | `meas.wgsll` / `meas.wgsh` / `meas.wgsllh` | WGS84 position extractors |
21//! | `meas.freq` / `meas.frequency` | Spectral frequency conversion |
22//! | `meas.rest` / `meas.restfreq` / `meas.restfrequency` | Rest-frequency helpers |
23//! | `meas.shift` / `meas.shiftfreq` / `meas.shiftfrequency` | Doppler shift helpers |
24//! | `meas.doppler` / `meas.redshift` | Doppler convention conversion |
25//! | `meas.radvel` / `meas.radialvelocity` | Radial velocity conversion |
26//! | `meas.hadec` / `meas.azel` / `meas.app` | Direction shortcut conversions |
27//! | `meas.riset` / `meas.riseset` | Rise/set UTC datetimes for a source |
28//! | `meas.em` / `meas.earthmagnetic` / `meas.emxyz` | Earth-magnetic vector conversion |
29//! | `meas.emang` / `meas.emangles` | Earth-magnetic conversion as angles |
30//! | `meas.emlen` / `meas.emlength` | Earth-magnetic conversion as field strength |
31//! | `meas.igrf` / `meas.igrfxyz` | IGRF model field vector |
32//! | `meas.igrfang` / `meas.igrfangles` | IGRF model field as angles |
33//! | `meas.igrflen` / `meas.igrflength` | IGRF model field strength |
34//! | `meas.igrflos` / `meas.igrflong` | IGRF line-of-sight and point-longitude helpers |
35//! | `meas.j2000` | Shortcut: direction → J2000 |
36//! | `meas.galactic` | Shortcut: direction → GALACTIC |
37//! | `meas.b1950` | Shortcut: direction → B1950 |
38//! | `meas.ecl` / `meas.ecliptic` | Shortcut: direction → ECLIPTIC |
39//! | `meas.gal` | Shortcut: direction → GALACTIC |
40//! | `meas.sgal` / `meas.supergal` / `meas.supergalactic` | Shortcut: direction → SUPERGAL |
41//!
42//! # C++ reference
43//!
44//! `meas/MeasUDF/Register.cc`, `EpochUDF.cc`, `DirectionUDF.cc`, etc.
45

143pub(crate) fn call_meas_function_with_provider(
144    name: &str,
145    args: &[ExprValue],
146    measures: MeasuresRef<'_>,
147) -> Result<ExprValue, TaqlError> {
148    let suffix = &name[5..]; // strip "meas."
149    match suffix {
150        "epoch" => meas_epoch(args, name, measures),
151        "last" | "lst" => meas_last(args, name, measures),
152        "help" => meas_help(args, name),
153        "dir" | "direction" => meas_dir(args, name, measures),
154        "dircos" | "directioncosine" => meas_dircos(args, name, measures),
155        "riset" | "riseset" => meas_riseset(args, name, measures),
156        "em" | "earthmagnetic" | "emxyz" => {
157            meas_earthmag(args, EarthMagneticOutput::Xyz, name, measures)
158        }
159        "emang" | "emangles" => meas_earthmag(args, EarthMagneticOutput::Angles, name, measures),
160        "emlen" | "emlength" => meas_earthmag(args, EarthMagneticOutput::Length, name, measures),
161        "igrf" | "igrfxyz" => meas_igrf(args, IgrfOutput::Xyz, name, measures),
162        "igrfang" | "igrfangles" => meas_igrf(args, IgrfOutput::Angles, name, measures),
163        "igrflen" | "igrflength" => meas_igrf(args, IgrfOutput::Length, name, measures),
164        "igrflos" => meas_igrf(args, IgrfOutput::Los, name, measures),
165        "igrflong" => meas_igrf(args, IgrfOutput::Long, name, measures),
166        "pos" | "position" => meas_pos(args, name, measures),
167        "itrfxyz" => meas_pos_extract(args, PositionRef::ITRF, PositionOutput::Xyz, name, measures),
168        "itrfll" | "itrflonlat" => meas_pos_extract(
169            args,
170            PositionRef::ITRF,
171            PositionOutput::LonLat,
172            name,
173            measures,
174        ),
175        "itrfh" | "itrfheight" => meas_pos_extract(
176            args,
177            PositionRef::ITRF,
178            PositionOutput::Height,
179            name,
180            measures,
181        ),
182        "itrfllh" => meas_pos_extract(
183            args,
184            PositionRef::ITRF,
185            PositionOutput::LonLatHeight,
186            name,
187            measures,
188        ),
189        "wgs" | "wgsxyz" => meas_pos_extract(
190            args,
191            PositionRef::WGS84,
192            PositionOutput::Xyz,
193            name,
194            measures,
195        ),
196        "wgsll" | "wgslonlat" => meas_pos_extract(
197            args,
198            PositionRef::WGS84,
199            PositionOutput::LonLat,
200            name,
201            measures,
202        ),
203        "wgsh" | "wgsheight" => meas_pos_extract(
204            args,
205            PositionRef::WGS84,
206            PositionOutput::Height,
207            name,
208            measures,
209        ),
210        "wgsllh" => meas_pos_extract(
211            args,
212            PositionRef::WGS84,
213            PositionOutput::LonLatHeight,
214            name,
215            measures,
216        ),
217        "freq" | "frequency" => meas_freq(args, name, measures),
218        "rest" | "restfreq" | "restfrequency" => meas_rest(args, name, measures),
219        "shift" | "shiftfreq" | "shiftfrequency" => meas_shift(args, name, measures),
220        "doppler" | "redshift" => meas_doppler(args, name),
221        "radvel" | "radialvelocity" => meas_radvel(args, name, measures),
222        "hadec" => meas_dir_shortcut(args, "HADEC", name, measures),
223        "azel" => meas_dir_shortcut(args, "AZEL", name, measures),
224        "app" | "apparent" => meas_dir_shortcut(args, "APP", name, measures),
225        "j2000" => meas_dir_shortcut(args, "J2000", name, measures),
226        "galactic" => meas_dir_shortcut(args, "GALACTIC", name, measures),
227        "b1950" => meas_dir_shortcut(args, "B1950", name, measures),
228        "ecl" | "ecliptic" => meas_dir_shortcut(args, "ECLIPTIC", name, measures),
229        "gal" => meas_dir_shortcut(args, "GALACTIC", name, measures),
230        "sgal" | "supergal" | "supergalactic" => {
231            meas_dir_shortcut(args, "SUPERGAL", name, measures)
232        }
233        "itrfd" | "itrfdir" | "itrfdirection" => meas_dir_shortcut(args, "ITRF", name, measures),
234        _ => Err(TaqlError::UnknownFunction {
235            name: name.to_string(),
236        }),
237    }
238}

264fn meas_help(args: &[ExprValue], fn_name: &str) -> Result<ExprValue, TaqlError> {
265    check_arity_range(fn_name, args, 0, 1)?;
266    if any_null(args) {
267        return Ok(ExprValue::Null);
268    }
269
270    let topic = match args {
271        [] => None,
272        [ExprValue::String(topic)] => Some(topic.to_ascii_lowercase()),
273        [other] => {
274            return Err(TaqlError::TypeError {
275                message: format!(
276                    "{fn_name}: expected optional string subtype, got {}",
277                    other.type_name()
278                ),
279            });
280        }
281        _ => unreachable!("arity checked above"),
282    };
283
284    Ok(ExprValue::String(render_meas_help(topic.as_deref())))
285}

287fn render_meas_help(topic: Option<&str>) -> String {
288    match topic {
289        None | Some("") => with_help_link(
290            [
291                help_position(false),
292                help_epoch(false),
293                help_direction(false),
294                help_earth_magnetic(false),
295                help_frequency(false),
296                help_radial_velocity(false),
297                help_doppler(false),
298            ]
299            .join("\n\n"),
300        ),
301        Some("position") | Some("pos") => with_help_link(help_position(true)),
302        Some("epoch") => with_help_link(help_epoch(true)),
303        Some("direction") | Some("dir") => with_help_link(help_direction(true)),
304        Some("earthmagnetic") | Some("em") => with_help_link(help_earth_magnetic(true)),
305        Some("frequency") | Some("freq") => with_help_link(help_frequency(true)),
306        Some("radialvelocity") | Some("radvel") | Some("rv") => {
307            with_help_link(help_radial_velocity(true))
308        }
309        Some("doppler") => with_help_link(help_doppler(true)),
310        Some(other) => format!(
311            "{other} is an unknown meas subtype; use pos(ition), epoch, dir(ection), \
312earthmagnetic (em), freq(uency) or radialvelocity (radvel)\n"
313        ),
314    }
315}

317fn with_help_link(mut section: String) -> String {
318    section.push('\n');
319    section.push_str(MEAS_HELP_URL);
320    section.push('\n');
321    section
322}

519/// Parse a reference type string from an `ExprValue::String`.
520fn parse_ref<R: FromStr>(val: &ExprValue, fn_name: &str) -> Result<R, TaqlError> {
521    let s = val.to_string_val()?;
522    R::from_str(&s).map_err(|_| TaqlError::TypeError {
523        message: format!("{fn_name}: unknown reference type \"{s}\""),
524    })
525}

535fn check_arity_range(
536    name: &str,
537    args: &[ExprValue],
538    min: usize,

543            name: name.to_string(),

557/// Build a `MeasFrame` from optional epoch/position/direction float args.
558///
559/// `extra` is the slice of optional arguments after the required ones.
560/// The interpretation depends on the calling function:
561/// - For direction UDFs: `[epoch, position]`
562/// - For numeric helpers: `[epoch, px, py, pz]`
563fn meas_frame(measures: MeasuresRef<'_>) -> MeasFrame {
564    measures.cloned().map_or_else(MeasFrame::new, |provider| {
565        MeasFrame::new().with_measures(provider)
566    })
567}
568
569fn build_frame_with_epoch_pos(
570    extra: &[ExprValue],
571    fn_name: &str,
572    measures: MeasuresRef<'_>,
573) -> Result<MeasFrame, TaqlError> {
574    let mut frame = meas_frame(measures);
575    if !extra.is_empty() {
576        let epoch_mjd = expr_to_mjd_days(&extra[0], fn_name)?;
577        frame = frame.with_epoch(MEpoch::from_mjd(epoch_mjd, EpochRef::UTC));
578    }
579    if extra.len() > 1 {
580        frame = frame.with_position(parse_position_input(&extra[1..], fn_name, measures)?);
581    }
582    Ok(frame)
583}
584
585/// Build a full `MeasFrame` from optional direction + epoch + position args.
586///
587/// For frequency/radvel: `[dir_lon, dir_lat, epoch, px, py, pz]`
588fn build_frame_with_dir_epoch_pos(
589    extra: &[ExprValue],
590    measures: MeasuresRef<'_>,
591) -> Result<MeasFrame, TaqlError> {
592    let mut frame = meas_frame(measures);
593    if extra.len() >= 2 {
594        let lon = expr_to_angle_rad(&extra[0], "meas frame direction")?;
595        let lat = expr_to_angle_rad(&extra[1], "meas frame direction")?;
596        frame = frame.with_direction(MDirection::from_angles(lon, lat, DirectionRef::J2000));
597    }
598    if extra.len() >= 3 {
599        let epoch_mjd = expr_to_mjd_days(&extra[2], "meas frame epoch")?;
600        frame = frame.with_epoch(MEpoch::from_mjd(epoch_mjd, EpochRef::UTC));
601    }
602    if extra.len() > 3 {
603        frame = frame.with_position(parse_position_input(&extra[3..], "meas frame", measures)?);
604    }
605    Ok(frame)
606}

621fn expr_to_unit_value(
622    value: &ExprValue,
623    default_unit: &str,
624    target_unit: &str,
625    what: &str,
626    fn_name: &str,
627) -> Result<f64, TaqlError> {
628    let quantity = expr_as_quantity(value, default_unit, fn_name)?;
629    let target = Unit::new(target_unit).map_err(|e| TaqlError::TypeError {
630        message: format!("{fn_name}: {e}"),
631    })?;
632    quantity
633        .convert(&target)
634        .map(|converted| converted.value())
635        .map_err(|_| TaqlError::TypeError {
636            message: format!(
637                "{fn_name}: expected {what} with units conformant to {target_unit}, got {}",
638                quantity.unit().name()
639            ),
640        })
641}

667fn parse_position_input(
668    args: &[ExprValue],
669    fn_name: &str,
670    measures: MeasuresRef<'_>,
671) -> Result<MPosition, TaqlError> {
672    match args {
673        [name] => {
674            let observatory = name.to_string_val()?;
675            let provider = measures.ok_or_else(|| TaqlError::TypeError {
676                message: format!("{fn_name}: observatory name {observatory:?} requires an explicit measures provider"),
677            })?;
678            MPosition::from_observatory_name(&observatory, provider.as_ref())
679                .map_err(|error| measure_err(fn_name, error))?
680                .ok_or_else(|| TaqlError::TypeError {
681                    message: format!("{fn_name}: unknown observatory {observatory:?}"),
682                })
683        }
684        [x, y, z] => Ok(MPosition::new_itrf(
685            expr_to_length_m(x, fn_name)?,
686            expr_to_length_m(y, fn_name)?,
687            expr_to_length_m(z, fn_name)?,
688        )),
689        [x, y, z, src_ref] => {
690            let src: PositionRef = parse_ref(src_ref, fn_name)?;
691            Ok(match src {
692                PositionRef::ITRF => MPosition::new_itrf(
693                    expr_to_length_m(x, fn_name)?,
694                    expr_to_length_m(y, fn_name)?,
695                    expr_to_length_m(z, fn_name)?,
696                ),
697                PositionRef::WGS84 => MPosition::new_wgs84(
698                    expr_to_angle_rad(x, fn_name)?,
699                    expr_to_angle_rad(y, fn_name)?,
700                    expr_to_length_m(z, fn_name)?,
701                ),
702            })
703        }
704        _ => Err(TaqlError::ArgumentCount {
705            name: fn_name.to_string(),
706            expected: "1 or 3..4".to_string(),
707            got: args.len(),
708        }),
709    }
710}

712fn parse_doppler_input(
713    value_arg: &ExprValue,
714    ref_arg: Option<&ExprValue>,
715    fn_name: &str,
716) -> Result<MDoppler, TaqlError> {
717    let value = match value_arg {
718        ExprValue::Quantity(q) => {
719            let target = Unit::new("m/s").expect("valid unit");
720            q.convert(&target)
721                .map(|converted| converted.value())
722                .map_err(|_| TaqlError::TypeError {
723                    message: format!(
724                        "{fn_name}: quantity doppler input must be a radial velocity, got {}",
725                        q.unit().name()
726                    ),
727                })?
728        }
729        _ => value_arg.to_float()?,
730    };
731    let refer = if let Some(arg) = ref_arg {
732        parse_ref::<DopplerRef>(arg, fn_name)?
733    } else {
734        DopplerRef::RADIO
735    };
736    Ok(MDoppler::new(value, refer))
737}

750enum ParsedDirectionInput {
751    Named(String),
752    Angles {
753        lon: f64,
754        lat: f64,
755        refer: DirectionRef,
756    },
757    Cosines {
758        xyz: [f64; 3],
759        refer: DirectionRef,
760    },
761}
762
763impl ParsedDirectionInput {
764    fn materialize(&self, frame: &MeasFrame) -> Result<MDirection, TaqlError> {
765        match self {
766            Self::Named(name) => {
767                MDirection::from_source_name(name, frame).map_err(|e| TaqlError::TypeError {
768                    message: e.to_string(),
769                })
770            }
771            Self::Angles { lon, lat, refer } => Ok(MDirection::from_angles(*lon, *lat, *refer)),
772            Self::Cosines { xyz, refer } => Ok(MDirection::from_cosines(*xyz, *refer)),
773        }
774    }
775}
776
777fn parse_direction_input(
778    args: &[ExprValue],
779    fn_name: &str,
780) -> Result<(ParsedDirectionInput, usize), TaqlError> {
781    let Some(first) = args.first() else {
782        return Err(TaqlError::ArgumentCount {
783            name: fn_name.to_string(),
784            expected: "direction argument".to_string(),
785            got: 0,
786        });
787    };
788
789    if let ExprValue::String(name) = first {
790        return Ok((ParsedDirectionInput::Named(name.clone()), 1));
791    }
792
793    if let ExprValue::Array(arr) = first {
794        let vals = extract_float_array(arr, fn_name)?;
795        let refer = if args.len() > 1 && matches!(args[1], ExprValue::String(_)) {
796            parse_ref(&args[1], fn_name)?
797        } else {
798            DirectionRef::J2000
799        };
800        return match vals.as_slice() {
801            [lon, lat] => Ok((
802                ParsedDirectionInput::Angles {
803                    lon: *lon,
804                    lat: *lat,
805                    refer,
806                },
807                if args.len() > 1 && matches!(args[1], ExprValue::String(_)) {
808                    2
809                } else {
810                    1
811                },
812            )),
813            [x, y, z] => Ok((
814                ParsedDirectionInput::Cosines {
815                    xyz: [*x, *y, *z],
816                    refer,
817                },
818                if args.len() > 1 && matches!(args[1], ExprValue::String(_)) {
819                    2
820                } else {
821                    1
822                },
823            )),
824            _ => Err(TaqlError::TypeError {
825                message: format!("{fn_name}: direction array must contain 2 or 3 values"),
826            }),
827        };
828    }
829
830    if args.len() < 2 {
831        return Err(TaqlError::ArgumentCount {
832            name: fn_name.to_string(),
833            expected: "2..3 direction scalars".to_string(),
834            got: args.len(),
835        });
836    }
837    let lon = expr_to_angle_rad(&args[0], fn_name)?;
838    let lat = expr_to_angle_rad(&args[1], fn_name)?;
839    let refer = if args.len() > 2 && matches!(args[2], ExprValue::String(_)) {
840        parse_ref(&args[2], fn_name)?
841    } else {
842        DirectionRef::J2000
843    };
844    Ok((
845        ParsedDirectionInput::Angles { lon, lat, refer },
846        if args.len() > 2 && matches!(args[2], ExprValue::String(_)) {
847            3
848        } else {
849            2
850        },
851    ))
852}

854fn parse_earthmag_xyz_input(
855    args: &[ExprValue],
856    fn_name: &str,
857) -> Result<(EarthMagneticParsedInput, EarthMagneticRef, usize), TaqlError> {
858    if args.is_empty() {
859        return Err(TaqlError::ArgumentCount {
860            name: fn_name.to_string(),
861            expected: "earth-magnetic vector".to_string(),
862            got: 0,
863        });
864    }
865
866    let (parsed, consumed) = match &args[0] {
867        ExprValue::Array(arr) => {
868            let vals = extract_float_array(arr, fn_name)?;
869            match vals.as_slice() {
870                [x, y, z] => (EarthMagneticParsedInput::Xyz([*x, *y, *z]), 1),
871                _ => {
872                    return Err(TaqlError::TypeError {
873                        message: format!(
874                            "{fn_name}: earth-magnetic vector array must contain 3 values"
875                        ),
876                    });
877                }
878            }
879        }
880        _ => {
881            if args.len() < 3 {
882                return Err(TaqlError::ArgumentCount {
883                    name: fn_name.to_string(),
884                    expected: "3 vector scalars".to_string(),
885                    got: args.len(),
886                });
887            }
888            parse_earthmag_scalars(args, fn_name)?
889        }
890    };
891
892    let src = if args.len() > consumed && matches!(args[consumed], ExprValue::String(_)) {
893        parse_ref(&args[consumed], fn_name)?
894    } else {
895        EarthMagneticRef::ITRF
896    };
897    Ok((
898        parsed,
899        src,
900        consumed
901            + if args.len() > consumed && matches!(args[consumed], ExprValue::String(_)) {
902                1
903            } else {
904                0
905            },
906    ))
907}

909enum EarthMagneticParsedInput {
910    Xyz([f64; 3]),
911    Angles {
912        lon_rad: f64,
913        lat_rad: f64,
914        length_nt: f64,
915    },
916}
917
918fn parse_earthmag_scalars(
919    args: &[ExprValue],
920    fn_name: &str,
921) -> Result<(EarthMagneticParsedInput, usize), TaqlError> {
922    let mode = match &args[0] {
923        ExprValue::Quantity(q) => {
924            let flux = Unit::new("T").expect("valid unit");
925            let angle = Unit::new("rad").expect("valid unit");
926            if q.unit().conformant(&flux) {
927                Some(true)
928            } else if q.unit().conformant(&angle) {
929                Some(false)
930            } else {
931                None
932            }
933        }
934        _ => None,
935    };
936
937    let parsed = match mode {
938        Some(true) => EarthMagneticParsedInput::Xyz([
939            expr_to_flux_nt(&args[0], fn_name)?,
940            expr_to_flux_nt(&args[1], fn_name)?,
941            expr_to_flux_nt(&args[2], fn_name)?,
942        ]),
943        Some(false) => EarthMagneticParsedInput::Angles {
944            lon_rad: expr_to_angle_rad(&args[0], fn_name)?,
945            lat_rad: expr_to_angle_rad(&args[1], fn_name)?,
946            length_nt: expr_to_flux_nt(&args[2], fn_name)?,
947        },
948        None => EarthMagneticParsedInput::Xyz([
949            args[0].to_float()?,
950            args[1].to_float()?,
951            args[2].to_float()?,
952        ]),
953    };
954
955    Ok((parsed, 3))
956}

972fn casacore_mvposition_length(length_m: f64) -> f64 {
973    let mut adjusted = length_m;
974    if adjusted < 0.0 && adjusted > -7.0e6 {
975        adjusted = adjusted / 1.0e7 + 743.569;
976    } else if adjusted > 743.568 && adjusted < 743.569 {
977        adjusted += 0.001;
978    }
979    if adjusted == 0.0 { 1.0e-6 } else { adjusted }
980}

982/// casacore `meas.wgs` / `meas.wgsxyz` return `MVPosition::getValue()` after
983/// conversion to `WGS84`, not the cleaner lon/lat/height triplet.
984fn casacore_wgs_xyz_values(position: &MPosition) -> [f64; 3] {
985    let converted = if position.refer() == PositionRef::WGS84 {
986        position.clone()
987    } else {
988        position
989            .convert_to(PositionRef::WGS84)
990            .expect("conversion to WGS84 should already have succeeded")
991    };
992    let [lon, lat, height] = converted.values();
993    let scale = casacore_mvposition_length(height);
994    let clat = lat.cos();
995    [
996        lon.cos() * clat * scale,
997        lon.sin() * clat * scale,
998        lat.sin() * scale,
999    ]
1000}
