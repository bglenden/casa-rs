17fn is_square_law_detector(name: &str) -> bool {
18    name.match_indices("BB_").any(|(start, _)| {
19        let suffix = &name.as_bytes()[start + 3..];
20        matches!(suffix, [b'0'..=b'9', rest @ ..] if rest.starts_with(b"#SQLD"))
21    })
22}

24fn has_column(windows: &MsSpectralWindow<'_>, column: &str) -> bool {
25    windows
26        .table()
27        .schema()
28        .is_some_and(|schema| schema.contains_column(column))
29}

31fn polarization_ids(ms: &MeasurementSet, count: usize) -> MsResult<Vec<BTreeSet<i32>>> {
32    let descriptions = ms.data_description()?;
33    let mut polarizations = vec![BTreeSet::new(); count];
34    for row in 0..descriptions.row_count() {
35        let spw = descriptions.spectral_window_id(row)?;
36        let ids = polarizations
37            .get_mut(spw as usize)
38            .ok_or_else(|| MsError::InvalidInput(format!("Unknown spectral window ID {spw}")))?;
39        ids.insert(descriptions.polarization_id(row)?);
40    }
41    Ok(polarizations)
42}

44fn selected_windows(
45    ms: &MeasurementSet,
46    chavg: bool,
47    fdm: bool,
48    sqld: bool,
49    tdm: bool,
50    wvr: bool,
51    complement: bool,
52) -> MsResult<Vec<i64>> {
53    let windows = ms.spectral_window()?;
54    let has_baseband = has_column(&windows, "BBC_NO");
55    let wvr_ids = if wvr {
56        water_vapor_windows(&windows)?
57    } else {
58        Vec::new()
59    };
60    let mut pol_ids = None;
61    let mut chosen = Vec::new();
62    for row in 0..windows.row_count() {
63        let mut matched = wvr_ids.binary_search(&row).is_ok();
64        if !matched && (chavg || fdm || tdm || (sqld && has_baseband)) {
65            let name = windows.name(row)?;
66            if sqld && has_baseband && is_square_law_detector(&name) {
67                matched = true;
68            } else if !name.contains("WVR") {
69                let channels = windows
70                    .table()
71                    .cell_accessor(row, "CHAN_WIDTH")?
72                    .array()?
73                    .len();
74                if channels == 1 && !name.contains("FULL_RES") {
75                    matched = chavg;
76                } else if fdm || tdm {
77                    // ALMA classifications include both normal TDM channel counts and
78                    // the counts after filtering edge channels.
79                    let mut is_fdm = windows.total_bandwidth(row)? < 2.0e9
80                        || channels >= 15
81                            && ![256, 128, 64, 32, 16, 248, 124, 62, 31].contains(&channels);
82                    if !is_fdm {
83                        if pol_ids.is_none() {
84                            pol_ids = Some(polarization_ids(ms, windows.row_count())?);
85                        }
86                        let ids = &pol_ids.as_ref().expect("polarizations were initialized")[row];
87                        if ids.len() == 1 {
88                            let id = *ids.first().expect("one polarization");
89                            let ncorrelations = ms.polarization()?.num_corr(id as usize)?;
90                            is_fdm = channels * ncorrelations as usize > 256;
91                        }
92                    }
93                    matched = if is_fdm { fdm } else { tdm };
94                }
95            }
96        }
97        if matched != complement {
98            chosen.push(row as i64);
99        }
100    }
101    Ok(chosen)
102}

104fn baseband_number(windows: &MsSpectralWindow<'_>, row: usize) -> MsResult<u32> {
105    if let ScalarValue::Int32(value) = windows.table().cell_accessor(row, "BBC_NO")?.scalar()? {
106        return Ok(*value as u32);
107    }
108    Err(MsError::InvalidInput(
109        "BBC_NO is not an Int32 column".into(),
110    ))
111}

113fn check_baseband(windows: &MsSpectralWindow<'_>) -> MsResult<()> {
114    if !has_column(windows, "BBC_NO") {
115        return Err(MsError::InvalidInput(
116            "This MS's SPECTRAL_WINDOW table does not have a BBC_NO column".into(),
117        ));
118    }
119    Ok(())
120}

122fn text_cell<'a>(table: &'a Table, row: usize, column: &str) -> MsResult<&'a str> {
123    match table.cell_accessor(row, column)?.scalar()? {
124        ScalarValue::String(value) => Ok(value),
125        _ => Err(MsError::InvalidInput(format!(
126            "{column} is not a string column"
127        ))),
128    }
129}

131fn integer_suffix<'a>(name: &'a str, prefix: &str, width: Option<usize>) -> Option<&'a str> {
132    name.match_indices(prefix).find_map(|(start, _)| {
133        let suffix = &name[start + prefix.len()..];
134        let digits = suffix.bytes().take_while(u8::is_ascii_digit).count();
135        let needed = width.unwrap_or(digits);
136        (needed > 0 && digits >= needed).then(|| &suffix[..needed])
137    })
138}

140fn receiver_bands(path: &Path, count: usize) -> MsResult<Vec<i64>> {
141    let path = path.join("ASDM_RECEIVER");
142    let mut bands = vec![-1; count];
143    if path.is_dir() {
144        let table = Table::open(TableOptions::new(&path))?;
145        if table.row_count() <= count {
146            // CASA uses the first receiver-table entry for repeated windows.
147            for row in (0..table.row_count()).rev() {
148                let band = text_cell(&table, row, "frequencyBand")?;
149                if let Some(band) = integer_suffix(band, "RB_", Some(2)) {
150                    let name = text_cell(&table, row, "spectralWindowId")?;
151                    let target = integer_suffix(name, "SpectralWindow_", None)
152                        .and_then(|window| window.parse::<usize>().ok())
153                        .and_then(|window| bands.get_mut(window))
154                        .ok_or_else(|| {
155                            MsError::InvalidInput(format!(
156                                "Unable to find spectral window ID for row {row} in {}",
157                                path.display()
158                            ))
159                        })?;
160                    *target = band.parse().expect("two decimal digits");
161                }
162            }
163        }
164    }
165    Ok(bands)
166}

168fn selected_name_numbers(
169    ms: &MeasurementSet,
170    selection: Indices,
171    prefix: &str,
172    use_receiver_table: bool,
173) -> MsResult<Vec<i64>> {
174    let windows = ms.spectral_window()?;
175    let indices = match selection {
176        Indices::EmptyString | Indices::BooleanVector => Indices::Empty,
177        selection => selection,
178    };
179    let mut bands = None;
180    indices
181        .rows(windows.row_count(), "Spectral window ID")?
182        .into_iter()
183        .map(|row| {
184            let name = windows.name(row)?;
185            if let Some(digits) = integer_suffix(&name, prefix, Some(2)) {
186                Ok(digits.parse().expect("two decimal digits"))
187            } else if use_receiver_table {
188                if bands.is_none() {
189                    let root = ms.path().expect("an attached MS has a path");
190                    bands = Some(receiver_bands(root, windows.row_count())?);
191                }
192                Ok(bands.as_ref().expect("receiver bands were initialized")[row])
193            } else {
194                Ok(-1)
195            }
196        })
197        .collect()
198}

242    #[pyo3(signature = (spw), text_signature = "($self, spw)")]
243    fn baseband(&self, py: Python<'_>, spw: i64) -> PyResult<u32> {
244        self.with_ms(py, |ms| {
245            let windows = ms.spectral_window()?;
246            let row = spw_row(&windows, spw)?;
247            check_baseband(&windows)?;
248            baseband_number(&windows, row)
249        })
250    }

252    #[pyo3(signature = (baseband = -1, sqldmode = "include"), text_signature = "($self, baseband=-1, sqldmode='include')")]
253    fn spwsforbaseband(
254        &self,
255        py: Python<'_>,
256        baseband: i64,
257        sqldmode: &str,
258    ) -> PyResult<Py<PyAny>> {
259        let selected = self.with_ms(py, |ms| {
260            let mode = sqldmode.as_bytes().first().map(u8::to_ascii_lowercase);
261            if !matches!(mode, Some(b'i' | b'e' | b'o')) {
262                return Err(MsError::InvalidInput(format!(
263                    "Unsupported sqldmode {sqldmode}. Must be either i(nclude), e(xclude), or o(nly)."
264                )));
265            }
266            let windows = ms.spectral_window()?;
267            check_baseband(&windows)?;
268            let mut groups = BTreeMap::<u32, Vec<i64>>::new();
269            for row in 0..windows.row_count() {
270                let group = baseband_number(&windows, row)?;
271                if baseband >= 0 && group as i64 != baseband {
272                    continue;
273                }
274                let entries = groups.entry(group).or_default();
275                if mode == Some(b'i')
276                    || is_square_law_detector(&windows.name(row)?) == (mode == Some(b'o'))
277                {
278                    entries.push(row as i64);
279                }
280            }
281            if baseband >= 0 {
282                Ok(int_array(groups.pop_first().map(|(_, values)| values).unwrap_or_default()))
283            } else {
284                Ok(CasaValue::record(groups.into_iter().map(|(id, values)| (id.to_string(), int_array(values)))))
285            }
286        })?;
287        selected.into_python(py)
288    }

290    #[pyo3(signature = (spw = Indices::Scalar(-1)), text_signature = "($self, spw=-1)")]
291    fn corrbit(&self, py: Python<'_>, spw: Indices) -> PyResult<Py<PyAny>> {
292        let value = self.with_ms(py, |ms| {
293            let windows = ms.spectral_window()?;
294            let column = "SDM_CORR_BIT";
295            let has_bits = has_column(&windows, column);
296            let bits = |row| {
297                if has_bits {
298                    text_cell(windows.table(), row, column).map(str::to_owned)
299                } else {
300                    Ok("UNKNOWN".to_owned())
301                }
302            };
303            let selected = match spw {
304                Indices::Scalar(id) if id >= 0 => {
305                    let row = spw_row(&windows, id)?;
306                    return bits(row).map(CasaValue::Text);
307                }
308                Indices::Scalar(_) => Indices::Empty,
309                Indices::Vector(ids) if ids.is_empty() => {
310                    return Ok(CasaValue::TextArray {
311                        shape: vec![0],
312                        values: Vec::new(),
313                    });
314                }
315                Indices::Vector(ids) => Indices::Vector(ids),
316                _ => {
317                    return Err(MsError::InvalidInput(
318                        "Parameter spw must either be an integer or a list of integers".into(),
319                    ));
320                }
321            };
322            let values = selected
323                .rows(windows.row_count(), "Spectral window ID")?
324                .into_iter()
325                .map(bits)
326                .collect::<MsResult<Vec<_>>>()?;
327            Ok(CasaValue::TextArray {
328                shape: vec![values.len()],
329                values,
330            })
331        })?;
332        value.into_python(py)
333    }
