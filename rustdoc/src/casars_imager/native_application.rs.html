18pub(super) fn execute(config: &CliConfig) -> Result<RunSummary, String> {
19    let started = Instant::now();
20    let result =
21        execute_continuum(application_request(config)).map_err(|error| error.to_string())?;
22    let native = result.outcome.output;
23    let clean_stop_reason = result.minor_stop_reason.map(|reason| match reason {
24        ContinuumStopReason::ThresholdReached => CleanStopReason::GlobalThresholdReached,
25        ContinuumStopReason::IterationBound => CleanStopReason::IterationLimitReached,
26        ContinuumStopReason::StalenessBound => CleanStopReason::MajorCycleLimitReached,
27    });
28    Ok(RunSummary {
29        warnings: Vec::new(),
30        gridded_samples: usize::try_from(native.scientific.normal_state().sample_count())
31            .map_err(|_| "native selected-sample count exceeds usize".to_string())?,
32        major_cycles: usize::from(native.final_major_receipt.is_some()) + 1,
33        minor_iterations: result.minor_iterations,
34        clean_stop_reason,
35        elapsed: started.elapsed(),
36        output_products: result.product_names,
37    })
38}

40fn application_request(config: &CliConfig) -> ContinuumImagingRequest {
41    ContinuumImagingRequest {
42        measurement_set: config.ms.clone(),
43        image_name: config.imagename.clone(),
44        image_size: config.imsize,
45        cell_arcsec: config.cell_arcsec,
46        field_ids: config.field_ids.clone(),
47        uv_range: config.uvrange.clone(),
48        intent: config.intent.clone(),
49        data_description: config.ddid,
50        spectral_window: config
51            .spw_selector
52            .clone()
53            .or_else(|| config.spw.map(|spw| spw.to_string())),
54        channel_start: config.channel_start,
55        channel_count: config.channel_count,
56        data_column: config.datacolumn.clone(),
57        algorithm: if config.dirty_only || config.niter == 0 {
58            ContinuumAlgorithm::Dirty
59        } else {
60            match config.deconvolver {
61                Deconvolver::Hogbom => ContinuumAlgorithm::Hogbom,
62                Deconvolver::Clark => ContinuumAlgorithm::Clark,
63                Deconvolver::Multiscale => ContinuumAlgorithm::Multiscale {
64                    scales_px: config
65                        .multiscale_scales
66                        .iter()
67                        .copied()
68                        .map(f64::from)
69                        .collect(),
70                },
71                Deconvolver::Mtmfs => ContinuumAlgorithm::Mtmfs {
72                    terms: config.nterms,
73                },
74            }
75        },
76        weighting: match config.weighting {
77            WeightingMode::Natural => ContinuumWeighting::Natural,
78            WeightingMode::Uniform => ContinuumWeighting::Uniform,
79            WeightingMode::Briggs { robust } => ContinuumWeighting::Briggs(f64::from(robust)),
80            WeightingMode::BriggsBwTaper { robust } => {
81                ContinuumWeighting::BriggsBandwidthTaper(f64::from(robust))
82            }
83        },
84        iterations: config.niter,
85        gain: f64::from(config.gain),
86        threshold_jy: f64::from(config.threshold_jy),
87        psf_cutoff: config.psf_cutoff,
88        beam_policy: match config.restoring_beam_mode {
89            RestoringBeamMode::PerPlane => ContinuumBeamPolicy::PerPlane,
90            RestoringBeamMode::Common => ContinuumBeamPolicy::Common,
91        },
92        task_requirements: task_requirements(config),
93    }
94}

96fn task_requirements(config: &CliConfig) -> Vec<TaskRequirement> {
97    let mut requirements = vec![match config.spectral_mode {
98        SpectralMode::Mfs => TaskRequirement::SerialCpu,
99        SpectralMode::Cube => TaskRequirement::SpectralCube,
100        SpectralMode::Cubedata => TaskRequirement::SpectralCubedata,
101    }];
102    if config.aw_project.is_some() {
103        requirements.push(TaskRequirement::AwProjection);
104    } else if config.w_term_mode == WTermMode::WProject {
105        requirements.push(TaskRequirement::WProjection);
106    } else if config.use_pointing && !config.force_standard_gridder {
107        requirements.push(TaskRequirement::MosaicGridder);
108    }
109    if config.outlier_file.is_some() {
110        requirements.push(TaskRequirement::FacetsOutliers);
111    }
112    if config.use_mask == CleanMaskMode::AutoMultiThreshold {
113        requirements.push(TaskRequirement::Automasking);
114    }
115    if config.use_mask == CleanMaskMode::AutoMultiThreshold
116        || !config.mask_boxes.is_empty()
117        || config.mask_image.is_some()
118    {
119        requirements.push(TaskRequirement::MaskProduct);
120    }
121    if config.start_model.is_some() {
122        requirements.push(TaskRequirement::StartModel);
123    }
124    if config.save_model != SaveModelMode::None {
125        requirements.push(TaskRequirement::ModelColumnWrite);
126    }
127    requirements.extend(backend_requirements(config));
128    if unsupported_native_controls(config) {
129        requirements.push(TaskRequirement::UnsupportedControls);
130    }
131    requirements
132}

134fn backend_requirements(config: &CliConfig) -> Vec<TaskRequirement> {
135    let mut requirements = Vec::new();
136    match config.standard_mfs_acceleration {
137        StandardMfsAccelerationPolicy::Auto => {
138            requirements.push(TaskRequirement::ExecutionAuto);
139        }
140        StandardMfsAccelerationPolicy::Cpu => {}
141        StandardMfsAccelerationPolicy::MultiCpu => {
142            requirements.push(TaskRequirement::FixedTileCpu);
143        }
144        StandardMfsAccelerationPolicy::Metal => {
145            requirements.push(TaskRequirement::MetalRowRunGroupedGridder);
146        }
147    }
148    if let Some(backend) = config.standard_mfs_backend.as_deref() {
149        requirements.push(match backend {
150            "cpu" | "serial" | "serial-cpu" => TaskRequirement::SerialCpu,
151            "fixed-tile" | "fixed-tile-cpu" => TaskRequirement::FixedTileCpu,
152            "metal" | "metal-gridder" => TaskRequirement::MetalGridder,
153            "metal-row-run" | "metal-row-run-gridder" => TaskRequirement::MetalRowRunGridder,
154            "metal-row-run-grouped" | "metal-row-run-grouped-gridder" => {
155                TaskRequirement::MetalRowRunGroupedGridder
156            }
157            _ => TaskRequirement::UnsupportedControls,
158        });
159    }
160    match config.imaging_fft_backend {
161        ImagingFftBackendPolicy::RustFft => {}
162        ImagingFftBackendPolicy::Auto => requirements.push(TaskRequirement::FftAuto),
163        ImagingFftBackendPolicy::Accelerate => requirements.push(TaskRequirement::Accelerate),
164        ImagingFftBackendPolicy::MetalMpsGraph => requirements.push(TaskRequirement::MetalMpsGraph),
165        ImagingFftBackendPolicy::Fftw => requirements.push(TaskRequirement::Fftw),
166    }
167    requirements
168}

170fn unsupported_native_controls(config: &CliConfig) -> bool {
171    config.phasecenter_field.is_some()
172        || config.phasecenter.is_some()
173        || config
174            .correlation
175            .as_deref()
176            .is_some_and(|plane| !plane.eq_ignore_ascii_case("I"))
177        || config.uv_taper.is_some()
178        || config.nmajor.is_some()
179        || config.nsigma != 0.0
180        || config.minor_cycle_length != 1000
181        || config.cyclefactor != 1.0
182        || config.min_psf_fraction != 0.05
183        || config.max_psf_fraction != 0.8
184        || config.hogbom_iteration_mode != super::HogbomIterationMode::Strict
185        || config.fullsummary
186        || config.pbcor
187        || config.write_pb
188        || config.chanchunks.is_some()
189        || config.per_channel_weight_density
190        || config.w_project_planes.is_some()
191        || config.standard_mfs_grid_threads.is_some()
192        || config.standard_mfs_tile_anchor.is_some()
193        || config.standard_mfs_residual_backend.is_some()
194        || config.standard_mfs_initial_dirty_backend.is_some()
195        || config.standard_mfs_metal_minor_cycle_chunk.is_some()
196        || config.standard_mfs_metal_grouped_input_cache.is_some()
197        || config.standard_mfs_memory_target_mb.is_some()
198        || config.standard_mfs_prepare_buffer_mb.is_some()
199        || config.imaging_memory_target_mb.is_some()
200        || config.imaging_memory_pressure_policy != ImagingMemoryPressurePolicy::Auto
201        || config.imaging_prepare_buffer_mb.is_some()
202        || config.imaging_row_block_rows.is_some()
203        || config.imaging_prepare_workers.is_some()
204        || config.imaging_read_ahead_blocks.is_some()
205        || config.imaging_fft_precision != ImagingFftPrecisionPolicy::Auto
206        || config.write_preview_pngs
207        || config.niter > config.minor_cycle_length
208        || !matches!(config.deconvolver, Deconvolver::Hogbom)
209}
