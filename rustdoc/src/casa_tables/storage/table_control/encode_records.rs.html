5pub(crate) fn write_ref_table_stream(
6    mut io: AipsIo,
7    contents: &RefTableDatContents,
8) -> Result<(), StorageError> {
9    let limit = u64::from(u32::MAX);
10    let wide = contents.nrrow >= limit
11        || contents.parent_nrrow >= limit
12        || contents.row_map.iter().any(|&row| row >= limit);
13    io.putstart("Table", if contents.nrrow >= limit { 3 } else { 2 })?;
14    if contents.nrrow >= limit {
15        io.put_u64(contents.nrrow)?;
16    } else {
17        io.put_u32(contents.nrrow as u32)?;
18    }
19    io.put_u32(if contents.big_endian { 0 } else { 1 })?;
20    io.put_string("RefTable")?;
21
22    io.putstart("RefTable", if wide { 3 } else { 2 })?;
23
24    // Parent table relative path.
25    io.put_string(&contents.parent_relative_path)?;
26
27    // Column name map: C++ serializes std::map<String, String> as a
28    // "SimpleOrderedMap" AipsIO object.
29    io.putstart("SimpleOrderedMap", 1)?;
30    io.put_string("")?; // old default value (unused)
31    io.put_u32(contents.column_name_map.len() as u32)?;
32    io.put_u32(1)?; // old increment (unused)
33    for (view_name, parent_name) in &contents.column_name_map {
34        io.put_string(view_name)?;
35        io.put_string(parent_name)?;
36    }
37    io.putend()?;
38
39    // Column names (version > 1): C++ writes as an "Array" AipsIO object (v3).
40    // Format: ndim(u32), shape[ndim](u32...), count(u32), then strings.
41    io.putstart("Array", 3)?;
42    io.put_u32(1)?; // ndim = 1 (Vector)
43    io.put_u32(contents.column_names.len() as u32)?; // shape[0]
44    io.put_u32(contents.column_names.len() as u32)?; // element count
45    for name in &contents.column_names {
46        io.put_string(name)?;
47    }
48    io.putend()?;
49
50    if wide {
51        io.put_u64(contents.parent_nrrow)?;
52    } else {
53        io.put_u32(contents.parent_nrrow as u32)?;
54    }
55
56    // Row order flag.
57    io.put_bool(contents.row_order)?;
58
59    // This RefTable's row count.
60    if wide {
61        io.put_u64(contents.nrrow)?;
62    } else {
63        io.put_u32(contents.nrrow as u32)?;
64    }
65
66    // Row number vector (chunked at 2^20, no count prefix).
67    let max_chunk: usize = 1 << 20;
68    for chunk in contents.row_map.chunks(max_chunk) {
69        if wide {
70            io.put_u64_slice(chunk, false)?;
71        } else {
72            let chunk32: Vec<u32> = chunk.iter().map(|&row| row as u32).collect();
73            io.put_u32_slice(&chunk32, false)?;
74        }
75    }
76
77    io.putend()?; // RefTable
78
79    io.putend()?; // Table
80    io.close()?;
81    Ok(())
82}

84/// Write table.dat for a ConcatTable.
85///
86/// Binary layout matches C++ `ConcatTable::writeConcatTable`:
87/// ```text
88/// putstart("Table", 2)
89///   nrrow (u32), endian (u32), "ConcatTable" (String)
90///   putstart("ConcatTable", 0)
91///     ntable (u32), table_path[0..ntable] (String each)
92///     putstart("Block", 1)                 // Block<String> subtable names
93///       count (u32), sub_table_name[0..count] (String each)
94///     putend()
95///   putend()
96/// putend()
97/// ```
98pub(crate) fn write_concat_table_dat(
99    path: &Path,
100    contents: &ConcatTableDatContents,
101) -> Result<(), StorageError> {
102    let mut io = AipsIo::open(path, AipsOpenOption::New)?;
103
104    // BaseTable uses version 3 once the row count exceeds a CASA Int.
105    let wide = contents.nrrow > i32::MAX as u64;
106    io.putstart("Table", if wide { 3 } else { 2 })?;
107    if wide {
108        io.put_u64(contents.nrrow)?;
109    } else {
110        io.put_u32(contents.nrrow as u32)?;
111    }
112    io.put_u32(if contents.big_endian { 0 } else { 1 })?;
113    io.put_string("ConcatTable")?;
114
115    // ConcatTable payload — version 0.
116    io.putstart("ConcatTable", 0)?;
117
118    // Number of constituent tables and their relative paths.
119    io.put_u32(contents.table_paths.len() as u32)?;
120    for table_path in &contents.table_paths {
121        io.put_string(table_path)?;
122    }
123
124    // Block<String> subtable names: C++ serializes as "Block" AipsIO object (v1).
125    io.putstart("Block", 1)?;
126    io.put_u32(contents.sub_table_names.len() as u32)?;
127    for name in &contents.sub_table_names {
128        io.put_string(name)?;
129    }
130    io.putend()?; // Block
131
132    io.putend()?; // ConcatTable
133
134    io.putend()?; // Table
135    io.close()?;
136    Ok(())
137}

139pub(super) fn write_table_desc(
140    io: &mut AipsIo,
141    desc: &TableDescContents,
142) -> Result<(), StorageError> {
143    io.putstart("TableDesc", 2)?;
144    io.put_string(&desc.name)?;
145    io.put_string(&desc.version)?;
146    io.put_string(&desc.comment)?;
147
148    write_table_record(io, &desc.table_keywords)?;
149    write_table_record(io, &desc.private_keywords)?;
150
151    // ColumnDescSet
152    io.put_u32(desc.columns.len() as u32)?;
153    for col in &desc.columns {
154        write_column_desc(io, col)?;
155    }
156
157    io.putend()?;
158    Ok(())
159}

161pub(super) fn write_column_desc(
162    io: &mut AipsIo,
163    col: &ColumnDescContents,
164) -> Result<(), StorageError> {
165    // ColumnDesc wrapper
166    io.put_u32(1)?; // version=1
167    io.put_string(&col.class_name)?;
168
169    // BaseColumnDesc
170    io.put_u32(1)?; // version=1
171    io.put_string(&col.col_name)?;
172    io.put_string(&col.comment)?;
173    io.put_string(&col.data_manager_type)?;
174    io.put_string(&col.data_manager_group)?;
175    io.put_i32(col.data_type as i32)?;
176    io.put_i32(col.option)?;
177    io.put_i32(col.nrdim)?;
178
179    // C++ writes shape for all non-scalar columns (if !isScalar_p)
180    if col.is_array {
181        write_iposition(io, &col.shape)?;
182    }
183
184    io.put_u32(col.max_length)?;
185    write_table_record(io, &col.keywords)?;
186
187    // Type-specific descriptor
188    io.put_u32(1)?; // version=1
189    if col.is_array {
190        io.put_bool(false)?; // hasDefault=false
191    } else {
192        write_default_value(io, col.data_type)?;
193    }
194
195    Ok(())
196}

198pub(super) fn write_default_value(
199    io: &mut AipsIo,
200    dt: CasacoreDataType,
201) -> Result<(), StorageError> {
202    match dt {
203        CasacoreDataType::TpBool => io.put_bool(false)?,
204        CasacoreDataType::TpUChar => io.put_u8(0)?,
205        CasacoreDataType::TpShort => io.put_i16(0)?,
206        CasacoreDataType::TpUShort => io.put_u16(0)?,
207        CasacoreDataType::TpInt => io.put_i32(0)?,
208        CasacoreDataType::TpUInt => io.put_u32(0)?,
209        CasacoreDataType::TpFloat => io.put_f32(0.0)?,
210        CasacoreDataType::TpDouble => io.put_f64(0.0)?,
211        CasacoreDataType::TpComplex => io.put_complex32(Complex32::new(0.0, 0.0))?,
212        CasacoreDataType::TpDComplex => io.put_complex64(Complex64::new(0.0, 0.0))?,
213        CasacoreDataType::TpString => io.put_string("")?,
214        CasacoreDataType::TpInt64 => io.put_i64(0)?,
215        // ScalarRecordColumnDesc has no default value.
216        CasacoreDataType::TpRecord => {}
217        _ => {
218            return Err(StorageError::FormatMismatch(format!(
219                "cannot write default for type {:?}",
220                dt
221            )));
222        }
223    }
224    Ok(())
225}

227pub(super) fn write_column_set(
228    io: &mut AipsIo,
229    contents: &TableDatContents,
230) -> Result<(), StorageError> {
231    let cs = &contents.column_set;
232
233    if let Some(storage) = cs.storage {
234        io.put_i32(-3)?;
235        io.put_u64(cs.nrrow)?;
236        io.put_i32(storage.kind)?;
237        io.put_i32(storage.block_size)?;
238    } else {
239        io.put_i32(-2)?;
240        io.put_u32(cs.nrrow as u32)?;
241    }
242    io.put_u32(cs.seq_count)?;
243
244    // Data managers
245    io.put_u32(cs.data_managers.len() as u32)?;
246    for dm in &cs.data_managers {
247        io.put_string(&dm.type_name)?;
248        io.put_u32(dm.seq_nr)?;
249    }
250
251    // PlainColumn entries
252    for (i, col) in cs.columns.iter().enumerate() {
253        io.put_u32(2)?; // version=2
254        io.put_string(
255            contents
256                .names_on_file
257                .get(&col.original_name)
258                .unwrap_or(&col.original_name),
259        )?;
260        // ScaColData or ArrColData derived data
261        io.put_u32(1)?; // version=1
262        io.put_u32(col.dm_seq_nr)?;
263        if col.is_array {
264            // ArrColData: write shapeColDef + optional shapeCol (IPosition)
265            let col_desc = &contents.table_desc.columns[i];
266            let has_shape = col_desc.nrdim > 0 && !col_desc.shape.is_empty();
267            io.put_bool(has_shape)?;
268            if has_shape {
269                write_iposition(io, &col_desc.shape)?;
270            }
271        }
272    }
273
274    // DM data blobs (empty for StManAipsIO, populated for StandardStMan)
275    for dm in &cs.data_managers {
276        io.put_u32(dm.data.len() as u32)?;
277        if !dm.data.is_empty() {
278            io.put_u8_slice(&dm.data, false)?;
279        }
280    }
281
282    Ok(())
283}

285pub(crate) fn write_iposition(io: &mut AipsIo, values: &[i32]) -> Result<(), StorageError> {
286    io.putstart("IPosition", 1)?;
287    io.put_u32(values.len() as u32)?;
288    io.put_i32_slice(values, false)?;
289    io.putend()?;
290    Ok(())
291}

293pub(super) fn write_table_record(
294    io: &mut AipsIo,
295    record: &RecordValue,
296) -> Result<(), StorageError> {
297    io.putstart("TableRecord", 1)?;
298    write_record_desc(io, record)?;
299    io.put_i32(1)?; // recordType = RecordInterface::Variable
300
301    for field in record.fields() {
302        write_record_field_value(io, &field.value)?;
303    }
304
305    io.putend()?;
306    Ok(())
307}

309/// Deserialize a `RecordValue` from a uChar byte buffer (AipsIO `TableRecord` format).
310///
311/// C++ `ScalarRecordColumnData` stores records as `Vector<uChar>` containing
312/// AipsIO-serialized `TableRecord` objects. This function reverses that encoding.
313pub(crate) fn deserialize_record_from_uchar(bytes: &[u8]) -> Result<RecordValue, StorageError> {
314    use casa_aipsio::ByteOrder;
315    use std::io::Cursor;
316
317    let mut io =
318        AipsIo::new_read_only_with_order(Cursor::new(bytes.to_vec()), ByteOrder::BigEndian);
319    read_table_record(&mut io)
320}

322/// Serialize a `RecordValue` to a uChar byte buffer (AipsIO `TableRecord` format).
323///
324/// Produces the same encoding that C++ `ScalarRecordColumnData::putRecord` writes:
325/// a `TableRecord` AipsIO object serialized to a `Vector<uChar>`.
326pub(crate) fn serialize_record_to_uchar(record: &RecordValue) -> Result<Vec<u8>, StorageError> {
327    use casa_aipsio::ByteOrder;
328    use std::io::Cursor;
329
330    let buf = Vec::new();
331    let mut io = AipsIo::new_write_only_with_order(Cursor::new(buf), ByteOrder::BigEndian);
332    write_table_record(&mut io, record)?;
333    let cursor: Cursor<Vec<u8>> = io
334        .into_inner_typed()
335        .map_err(|e| StorageError::FormatMismatch(format!("serialize_record_to_uchar: {e}")))?;
336    Ok(cursor.into_inner())
337}

346pub(super) fn write_record_desc(io: &mut AipsIo, record: &RecordValue) -> Result<(), StorageError> {
347    io.putstart("RecordDesc", 2)?;
348    io.put_i32(record.fields().len() as i32)?;
349
350    for field in record.fields() {
351        io.put_string(&field.name)?;
352        let dt = value_to_casacore_data_type(&field.value)?;
353        io.put_i32(dt as i32)?;
354
355        if dt == CasacoreDataType::TpRecord {
356            // C++ writes an empty sub-record schema here (nfields=0).
357            // The actual fields are written in write_record_field_value as
358            // a full TableRecord (Variable record type).
359            write_empty_record_desc(io)?;
360        } else if dt == CasacoreDataType::TpTable {
361            // Table keyword entries carry an additional table-desc name slot.
362            // We do not model named table descriptors, so persist an empty name
363            // while keeping the wire layout aligned with casacore.
364            io.put_string("")?;
365        } else if is_array_data_type(dt) {
366            // A keyword array is of variable shape, as `RecordDesc::addField`
367            // without a shape makes it: `IPosition(1, -1)`. The array's own
368            // shape is written with its data (`write_array_record_field`).
369            // Writing the current shape here would fix the field to it, and
370            // `TableRecordRep::defineDataField` then refuses a later update of
371            // another shape (`Record::define - fixed array conformance error`).
372            write_iposition(io, &[-1])?;
373        }
374
375        io.put_string("")?; // comment
376    }
377
378    io.putend()?;
379    Ok(())
380}

382pub(super) fn write_record_field_value(io: &mut AipsIo, value: &Value) -> Result<(), StorageError> {
383    match value {
384        Value::Scalar(sv) => match sv {
385            ScalarValue::Bool(v) => io.put_bool(*v)?,
386            ScalarValue::UInt8(v) => io.put_u8(*v)?,
387            ScalarValue::UInt16(v) => io.put_u16(*v)?,
388            ScalarValue::UInt32(v) => io.put_u32(*v)?,
389            ScalarValue::Int16(v) => io.put_i16(*v)?,
390            ScalarValue::Int32(v) => io.put_i32(*v)?,
391            ScalarValue::Int64(v) => io.put_i64(*v)?,
392            ScalarValue::Float32(v) => io.put_f32(*v)?,
393            ScalarValue::Float64(v) => io.put_f64(*v)?,
394            ScalarValue::Complex32(v) => io.put_complex32(*v)?,
395            ScalarValue::Complex64(v) => io.put_complex64(*v)?,
396            ScalarValue::String(v) => io.put_string(v)?,
397        },
398        Value::TableRef(path) => io.put_string(path)?,
399        Value::Record(record) => {
400            write_table_record(io, record)?;
401        }
402        Value::Array(av) => {
403            write_array_record_field(io, av)?;
404        }
405    }
406    Ok(())
407}

409/// Write an array-typed record field to AipsIO.
410///
411/// C++ serializes `Array<T>` as: `"Array"` v3 → Int ndim → IPosition shape → count + data.
412pub(super) fn write_array_record_field(
413    io: &mut AipsIo,
414    av: &ArrayValue,
415) -> Result<(), StorageError> {
416    // C++ RecordRep uses type-specific AipsIO names for arrays.
417    let type_name = match av {
418        ArrayValue::Bool(_) => "Array<void>",
419        ArrayValue::UInt8(_) => "Array<uChar>",
420        ArrayValue::Int16(_) => "Array<short>",
421        ArrayValue::UInt16(_) => "Array<short>",
422        ArrayValue::Int32(_) => "Array<Int>",
423        ArrayValue::UInt32(_) => "Array<uInt>",
424        ArrayValue::Int64(_) => "Array<Int64>",
425        ArrayValue::Float32(_) => "Array<float>",
426        ArrayValue::Float64(_) => "Array<double>",
427        ArrayValue::Complex32(_) => "Array<void>",
428        ArrayValue::Complex64(_) => "Array<void>",
429        ArrayValue::String(_) => "Array<String>",
430    };
431    io.putstart(type_name, 3)?;
432
433    // Write ndim as u32, then shape as ndim × u32 (NOT as IPosition).
434    let ndim = av.ndim() as u32;
435    io.put_u32(ndim)?;
436    if ndim > 0 {
437        for &d in av.shape() {
438            io.put_u32(d as u32)?;
439        }
440        let nelem = av.len() as u32;
441        io.put_u32(nelem)?;
442        write_array_elements(io, av)?;
443    }
444
445    io.putend()?;
446    Ok(())
447}

449/// Write an array's elements in casacore's order, the first axis fastest,
450/// whatever the array's memory layout: an array held in C layout, or a
451/// strided view, is written as the logical array it is, and an array in
452/// Fortran layout is read straight from memory, as before. The reader
453/// ([`read_array_record_field`]) rebuilds Fortran-layout arrays.
454pub(super) fn write_array_elements(io: &mut AipsIo, av: &ArrayValue) -> Result<(), StorageError> {
455    match av {
456        ArrayValue::Bool(a) => {
457            // casacore's TypeIO stores Booleans bit-packed, eight to a byte;
458            // one byte each is a record casacore refuses ("part of object
459            // not read").
460            let flags: Vec<bool> = casacore_order(a).copied().collect();
461            io.put_bool_slice(&flags, false)?;
462            Ok(())
463        }
464        ArrayValue::UInt8(a) => put_each(io, a, AipsIo::put_u8),
465        ArrayValue::Int16(a) => put_each(io, a, AipsIo::put_i16),
466        ArrayValue::UInt16(a) => put_each(io, a, AipsIo::put_u16),
467        ArrayValue::Int32(a) => put_each(io, a, AipsIo::put_i32),
468        ArrayValue::UInt32(a) => put_each(io, a, AipsIo::put_u32),
469        ArrayValue::Float32(a) => put_each(io, a, AipsIo::put_f32),
470        ArrayValue::Float64(a) => put_each(io, a, AipsIo::put_f64),
471        ArrayValue::Int64(a) => put_each(io, a, AipsIo::put_i64),
472        ArrayValue::Complex32(a) => put_each(io, a, AipsIo::put_complex32),
473        ArrayValue::Complex64(a) => put_each(io, a, AipsIo::put_complex64),
474        ArrayValue::String(a) => {
475            for s in casacore_order(a) {
476                io.put_string(s)?;
477            }
478            Ok(())
479        }
480    }
