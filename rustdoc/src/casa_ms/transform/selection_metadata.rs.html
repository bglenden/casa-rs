19/// CASA's visibility iterator groups ordinary MS rows by the original IDs
20/// before ordering each group by TIME (MSTransformManager::setIterationApproach).
21pub(super) fn sort_selected_rows(
22    table: &Table,
23    request: &MsTransformRequest,
24    rows: Vec<usize>,
25    descriptions: Vec<i32>,
26) -> Result<(Vec<usize>, Vec<i32>), MsTransformError> {
27    const GROUPS: [&str; 5] = [
28        "OBSERVATION_ID",
29        "ARRAY_ID",
30        "SCAN_NUMBER",
31        "STATE_ID",
32        "FIELD_ID",
33    ];
34    let read = |name| {
35        table
36            .column_accessor(name)
37            .and_then(|column| column.scalar_cells_owned_for_rows(&rows))
38            .map_err(|source| MsTransformError::MutateMeasurementSet {
39                path: request.input_ms.display().to_string(),
40                source: Box::new(source),
41            })
42    };
43    let mut groups = vec![[0; GROUPS.len()]; rows.len()];
44    for (slot, name) in GROUPS.into_iter().enumerate() {
45        for ((group, value), &row) in groups.iter_mut().zip(read(name)?).zip(&rows) {
46            group[slot] = scalar_i32(value.as_ref(), name, row)?;
47        }
48    }
49    let times = read("TIME")?;
50    let mut ordering = groups
51        .into_iter()
52        .zip(times)
53        .zip(rows)
54        .zip(descriptions)
55        .map(|(((group, time), row), ddid)| {
56            Ok((group, ddid, scalar_f64(time.as_ref(), "TIME", row)?, row))
57        })
58        .collect::<Result<Vec<_>, MsTransformError>>()?;
59    ordering.sort_unstable_by(|left, right| {
60        left.0
61            .cmp(&right.0)
62            .then_with(|| left.1.cmp(&right.1))
63            .then_with(|| left.2.total_cmp(&right.2))
64            .then_with(|| left.3.cmp(&right.3))
65    });
66    Ok(ordering
67        .into_iter()
68        .map(|(_, ddid, _, row)| (row, ddid))
69        .unzip())
70}

72pub(super) struct RowIdMaps {
73    antenna: Option<BTreeMap<i32, i32>>,
74    observation: Option<BTreeMap<i32, i32>>,
75    state: Option<BTreeMap<i32, i32>>,
76}
77
78impl RowIdMaps {
79    pub(super) fn for_column(&self, column: &str) -> Option<&BTreeMap<i32, i32>> {
80        match column {
81            "ANTENNA1" | "ANTENNA2" => self.antenna.as_ref(),
82            "OBSERVATION_ID" => self.observation.as_ref(),
83            "STATE_ID" => self.state.as_ref(),
84            _ => None,
85        }
86    }
87}

89pub(super) fn reindex_observation_and_state(
90    input: &MeasurementSet,
91    request: &MsTransformRequest,
92    antenna: Option<BTreeMap<i32, i32>>,
93) -> Result<RowIdMaps, MsTransformError> {
94    let observation = request
95        .selection
96        .observation
97        .as_deref()
98        .filter(|text| !text.trim().is_empty())
99        .map(|text| {
100            let table = input
101                .observation()
102                .map_err(|source| metadata_error(request, source))?;
103            let ranges = parse_numeric_row_id_ranges(text, "observation")
104                .map_err(|source| metadata_error(request, source))?;
105            let selected = (0..table.row_count() as i32)
106                .filter(|&id| {
107                    ranges
108                        .iter()
109                        .any(|&(first, last)| (first..=last).contains(&id))
110                })
111                .collect::<Vec<_>>();
112            reindex_table(request, SubtableId::Observation, selected)
113        })
114        .transpose()?;
115    let state = request
116        .selection
117        .intent
118        .as_deref()
119        .filter(|text| !text.trim().is_empty())
120        .map(|text| {
121            let selected = parse_state_selector(input, text)
122                .map_err(|source| metadata_error(request, source))?;
123            reindex_table(request, SubtableId::State, selected)
124        })
125        .transpose()?;
126    Ok(RowIdMaps {
127        antenna,
128        observation,
129        state,
130    })
131}

133fn reindex_table(
134    request: &MsTransformRequest,
135    id: SubtableId,
136    selected: Vec<i32>,
137) -> Result<BTreeMap<i32, i32>, MsTransformError> {
138    let name = id.name();
139    let rows = selected.iter().map(|&row| row as usize).collect::<Vec<_>>();
140    rewrite_selected_table_rows(
141        &request.input_ms.join(name),
142        &request.output_ms.join(name),
143        &rows,
144        |_, row| Ok(row),
145    )?;
146    Ok(selected
147        .into_iter()
148        .enumerate()
149        .map(|(out, input)| (input, out as i32))
150        .collect())
151}

153pub(super) fn reindex_fields_and_sources(
154    input: &MeasurementSet,
155    request: &MsTransformRequest,
156    field: Option<&str>,
157    spw_ids: &BTreeMap<i32, i32>,
158) -> Result<Option<BTreeMap<i32, i32>>, MsTransformError> {
159    let fields = input
160        .field()
161        .map_err(|source| metadata_error(request, source))?;
162    let selected = field
163        .filter(|text| !text.trim().is_empty())
164        .map(|text| {
165            input
166                .resolve_field_selector(text)
167                .map_err(|source| metadata_error(request, source))
168        })
169        .transpose()?;
170    let field_rows = match &selected {
171        Some(ids) => ids.iter().map(|&id| id as usize).collect::<Vec<_>>(),
172        None => (0..fields.row_count()).collect(),
173    };
174    let mut source_map = BTreeMap::new();
175    let has_source = input.subtable(SubtableId::Source).is_some();
176    let mut field_sources = BTreeMap::new();
177    let mut sources_changed = false;
178    for &index in &field_rows {
179        let old_id = fields
180            .source_id(index)
181            .map_err(|source| metadata_error(request, source))?;
182        let new_id = if has_source && old_id >= 0 {
183            let next_id = source_map.len() as i32;
184            *source_map.entry(old_id).or_insert(next_id)
185        } else {
186            -1
187        };
188        sources_changed |= new_id != old_id;
189        field_sources.insert(index, new_id);
190    }
191    if selected.is_some() || sources_changed {
192        let name = SubtableId::Field.name();
193        rewrite_selected_table_rows(
194            &request.input_ms.join(name),
195            &request.output_ms.join(name),
196            &field_rows,
197            |index, mut row| {
198                row.upsert(
199                    "SOURCE_ID",
200                    Value::Scalar(ScalarValue::Int32(field_sources[&index])),
201                );
202                Ok(row)
203            },
204        )?;
205    }
206    if let Some(source) = input.subtable(SubtableId::Source) {
207        reindex_sources(source, request, &source_map, spw_ids)?;
208    }
209    Ok(selected.map(|ids| {
210        ids.into_iter()
211            .enumerate()
212            .map(|(new, old)| (old, new as i32))
213            .collect()
214    }))
215}

217fn reindex_sources(
218    source: &Table,
219    request: &MsTransformRequest,
220    source_map: &BTreeMap<i32, i32>,
221    spw_map: &BTreeMap<i32, i32>,
222) -> Result<(), MsTransformError> {
223    let mut selected = BTreeMap::new();
224    let mut changed = false;
225    for index in 0..source.row_count() {
226        let id =
227            get_i32(source, index, "SOURCE_ID").map_err(|error| metadata_error(request, error))?;
228        let spw = get_i32(source, index, "SPECTRAL_WINDOW_ID")
229            .map_err(|error| metadata_error(request, error))?;
230        let new_spw = if spw == -1 {
231            Some(-1)
232        } else {
233            spw_map.get(&spw).copied()
234        };
235        if let (Some(&new_id), Some(new_spw)) = (source_map.get(&id), new_spw) {
236            selected.insert(index, (new_id, new_spw));
237            changed |= (new_id, new_spw) != (id, spw);
238        }
239    }
240    if changed || selected.len() != source.row_count() {
241        let name = SubtableId::Source.name();
242        let rows = selected.keys().copied().collect::<Vec<_>>();
243        rewrite_selected_table_rows(
244            &request.input_ms.join(name),
245            &request.output_ms.join(name),
246            &rows,
247            |index, mut row| {
248                let (id, spw) = selected[&index];
249                row.upsert("SOURCE_ID", Value::Scalar(ScalarValue::Int32(id)));
250                row.upsert("SPECTRAL_WINDOW_ID", Value::Scalar(ScalarValue::Int32(spw)));
251                Ok(row)
252            },
253        )?;
254    }
255    Ok(())
256}

258pub(super) fn selected_antenna_ids(
259    input: &MeasurementSet,
260    antenna: Option<&str>,
261    rows: &[ResolvedMsSelectionRow],
262) -> Result<Option<Vec<i32>>, MsTransformError> {
263    let Some(text) = antenna.filter(|text| !text.trim().is_empty()) else {
264        return Ok(None);
265    };
266    let error = |source| MsTransformError::SelectRows {
267        path: input
268            .main_table()
269            .path()
270            .map_or_else(String::new, |path| path.display().to_string()),
271        source,
272    };
273    let antenna_count = input.antenna().map_err(error)?.row_count();
274    let ids = input.transform_antenna_ids(text).map_err(error)?;
275    // Expressions with negation or a length/regex term can select rows using
276    // antennas absent from the parser's positive ID lists. Keep their references
277    // valid even when CASA's remapping would yield a negative MAIN antenna ID.
278    let compact = ids.len() < antenna_count
279        && rows.iter().all(|row| {
280            ids.binary_search(&row.antenna1_id).is_ok()
281                && ids.binary_search(&row.antenna2_id).is_ok()
282        });
283    Ok(compact.then_some(ids))
284}

286pub(super) fn reindex_antenna_subtables(
287    input: &MeasurementSet,
288    request: &MsTransformRequest,
289    antenna_ids: Option<&[i32]>,
290    spw_ids: &BTreeMap<i32, i32>,
291    time: Option<&TimeSelection>,
292) -> Result<Option<BTreeMap<i32, i32>>, MsTransformError> {
293    let antenna_map = antenna_ids.map(|ids| {
294        ids.iter()
295            .enumerate()
296            .map(|(new, &old)| (old, new as i32))
297            .collect::<BTreeMap<_, _>>()
298    });
299    if let Some(ids) = antenna_ids {
300        let name = SubtableId::Antenna.name();
301        let rows = ids.iter().map(|&id| id as usize).collect::<Vec<_>>();
302        rewrite_selected_table_rows(
303            &request.input_ms.join(name),
304            &request.output_ms.join(name),
305            &rows,
306            |_, row| Ok(row),
307        )?;
308    }
309    let spw_count = input
310        .spectral_window()
311        .map_err(|source| metadata_error(request, source))?
312        .row_count();
313    if antenna_map.is_some() || spw_ids.len() != spw_count {
314        for name in ["FEED", "SYSCAL", "CALDEVICE", "SYSPOWER"] {
315            reindex_subtable(request, name, antenna_map.as_ref(), Some(spw_ids), None)?;
316        }
317    }
318    if antenna_map.is_some() {
319        reindex_subtable(request, "WEATHER", antenna_map.as_ref(), None, None)?;
320    }
321    if antenna_map.is_some() || time.is_some() {
322        reindex_subtable(request, "POINTING", antenna_map.as_ref(), None, time)?;
323    }
324    Ok(antenna_map)
325}

327fn reindex_subtable(
328    request: &MsTransformRequest,
329    name: &str,
330    antenna_map: Option<&BTreeMap<i32, i32>>,
331    spw_map: Option<&BTreeMap<i32, i32>>,
332    time: Option<&TimeSelection>,
333) -> Result<(), MsTransformError> {
334    let source = request.input_ms.join(name);
335    if !source.exists() {
336        return Ok(());
337    }
338    let table = Table::open(TableOptions::new(&source)).map_err(|source| {
339        MsTransformError::MutateMeasurementSet {
340            path: request.input_ms.display().to_string(),
341            source: Box::new(source),
342        }
343    })?;
344    let mut ids = BTreeMap::new();
345    for row in 0..table.row_count() {
346        let antenna =
347            get_i32(&table, row, "ANTENNA_ID").map_err(|source| metadata_error(request, source))?;
348        let new_antenna = match (antenna_map, antenna >= 0) {
349            (Some(map), true) => map.get(&antenna).copied(),
350            (_, false) if name != "WEATHER" => None,
351            _ => Some(antenna),
352        };
353        let Some(new_antenna) = new_antenna else {
354            continue;
355        };
356        let new_spw = if let Some(map) = spw_map {
357            let spw = get_i32(&table, row, "SPECTRAL_WINDOW_ID")
358                .map_err(|source| metadata_error(request, source))?;
359            if spw < 0 {
360                Some(spw)
361            } else if let Some(&new_spw) = map.get(&spw) {
362                Some(new_spw)
363            } else {
364                continue;
365            }
366        } else {
367            None
368        };
369        if let Some(time) = time {
370            let epoch =
371                get_f64(&table, row, "TIME").map_err(|source| metadata_error(request, source))?;
372            // CASA's POINTING filter uses the recorded interval endpoints.
373            // Those can differ from the predicate that selects MAIN rows.
374            if !time
375                .time_list()
376                .any(|term| epoch >= term.start_s && epoch <= term.end_s)
377            {
378                continue;
379            }
380        }
381        ids.insert(row, (new_antenna, new_spw));
382    }
383    let rows = ids.keys().copied().collect::<Vec<_>>();
384    rewrite_selected_table_rows(
385        &source,
386        &request.output_ms.join(name),
387        &rows,
388        |index, mut row| {
389            let (antenna, spw) = ids[&index];
390            row.upsert("ANTENNA_ID", Value::Scalar(ScalarValue::Int32(antenna)));
391            if let Some(spw) = spw {
392                row.upsert("SPECTRAL_WINDOW_ID", Value::Scalar(ScalarValue::Int32(spw)));
393            }
394            Ok(row)
395        },
396    )
397}

399fn metadata_error(request: &MsTransformRequest, source: MsError) -> MsTransformError {
400    MsTransformError::SpectralMetadata {
401        path: request.input_ms.display().to_string(),
402        reason: source.to_string(),
403    }

406/// Project only selected source cells. A narrow POINTING selection can have
407/// very few output rows even when the input covers an entire observation.
408pub(super) fn read_subtable_rows(
409    table: &Table,
410    indices: &[usize],
411) -> Result<Vec<RecordValue>, TableError> {
412    let schema = table
413        .schema()
414        .ok_or_else(|| TableError::Schema("missing subtable schema".into()))?;
415    let mut rows = vec![RecordValue::new(Vec::new()); indices.len()];
416    for column in schema.columns() {
417        let name = column.name();
418        let cells = table.column_accessor(name)?;
419        let values = match column.column_type() {
420            ColumnType::Scalar => cells
421                .scalar_cells_owned_for_rows(indices)?
422                .into_iter()
423                .map(|cell| cell.map(Value::Scalar))
424                .collect(),
425            ColumnType::Array(_) => cells
426                .array_cells_owned_uncached(indices)?
427                .into_iter()
428                .map(|cell| cell.map(Value::Array))
429                .collect(),
430            ColumnType::Record => indices
431                .iter()
432                .map(|&index| {
433                    cells
434                        .record_cell(index)
435                        .map(|cell| Some(Value::Record(cell)))
436                })
437                .collect::<Result<Vec<_>, _>>()?,
438        };
439        for (row, value) in rows.iter_mut().zip(values) {
440            if let Some(value) = value {
441                row.upsert(name, value);
442            }
443        }
444    }
445    Ok(rows)
