17pub(crate) const CORRELATOR_EFFICIENCY: f32 = 0.932;
18const FRINGE_ROTATION_SCALE: f32 = 1.176;
19const FIELD_SCAN_TOLERANCE_SECONDS: f64 = 0.1;
20const MISSING_NOISE: [f32; 2] = [-1.0; 2];

27pub(crate) fn input_paths(
28    ms: &MeasurementSet,
29    path: &Path,
30) -> Result<(PathBuf, PathBuf), GencalError> {
31    let resolve = |name| {
32        ms.main_table()
33            .keywords()
34            .get(name)
35            .and_then(Value::as_table_ref)
36            .map(|reference| casa_tables::add_directory(reference, path))
37            .ok_or_else(|| invalid(format!("RecordInterface: field {name} is unknown")))
38    };
39    Ok((resolve("SYSPOWER")?, resolve("CALDEVICE")?))
40}

42pub(super) fn rows(
43    ms: &MeasurementSet,
44    power_path: &Path,
45    noise_path: &Path,
46    caltype: GencalType,
47    selection: Option<&casa_ms::MsSelection>,
48) -> Result<Vec<PriorCalRow>, GencalError> {
49    let power = open(power_path, "SYSPOWER")?;
50    let noise = open(noise_path, "CALDEVICE")?;
51    validate_scalar_columns(&power, &[("INTERVAL", PrimitiveType::Float64)])?;
52    validate_scalar_columns(
53        &noise,
54        &[
55            (COL_TIME, PrimitiveType::Float64),
56            ("INTERVAL", PrimitiveType::Float64),
57            ("NUM_CAL_LOAD", PrimitiveType::Int32),
58        ],
59    )?;
60    let antennas = ms.antenna().map_err(invalid)?.row_count();
61    let windows = ms.spectral_window().map_err(invalid)?.row_count();
62    let labels = visibility_labels(ms, selection)?;
63    let noise = noise_temperatures(&noise, antennas, windows)?;
64    let mut samples = (0..power.row_count())
65        .map(|row| {
66            let time = get_f64(&power, row, COL_TIME)?;
67            if !time.is_finite() {
68                return Err(invalid(format!("Invalid TIME in SYSPOWER row {row}")));
69            }
70            let (window, antenna) = ids(&power, row, windows, antennas)?;
71            Ok((time, window, antenna, row))
72        })
73        .collect::<Result<Vec<_>, _>>()?;
74    samples.sort_by(|a, b| a.0.total_cmp(&b.0).then(a.1.cmp(&b.1)));
75
76    let mut output = Vec::new();
77    let mut gains = vec![([1.0; 4], false); antennas];
78    for group in samples.chunk_by(|a, b| a.0 == b.0 && a.1 == b.1) {
79        let (time, window, _, _) = group[0];
80        gains.fill(([1.0; 4], false));
81        for &(_, _, antenna, row) in group {
82            let temperature = noise
83                .get(&(window, antenna))
84                .copied()
85                .unwrap_or(MISSING_NOISE);
86            gains[antenna] = power_parameters(&power, row, temperature, caltype)?;
87        }
88        let (field, scan) = labels_at(&labels, time);
89        for (antenna, &(values, good)) in gains.iter().enumerate() {
90            let flags = ArrayD::from_elem(IxDyn(&[4, 1]), !good);
91            output.push(PriorCalRow {
92                time_seconds: time,
93                interval_seconds: 0.0,
94                field_id: field,
95                spw_id: window as i32,
96                antenna_id: antenna as i32,
97                observation_id: -1,
98                scan_number: scan,
99                parameters: ArrayValue::Float32(
100                    ArrayD::from_shape_vec(IxDyn(&[4, 1]), values.to_vec())
101                        .expect("four parameters"),
102                ),
103                snr_value: 1.0,
104                preserved: Some(RecordValue::new(vec![RecordField::new(
105                    COL_FLAG,
106                    Value::Array(ArrayValue::Bool(flags)),
107                )])),
108            });
109        }
110    }
111    if output.is_empty() {
112        return Err(invalid("SYSPOWER contains no switched-power samples"));
113    }
114    Ok(output)
115}

117fn open(path: &Path, name: &str) -> Result<Table, GencalError> {
118    Table::open(TableOptions::new(path)).map_err(|source| {
119        invalid(format!(
120            "The {name} subtable is not readable at {}: {source}",
121            path.display()
122        ))
123    })
124}

126fn validate_scalar_columns(
127    table: &Table,
128    columns: &[(&str, PrimitiveType)],
129) -> Result<(), GencalError> {
130    for &(name, expected) in columns {
131        let column = table
132            .schema()
133            .and_then(|schema| schema.column(name))
134            .ok_or_else(|| invalid(format!("Table column {name} is unknown")))?;
135        if *column.column_type() != ColumnType::Scalar || column.data_type() != Some(expected) {
136            return Err(invalid(format!(
137                "Table column {name} has the wrong scalar type"
138            )));
139        }
140    }
141    Ok(())
142}

144fn ids(
145    table: &Table,
146    row: usize,
147    windows: usize,
148    antennas: usize,
149) -> Result<(usize, usize), GencalError> {
150    let id = |name, count| {
151        let value = table
152            .cell_accessor(row, name)
153            .and_then(|cell| cell.scalar())
154            .map_err(invalid)?;
155        match value {
156            ScalarValue::Int32(value) if *value >= 0 && (*value as usize) < count => {
157                Ok(*value as usize)
158            }
159            _ => Err(invalid(format!("Invalid {name} in row {row}"))),
160        }
161    };
162    Ok((
163        id("SPECTRAL_WINDOW_ID", windows)?,
164        id("ANTENNA_ID", antennas)?,
165    ))
166}

168fn float_array<'a>(
169    table: &'a Table,
170    row: usize,
171    name: &str,
172) -> Result<&'a ArrayD<f32>, GencalError> {
173    match table
174        .cell_accessor(row, name)
175        .and_then(|cell| cell.array())
176        .map_err(invalid)?
177    {
178        ArrayValue::Float32(array) => Ok(array),
179        _ => Err(invalid(format!(
180            "{name} in row {row} must be a Float32 array"
181        ))),
182    }
183}

185fn noise_temperatures(
186    table: &Table,
187    antennas: usize,
188    windows: usize,
189) -> Result<HashMap<(usize, usize), [f32; 2]>, GencalError> {
190    let mut noise = HashMap::new();
191    for row in 0..table.row_count() {
192        let key = ids(table, row, windows, antennas)?;
193        if noise.contains_key(&key) {
194            continue;
195        }
196        let values = float_array(table, row, "NOISE_CAL")?;
197        let nominal = match values.len() {
198            1 => [*values.first().expect("one value"); 2],
199            2 => {
200                let mut values = values.iter();
201                [
202                    *values.next().expect("two values"),
203                    *values.next().expect("two values"),
204                ]
205            }
206            4 if values.shape() == [2, 2] => [values[[0, 0]], values[[0, 1]]],
207            4 => {
208                return Err(invalid(
209                    "NOISE_CAL with four entries must have shape [2, 2]",
210                ));
211            }
212            _ => MISSING_NOISE,
213        };
214        noise.insert(key, nominal);
215    }
216    Ok(noise)
217}

219fn power_parameters(
220    table: &Table,
221    row: usize,
222    noise: [f32; 2],
223    caltype: GencalType,
224) -> Result<([f32; 4], bool), GencalError> {
225    let receptors = |name| {
226        let values = float_array(table, row, name)?;
227        if values.shape() != [2] {
228            return Err(invalid(format!(
229                "{name} in row {row} must have two receptors"
230            )));
231        }
232        Ok([values[[0]], values[[1]]])
233    };
234    let sum = receptors("SWITCHED_SUM")?;
235    let diff = receptors("SWITCHED_DIFF")?;
236    let rq = receptors("REQUANTIZER_GAIN")?;
237    let good = |values: [f32; 2]| values.iter().all(|&value| value > f32::EPSILON);
238    // CASA's SWPOW validity check also requires the RQ plane, including for
239    // the evlagain alias, while the gain formula itself uses diff and noise.
240    let valid = match caltype {
241        GencalType::Rq => good(rq),
242        GencalType::Swpwts => good(sum) && good(rq),
243        _ => good(sum) && good(diff) && good(noise) && good(rq),
244    };
245    let mut parameters = [1.0; 4];
246    if valid {
247        let digital = (CORRELATOR_EFFICIENCY * FRINGE_ROTATION_SCALE).sqrt();
248        for receptor in 0..2 {
249            let (sum, diff, rq, noise) =
250                (sum[receptor], diff[receptor], rq[receptor], noise[receptor]);
251            let (gain, tsys) = match caltype {
252                GencalType::Rq => (rq, 1.0 / (rq * rq)),
253                GencalType::Swpwts => (rq, (sum / 2.0) / (rq * rq)),
254                GencalType::SwpOverRq => ((diff / noise).sqrt() / rq, noise * sum / diff / 2.0),
255                _ => ((diff / noise).sqrt(), noise * sum / diff / 2.0),
256            };
257            parameters[receptor * 2] = gain * digital;
258            parameters[receptor * 2 + 1] = tsys;
259        }
260    }
261    Ok((parameters, valid))
262}

264fn visibility_labels(
265    ms: &MeasurementSet,
266    selection: Option<&casa_ms::MsSelection>,
267) -> Result<VisibilityLabels, GencalError> {
268    let columns = ms
269        .main_table()
270        .required_scalar_columns_owned(&[
271            COL_TIME,
272            COL_FIELD_ID,
273            COL_SCAN_NUMBER,
274            COL_OBSERVATION_ID,
275            COL_ARRAY_ID,
276        ])
277        .map_err(invalid)?;
278    let RequiredScalarColumnValues::Float64(times) = &columns[COL_TIME] else {
279        return Err(invalid("MeasurementSet TIME is not Float64"));
280    };
281    let ids = |name| match &columns[name] {
282        RequiredScalarColumnValues::Int32(values) => Ok(values.as_slice()),
283        _ => Err(invalid(format!("MeasurementSet {name} is not Int32"))),
284    };
285    let fields = ids(COL_FIELD_ID)?;
286    let scans = ids(COL_SCAN_NUMBER)?;
287    let observations = ids(COL_OBSERVATION_ID)?;
288    let arrays = ids(COL_ARRAY_ID)?;
289    let mut fields_by_time = HashMap::new();
290    let mut scan_limits = HashMap::new();
291    let observe = |row: usize| {
292        let time = times[row];
293        if !time.is_finite() {
294            return;
295        }
296        fields_by_time
297            .entry(time.to_bits())
298            .and_modify(|field: &mut i32| *field = (*field).min(fields[row]))
299            .or_insert(fields[row]);
300        let limits = scan_limits
301            .entry((observations[row], arrays[row], scans[row]))
302            .or_insert((time, time));
303        limits.0 = limits.0.min(time);
304        limits.1 = limits.1.max(time);
305    };
306    if let Some(selection) = selection {
307        crate::session::resolve_casa_calibration_selection(ms, selection)
308            .map_err(invalid)?
309            .row_indices()
310            .for_each(observe);
311    } else {
312        (0..times.len()).for_each(observe);
313    }
314    // At any visibility time CASA's set reports the smallest field ID.
315    let mut fields_by_time = fields_by_time
316        .into_iter()
317        .map(|(time, field)| (f64::from_bits(time), field))
318        .collect::<Vec<_>>();
319    fields_by_time.sort_unstable_by(|a, b| a.0.total_cmp(&b.0));
320    Ok(VisibilityLabels {
321        fields_by_time,
322        scan_limits: scan_limits
323            .into_iter()
324            .map(|((_, _, scan), limits)| (scan, limits))
325            .collect(),
326    })
327}

329fn labels_at(labels: &VisibilityLabels, time: f64) -> (i32, i32) {
330    let (low, high) = (
331        time - FIELD_SCAN_TOLERANCE_SECONDS,
332        time + FIELD_SCAN_TOLERANCE_SECONDS,
333    );
334    let fields = &labels.fields_by_time;
335    let first = fields.partition_point(|&(timestamp, _)| timestamp < low);
336    // MSMetadata inserts fields at the first time above the high endpoint
337    // before exiting its search. Scan matching instead tests the full span.
338    let after = fields.partition_point(|&(timestamp, _)| timestamp <= high);
339    let end = (after + 1).min(fields.len());
340    let field = fields[first..end]
341        .iter()
342        .map(|&(_, field)| field)
343        .min()
344        .unwrap_or(-1);
345    let scan = labels
346        .scan_limits
347        .iter()
348        .filter(|&&(_, (start, end))| start <= high && end >= low)
349        .map(|&(scan, _)| scan)
350        .min()
351        .unwrap_or(-1);
352    (field, scan)
353}
