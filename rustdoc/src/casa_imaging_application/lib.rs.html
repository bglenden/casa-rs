51/// What a run needs besides its request: the host, how much of it the run
52/// may use, the cancellation the caller sets, and where the run summary
53/// goes.
54#[derive(Clone, Debug)]
55pub struct RunContext {
56    /// The host the run admits its phases against.
57    pub host: HostResources,
58    /// How much of the host the run may use.
59    pub policy: ResourcePolicy,
60    /// Set (SIGINT in `casars-imager`) to stop the run at the next block
61    /// boundary or phase; nothing is published.
62    pub cancel: Cancel,
63    /// Where the run writes its summary, whether it completes or fails;
64    /// `None` writes no file.
65    pub summary: Option<SummaryTarget>,
66}

195/// Stable application spelling of the scientific minor-cycle terminal reason.
196#[derive(Clone, Copy, Debug, PartialEq, Eq)]
197pub enum NativeMinorCycleStopReason {
198    /// The normalized residual peak fell below the requested threshold.
199    ThresholdReached,
200    /// The bounded minor-cycle iteration budget was exhausted.
201    IterationBound,
202    /// A plane's peak residual rose more than 10% above its minimum in the
203    /// cycle (CASA's plane stop code 4, any solver).
204    Diverged,
205}
206
207impl NativeMinorCycleStopReason {
208    /// The cycle's reason from its planes' CASA stop codes: divergence in any
209    /// plane, else an iteration limit or early exit in any plane, else a
210    /// threshold.
211    #[must_use]
212    pub fn from_planes(stops: &[casa_imaging_deconvolution::PlaneStop]) -> Self {
213        use casa_imaging_deconvolution::PlaneStop;
214        if stops.contains(&PlaneStop::Diverged) {
215            Self::Diverged
216        } else if stops
217            .iter()
218            .any(|stop| matches!(stop, PlaneStop::Iterations | PlaneStop::Exited))
219        {
220            Self::IterationBound
221        } else {
222            Self::ThresholdReached
223        }
224    }
225}

227/// Run one imaging request on the installed implementation: validate it,
228/// compile it against its MeasurementSet, check once that the
229/// implementation can run it ([`Unsupported`]), and run it.
230pub fn execute(
231    request: &ImagingRequest,
232    context: RunContext,
233) -> Result<ImagingOutcome, ApplicationDispatchError> {
234    request
235        .validate()
236        .map_err(ApplicationDispatchError::Request)?;
237    let prepared =
238        compile::prepare(request, &context).map_err(ApplicationDispatchError::Preparation)?;
239    let resolved = resolve_selected_observation(prepared.observation.clone())
240        .map_err(|error| ApplicationDispatchError::Preparation(error.into()))?;
241    let (snapshot, access) = resolved.into_parts();
242    let observation = compile_observation(snapshot)
243        .map_err(|error| ApplicationDispatchError::Preparation(error.into()))?;
244    let problem = compile(ProblemInput::new(
245        prepared.specification,
246        prepared.geometry,
247        observation,
248        prepared.model_lifecycle,
249    ))
250    .map_err(ApplicationDispatchError::Compile)?;
251    availability::check(
252        &problem,
253        request.backend,
254        request.gridprecision,
255        &context.host,
256    )
257    .map_err(ApplicationDispatchError::Unavailable)?;
258    let native = prepared
259        .deployment
260        .deploy()
261        .map_err(ApplicationDispatchError::Preparation)?;
262    let input = NativeInput {
263        observation: prepared.observation,
264        initial_access: access,
265        write_model_column: prepared.write_model_column,
266        write_corrected_data: prepared.write_corrected_data,
267        masks: prepared.masks,
268        minor_cycle_image_response: prepared.minor_cycle_image_response,
269    };
270    run_native(&problem, input, native).map_err(|error| match error {
271        NativeError::Admission(admission) => ApplicationDispatchError::Admission(admission),
272        NativeError::Cancelled(_) => ApplicationDispatchError::Cancelled,
273        NativeError::Other(error) => ApplicationDispatchError::Native(error),
274    })
275}

277pub(crate) fn visibility_write_selection(
278    problem: &CompiledProblem,
279    selected: Arc<ObservationSelection>,
280) -> Result<Arc<ObservationSelection>, ApplicationError> {
281    let Some(transform) = problem.visibility_transform() else {
282        return Ok(selected);
283    };
284    let spectral_windows = selected
285        .spectral_windows()
286        .iter()
287        .map(|selection| {
288            let output_channels = selection
289                .channel_indices()
290                .iter()
291                .copied()
292                .filter(|channel| {
293                    transform.rules().iter().any(|rule| {
294                        rule.spectral_window_id() == selection.spectral_window_id()
295                            && rule
296                                .channel_use(*channel)
297                                .is_some_and(|role| role.contributes_to_output())
298                    })
299                })
300                .collect::<Vec<_>>();
301            if output_channels.is_empty() {
302                return Err(boxed(
303                    "continuum transform selected no visibility-write output channels",
304                ));
305            }
306            Ok(SpectralWindowSelection::new(
307                selection.spectral_window_id(),
308                output_channels,
309            ))
310        })
311        .collect::<Result<Vec<_>, ApplicationError>>()?;
312    Ok(Arc::new(ObservationSelection::new(
313        selected.rows().clone(),
314        selected.rows_filter().clone(),
315        selected.data_descriptions().to_vec(),
316        spectral_windows,
317        selected.correlations().to_vec(),
318    )))
319}

321/// Failure before or within the installed whole-run implementation.
322#[derive(Debug)]
323pub enum ApplicationDispatchError {
324    /// The request's parameters contradict each other.
325    Request(InvalidRequest),
326    /// The request could not be compiled against its MeasurementSet, or
327    /// the compiled run could not be deployed; nothing ran.
328    Preparation(PrepareError),
329    /// Backend-independent request compilation failed.
330    Compile(CompileProblemError),
331    /// The installed implementation cannot run the compiled problem.
332    Unavailable(availability::ImplementationUnavailable),
333    /// A phase's memory did not fit the resource policy; nothing was
334    /// published.
335    Admission(Admission),
336    /// The run was cancelled; nothing was published.
337    Cancelled,
338    /// The sole installed implementation failed.
339    Native(ApplicationError),
340}
341
342impl fmt::Display for ApplicationDispatchError {
343    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
344        match self {
345            Self::Request(error) => error.fmt(formatter),
346            Self::Preparation(error) => {
347                write!(formatter, "imaging application preparation failed: {error}")
348            }
349            Self::Compile(error) => {
350                write!(formatter, "imaging request compilation failed: {error}")
351            }
352            Self::Unavailable(error) => error.fmt(formatter),
353            Self::Admission(error) => write!(formatter, "native imaging run refused: {error}"),
354            Self::Cancelled => write!(formatter, "native imaging run cancelled"),
355            Self::Native(error) => write!(formatter, "native imaging run failed: {error}"),
356        }
357    }
358}
