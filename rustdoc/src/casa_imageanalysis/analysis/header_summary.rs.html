5pub(super) fn imhead_typed<T>(
6    path: &Path,
7    image: &PagedImage<T>,
8    pixel_type: String,
9) -> Result<ImageHeaderSummary, ImageError>
10where
11    T: ImagePixel,
12{
13    let info = image.image_info()?;
14    let default_mask = image.default_mask_name();
15    let has_mask = default_mask.is_some();
16    Ok(ImageHeaderSummary {
17        imagename: path.display().to_string(),
18        pixel_type,
19        shape: image.shape().to_vec(),
20        tile_shape: image.tile_shape().to_vec(),
21        units: image.units().to_string(),
22        object_name: info.object_name.clone(),
23        observer: image.coordinates().obs_info().observer.clone(),
24        telescope: image.coordinates().obs_info().telescope.clone(),
25        observation_date: casa_core::quanta::format::format_time(
26            image
27                .coordinates()
28                .obs_info()
29                .date
30                .as_ref()
31                .map_or(0.0, |epoch| epoch.value().as_mjd()),
32            casa_core::quanta::format::YMD,
33            12,
34        ),
35        direction_reference: (0..image.coordinates().n_coordinates()).find_map(|index| match image
36            .coordinates()
37            .coordinate(index)
38        {
39            casa_coordinates::CoordinateModel::Direction(direction) => {
40                Some(direction.direction_ref().as_str().to_string())
41            }
42            _ => None,
43        }),
44        direction_projection: (0..image.coordinates().n_coordinates()).find_map(
45            |index| match image.coordinates().coordinate(index) {
46                casa_coordinates::CoordinateModel::Direction(direction) => {
47                    Some(projection_header::projection_header(direction))
48                }
49                _ => None,
50            },
51        ),
52        spectral: spectral_header(image)?.map(Box::new),
53        image_type: header_image_type(image)?.to_string(),
54        default_mask,
55        has_mask,
56        masks: image.mask_names(),
57        regions: image.region_names(),
58        restoring_beam: info
59            .beam_set
60            .single_beam()
61            .and_then(|beam| format_beam(beam, image.restoring_beam_record()))
62            .map(Box::new),
63        has_multiple_beams: info.beam_set.is_multi(),
64        per_plane_beams: per_plane_beam_summary(image, &info.beam_set)?.map(Box::new),
65        extrema: None,
66        axes: axis_summaries(image.coordinates(), image.shape(), false),
67        world_axes: axis_summaries(image.coordinates(), image.shape(), true),
68    })
69}

83pub(super) fn summarize_beam_set(
84    coordinates: &CoordinateSystem,
85    shape: &[usize],
86    beam_set: &ImageBeamSet,
87    beam_record: Option<&RecordValue>,
88) -> Result<Option<PerPlaneBeamSummary>, ImageError> {
89    if beam_set.single_beam().is_some_and(|beam| beam.is_null()) {
90        return Err(ImageError::InvalidMetadata(
91            "Invalid per plane beam found".into(),
92        ));
93    }
94    if !beam_set.is_multi() {
95        return Ok(None);
96    }
97    let channels = beam_set.n_channels();
98    let stokes = beam_set.n_stokes();
99    let mut image_channels = 1;
100    let mut image_stokes = 1;
101    for index in 0..coordinates.n_coordinates() {
102        let coordinate = coordinates.coordinate(index);
103        let extent = || {
104            coordinates
105                .coordinate_pixel_axes(index)
106                .and_then(|axes| axes.into_iter().flatten().next())
107                .and_then(|axis| shape.get(axis).copied())
108                .unwrap_or(1)
109        };
110        match coordinate.coordinate_type() {
111            CoordinateType::Spectral => image_channels = extent(),
112            CoordinateType::Stokes => image_stokes = extent(),
113            _ => {}
114        }
115    }
116    if channels != 1 && channels != image_channels {
117        return Err(ImageError::InvalidMetadata(
118            "Number of channels is not consistent".into(),
119        ));
120    }
121    if stokes != 1 && stokes != image_stokes {
122        return Err(ImageError::InvalidMetadata(
123            "Number of polarizations is not consistent".into(),
124        ));
125    }
126    let mut beams = Vec::with_capacity(channels * stokes);
127    let mut areas = Vec::with_capacity(channels * stokes);
128    let area_unit = casa_core::quanta::Unit::new("arcsec2")
129        .map_err(|error| ImageError::InvalidMetadata(error.to_string()))?;
130    for polarization in 0..stokes {
131        for channel in 0..channels {
132            let index = polarization * channels + channel;
133            let beam = *beam_set.beam(channel, polarization);
134            let raw = beam_record
135                .and_then(|record| record.get(&format!("*{index}")))
136                .and_then(|value| match value {
137                    Value::Record(record) => Some(record),
138                    _ => None,
139                });
140            let formatted = format_beam(beam, raw).ok_or_else(|| {
141                ImageError::InvalidMetadata("Invalid per plane beam found".into())
142            })?;
143            // CASA multiplies the stored quantities before converting their
144            // product to arcsec². That order affects ties in the median.
145            let major_unit = casa_core::quanta::Unit::new(&formatted.major.unit)
146                .map_err(|error| ImageError::InvalidMetadata(error.to_string()))?;
147            let minor_unit = casa_core::quanta::Unit::new(&formatted.minor.unit)
148                .map_err(|error| ImageError::InvalidMetadata(error.to_string()))?;
149            let ratio =
150                area_unit.val().factor / (major_unit.val().factor * minor_unit.val().factor);
151            let area = std::f64::consts::PI / (4.0 * 2.0_f64.ln())
152                * (formatted.major.value * formatted.minor.value / ratio);
153            beams.push(formatted);
154            areas.push((index, area));
155        }
156    }
157    areas.sort_by(|left, right| left.1.total_cmp(&right.1));
158    let median_index = areas[areas.len() / 2].0;
159    Ok(Some(PerPlaneBeamSummary {
160        channels,
161        stokes,
162        beams,
163        median_index,
164    }))
165}

167pub(super) fn spectral_header<T: ImagePixel>(
168    image: &PagedImage<T>,
169) -> Result<Option<SpectralHeaderSummary>, ImageError> {
170    let Some(spectral) = image.first_spectral_coordinate()? else {
171        return Ok(None);
172    };
173    Ok(Some(spectral_summary(&spectral)))
174}

176pub(super) fn spectral_summary(
177    spectral: &casa_coordinates::SpectralCoordinate,
178) -> SpectralHeaderSummary {
179    let unit = spectral
180        .axis_units()
181        .into_iter()
182        .next()
183        .unwrap_or_else(|| "Hz".into());
184    // The stored record's rest frequencies are already in its axis unit.
185    SpectralHeaderSummary {
186        reference_frame: if spectral.has_undefined_frame() {
187            "Undefined".into()
188        } else {
189            spectral.frequency_ref().as_str().into()
190        },
191        unit,
192        rest_frequency: spectral.rest_frequency(),
193        rest_frequencies: spectral.rest_frequencies().to_vec(),
194    }
195}

197pub(super) fn format_beam(
198    beam: GaussianBeam,
199    record: Option<&RecordValue>,
200) -> Option<RestoringBeamSummary> {
201    if beam.is_null() {
202        return None;
203    }
204    let major_arcsec = beam.major_in("arcsec").ok()?;
205    let minor_arcsec = beam.minor_in("arcsec").ok()?;
206    let position_angle_deg = beam.position_angle_in("deg").ok()?;
207    let fallback = |value, unit: &str| BeamQuantity {
208        value,
209        unit: unit.into(),
210    };
211    let major = record
212        .and_then(|r| beam_quantity(r, "major"))
213        .unwrap_or_else(|| fallback(major_arcsec, "arcsec"));
214    let minor = record
215        .and_then(|r| beam_quantity(r, "minor"))
216        .unwrap_or_else(|| fallback(minor_arcsec, "arcsec"));
217    let position_angle = record
218        .and_then(|r| beam_quantity(r, "positionangle"))
219        .unwrap_or_else(|| fallback(position_angle_deg, "deg"));
220    let quantity =
221        casa_core::quanta::Quantity::new(position_angle.value, &position_angle.unit).ok()?;
222    let wrapped = casa_images::wrap_position_angle(&quantity).ok()?;
223    let wrapped_position_angle = BeamQuantity {
224        value: wrapped.value(),
225        unit: wrapped.unit().name().into(),
226    };
227    Some(RestoringBeamSummary {
228        major_arcsec,
229        minor_arcsec,
230        position_angle_deg,
231        major,
232        minor,
233        position_angle,
234        wrapped_position_angle,
235    })
236}

238pub(super) fn beam_quantity(record: &RecordValue, key: &str) -> Option<BeamQuantity> {
239    let Value::Record(quantity) = record.get(key)? else {
240        return None;
241    };
242    let quantity = casa_images::quantity_from_record(quantity).ok()?;
243    Some(BeamQuantity {
244        value: quantity.value(),
245        unit: casa_core::quanta::unit::normalize_casa_unit_name(quantity.unit().name()),
246    })
247}

249pub(super) fn axis_summaries(
250    coordinates: &CoordinateSystem,
251    shape: &[usize],
252    world: bool,
253) -> Vec<ImageAxisSummary> {
254    let mut summaries = Vec::new();
255    for coord_idx in 0..coordinates.n_coordinates() {
256        let coord = coordinates.coordinate(coord_idx);
257        let mut refs = coord.reference_value();
258        let refpix = coord.reference_pixel();
259        let mut increments = coord.increment();
260        let names = coord.axis_names();
261        let mut units = coord.axis_units();
262        // A direction coordinate presents its values in the units it is
263        // stored in, as casacore's does, though it computes in radians.
264        if let casa_coordinates::CoordinateModel::Direction(direction) = coord {
265            (refs, increments) = direction.reference_and_increment_in_world_units();
266            units = direction.visible_world_units();
267        }
268        let pixel_axes = coordinates
269            .coordinate_pixel_axes(coord_idx)
270            .unwrap_or_default();
271        let world_axes;
272        let axes = if world {
273            world_axes = coordinates
274                .coordinate_world_axes(coord_idx)
275                .unwrap_or_default();
276            &world_axes
277        } else {
278            &pixel_axes
279        };
280        for (axis_in_coord, &axis) in axes.iter().enumerate() {
281            let Some(axis) = axis else {
282                continue;
283            };
284            summaries.push(ImageAxisSummary {
285                axis,
286                coordinate_type: format!("{:?}", coord.coordinate_type()),
287                name: names
288                    .get(axis_in_coord)
289                    .cloned()
290                    .unwrap_or_else(|| format!("Axis{axis}")),
291                shape: pixel_axes
292                    .get(axis_in_coord)
293                    .copied()
294                    .flatten()
295                    .and_then(|pixel| shape.get(pixel))
296                    .copied()
297                    .unwrap_or(0),
298                reference_value: *refs.get(axis_in_coord).unwrap_or(&0.0),
299                reference_pixel: *refpix.get(axis_in_coord).unwrap_or(&0.0),
300                increment: *increments.get(axis_in_coord).unwrap_or(&0.0),
301                unit: units.get(axis_in_coord).cloned().unwrap_or_default(),
302                stokes: match coord {
303                    casa_coordinates::CoordinateModel::Stokes(stokes) => {
304                        stokes.stokes().iter().map(ToString::to_string).collect()
305                    }
306                    _ => Vec::new(),
307                },
308            });
309        }
310    }
311    summaries.sort_unstable_by_key(|axis| axis.axis);
312    if world {
313        return summaries;
314    }
315    for (axis, &extent) in shape.iter().enumerate().skip(summaries.len()) {
316        summaries.push(ImageAxisSummary {
317            axis,
318            coordinate_type: "Linear".to_string(),
319            name: format!("Axis{axis}"),
320            shape: extent,
321            reference_value: 0.0,
322            reference_pixel: 0.0,
323            increment: 1.0,
324            unit: String::new(),
325            stokes: Vec::new(),
326        });
327    }
328    summaries
329}
