55fn slice(start: &[usize], shape: &[usize], stride: &[usize]) -> Vec<SliceInfoElem> {
56    start
57        .iter()
58        .zip(shape)
59        .zip(stride)
60        .map(|((&start, &count), &stride)| SliceInfoElem::Slice {
61            start: start as isize,
62            end: Some((start + (count - 1) * stride + 1) as isize),
63            step: stride as isize,
64        })
65        .collect()
66}

68fn selected_image_info(original: &ImageHandle, selection: &Selection) -> PyResult<RecordValue> {
69    let mut info = image_info(original).map_err(raised)?;
70    if info.image_type == casa_images::ImageType::Undefined {
71        info.image_type = casa_images::ImageType::Intensity;
72    }
73    let stored = image_info_record(original);
74    let mut record = stored.cloned().unwrap_or_else(|| info.to_record());
75    let beams = &info.beam_set;
76    if !beams.is_multi() {
77        return Ok(record);
78    }
79    let (spectral, stokes) = plane_axes(&original.coordinate_record())?;
80    for (axis, planes, diagnostic) in [
81        (
82            spectral,
83            beams.n_channels(),
84            "Number of channels is not consistent",
85        ),
86        (
87            stokes,
88            beams.n_stokes(),
89            "Number of polarizations is not consistent",
90        ),
91    ] {
92        if planes != 1 && planes != axis_length(original, axis)? {
93            return Err(raised(diagnostic));
94        }
95    }
96    let indices = |axis: Option<usize>, planes| match axis {
97        Some(axis) if planes > 1 => (0..selection.shape[axis])
98            .map(|index| selection.start[axis] + index * selection.stride[axis])
99            .collect::<Vec<_>>(),
100        _ => vec![0],
101    };
102    let channels = indices(spectral, beams.n_channels());
103    let polarizations = indices(stokes, beams.n_stokes());
104    if channels.len() == beams.n_channels() && polarizations.len() == beams.n_stokes() {
105        return Ok(record);
106    }
107    subset_beam_record(&mut record, stored, beams, &channels, &polarizations).map_err(raised)?;
108    Ok(record)
109}

111impl CopyImage<'_> {
112    fn copy_pixels<T: ImagePixel>(
113        selection: &Selection,
114        coordinates: CoordinateSystem,
115        path: Option<&Path>,
116        expression_mask: Option<&ArrayD<bool>>,
117        region: &ImageRegionSection,
118        mut read: impl FnMut(&[usize], &[usize], &[usize]) -> PyResult<PixelSection<T>>,
119    ) -> PyResult<Storage<T>> {
120        let shape = &selection.shape;
121        let output_shape = selection.projected(shape);
122        let mut storage = if let Some(path) = path {
123            let tiled = TiledShape::new(output_shape.clone(), T::ELEM_SIZE).map_err(raised)?;
124            let tile = tiled.tile_shape().to_vec();
125            let bytes = 2 * tile.iter().product::<usize>() * T::ELEM_SIZE;
126            Storage::Paged(Box::new(
127                PagedImage::create_with_tile_shape_and_cache(
128                    output_shape,
129                    tile,
130                    coordinates,
131                    path,
132                    bytes,
133                )
134                .map_err(raised)?,
135            ))
136        } else {
137            Storage::create(output_shape, coordinates, None, None).map_err(raised)?
138        };
139        let total = shape
140            .iter()
141            .try_fold(1usize, |size, &axis| size.checked_mul(axis))
142            .ok_or_else(|| raised("Image size is too large"))?;
143        let mut axis = 0;
144        let mut prefix = 1;
145        let quota = 1_048_576;
146        while axis + 1 < shape.len() && shape[axis] <= quota / prefix {
147            prefix *= shape[axis];
148            axis += 1;
149        }
150        let mut block_shape = vec![1; shape.len()];
151        block_shape[..axis].copy_from_slice(&shape[..axis]);
152        let mut offset = 0;
153        let mut wrote_mask = false;
154        while offset < total {
155            let mut position = offset;
156            let start = shape
157                .iter()
158                .map(|length| {
159                    let index = position % length;
160                    position /= length;
161                    index
162                })
163                .collect::<Vec<_>>();
164            block_shape[axis] = (shape[axis] - start[axis]).min(quota / prefix);
165            let source = start
166                .iter()
167                .zip(&selection.start)
168                .zip(&selection.stride)
169                .map(|((&index, &base), &step)| base + index * step)
170                .collect::<Vec<_>>();
171            let (pixels, mask) = read(&source, &block_shape, &selection.stride)?;
172            let mut mask = and_mask(
173                mask,
174                region.mask_window(&start, &block_shape).map_err(raised)?,
175            );
176            let output_start = selection.projected(&start);
177            let pixels = selection.squeeze(pixels);
178            with_storage!(&mut storage, image => image.put_slice(&pixels, &output_start))
179                .map_err(raised)?;
180            let expression = expression_mask
181                .map(|mask| mask.slice(slice(&source, &block_shape, &selection.stride).as_slice()));
182            if let Some(passes) = expression {
183                if let Some(pixels_valid) = mask.as_mut() {
184                    Zip::from(pixels_valid)
185                        .and(passes)
186                        .for_each(|pixel, passes| *pixel &= *passes);
187                } else {
188                    mask = Some(passes.to_owned());
189                }
190            }
191            if let Some(mask) = mask
192                && mask.iter().any(|value| !value)
193            {
194                let mask = selection.squeeze(mask);
195                if !wrote_mask {
196                    with_storage!(&mut storage, image => image.create_filled_mask("mask0", true))
197                        .map_err(raised)?;
198                    wrote_mask = true;
199                }
200                with_storage!(&mut storage, image => image.put_mask_slice("mask0", &mask, &output_start))
201                    .map_err(raised)?;
202            }
203            offset += block_shape[axis] * prefix;
204        }
205        if wrote_mask {
206            with_storage!(&mut storage, image => image.set_default_mask("mask0"))
207                .map_err(raised)?;
208        }
209        Ok(storage)
210    }

212    fn build(
213        &self,
214        original: &ImageHandle,
215        path: Option<&Path>,
216        add_history: bool,
217        origin: Option<&TextOrigin>,
218    ) -> PyResult<ImageHandle> {
219        if original.image_type_name() == "MIRIADImage"
220            && original.coordinates().n_pixel_axes() != original.shape().len()
221        {
222            return Err(raised(
223                "Failed AlwaysAssert setCoordinateInfo (coordinateInfo)",
224            ));
225        }
226        let section = match &self.region {
227            CopyRegion::Record(region) => {
228                let region = (!region.is_empty()).then_some(SubimageRegion::Record(region));
229                resolve_region_section(original, region, false)?
230            }
231            CopyRegion::Text(text) if text.is_empty() => {
232                resolve_region_section(original, None, false)?
233            }
234            CopyRegion::Text(text) => {
235                let origin = origin.expect("text origin resolved before copying");
236                let coordinates = origin
237                    .coordinates
238                    .as_ref()
239                    .map(|(coords, shape)| (coords, shape.as_slice()));
240                let table = origin.table.as_deref();
241                if let ImageHandle::Float(storage) = original {
242                    storage.with_float_view(|image| {
243                        ImageRegionSection::resolve_copy_text(image, text, coordinates, table)
244                    })
245                } else {
246                    with_pixels!(original, storage => with_storage!(storage, image =>
247                        ImageRegionSection::resolve_copy_text(image.as_ref(), text, coordinates, table)))
248                }
249                .map_err(raised)?
250            }
251        };
252        let mut selection = Selection::new(&section);
253        if self.dropdeg {
254            selection.drop_degenerate()?;
255        }
256        let coordinates = original.coordinates();
257        let info = selected_image_info(original, &selection)?;
258        let record =
259            selection.coordinates(original.coordinate_record(), coordinates, original.shape())?;
260        let coordinates = CoordinateSystem::from_record(&record).map_err(raised)?;
261        let mask = if self.mask.is_empty() {
262            None
263        } else {
264            let target =
265                (self.overwrite && !self.outfile.is_empty()).then(|| Path::new(self.outfile));
266            Some(super::saved_expression::evaluate_mask(
267                &self.mask,
268                original.shape(),
269                target,
270            )?)
271        };
272        macro_rules! copy_native {
273            ($storage:expr, $variant:ident) => {
274                ImageHandle::$variant(with_storage!($storage, image => {
275                    Self::copy_pixels(&selection, coordinates, path, mask.as_ref(), &section, |start, shape, stride| {
276                        let pixels = image.get_slice_with_stride(start, shape, stride).map_err(raised)?;
277                        let mask = ImageInterface::get_mask_slice(image.as_ref(), start, shape, stride).map_err(raised)?;
278                        Ok((pixels, mask))
279                    })
280                })?)
281            };
282        }
283        let mut copy = if let Some(fits) = original.fits() {
284            ImageHandle::Float(Self::copy_pixels(
285                &selection,
286                coordinates,
287                path,
288                mask.as_ref(),
289                &section,
290                |start, shape, stride| {
291                    let pixels = fits.read(start, shape, stride)?;
292                    let mask = pixels.mapv(|value| !value.is_nan());
293                    Ok((pixels, Some(mask)))
294                },
295            )?)
296        } else {
297            match original {
298                ImageHandle::Float(storage) => copy_native!(storage, Float),
299                ImageHandle::Double(storage) => copy_native!(storage, Double),
300                ImageHandle::Complex(storage) => copy_native!(storage, Complex),
301                ImageHandle::DComplex(storage) => copy_native!(storage, DComplex),
302            }
303        };
304        let units =
305            with_pixels!(original, storage => with_storage!(storage, image => image.units()));
306        let misc =
307            with_pixels!(original, storage => with_storage!(storage, image => image.misc_info()));
308        let history = with_pixels!(original, storage => match storage {
309            Storage::Paged(image) => image_history_rows(image.name().expect("native image has a path")).map_err(raised)?,
310            Storage::Temp { image, .. } => image.history_rows().map_err(raised)?.into_owned(),
311        });
312        with_pixels!(&mut copy, storage => with_storage!(storage, image => (|| {
313            image.set_units(units)?;
314            image.set_image_info_record(info)?;
315            image.set_misc_info(misc)?;
316            image.append_history_entries(&history)
317        })()))
318        .map_err(raised)?;
319        copy.set_coordinate_record(record)?;
320        if add_history {
321            let input = if self.method == "fromimage" {
322                format!(
323                    "outfile={:?}, infile={:?}, region={}, mask={}",
324                    self.outfile, self.infile, self.region_history, self.mask_history
325                )
326            } else {
327                format!(
328                    "infile={:?}, outfile={:?}, region={}, vmask={}",
329                    self.infile, self.outfile, self.region_history, self.mask_history
330                )
331            };
332            copy.record_history(
333                self.method,
334                format!(
335                    "ia.{}({input}, dropdeg={}, overwrite={})",
336                    self.method, self.dropdeg, self.overwrite
337                ),
338            )?;
339        }
340        copy.save().map_err(raised)?;
341        Ok(copy)
342    }

344    fn text_origin(
345        &self,
346        attached: Option<&super::attachment::AttachedImage>,
347    ) -> PyResult<Option<TextOrigin>> {
348        let CopyRegion::Text(text) = &self.region else {
349            return Ok(None);
350        };
351        let use_attached_coordinates = self.method == "fromimage" || self.infile.is_empty();
352        if !use_attached_coordinates && text.is_empty() {
353            return Ok(None);
354        }
355        let attached = attached.ok_or_else(|| {
356            if use_attached_coordinates {
357                raised("No attached image. Cannot use a string value for region")
358            } else {
359                raised("Logic error at File: src/tools/image/image_cmpt.cc, line 4055")
360            }
361        })?;
362        if text.is_empty() {
363            return Ok(None);
364        }
365        let image = attached.lock();
366        Ok(Some(TextOrigin {
367            coordinates: use_attached_coordinates
368                .then(|| (image.coordinates().clone(), image.shape().to_vec())),
369            table: image.path().map(Path::to_path_buf),
370        }))
371    }

373    fn run(
374        &self,
375        expanded: PyResult<Cow<'_, str>>,
376        history: bool,
377        attached: Option<&super::attachment::AttachedImage>,
378    ) -> PyResult<super::attachment::AttachedImage> {
379        let use_attached_coordinates = self.method == "fromimage" || self.infile.is_empty();
380        let (original, origin) = if use_attached_coordinates {
381            let origin = self.text_origin(attached)?;
382            (open_image(self.infile, expanded?.as_ref())?, origin)
383        } else {
384            let original = open_image(self.infile, expanded?.as_ref())?;
385            if matches!(&self.region, CopyRegion::Text(_))
386                && !matches!(&*original.lock(), ImageHandle::Float(_))
387            {
388                return Err(raised(format!(
389                    "Logic Error: {} has a different data type than the data type of the requested object",
390                    self.infile
391                )));
392            }
393            (original, self.text_origin(attached)?)
394        };
395        if self.overwrite
396            && !self.outfile.is_empty()
397            && let Ok(target) = Path::new(self.outfile).canonicalize()
398        {
399            let source = original
400                .lock()
401                .path()
402                .ok_or_else(|| raised("The input image has no file name"))?
403                .canonicalize()
404                .map_err(raised)?;
405            if target != source && (target.starts_with(&source) || source.starts_with(&target)) {
406                return Err(raised(
407                    "Cannot overwrite an output that contains the source image or is part of it",
408                ));
409            }
410            if original
411                .operand_paths()
412                .iter()
413                .any(|path| target.starts_with(path) || path.starts_with(&target))
414            {
415                return Err(raised(
416                    "Cannot overwrite an output that overlaps a saved image expression operand",
417                ));
418            }
419        }
420        let mut live_coordinates = None;
421        let mut output = publish_image(self.outfile, self.overwrite, |path| {
422            let mut image = original.lock();
423            original.require_lock(&mut image, casa_tables::LockType::Read)?;
424            let copied = self.build(&image, path, history, origin.as_ref())?;
425            image.release_requested_lock()?;
426            if copied
427                .coordinates()
428                .find_coordinate(CoordinateType::Tabular)
429                .is_some()
430            {
431                live_coordinates = AttachedImage::capture_live_coordinates(&copied)?;
432            }
433            Ok(copied)
434        })?;
435        if let Some((record, system)) = live_coordinates {
436            output.keep_live_coordinates(record, system);
437        }
438        Ok(output)
439    }
440}

442fn copy_region_argument(
443    py: Python<'_>,
444    raw: Option<&Bound<'_, PyAny>>,
445    default_history: &str,
446) -> PyResult<(CopyRegion, String)> {
447    let empty_record = || CopyRegion::Record(RecordValue::default());
448    let Some(raw) = raw else {
449        return Ok((empty_record(), default_history.into()));
450    };
451    if let Ok(dict) = raw.cast::<PyDict>() {
452        return Ok((
453            CopyRegion::Record(casa_record(py, Some(dict))?),
454            "{...}".into(),
455        ));
456    }
457    if let Ok(text) = raw.cast::<PyString>() {
458        let text = text.to_str()?.to_owned();
459        let history = format!("{text:?}");
460        return Ok((CopyRegion::Text(text), history));
461    }
462    if is_boolean_sequence(raw)? {
463        let values = if let Ok(array) = raw.cast::<PyArrayDyn<bool>>() {
464            array
465                .readonly()
466                .as_array()
467                .reversed_axes()
468                .iter()
469                .copied()
470                .collect()
471        } else {
472            raw.extract::<Vec<bool>>()?
473        };
474        let history = values
475            .into_iter()
476            .map(|value| value.to_string())
477            .collect::<Vec<_>>()
478            .join(",");
479        return Ok((empty_record(), format!("[{history}]")));
480    }
481    if raw.extract::<Variant>()?.is_empty_sequence() {
482        return Ok((empty_record(), "[]".into()));
483    }
484    Err(raised("Unsupported type for region"))
485}
486
487fn mask_argument(raw: Option<&Bound<'_, PyAny>>) -> PyResult<(String, String)> {
488    let Some(raw) = raw else {
489        return Ok((String::new(), "\"\"".into()));
490    };
491    if let Ok(text) = raw.cast::<PyString>() {
492        let text = text.to_str()?.to_owned();
493        let history = format!("{text:?}");
494        return Ok((text, history));
495    }
496    let value: Variant = raw.extract()?;
497    if value.is_empty_sequence() {
498        return Ok((String::new(), "[]".into()));
499    }
500    Err(raised("Mask is not understood, try a valid LEL string"))
501}

503#[pymethods]
504#[allow(clippy::too_many_arguments)]
505impl PyImageTool {
506    #[pyo3(signature = (outfile="", infile="", region=None, mask=None, dropdeg=false, overwrite=false), text_signature = "($self, outfile='', infile='', region=[], mask='', dropdeg=False, overwrite=False)")]
507    fn fromimage(
508        &self,
509        py: Python<'_>,
510        outfile: &str,
511        infile: &str,
512        region: Option<&Bound<'_, PyAny>>,
513        mask: Option<&Bound<'_, PyAny>>,
514        dropdeg: bool,
515        overwrite: bool,
516    ) -> PyResult<bool> {
517        let method = "fromimage";
518        let (region, region_history) = copy_region_argument(py, region, "[]")?;
519        let (mask, mask_history) = mask_argument(mask)?;
520        let expanded = resolve_image_path(py, infile);
521        let request = CopyImage {
522            method,
523            infile,
524            outfile,
525            region,
526            region_history,
527            mask,
528            mask_history,
529            dropdeg,
530            overwrite,
531        };
532        py.detach(|| {
533            let mut state = self.state();
534            state.image = Some(request.run(expanded, state.history, state.image.as_ref())?);
535            Ok(true)
536        })
537    }
538
539    #[pyo3(signature = (infile="", outfile="", region=None, mask=None, dropdeg=false, overwrite=false), text_signature = "($self, infile='', outfile='', region={}, mask='', dropdeg=False, overwrite=False)")]
540    fn newimagefromimage(
541        &self,
542        py: Python<'_>,
543        infile: &str,
544        outfile: &str,
545        region: Option<&Bound<'_, PyAny>>,
546        mask: Option<&Bound<'_, PyAny>>,
547        dropdeg: bool,
548        overwrite: bool,
549    ) -> PyResult<Py<Self>> {
550        let method = "newimagefromimage";
551        let (region, region_history) = copy_region_argument(py, region, "{...}")?;
552        let (mask, mask_history) = mask_argument(mask)?;
553        let expanded = resolve_image_path(py, infile);
554        let request = CopyImage {
555            method,
556            infile,
557            outfile,
558            region,
559            region_history,
560            mask,
561            mask_history,
562            dropdeg,
563            overwrite,
564        };
565        let output = py.detach(|| {
566            let state = self.state();
567            request.run(expanded, state.history, state.image.as_ref())
568        })?;
569        Py::new(py, Self::attached(output))
570    }
