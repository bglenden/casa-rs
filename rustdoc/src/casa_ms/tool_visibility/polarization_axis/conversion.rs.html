2//! CASA-normalized complex polarization transforms for the visibility tool.
3
4use casa_coordinates::StokesType;
5use num_complex::{Complex32, Complex64};
6
7use super::{MsResult, input};
8
9pub(super) struct Conversion {
10    input_count: usize,
11    outputs: Result<Vec<Product>, &'static str>,
12    stokes: Option<[LinearCombination; 4]>,
13}
14
15struct Product {
16    value: ProductValue,
17    flags: Vec<usize>,
18    noise: Vec<(usize, f32)>,
19}
20
21enum ProductValue {
22    Linear(LinearCombination),
23    Derived(StokesType),
24}
25
26type LinearCombination = Vec<(usize, Complex32)>;
27
28impl Conversion {
29    pub(super) fn new(codes: &[i32], outputs: &[StokesType]) -> Self {
30        let mut conversion = Self {
31            input_count: codes.len(),
32            outputs: Ok(Vec::with_capacity(outputs.len())),
33            stokes: None,
34        };
35        let first = codes.first().copied().unwrap_or(0);
36        if !(1..=20).contains(&first) {
37            conversion.outputs = Err("Stokes conversion input frame is unsupported");
38            return conversion;
39        }
40        if codes
41            .iter()
42            .any(|code| !(1..=20).contains(code) || (code - 1) / 4 != (first - 1) / 4)
43        {
44            conversion.outputs = Err("Stokes conversion cannot mix input frames");
45            return conversion;
46        }
47        let basis = codes
48            .iter()
49            .copied()
50            .map(stokes_contribution)
51            .collect::<Vec<_>>();
52        let plans = conversion.outputs.as_mut().expect("valid basis");
53        for &output in outputs {
54            if (1..=20).contains(&output.code()) {
55                plans.push(Product::linear(coefficients(&basis, |stokes| {
56                    project_product(output.code(), stokes)
57                })));
58            } else if (21..=27).contains(&output.code()) {
59                // CASA recognizes these generic and single-dish products but
60                // leaves their Complex output at zero when converting.
61                plans.push(Product::linear(Vec::new()));
62            } else if matches!(
63                output,
64                StokesType::Ptotal
65                    | StokesType::Plinear
66                    | StokesType::PFtotal
67                    | StokesType::PFlinear
68                    | StokesType::Pangle
69            ) {
70                let iquv = conversion.stokes.get_or_insert_with(|| {
71                    std::array::from_fn(|index| coefficients(&basis, |stokes| stokes[index]))
72                });
73                plans.push(Product::derived(output, iquv, codes.len()));
74            } else {
75                conversion.outputs = Err("Stokes conversion output type is unsupported");
76                break;
77            }
78        }
79        conversion
80    }
81
82    pub(super) fn append(
83        &self,
84        samples: &[Complex32],
85        output: &mut Vec<Complex32>,
86    ) -> MsResult<()> {
87        let products = self.products(samples.len())?;
88        let iquv = self
89            .stokes
90            .as_ref()
91            .map(|terms| terms.each_ref().map(|term| sum(term, samples)));
92        for product in products {
93            let value = match &product.value {
94                ProductValue::Linear(terms) => sum(terms, samples),
95                ProductValue::Derived(stokes) => {
96                    derive(*stokes, iquv.expect("derived basis exists"))
97                }
98            };
99            output.push(value);
100        }
101        Ok(())
102    }
103
104    pub(super) fn append_flags(&self, samples: &[bool], output: &mut Vec<bool>) -> MsResult<()> {
105        for product in self.products(samples.len())? {
106            output.push(product.flags.iter().any(|&index| samples[index]));
107        }
108        Ok(())
109    }
110
111    pub(super) fn append_weights(&self, samples: &[f32], output: &mut Vec<f32>) -> MsResult<()> {
112        for product in self.products(samples.len())? {
113            let variance = product
114                .noise
115                .iter()
116                .try_fold(0.0, |variance, &(index, square)| {
117                    let weight = samples[index];
118                    (weight.is_finite() && weight > 0.0).then_some(variance + square / weight)
119                });
120            output.push(
121                variance
122                    .filter(|&value| value > 0.0)
123                    .map_or(0.0, f32::recip),
124            );
125        }
126        Ok(())
127    }
128
129    fn products(&self, count: usize) -> MsResult<&[Product]> {
130        if count != self.input_count {
131            return Err(input(
132                "visibility correlation count does not match its polarization table",
133            ));
134        }
135        self.outputs.as_deref().map_err(|&error| input(error))
136    }
137}

139impl Product {
140    fn linear(terms: LinearCombination) -> Self {
141        Self {
142            flags: terms.iter().map(|&(index, _)| index).collect(),
143            noise: terms
144                .iter()
145                .map(|&(index, term)| (index, term.norm_sqr()))
146                .collect(),
147            value: ProductValue::Linear(terms),
148        }
149    }
150
151    fn derived(kind: StokesType, iquv: &[LinearCombination; 4], input_count: usize) -> Self {
152        let terms = match kind {
153            StokesType::Ptotal | StokesType::PFtotal => &iquv[1..4],
154            _ => &iquv[1..3],
155        };
156        let magnitude = |index| {
157            terms
158                .iter()
159                .flat_map(|group| group.iter())
160                .filter_map(|&(sample, factor)| (sample == index).then_some(factor.norm()))
161                .sum::<f32>()
162                / terms.len() as f32
163        };
164        let mut flags = Vec::new();
165        let mut noise = Vec::new();
166        for index in 0..input_count {
167            let weight = magnitude(index);
168            if weight > 0.0 {
169                noise.push((index, weight * weight));
170            }
171            if weight > 0.0
172                || kind == StokesType::PFtotal
173                || (kind == StokesType::PFlinear
174                    && iquv[0].iter().any(|&(sample, _)| sample == index))
175            {
176                flags.push(index);
177            }
178        }
179        Self {
180            value: ProductValue::Derived(kind),
181            flags,
182            noise,
183        }
184    }
185}

187fn coefficients(
188    basis: &[[Complex64; 4]],
189    mut coefficient: impl FnMut([Complex64; 4]) -> Complex64,
190) -> LinearCombination {
191    basis
192        .iter()
193        .copied()
194        .enumerate()
195        .filter_map(|(index, stokes)| {
196            let coefficient = coefficient(stokes);
197            (coefficient.norm_sqr() > 1e-24).then_some((
198                index,
199                Complex32::new(coefficient.re as f32, coefficient.im as f32),
200            ))
201        })
202        .collect()
203}
204
205fn sum(terms: &LinearCombination, samples: &[Complex32]) -> Complex32 {
206    terms
207        .iter()
208        .map(|&(index, scalar)| scalar * samples[index])
209        .sum()
210}
211
212fn derive(kind: StokesType, [i, q, u, v]: [Complex32; 4]) -> Complex32 {
213    let linear = q.norm_sqr() + u.norm_sqr();
214    let value = match kind {
215        StokesType::Ptotal => (linear + v.norm_sqr()).sqrt(),
216        StokesType::Plinear => linear.sqrt(),
217        StokesType::PFtotal => (linear + v.norm_sqr()).sqrt() / i.norm(),
218        StokesType::PFlinear => linear.sqrt() / i.norm(),
219        StokesType::Pangle => 0.5 * u.re.atan2(q.re),
220        _ => unreachable!("only derived outputs are compiled"),
221    };
222    Complex32::new(value, 0.0)
223}

225/// Treat one stored product as nonzero and reverse its receptor basis rotation.
226fn stokes_contribution(code: i32) -> [Complex64; 4] {
227    if code <= 4 {
228        let mut result = [Complex64::ZERO; 4];
229        result[code as usize - 1] = Complex64::ONE;
230        return result;
231    }
232    let (left, right, scale) = receptors(code);
233    let xx = left[0].conj() * right[0] * scale;
234    let xy = left[0].conj() * right[1] * scale;
235    let yx = left[1].conj() * right[0] * scale;
236    let yy = left[1].conj() * right[1] * scale;
237    [xx + yy, xx - yy, xy + yx, -Complex64::I * (xy - yx)]
238}
239
240fn project_product(code: i32, [i, q, u, v]: [Complex64; 4]) -> Complex64 {
241    if code <= 4 {
242        return [i, q, u, v][code as usize - 1];
243    }
244    let (left, right, scale) = receptors(code);
245    let x = (i + q) * right[0].conj() + (u + Complex64::I * v) * right[1].conj();
246    let y = (u - Complex64::I * v) * right[0].conj() + (i - q) * right[1].conj();
247    (left[0] * x + left[1] * y) * (0.5 / scale)
248}

250/// Jones vectors for the left and right receptors in the linear basis. CASA
251/// normalizes same-basis products by 1/2 and mixed products by sqrt(2)/4.
252fn receptors(code: i32) -> ([Complex64; 2], [Complex64; 2], f64) {
253    use std::f64::consts::FRAC_1_SQRT_2;
254    let linear = [
255        [Complex64::ONE, Complex64::ZERO],
256        [Complex64::ZERO, Complex64::ONE],
257    ];
258    let circular = [
259        [Complex64::ONE * FRAC_1_SQRT_2, Complex64::I * FRAC_1_SQRT_2],
260        [
261            Complex64::ONE * FRAC_1_SQRT_2,
262            -Complex64::I * FRAC_1_SQRT_2,
263        ],
264    ];
265    let index = (code - 5) as usize;
266    let (left, right, scale) = match index / 4 {
267        0 => (circular, circular, 0.5),
268        1 => (linear, linear, 0.5),
269        2 => (circular, linear, 0.5 * FRAC_1_SQRT_2),
270        3 => (linear, circular, 0.5 * FRAC_1_SQRT_2),
271        _ => unreachable!("polarization type validated before projection"),
272    };
273    (left[index % 4 / 2], right[index % 2], scale)
274}
