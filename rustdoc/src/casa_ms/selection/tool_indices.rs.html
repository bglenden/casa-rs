15/// The table identifiers and channel ranges recorded while parsing a CASA MS
16/// selection. These are the clause results, independently of the rows selected
17/// by applying the complete predicate.
18#[derive(Default)]
19pub struct MsSelectionIndices {
20    /// FIELD IDs, in the FIELD selector's canonical order.
21    pub field: Vec<i32>,
22    /// SCAN_NUMBERs, in their requested order, with repeats.
23    pub scan: Vec<i32>,
24    /// The requested identifiers present in OBSERVATION.
25    pub observation: Vec<i32>,
26    /// STATE IDs resolved from the intent.
27    pub state: Vec<i32>,
28    /// First-side antenna identifiers, signed for negated terms.
29    pub antenna1: Vec<i32>,
30    /// Second-side antenna identifiers, signed for negated pairs.
31    pub antenna2: Vec<i32>,
32    /// Baselines in the order the antenna grammar produced them. A supplied
33    /// antenna clause has two columns even when it names no pairs.
34    pub baselines: Option<Vec<[i32; 2]>>,
35    /// Spectral window identifiers in the order the SPW grammar emits.
36    pub spw: Vec<i32>,
37    /// Inclusive `[window, first channel, last channel, stride]` ranges.
38    pub channel: Vec<[i32; 4]>,
39    /// DATA_DESCRIPTION IDs matching the SPW clause.
40    pub spwdd: Vec<i32>,
41    /// DATA_DESCRIPTION IDs matching the polarization clause.
42    pub poldd: Vec<i32>,
43}
44
45impl MsSelectionIndices {
46    /// Resolve the data descriptions accepted by both applicable clauses.
47    pub fn descriptions(&self) -> Vec<i32> {
48        match (self.spwdd.is_empty(), self.poldd.is_empty()) {
49            (_, true) => self.spwdd.clone(),
50            (true, _) => self.poldd.clone(),
51            _ => {
52                let mut ids = self
53                    .poldd
54                    .iter()
55                    .copied()
56                    .filter(|id| self.spwdd.contains(id))
57                    .collect::<Vec<_>>();
58                ids.sort_unstable();
59                ids.dedup();
60                ids
61            }
62        }
63    }
64}

66/// The compiled row predicate of a MeasurementSet tool selection.
67pub struct ParsedMsToolSelection {
68    compiled: CompiledMsSelection,
69    rows_per_block: usize,
70    taql_is_row_predicate: bool,
71    constant_taql: Option<bool>,
72    correlations: Vec<ToolCorrelationSelection>,
73}

75/// Stored visibility correlations accepted by a CASA polarization clause.
76pub struct ToolCorrelationSelection {
77    /// DATA_DESCRIPTION row to which this projection applies.
78    pub description_id: i32,
79    /// Indices in the stored correlation axis, in requested order.
80    pub indices: Vec<usize>,
81    /// The CASA types in the projected correlation axis.
82    pub types: Vec<StokesType>,
83}
84
85impl ParsedMsToolSelection {
86    /// Whether an expression constrains the rows of the supplied view.
87    pub fn has_row_expression(&self) -> bool {
88        !self.compiled.where_clauses().is_empty()
89    }
90
91    /// A true predicate independent of rows retains the existing live or
92    /// saved view because it has no additional row restriction.
93    pub fn preserves_current_view(&self) -> bool {
94        self.constant_taql == Some(true)
95    }
96
97    /// Take the in-row correlation projections from this selection.
98    pub fn into_correlations(self) -> Vec<ToolCorrelationSelection> {
99        self.correlations
100    }
101
102    /// Apply the predicate and return matching indices in the supplied view.
103    ///
104    /// # Errors
105    /// A selection column or predicate cannot be evaluated.
106    pub fn select_rows(&self, ms: &MeasurementSet) -> MsResult<Vec<usize>> {
107        if !self.taql_is_row_predicate {
108            return Err(MsError::InvalidInput(
109                "TaQL row predicate is not a Boolean scalar".into(),
110            ));
111        }
112        if let Some(include) = self.constant_taql {
113            return Ok(if include {
114                (0..ms.row_count()).collect()
115            } else {
116                Vec::new()
117            });
118        }
119        self.compiled
120            .apply_internal(ms, Some(self.rows_per_block), false)
121    }
122}

124impl MeasurementSet {
125    /// Select rows whose stored TIME is a member of the supplied second values.
126    /// The public `ms.select` record uses exact comparisons for its `times` field.
127    ///
128    /// # Errors
129    /// Returns an error when the MAIN TIME column cannot be read as doubles.
130    pub fn select_ms_tool_times(&self, mut times: Vec<f64>) -> MsResult<Vec<usize>> {
131        times.retain(|value| !value.is_nan());
132        if times.is_empty() {
133            return Ok(Vec::new());
134        }
135        times.sort_unstable_by(f64::total_cmp);
136        let rows_per_block = rows_per_block_for_budget(default_selection_budget()?)
137            .map_err(|error| MsError::InvalidInput(error.to_string()))?;
138        let table = self.main_table();
139        let mut selected = Vec::new();
140        for start in (0..table.row_count()).step_by(rows_per_block) {
141            let end = start.saturating_add(rows_per_block).min(table.row_count());
142            let rows = (start..end).collect::<Vec<_>>();
143            let values = super::columns::load_f64_column(table, "TIME", &rows)?;
144            selected.extend(rows.into_iter().zip(values).filter_map(|(row, time)| {
145                let inside = !time.is_nan()
146                    && times
147                        .binary_search_by(|value| {
148                            value.partial_cmp(&time).expect("NaN times were removed")
149                        })
150                        .is_ok();
151                inside.then_some(row)
152            }));
153        }
154        Ok(selected)
155    }
156
157    /// Compile one CASA MS tool selection and update its auxiliary lists.
158    /// An omitted antenna, field, window, or polarization clause retains
159    /// its previous list. The compiled predicate describes only this request.
160    ///
161    /// # Errors
162    /// A supplied selector cannot be parsed or names invalid metadata.
163    pub fn parse_ms_tool_selection(
164        &self,
165        request: &MsSelection,
166        indices: &mut MsSelectionIndices,
167    ) -> MsResult<ParsedMsToolSelection> {
168        let budget = default_selection_budget()?;
169        let rows_per_block = rows_per_block_for_budget(budget)
170            .map_err(|error| MsError::InvalidInput(error.to_string()))?;
171        indices.update_antenna_field_and_spw(self, request)?;
172        indices.scan.clear();
173        if let Some(text) = nonempty(request.scan.as_deref()) {
174            indices.scan = scan_indices(text, rows_per_block)?;
175        }
176        let mut rows = request.clone();
177        rows.correlation = None;
178        let mut correlations = Vec::new();
179        if let Some(text) = nonempty(request.correlation.as_deref()) {
180            correlations = polarization_projections(self, text)?;
181            indices.poldd = correlations
182                .iter()
183                .map(|plan| plan.description_id)
184                .collect();
185            if !indices.poldd.is_empty() {
186                rows.data_description = Some(int_list(&indices.poldd));
187            }
188        }
189        let taql = rows.msselect.take();
190        let (taql_type, constant_taql) =
191            if let Some(text) = nonempty(taql.as_deref().map(str::trim)) {
192                super::taql_selection::validate_predicate(
193                    self,
194                    &format!("SELECT * WHERE ({text})"),
195                    Some(rows_per_block),
196                )?
197            } else {
198                (RecordExpressionType::PossibleRowPredicate, None)
199            };
200        let taql_is_row_predicate = taql_type == RecordExpressionType::PossibleRowPredicate;
201        if (taql_type == RecordExpressionType::NonBoolean && rows.has_text_selection())
202            || (!taql_is_row_predicate && rows.timerange.is_some())
203        {
204            return Err(MsError::InvalidInput(
205                "TaQL expression cannot be combined with the other row selectors".into(),
206            ));
207        }
208        let constant_taql = if rows.has_text_selection() {
209            None
210        } else {
211            constant_taql
212        };
213        rows.msselect = taql;
214        let compiled =
215            rows.compile_with_row_grammar(self, RowGrammar::Transform, Some(rows_per_block))?;
216        indices.state.clone_from(&compiled.state_ids);
217        indices.observation.clear();
218        if let Some(text) = nonempty(request.observation.as_deref()) {
219            indices.observation = observation_indices(text, self.observation()?.row_count())?;
220        }
221        Ok(ParsedMsToolSelection {
222            compiled,
223            rows_per_block,
224            taql_is_row_predicate,
225            constant_taql,
226            correlations,
227        })
228    }
229}

231impl MsSelectionIndices {
232    fn update_antenna_field_and_spw(
233        &mut self,
234        ms: &MeasurementSet,
235        request: &MsSelection,
236    ) -> MsResult<()> {
237        if let Some(text) = nonempty(request.antenna.as_deref()) {
238            self.antenna1.clear();
239            self.antenna2.clear();
240            self.baselines = Some(Vec::new());
241            let AntennaSelectionIndices {
242                first,
243                second,
244                baselines,
245            } = ms.antenna_selection_indices(text)?;
246            self.antenna1 = first;
247            self.antenna2 = second;
248            self.baselines = Some(baselines);
249        }
250        if let Some(text) = nonempty(request.field.as_deref()) {
251            self.field.clear();
252            self.field = ms.resolve_field_selector(text)?;
253            if self.field.is_empty() {
254                return Err(MsError::InvalidInput(format!(
255                    "field selector {text:?} did not match any existing fields"
256                )));
257            }
258        }
259        if let Some(text) = nonempty(request.spw.as_deref()) {
260            self.spw.clear();
261            let spw = ms.resolve_spw_selector(text)?;
262            self.spw = spw.spw_ids().iter().map(|&value| value as i32).collect();
263            self.spwdd = spw
264                .data_description_ids()
265                .iter()
266                .map(|&value| value as i32)
267                .collect();
268            self.channel = spw
269                .channel_list()
270                .iter()
271                .map(|row| [row.spw_id as i32, row.start, row.stop, row.step.max(1)])
272                .collect();
273        }
274        Ok(())
275    }
276}

282/// CASA's auxiliary SCAN_NUMBER list uses a 1000-scan bound for an
283/// open-ended greater-than expression. The row predicate stays unbounded.
284const CASA_MAX_SCAN: i64 = 1000;
285
286fn scan_indices(text: &str, max_ids: usize) -> MsResult<Vec<i32>> {
287    let mut values = Vec::new();
288    for item in text.split(',').map(trim_row_id_space) {
289        let (first, last) = selector_interval(item, "scan")?;
290        let first = first.max(0);
291        let last = if last == i64::from(i32::MAX) && item.starts_with('>') {
292            CASA_MAX_SCAN + i64::from(!item.starts_with(">="))
293        } else {
294            last
295        };
296        if first > last {
297            continue;
298        }
299        let count = usize::try_from(last - first + 1)
300            .map_err(|_| MsError::InvalidInput("scan index list is too large".into()))?;
301        if count > max_ids.saturating_sub(values.len()) {
302            return Err(MsError::InvalidInput(
303                "scan index list exceeds the selection memory budget".into(),
304            ));
305        }
306        values.extend((first..=last).map(|value| value as i32));
307    }
308    Ok(values)
309}

311fn observation_indices(text: &str, table_rows: usize) -> MsResult<Vec<i32>> {
312    let intervals = text
313        .split(',')
314        .map(trim_row_id_space)
315        .map(|item| selector_interval(item, "observation"))
316        .collect::<MsResult<Vec<_>>>()?;
317    Ok((0..table_rows)
318        .filter(|&row| {
319            intervals
320                .iter()
321                .any(|&(first, last)| (first..=last).contains(&(row as i64)))
322        })
323        .map(|row| row as i32)
324        .collect())
325}

327fn selector_interval(text: &str, label: &str) -> MsResult<(i64, i64)> {
328    row_id_interval(text)
329        .ok_or_else(|| MsError::InvalidInput(format!("invalid {label} selector term {text:?}")))
330}

332/// CASA's polarization parser accepts semicolon-separated correlation terms.
333/// A term may name its own spectral windows before the colon.
334fn polarization_projections(
335    ms: &MeasurementSet,
336    text: &str,
337) -> MsResult<Vec<ToolCorrelationSelection>> {
338    let data = ms.data_description()?;
339    let polarization = ms.polarization()?;
340    let mut result = Vec::new();
341    let descriptions = (0..data.row_count())
342        .filter_map(|row| match data.flag_row(row) {
343            Ok(true) => None,
344            Ok(false) => Some(Ok(row)),
345            Err(error) => Some(Err(error)),
346        })
347        .map(|row| {
348            let row = row?;
349            Ok((
350                row,
351                data.spectral_window_id(row)?,
352                data.polarization_id(row)?,
353            ))
354        })
355        .collect::<MsResult<Vec<_>>>()?;
356    for term in text
357        .split(';')
358        .map(str::trim)
359        .filter(|term| !term.is_empty())
360    {
361        let fields = term
362            .split(':')
363            .map(str::trim)
364            .filter(|part| !part.is_empty())
365            .collect::<Vec<_>>();
366        let (window, correlation) = match fields.as_slice() {
367            [correlation] => ("*", *correlation),
368            [window, correlation] => (*window, *correlation),
369            _ => {
370                return Err(MsError::InvalidInput(format!(
371                    "invalid polarization term {term:?}"
372                )));
373            }
374        };
375        let codes = correlation
376            .split(',')
377            .map(str::trim)
378            .filter(|code| !code.is_empty())
379            .map(|code| {
380                StokesType::from_name(code)
381                    .filter(|stokes| *stokes != StokesType::Undefined)
382                    .ok_or_else(|| MsError::NoPolarizationMatch(correlation.into()))
383            })
384            .collect::<MsResult<Vec<_>>>()?;
385        if codes.is_empty() {
386            return Err(MsError::InvalidInput(
387                "polarization selector was empty".into(),
388            ));
389        }
390        let spectral_windows = ms.resolve_spw_selector(window)?;
391        let mut matches = Vec::new();
392        for pol_id in 0..polarization.row_count() {
393            let available = polarization.corr_type(pol_id)?;
394            let Some(positions) = codes
395                .iter()
396                .map(|code| available.iter().position(|value| *value == code.code()))
397                .collect::<Option<Vec<_>>>()
398            else {
399                continue;
400            };
401            for &window in spectral_windows.spw_ids() {
402                if let Some(&(row, _, _)) = descriptions
403                    .iter()
404                    .find(|&&(_, spw, pol)| spw == window as i32 && pol == pol_id as i32)
405                {
406                    matches.push((row as i32, positions.clone()));
407                }
408            }
409        }
410        if matches.is_empty() {
411            return Err(MsError::NoPolarizationMatch(correlation.into()));
412        }
413        for (description_id, positions) in matches {
414            append_correlations(&mut result, description_id, positions, &codes);
415        }
416    }
417    Ok(result)
418}
419
420fn append_correlations(
421    projections: &mut Vec<ToolCorrelationSelection>,
422    description_id: i32,
423    positions: Vec<usize>,
424    codes: &[StokesType],
425) {
426    let index = projections
427        .iter()
428        .position(|plan| plan.description_id == description_id)
429        .unwrap_or_else(|| {
430            projections.push(ToolCorrelationSelection {
431                description_id,
432                indices: Vec::new(),
433                types: Vec::new(),
434            });
435            projections.len() - 1
436        });
437    let plan = &mut projections[index];
438    for (position, &code) in positions.into_iter().zip(codes) {
439        if !plan.indices.contains(&position) {
440            plan.indices.push(position);
441            plan.types.push(code);
442        }
443    }
