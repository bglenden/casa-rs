2//! Typed accessor for frequency columns in the SPECTRAL_WINDOW subtable.
3//!
4//! Returns channel frequencies as [`MFrequency`] values with the correct
5//! reference frame, determined by the MEAS_FREQ_REF column.
6//!
7//! Cf. C++ `MSSpWindowColumns::chanFreqMeas()`.
8
9use casa_tables::Table;
10use casa_tables::table_measures::{
11    MeasRefDesc, MeasureType, TableMeasDesc, frequency_ref_for_table_code,
12};
13use casa_types::measures::frequency::{FrequencyRef, MFrequency};
14use casa_types::{ArrayValue, ScalarValue, Value};
15
16use crate::error::{MsError, MsResult};
17use crate::subtables::{get_array, get_f64, get_i32, get_string};

19/// Typed accessor for channel frequencies.
20pub struct ChanFreqColumn<'a> {
21    table: &'a Table,
22    channels_reference: FrequencyReference,
23    centre_reference: FrequencyReference,
24}
25
26enum FrequencyReference {
27    Missing,
28    Invalid,
29    Incomplete,
30    Present(TableMeasDesc),
31}
32
33impl FrequencyReference {
34    fn read(table: &Table, column: &str) -> Self {
35        if let Some(desc) = TableMeasDesc::reconstruct(table, column) {
36            Self::Present(desc)
37        } else if let Some(Value::Record(info)) = table
38            .column_keywords(column)
39            .and_then(|keywords| keywords.get("MEASINFO"))
40            && matches!(info.get("type"), Some(Value::Scalar(ScalarValue::String(t))) if t.eq_ignore_ascii_case("frequency"))
41            && info.get("Ref").is_none()
42            && info.get("VarRefCol").is_none()
43        {
44            // casacore gives a frequency measure with no Ref its code-zero
45            // default (REST); it does not consult MEAS_FREQ_REF in this case.
46            Self::Incomplete
47        } else if table
48            .column_keywords(column)
49            .is_some_and(|keywords| keywords.get("MEASINFO").is_some())
50        {
51            Self::Invalid
52        } else {
53            Self::Missing
54        }
55    }
56
57    fn or<'a>(&'a self, other: &'a Self) -> &'a Self {
58        if matches!(self, Self::Missing) {
59            other
60        } else {
61            self
62        }
63    }
64}
65
66impl<'a> ChanFreqColumn<'a> {
67    /// Create a channel frequency column accessor over the SPECTRAL_WINDOW subtable.
68    pub fn new(table: &'a Table) -> Self {
69        let channels_reference = FrequencyReference::read(table, "CHAN_FREQ");
70        let centre_reference = FrequencyReference::read(table, "REF_FREQUENCY");
71        Self {
72            table,
73            channels_reference,
74            centre_reference,
75        }
76    }
77
78    /// Read channel frequencies as [`MFrequency`] values for a spectral window row.
79    ///
80    /// Uses the MEAS_FREQ_REF column to determine the reference frame.
81    pub fn get_frequencies(&self, row: usize) -> MsResult<Vec<MFrequency>> {
82        let refer = self.frequency_reference(row)?;
83        let arr = get_array(self.table, row, "CHAN_FREQ")?;
84
85        match arr {
86            ArrayValue::Float64(a) => Ok(a.iter().map(|&hz| MFrequency::new(hz, refer)).collect()),
87            other => Err(MsError::ColumnTypeMismatch {
88                column: "CHAN_FREQ".to_string(),
89                table: "SPECTRAL_WINDOW".to_string(),
90                expected: "Float64 array".to_string(),
91                found: format!("{:?}", other.primitive_type()),
92            }),
93        }
94    }
95
96    /// Read the reference frequency as an [`MFrequency`].
97    pub fn get_ref_frequency(&self, row: usize) -> MsResult<MFrequency> {
98        let refer = self.reference_frequency_reference(row)?;
99        let hz = get_f64(self.table, row, "REF_FREQUENCY")?;
100        Ok(MFrequency::new(hz, refer))
101    }
102
103    /// Resolve the row's frequency frame using its table-specific measure
104    /// reference mapping, including codes inferred by casacore for old tables.
105    /// Without measure metadata, use the standard `MEAS_FREQ_REF` codes.
106    ///
107    /// # Errors
108    ///
109    /// The reference column or its measure mapping is invalid or unreadable.
110    pub fn frequency_reference(&self, row: usize) -> MsResult<FrequencyRef> {
111        self.resolve_reference(
112            row,
113            self.channels_reference.or(&self.centre_reference),
114            "CHAN_FREQ",
115        )
116    }
117
118    /// The visibility frame for a channel column whose frequency measure
119    /// descriptor lacks a reference. CASA's visibility iterator reads the
120    /// measurement-frame column in that case, while the measure accessor's
121    /// default remains REST.
122    pub(crate) fn visibility_frequency_reference(&self, row: usize) -> MsResult<FrequencyRef> {
123        if matches!(
124            self.channels_reference.or(&self.centre_reference),
125            FrequencyReference::Incomplete
126        ) {
127            let code = get_i32(self.table, row, "MEAS_FREQ_REF")?;
128            FrequencyRef::from_casacore_code(code)
129                .ok_or_else(|| invalid_code("MEAS_FREQ_REF", code))
130        } else {
131            self.frequency_reference(row)
132        }
133    }
134
135    /// Resolve the reference-frequency column's frame, which can be distinct
136    /// from the channel-frequency column's frame.
137    ///
138    /// # Errors
139    ///
140    /// The reference column or its measure mapping is invalid or unreadable.
141    pub fn reference_frequency_reference(&self, row: usize) -> MsResult<FrequencyRef> {
142        self.resolve_reference(
143            row,
144            self.centre_reference.or(&self.channels_reference),
145            "REF_FREQUENCY",
146        )
147    }
148
149    fn resolve_reference(
150        &self,
151        row: usize,
152        reference: &FrequencyReference,
153        column: &str,
154    ) -> MsResult<FrequencyRef> {
155        if matches!(reference, FrequencyReference::Invalid) {
156            return Err(MsError::Table(casa_tables::TableError::Storage(format!(
157                "invalid MEASINFO for SPECTRAL_WINDOW.{column}"
158            ))));
159        }
160        if matches!(reference, FrequencyReference::Incomplete) {
161            return Ok(FrequencyRef::REST);
162        }
163        if let FrequencyReference::Present(desc) = reference {
164            if desc.measure_type() != MeasureType::Frequency {
165                return Err(MsError::ColumnTypeMismatch {
166                    column: column.to_string(),
167                    table: "SPECTRAL_WINDOW".to_string(),
168                    expected: "frequency measure".to_string(),
169                    found: desc.measure_type().as_str().to_string(),
170                });
171            }
172            match desc.ref_desc() {
173                MeasRefDesc::Fixed { refer } => return Ok(refer.parse()?),
174                MeasRefDesc::VariableString { ref_column } => {
175                    return Ok(get_string(self.table, row, ref_column)?.parse()?);
176                }
177                MeasRefDesc::VariableInt {
178                    ref_column,
179                    tab_ref_types,
180                    tab_ref_codes,
181                } => {
182                    let code = get_i32(self.table, row, ref_column)?;
183                    return frequency_ref_for_table_code(tab_ref_types, tab_ref_codes, code)
184                        .ok_or_else(|| invalid_code(ref_column, code));
185                }
186            }
187        }
188        let code = get_i32(self.table, row, "MEAS_FREQ_REF")?;
189        FrequencyRef::from_casacore_code(code).ok_or_else(|| invalid_code("MEAS_FREQ_REF", code))
190    }
191}
192
193fn invalid_code(column: &str, code: i32) -> MsError {
194    MsError::InvalidMeasureCode {
195        table: "SPECTRAL_WINDOW".to_string(),
196        column: column.to_string(),
197        code,
198    }
199}

209    fn make_spw_table(freq_ref: i32, ref_freq: f64, chan_freqs: &[f64]) -> Table {
210        let schema =
211            build_table_schema(schema::spectral_window::REQUIRED_COLUMNS).expect("valid schema");
212        let mut table = Table::with_schema(schema);
213
214        let widths = vec![1.0e6; chan_freqs.len()];
215        let n = chan_freqs.len();
216
217        let make_arr = |v: &[f64]| -> ArrayValue {
218            ArrayValue::Float64(ArrayD::from_shape_vec(vec![v.len()], v.to_vec()).unwrap())
219        };
220
221        let row = RecordValue::new(vec![
222            RecordField::new("CHAN_FREQ", Value::Array(make_arr(chan_freqs))),
223            RecordField::new("CHAN_WIDTH", Value::Array(make_arr(&widths))),
224            RecordField::new("EFFECTIVE_BW", Value::Array(make_arr(&widths))),
225            RecordField::new("FLAG_ROW", Value::Scalar(ScalarValue::Bool(false))),
226            RecordField::new("FREQ_GROUP", Value::Scalar(ScalarValue::Int32(0))),
227            RecordField::new(
228                "FREQ_GROUP_NAME",
229                Value::Scalar(ScalarValue::String(String::new())),
230            ),
231            RecordField::new("IF_CONV_CHAIN", Value::Scalar(ScalarValue::Int32(0))),
232            RecordField::new("MEAS_FREQ_REF", Value::Scalar(ScalarValue::Int32(freq_ref))),
233            RecordField::new("NAME", Value::Scalar(ScalarValue::String(String::new()))),
234            RecordField::new("NET_SIDEBAND", Value::Scalar(ScalarValue::Int32(1))),
235            RecordField::new("NUM_CHAN", Value::Scalar(ScalarValue::Int32(n as i32))),
236            RecordField::new(
237                "REF_FREQUENCY",
238                Value::Scalar(ScalarValue::Float64(ref_freq)),
239            ),
240            RecordField::new("RESOLUTION", Value::Array(make_arr(&widths))),
241            RecordField::new(
242                "TOTAL_BANDWIDTH",
243                Value::Scalar(ScalarValue::Float64(n as f64 * 1.0e6)),
244            ),
245        ]);
246        table.add_row(row).unwrap();
247        table
248    }

250    #[test]
251    fn chan_freq_with_topo_ref() {
252        let freqs = [1.0e9, 1.001e9, 1.002e9];
253        let table = make_spw_table(5, 1.0e9, &freqs); // 5 = TOPO
254
255        let col = ChanFreqColumn::new(&table);
256        let mfreqs = col.get_frequencies(0).unwrap();
257        assert_eq!(mfreqs.len(), 3);
258        assert_eq!(mfreqs[0].refer(), FrequencyRef::TOPO);
259        assert!((mfreqs[0].hz() - 1.0e9).abs() < 1.0);
260        assert!((mfreqs[2].hz() - 1.002e9).abs() < 1.0);
261    }

272    #[test]
273    fn table_reference_mapping_takes_precedence_over_canonical_integer_codes() {
274        for (code, names, codes, expected) in [
275            (
276                64,
277                vec!["LSRK", "Undefined"],
278                vec![64, 9],
279                FrequencyRef::LSRK,
280            ),
281            (
282                9,
283                vec!["LSRK", "Undefined"],
284                vec![64, 9],
285                FrequencyRef::Undefined,
286            ),
287            (36, vec!["LSRK"], vec![27], FrequencyRef::Undefined),
288        ] {
289            let mut table = make_spw_table(code, 1.4e9, &[1.4e9]);
290            TableMeasDesc::new_variable_int(
291                "CHAN_FREQ",
292                MeasureType::Frequency,
293                "MEAS_FREQ_REF",
294                names.into_iter().map(str::to_string).collect(),
295                codes,
296            )
297            .unwrap()
298            .write(&mut table)
299            .unwrap();
300            let column = ChanFreqColumn::new(&table);
301            assert_eq!(column.frequency_reference(0).unwrap(), expected);
302            assert_eq!(column.get_frequencies(0).unwrap()[0].refer(), expected);
303            assert_eq!(column.get_ref_frequency(0).unwrap().refer(), expected);
304        }

383        let ms = crate::MeasurementSet::create_memory(crate::MeasurementSetBuilder::new()).unwrap();
384        let spw = ms

403                frequency_ref_for_table_code(tab_ref_types, tab_ref_codes, 64),
404                Some(FrequencyRef::Undefined)
