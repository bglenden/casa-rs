20/// Bounded metadata acquisition; pixel generation remains in the admitted phase.
21pub(super) fn resolve(
22    request: &ImagingRequest,
23    aw: &AwProjection,
24    surveyed: &Surveyed<'_>,
25    first_cross: FirstCrossSample,
26    context: &crate::RunContext,
27) -> Result<NativeAwRequestInput, PrepareError> {
28    let AwCfSource::NativeEvla {
29        evla_surface,
30        native_cf_working_size,
31        native_cf_oversampling,
32        native_cf_maximum_cells,
33        ..
34    } = &aw.cf_source
35    else {
36        unreachable!("native AW resolves only a native cache");
37    };
38    let surface = evla_dish(surveyed.catalogue, first_cross, evla_surface, aw.aterm)?;
39    let reference_frequency_hz = surveyed.spectral.reference_frequency_hz;
40    let frequencies = frequency_groups((surveyed.reference, surveyed.survey), surveyed.spectral);
41    let feed_angle = receptor_zero_angle(
42        surveyed.catalogue,
43        surveyed.msv2,
44        frequencies[0].spectral_window,
45    )? as f32;
46    let frequencies = if aw.wideband() {
47        frequencies
48    } else {
49        reference_group(frequencies, reference_frequency_hz)
50    };
51    // AW projection images square grids of square cells (the availability
52    // gate refuses others), so x stands for both axes.
53    let sky_cell = request.cell[0].to_radians() / 3600.0;
54    let planes = aw
55        .wprojplanes
56        .expect("a validated AW request names its W planes")
57        .get();
58    let (w_values, w_increment) = w_grid(planes, sky_cell);
59    let working_cell = sky_cell * *native_cf_oversampling as f64 * request.imsize[0] as f64
60        / *native_cf_working_size as f64;
61    let mueller_elements = mueller_elements(&request.stokes)?;
62    let pa_values = if aw.computepastep == 360.0 {
63        vec![f64::from(feed_pa(
64            surveyed,
65            first_cross.time_mjd_seconds,
66            first_cross.field_id,
67            feed_angle,
68        )?)]
69    } else {
70        let [window] = surveyed.survey.spectral_windows.as_slice() else {
71            return Err(PrepareError::NativeAwPaScope);
72        };
73        if surveyed.survey.fields.len() != 1 || request.specmode != crate::SpecMode::Mfs {
74            return Err(PrepareError::NativeAwPaScope);
75        }
76        let groups_per_pa = frequencies.len() * w_values.len();
77        let cell_limit = *native_cf_maximum_cells / (groups_per_pa * mueller_elements.len());
78        let key_limit = (usize::from(u16::MAX) + 1) / groups_per_pa;
79        let epochs = surveyed.survey.window_epochs(window.spw_id);
80        let limit = cell_limit.min(key_limit).min(epochs.len());
81        let _admission = casa_imaging_runtime::admit(
82            &context.host,
83            &context.policy,
84            &casa_imaging_runtime::Demand {
85                phase: "native AW PA centres",
86                memory: (2 * limit * size_of::<f64>()) as u64,
87            },
88        )?;
89        let angles = epochs
90            .iter()
91            .filter(|epoch| epoch.has_unflagged_cross)
92            .map(|epoch| feed_pa(surveyed, epoch.time_mjd_seconds, epoch.field_id, feed_angle));
93        planned_pa_centres(angles, aw.pa_tolerance_rad(), limit)?
94    };
95    Ok(NativeAwRequestInput {
96        surface,
97        antenna_diameter_m: 25.0,
98        frequencies,
99        w_values,
100        w_increment,
101        pa_values,
102        mueller_elements,
103        reference_frequency_hz,
104        grid: NativeAwGrid {
105            size: *native_cf_working_size,
106            sky_increment_rad: [-working_cell, working_cell],
107            oversampling: *native_cf_oversampling,
108        },
109        // `AWProjectFT::makeCFObject`: the W term only over more than one
110        // plane, wideband cells only with the aperture.
111        terms: NativeAwTerms {
112            aperture: aw.aterm,
113            w_term: planes > 1,
114            prolate_spheroidal: aw.psterm,
115            wideband: aw.wideband(),
116            conjugate_beams: aw.conjbeams,
117        },
118        maximum_cells: *native_cf_maximum_cells,
119    })
120}

122/// Antenna zero's Float physical PA plus feed zero's first receptor angle.
123pub(super) fn feed_pa(
124    surveyed: &Surveyed<'_>,
125    time_s: f64,
126    field: i32,
127    feed_rad: f32,
128) -> Result<f32, PrepareError> {
129    let field = usize::try_from(field).expect("a surveyed field is a FIELD row");
130    let angle = surveyed.engine.parallactic_angle(time_s, field, 0)? as f32;
131    Ok(angle + feed_rad)
132}

134/// Generate a centre at the first uncovered selected feed angle.
135/// `AWProjectFT::findConvFunction` fills a cache miss with that angle.
136/// Check the persisted Float-degree centres so native cold and warm
137/// lookups have the same coverage and first-candidate order.
138fn planned_pa_centres(
139    samples: impl IntoIterator<Item = Result<f32, PrepareError>>,
140    step_rad: f32,
141    limit: usize,
142) -> Result<Vec<f64>, PrepareError> {
143    use casa_imaging_model::NativeAwRequestError;
144    use casa_imaging_operator::{first_covered_pa, stored_pa_rad};
145
146    let import_tolerance = 1.0_f32.to_radians();
147    let mut centres_deg = Vec::with_capacity(limit);
148    for pa in samples {
149        let pa = pa?;
150        if !pa.is_finite() {
151            return Err(NativeAwRequestError::InvalidSampling.into());
152        }
153        if first_covered_pa(&centres_deg, pa, step_rad).is_some() {
154            continue;
155        }
156        let degrees = f64::from(pa).to_degrees();
157        let stored = stored_pa_rad(degrees);
158        if (stored - pa).abs() >= step_rad
159            || first_covered_pa(&centres_deg, stored, import_tolerance).is_some()
160        {
161            return Err(PrepareError::NativeAwPaSpacing);
162        }
163        if centres_deg.len() == limit {
164            return Err(NativeAwRequestError::InvalidCatalog.into());
165        }
166        let at = centres_deg.partition_point(|centre| *centre < degrees);
167        centres_deg.insert(at, degrees);
168    }
169    // The original Float radians generated each Double degree header.
170    // Converting back recovers the input Float used to build the aperture.
171    for centre in &mut centres_deg {
172        *centre = f64::from(centre.to_radians() as f32);
173    }
174    centres_deg.shrink_to_fit();
175    Ok(centres_deg)
176}

178/// The EVLA dish surface, after checking that the observation is EVLA with
179/// the homogeneous 25 m dishes the native model describes. CASA images a
180/// `VLA` observation through the same `EVLAAperture`
181/// (`AWProjectFT::createTelescopeATerm`) but with the old VLA's feeds,
182/// which the native model lacks; without the aperture (`aterm=False`) the
183/// cells hold no feed, so either name serves.
184fn evla_dish(
185    catalogue: &Catalogue,
186    first_cross: FirstCrossSample,
187    evla_surface: &std::path::Path,
188    aperture: bool,
189) -> Result<EvlaDishSurface, PrepareError> {
190    let telescope = catalogue
191        .observations
192        .get(usize::try_from(first_cross.observation_id).expect("an observation is a row"))
193        .map(|observation| observation.telescope.trim());
194    let modelled = match telescope {
195        Some("EVLA") => true,
196        Some("VLA") => !aperture,
197        _ => false,
198    };
199    if !modelled {
200        return Err(PrepareError::NativeAwTelescope);
201    }
202    if super::instrument::dish_diameters(catalogue)
203        .iter()
204        .any(|diameter| *diameter != 25.0)
205    {
206        return Err(PrepareError::NativeAwDishes);
207    }
208    // The reference data is an explicit input with bounded acquisition, never
209    // discovered through a CASA installation or downloaded during execution.
210    let mut bytes = Vec::new();
211    std::fs::File::open(evla_surface)
212        .and_then(|file| file.take(1_048_577).read_to_end(&mut bytes))
213        .map_err(|source| PrepareError::EvlaSurface {
214            path: evla_surface.to_path_buf(),
215            source,
216        })?;
217    if bytes.len() > 1_048_576 {
218        return Err(PrepareError::NativeAwSurfaceSize);
219    }
220    let text = std::str::from_utf8(&bytes).map_err(|_| PrepareError::EvlaSurfaceText {
221        path: evla_surface.to_path_buf(),
222    })?;
223    Ok(EvlaDishSurface::from_surface_text(text)?)
224}

226/// One frequency group per selected SPW of the store `store` surveyed
227/// (`survey`), in increasing CF frequency.
228fn frequency_groups(
229    (store, survey): (usize, &Survey),
230    spectral: &PreparedSpectralAxis<'_>,
231) -> Vec<NativeAwFrequencyGroup> {
232    let mut frequencies = survey
233        .spectral_windows
234        .iter()
235        .map(|window| {
236            let channel_frequencies_hz = spectral
237                .window_channels(store, window.spw_id)
238                .iter()
239                .map(|index| {
240                    *window
241                        .frequencies_hz
242                        .get(*index)
243                        .expect("selected channels index their window's frequency axis")
244                })
245                .collect::<Vec<_>>();
246            // TransformMachines2::makeFreqValList selects each SPW's high endpoint.
247            let cf_frequency_hz = channel_frequencies_hz
248                .iter()
249                .copied()
250                .reduce(f64::max)
251                .expect("the spectral axis selects at least one channel of every window");
252            NativeAwFrequencyGroup {
253                spectral_window: u32::try_from(window.spw_id)
254                    .expect("SPW ids are nonnegative stored i32 values"),
255                channel_frequencies_hz,
256                cf_frequency_hz,
257            }
258        })
259        .collect::<Vec<_>>();
260    frequencies.sort_by(|left, right| left.cf_frequency_hz.total_cmp(&right.cf_frequency_hz));
261    frequencies
262}

264/// The one frequency group of a run without wideband cells:
265/// `AWConvFunc::makeFreqValList` samples the aperture at the image
266/// reference frequency alone, whatever the windows. The group names the
267/// lowest window id and holds every selected channel, which the cache's
268/// manifest records.
269fn reference_group(
270    groups: Vec<NativeAwFrequencyGroup>,
271    reference_frequency_hz: f64,
272) -> Vec<NativeAwFrequencyGroup> {
273    let spectral_window = groups
274        .iter()
275        .map(|group| group.spectral_window)
276        .min()
277        .expect("the spectral axis selects a window");
278    let mut channel_frequencies_hz = groups
279        .into_iter()
280        .flat_map(|group| group.channel_frequencies_hz)
281        .collect::<Vec<_>>();
282    channel_frequencies_hz.sort_by(f64::total_cmp);
283    vec![NativeAwFrequencyGroup {
284        spectral_window,
285        channel_frequencies_hz,
286        cf_frequency_hz: reference_frequency_hz,
287    }]
288}

290/// The W values of `planes` planes and their increment. CASA AWConvFunc
291/// derives its W grid from the requested field of view, not the observed W
292/// envelope: maxUVW=1/(4*sky_increment), w=i²/wScale. One plane is the
293/// co-planar w = 0 with no scale (`wScale` stays zero, `makeWValList`).
294fn w_grid(planes: usize, sky_cell: f64) -> (Vec<f64>, f64) {
295    if planes <= 1 {
296        return (vec![0.0], 0.0);
297    }
298    let max_w = 1.0 / (sky_cell * 4.0);
299    let w_increment = ((planes - 1) * (planes - 1)) as f32 as f64 / max_w;
300    let w_values = (0..planes)
301        .map(|index| (index * index) as f64 / w_increment)
302        .collect();
303    (w_values, w_increment)
304}

306/// The Mueller elements of the imaged polarization. The catalog routes
307/// each hand through its own element and, for the conjugate baseline, the
308/// opposite hand (`makeConjPolMap`), so a single-hand image still needs
309/// both diagonal elements.
310fn mueller_elements(stokes: &[PolarizationCoordinate]) -> Result<Vec<usize>, PrepareError> {
311    match stokes {
312        [PolarizationCoordinate::CircularRr]
313        | [PolarizationCoordinate::CircularLl]
314        | [PolarizationCoordinate::StokesI] => Ok(vec![0, 15]),
315        _ => Err(PrepareError::NativeAwPolarization),
316    }
317}

319/// Antenna 0's first receptor angle in window `spw`, radians: the feed
320/// angle CASA's AW projection adds to the parallactic angle. CASA reads it
321/// as casacore's `MSIter` states feed 0's (`feed_pa`:
322/// `receptorAngles().xyPlane(0)`), whatever feeds MAIN's rows name and
323/// whether or not antenna 0 has a selected baseline; on an MSv2 the adapter
324/// states that `FEED` row ([`Msv2Observation::feed_receptors`]). An MSv4
325/// numbers no feeds: each partition's `antenna_xds` states the receptors of
326/// its antennas, so there the angle is antenna 0's as the window's
327/// partitions, selected or not, state it, which must agree.
328pub(super) fn receptor_zero_angle(
329    catalogue: &Catalogue,
330    msv2: Option<&Msv2Observation>,
331    spw: u32,
332) -> Result<f64, PrepareError> {
333    let radians = Unit::new("rad").expect("radians are a unit");
334    let first_angle = |receptors: &Receptors| {
335        receptors
336            .angles
337            .first()
338            .and_then(|angle| angle.get_value_in(&radians).ok())
339            .filter(|value| value.is_finite())
340            .ok_or(PrepareError::FeedAngleNotFinite)
341    };
342    if let Some(ms) = msv2 {
343        return ms
344            .feed_receptors(AntennaId(0), 0, WindowId(spw))
345            .ok_or(PrepareError::NoApplicableFeed)
346            .and_then(first_angle);
347    }
348    let mut angle = None;
349    for entry in catalogue
350        .partitions
351        .iter()
352        .filter(|entry| entry.window == WindowId(spw))
353    {
354        let Some(receptors) = entry.receptors.get(&AntennaId(0)) else {
355            continue;
356        };
357        let value = first_angle(receptors)?;
358        if angle.is_some_and(|previous: f64| previous.to_bits() != value.to_bits()) {
359            return Err(PrepareError::FeedAngleAmbiguous);
360        }
361        angle = Some(value);
362    }
363    angle.ok_or(PrepareError::NoApplicableFeed)
364}
