8impl SubimageSelection {
9    pub(in crate::analysis::imsubimage) fn overlaps_window(
10        &self,
11        start: &[usize],
12        shape: &[usize],
13    ) -> bool {
14        if self.spatial_shapes.is_empty() && self.saved_mask.is_none() {
15            return true;
16        }
17        let [x, y] = self.plane;
18        let overlaps_spatial = self.spatial_shapes.iter().any(|candidate| {
19            let (x0, y0, x1, y1) = shape_candidate_bounds(
20                candidate,
21                self.start[x] + self.shape[x],
22                self.start[y] + self.shape[y],
23            );
24            start[x] <= x1 && start[x] + shape[x] > x0 && start[y] <= y1 && start[y] + shape[y] > y0
25        });
26        overlaps_spatial
27            || self.saved_mask.as_ref().is_some_and(|saved| match saved {
28                SavedRegionMask::Compound(compound) => compound.overlaps(start, shape),
29                SavedRegionMask::Polygon { .. } => true,
30            })
31    }

33    pub(in crate::analysis) fn apply_selection_masks(&self, mask: &mut Option<ArrayD<bool>>) {
34        self.apply_window_masks(mask, &self.start, &self.shape);
35    }

37    pub(in crate::analysis::imsubimage) fn apply_window_masks(
38        &self,
39        mask: &mut Option<ArrayD<bool>>,
40        start: &[usize],
41        shape: &[usize],
42    ) {
43        let [px, py] = self.plane;
44        let box_mask = (!self.spatial_shapes.is_empty()).then(|| {
45            spatial_mask(shape, [px, py], |x, y| {
46                self.spatial_shapes
47                    .iter()
48                    .any(|spatial| shape_contains(spatial, start[px] + x, start[py] + y))
49            })
50        });
51        let region_mask = self.saved_mask.as_ref().map(|saved| match saved {
52            SavedRegionMask::Polygon {
53                axes: [x, y],
54                plane,
55            } => spatial_mask(shape, [*x, *y], |dx, dy| {
56                plane.contains(start[*x] + dx, start[*y] + dy)
57            }),
58            SavedRegionMask::Compound(region) => {
59                let mut pixel = start.to_vec();
60                ArrayD::from_shape_fn(IxDyn(shape), |idx| {
61                    for ((pixel, &start), &offset) in pixel.iter_mut().zip(start).zip(idx.slice()) {
62                        *pixel = start + offset;
63                    }
64                    region.contains(&mut pixel)
65                })
66            }
67        });
68
69        match (box_mask, region_mask) {
70            (Some(mut bm), Some(pm)) => {
71                // `box` and `region` combine by spatial union (CASA fromBCS).
72                Zip::from(&mut bm).and(&pm).for_each(|b, &p| *b = *b || p);
73                if !bm.iter().all(|&v| v) {
74                    and_mask(mask, bm);
75                }
76            }
77            (Some(bm), None) => {
78                if !bm.iter().all(|&v| v) {
79                    and_mask(mask, bm);
80                }
81            }
82            (None, Some(pm)) => {
83                if !pm.iter().all(|&v| v) {
84                    and_mask(mask, pm);
85                }
86            }
87            (None, None) => {}
88        }
89
90        if !self.qualified_shapes.is_empty() {
91            let qualified = ArrayD::from_shape_fn(IxDyn(shape), |idx| {
92                let x = start[px] + idx[px];
93                let y = start[py] + idx[py];
94                // Each containing CRTF row adds or removes the pixel in document order.
95                self.qualified_shapes
96                    .iter()
97                    .rev()
98                    .find(|region| {
99                        let contains = |axis: Option<usize>, selected: Option<&[usize]>| {
100                            axis.zip(selected).is_none_or(|(axis, selected)| {
101                                selected.contains(&(start[axis] + idx[axis]))
102                            })
103                        };
104                        contains(self.channel_axis, region.planes.channels.as_deref())
105                            && contains(self.stokes_axis, region.planes.stokes.as_deref())
106                            && shape_contains(&region.spatial, x, y)
107                    })
108                    .is_some_and(|region| !region.excluded)
109            });
110            if qualified.iter().any(|&value| !value) {
111                and_mask(mask, qualified);
112            }
113        }
114
115        if let Some(ax) = self.channel_axis
116            && !self.channel_spans.is_empty()
117        {
118            mask_skipped_axis(mask, shape, ax, start[ax], |channel| {
119                self.channel_spans.iter().any(|span| span.contains(channel))
120            });
121        }
122
123        if let (Some(ax), Some(indices)) = (self.stokes_axis, &self.stokes_indices)
124            && indices.len() != self.shape[ax]
125        {
126            mask_skipped_axis(mask, shape, ax, start[ax], |index| indices.contains(&index));
127        }
128    }
129}

131fn spatial_mask(
132    shape: &[usize],
133    [x, y]: [usize; 2],
134    includes: impl Fn(usize, usize) -> bool,
135) -> ArrayD<bool> {
136    let plane = Array2::from_shape_fn((shape[x], shape[y]), |(x, y)| includes(x, y));
137    let plane = if x < y { plane } else { plane.reversed_axes() };
138    let mut mask = plane.into_dyn();
139    for axis in 0..shape.len() {
140        if axis != x && axis != y {
141            mask = mask.insert_axis(Axis(axis));
142        }
143    }
144    if mask.shape() == shape {
145        return mask;
146    }
147    mask.broadcast(IxDyn(shape))
148        .expect("spatial mask matches the window plane")
149        .to_owned()
150}
151
152fn and_mask(dst: &mut Option<ArrayD<bool>>, rhs: ArrayD<bool>) {
153    *dst = Some(match dst.take() {
154        Some(mut lhs) => {
155            Zip::from(&mut lhs).and(&rhs).for_each(|a, &b| *a = *a && b);
156            lhs
157        }
158        None => rhs,
159    });
160}
161
162fn mask_skipped_axis(
163    mask: &mut Option<ArrayD<bool>>,
164    shape: &[usize],
165    axis: usize,
166    start: usize,
167    kept: impl Fn(usize) -> bool,
168) {
169    let mut m = mask
170        .take()
171        .unwrap_or_else(|| ArrayD::from_elem(IxDyn(shape), true));
172    for (local, mut slice) in m.axis_iter_mut(Axis(axis)).enumerate() {
173        if !kept(start + local) {
174            slice.fill(false);
175        }
176    }
177    *mask = Some(m);
178}
