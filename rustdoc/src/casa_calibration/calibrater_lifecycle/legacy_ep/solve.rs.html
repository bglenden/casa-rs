43struct ConvolutionSelection<'a> {
44    solver: &'a LegacyEpSolver,
45    model: &'a PreparedEpModel,
46    selected_angle: f64,
47    selected: Arc<EpCacheProducts>,
48    host: Option<HostResources>,
49}
50
51impl ConvolutionSelection<'_> {
52    fn at(&mut self, angle: f64) -> MsResult<&Arc<EpCacheProducts>> {
53        if angle != self.selected_angle
54            && (angle - self.selected_angle).abs().to_degrees() > self.solver.angle_increment_deg
55        {
56            if self.host.is_none() {
57                self.host = Some(HostResources::detect().map_err(invalid)?);
58            }
59            let buffer = FirstBuffer {
60                correlations: self.model.correlation_planes.clone(),
61                frequency_hz: self.model.reference_frequency_hz,
62                angle_rad: angle,
63            };
64            let products = convolution(
65                &self.solver.cache,
66                (self.solver.angle_increment_deg, 1),
67                self.host.as_ref().expect("host resources detected"),
68                &buffer,
69                [self.model.side, self.model.side, 4, 1],
70                self.model.cell_rad,
71                &self.model.coordinates,
72            )?;
73            self.selected = Arc::new(products);
74            self.selected_angle = angle;
75        }
76        Ok(&self.selected)
77    }
78}

80impl<'a> Rows<'a> {
81    fn new(
82        ms: &MeasurementSet,
83        selected: &'a ResolvedMsSelection,
84        interval: f64,
85    ) -> MsResult<Self> {
86        let rows = selected.row_indices().collect::<Vec<_>>();
87        let keys = ms
88            .main_table()
89            .required_scalar_columns_owned_for_rows(&["ARRAY_ID", "SCAN_NUMBER"], &rows)?;
90        let [
91            Some(RequiredScalarColumnValues::Int32(arrays)),
92            Some(RequiredScalarColumnValues::Int32(scans)),
93        ] = [keys.get("ARRAY_ID"), keys.get("SCAN_NUMBER")]
94        else {
95            return Err(invalid("EP solve requires ARRAY_ID and SCAN_NUMBER"));
96        };
97        let mut ordered = selected
98            .selected_rows
99            .iter()
100            .zip(arrays)
101            .zip(scans)
102            .map(|((row, &array), &scan)| {
103                (
104                    row,
105                    [
106                        array,
107                        if interval == 0. { scan } else { 0 },
108                        row.field_id,
109                        row.data_desc_id,
110                    ],
111                )
112            })
113            .collect::<Vec<_>>();
114        ordered.sort_by(|(a, ka), (b, kb)| {
115            ka.cmp(kb)
116                .then(a.time_mjd_seconds.total_cmp(&b.time_mjd_seconds))
117        });
118        Ok(Self {
119            ordered,
120            start: 0,
121            interval,
122        })
123    }
124
125    fn next(&mut self) -> Option<&[(&'a ResolvedMsSelectionRow, [i32; 4])]> {
126        let start = self.start;
127        let &(first, key) = self.ordered.get(start)?;
128        self.start += 1;
129        while let Some((row, next_key)) = self.ordered.get(self.start) {
130            if *next_key != key
131                || (self.interval > 0.
132                    && row.time_mjd_seconds - first.time_mjd_seconds >= self.interval)
133            {
134                break;
135            }
136            self.start += 1;
137        }
138        Some(&self.ordered[start..self.start])
139    }
140}

142impl LegacyEpSolver {
143    pub(in crate::calibrater_lifecycle) fn fit_attached(
144        &self,
145        ms: &MeasurementSet,
146        selected: &ResolvedMsSelection,
147    ) -> MsResult<usize> {
148        let model = self
149            .model
150            .as_ref()
151            .ok_or_else(|| invalid("EP solve needs an image model"))?;
152        let engine = MsCalEngine::new(ms)?;
153        let antenna_count = ms.antenna()?.row_count();
154        let mut rows = Rows::new(ms, selected, self.interval_seconds)?;
155        let mut results = Vec::new();
156        let mut kernels = ConvolutionSelection {
157            solver: self,
158            model,
159            selected_angle: model.initial_angle_rad,
160            selected: Arc::clone(&model.products),
161            host: None,
162        };
163        while let Some(chunk) = rows.next() {
164            let first = chunk[0].0;
165            let input = crate::timing::Phase::start("ep_gather");
166            let Some(LoadedChunk {
167                baselines,
168                reference_time,
169            }) = load_chunk(ms, selected, &engine, chunk, &mut kernels)?
170            else {
171                continue;
172            };
173            if baselines
174                .iter()
175                .any(|baseline| baseline.antennas.iter().any(|&id| id >= antenna_count))
176            {
177                return Err(invalid("EP visibility references an unknown antenna"));
178            }
179            drop(input);
180            let _fit = crate::timing::Phase::start("ep_fit");
181            // EPJones::solveRPar aliases pointPar_, which guessPar zeros
182            // at the beginning of every independent solution interval.
183            let mut offsets = vec![[0.; 4]; antenna_count];
184            if fit::solve(model, &baselines, &mut offsets) {
185                results.push(Slot {
186                    spw: first.spectral_window_id,
187                    field: first.field_id,
188                    time: reference_time,
189                    offsets,
190                });
191            }
192        }
193        if results.is_empty() {
194            return Err(invalid("No valid calibration solutions; no table written."));
195        }
196        let _write = crate::timing::Phase::start("ep_write");
197        table::write(ms, self, &results)?;
198        Ok(results.len() * antenna_count)
199    }
200}

202fn load_chunk(
203    ms: &MeasurementSet,
204    selected: &ResolvedMsSelection,
205    geometry: &MsCalEngine,
206    chunk: &[(&ResolvedMsSelectionRow, [i32; 4])],
207    kernels: &mut ConvolutionSelection<'_>,
208) -> MsResult<Option<LoadedChunk>> {
209    let first = chunk[0].0;
210    let rows = chunk
211        .iter()
212        .map(|(row, _)| *row)
213        .filter(|row| !row.flag_row)
214        .collect::<Vec<_>>();
215    if rows.is_empty() {
216        return Ok(None);
217    }
218    let model = kernels.model;
219    let spectral = Window::prepare(ms, selected, model, first)?;
220    let hand_count = spectral.input_hands.len();
221    let field = first.field_id as usize;
222    let projection = projection(geometry, model, field)?;
223    let preaverage_seconds = kernels.solver.preaverage_seconds;
224    let mut accumulated = BTreeMap::<_, Baseline>::new();
225    let mut angles = BTreeMap::new();
226    let mut epoch_start = first.time_mjd_seconds;
227    let mut bucket = 0;
228    let mut weighted_time = 0.;
229    let mut time_weight = 0.;
230    for block in rows.chunks(256) {
231        let inputs = visibility_cells(ms, block)?;
232        for (row, cells) in block.iter().zip(inputs) {
233            let visibility = Visibility::decode(cells, hand_count, spectral.frequencies.len())?;
234            let weight = visibility.weight();
235            if weight <= 0. || !weight.is_finite() {
236                continue;
237            }
238            // syncSolveMeta uses flagRow and the weighted clock of the first VB.
239            if row.time_mjd_seconds == first.time_mjd_seconds {
240                weighted_time += row.time_mjd_seconds * f64::from(weight);
241                time_weight += f64::from(weight);
242            }
243            if row.antenna1_id == row.antenna2_id {
244                continue;
245            }
246            if row.time_mjd_seconds - epoch_start >= preaverage_seconds
247                && row.time_mjd_seconds != epoch_start
248            {
249                epoch_start = row.time_mjd_seconds;
250                bucket += 1;
251            }
252            let angle = match angles.entry(row.time_mjd_seconds.to_bits()) {
253                std::collections::btree_map::Entry::Occupied(slot) => *slot.get(),
254                std::collections::btree_map::Entry::Vacant(slot) => *slot.insert(f64::from(
255                    geometry.feed_parallactic_angle(row.time_mjd_seconds, field, 0)? as f32,
256                )),
257            };
258            let kernels = kernels.at(angle)?;
259            let channels = spectral
260                .selected
261                .iter()
262                .copied()
263                .filter(|&channel| (0..hand_count).all(|hand| !visibility.flags[[hand, channel]]))
264                .collect::<Vec<_>>();
265            if channels.is_empty() {
266                continue;
267            }
268            let antennas = [row.antenna1_id as usize, row.antenna2_id as usize];
269            let baseline = accumulated
270                .entry((bucket, antennas))
271                .or_insert_with(|| Baseline {
272                    antennas,
273                    observations: Vec::new(),
274                    good_channels: Default::default(),
275                });
276            baseline.observations.push(spectral.observe(
277                model,
278                kernels,
279                &visibility,
280                &channels,
281                &projection,
282                angle,
283            ));
284            baseline.good_channels.extend(channels);
285        }
286    }
287    Ok((time_weight > 0.).then(|| LoadedChunk {
288        reference_time: weighted_time / time_weight,
289        baselines: accumulated.into_values().collect(),
290    }))
291}

293fn projection(
294    geometry: &MsCalEngine,
295    model: &PreparedEpModel,
296    field: usize,
297) -> MsResult<GridFtProjection> {
298    let CoordinateModel::Direction(direction) = model.coordinates.coordinate(0) else {
299        return Err(invalid("EP model is missing its direction coordinate"));
300    };
301    let centre = direction.reference_value();
302    let target = MDirection::from_angles(centre[0], centre[1], direction.direction_ref());
303    let source = geometry.field_direction_j2000(field)?;
304    Ok(GridFtProjection::between_directions(source, &target))
305}

307struct Window {
308    frequencies: Vec<f64>,
309    selected: Vec<usize>,
310    matches: Vec<i32>,
311    input_hands: Vec<usize>,
312}
313
314impl Window {
315    fn prepare(
316        ms: &MeasurementSet,
317        selected: &ResolvedMsSelection,
318        model: &PreparedEpModel,
319        first: &ResolvedMsSelectionRow,
320    ) -> MsResult<Self> {
321        let spw = first.spectral_window_id;
322        let frequencies = ms.spectral_window()?.chan_freq(spw as usize)?;
323        let selected = selected.channel_selections.get(&spw).map_or_else(
324            || Ok((0..frequencies.len()).collect()),
325            |selection| selection.indices(frequencies.len()),
326        )?;
327        let CoordinateModel::Spectral(axis) = model.coordinates.coordinate(2) else {
328            return Err(invalid("EP model is missing its spectral coordinate"));
329        };
330        let matches = crate::ft::image::match_channels(axis, 1, &frequencies);
331        let input_hands = input_hand_indices(ms, first.polarization_id, &model.correlation_planes)?;
332        Ok(Self {
333            frequencies,
334            selected,
335            matches,
336            input_hands,
337        })
338    }
339
340    fn observe(
341        &self,
342        model: &PreparedEpModel,
343        products: &Arc<EpCacheProducts>,
344        visibility: &Visibility,
345        channels: &[usize],
346        projection: &GridFtProjection,
347        angle: f64,
348    ) -> Observation {
349        let ([u, v, _], phase_m) = projection.apply(visibility.uvw);
350        let hands = std::array::from_fn(|hand| {
351            let Some(&input_hand) = self.input_hands.get(hand) else {
352                return fit::ObservedHand::default();
353            };
354            let mut result = fit::ObservedHand {
355                weight: visibility.weights[input_hand],
356                ..Default::default()
357            };
358            for &channel in channels {
359                result.data += visibility.data[[input_hand, channel]];
360                // A single circular hand has no conjugate kernel to sample.
361                if self.input_hands.len() == 1 || self.matches[channel] < 0 {
362                    continue;
363                }
364                let scale = self.frequencies[channel] / SPEED_OF_LIGHT;
365                let pol = model.correlation_planes[hand] as usize - StokesType::RR as usize;
366                result.samples.extend(PointingSample::new(
367                    model,
368                    products,
369                    (pol, 1 - hand),
370                    [-u * scale, -v * scale],
371                    angle,
372                    std::f64::consts::TAU * phase_m * scale,
373                ));
374            }
375            result
376        });
377        Observation {
378            weight: visibility.weight(),
379            hands,
380        }
381    }
382}

384struct Visibility {
385    data: ArrayD<Complex32>,
386    flags: ArrayD<bool>,
387    uvw: [f64; 3],
388    weights: ArrayD<f32>,
389}
390
391impl Visibility {
392    fn decode(cells: [Option<ArrayValue>; 4], hands: usize, channels: usize) -> MsResult<Self> {
393        let [
394            Some(ArrayValue::Complex32(data)),
395            Some(ArrayValue::Bool(flags)),
396            Some(ArrayValue::Float64(uvw)),
397            Some(ArrayValue::Float32(weights)),
398        ] = cells
399        else {
400            return Err(invalid(
401                "EP requires defined DATA, FLAG, UVW and WEIGHT cells",
402            ));
403        };
404        if data.shape() != [hands, channels]
405            || data.shape() != flags.shape()
406            || uvw.shape() != [3]
407            || weights.shape() != [hands]
408        {
409            return Err(invalid(
410                "EP visibility shape disagrees with its spectral window",
411            ));
412        }
413        Ok(Self {
414            data,
415            flags,
416            uvw: [uvw[0], uvw[1], uvw[2]],
417            weights,
418        })
419    }
420
421    fn weight(&self) -> f32 {
422        self.weights.iter().copied().sum::<f32>() / self.weights.len() as f32
423    }
424}

426fn visibility_cells(
427    ms: &MeasurementSet,
428    rows: &[&ResolvedMsSelectionRow],
429) -> MsResult<impl Iterator<Item = [Option<ArrayValue>; 4]> + use<>> {
430    let indices = rows.iter().map(|row| row.row_index).collect::<Vec<_>>();
431    let read = |column| {
432        ms.main_table()
433            .column_accessor(column)?
434            .array_cells_owned_uncached(&indices)
435    };
436    Ok(read("DATA")?
437        .into_iter()
438        .zip(read("FLAG")?)
439        .zip(read("UVW")?)
440        .zip(read("WEIGHT")?)
441        .map(|(((data, flags), uvw), weights)| [data, flags, uvw, weights]))
442}

444fn input_hand_indices(
445    ms: &MeasurementSet,
446    polarization: i32,
447    model: &[StokesType],
448) -> MsResult<Vec<usize>> {
449    let correlations = ms.polarization()?.corr_type(polarization as usize)?;
450    let incompatible =
451        || invalid("EP selected a different set of circular hands from the image model");
452    if correlations.len() != model.len() {
453        return Err(incompatible());
454    }
455    model
456        .iter()
457        .map(|hand| {
458            correlations
459                .iter()
460                .position(|&code| code == *hand as i32)
461                .ok_or_else(incompatible)
462        })
463        .collect()
