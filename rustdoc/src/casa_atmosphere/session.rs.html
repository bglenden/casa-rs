2//! The profile, windows and sky settings owned by one atmosphere tool.
3
4use crate::{
5    AbsorptionProfile, AtmosphereError, AtmosphereProfile, ProfileParameters, Sky, SpectralGrid,
6};
7
8/// Independent mutable state for the atmosphere tool.
9#[derive(Debug, Default)]
10pub struct AtmosphereSession {
11    profile: Option<AtmosphereProfile>,
12    spectrum: Option<AtmosphereSpectrum>,
13}
14
15/// The evaluated ATM channels and the current radiative-transfer settings.
16#[derive(Debug)]
17pub struct AtmosphereSpectrum {
18    /// Absorption and refractivity at the ground-profile water column.
19    pub absorption: AbsorptionProfile,
20    /// User water column, airmass and background used by sky queries.
21    pub sky: Sky,
22}
23
24impl AtmosphereSession {
25    /// The attached atmospheric profile, when initialized.
26    #[must_use]
27    pub fn profile(&self) -> Option<&AtmosphereProfile> {
28        self.profile.as_ref()
29    }
30
31    /// The configured spectral windows and sky settings, when initialized.
32    #[must_use]
33    pub fn spectrum(&self) -> Option<&AtmosphereSpectrum> {
34        self.spectrum.as_ref()
35    }
36
37    /// Mutably access the configured sky for an explicit setting change.
38    pub fn sky_mut(&mut self) -> Option<&mut Sky> {
39        self.spectrum.as_mut().map(|spectrum| &mut spectrum.sky)
40    }
41
42    /// Clear this tool's profile, windows and sky settings.
43    pub fn close(&mut self) {
44        *self = Self::default();
45    }
46
47    /// Attach a new profile and clear the spectral state.
48    pub fn set_profile(&mut self, profile: AtmosphereProfile) {
49        self.profile = Some(profile);
50        self.spectrum = None;
51    }
52
53    /// Replace the spectral windows and reset to the ground column at zenith.
54    /// Return false when the tool has no atmospheric profile.
55    pub fn set_grid(&mut self, grid: SpectralGrid) -> bool {
56        let Some(profile) = &self.profile else {
57            return false;
58        };
59        self.spectrum = Some(AtmosphereSpectrum {
60            absorption: AbsorptionProfile::new(profile.clone(), grid),
61            sky: Sky::zenith(profile.ground_water_column_m()),
62        });
63        true
64    }
65
66    /// Append one regular window and reset sky settings. Preserve all computed
67    /// channels already in the spectrum. An unconfigured tool returns `None`.
68    ///
69    /// # Errors
70    /// Invalid channel count or frequencies, as in [`SpectralGrid::add_regular`].
71    pub fn append_window(
72        &mut self,
73        channels: usize,
74        reference_channel: usize,
75        reference_frequency_hz: f64,
76        separation_hz: f64,
77    ) -> Result<Option<usize>, AtmosphereError> {
78        let Some(spectrum) = &mut self.spectrum else {
79            return Ok(None);
80        };
81        let index = spectrum.absorption.append_regular_window(
82            channels,
83            reference_channel,
84            reference_frequency_hz,
85            separation_hz,
86        )?;
87        spectrum.sky = Sky::zenith(spectrum.absorption.ground_water_column_m());
88        Ok(Some(index))
89    }
90
91    /// Apply the six mutable weather parameters using ATM's rebuild thresholds.
92    /// A successful call resets user water even if the effective profile was
93    /// already within the thresholds. The result says if the profile changed.
94    ///
95    /// # Errors
96    /// A requested rebuild has invalid physical parameters.
97    pub fn update_profile(
98        &mut self,
99        requested: &ProfileParameters,
100    ) -> Result<bool, AtmosphereError> {
101        let Some(profile) = &self.profile else {
102            return Ok(false);
103        };
104        let current = profile.parameters();
105        let changed = site_changed(current, requested);
106        if changed {
107            let effective = ProfileParameters {
108                altitude_m: requested.altitude_m,
109                ground_pressure_hpa: requested.ground_pressure_hpa,
110                ground_temperature_k: requested.ground_temperature_k,
111                lapse_rate_k_per_km: requested.lapse_rate_k_per_km,
112                relative_humidity_percent: requested.relative_humidity_percent,
113                water_vapour_scale_height_km: requested.water_vapour_scale_height_km,
114                ..current.clone()
115            };
116            let updated = AtmosphereProfile::new(&effective)?;
117            if let Some(spectrum) = &mut self.spectrum {
118                spectrum.absorption =
119                    AbsorptionProfile::new(updated.clone(), spectrum.absorption.grid().clone());
120            }
121            self.profile = Some(updated);
122        }
123        if let Some(spectrum) = &mut self.spectrum {
124            spectrum.sky.water_column_m = spectrum.absorption.ground_water_column_m();
125        }
126        Ok(changed)
127    }
128}
129
130fn site_changed(current: &ProfileParameters, requested: &ProfileParameters) -> bool {
131    // ATM compares pressure in pascals, scale height in metres and
132    // humidity as a fraction. Convert both operands before subtracting.
133    (requested.altitude_m - current.altitude_m).abs() > 1.0
134        || (requested.ground_pressure_hpa * 100.0 - current.ground_pressure_hpa * 100.0).abs()
135            > 99.0
136        || (requested.ground_temperature_k - current.ground_temperature_k).abs() > 0.3
137        || (requested.lapse_rate_k_per_km - current.lapse_rate_k_per_km).abs() > 0.01
138        || (requested.relative_humidity_percent / 100.0 - current.relative_humidity_percent / 100.0)
139            .abs()
140            > 1.0
141        || (requested.water_vapour_scale_height_km * 1000.0
142            - current.water_vapour_scale_height_km * 1000.0)
143            .abs()
144            > 20.0
145}
