5/// Apply rotation matrix to direction vector: result = R * v.
6pub(super) fn rotate(r: &[[f64; 3]; 3], v: &[f64; 3]) -> [f64; 3] {
7    let mut result = [0.0; 3];
8    sofars::vm::rxp(r, v, &mut result);
9    result
10}
11
12/// Apply transpose of rotation matrix: result = R^T * v.
13pub(super) fn rotate_t(r: &[[f64; 3]; 3], v: &[f64; 3]) -> [f64; 3] {
14    let mut result = [0.0; 3];
15    sofars::vm::trxp(r, v, &mut result);
16    result
17}

23/// ICRS-to-J2000 frame tie rotation matrix (IAU 2000).
24///
25/// The frame tie between ICRS and the dynamical mean equator/equinox at
26/// J2000.0 is a tiny rotation (~17 mas). We extract the frame bias matrix
27/// `rb` from `sofars::pnp::bp00` evaluated at J2000.0, as casacore's
28/// `MeasTable::ICRSToJ2000` uses `frameBias00`. The IAU 2006 matrix differs
29/// by about 0.3 microarcseconds, observable in simulated FIELD directions.
30///
31/// v_J2000 = ICRS_TO_J2000 * v_ICRS
32pub(super) fn icrs_to_j2000_matrix() -> [[f64; 3]; 3] {
33    // bp00 returns (rb, rp, rbp) where rb is the frame bias.
34    let (rb, _, _) = sofars::pnp::bp00(MJD_OFFSET, 51544.5);
35    rb
36}

47/// Compute TT JD pair (uncached, used by ConvCtx).
48pub(super) fn compute_tt_jd(frame: &MeasFrame) -> Result<(f64, f64), MeasureError> {
49    let epoch = frame.epoch().ok_or(MeasureError::MissingFrameData {
50        what: "epoch (for precession/nutation)",
51    })?;
52
53    if epoch.refer() == EpochRef::TT {
54        return Ok(epoch.value().as_jd_pair());
55    }
56
57    let tt = epoch.convert_to(EpochRef::TT, frame)?;
58    Ok(tt.value().as_jd_pair())
59}

87/// The equatorial-to-ecliptic rotation about x by obliquity `eps`; its
88/// transpose converts back.
89pub(super) fn obliquity_rotation(eps: f64) -> [[f64; 3]; 3] {
90    let mut r = [[0.0; 3]; 3];
91    sofars::vm::ir(&mut r);
92    sofars::vm::rx(eps, &mut r);
93    r
94}

96/// The J2000-to-ECLIPTIC rotation (`MeasMath::deapplyECLIPtoJ2000`).
97pub(super) fn j2000_ecliptic_rotation() -> [[f64; 3]; 3] {
98    obliquity_rotation(casacore_frames::j2000_obliquity_rad())
99}

101/// Rotate equatorial to ecliptic coordinates by `eps`, or back.
102pub(super) fn rotate_obliquity(eps: f64, cosines: &[f64; 3], to_ecliptic: bool) -> [f64; 3] {
103    let m = obliquity_rotation(eps);
104    if to_ecliptic {
105        rotate(&m, cosines)
106    } else {
107        rotate_t(&m, cosines)
108    }
109}

111/// The B1950 conversions' epoch term: the frame's UT1 epoch in Julian
112/// centuries from J2000, or 0 without an epoch, as casacore's
113/// `MeasMath::applyJ2000toB1950` takes it.
114///
115/// Without Earth-orientation data for the epoch, casacore's
116/// `MeasTable::dUT1` returns 0, so UT1 is taken as UTC, with casacore's note
117/// (`casacore_frames::note_missing_dut1`). The term moves a direction by
118/// about 1e-15 rad per second of time. Every other UT1 conversion in casa-rs
119/// still requires the data.
120pub(super) fn b1950_epoch_centuries(frame: &MeasFrame, ctx: &ConvCtx) -> Result<f64, MeasureError> {
121    let Some(epoch) = frame.epoch() else {
122        return Ok(casacore_frames::b1950_epoch_centuries(None));
123    };
124    let ut1_mjd = match get_ut1_jd(frame, ctx) {
125        Ok((ut_a, ut_b)) => (ut_a - MJD_OFFSET) + ut_b,
126        Err(MeasureError::MissingFrameData { what }) if what == MISSING_DUT1 => {
127            let ut1_mjd = epoch
128                .convert_to(EpochRef::UT1, &frame.clone().with_dut1(0.0))?
129                .value()
130                .as_mjd();
131            casacore_frames::note_missing_dut1(ut1_mjd);
132            ut1_mjd
133        }
134        Err(error) => return Err(error),
135    };
136    Ok(casacore_frames::b1950_epoch_centuries(Some(ut1_mjd)))
137}

163/// Get the observer geodetic latitude from the frame position.
164pub(super) fn get_latitude(frame: &MeasFrame) -> Result<f64, MeasureError> {
165    let pos = frame.position().ok_or(MeasureError::MissingFrameData {
166        what: "position (for observer latitude)",
167    })?;
168    // MCFrame uses zero geodetic latitude at the geocentre. SOFA's
169    // degenerate Cartesian-to-geodetic convention chooses the pole there.
170    Ok(if pos.as_itrf() == [0.; 3] {
171        0.
172    } else {
173        pos.latitude_rad()
174    })
175}

193/// Compute UT1 JD pair (uncached, used by ConvCtx).
194pub(super) fn compute_ut1_jd(frame: &MeasFrame) -> Result<(f64, f64), MeasureError> {
195    let epoch = frame.epoch().ok_or(MeasureError::MissingFrameData {
196        what: "epoch (for Earth rotation)",
197    })?;
198
199    if epoch.refer() == EpochRef::UT1 {
200        return Ok(epoch.value().as_jd_pair());
201    }
202
203    let ut1 = epoch.convert_to(EpochRef::UT1, frame)?;
204    Ok(ut1.value().as_jd_pair())
205}

207/// Get polar motion (xp, yp) in radians from the frame's EOP table.
208///
209/// Returns a typed missing-data error when the explicit provider has no EOP
210/// sample for the requested epoch; scientific conversions do not silently
211/// substitute zero polar motion.
212pub(super) fn get_polar_motion_rad(
213    frame: &MeasFrame,
214    epoch_mjd: f64,
215) -> Result<(f64, f64), MeasureError> {
216    const ARCSEC_TO_RAD: f64 = std::f64::consts::PI / (180.0 * 3600.0);
217    match frame.polar_motion_for_mjd(epoch_mjd)? {
218        Some((xp_arcsec, yp_arcsec)) => Ok((xp_arcsec * ARCSEC_TO_RAD, yp_arcsec * ARCSEC_TO_RAD)),
219        None => Err(MeasureError::MissingFrameData {
220            what: "polar-motion measures data for epoch",
221        }),
222    }
223}

225/// Get the geocentric radius in metres from the frame position.
226pub(super) fn get_geocentric_radius(frame: &MeasFrame) -> Result<f64, MeasureError> {
227    let pos = frame.position().ok_or(MeasureError::MissingFrameData {
228        what: "position (for geocentric radius)",
229    })?;
230    let xyz = pos.as_itrf();
231    Ok((xyz[0] * xyz[0] + xyz[1] * xyz[1] + xyz[2] * xyz[2]).sqrt())
232}

234/// Compute the diurnal aberration factor v/c for an observer.
235///
236/// Returns the ratio of the observer's rotational velocity to the speed of light.
237/// C++ casacore computes: `(2π × radius) / SECinDAY × UTtoST(T) / c`.
238/// The UTtoST ratio is approximately 1.00273790935 (sidereal/solar time ratio).
239pub(super) fn diurnal_aberration_factor(radius_m: f64) -> f64 {
240    const C: f64 = 299_792_458.0; // speed of light in m/s
241    const SIDEREAL_RATIO: f64 = 1.002_737_909_35;
242    (2.0 * std::f64::consts::PI * radius_m) / 86400.0 * SIDEREAL_RATIO / C
243}

245/// Compute the polar motion Euler matrix R = Ry(xp) × Rx(yp) × Rz(angle).
246///
247/// Returns the full 3×3 rotation matrix. This matches C++ casacore's
248/// `MeasTable::polarMotion(tdb)` with the third Euler angle set to LAST.
249pub(super) fn polar_motion_euler(xp: f64, yp: f64, last: f64) -> [[f64; 3]; 3] {
250    // Ry(xp) × Rx(yp) × Rz(last)
251    let (sx, cx) = xp.sin_cos();
252    let (sy, cy) = yp.sin_cos();
253    let (sl, cl) = last.sin_cos();
254
255    // Row 0 of Ry(xp):  [cx, 0, sx]
256    // Row 1 of Ry(xp):  [0, 1, 0]
257    // Row 2 of Ry(xp):  [-sx, 0, cx]
258    //
259    // Rx(yp): [1, 0, 0; 0, cy, -sy; 0, sy, cy]
260    // Rz(last): [cl, -sl, 0; sl, cl, 0; 0, 0, 1]
261    //
262    // Compute Rx(yp) × Rz(last) first, then Ry(xp) × result.
263    let rxz = [
264        [cl, -sl, 0.0],
265        [cy * sl, cy * cl, -sy],
266        [sy * sl, sy * cl, cy],
267    ];
268    // Ry(xp) × rxz
269    [
270        [
271            cx * rxz[0][0] + sx * rxz[2][0],
272            cx * rxz[0][1] + sx * rxz[2][1],
273            cx * rxz[0][2] + sx * rxz[2][2],
274        ],
275        rxz[1],
276        [
277            -sx * rxz[0][0] + cx * rxz[2][0],
278            -sx * rxz[0][1] + cx * rxz[2][1],
279            -sx * rxz[0][2] + cx * rxz[2][2],
280        ],
281    ]
282}

284/// Compute the bias-precession-nutation matrix, dispatching on IAU model.
285///
286/// - IAU 1976/1980: `pnm80` (Lieske precession + Wahr nutation, no frame bias)
287/// - IAU 2006/2000A: `pnm00a` (IAU 2000 precession + IAU 2000A nutation + frame bias),
288///   matching C++ casacore's combined `frameBias00 * Precession(IAU2000) * Nutation(IAU2000A)`.
289pub(super) fn bias_precession_nutation(frame: &MeasFrame, tt1: f64, tt2: f64) -> [[f64; 3]; 3] {
290    match frame.iau_model() {
291        IauModel::Iau1976_1980 => sofars::pnp::pnm80(tt1, tt2),
292        IauModel::Iau2006_2000A => sofars::pnp::pnm00a(tt1, tt2),
293    }
294}

313/// Apply the relativistic stellar-aberration transform used by casacore's
314/// `MeasMath::applyAberration`. Passing `-beta` applies the exact inverse
315/// Lorentz transform.
316pub(super) fn apply_relativistic_aberration(direction: [f64; 3], beta: [f64; 3]) -> [f64; 3] {
317    let beta_squared = beta
318        .iter()
319        .map(|component| component * component)
320        .sum::<f64>();
321    let gamma_inverse = (1.0 - beta_squared).sqrt();
322    let dot = direction
323        .iter()
324        .zip(beta.iter())
325        .map(|(direction, beta)| direction * beta)
326        .sum::<f64>();
327    let beta_factor = 1.0 + dot / (1.0 + gamma_inverse);
328    let denominator = 1.0 + dot;
329    let shifted = std::array::from_fn(|axis| {
330        (gamma_inverse * direction[axis] + beta_factor * beta[axis]) / denominator
331    });
332    sofars::vm::pn(&shifted).1
333}

335/// Geocentric direction and distance of the Sun in the J2000-compatible SOFA
336/// ephemeris axes. Casacore's deflection formula depends only weakly on the
337/// sub-milliarcsecond ephemeris-series difference.
338pub(super) fn geocentric_sun_position(frame: &MeasFrame) -> Result<SolarPosition, MeasureError> {
339    let tdb_mjd = tdb_mjd(frame)?;
340    SOLAR_POSITION_CACHE.with(|cache| {
341        cache.get_or_compute(tdb_mjd, || {
342            let (earth_heliocentric, _) = sofars::eph::epv00(MJD_OFFSET, tdb_mjd)
343                .ok_or(MeasureError::SofarsError { code: -1 })?;
344            let sun = [
345                -earth_heliocentric[0][0],
346                -earth_heliocentric[0][1],
347                -earth_heliocentric[0][2],
348            ];
349            let (distance_au, unit) = sofars::vm::pn(&sun);
350            Ok((unit, distance_au))
351        })
352    })
353}

355// Reuse the exact ephemeris input across antenna frames at the same epoch.
356// One entry per thread bounds residency independently of MS size and avoids
357// sharing mutable state between the admitted geometry workers.
358pub(super) type SolarPosition = ([f64; 3], f64);
359
360pub(super) struct SolarPositionCache {
361    pub(super) latest: Cell<Option<(u64, SolarPosition)>>,
362}
363
364impl SolarPositionCache {
365    pub(super) const fn new() -> Self {
366        Self {
367            latest: Cell::new(None),
368        }
369    }
370
371    pub(super) fn get_or_compute(
372        &self,
373        tdb_mjd: f64,
374        compute: impl FnOnce() -> Result<SolarPosition, MeasureError>,
375    ) -> Result<SolarPosition, MeasureError> {
376        let key = tdb_mjd.to_bits();
377        if let Some((previous, position)) = self.latest.get()
378            && previous == key
379        {
380            return Ok(position);
381        }
382        let position = compute()?;
383        self.latest.set(Some((key, position)));
384        Ok(position)
385    }
386}

392/// Casacore `MeasMath::applySolarPos`: convert a J2000 mean direction to the
393/// corresponding natural direction before annual aberration.
394pub(super) fn apply_solar_deflection(
395    direction: [f64; 3],
396    frame: &MeasFrame,
397) -> Result<[f64; 3], MeasureError> {
398    const SOLAR_SEMIDIAMETER_RAD_AT_ONE_AU: f64 = 0.004_652_472_638;
399    const TWO_GM_SUN_OVER_C2_AU: f64 = 1.974e-8;
400
401    let (sun_direction, sun_distance_au) = geocentric_sun_position(frame)?;
402    let dot = direction
403        .iter()
404        .zip(sun_direction.iter())
405        .map(|(direction, sun)| direction * sun)
406        .sum::<f64>();
407    let limb_tolerance = 1.0 - (SOLAR_SEMIDIAMETER_RAD_AT_ONE_AU / sun_distance_au).cos();
408    if (dot - 1.0).abs() <= limb_tolerance {
409        return Ok(direction);
410    }
411
412    let scale = (-TWO_GM_SUN_OVER_C2_AU / sun_distance_au) / (1.0 - dot);
413    let shifted = std::array::from_fn(|axis| {
414        direction[axis] + (sun_direction[axis] - dot * direction[axis]) * scale
415    });
416    Ok(sofars::vm::pn(&shifted).1)
417}

419/// Casacore `MeasMath::deapplySolarPos`: invert the very small solar
420/// deflection. Fixed-point correction converges quadratically here because
421/// the forward displacement is O(1e-8) away from the solar limb.
422pub(super) fn remove_solar_deflection(
423    natural_direction: [f64; 3],
424    frame: &MeasFrame,
425) -> Result<[f64; 3], MeasureError> {
426    let mut mean_direction = natural_direction;
427    for _ in 0..5 {
428        let predicted_natural = apply_solar_deflection(mean_direction, frame)?;
429        let corrected = std::array::from_fn(|axis| {
430            mean_direction[axis] + natural_direction[axis] - predicted_natural[axis]
431        });
432        mean_direction = sofars::vm::pn(&corrected).1;
433    }
434    Ok(mean_direction)
435}
