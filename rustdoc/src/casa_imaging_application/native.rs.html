73/// Why a native run did not complete.
74#[derive(Debug, thiserror::Error)]
75pub(crate) enum NativeError {
76    /// A phase's memory did not fit the resource policy.
77    #[error(transparent)]
78    Admission(#[from] Admission),
79    /// The run was cancelled.
80    #[error(transparent)]
81    Cancelled(#[from] Cancelled),
82    /// Any other failure.
83    #[error("{0}")]
84    Other(ApplicationError),
85}

87impl From<NativeError> for ApplicationDispatchError {
88    fn from(error: NativeError) -> Self {
89        match error {
90            NativeError::Admission(admission) => Self::Admission(admission),
91            NativeError::Cancelled(_) => Self::Cancelled,
92            NativeError::Other(error) => Self::Native(error),
93        }
94    }
95}

171/// The batch route of `problem` on `observation`, the observation compile
172/// opened: each store's selection resolved to its partitions and its
173/// geometry compiled, and the batches of every store admitted
174/// ([`imaging::admit_batches`]). A run that writes the model (`model_data`)
175/// or the corrected data (`corrected_data`) reads each store through the
176/// update session that writes them ([`imaging::WriteBack`]);
177/// `predicts_final_model` says which channels of each time the model's
178/// write takes.
179pub(crate) fn batch_route(
180    problem: &CompiledProblem,
181    observation: OpenedObservation,
182    (model_data, corrected_data, predicts_final_model): (bool, bool, bool),
183    runtime: &ApplicationRuntime,
184) -> Result<Admitted<imaging::BatchRoute>, NativeError> {
185    let other = |error: Box<dyn std::error::Error + Send + Sync>| NativeError::Other(error);
186    let OpenedObservation {
187        stores,
188        measures,
189        phase_table,
190    } = observation;
191    let mut opening = Reservation::none();
192    let mut routes = Vec::with_capacity(stores.len());
193    for (source, Admitted { value, reservation }) in stores.into_iter().enumerate() {
194        opening.join(reservation);
195        let selection = resolve_partitions(problem, source, value.observation())
196            .map_err(|error| other(Box::new(error)))?;
197        let plan = GeometryPlan::compile(
198            problem,
199            value.catalogue(),
200            Arc::clone(&measures),
201            phase_table.clone(),
202        )
203        .map_err(|error| other(Box::new(error)))?;
204        let pointing_budget = if plan.reads_pointing() {
205            value
206                .observation()
207                .pointing_bytes()
208                .map_err(|error| other(Box::new(error)))?
209        } else {
210            0
211        };
212        routes.push((value, selection, plan, pointing_budget));
213    }
214    let Admitted {
215        value: batching,
216        reservation,
217    } = imaging::admit_batches(
218        problem,
219        &routes
220            .iter()
221            .map(
222                |(store, selection, plan, pointing_budget)| imaging::StoreBatches {
223                    observation: store.observation(),
224                    selection,
225                    plan,
226                    pointing_budget: *pointing_budget,
227                },
228            )
229            .collect::<Vec<_>>(),
230        imaging::row_samples(problem)?,
231        (&runtime.host, &runtime.resource_policy),
232    )?;
233    opening.join(reservation);
234    let write_back = (model_data || corrected_data).then(|| {
235        imaging::WriteBack::new(problem, model_data, corrected_data, !predicts_final_model)
236    });
237    let stores = routes
238        .into_iter()
239        .map(|(opened, selection, plan, pointing_budget)| {
240            let store = match &write_back {
241                Some(write_back) => imaging::RouteStore::Write(Box::new(
242                    write_back
243                        .session(opened)
244                        .map_err(|error| other(Box::new(error)))?,
245                )),
246                None => imaging::RouteStore::Read(opened.into_observation()),
247            };
248            Ok(imaging::StoreRoute {
249                store,
250                selection,
251                plan,
252                pointing_budget,
253            })
254        })
255        .collect::<Result<Vec<_>, NativeError>>()?;
256    Ok(Admitted {
257        value: imaging::BatchRoute {
258            stores,
259            batching,
260            write_back,
261        },
262        reservation: opening,
263    })
264}

266/// The end of a run's preparation: check the products against `problem`
267/// and admit the source, its pass storage sized for the policy's streaming
268/// workers ([`casa_imaging_runtime::ResourcePolicy::cores`]), the most the run's setup then
269/// chooses.
270pub(crate) fn prepare(
271    problem: &CompiledProblem,
272    input: NativeInput,
273    native: ApplicationNative,
274) -> Result<NativeRun, NativeError> {
275    let mvc_taylor = input
276        .mvc_taylor
277        .map(|view| view.compile_from_cube(problem))
278        .transpose()
279        .map_err(|error| NativeError::Other(Box::new(error)))?;
280    let ApplicationNative {
281        runtime,
282        mut publication,
283        aw_catalog,
284        awp2,
285    } = native;
286    if let Some(existing) = &input.existing
287        && let Some(beam) = existing.mosaic_beam()
288    {
289        publication.controls = publication
290            .controls
291            .preserving_mosaic_products(beam.reading_bytes(), existing.has_higher_primary_beam());
292    }
293    publication.controls.validate_for_problem(problem)?;
294    let keeps_pb_mask = problem.products().validity().uncorrected_mask()
295        == casa_imaging_model::UncorrectedImageMaskPolicy::PrimaryBeam;
296    let held = |major_cycles| {
297        let controls = controls_after_cycles(
298            publication.controls,
299            input.existing.as_ref(),
300            major_cycles,
301            keeps_pb_mask,
302        );
303        products_floor(problem, &publication, controls)
304    };
305    let products = if problem.reconstruction().controls().max_minor_iterations() == 0 {
306        held(0)?
307    } else {
308        held(0)?.max(held(1)?)
309    };
310    let measures = input
311        .psf_view
312        .as_ref()
313        .map(|_| Arc::clone(&input.observation.measures));
314    let Admitted {
315        value: route,
316        reservation: source,
317    } = batch_route(
318        problem,
319        input.observation,
320        (
321            input.write_model_column,
322            input.write_corrected_data,
323            input.predicts_final_model,
324        ),
325        &runtime,
326    )?;
327    let psf_view = input
328        .psf_view
329        .zip(measures)
330        .map(|(view, measures)| psf::PsfRoute::new(view, measures, &route, &runtime))
331        .transpose()?;
332    Ok(NativeRun {
333        mvc_taylor,
334        psf_view,
335        runtime,
336        publication,
337        aw_catalog,
338        awp2,
339        route: Box::new(route),
340        source,
341        products,
342        masks: input.masks,
343        start_model: input.start_model,
344        outlier_models: input.outlier_models,
345        outlier_psfs: input.outlier_psfs,
346        existing: input.existing,
347        forms: input.forms,
348        image_response: input.minor_cycle_image_response,
349        virtual_model: input.virtual_model,
350    })
351}

353struct SavedBeams {
354    mosaic: Option<casa_imaging_products::SavedMosaicBeam>,
355    psf: Option<Arc<[casa_imaging_deconvolution::RestoringBeam]>>,
356    psf_pixels: Option<Arc<[casa_imaging_deconvolution::RestoringBeam]>>,
357    outliers: Vec<Option<Arc<[casa_imaging_deconvolution::RestoringBeam]>>>,
358    outlier_pixels: Vec<Option<Arc<[casa_imaging_deconvolution::RestoringBeam]>>>,
359    mask_pixels: Option<Arc<[casa_imaging_deconvolution::RestoringBeam]>>,
360}

362impl SavedBeams {
363    fn read(
364        existing: Option<&crate::restart::ExistingProducts>,
365        outlier_psfs: &[Option<crate::restart::ExistingProducts>],
366    ) -> Self {
367        use crate::restart::ExistingProducts;
368        Self {
369            mosaic: existing.and_then(ExistingProducts::mosaic_beam).cloned(),
370            psf: existing.and_then(ExistingProducts::psf_beams),
371            psf_pixels: existing.and_then(ExistingProducts::psf_pixel_beams),
372            outliers: outlier_psfs
373                .iter()
374                .map(|saved| saved.as_ref().and_then(ExistingProducts::psf_beams))
375                .collect(),
376            outlier_pixels: outlier_psfs
377                .iter()
378                .map(|saved| saved.as_ref().and_then(ExistingProducts::psf_pixel_beams))
379                .collect(),
380            mask_pixels: existing.and_then(ExistingProducts::mask_beams),
381        }
382    }
383}

385fn saved_psf_controls(
386    mut publication: ApplicationPublication,
387    forms: imaging::Forms,
388    missing_beam: bool,
389    problem: &CompiledProblem,
390) -> Result<ApplicationPublication, ProductsError> {
391    if !forms.psf && missing_beam && problem.reconstruction().controls().max_minor_iterations() > 0
392    {
393        publication.controls = publication
394            .controls
395            .with_psf_cutoff(casa_imaging_deconvolution::DEFAULT_PSF_FIT_CUTOFF)?;
396    }
397    if !forms.psf {
398        publication.controls = publication.controls.with_reused_single_psf();
399    }
400    Ok(publication)
401}

403/// Run the major-cycle passes, the minor cycles between them and any
404/// visibility write, then publish the products. `summary` collects the
405/// phases and totals as they end; a completed run's outcome takes it once
406/// the caller has finished it.
407pub(crate) fn run(
408    problem: &CompiledProblem,
409    run: NativeRun,
410    summary: &mut RunSummary,
411) -> Result<impl FnOnce(RunSummary) -> ImagingOutcome + use<>, NativeError> {
412    let NativeRun {
413        mvc_taylor,
414        psf_view,
415        runtime,
416        publication,
417        aw_catalog,
418        awp2,
419        mut route,
420        source,
421        products,
422        masks,
423        start_model,
424        outlier_models,
425        outlier_psfs,
426        existing,
427        forms,
428        image_response,
429        virtual_model,
430    } = run;
431    let SavedBeams {
432        mosaic: saved_mosaic_beam,
433        psf: saved_psf_beams,
434        psf_pixels: psf_beams,
435        outliers: outlier_psf_beams,
436        outlier_pixels: outlier_psf_pixel_beams,
437        mask_pixels: mask_beams,
438    } = SavedBeams::read(existing.as_ref(), &outlier_psfs);
439    // An unlabelled saved PSF is fitted by getBeamSet's default cutoff at
440    // minor-cycle initialization. Publication must reuse that fit's cutoff.
441    let publication = saved_psf_controls(publication, forms, psf_beams.is_none(), problem)?;
442    let vla_selection_hz = publication
443        .vla_selection_store
444        .map(|store| vla_selection_frequency(problem, &route, store));
445    let teams = OnceCell::new();
446    let primary_beam = match publication.controls.primary_beam_model() {
447        Some(model) => {
448            AnalyticPrimaryBeam::new(problem, model, publication.primary_beam_centres.as_ref())?
449                .map(|beam| {
450                    vla_selection_hz.map_or(beam, |frequency| {
451                        beam.with_vla_selection_frequency(frequency)
452                    })
453                })
454        }
455        None => None,
456    };
457    let separate_psf = psf_view
458        .map(|view| view.form(&mut route, &runtime, summary))
459        .transpose()?;
460    // The run's source borrows the batch route, which is released with the
461    // source's charge once the run has finished with it.
462    let visibility_write = route
463        .write_back()
464        .map(|_| imaging::VisibilityWriteTarget(&route));
465    let outcome = imaging::run(
466        imaging::ImagingInputs {
467            problem,
468            mvc: mvc_taylor.as_ref().and_then(|view| {
469                (view
470                    .problem
471                    .reconstruction()
472                    .controls()
473                    .max_minor_iterations()
474                    > 0)
475                .then_some(imaging::MvcCycleSpec {
476                    taylor_problem: &view.problem,
477                    start: view.start.as_ref(),
478                    expansion_reference_hz: view.expansion_reference_hz,
479                    controls: publication.controls,
480                    centres: publication.primary_beam_centres.as_ref(),
481                })
482            }),
483            psf_cutoff: if !forms.psf && psf_beams.is_none() {
484                casa_imaging_deconvolution::DEFAULT_PSF_FIT_CUTOFF
485            } else {
486                publication.controls.psf_cutoff()
487            },
488            psf_beams,
489            outlier_psf_beams: outlier_psf_pixel_beams,
490            mask_beams,
491            source: &route,
492            separate_psf,
493            masks,
494            start_model,
495            outlier_models,
496            existing: existing.clone(),
497            forms,
498            image_response,
499            primary_beam,
500            visibility_write,
501            products: (products, source.memory()),
502            host: runtime.host,
503            policy: runtime.resource_policy,
504            teams: &teams,
505            cancel: runtime.cancel.clone(),
506            spill_directory: &runtime.spill_directory,
507            aw_catalog,
508            awp2,
509            backend: runtime.backend,
510            grid_precision: runtime.grid_precision,
511            ft_padding: runtime.ft_padding,
512            diagnostics: runtime.diagnostics,
513        },
514        summary,
515    )?;
516    drop(route);
517    drop(source);
518    let team = teams
519        .get()
520        .expect("a run that completed formed its teams")
521        .compute();
522    let mut published = Vec::new();
523    let mut lanes = (0, None);
524    let mvc_run = mvc_taylor.is_some();
525    // CASA creates the cube model when it invokes its first minor cycle.
526    let cube_model_created = outcome.mvc_taylor.is_none()
527        || !outcome.minor_cycles.is_empty()
528        || mvc_taylor.as_ref().is_some_and(|view| view.start.is_some());
529    let products: Result<_, NativeError> = run_phase(
530        "products",
531        &runtime.cancel,
532        runtime.diagnostics,
533        summary,
534        || {
535            publish_run_products(
536                RunProductInputs {
537                    problem,
538                    source: ProductSource {
539                        scientific: outcome.scientific,
540                        masks: outcome.masks,
541                        facet_sum_weights: outcome.facet_sum_weights,
542                        mem_alpha: outcome.mem_alpha,
543                        saved_psf_beams,
544                        outlier_psf_beams,
545                        vla_selection_hz,
546                        primary_beam: saved_mosaic_beam
547                            .as_ref()
548                            .map(ProductPrimaryBeam::SavedMosaic),
549                    },
550                    taylor_view: mvc_taylor,
551                    taylor_completion: outcome.mvc_taylor,
552                    cube_model_created,
553                    major_cycles: outcome.major_cycle_count,
554                },
555                &runtime,
556                ApplicationPublication {
557                    controls: controls_after_cycles(
558                        publication.controls,
559                        existing.as_ref(),
560                        outcome.major_cycle_count,
561                        problem.products().validity().uncorrected_mask()
562                            == casa_imaging_model::UncorrectedImageMaskPolicy::PrimaryBeam,
563                    ),
564                    ..publication
565                },
566                team,
567                &mut published,
568                &mut lanes,
569            )
570        },
571    );
572    summary.products = published;
573    (summary.product_lanes, summary.product_lanes_reason) = lanes;
574    let (products, taylor_names, names_reservation) = products?;
575    let mut retained = outcome.retained;
576    retained.join(names_reservation);
577    for plan in &virtual_model {
578        run_phase(
579            "virtual model",
580            &runtime.cancel,
581            runtime.diagnostics,
582            summary,
583            || {
584                crate::virtual_model::save(
585                    problem,
586                    products.scientific.final_model(),
587                    plan,
588                    &runtime,
589                )
590            },
591        )?;
592    }
593    let workers = team.workers();
594    let problem = problem.clone();
595    Ok(move |summary| ImagingOutcome {
596        problem,
597        minor_cycles: outcome.minor_cycles,
598        plane_rows: outcome.plane_rows,
599        final_residual_statistics: outcome.final_residual_statistics,
600        stop: outcome.stop,
601        major_cycle_count: outcome.major_cycle_count,
602        total_minor_iterations: outcome.total_minor_iterations,
603        total_actual_minor_iterations: outcome.total_actual_minor_iterations,
604        workers,
605        planes_per_wave: outcome.planes_per_wave,
606        visibility_products: outcome.visibility_products,
607        summary,
608        scientific: products.scientific,
609        planned_products: products.planned,
610        products: products.published,
611        mvc_taylor_names: taylor_names,
612        mvc_run,
613        _retained: retained,
614    })
615}

617/// CASA selects the cube's VLA aperture through its LSRK conversion layer,
618/// but evaluates it at each output plane's own frequency.
619fn vla_selection_frequency(
620    problem: &CompiledProblem,
621    route: &imaging::BatchRoute,
622    store: usize,
623) -> f64 {
624    let factor = route.stores[store]
625        .plan
626        .output_lsrk_factor()
627        .expect("a tracked cube has an LSRK conversion layer");
628    problem
629        .geometry()
630        .spectral()
631        .channel_centre_hz(0)
632        .expect("a cube has a first channel")
633        * factor
634}

646/// Reduce MVC's cube before staging both families, or publish an ordinary run.
647fn publish_run_products(
648    inputs: RunProductInputs<'_>,
649    runtime: &ApplicationRuntime,
650    publication: ApplicationPublication,
651    team: &WorkerTeam,
652    published: &mut Vec<String>,
653    lanes: &mut (usize, Option<String>),
654) -> Result<(Products, Vec<String>, Reservation), NativeError> {
655    let RunProductInputs {
656        problem,
657        source,
658        taylor_view,
659        taylor_completion,
660        cube_model_created,
661        major_cycles,
662    } = inputs;
663    if let Some(view) = taylor_view {
664        let crate::compile::CompiledTaylorView {
665            problem: taylor_problem,
666            coordinates,
667            expansion_reference_hz,
668            reservation: _view_reservation,
669            start,
670        } = view;
671        let saved = start.as_ref().and_then(|start| start.saved_psf.as_ref());
672        let taylor = if let Some(taylor) = taylor_completion {
673            taylor
674        } else {
675            crate::mvc::assemble_taylor_completion(
676                crate::mvc::TaylorAssemblyInputs {
677                    cube_problem: problem,
678                    cube: &source.scientific,
679                    taylor_problem: &taylor_problem,
680                    expansion_reference_hz,
681                    controls: publication.controls,
682                    centres: publication.primary_beam_centres.as_ref(),
683                    read_saved_residual: false,
684                },
685                (&runtime.host, &runtime.resource_policy, &runtime.cancel),
686            )?
687        };
688        let products = publish_mvc_products(
689            MvcProductSource {
690                cube_problem: problem,
691                cube_model_created,
692                cube: source,
693                taylor_problem: &taylor_problem,
694                taylor,
695                taylor_coordinates: coordinates,
696                saved,
697                saved_residual_mask_bytes: (major_cycles == 0)
698                    .then(|| saved.and_then(|saved| saved.taylor.saved_residual_mask_bytes()))
699                    .flatten(),
700            },
701            runtime,
702            publication,
703            team,
704            published,
705            lanes,
706        )?;
707        Ok((
708            products.cube,
709            products.taylor_names,
710            products.names_reservation,
711        ))
712    } else {
713        if taylor_completion.is_some() {
714            return Err(NativeError::Other(
715                std::io::Error::other("MVC Taylor completion has no compiled view").into(),
716            ));
717        }
718        let products = publish_products(
719            problem,
720            source,
721            runtime,
722            publication,
723            team,
724            published,
725            lanes,
726        )?;
727        Ok((products, Vec::new(), Reservation::none()))
728    }
729}

735    published: PublishedContinuumGeneration,

738/// What the products are formed from: the final completion, its
739/// reconstruction masks and the faceted domains' facets' own sums of
740/// weights.
741enum ProductPrimaryBeam<'a> {
742    MvcCube(
743        &'a [f32],
744        Option<&'a casa_imaging_products::SavedMosaicBeam>,
745        bool,
746    ),
747    MvcTaylor(&'a [f32], Option<&'a crate::compile::MvcSavedPsf>),
748    SavedMosaic(&'a casa_imaging_products::SavedMosaicBeam),
749}

751struct ProductSource<'a> {
752    scientific: MajorCycleCompletion,
753    masks: Option<ReconstructionMaskSet>,
754    facet_sum_weights: FacetSumWeights,
755    mem_alpha: Option<f32>,
756    saved_psf_beams: Option<std::sync::Arc<[casa_imaging_deconvolution::RestoringBeam]>>,
757    outlier_psf_beams: Vec<Option<std::sync::Arc<[casa_imaging_deconvolution::RestoringBeam]>>>,
758    vla_selection_hz: Option<f64>,
759    /// A separately admitted PB, borrowed until the family is staged.
760    primary_beam: Option<ProductPrimaryBeam<'a>>,
761}

763/// CASA keeps the saved residual in image units until the first major
764/// cycle, then retains its mask when residual division leaves it in place.
765fn controls_after_cycles(
766    controls: casa_imaging_products::ContinuumProductControls,
767    existing: Option<&crate::restart::ExistingProducts>,
768    major_cycles: usize,
769    keeps_pb_mask: bool,
770) -> casa_imaging_products::ContinuumProductControls {
771    let Some(existing) = existing else {
772        return controls;
773    };
774    let Some(bytes) = existing.saved_residual_mask_bytes() else {
775        return controls;
776    };
777    if major_cycles == 0 {
778        controls.preserving_weighted_residuals(bytes)
779    } else if keeps_pb_mask && existing.reads_mosaic_residual() && bytes > 0 {
780        controls.with_saved_residual_mask(bytes)
781    } else {
782        controls
783    }
784}

786/// What `problem`'s products hold on one lane, before the run
787/// (`PlannedContinuumGeneration::planned_demand`): the least they can run
788/// on, which the run's setup fits beside it.
789fn products_floor(
790    problem: &CompiledProblem,
791    publication: &ApplicationPublication,
792    controls: casa_imaging_products::ContinuumProductControls,
793) -> Result<u64, NativeError> {
794    Ok(PlannedContinuumGeneration::for_problem(problem, &controls)?
795        .planned_demand(problem, ProductStoragePlan::new(1, 1)?, &publication.writer)?
796        .peak_residency_bytes())
797}

799/// Generate every planned product member into the writer's private staging,
800/// one output channel per window on the run's team, and publish them,
801/// adding each image published to `published`; a cancelled run publishes
802/// nothing. `lanes` receives the lanes the products ran on and, when fewer
803/// than the team's workers, why.
804fn publish_products(
805    problem: &CompiledProblem,
806    source: ProductSource<'_>,
807    runtime: &ApplicationRuntime,
808    mut publication: ApplicationPublication,
809    team: &WorkerTeam,
810    published: &mut Vec<String>,
811    lanes: &mut (usize, Option<String>),
812) -> Result<Products, NativeError> {
813    if let Some(alpha) = source.mem_alpha {
814        publication.writer.set_mem_alpha(alpha)?;
815    }
816    let (products, held) =
817        stage_products(problem, source, runtime, &publication, team, lanes, false)?;
818    if runtime.cancel.is_cancelled() {
819        // The staged members are removed when the writer drops.
820        return Err(Cancelled.into());
821    }
822    publication
823        .writer
824        .publish(|image: &Path| published.push(image.to_string_lossy().into_owned()))?;
825    // The charge covers the staged records until they are published.
826    drop(held);
827    Ok(products)
828}

837    cube_model_created: bool,
838    cube: ProductSource<'a>,

844        casa_imaging_model::ImageDomainRole,
845        casa_coordinates::CoordinateSystem,
846    )>,

849/// The cube remains the run's scientific result; Taylor names are separately
850/// charged so the task's artifact inventory can report both families.
851#[cfg_attr(
852    not(test),
853    allow(dead_code, reason = "MVC publication is being wired in #1069")
854)]
855struct MvcProducts {
856    cube: Products,
857    taylor_names: Vec<String>,
858    names_reservation: Reservation,

861/// Stage both MVC families before publishing either. A failure or cancellation
862/// drops the shared writer and discards its staged members.
863#[cfg_attr(
864    not(test),
865    allow(dead_code, reason = "MVC publication is being wired in #1069")
866)]
867fn publish_mvc_products(
868    source: MvcProductSource<'_>,
869    runtime: &ApplicationRuntime,
870    publication: ApplicationPublication,
871    team: &WorkerTeam,
872    published: &mut Vec<String>,
873    lanes: &mut (usize, Option<String>),
874) -> Result<MvcProducts, NativeError> {
875    if runtime.cancel.is_cancelled() {
876        return Err(Cancelled.into());
877    }
878    let MvcProductSource {
879        cube_problem,
880        cube_model_created,
881        cube,
882        taylor_problem,
883        taylor,
884        taylor_coordinates,
885        saved,
886        saved_residual_mask_bytes,
887    } = source;
888    let Admitted {
889        value:
890            crate::mvc::TaylorCompletion {
891                scientific: taylor_science,
892                primary_beam,
893            },
894        reservation: taylor_reservation,
895    } = taylor;
896    let ProductSource {
897        scientific,
898        masks,
899        facet_sum_weights,
900        mem_alpha: _,
901        saved_psf_beams,
902        outlier_psf_beams,
903        vla_selection_hz,
904        primary_beam: _,
905    } = cube;
906    let taylor_publication = ApplicationPublication {
907        controls: saved_residual_mask_bytes.map_or(publication.controls, |bytes| {
908            publication.controls.preserving_weighted_residuals(bytes)
909        }),
910        primary_beam_centres: publication.primary_beam_centres.clone(),
911        vla_selection_store: None,
912        writer: publication.writer.with_coordinates(taylor_coordinates),
913    };
914    let (cube, cube_charge) = stage_products(
915        cube_problem,
916        ProductSource {
917            scientific,
918            masks: None,
919            facet_sum_weights,
920            mem_alpha: None,
921            saved_psf_beams,
922            outlier_psf_beams,
923            vla_selection_hz,
924            primary_beam: Some(ProductPrimaryBeam::MvcCube(
925                &primary_beam,
926                saved.map(|saved| &saved.cube_pb),
927                saved_residual_mask_bytes.is_some(),
928            )),
929        },
930        runtime,
931        &publication,
932        team,
933        lanes,
934        !cube_model_created,
935    )?;
936    let (taylor, taylor_charge) = stage_products(
937        taylor_problem,
938        ProductSource {
939            scientific: taylor_science,
940            masks,
941            facet_sum_weights: Default::default(),
942            mem_alpha: None,
943            saved_psf_beams: saved.and_then(|saved| saved.taylor.psf_beams()),
944            outlier_psf_beams: Vec::new(),
945            vla_selection_hz: None,
946            primary_beam: Some(ProductPrimaryBeam::MvcTaylor(&primary_beam, saved)),
947        },
948        runtime,
949        &taylor_publication,
950        team,
951        lanes,
952        false,
953    )?;
954    let count = taylor.planned.members().len();
955    let overflow = || ProductsError::ResourceDemandOverflow("MVC product names");
956    let bytes = taylor.planned.members().iter().try_fold(
957        count
958            .checked_mul(size_of::<String>())
959            .ok_or_else(overflow)?,
960        |sum, member| sum.checked_add(member.name().len()).ok_or_else(overflow),
961    )?;
962    let names_reservation = admit(
963        &runtime.host,
964        &runtime.resource_policy,
965        &Demand {
966            phase: "MVC product inventory",
967            memory: u64::try_from(bytes).map_err(|_| overflow())?,
968        },
969    )?;
970    let mut taylor_names = Vec::with_capacity(count);
971    taylor_names.extend(
972        taylor
973            .planned
974            .members()
975            .iter()
976            .map(|member| member.name().to_owned()),
977    );
978    if runtime.cancel.is_cancelled() {
979        return Err(Cancelled.into());
980    }
981    publication
982        .writer
983        .publish(|image: &Path| published.push(image.to_string_lossy().into_owned()))?;
984    drop(taylor);
985    drop(taylor_charge);
986    drop(cube_charge);
987    drop(taylor_reservation);
988    Ok(MvcProducts {
989        cube,
990        taylor_names,
991        names_reservation,
992    })
993}

995/// Stage one product family without publishing it. The caller holds its
996/// admission until every staged family has been published or discarded.
997fn stage_products(
998    problem: &CompiledProblem,
999    source: ProductSource<'_>,
1000    runtime: &ApplicationRuntime,
1001    publication: &ApplicationPublication,
1002    team: &WorkerTeam,
1003    lanes: &mut (usize, Option<String>),
1004    omit_uncreated_mvc_cube_model: bool,
1005) -> Result<(Products, casa_imaging_runtime::Reservation), NativeError> {
1006    let ProductSource {
1007        scientific,
1008        masks: reconstruction_masks,
1009        facet_sum_weights,
1010        mem_alpha: _,
1011        saved_psf_beams,
1012        outlier_psf_beams,
1013        vla_selection_hz,
1014        primary_beam,
1015    } = source;
1016    let mut controls = publication.controls;
1017    let mut inputs = ContinuumProductInputs::from_major_cycle(problem, &scientific);
1018    let mut preserve_cube_residual = false;
1019    if let Some(beam) = primary_beam {
1020        let saved = match beam {
1021            ProductPrimaryBeam::MvcCube(beam, saved, preserve_residual) => {
1022                preserve_cube_residual = preserve_residual;
1023                if !preserve_residual {
1024                    inputs = inputs.with_cube_taylor_primary_beam(beam)?;
1025                }
1026                saved.map(|beam| (beam, false))
1027            }
1028            ProductPrimaryBeam::MvcTaylor(beam, saved) => {
1029                inputs = inputs
1030                    .with_taylor_primary_beam(beam)?
1031                    .with_taylor_residual_sum_weight(1.0)?;
1032                saved.map(|saved| (&saved.taylor_pb, saved.taylor.has_higher_primary_beam()))
1033            }
1034            ProductPrimaryBeam::SavedMosaic(beam) => {
1035                inputs = inputs.with_saved_mosaic_beam(beam)?;
1036                None
1037            }
1038        };
1039        if let Some((beam, higher)) = saved {
1040            inputs = inputs.with_saved_mosaic_beam(beam)?;
1041            controls = controls.preserving_mosaic_products(beam.reading_bytes(), higher);
1042        }
1043    }
1044    if let Some(frequency) = vla_selection_hz {
1045        inputs = inputs.with_vla_selection_frequency(frequency);
1046    }
1047    if let Some(beams) = saved_psf_beams.as_deref() {
1048        inputs = inputs.with_saved_psf_beams(beams)?;
1049    }
1050    if !outlier_psf_beams.is_empty() {
1051        inputs = inputs.with_saved_outlier_psf_beams(&outlier_psf_beams)?;
1052    }
1053    if let Some(masks) = reconstruction_masks.as_ref() {
1054        inputs = match masks {
1055            ReconstructionMaskSet::Shared(mask) => inputs.with_reconstruction_mask(mask),
1056            ReconstructionMaskSet::Domains(masks) => inputs.with_domain_reconstruction_masks(masks),
1057        };
1058    }
1059    if let Some(centres) = publication.primary_beam_centres.as_ref() {
1060        inputs = inputs.with_primary_beam_centres(centres);
1061    }
1062    if !facet_sum_weights.is_empty() {
1063        inputs = inputs.with_facet_sum_weights(&facet_sum_weights);
1064    }
1065    let mut planned = PlannedContinuumGeneration::new(&inputs, &controls)?;
1066    if omit_uncreated_mvc_cube_model {
1067        planned = planned.without_uncreated_mvc_cube_model();
1068    }
1069    if preserve_cube_residual {
1070        planned = planned.without_mvc_cube_residual();
1071    }
1072    // The products' scratch grows with their lanes, and the gridding's
1073    // workers were chosen without it: the products take the most lanes, up
1074    // to the team's, whose demand fits what the policy leaves free (#763
1075    // review), and one when none does, which admission then refuses.
1076    let free = free_memory(&runtime.host, &runtime.resource_policy);
1077    let demand_on = |count: usize| -> Result<_, NativeError> {
1078        let window = ProductStoragePlan::new(1, count)?;
1079        Ok((
1080            window,
1081            planned.demand(&inputs, window, &publication.writer)?,
1082        ))
1083    };
1084    let workers = team.workers();
1085    let (mut window, mut demand) = demand_on(workers)?;
1086    debug_assert_eq!(
1087        planned.planned_demand(problem, window, &publication.writer)?,
1088        demand,
1089        "the products' demand before the run is the run's"
1090    );
1091    let needed = demand.peak_residency_bytes();
1092    if needed > free {
1093        for count in (1..workers).rev() {
1094            (window, demand) = demand_on(count)?;
1095            if demand.peak_residency_bytes() <= free {
1096                break;
1097            }
1098        }
1099        let count = window.maximum_workers();
1100        let gigabytes = |bytes: u64| format!("{:.2} GB", bytes as f64 / 1e9);
1101        let reason = if demand.peak_residency_bytes() <= free {
1102            format!(
1103                "{workers} lanes need {}, {count} fit {} free",
1104                gigabytes(needed),
1105                gigabytes(free)
1106            )
1107        } else {
1108            format!(
1109                "{workers} lanes need {} and no fewer fit {} free",
1110                gigabytes(needed),
1111                gigabytes(free)
1112            )
1113        };
1114        if count < workers {
1115            tracing::warn!("products on {count} lanes, not the team's {workers}: {reason}");
1116            lanes.1 = Some(reason);
1117        }
1118    }
1119    lanes.0 = window.maximum_workers();
1120    let held = admit(
1121        &runtime.host,
1122        &runtime.resource_policy,
1123        &Demand {
1124            phase: "products",
1125            memory: demand.peak_residency_bytes(),
1126        },
1127    )?;
1128    let generation =
1129        produce_continuum_members(&planned, &inputs, window, team, &publication.writer)?;
1130    Ok((
1131        Products {
1132            planned,
1133            scientific,
1134            published: generation,
1135        },
1136        held,
1137    ))
1138}
