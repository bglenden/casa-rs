35/// A new MeasurementSet v2 planned from observations, checked before
36/// anything is written ([`CreationPlan::new`]) and then written
37/// ([`CreationPlan::write`]).
38///
39/// The inputs' catalogues are merged as CASA's `concat` merges
40/// MeasurementSets' subtables (`casa_observation::merge`), their scans
41/// offset as it offsets them, and their partitions combined by the key the
42/// MSv2 reader partitions rows by; a combined partition's inputs must not
43/// acquire one time and baseline twice. The selected data groups
44/// ([`CreationOptions::data_groups`], by default every input's) become
45/// `DATA` (`base`), `CORRECTED_DATA` (`corrected`) and `MODEL_DATA`
46/// (`model`), sharing the first group's flags, weights and UVW; an input
47/// without `corrected` is written from its `base`, and one without `model`
48/// with CASA's initial model. Every grid cell of one input becomes one
49/// row; padding becomes a row whose every `FLAG` is set, with NaN data,
50/// weights and `UVW`, the grid's interval, no exposure and `FLAG_ROW` false
51/// (its flags already exclude it), as D23 decided. With
52/// [`PaddingRows::Omitted`], which partitions combined from several inputs
53/// need, padding becomes no row. `WEIGHT` is the input's times its
54/// [`CreationInput::weight_scale`], and `SIGMA` is derived from it: a
55/// loss ([`Loss::SigmaDerived`], which the caller must accept as
56/// [`LossKind::Sigma`](super::LossKind::Sigma)) unless the input's stored
57/// `SIGMA` is already `1/√WEIGHT` to within one unit in the last place
58/// (an MSv4, or an MSv2 with `WEIGHT_SPECTRUM`, always takes it). Names
59/// are the inputs': an
60/// MSv4 converted from an MSv2 holds XRADIO's suffixed names
61/// (`NAME_FIELDID`, `NAME_STATION`, ...), so a round trip renames.
62///
63/// A MeasurementSet's row count is fixed before its rows are written. With
64/// padding omitted it is the acquisitions, which only the weights record,
65/// so the plan reads them to count; otherwise it is the grids' cells.
66pub struct CreationPlan {
67    inputs: Vec<Input>,
68    pointing: Vec<PointingPlan>,
69    combined: Combined,
70    /// Each data column, and the units it declares.
71    columns: Vec<(OptionalMainColumn, Option<String>)>,
72    losses: Vec<Loss>,
73    occupancy: Occupancy,
74    output: Catalogue,
75    /// MAIN's write plan, within the memory admitted.
76    main: MeasurementSetWritePlan,
77    charges: MemoryCharges,
78    /// What writing takes from the options: the `HISTORY` rows, the
79    /// padding policy and the batch budget.
80    history: Vec<HistoryRow>,
81    padding: PaddingRows,
82    batch_bytes: usize,
83    /// The date the ephemeris tables written state.
84    creation_date: String,
85    publication: Publication,
86}
87
88/// How the planned MeasurementSet is published.
89enum Publication {
90    /// Renamed from its staging directory to a new path.
91    New(PathBuf),
92    /// Exchanged with the MeasurementSet it replaces, whose lock the plan
93    /// holds.
94    Replace(Replacing),
95}

98    /// Open `inputs` and plan the MeasurementSet created from them at
99    /// `destination`, in input order, reading what the plan must know before
100    /// writing: each input's losses, its pointing, its grids and, with
101    /// padding omitted, its acquisitions. A [`Destination::Replace`]
102    /// destination is locked first. What making the plan holds, and what
103    /// writing it will hold, are admitted against
104    /// [`CreationOptions::memory_bytes`] before they are allocated
105    /// ([`Self::charges`]).
106    ///
107    /// # Errors
108    ///
109    /// Nothing is written by a plan. Memory beyond
110    /// [`CreationOptions::memory_bytes`], or a batch of
111    /// [`CreationOptions::batch_bytes`] that cannot hold one integration of
112    /// every selected group ([`ObservationError::OverBudget`]). An existing
113    /// path for a
114    /// [`Destination::New`], or for a [`Destination::Replace`] no
115    /// MeasurementSet, one another process holds for writing, or a
116    /// filesystem that cannot exchange directories. A weight scale that is
117    /// not finite and positive, no input, or padding flagged where
118    /// partitions combine ([`ObservationError::InvalidRequest`]); a loss not
119    /// accepted
120    /// ([`ObservationError::InvalidRequest`]); fields whose direction is a
121    /// polynomial in time, fields that follow an ephemeris in a combination
122    /// of several inputs, or an ephemeris whose samples are not evenly
123    /// spaced or whose radial velocities are not geocentric; a data group
124    /// without an MSv2 column, or selected twice; times in another scale
125    /// than UTC; feeds other than 0; a window without channels; fields in
126    /// another frame than J2000; a scan without a casacore number; feeds
127    /// whose receptors differ; a correlation product the feeds cannot name
128    /// (a Stokes product, or a receptor they lack); one column's groups in
129    /// different units; what the merge and the combination refuse; a time
130    /// and baseline two inputs both acquire, or two fields or scans at one
131    /// time, or two intervals, of a combined partition; inputs with nothing
132    /// to write. Weights that vary over channels (an MSv2 `WEIGHT` holds one
133    /// per correlation), refused here when padding is omitted and otherwise
134    /// when writing reaches them. Any error opening or reading an input.
135    pub fn new(
136        inputs: &[CreationInput],
137        options: &CreationOptions,
138        destination: Destination,
139    ) -> Result<Self, ObservationError> {
140        if inputs.is_empty() {
141            return Err(request_error("a MeasurementSet is created from an input"));
142        }
143        let publication = match destination {
144            Destination::New(path) => {
145                if path.symlink_metadata().is_ok() {
146                    return Err(request_error(format!(
147                        "{} already exists; a new MeasurementSet is created at a free path",
148                        path.display()
149                    )));
150                }
151                Publication::New(path)
152            }
153            Destination::Replace(path) => Publication::Replace(Replacing::hold(&path)?),
154        };
155        let mut ledger = Ledger::new(options.memory_bytes);
156        // The plan keeps a copy of the HISTORY rows to write.
157        ledger.admit("the HISTORY rows", history_bytes(&options.history))?;
158        let inputs = inputs
159            .iter()
160            .map(|input| Input::open(input, &mut ledger))
161            .collect::<Result<Vec<_>, _>>()?;
162        let groups = selected_groups(&inputs, options.data_groups.as_deref())?;
163        let names = groups
164            .iter()
165            .map(|group| column(group))
166            .collect::<Result<Vec<_>, _>>()?;
167        let mut found = Vec::new();
168        let mut pointing = Vec::with_capacity(inputs.len());
169        for input in &inputs {
170            let (path, observation) = (&input.path, input.observation());
171            let catalogue = observation.catalogue();
172            // An MSv2's scan opens it again and loads the subtables its
173            // catalogue was read from.
174            let scans = losses::scan_bytes(catalogue)
175                + input.msv2().map_or(0, Msv2Observation::tables_bytes);
176            found.extend(ledger.during("an input's loss scans", scans, |_| {
177                losses::inventory(path, catalogue, &groups, losses::Store::Msv2)
178            })?);
179            let scan = pointing_plan::planning_bytes(path, observation, options.batch_bytes)?;
180            let (plan, lost) = ledger.during("an input's pointing scan", scan, |_| {
181                pointing_plan::plan(path, observation, losses::Store::Msv2, options.batch_bytes)
182            })?;
183            found.extend(lost);
184            pointing.push(plan);
185        }
186        // One POINTING holds one frame: the inputs' must agree.
187        let mut frames = pointing.iter().filter_map(PointingPlan::frame);
188        if let Some(first) = frames.next()
189            && frames.any(|frame| frame != first)
190        {
191            return Err(ObservationError::Unsupported {
192                what: "writing MSv2 pointing of inputs in different frames",
193            });
194        }
195        found.sort();
196        found.dedup();
197        losses::refuse_unaccepted(&found, &options.accept)?;
198        for input in &inputs {
199            refuse_unwritable(input.observation().catalogue())?;
200        }
201        // A conversion writes the ephemerides its fields follow (#942);
202        // combining them, as concat merges ephemeris tables, is not done.
203        if inputs.len() > 1
204            && inputs.iter().any(|input| {
205                let catalogue = input.observation().catalogue();
206                catalogue
207                    .fields
208                    .iter()
209                    .any(|field| field.ephemeris.is_some())
210            })
211        {
212            return Err(ObservationError::Unsupported {
213                what: "combining fields that follow an ephemeris from several inputs",
214            });
215        }
216        let merging = merge_bound(
217            inputs
218                .iter()
219                .map(|input| catalogue_bytes(input.observation().catalogue()))
220                .sum(),
221        );
222        ledger.admit("merging the catalogues", merging)?;
223        let combined = combine::combine(&inputs, &groups, &options.tolerance)?;
224        ledger.trim("the merged catalogue", merging, combined_bytes(&combined))?;
225        if options.padding == PaddingRows::Flagged
226            && combined
227                .contributors
228                .iter()
229                .any(|contributors| contributors.len() > 1)
230        {
231            return Err(request_error(
232                "partitions combined from several inputs are written with padding omitted \
233                 (PaddingRows::Omitted): a padded row of one input could fall on an \
234                 acquisition of another",
235            ));
236        }
237        refuse_unwritable(&combined.catalogue)?;
238        found.extend(combined.losses.iter().cloned());
239        found.extend(msv2_ephemeris::filled_rows(&combined.catalogue));
240        found.extend(
241            msv2_ephemeris::followed(&combined.catalogue)
242                .0
243                .into_iter()
244                .filter_map(|ephemeris| losses::extrapolated(ephemeris)),
245        );
246        let grids = occupancy::grids(&inputs, &combined, &mut ledger)?;
247        // What writing the subtables defaults or leaves out, found by writing
248        // them in memory, and where they put each partition; the catalogue
249        // read back from them stays.
250        let promised = written_bound(&combined);
251        ledger.admit("the written catalogue", promised)?;
252        let rehearsing = subtables_bytes(&combined.catalogue);
253        let msv2_tables::Rehearsal {
254            layout,
255            losses: rehearsed,
256            read_back,
257        } = ledger.during("the subtables rehearsed", rehearsing, |_| {
258            msv2_tables::rehearse(&combined.catalogue, &grids.spans, &combined.intents)
259        })?;
260        found.extend(rehearsed);
261        found.sort();
262        found.dedup();
263        losses::refuse_unaccepted(&found, &options.accept)?;
264        let columns: Vec<(OptionalMainColumn, Option<String>)> = names
265            .into_iter()
266            .zip(&combined.units)
267            .map(|(name, units)| (name, unit(units)))
268            .collect();
269        let output = written(&combined, &layout, read_back);
270        ledger.trim("the written catalogue", promised, catalogue_bytes(&output))?;
271        let occupancy = grids.occupancy(
272            &inputs,
273            &combined,
274            (options.padding, options.batch_bytes),
275            &mut ledger,
276        )?;
277        if occupancy.rows.iter().sum::<usize>() == 0 {
278            return Err(request_error(
279                "the inputs hold nothing to write as an MSv2 row",
280            ));
281        }
282        let data_columns: Vec<&'static str> =
283            columns.iter().map(|(column, _)| column.name()).collect();
284        let writing = writing::admit(
285            &mut ledger,
286            (&inputs, &combined, &pointing),
287            (&data_columns, &occupancy, &options.history),
288            options.batch_bytes,
289        )?;
290        Ok(Self {
291            inputs,
292            pointing,
293            combined,
294            columns,
295            losses: found,
296            occupancy,
297            output,
298            main: writing.plan,
299            charges: writing.charges,
300            history: options.history.clone(),
301            padding: options.padding,
302            batch_bytes: options.batch_bytes,
303            creation_date: options.creation_date.clone(),
304            publication,
305        })
306    }

344    /// Write the planned MeasurementSet, by one thread, in a staging
345    /// directory beside its destination, then publish it: renamed into a
346    /// new path, or exchanged with the MeasurementSet it replaces, which is
347    /// then deleted. No input is written, except a replaced one, whose
348    /// directory then holds the new MeasurementSet.
349    ///
350    /// # Errors
351    ///
352    /// A destination that appeared or changed since the plan was made.
353    /// Weights that vary over channels, when padding was not omitted. Then
354    /// any error reading an input or writing the MeasurementSet, which then
355    /// is not published: the stage is removed, and a replaced
356    /// MeasurementSet is as it was.
357    pub fn write(self) -> Result<CreationReport, ObservationError> {
358        let stage = match &self.publication {
359            Publication::New(path) => Stage::reserve(path, "casa-rs-creating")?,
360            Publication::Replace(replacing) => {
361                Stage::reserve(&replacing.destination, "casa-rs-replacing")?
362            }
363        };
364        let created = self.create(stage.path());
365        let Self {
366            inputs,
367            publication,
368            losses,
369            ..
370        } = self;
371        // The inputs' read handles go first: a replaced MeasurementSet's
372        // directory is exchanged and deleted.
373        drop(inputs);
374        let published = created.and_then(|report| match publication {
375            Publication::New(path) => stage.publish_new(&path).map(|()| report),
376            Publication::Replace(replacing) => replacing
377                .publish(&stage)
378                .map(|leftover| CreationReport { leftover, ..report }),
379        });
380        match published {
381            Ok(report) => {
382                stage.release();
383                Ok(CreationReport { losses, ..report })
384            }
385            Err(error) => {
386                // The stage, which the plan reserved, is the only output.
387                stage.remove();
388                Err(error)
389            }
390        }
391    }

393    /// Create the MeasurementSet at `staging`: its subtables, then MAIN's
394    /// rows from a lockstep read of every input partition.
395    fn create(&self, staging: &Path) -> Result<CreationReport, ObservationError> {
396        let catalogue = &self.combined.catalogue;
397        let optional: Vec<OptionalMainColumn> =
398            self.columns.iter().map(|(column, _)| *column).collect();
399        let builder = msv2_tables::builder(catalogue, &optional);
400        let mut ms = MeasurementSet::create(staging, builder).map_err(storage)?;
401        // The MAIN writers write into the directory before the save.
402        fs::create_dir_all(staging).map_err(storage)?;
403        for (column, unit) in &self.columns {
404            if let Some(unit) = unit {
405                TableQuantumDesc::with_unit(column.name(), unit)
406                    .write(ms.main_table_mut())
407                    .map_err(storage)?;
408            }
409        }
410        let (layout, _) = msv2_tables::populate(
411            &mut ms,
412            catalogue,
413            &self.occupancy.spans,
414            &self.combined.intents,
415        )?;
416        msv2_ephemeris::mark_fields(&mut ms, catalogue)?;
417        let pointed: Vec<Pointed<'_>> = self
418            .inputs
419            .iter()
420            .zip(&self.pointing)
421            .zip(&self.combined.rows)
422            .map(|((input, plan), map)| Pointed {
423                observation: input.observation(),
424                plan,
425                antennas: &map.antennas,
426            })
427            .collect();
428        msv2_pointing::write(&mut ms, &pointed)?;
429        super::msv2_identity::history(&mut ms, &self.history)?;
430        let names = optional.iter().map(|column| column.name()).collect();
431        let mut main = MainRows::start(
432            (staging, self.main.clone()),
433            catalogue,
434            names,
435            &self.occupancy,
436            layout,
437            self.padding,
438        )?;
439        let mut report = CreationReport::default();
440        let mut row_states = RowStates::default();
441        for (input, map) in self.inputs.iter().zip(&self.combined.rows) {
442            let observation = input.observation();
443            let mut ids = observation.catalogue().partition_ids();
444            ids.sort_by_key(|id| map.partitions[id.0 as usize].output);
445            for id in ids {
446                let placement = &map.partitions[id.0 as usize];
447                let per_row = match (&placement.state, input.msv2()) {
448                    (StateIds::PerRow(_), Some(msv2)) => Some(msv2),
449                    _ => None,
450                };
451                row_states.restart();
452                let read = lockstep::read(
453                    observation,
454                    id,
455                    &placement.groups,
456                    Roles::ALL,
457                    (self.batch_bytes, None),
458                    |batches| {
459                        let states = per_row
460                            .map(|msv2| row_states.of(msv2, id, &batches[0]))
461                            .transpose()?;
462                        main.push(map, batches, states)
463                    },
464                )?;
465                report.partitions += 1;
466                report.batches += read.batches;
467                report.samples += read.samples;
468            }
469        }
470        report.rows = main.finish(&mut ms)?;
471        msv2_ephemeris::write_tables(staging, catalogue, &self.creation_date)?;
472        Ok(report)
473    }
474}

476/// The data groups to write: `requested`, or every input's, `base` first.
477fn selected_groups(
478    inputs: &[Input],
479    requested: Option<&[String]>,
480) -> Result<Vec<String>, ObservationError> {
481    let groups: Vec<String> = match requested {
482        Some(requested) => requested.to_vec(),
483        None => {
484            let mut groups = vec!["base".to_string()];
485            for input in inputs {
486                for entry in &input.observation().catalogue().partitions {
487                    for group in &entry.data_groups {
488                        if !groups.contains(&group.name) {
489                            groups.push(group.name.clone());
490                        }
491                    }
492                }
493            }
494            groups
495        }
496    };
497    if groups.first().map(String::as_str) != Some("base") {
498        return Err(request_error(
499            "a MeasurementSet is written with the base data group first",
500        ));
501    }
502    if let Some(group) = groups
503        .iter()
504        .enumerate()
505        .find_map(|(index, group)| groups[..index].contains(group).then_some(group))
506    {
507        return Err(request_error(format!(
508            "data group {group} is selected twice"
509        )));
510    }
511    Ok(groups)
512}

514/// The MSv2 column of a data group, as XRADIO's converter names them.
515fn column(group: &str) -> Result<OptionalMainColumn, ObservationError> {
516    match group {
517        "base" => Ok(OptionalMainColumn::Data),
518        "corrected" => Ok(OptionalMainColumn::CorrectedData),
519        "model" => Ok(OptionalMainColumn::ModelData),
520        other => Err(request_error(format!(
521            "data group {other} has no MSv2 column: base (DATA), corrected \
522             (CORRECTED_DATA) and model (MODEL_DATA) have"
523        ))),
524    }
525}

527/// The `QuantumUnits` a column of `units` declares: none for undeclared
528/// units, as CASA's `DATA` declares none.
529fn unit(units: &VisibilityUnits) -> Option<String> {
530    match units {
531        VisibilityUnits::Undeclared => None,
532        VisibilityUnits::Jansky => Some("Jy".to_string()),
533        VisibilityUnits::Declared(unit) => Some(unit.clone()),
534    }

537/// Refuse what the MSv2 written cannot say: another time scale than UTC
538/// (`TIME` is written with the MeasurementSet's default reference), a feed
539/// other than the one `FEED` row per antenna the writer writes, a window
540/// without channels, a field in another frame than J2000, what its
541/// subtables cannot say ([`msv2_tables::refuse_unwritable`]) and an
542/// ephemeris its table cannot hold ([`msv2_ephemeris::refuse_unwritable`]).
543fn refuse_unwritable(catalogue: &Catalogue) -> Result<(), ObservationError> {
544    if catalogue
545        .fields
546        .iter()
547        .any(|field| field.phase_centre.refer() != DirectionRef::J2000)
548    {
549        return Err(ObservationError::Unsupported {
550            what: "writing MSv2 fields in another frame than J2000",
551        });
552    }
553    for entry in &catalogue.partitions {
554        if entry.frequencies_hz.is_empty() {
555            return Err(ObservationError::Unsupported {
556                what: "writing an MSv2 spectral window without channels",
557            });
558        }
559        if entry.time_scale != EpochRef::UTC {
560            return Err(ObservationError::Unsupported {
561                what: "writing MSv2 times in another scale than UTC",
562            });
563        }
564        if entry.feeds.is_some_and(|feeds| feeds != [0, 0]) {
565            return Err(ObservationError::Unsupported {
566                what: "writing MSv2 rows of feeds other than 0",
567            });
568        }
569    }
570    msv2_tables::refuse_unwritable(catalogue)?;
571    msv2_ephemeris::refuse_unwritable(catalogue)
572}

574/// What reopening the written MeasurementSet gives of `combined`, whose
575/// subtables `layout` places and read back as `subtables` with MAIN's
576/// `uvw_frame`: every entity as the adapter reads it from the subtables
577/// written (the members written as MSv2's defaults, and the states written
578/// for intents, included); each partition naming the `SPECTRAL_WINDOW` row
579/// and the states and feeds its rows are written with, its data groups' UVW
580/// in the frame MAIN's keywords give, in the order the MSv2 reader keys
581/// them (data description, `OBS_MODE`, observation).
582fn written(
583    combined: &Combined,
584    layout: &msv2_tables::Layout,
585    (subtables, uvw_frame): (Catalogue, casa_observation::UvwFrame),
586) -> Catalogue {
587    let mut catalogue = Catalogue {
588        scans: combined.catalogue.scans.clone(),
589        partitions: combined.catalogue.partitions.clone(),
590        ..subtables
591    };
592    // The ephemerides fields follow are written as tables of their own
593    // beside the subtables rehearsed (`msv2_ephemeris`).
594    for (field, merged) in catalogue.fields.iter_mut().zip(&combined.catalogue.fields) {
595        field.ephemeris.clone_from(&merged.ephemeris);
596    }
597    for (output, entry) in catalogue.partitions.iter_mut().enumerate() {
598        entry.window = casa_observation::WindowId(
599            u32::try_from(layout.spectral_window[output]).expect("a written window row"),
600        );
601        // Every row is written with feeds 0, which `refuse_unwritable` has
602        // checked an MSv2 input states.
603        entry.feeds = Some([0, 0]);
604        for group in &mut entry.data_groups {
605            group.uvw_frame = uvw_frame;
606        }
607        let mut states: Vec<casa_observation::StateId> = combined.contributors[output]
608            .iter()
609            .flat_map(
610                |(input, id)| match &combined.rows[*input].partitions[id.0 as usize].state {
611                    StateIds::One(state) => vec![*state],
612                    StateIds::Intents => layout.intent_states[output].into_iter().collect(),
613                    StateIds::PerRow(states) => states.clone(),
614                },
615            )
616            .filter_map(|state| u32::try_from(state).ok().map(casa_observation::StateId))
617            .collect();
618        states.sort_unstable();
619        states.dedup();
620        entry.states = states;
621    }
622    let mut order: Vec<usize> = (0..catalogue.partitions.len()).collect();
623    order.sort_by(|a, b| {
624        (
625            layout.data_description[*a],
626            &combined.modes[*a],
627            catalogue.partitions[*a].observation,
628        )
629            .cmp(&(
630                layout.data_description[*b],
631                &combined.modes[*b],
632                catalogue.partitions[*b].observation,
633            ))
634    });
635    let partitions = std::mem::take(&mut catalogue.partitions);
636    catalogue.partitions = order
637        .into_iter()
638        .map(|index| partitions[index].clone())
639        .collect();
640    catalogue
641}

643/// An in-memory MeasurementSet holding no rows and the subtables an MSv2
644/// converted from `observation` would hold, and the `SPECTRAL_WINDOW` row
645/// of each partition: the namespace a calibration table solved on a store
646/// without one (MSv4) persists. Antennas, fields and sessions keep their
647/// catalogue order.
648///
649/// # Errors
650///
651/// As [`CreationPlan::new`] for what an MSv2's subtables cannot say.
652#[cfg(feature = "msv4")]
653pub(crate) fn catalogue_subtables(
654    observation: &dyn casa_observation::Observation,
655) -> Result<(MeasurementSet, Vec<i32>), ObservationError> {
656    let catalogue = observation.catalogue();
657    refuse_unwritable(catalogue)?;
658    let spans = match catalogue
659        .partitions
660        .first()
661        .and_then(|entry| entry.data_groups.first())
662    {
663        Some(group) => occupancy::spans(observation, &group.name)?,
664        None => super::msv2_rows::Spans {
665            time_range: [0.0, 0.0],
666            observation_ranges: vec![[0.0, 0.0]; catalogue.observations.len()],
667            field_first: vec![None; catalogue.fields.len()],
668        },
669    };
670    let mut ms =
671        MeasurementSet::create_memory(msv2_tables::builder(catalogue, &[])).map_err(storage)?;
672    let intents = super::msv2_identity::intent_partitions(catalogue);
673    let (layout, _) = msv2_tables::populate(&mut ms, catalogue, &spans, &intents)?;
674    Ok((ms, layout.spectral_window))
675}
