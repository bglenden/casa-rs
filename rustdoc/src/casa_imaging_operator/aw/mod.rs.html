45/// A native A/W request or numerical calculation could not be completed.
46#[derive(Clone, Copy, Debug, Error, PartialEq, Eq)]
47pub enum NativeAwGenerationError {
48    /// Dish samples are not finite, regularly spaced, or physically valid.
49    #[error("EVLA dish surface requires finite, uniformly spaced radius/height/slope samples")]
50    InvalidSurface,
51    /// The requested receiver band is outside the implemented EVLA proof.
52    #[error("native EVLA aperture supports only L, S and C receiver bands (0.9 to 8 GHz)")]
53    UnsupportedFrequency,
54    /// Aperture geometry or numerical sampling is invalid.
55    #[error(
56        "native aperture grid requires a finite positive cell size, frequency, and even extent"
57    )]
58    InvalidGrid,
59    /// The caller did not supply the declared numerical workspace.
60    #[error("native A/W generation workspace does not fit the requested grid")]
61    WorkspaceMismatch,
62    /// A generated value or normalization is invalid.
63    #[error("native A/W generation produced an invalid numerical value")]
64    InvalidNumerics,
65}

68#[derive(Debug, Error)]
69pub enum AwCatalogError {

76        detail: String,

91fn cache_error(path: impl AsRef<Path>, detail: impl Into<String>) -> AwCatalogError {

140/// One (PA, frequency, w) group: its Mueller elements in the catalog's
141/// element order.
142#[derive(Clone, Debug)]
143struct Group {
144    cells: Vec<CellFiles>,
145    /// The largest declared half support over the group's imaging cells:
146    /// the imaging dense cell is padded to it, and `AWVisResampler`'s
147    /// `onGrid` test for a data row uses it (`CF Support: 4 (6)` in CASA's
148    /// log is the imaging support with the weight support in brackets).
149    imaging_half_support: [u16; 2],
150    /// The same over the weight cells, which the PSF and weight image use.
151    weight_half_support: [u16; 2],
152    /// The smallest declared half support over both kinds: the row filter
153    /// keeps every row some cell of the group fits.
154    smallest_half_support: [u16; 2],
155}
156
157impl Group {
158    fn new(cells: Vec<CellFiles>) -> Self {
159        let largest = |select: fn(&CellFiles) -> &CellHeader| {
160            cells.iter().map(select).fold([0_u16; 2], |half, header| {
161                [
162                    half[0].max(header.support[0]),
163                    half[1].max(header.support[1]),
164                ]
165            })
166        };
167        let imaging_half_support = largest(|cell| &cell.imaging);
168        let weight_half_support = largest(|cell| &cell.weight);
169        let smallest_half_support = cells
170            .iter()
171            .flat_map(|cell| [&cell.imaging, &cell.weight])
172            .fold([u16::MAX; 2], |half, header| {
173                [
174                    half[0].min(header.support[0]),
175                    half[1].min(header.support[1]),
176                ]
177            });
178        Self {
179            cells,
180            imaging_half_support,
181            weight_half_support,
182            smallest_half_support,
183        }
184    }
185}

187/// The support `locate_sample` tests a row against: the smallest cell the
188/// gridding group or the prediction group can select. Every mode re-tests
189/// the support of the cell it reads (`AWVisResampler::DataToGrid` and
190/// `GridToData` call `onGrid` with the selected cell's support), so the
191/// filter only has to keep every row CASA grids with any of them.
192fn placement_half_support(gridding: &Group, prediction: &Group) -> [u16; 2] {
193    let (g, p) = (
194        gridding.smallest_half_support,
195        prediction.smallest_half_support,
196    );
197    [g[0].min(p[0]), g[1].min(p[1])]
198}

225/// The AW-projection kernel set over a CASA convolution-function cache.
226///
227/// Cells are keyed on the parallactic-angle cell, the frequency cell and
228/// the w-plane (`CfKey::group`); the Mueller elements live inside the cell
229/// as planes, routed by [`MuellerRouting`] with CASA's selection of both
230/// the cell and its input polarization (`AWVisResampler`). Prediction
231/// divides once by the kernel sum and `sumwt += W·|N|`
232/// ([`KernelNormalisation::KernelSum`]); the pointing ramp applies to
233/// every row ([`ConvolutionFunctionSet::pointing_ramp`]). A row whose
234/// parallactic angle lies between cells takes the nearest cell; CASA would
235/// rotate it (`AWConvFunc::rotateCF`, `rotatepastep`), which the frozen
236/// 360° steps of the EVLA cache make the same choice.
237///
238/// # Panics
239///
240/// [`ConvolutionFunctionSet::taps`] and [`ConvolutionFunctionSet::weight_taps`]
241/// are infallible: a cell image that was validated at open but cannot be
242/// read when a row first needs it is a catalog broken under the run, and
243/// the catalog panics with the cell's path rather than gridding without it.
244pub struct AwCatalog {
245    root: PathBuf,
246    indexing: AwIndexing,
247    pa_deg: Vec<f64>,
248    frequencies_hz: Vec<f64>,
249    w_values: Vec<f64>,
250    w_increment: f64,
251    mueller_elements: Vec<u32>,
252    groups: Vec<Group>,
253    oversampling: u16,
254    max_half_support: [u16; 2],
255    mueller: MuellerRouting,
256    correction: ImageCorrection,
257    lru: Mutex<Lru>,
258}

273impl AwCatalog {
274    /// Open the cache at `root`: read every `CFS_*`/`WTCFS_*` image's
275    /// coordinates and misc info (`CFCache::fillCFListFromDisk`), sort the
276    /// unique parallactic angles, frequencies, w values and Mueller
277    /// elements, and build the routing of `polarization`'s correlations
278    /// over `geometry`'s grid. Pixels are read on demand, at most
279    /// `bound_bytes` of cells resident.
280    ///
281    /// # Errors
282    ///
283    /// The directory cannot be listed, a cell is unreadable or malformed,
284    /// a cell lacks its partner, a (PA, frequency, w) group lacks a Mueller
285    /// element, or the catalog has no element for a selected correlation.
286    pub fn open_casa(
287        root: impl AsRef<Path>,
288        indexing: AwIndexing,
289        geometry: &GridGeometry,
290        polarization: &PolarizationRouting,
291        bound_bytes: usize,
292    ) -> Result<Self, AwCatalogError> {
293        let root = root.as_ref().to_path_buf();
294        let entries = std::fs::read_dir(&root)
295            .map_err(|error| cache_error(&root, format!("cannot list: {error}")))?;
296        let mut imaging = HashMap::new();
297        let mut weight = HashMap::new();
298        for entry in entries {
299            let entry =
300                entry.map_err(|error| cache_error(&root, format!("cannot list: {error}")))?;
301            let name = entry.file_name();
302            let Some(name) = name.to_str() else {
303                continue;
304            };
305            if let Some(rest) = name.strip_prefix(WEIGHT_PREFIX) {
306                weight.insert(rest.to_string(), read_header(&entry.path())?);
307            } else if let Some(rest) = name.strip_prefix(IMAGING_PREFIX) {
308                imaging.insert(rest.to_string(), read_header(&entry.path())?);
309            }
310        }
311        if imaging.is_empty() {
312            return Err(cache_error(&root, "no CFS_ cell images"));
313        }
314        // A native cell's working field is the image field times the
315        // oversampling: `working_size · working_increment = sampling · n · Δx`.
316        let [grid_n, _] = geometry.grid_shape();
317        let image_increment = geometry.image().increment_rad[0].abs();
318        for header in imaging.values() {
319            let Some(working_field) = header.working_field_rad else {
320                continue;
321            };
322            let expected = image_increment * f64::from(header.sampling) * grid_n as f64;
323            if (working_field - expected).abs() > 1.0e-6 * expected {
324                return Err(cache_error(
325                    &header.path,
326                    format!(
327                        "generated for a working field of {working_field} rad, the image needs \
328                         {expected} rad: the cache belongs to another image geometry"
329                    ),
330                ));
331            }
332        }
333        let mut cells = Vec::with_capacity(imaging.len());
334        for (rest, header) in imaging {
335            let partner = weight
336                .remove(&rest)
337                .ok_or_else(|| cache_error(&header.path, "paired WTCFS_ cell is missing"))?;
338            if partner.mueller != header.mueller
339                || partner.w_value.to_bits() != header.w_value.to_bits()
340                || partner.pa_deg.to_bits() != header.pa_deg.to_bits()
341                || partner.sampling != header.sampling
342            {
343                return Err(cache_error(
344                    &partner.path,
345                    "WTCFS_ cell disagrees with its CFS_ cell",
346                ));
347            }
348            cells.push(CellFiles {
349                imaging: header,
350                weight: partner,
351            });
352        }
353        if let Some((_, orphan)) = weight.into_iter().next() {
354            return Err(cache_error(&orphan.path, "paired CFS_ cell is missing"));
355        }
356        let unique = |values: Vec<f64>| {
357            let mut values = values;
358            values.sort_by(f64::total_cmp);
359            values.dedup_by(|a, b| a.to_bits() == b.to_bits());
360            values
361        };
362        let pa_deg = unique(cells.iter().map(|cell| cell.imaging.pa_deg).collect());
363        let frequencies_hz = unique(cells.iter().map(|cell| cell.imaging.frequency_hz).collect());
364        let w_values = unique(cells.iter().map(|cell| cell.imaging.w_value).collect());
365        let mut mueller_elements = cells
366            .iter()
367            .map(|cell| cell.imaging.mueller)
368            .collect::<Vec<_>>();
369        mueller_elements.sort_unstable();
370        mueller_elements.dedup();
371        let w_increment = cells[0].imaging.w_increment;
372        let oversampling = cells[0].imaging.sampling;
373        if cells.iter().any(|cell| {
374            cell.imaging.w_increment.to_bits() != w_increment.to_bits()
375                || cell.imaging.sampling != oversampling
376        }) {
377            return Err(cache_error(&root, "cells disagree on WIncr or Sampling"));
378        }
379        let index_of = |values: &[f64], value: f64| {
380            values
381                .iter()
382                .position(|listed| listed.to_bits() == value.to_bits())
383                .expect("listed value")
384        };
385        let groups_len = pa_deg.len() * frequencies_hz.len() * w_values.len();
386        let mut groups: Vec<Vec<Option<CellFiles>>> = (0..groups_len)
387            .map(|_| vec![None; mueller_elements.len()])
388            .collect();
389        let mut max_half_support = [0_u16; 2];
390        for cell in cells {
391            let group = (index_of(&pa_deg, cell.imaging.pa_deg) * frequencies_hz.len()
392                + index_of(&frequencies_hz, cell.imaging.frequency_hz))
393                * w_values.len()
394                + index_of(&w_values, cell.imaging.w_value);
395            let plane = mueller_elements
396                .binary_search(&cell.imaging.mueller)
397                .expect("listed element");
398            if groups[group][plane].is_some() {
399                return Err(cache_error(&cell.imaging.path, "duplicate cell key"));
400            }
401            for (axis, largest) in max_half_support.iter_mut().enumerate() {
402                *largest = (*largest)
403                    .max(cell.imaging.support[axis])
404                    .max(cell.weight.support[axis]);
405            }
406            groups[group][plane] = Some(cell);
407        }
408        let groups = groups
409            .into_iter()
410            .map(|planes| {
411                let cells = planes
412                    .into_iter()
413                    .collect::<Option<Vec<_>>>()
414                    .ok_or_else(|| {
415                        cache_error(&root, "a (PA, frequency, w) group lacks a Mueller element")
416                    })?;
417                Ok(Group::new(cells))
418            })
419            .collect::<Result<Vec<_>, AwCatalogError>>()?;
420        let mueller = routing(&mueller_elements, polarization).ok_or_else(|| {
421            cache_error(
422                &root,
423                "selected correlations have missing or overlapping Mueller routes",
424            )
425        })?;
426        let [nx, ny] = geometry.grid_shape();
427        // `AWProjectFT::getImage` and `getWeightImage` divide every image
428        // by the sampling sinc `sin(x)/x`, `x = π(i − n/2)/(n·sampling)`;
429        // the spheroidal lives in the cells under `psterm`, and
430        // `initializeToVis` leaves the model uncorrected (its `sincConv`
431        // is reset to one).
432        let sampling_correction = |len: usize| {
433            (0..len)
434                .map(|index| {
435                    if index == len / 2 {
436                        return 1.0;
437                    }
438                    let x = std::f64::consts::PI * (index as f64 - (len / 2) as f64)
439                        / (len as f64 * f64::from(oversampling));
440                    x / x.sin()
441                })
442                .collect::<Vec<_>>()
443        };
444        Ok(Self {
445            root,
446            indexing,
447            pa_deg,
448            frequencies_hz,
449            w_values,
450            w_increment,
451            mueller_elements,
452            groups,
453            oversampling,
454            max_half_support,
455            mueller,
456            correction: ImageCorrection::split(
457                sampling_correction(nx),
458                sampling_correction(ny),
459                vec![1.0; nx],
460                vec![1.0; ny],
461            ),
462            lru: Mutex::new(Lru {
463                loaded: HashMap::new(),
464                clock: 0,
465                bytes: 0,
466                bound: bound_bytes,
467            }),
468        })
469    }

471    /// Generate every cell of `input` natively into `root` as CASA-format
472    /// images (`CFS_<pa>_0_CF_<f>_<w>_<m>.im` and the `WTCFS_` partner, one
473    /// baseline type) so [`AwCatalog::open_casa`] serves them. Existing
474    /// cells of the same names are replaced.
475    ///
476    /// # Errors
477    ///
478    /// Generation fails or an image cannot be written.
479    pub fn generate_native(
480        root: impl AsRef<Path>,
481        input: &NativeAwRequestInput,
482        only_missing: bool,
483    ) -> Result<(), AwCatalogError> {
484        let root = root.as_ref();
485        std::fs::create_dir_all(root)
486            .map_err(|error| cache_error(root, format!("cannot create: {error}")))?;
487        let model = EvlaApertureModel::new(input.surface.clone());
488        let grid = input.grid;
489        let mut workspace: Option<EvlaAwWorkspace> = None;
490        for (ipa, &pa) in input.pa_values.iter().enumerate() {
491            for (inu, group) in input.frequencies.iter().enumerate() {
492                let frequency = group.cf_frequency_hz;
493                let conjugate = conjugate_frequency(frequency, input.reference_frequency_hz);
494                // `AWConvFunc::makeConvFunction2`: the conjugate beam uses the
495                // nearest listed frequency to `√(2f₀² − f²)`, the first on a tie
496                // (`SynthesisUtils::nearestValue`).
497                let conjugate_frequency = if input.terms.conjugate_beams {
498                    nearest(
499                        &input
500                            .frequencies
501                            .iter()
502                            .map(|group| group.cf_frequency_hz)
503                            .collect::<Vec<_>>(),
504                        conjugate,
505                    )
506                    .1
507                } else {
508                    frequency
509                };
510                // Only the W screen depends on w: the aperture and the weight
511                // cell are formed once per (PA, frequency, Mueller element).
512                for (im, &mueller) in input.mueller_elements.iter().enumerate() {
513                    for (iw, &w) in input.w_values.iter().enumerate() {
514                        let suffix = format!("{ipa}_0_CF_{inu}_{iw}_{im}.im");
515                        let imaging_path = root.join(format!("{IMAGING_PREFIX}{suffix}"));
516                        let weight_path = root.join(format!("{WEIGHT_PREFIX}{suffix}"));
517                        if only_missing && imaging_path.is_dir() && weight_path.is_dir() {
518                            continue;
519                        }
520                        let request = EvlaAwCellRequest {
521                            size: grid.size,
522                            sky_increment_rad: grid.sky_increment_rad,
523                            frequency_hz: frequency,
524                            conjugate_frequency_hz: conjugate_frequency,
525                            w_wavelengths: w,
526                            parallactic_angle_rad: pa,
527                            mueller,
528                            oversampling: grid.oversampling,
529                            prolate_spheroidal: input.terms.prolate_spheroidal,
530                            aperture: input.terms.aperture,
531                        };
532                        let workspace = match &mut workspace {
533                            Some(workspace) => workspace,
534                            None => workspace.insert(EvlaAwWorkspace::new(model.clone(), request)?),
535                        };
536                        workspace.prepare(request)?;
537                        let pair = workspace.realize(request)?;
538                        let misc = CellMisc {
539                            sampling: grid.oversampling as f64,
540                            pa_deg: pa.to_degrees(),
541                            mueller,
542                            w_value: w,
543                            w_increment: input.w_increment,
544                            conjugate_frequency,
545                            conjugate_mueller: MUELLER_ELEMENTS - 1 - mueller as u32,
546                            diameter_m: input.antenna_diameter_m,
547                            sky_increment_rad: grid.sky_increment_rad[1].abs(),
548                            working_size: grid.size,
549                        };
550                        write_cell(&imaging_path, &pair.imaging, frequency, grid, &misc)?;
551                        write_cell(&weight_path, &pair.weight, frequency, grid, &misc)?;
552                    }
553                }
554            }
555        }
556        Ok(())
557    }

559    /// How many of the cells `input` names exist under `root` (both the
560    /// `CFS_` and `WTCFS_` images), and how many it names.
561    pub fn native_cells_present(
562        root: impl AsRef<Path>,
563        input: &NativeAwRequestInput,
564    ) -> (usize, usize) {
565        let root = root.as_ref();
566        let mut present = 0;
567        let mut expected = 0;
568        for ipa in 0..input.pa_values.len() {
569            for inu in 0..input.frequencies.len() {
570                for iw in 0..input.w_values.len() {
571                    for im in 0..input.mueller_elements.len() {
572                        let suffix = format!("{ipa}_0_CF_{inu}_{iw}_{im}.im");
573                        expected += 1;
574                        if root.join(format!("{IMAGING_PREFIX}{suffix}")).is_dir()
575                            && root.join(format!("{WEIGHT_PREFIX}{suffix}")).is_dir()
576                        {
577                            present += 1;
578                        }
579                    }
580                }
581            }
582        }
583        (present, expected)
584    }

586    /// Remove every `CFS_` and `WTCFS_` cell under `root`, so a regenerated
587    /// catalog carries no member of an earlier request.
588    pub fn clear_native(root: impl AsRef<Path>) -> Result<(), AwCatalogError> {
589        let root = root.as_ref();
590        if !root.is_dir() {
591            return Ok(());
592        }
593        let entries = std::fs::read_dir(root)
594            .map_err(|error| cache_error(root, format!("cannot list: {error}")))?;
595        for entry in entries {
596            let entry =
597                entry.map_err(|error| cache_error(root, format!("cannot list: {error}")))?;
598            let name = entry.file_name();
599            let Some(name) = name.to_str() else {
600                continue;
601            };

665    /// Bytes the catalog holds at most while it lives with `holders`
666    /// workers fetching cells: its index and corrections; the cache's bound
667    /// and one group past it, which a fetch inserts before the cache
668    /// evicts; and for each holder the largest group, which its
669    /// [`CellHold`] keeps alive after eviction, and the planes it reads to
670    /// build one (each read as an image array and its copy).
671    #[must_use]
672    pub fn charge_bytes(&self, holders: usize) -> u64 {
673        let header = |header: &CellHeader| size_of::<CellHeader>() + header.path.as_os_str().len();
674        let index = self.groups.capacity() * size_of::<Group>()
675            + self
676                .groups
677                .iter()
678                .flat_map(|group| &group.cells)
679                .map(|cell| header(&cell.imaging) + header(&cell.weight))
680                .sum::<usize>()
681            + (self.pa_deg.len() + self.frequencies_hz.len() + self.w_values.len())
682                * size_of::<f64>()
683            + self.mueller_elements.len() * size_of::<u32>();
684        let group = |group: &Group| {
685            let planes = group.cells.len();
686            dense_cell_bytes(planes, self.oversampling, group.imaging_half_support)
687                + dense_cell_bytes(planes, self.oversampling, group.weight_half_support)
688        };
689        let read = |group: &Group| {
690            let plane = |header: &CellHeader| header.shape[0] * header.shape[1];
691            let imaging = group
692                .cells
693                .iter()
694                .map(|cell| plane(&cell.imaging))
695                .sum::<usize>();
696            let weight = group
697                .cells
698                .iter()
699                .map(|cell| plane(&cell.weight))
700                .sum::<usize>();
701            (2 * imaging.max(weight) * size_of::<Complex32>()) as u64
702        };
703        let largest = self.groups.iter().map(group).max().unwrap_or(0);
704        let reading = self.groups.iter().map(read).max().unwrap_or(0);
705        let bound = self.lru.lock().expect("cache lock").bound as u64;
706        index as u64
707            + self.correction.resident_bytes()
708            + bound
709            + (holders as u64 + 1) * largest
710            + holders as u64 * reading
711    }

713    /// The parallactic-angle cell of `pa_deg`: the nearest listed angle on
714    /// the circle (`CFCache::fillCFListFromDisk` selects cells within `dPA`
715    /// and `SynthesisUtils::stdNearestValue` picks among them).
716    #[must_use]
717    pub fn pa_cell(&self, pa_deg: f64) -> usize {
718        self.pa_deg
719            .iter()
720            .enumerate()
721            .fold((0, f64::INFINITY), |best, (index, listed)| {
722                let distance = circular_degrees(pa_deg - listed).abs();
723                if distance < best.1 {
724                    (index, distance)
725                } else {
726                    best
727                }
728            })
729            .0
730    }
731
732    /// The frequency cell of a row at `freq_hz`: the nearest listed
733    /// frequency to `√(2f₀² − f²)` under `conjbeams`, to `f` otherwise
734    /// (`CFBuffer::initMaps`, `nearestFreqNdx`).
735    #[must_use]
736    pub fn frequency_cell(&self, freq_hz: f64) -> usize {
737        let target = if self.indexing.conjugate_beams {
738            conjugate_frequency(freq_hz, self.indexing.image_reference_hz)
739        } else {
740            freq_hz
741        };
742        nearest(&self.frequencies_hz, target).0
743    }
744
745    /// The w-plane of `w_lambda` (`CFBuffer::nearestWNdx`, CAS-13191).
746    #[must_use]
747    pub fn w_cell(&self, w_lambda: f64) -> usize {
748        let index = (self.w_increment * w_lambda.abs()).sqrt().round() as usize;
749        index.min(self.w_values.len() - 1)
750    }
751
752    /// The group of a (PA, frequency, w) cell triple.
753    #[must_use]
754    pub fn group_index(&self, pa: usize, frequency: usize, w: usize) -> usize {
755        (pa * self.frequencies_hz.len() + frequency) * self.w_values.len() + w
756    }
757
758    /// Load group `group`'s cells, or fetch them from the cache, and park
759    /// them in `hold`.
760    fn loaded(&self, group: usize) -> (Arc<DenseCell>, Arc<DenseCell>) {
761        let mut lru = self.lru.lock().expect("cache lock");
762        lru.clock += 1;
763        let now = lru.clock;
764        if let Some(loaded) = lru.loaded.get_mut(&group) {
765            loaded.last_use = now;
766            return (loaded.imaging.clone(), loaded.weight.clone());
767        }
768        let mut loaded = load_group(&self.groups[group], self.oversampling)
769            .unwrap_or_else(|error| panic!("AW cell group {group} cannot be read: {error}"));
770        loaded.last_use = now;
771        let pair = (loaded.imaging.clone(), loaded.weight.clone());
772        lru.bytes += loaded.bytes();
773        lru.loaded.insert(group, loaded);
774        while lru.bytes > lru.bound && lru.loaded.len() > 1 {
775            let oldest = lru
776                .loaded
777                .iter()
778                .min_by_key(|(_, loaded)| loaded.last_use)
779                .map(|(group, _)| *group)
780                .expect("a loaded group");
781            if let Some(cell) = lru.loaded.remove(&oldest) {
782                lru.bytes -= cell.bytes();
783            }
784        }
785        pair
786    }

883/// `AWVisResampler`'s routes through diagonal Mueller cells. Each selected
884/// correlation (the `outer` loop of `DataToGrid` and `GridToData`) has the
885/// element `e` of its own hand (`RR, RL, LR, LL` → `0, 5, 10, 15`,
886/// `AWProjectFT::makeCFPolMap`) and the partner `15 − e` (`RR ↔ LL`,
887/// `RL ↔ LR`), or `e` again when the partner hand is not a selected
888/// correlation (`PolOuterProduct::makeConjPol2CFMat` keeps the direct
889/// entry). `getConvFunc_p` selects the own cell for `w > 0` and the
890/// partner's otherwise in the adjoint, the reverse in the forward
891/// transform (`GridToData` passes `mNdx` and `conjMNdx` swapped); the hand
892/// of the selected cell is `element % npol`, CASA's index into the
893/// visibility axis, so
894///
895/// `DataToGrid`: `grid[pol_map[outer]] += vis[hand(cell)] ⋆ cell`,
896/// `GridToData`: `vis[outer] += model[pol_map[hand(cell)]] ⋆ cell`.
897///
898/// `None` when a cell is missing or two outer correlations route one
899/// (grid, visibility) pair through different cells (a non-canonical
900/// three-correlation selection, where `element % npol` collides).
901fn routing(elements: &[u32], polarization: &PolarizationRouting) -> Option<MuellerRouting> {
902    let pol_map = polarization.pol_map();
903    let correlations = polarization.correlations();
904    let plane_of = |element: u32| {
905        elements
906            .binary_search(&element)
907            .ok()
908            .map(|plane| u8::try_from(plane).expect("planes fit u8"))
909    };
910    let selected = |element: u32| {
911        correlations
912            .iter()
913            .zip(pol_map)
914            .any(|(correlation, target)| {
915                target.is_some() && mueller_element(*correlation) == Some(element)
916            })
917    };
918    let hand = |element: u32| element as usize % pol_map.len();
919    let empty = || vec![vec![None; pol_map.len()]; polarization.grid_pols()];
920    let mut adjoint = [empty(), empty()];
921    let mut forward = [empty(), empty()];
922    for (outer, (correlation, target)) in correlations.iter().zip(pol_map).enumerate() {
923        let Some(gpol) = *target else {
924            continue;
925        };
926        let own = mueller_element(*correlation)?;
927        let partner = Some(MUELLER_ELEMENTS - 1 - own)
928            .filter(|partner| selected(*partner))
929            .unwrap_or(own);
930        for positive in [false, true] {
931            let cell = if positive { own } else { partner };
932            let plane = plane_of(cell)?;
933            if adjoint[usize::from(positive)][usize::from(gpol)][hand(cell)]
934                .replace(plane)
935                .is_some()
936            {
937                return None;
938            }
939            if let Some(input_grid) = pol_map[hand(cell)] {
940                forward[usize::from(!positive)][usize::from(input_grid)][outer] = Some(plane);
941            }
942        }
943    }
944    Some(MuellerRouting { adjoint, forward })
945}
946
947/// The diagonal Mueller element of a parallel or cross hand.
948fn mueller_element(correlation: CorrelationType) -> Option<u32> {
949    Some(match correlation {
950        CorrelationType::CircularRr | CorrelationType::LinearXx => 0,
951        CorrelationType::CircularRl | CorrelationType::LinearXy => 5,
952        CorrelationType::CircularLr | CorrelationType::LinearYx => 10,
953        CorrelationType::CircularLl | CorrelationType::LinearYy => 15,
954        _ => return None,
955    })
956}
957
958/// `SynthesisUtils::conjFreq`: `√(2f₀² − f²)`.
959fn conjugate_frequency(frequency_hz: f64, reference_hz: f64) -> f64 {
960    (2.0 * reference_hz * reference_hz - frequency_hz * frequency_hz).sqrt()
961}
962
963/// `SynthesisUtils::nearestValue`: the first listed value at the smallest
964/// distance.
965fn nearest(values: &[f64], target: f64) -> (usize, f64) {
966    let mut best = (0, f64::INFINITY);
967    for (index, value) in values.iter().enumerate() {
968        let distance = (value - target).abs();
969        if distance < best.1 {
970            best = (index, distance);
971        }
972    }
973    (best.0, values[best.0])
974}
975
976fn circular_degrees(value: f64) -> f64 {
977    (value + 180.0).rem_euclid(360.0) - 180.0
978}
