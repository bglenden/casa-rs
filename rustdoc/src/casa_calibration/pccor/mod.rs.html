20/// Request to execute pulse-calibration alignment against a fringe solution.
21pub struct PccorRequest {
22    /// Input MeasurementSet.
23    pub vis: PathBuf,
24    /// Output calibration table.
25    pub output: PathBuf,
26    /// Reference antenna selector.
27    pub refant: String,
28    /// Time range selector.
29    pub timerange: String,
30    /// Scan selector.
31    pub scan: String,
32    /// Spectral window selector.
33    pub spw: String,
34    /// Antenna selector.
35    pub antenna: String,
36    /// Apply cable-cal correction.
37    pub cable: bool,
38    /// Optional fringe-fit calibration table.
39    pub fringe_table: Option<PathBuf>,
40}

42/// Errors returned by `run_pccor`.
43#[derive(Debug, Error)]
44pub enum PccorError {
45    /// Invalid selection.
46    #[error("{0}")]
47    Selection(String),
48    /// Invalid or missing fringe-fit table.
49    #[error("{0}")]
50    FringeTable(String),
51    /// Index failure.
52    #[error("{0}")]
53    Index(String),
54    /// Execution failure.
55    #[error("{0}")]
56    Execution(String),
57    /// MeasurementSet error.
58    #[error(transparent)]
59    MeasurementSet(#[from] casa_ms::MsError),
60    /// Table error.
61    #[error(transparent)]
62    Table(#[from] casa_tables::TableError),
63}

65/// Per-window aligned pulse-calibration report for one antenna.
66pub struct PccorWindowReport {
67    /// Spectral window id.
68    pub window: i32,
69    /// Sample timestamps.
70    pub time: Vec<f64>,
71    /// Scan numbers (`NaN` when unavailable).
72    pub scan: Vec<f64>,
73    /// Field ids (`NaN` when unavailable).
74    pub field_id: Vec<f64>,
75    /// Aligned phase values shaped `[2, samples]` row-major by hand.
76    pub phase: Vec<f64>,
77    /// Aligned delay values shaped `[2, samples]` row-major by hand.
78    pub delay: Vec<f64>,
79    /// Wrapped fringe phase `[hand 0, hand 1]`.
80    pub ff_phase: Vec<f64>,
81    /// Fringe delay `[hand 0, hand 1]`.
82    pub ff_delay: Vec<f64>,
83    /// Mean pulse-cal phase tones per hand.
84    pub pc_mean_phase: [Vec<f64>; 2],
85    /// Mean pulse-cal delay `[hand 0, hand 1]`.
86    pub pc_mean_delay: Vec<f64>,
87    /// Reference antenna mean pulse-cal phase tones per hand.
88    pub pc_refant_phase: [Vec<f64>; 2],
89    /// Reference antenna mean pulse-cal delay `[hand 0, hand 1]`.
90    pub pc_refant_delay: Vec<f64>,
91}

101/// Output of `run_pccor`.
102pub struct PccorReport {
103    /// Antenna reports in table-return order.
104    pub antennas: Vec<PccorAntennaReport>,
105    /// Number of rows written to `output`.
106    pub rows_written: usize,
107}

109/// Execute `pccor` alignment and write the calibration table.
110pub fn run_pccor(request: &PccorRequest) -> Result<PccorReport, PccorError> {
111    let open = crate::timing::Phase::start("pccor_select_read");
112    let ms = MeasurementSet::open(&request.vis)?;
113    let selection = Selection::read(&ms, request)?;
114    let pulses = pulse::read(&request.vis, &selection)?;
115    drop(open);
116
117    let fit = crate::timing::Phase::start("pccor_reference_fringe");
118    let fringes = fringe::read_or_solve(&ms, request, &selection)?;
119    drop(fit);
120    if selection.windows.is_empty() {
121        return Err(PccorError::Index("list index out of range".into()));
122    }
123
124    let align = crate::timing::Phase::start("pccor_align");
125    let mut antennas = Vec::with_capacity(selection.antennas.len());
126    let mut seen_antennas = BTreeSet::new();
127    let mut table_rows = Vec::new();
128    for &antenna in &selection.antennas {
129        let first_antenna = seen_antennas.insert(antenna);
130        let mut windows = Vec::with_capacity(selection.windows.len());
131        let mut seen_windows = BTreeSet::new();
132        for &window in &selection.windows {
133            let pair = (antenna, window);
134            if request.cable && pulses[&pair].missing_single_tone {
135                return Err(PccorError::Selection(
136                    "non-broadcastable output operand with shape (2,1) doesn't match the broadcast shape (2,0)".into(),
137                ));
138            }
139            let options = Alignment {
140                pulse: &pulses[&pair],
141                reference: &pulses[&(selection.reference, window)],
142                fringe: &fringes[&pair],
143                cable: request.cable,
144            };
145            let aligned = options.align(&selection);
146            table_rows.extend(aligned.iter().filter_map(|row| {
147                row.scan.map(|scan| table::Row {
148                    antenna,
149                    window,
150                    time: row.time,
151                    field: scan.field,
152                    scan: scan.number,
153                    phase: row.phase,
154                    delay: row.delay,
155                })
156            }));
157            if first_antenna && seen_windows.insert(window) {
158                windows.push(options.report(window, aligned));
159            }
160        }
161        if first_antenna {
162            let name = selection.antenna_names[antenna as usize].clone();
163            antennas.push(PccorAntennaReport { name, windows });
164        }
165    }
166    drop(align);
167    let publish = crate::timing::Phase::start("pccor_write");
168    let rows_written = table_rows.len();
169    table::write(&ms, &request.output, selection.reference, table_rows)?;
170    drop(publish);
171    Ok(PccorReport {
172        antennas,
173        rows_written,
174    })
175}

177struct Alignment<'a> {
178    pulse: &'a PulseSeries,
179    reference: &'a PulseSeries,
180    fringe: &'a FringeSolution,
181    cable: bool,
182}
183
184struct Aligned<'a> {
185    time: f64,
186    scan: Option<&'a selection::Scan>,
187    phase: [f64; 2],
188    delay: [f64; 2],
189}
190
191impl Alignment<'_> {
192    fn align<'a>(&self, selection: &'a Selection) -> Vec<Aligned<'a>> {
193        let fringe = self.fringe;
194        let fallback = match self.pulse.samples.as_slice() {
195            [] => true,
196            [only] => only.delay[0] == 0.,
197            _ => false,
198        };
199        if fallback {
200            return vec![Aligned {
201                time: fringe.time,
202                scan: selection.scan_at(fringe.time),
203                phase: fringe.phase.map(wrap_phase),
204                delay: fringe.delay,
205            }];
206        }
207        self.pulse
208            .samples
209            .iter()
210            .map(|sample| {
211                let cable = if self.cable { sample.cable } else { 0. };
212                let phase = [0, 1].map(|hand| {
213                    wrap_phase(
214                        fringe.phase[hand] + sample.phase[hand] - self.pulse.mean_phase[hand][0],
215                    )
216                });
217                let delay = [0, 1].map(|hand| {
218                    sample.delay[hand] + fringe.delay[hand]
219                        - (self.pulse.mean_delay[hand] - self.reference.mean_delay[hand])
220                        + cable
221                });
222                Aligned {
223                    time: sample.time,
224                    scan: selection.scan_at(sample.time),
225                    phase,
226                    delay,
227                }
228            })
229            .collect()
230    }
231
232    fn report(&self, window: i32, aligned: Vec<Aligned<'_>>) -> PccorWindowReport {
233        PccorWindowReport {
234            window,
235            time: aligned.iter().map(|row| row.time).collect(),
236            scan: aligned
237                .iter()
238                .map(|row| row.scan.map_or(f64::NAN, |scan| f64::from(scan.number)))
239                .collect(),
240            field_id: aligned
241                .iter()
242                .map(|row| row.scan.map_or(f64::NAN, |scan| f64::from(scan.field)))
243                .collect(),
244            phase: aligned.iter().flat_map(|row| row.phase).collect(),
245            delay: aligned.iter().flat_map(|row| row.delay).collect(),
246            ff_phase: self.fringe.phase.map(wrap_phase).to_vec(),
247            ff_delay: self.fringe.delay.to_vec(),
248            pc_mean_phase: self.pulse.mean_phase.clone(),
249            pc_mean_delay: self.pulse.mean_delay.to_vec(),
250            pc_refant_phase: self.reference.mean_phase.clone(),
251            pc_refant_delay: self.reference.mean_delay.to_vec(),
252        }
253    }
254}
255
256fn wrap_phase(phase: f64) -> f64 {
257    (phase + PI).rem_euclid(TAU) - PI
