32/// Return an image's coordinate template, or regrid its pixels onto a template.
33///
34/// A dictionary template holds the two fields returned by `template='get'`.
35/// A template image supplies its coordinates and dimensions in the same form.
36///
37/// # Errors
38/// Returns an error for unreadable images, incompatible templates, unsupported
39/// pixel or coordinate types, and failed output publication.
40pub fn imregrid(request: &ImregridRequest) -> Result<ImregridOutcome, ImageError> {
41    let input = AnyPagedImage::open(&request.imagename)?;
42    if matches!(&request.template, ImregridTemplate::Image(path) if path.as_os_str().eq_ignore_ascii_case("get"))
43    {
44        let shape = input
45            .shape()
46            .iter()
47            .map(|&length| i32::try_from(length))
48            .collect::<Result<Vec<_>, _>>()
49            .map_err(|_| invalid("image shape exceeds the range of CASA's template record"))?;
50        let mut coordinates = coordinate_record(&input);
51        // casacore's coordsys.torecord() returns the spectral WCS string
52        // without the fixed-width padding found in stored image records.
53        for field in coordinates.fields_mut() {
54            if field.name.starts_with("spectral")
55                && let Value::Record(spectral) = &mut field.value
56                && let Some(Value::Record(wcs)) = spectral.get_mut("wcs")
57                && let Some(Value::Scalar(ScalarValue::String(ctype))) = wcs.get_mut("ctype")
58            {
59                ctype.truncate(ctype.trim_end_matches('\0').len());
60            }
61        }
62        let template = Value::Record(RecordValue::new(vec![
63            casa_types::RecordField::new("csys", Value::Record(coordinates)),
64            casa_types::RecordField::new("shap", Value::Array(ArrayValue::from_i32_vec(shape))),
65        ]));
66        return Ok(ImregridOutcome::Template(
67            casacore_value_to_casa(&template).map_err(invalid)?,
68        ));
69    }
70    let template = read_template(&request.template)?;
71    match input {
72        AnyPagedImage::Float32(image) => write_product(&image, request, &template),
73        AnyPagedImage::Float64(image) => write_product(&image, request, &template),
74        AnyPagedImage::Complex32(image) => write_product(&image, request, &template),
75        AnyPagedImage::Complex64(image) => write_product(&image, request, &template),
76    }
77    .map(ImregridOutcome::Product)
78}

80fn image_coordinates<T: ImagePixel>(image: &PagedImage<T>) -> RecordValue {
81    match image.table().keywords().get("coords") {
82        Some(Value::Record(record)) => record.clone(),
83        _ => image.coordinates().to_record(),
84    }
85}

87fn coordinate_record(image: &AnyPagedImage) -> RecordValue {
88    match image {
89        AnyPagedImage::Float32(image) => image_coordinates(image),
90        AnyPagedImage::Float64(image) => image_coordinates(image),
91        AnyPagedImage::Complex32(image) => image_coordinates(image),
92        AnyPagedImage::Complex64(image) => image_coordinates(image),
93    }
94}

96enum Template {
97    Grid(RegridTemplate),
98    Frame(DirectionRef),
99}

101fn read_template(template: &ImregridTemplate) -> Result<Template, ImageError> {
102    let (shape, record) = match template {
103        ImregridTemplate::Image(path) => {
104            if let Some(reference) = path.to_str().and_then(frame_from_template) {
105                return Ok(Template::Frame(reference));
106            }
107            if !path.exists() {
108                return Err(ImageError::TaskArgumentType {
109                    parameter: "template",
110                    message: format!("Cannot read template image {}", path.display()),
111                });
112            }
113            let image = AnyPagedImage::open(path)?;
114            (image.shape().to_vec(), coordinate_record(&image))
115        }
116        ImregridTemplate::Record(fields) => {
117            let Some(value) = fields.get("csys") else {
118                return Err(invalid("template record has no csys field"));
119            };
120            let Value::Record(record) = parameter_value_to_casacore(value).map_err(invalid)? else {
121                return Err(invalid("template csys must be a coordinate record"));
122            };
123            let Some(ParameterValue::Array(values)) = fields.get("shap") else {
124                return Err(invalid("template shap must be an array of axis lengths"));
125            };
126            let shape = values
127                .iter()
128                .map(|value| match value {
129                    ParameterValue::Integer(length) if *length > 0 => usize::try_from(*length).ok(),
130                    _ => None,
131                })
132                .collect::<Option<Vec<_>>>()
133                .ok_or_else(|| invalid("template shape must contain positive integers"))?;
134            (shape, record)
135        }
136    };
137    let coordinates = CoordinateSystem::from_record(&record)?;
138    if shape.len() != coordinates.n_pixel_axes() {
139        return Err(invalid(
140            "template shape must match the number of pixel axes",
141        ));
142    }
143    Ok(Template::Grid(RegridTemplate { coordinates, shape }))
144}

146fn write_product<T: RegridPixel>(
147    input: &PagedImage<T>,
148    request: &ImregridRequest,
149    template: &Template,
150) -> Result<ImageRegridSummary, ImageError> {
151    let output = match (&request.template, request.output.as_os_str().is_empty()) {
152        (ImregridTemplate::Image(_), true) => {
153            let mut name = request.imagename.as_os_str().to_os_string();
154            name.push(".regridded");
155            PathBuf::from(name)
156        }
157        _ => request.output.clone(),
158    };
159    // The same-frame CASA path uses image.subimage with its overwrite default.
160    let same_frame = match template {
161        Template::Frame(reference) => input
162            .coordinates()
163            .find_coordinate(CoordinateType::Direction)
164            .is_some_and(|index| {
165                matches!(input.coordinates().coordinate(index),
166                CoordinateModel::Direction(direction) if direction.direction_ref() == *reference)
167            }),
168        Template::Grid(_) => false,
169    };
170    let overwrite = request.overwrite && !same_frame;
171    let destination = if output.as_os_str().is_empty() {
172        None
173    } else {
174        let path = std::path::absolute(&output).map_err(output_io)?;
175        check_output(request, &path, overwrite)?;
176        Some(path)
177    };
178    let method = RegridMethod::from_casa(&request.interpolation)?;
179    let (regridded, beam_rotation) = match template {
180        Template::Grid(grid) => {
181            let record_shape;
182            let shape = if matches!(request.template, ImregridTemplate::Record(_)) {
183                record_shape = grid
184                    .shape
185                    .iter()
186                    .map(|&length| length as i64)
187                    .collect::<Vec<_>>();
188                &record_shape
189            } else {
190                &request.shape
191            };
192            (
193                regrid_task(
194                    input,
195                    grid,
196                    TaskRegridOptions {
197                        axes: &request.axes,
198                        shape,
199                        as_velocity: request.asvelocity,
200                        decimate: request.decimate,
201                        replicate: request.replicate,
202                        method,
203                        image_tool: None,
204                    },
205                )?,
206                0.0,
207            )
208        }
209        Template::Frame(reference) => {
210            regrid_direction_frame(input, *reference, method, request.decimate)?
211        }
212    };
213    let Regridded {
214        data,
215        mask,
216        coordinates,
217        ..
218    } = regridded;
219    let shape = data.shape().to_vec();
220    let valid_pixels = mask.iter().filter(|&&bit| bit).count();
221    if let Some(destination) = destination {
222        let input_info = input.image_info()?;
223        let info = output_image_info(
224            input_info.clone(),
225            input.coordinates(),
226            &coordinates,
227            beam_rotation,
228        );
229        let stage = tempfile::tempdir_in(destination.parent().ok_or(ImageError::NotPersistent)?)
230            .map_err(output_io)?;
231        let product = stage.path().join("image");
232        let mut image = PagedImage::<T>::create(shape.clone(), coordinates, &product)?;
233        image.set_units(input.units())?;
234        if info == input_info
235            && let Some(Value::Record(record)) = input.table().keywords().get("imageinfo")
236        {
237            image.set_image_info_record(record.clone())?;
238        } else {
239            image.set_image_info(&info)?;
240        }
241        image.set_misc_info(input.misc_info())?;
242        image.put_slice(&data, &vec![0; shape.len()])?;
243        if valid_pixels < data.len() {
244            image.put_mask("mask0", &mask)?;
245            image.set_default_mask("mask0")?;
246        }
247        image.append_history_entries(&crate::image::image_history_rows(&request.imagename)?)?;
248        image.append_history_at(
249            "::imregrid",
250            &[
251                concat!("version: casa-rs ", env!("CARGO_PKG_VERSION")).into(),
252                format!(
253                    "imregrid(imagename={:?}, output={:?}, interpolation={:?})",
254                    request.imagename, output, request.interpolation
255                ),
256            ],
257        )?;
258        image.save()?;
259        drop(image);
260        match rename_without_replacing(&product, &destination) {
261            Err(error) if error.kind() == io::ErrorKind::AlreadyExists && overwrite => {
262                check_output(request, &destination, overwrite)?;
263                let _locks = lock_table_tree_for_replacement(&destination)?;
264                exchange_directories(&product, &destination).map_err(output_io)?;
265            }
266            result => result.map_err(output_io)?,
267        }
268    }
269    Ok(ImageRegridSummary {
270        imagename: request.imagename.display().to_string(),
271        template: match &request.template {
272            ImregridTemplate::Image(path) => path.display().to_string(),
273            ImregridTemplate::Record(_) => "coordinate record".into(),
274        },
275        output: output.display().to_string(),
276        shape,
277        valid_pixels,
278    })
279}

281fn output_image_info(
282    mut info: ImageInfo,
283    source: &CoordinateSystem,
284    output: &CoordinateSystem,
285    rotation: f64,
286) -> ImageInfo {
287    if info.beam_set.is_empty() {
288        return info;
289    }
290    let stokes = |system: &CoordinateSystem| {
291        system
292            .find_coordinate(CoordinateType::Stokes)
293            .and_then(|index| match system.coordinate(index) {
294                CoordinateModel::Stokes(coordinate) => Some(coordinate.stokes().to_vec()),
295                _ => None,
296            })
297    };
298    let selections = if info.beam_set.n_stokes() > 1 {
299        stokes(source).zip(stokes(output)).map(|(source, target)| {
300            target
301                .iter()
302                .filter_map(|value| source.iter().position(|candidate| candidate == value))
303                .collect::<Vec<_>>()
304        })
305    } else {
306        None
307    };
308    if rotation == 0.0
309        && selections
310            .as_ref()
311            .is_none_or(|planes| planes.len() == info.beam_set.n_stokes())
312    {
313        return info;
314    }
315    let planes = selections.unwrap_or_else(|| (0..info.beam_set.n_stokes()).collect());
316    info.beam_set = ImageBeamSet::from_grid(
317        (0..info.beam_set.n_channels())
318            .map(|channel| {
319                planes
320                    .iter()
321                    .map(|&pol| {
322                        let mut beam = *info.beam_set.beam(channel, pol);
323                        beam.position_angle += rotation;
324                        beam
325                    })
326                    .collect()
327            })
328            .collect(),
329    );
330    info
331}

333fn check_output(
334    request: &ImregridRequest,
335    destination: &Path,
336    overwrite: bool,
337) -> Result<(), ImageError> {
338    ensure_outfile_does_not_alias_input(destination, &request.imagename)?;
339    if let ImregridTemplate::Image(path) = &request.template {
340        ensure_outfile_does_not_alias_input(destination, path)?;
341    }
342    match destination.symlink_metadata() {
343        Err(error) if error.kind() == io::ErrorKind::NotFound => Ok(()),
344        Err(error) => Err(output_io(error)),
345        Ok(_) if !overwrite => Err(invalid(format!(
346            "imregrid output already exists: {}",
347            destination.display()
348        ))),
349        Ok(metadata)
350            if metadata.is_dir()
351                && Table::open_metadata_only(TableOptions::new(destination))
352                    .is_ok_and(|table| table.info().table_type == "Image") =>
353        {
354            Ok(())
355        }
356        Ok(_) => Err(invalid(format!(
357            "{} is not a CASA image directory",
358            destination.display()
359        ))),
360    }
361}
