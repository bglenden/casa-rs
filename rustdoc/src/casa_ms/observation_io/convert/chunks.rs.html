2//! XRADIO's default chunk shape of an MSv4 dataset ([`xradio_chunks`]).
3//!
4//! XRADIO's converter (`default_main_chunksize`, XRADIO 1.3) chunks a main
5//! dataset along time only, every other axis whole, with as many times as
6//! make the chunk of its largest data variable about
7//! [`DEFAULT_CHUNK_BYTES`] uncompressed, and at least one. Only where one
8//! time of that variable is more than Blosc encodes in one buffer
9//! ([`BLOSC_MAX_BUFFER_BYTES`]) does it split frequency, and then
10//! baselines, to chunks of about [`DEFAULT_CHUNK_BYTES`]. Each dataset is
11//! sized by itself, so a continuum window of few channels gets as many
12//! times a chunk as a line window of many gets few.
13
14/// XRADIO's `DEFAULT_MAIN_CHUNK_BYTES`: the uncompressed bytes of the
15/// largest variable's chunk the default aims at.
16pub const DEFAULT_CHUNK_BYTES: u64 = 128 << 20;
17
18/// XRADIO's `BLOSC_MAX_BUFFER_BYTES`: the largest buffer Blosc encodes.
19pub const BLOSC_MAX_BUFFER_BYTES: u64 = (1 << 31) - 1;
20
21/// Bytes of one `VISIBILITY` element (`complex64`), the widest per channel
22/// of what the writer stores.
23const VISIBILITY_BYTES: u64 = 8;
24/// Bytes of one baseline's `UVW` (three `float64`), the widest variable
25/// without a frequency axis.
26const UVW_BYTES: u64 = 24;
27
28/// Bytes of the largest variable's chunk of one time, `baselines`
29/// baselines, `channels` channels and `correlations` correlations.
30fn chunk_bytes(baselines: u64, channels: u64, correlations: u64) -> u64 {
31    (VISIBILITY_BYTES * baselines * channels * correlations).max(UVW_BYTES * baselines)
32}
33
34/// The chunk shape `[time, baseline, frequency, polarization]` XRADIO's
35/// converter gives a dataset of `extents` (in the same order), 0 for a
36/// whole axis.
37#[must_use]
38pub fn xradio_chunks(extents: [usize; 4]) -> [u64; 4] {
39    let [times, baselines, channels, correlations] = extents.map(|extent| extent as u64);
40    let per_time = chunk_bytes(baselines, channels, correlations).max(1);
41    let mut chunks = [
42        (DEFAULT_CHUNK_BYTES / per_time).clamp(1, times.max(1)),
43        0,
44        0,
45        0,
46    ];
47    if per_time <= BLOSC_MAX_BUFFER_BYTES {
48        return chunks;
49    }
50    // One time is too wide for Blosc: chunks of one time, then frequency
51    // split, then baselines, until the chunk fits.
52    let per_channel = (VISIBILITY_BYTES * baselines * correlations).max(1);
53    let channel_chunk = (DEFAULT_CHUNK_BYTES / per_channel).clamp(1, channels.max(1));
54    chunks[2] = channel_chunk;
55    if chunk_bytes(baselines, channel_chunk, correlations) > BLOSC_MAX_BUFFER_BYTES {
56        let per_baseline = chunk_bytes(1, channel_chunk, correlations).max(1);
57        chunks[1] = (DEFAULT_CHUNK_BYTES / per_baseline).clamp(1, baselines.max(1));
58    }
59    chunks
60}

62#[cfg(test)]
63mod tests {
64    use super::*;
65
66    #[test]
67    fn a_dataset_is_chunked_along_time_to_about_128_mib() {
68        // pilot.ms: 351 baselines, 64 channels, 4 correlations; one time of
69        // VISIBILITY is 718,848 bytes, so 186 times make 128 MiB.
70        assert_eq!(xradio_chunks([1000, 351, 64, 4]), [186, 0, 0, 0]);
71        // Never more times than the dataset holds.
72        assert_eq!(xradio_chunks([12, 351, 64, 4]), [12, 0, 0, 0]);
73        // A time wider than 128 MiB still makes a chunk of one time.
74        assert_eq!(xradio_chunks([100, 1225, 4096, 4]), [1, 0, 0, 0]);
75        // UVW is the widest variable of a single channel and correlation.
76        assert_eq!(
77            xradio_chunks([100_000, 1000, 1, 1]),
78            [DEFAULT_CHUNK_BYTES / 24_000, 0, 0, 0]
79        );
80    }
81
82    #[test]
83    fn each_dataset_is_sized_by_itself() {
84        // A one-channel average beside a 3,840-channel window, as raw ALMA
85        // data holds them: each chunk is about 128 MiB (the average's of
86        // UVW, its widest variable).
87        let line = xradio_chunks([10_000, 990, 3840, 2]);
88        let average = xradio_chunks([10_000, 990, 1, 2]);
89        assert_eq!(line, [2, 0, 0, 0]);
90        assert_eq!(average, [5648, 0, 0, 0]);
91    }
92
93    #[test]
94    fn frequency_and_then_baselines_split_only_past_blosc_s_limit() {
95        // 2.4 GiB a time: one time, 3,423 channels a chunk.
96        assert_eq!(xradio_chunks([10, 1225, 65_536, 4]), [1, 0, 3423, 0]);
97        // A time of one channel still past the limit splits baselines too.
98        assert_eq!(xradio_chunks([10, 100_000_000, 2, 4]), [1, 4_194_304, 1, 0]);
99    }
100
101    #[test]
102    fn an_empty_dataset_is_one_time_of_whole_axes() {
103        assert_eq!(xradio_chunks([0, 0, 0, 0]), [1, 0, 0, 0]);
104    }
105}
