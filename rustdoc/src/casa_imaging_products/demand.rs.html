19/// Checked owned-array and metadata demand for one planned continuum generation.
20///
21/// It describes the products' payload, validity and working arrays, and what
22/// the image writer holds, so the run admits the whole phase once before
23/// production starts. The generated summary's owned metadata and temporary
24/// beam arrays are included; allocator bookkeeping, the borrowed plan and
25/// borrowed reconstruction inputs are not charged here.
26#[derive(Debug, Clone, Copy, PartialEq, Eq)]
27pub struct ContinuumGenerationDemand {
28    storage_plan: ProductStoragePlan,
29    algorithm_scratch_bytes: u64,
30    retained_metadata_bytes: u64,
31    beam_scratch_bytes: u64,
32    writer_bytes: u64,
33    maximum_member_payload_bytes: u64,
34    maximum_member_validity_bytes: u64,
35    maximum_window_payload_bytes: u64,
36    maximum_window_validity_bytes: u64,
37    peak_residency_bytes: u64,
38}

104/// The shapes a generation's demand depends on, as the run's final normal
105/// state and model have them: the compiled problem fixes them before the
106/// run ([`Self::of_problem`]), and the completion carries them after it
107/// ([`Self::of_inputs`]).
108struct Shapes<'a> {
109    problem: &'a CompiledProblem,
110    basis: NormalBasis,
111    /// Image domains.
112    domains: usize,
113    /// The normal state's channel planes.
114    channels: usize,
115    polarizations: usize,
116    /// The normal state's plane: its primary domain's.
117    plane: [usize; 2],
118    /// Reconstruction-coefficient terms and normal moments.
119    terms: usize,
120    moments: usize,
121    /// Each model domain's plane, in canonical order.
122    model_planes: Vec<[usize; 2]>,
123}
124
125impl<'a> Shapes<'a> {
126    fn of_inputs(inputs: &'a ContinuumProductInputs<'_>) -> Self {
127        let state = inputs.normal_state();
128        Self {
129            problem: inputs.problem(),
130            basis: state.basis(),
131            domains: state.domain_count(),
132            channels: state.channel_count(),
133            polarizations: state.polarization_count(),
134            plane: state.shape(),
135            terms: state.coefficient_term_count(),
136            moments: state.normal_moment_count(),
137            model_planes: inputs
138                .final_model()
139                .shape()
140                .domains()
141                .iter()
142                .map(|domain| domain.pixels())
143                .collect(),
144        }
145    }
146
147    /// The shapes a run of `problem` completes with: the model it compiled
148    /// (`ModelLifecycleContract::target`), and a normal state of one channel
149    /// plane per model coefficient (one for a Taylor basis, whose terms and
150    /// `2·terms − 1` moments are its own) on the primary domain's plane.
151    fn of_problem(problem: &'a CompiledProblem) -> Self {
152        let model = problem.model_lifecycle().target();
153        let model_planes: Vec<[usize; 2]> = model
154            .domains()
155            .iter()
156            .map(|domain| domain.pixels())
157            .collect();
158        let (basis, channels, terms, moments) = match problem.reconstruction().basis() {
159            ReconstructionBasis::Taylor { terms } => (NormalBasis::Taylor, 1, terms, 2 * terms - 1),
160            ReconstructionBasis::ChannelLocal { .. } => {
161                (NormalBasis::ChannelLocal, model.coefficients(), 1, 1)
162            }
163            _ => (NormalBasis::Constant, model.coefficients(), 1, 1),
164        };
165        Self {
166            problem,
167            basis,
168            domains: model_planes.len(),
169            channels,
170            polarizations: model.polarizations(),
171            plane: model_planes[0],
172            terms,
173            moments,
174            model_planes,
175        }
176    }
177}

189impl PlannedContinuumGeneration {
190    /// Derive current-catalog array and metadata demand for these inputs, the
191    /// ones this plan was made from.
192    ///
193    /// # Errors
194    ///
195    /// Fails when the storage plan cannot lay out a member or a byte
196    /// calculation overflows.
197    pub fn demand(
198        &self,
199        inputs: &ContinuumProductInputs<'_>,
200        storage_plan: ProductStoragePlan,
201        writer: &CasaImageWriter,
202    ) -> Result<ContinuumGenerationDemand, ProductsError> {
203        self.demand_of(&Shapes::of_inputs(inputs), storage_plan, writer)
204    }

206    /// [`Self::demand`] before the run: from the shapes its problem fixes,
207    /// which its completion then has, so that a run can tell before its
208    /// passes whether its products fit. A plan of `problem`
209    /// ([`Self::for_problem`]) gives the demand [`Self::demand`] gives once
210    /// the run completes.
211    ///
212    /// # Errors
213    ///
214    /// As [`Self::demand`].
215    pub fn planned_demand(
216        &self,
217        problem: &CompiledProblem,
218        storage_plan: ProductStoragePlan,
219        writer: &CasaImageWriter,
220    ) -> Result<ContinuumGenerationDemand, ProductsError> {
221        self.demand_of(&Shapes::of_problem(problem), storage_plan, writer)
222    }

224    fn demand_of(
225        &self,
226        shapes: &Shapes<'_>,
227        storage_plan: ProductStoragePlan,
228        writer: &CasaImageWriter,
229    ) -> Result<ContinuumGenerationDemand, ProductsError> {
230        let bounds = self.member_bounds(storage_plan)?;
231        let maximum_windows = bounds.windows;
232        let beam_jobs = if self.requires_beam() {
233            shapes
234                .domains
235                .checked_mul(shapes.channels)
236                .and_then(|count| count.checked_mul(shapes.polarizations))
237                .ok_or(ProductsError::ResourceDemandOverflow(
238                    "generation beam count",
239                ))?
240        } else {
241            0
242        };
243        let workers = match shapes.basis {
244            NormalBasis::Taylor => 1,
245            _ if cfg!(unix)
246                && maximum_windows == 1
247                && self.members().iter().any(|member| {
248                    matches!(
249                        member.role(),
250                        ProductRole::RestoredImage(_) | ProductRole::PbCorrectedImage(_)
251                    )
252                }) =>
253            {
254                storage_plan.maximum_workers()
255            }
256            _ => storage_plan
257                .maximum_workers()
258                .min(maximum_windows.max(beam_jobs)),
259        };
260        let storage_plan = ProductStoragePlan::new(storage_plan.maximum_channels(), workers)?;
261        let algorithm_scratch_bytes =
262            self.algorithm_scratch_bytes(shapes, &bounds, workers.min(maximum_windows))?;
263        let (retained_metadata_bytes, member_beam_bytes, beam_scratch_bytes) =
264            self.metadata_demand(shapes, workers.min(beam_jobs))?;
265        let writer_bytes =
266            writer.residency_bytes(self, bounds.window_payload, bounds.window_validity)?;
267        let peak_residency_bytes = if matches!(shapes.basis, NormalBasis::Taylor) {
268            // The coupled family is built whole, then published member by
269            // member: it peaks at the larger stage, not at their sum (#738).
270            let stages = taylor_stages(shapes, beam_scratch_bytes)?;
271            let publishing = [
272                bounds.member_payload,
273                bounds.member_validity,
274                bounds.window_payload,
275                bounds.window_validity,
276                writer_bytes,
277            ]
278            .into_iter()
279            .try_fold(stages.retained, |sum, bytes| {
280                checked_add(sum, bytes, "Taylor publication residency")
281            })?;
282            checked_add(
283                retained_metadata_bytes,
284                stages.building.max(publishing),
285                "Taylor generation residency",
286            )?
287        } else {
288            let transient_bytes = checked_add(
289                algorithm_scratch_bytes,
290                member_beam_bytes,
291                "active member preparation",
292            )?
293            .max(beam_scratch_bytes);
294            checked_add(
295                checked_add(
296                    retained_metadata_bytes,
297                    transient_bytes,
298                    "product generation residency",
299                )?,
300                writer_bytes,
301                "product generation and writer residency",
302            )?
303        };
304        Ok(ContinuumGenerationDemand {
305            storage_plan,
306            algorithm_scratch_bytes,
307            retained_metadata_bytes,
308            beam_scratch_bytes,
309            writer_bytes,
310            maximum_member_payload_bytes: bounds.member_payload,
311            maximum_member_validity_bytes: bounds.member_validity,
312            maximum_window_payload_bytes: bounds.window_payload,
313            maximum_window_validity_bytes: bounds.window_validity,
314            peak_residency_bytes,
315        })
316    }

318    /// The largest member and window arrays, and the most windows a member
319    /// takes.
320    fn member_bounds(
321        &self,
322        storage_plan: ProductStoragePlan,
323    ) -> Result<MemberBounds, ProductsError> {
324        let mut bounds = MemberBounds {
325            windows: 1,
326            ..MemberBounds::default()
327        };
328        for member in self.members() {
329            let values = member.payload_values();
330            bounds.member_payload = bounds
331                .member_payload
332                .max(bytes_for::<f32>(values, "member payload")?);
333            bounds.member_validity = bounds
334                .member_validity
335                .max(bytes_for::<bool>(values, "member validity")?);
336            let layout = storage_plan.layout(member.axes())?;
337            bounds.windows = bounds
338                .windows
339                .max(layout.shape()[layout.spectral_axis()].div_ceil(layout.maximum_channels()));
340            let window_values = layout.maximum_values();
341            bounds.window_payload = bounds
342                .window_payload
343                .max(bytes_for::<f32>(window_values, "product window payload")?);
344            bounds.window_validity = bounds
345                .window_validity
346                .max(bytes_for::<bool>(window_values, "product window validity")?);
347        }
348        Ok(bounds)
349    }

351    /// The products' working arrays: the algorithm's scratch and an active
352    /// window per lane.
353    fn algorithm_scratch_bytes(
354        &self,
355        shapes: &Shapes<'_>,
356        bounds: &MemberBounds,
357        window_workers: usize,
358    ) -> Result<u64, ProductsError> {
359        let taylor = matches!(shapes.basis, NormalBasis::Taylor);
360        let mut bytes = if taylor {
361            checked_add(
362                taylor_scratch_bytes(shapes)?,
363                checked_add(
364                    bounds.member_payload,
365                    bounds.member_validity,
366                    "coupled member arrays",
367                )?,
368                "coupled member production scratch",
369            )?
370        } else {
371            generic_scratch_bytes(self, shapes)?
372        };
373        bytes = checked_add(
374            bytes,
375            checked_add(
376                bounds.window_payload,
377                bounds.window_validity,
378                "active product window",
379            )?,
380            "windowed production scratch",
381        )?;
382        // Each lane can retain a completed window while another lane still owns
383        // its plane workspace. Beam fitting joins before these windows begin.
384        bytes = checked_mul(bytes, window_workers as u64, "parallel product windows")?;
385        if !taylor {
386            bytes = checked_add(
387                bytes,
388                bytes_for::<Option<crate::ProductWindow>>(window_workers, "product window slots")?,
389                "parallel product slots",
390            )?;
391        }
392        Ok(bytes)
393    }

395    fn metadata_demand(
396        &self,
397        shapes: &Shapes<'_>,
398        workers: usize,
399    ) -> Result<(u64, u64, u64), ProductsError> {
400        let domains = shapes.domains;
401        let fitted = if !self.requires_beam() {
402            0
403        } else {
404            match shapes.basis {
405                NormalBasis::Taylor => 1,
406                _ => domains
407                    .checked_mul(shapes.channels)
408                    .and_then(|count| count.checked_mul(shapes.polarizations))
409                    .ok_or(ProductsError::ResourceDemandOverflow(
410                        "generation beam count",
411                    ))?,
412            }
413        };
414        let restoring_policy = shapes.problem.products().restoring_beam();
415        let restoring = match (restoring_policy, shapes.basis) {
416            (RestoringBeamPolicy::None, _) => 0,
417            (_, NormalBasis::Taylor) => 1,
418            _ => fitted,
419        };
420        let mut retained = checked_add(
421            size_of::<PublishedContinuumGeneration>() as u64,
422            bytes_for::<PublishedMember>(self.members().len(), "published members")?,
423            "generation summary",
424        )?;
425        for count in [fitted, restoring] {
426            retained = checked_add(
427                retained,
428                bytes_for::<Option<RestoringBeam>>(count, "generation beams")?,
429                "generation metadata",
430            )?;
431        }
432        let mut active_member_beams = 0;
433        for member in self.members() {
434            let beams = match member.beam_rule() {
435                ProductBeamRule::None
436                | ProductBeamRule::Restoring(RestoringBeamPolicy::None)
437                | ProductBeamRule::Metadata(RestoringBeamPolicy::None) => 0,
438                ProductBeamRule::Fitted => fitted / domains,
439                ProductBeamRule::Restoring(RestoringBeamPolicy::Common)
440                | ProductBeamRule::Metadata(RestoringBeamPolicy::Common) => 1,
441                _ => restoring / domains,
442            };
443            let beam_bytes = bytes_for::<Option<RestoringBeam>>(beams, "member beams")?;
444            active_member_beams = active_member_beams.max(beam_bytes);
445            for bytes in [
446                bytes_for::<u8>(member.name().len(), "member name")?,
447                beam_bytes,
448            ] {
449                retained = checked_add(retained, bytes, "member metadata")?;
450            }
451        }
452        let common_scratch = if restoring_policy == RestoringBeamPolicy::Common
453            && matches!(
454                shapes.basis,
455                NormalBasis::Constant | NormalBasis::ChannelLocal
456            ) {
457            checked_add(
458                bytes_for::<RestoringBeam>(fitted / domains, "common valid beams")?,
459                checked_mul(
460                    bytes_for::<casa_numerics::EllipticalGaussian>(
461                        fitted / domains,
462                        "common numeric beams",
463                    )?,
464                    2,
465                    "common converted and mutable numeric beams",
466                )?,
467                "common beam scratch",
468            )?
469        } else {
470            0
471        };
472        let mut fitting_scratch = 0;
473        if self.requires_beam() {
474            for &shape in &shapes.model_planes {
475                let cells = checked_shape_values([shape[0], shape[1], 1, 1])?;
476                let input = bytes_for::<f32>(cells, "beam real PSF")?;
477                // Paged PSFs own the decoded complex plane while converting to
478                // f32; fitting then retains only f32 and the fit workspace.
479                let conversion = bytes_for::<Complex64>(cells, "beam complex PSF")?;
480                fitting_scratch = fitting_scratch.max(
481                    checked_add(conversion, input, "beam input conversion")?.max(checked_add(
482                        input,
483                        casa_imaging_deconvolution::psf_fit_workspace_bytes(shape),
484                        "beam fitting workspace",
485                    )?),
486                );
487            }
488            fitting_scratch = checked_mul(fitting_scratch, workers as u64, "parallel beam fits")?;
489            fitting_scratch = checked_add(
490                fitting_scratch,
491                bytes_for::<Option<Option<RestoringBeam>>>(workers, "beam summary slots")?,
492                "parallel beam summaries",
493            )?;
494        }
495        Ok((
496            retained,
497            active_member_beams,
498            active_member_beams.max(common_scratch).max(fitting_scratch),
499        ))
500    }
501}

503fn generic_scratch_bytes(
504    planned: &PlannedContinuumGeneration,
505    shapes: &Shapes<'_>,
506) -> Result<u64, ProductsError> {
507    let requires_validity = planned
508        .members()
509        .iter()
510        .any(|member| member.validity() != ProductValidityRule::All);
511    let requires_restoration = planned
512        .members()
513        .iter()
514        .any(|member| matches!(member.role(), ProductRole::RestoredImage(_)));
515    let mut maximum = 0;
516    let mut normal_window = 0;
517    for &shape in &shapes.model_planes {
518        let cells = checked_shape_values([shape[0], shape[1], 1, 1])?;
519        let plane = bytes_for::<f32>(cells, "generic plane")?;
520        let validity = bytes_for::<bool>(cells, "generic validity plane")?;
521        // The full-window allowance conservatively bounds field-selective reads,
522        // including the scalar-to-complex conversion overlap.
523        normal_window = checked_add(
524            normal_window,
525            casa_imaging_deconvolution::normal_state_window_residency_bytes(
526                shape,
527                shapes.polarizations,
528                1,
529            )?,
530            "generic normal-state input window",
531        )?;
532        let mut scratch = if requires_validity { validity } else { 0 };
533
534        // Normalization overlaps one converted source plane with its result.
535        scratch = scratch.max(checked_mul(plane, 2, "generic converted plane pair")?);
536        if requires_restoration {
537            // Restoration retains its restored result and normalized residual while
538            // a Gaussian kernel and one FFTW convolution workspace live.
539            scratch = scratch.max(checked_add(
540                checked_mul(plane, 3, "generic restoration planes")?,
541                fft_convolution_workspace_bytes(shape)?,
542                "generic restoration scratch",
543            )?);
544        }
545        maximum = maximum.max(checked_add(
546            scratch,
547            bytes_for::<casa_imaging_model::ModelSample>(cells, "generic model input plane")?,
548            "generic loaded model and algorithm scratch",
549        )?);
550    }
551    checked_add(
552        maximum,
553        normal_window,
554        "generic loaded normal state and algorithm scratch",
555    )
556}

558fn taylor_scratch_bytes(shapes: &Shapes<'_>) -> Result<u64, ProductsError> {
559    let parts = TaylorParts::of(shapes)?;
560    checked_add(
561        checked_add(parts.retained, parts.transient, "Taylor algorithm scratch")?,
562        parts.restoration.unwrap_or(0),
563        "Taylor restoration scratch",
564    )
565}

567/// What the coupled Taylor family holds while `TaylorProducts::build`
568/// builds it, at its largest stage, and once it is built
569/// ([`taylor_stages`]).
570struct TaylorStages {
571    building: u64,
572    retained: u64,
573}
574
575/// The stages of the coupled Taylor family (#738). It reads its PSF and
576/// weight moments, residual and model terms and sensitivity first, solves
577/// the principal residuals, fits the beam and restores each term; the PB,
578/// PB-corrected, spectral-index and mask planes come after the
579/// restoration's FFT workspace is gone. Building peaks at the largest of:
580/// the read family and the principal residuals beside the beam fit
581/// (`beam_scratch`) or beside restoration (its kernel, the restored terms
582/// and one FFT convolution's workspace); or every retained plane beside
583/// the principal residuals, the PB0 source and the normal solve.
584fn taylor_stages(shapes: &Shapes<'_>, beam_scratch: u64) -> Result<TaylorStages, ProductsError> {
585    let parts = TaylorParts::of(shapes)?;
586    let planes = |count: u64| checked_mul(count, parts.plane, "Taylor stage planes");
587    // PSF and weight moments, residual and model terms, the sensitivity.
588    let read = checked_add(
589        planes(2 * parts.moments + 2 * parts.terms + 1)?,
590        checked_mul(parts.moments, size_of::<f32>() as u64, "Taylor sum weights")?,
591        "Taylor read family",
592    )?;
593    let solved = checked_add(read, planes(parts.terms)?, "Taylor principal residuals")?;
594    let mut building = checked_add(parts.retained, parts.transient, "Taylor built family")?
595        .max(checked_add(solved, beam_scratch, "Taylor beam fit")?);
596    if let Some(restoration) = parts.restoration {
597        // The kernel and the workspace's output are in `restoration`; the
598        // terms restored before the last are beside it.
599        building = building.max(checked_add(
600            checked_add(solved, restoration, "Taylor restoration")?,
601            planes(parts.terms.saturating_sub(1))?,
602            "Taylor restored terms",
603        )?);
604    }
605    Ok(TaylorStages {
606        building,
607        retained: parts.retained,
608    })
609}

611/// The pieces of the coupled Taylor family's demand.
612struct TaylorParts {
613    /// One `f32` plane.
614    plane: u64,
615    terms: u64,
616    moments: u64,
617    /// Every plane the built family keeps, its validity and sum weights.
618    retained: u64,
619    /// The principal residuals and PB0 source the build keeps beside the
620    /// family, and the normal solve.
621    transient: u64,
622    /// The restoring kernel and one FFT convolution's workspace, when the
623    /// products restore.
624    restoration: Option<u64>,
625}
626
627impl TaylorParts {
628    fn of(shapes: &Shapes<'_>) -> Result<Self, ProductsError> {
629        let shape = shapes.plane;
630        let cells = checked_shape_values([shape[0], shape[1], 1, 1])?;
631        let terms = u64::try_from(shapes.terms)
632            .map_err(|_| ProductsError::ResourceDemandOverflow("Taylor term count"))?;
633        let moments = u64::try_from(shapes.moments)
634            .map_err(|_| ProductsError::ResourceDemandOverflow("Taylor moment count"))?;
635        let cells = u64::try_from(cells)
636            .map_err(|_| ProductsError::ResourceDemandOverflow("Taylor cell count"))?;
637
638        // TaylorProducts retains PSF and weight moments; residual, model,
639        // restored, PB and PB-corrected term families; alpha, PB-corrected alpha,
640        // alpha-error, sensitivity and mask;
641        // two validity planes; and scalar sum weights until every planned member
642        // has been copied into the produced generation.
643        let retained_f32_planes = checked_add(
644            checked_mul(moments, 2, "Taylor PSF and weight planes")?,
645            checked_add(
646                checked_mul(terms, 5, "Taylor term families")?,
647                5,
648                "Taylor scalar product planes",
649            )?,
650            "Taylor retained plane count",
651        )?;
652        let retained = checked_add(
653            checked_mul(
654                checked_mul(retained_f32_planes, cells, "Taylor retained values")?,
655                size_of::<f32>() as u64,
656                "Taylor retained payload bytes",
657            )?,
658            checked_add(
659                checked_mul(
660                    checked_mul(2, cells, "Taylor validity values")?,
661                    size_of::<bool>() as u64,
662                    "Taylor validity bytes",
663                )?,
664                checked_mul(moments, size_of::<f32>() as u64, "Taylor sum weights")?,
665                "Taylor validity and sum-weight bytes",
666            )?,
667            "Taylor retained scratch",
668        )?;
669
670        // Construction additionally retains the principal residual family and PB0
671        // source plane. The normal solve owns three dense term matrices plus two
672        // term vectors at its peak. This intentionally follows the product owner's
673        // concrete current data structures, rather than a runtime approximation.
674        let transient_planes = checked_mul(
675            checked_add(terms, 1, "Taylor transient plane count")?,
676            checked_mul(
677                cells,
678                size_of::<f32>() as u64,
679                "Taylor transient plane bytes",
680            )?,
681            "Taylor transient planes",
682        )?;
683        let term_square = checked_mul(terms, terms, "Taylor normal matrix values")?;
684        let solve = checked_mul(
685            checked_add(
686                checked_mul(term_square, 3, "Taylor normal matrices")?,
687                checked_mul(terms, 2, "Taylor solve vectors")?,
688                "Taylor solve values",
689            )?,
690            size_of::<f64>() as u64,
691            "Taylor solve bytes",
692        )?;
693        let restoration = if shapes.problem.products().restoring_beam() != RestoringBeamPolicy::None
694        {
695            Some(checked_add(
696                checked_mul(cells, size_of::<f32>() as u64, "Taylor restoring kernel")?,
697                fft_convolution_workspace_bytes(shape)?,
698                "Taylor convolution scratch",
699            )?)
700        } else {
701            None
702        };
703        Ok(Self {
704            plane: checked_mul(cells, size_of::<f32>() as u64, "Taylor plane bytes")?,
705            terms,
706            moments,
707            retained,
708            transient: checked_add(transient_planes, solve, "Taylor transient scratch")?,
709            restoration,
710        })
711    }
712}
713
714fn fft_convolution_workspace_bytes(shape: [usize; 2]) -> Result<u64, ProductsError> {
715    let cells = checked_shape_values([shape[0], shape[1], 1, 1])?;
716    let complex_planes = bytes_for::<Complex64>(
717        cells
718            .checked_mul(2)
719            .ok_or(ProductsError::ResourceDemandOverflow("FFT complex planes"))?,
720        "FFT complex planes",
721    )?;
722    let output = bytes_for::<f32>(cells, "FFT output plane")?;
723    let planning = cells
724        .checked_add(64)
725        .and_then(|values| values.checked_mul(size_of::<Complex64>()))
726        .and_then(|bytes| bytes.checked_add(cells.checked_mul(size_of::<usize>())?))
727        .ok_or(ProductsError::ResourceDemandOverflow("FFTW planning"))?;
728    checked_add(
729        checked_add(complex_planes, output, "FFT planes and output")?,
730        u64::try_from(planning)
731            .map_err(|_| ProductsError::ResourceDemandOverflow("FFTW planning"))?,
732        "FFT convolution workspace",
733    )
734}
735
736fn checked_shape_values(shape: [usize; 4]) -> Result<usize, ProductsError> {
737    shape.into_iter().try_fold(1usize, |values, extent| {
738        values
739            .checked_mul(extent)
740            .ok_or(ProductsError::ResourceDemandOverflow("member shape"))
741    })
742}
743
744fn bytes_for<T>(count: usize, what: &'static str) -> Result<u64, ProductsError> {
745    let count = u64::try_from(count).map_err(|_| ProductsError::ResourceDemandOverflow(what))?;
746    checked_mul(count, size_of::<T>() as u64, what)
747}
748
749fn checked_add(left: u64, right: u64, what: &'static str) -> Result<u64, ProductsError> {
750    left.checked_add(right)
751        .ok_or(ProductsError::ResourceDemandOverflow(what))
752}
753
754fn checked_mul(left: u64, right: u64, what: &'static str) -> Result<u64, ProductsError> {
755    left.checked_mul(right)
756        .ok_or(ProductsError::ResourceDemandOverflow(what))
757}
