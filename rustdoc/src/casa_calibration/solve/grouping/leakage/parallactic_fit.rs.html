9/// CASA applies the selected amplitude or phase constraint to averaged
10/// coherencies and scales phase-only weights by the measured power.
11pub(super) fn solve_sample(mode: GainSolveMode, value: Complex32, weight: f32) -> (Complex32, f32) {
12    match mode {
13        GainSolveMode::Amplitude => (value.norm().into(), weight),
14        GainSolveMode::Phase if value != Complex32::ZERO => {
15            let amplitude = value.norm();
16            (value / amplitude, weight * (amplitude * amplitude))
17        }
18        _ => (value, weight),
19    }
20}

22#[derive(Clone, Default)]
23pub(super) struct SegmentChannel {
24    weight: f64,
25    corrected: Complex64,
26    model: Complex64,
27    alignment: Complex64,
28}
29
30impl SegmentChannel {
31    pub(super) fn has_weight(&self) -> bool {
32        self.weight > 0.0
33    }
34
35    pub(super) fn observe(
36        &mut self,
37        [xy, yx]: [Complex32; 2],
38        [mx, my]: [Complex32; 2],
39        [wx, wy]: [f32; 2],
40    ) {
41        self.weight += f64::from(wx) + f64::from(wy);
42        self.corrected += complex32_to_64(wx * xy) + complex32_to_64(wy * yx.conj());
43        self.model += complex32_to_64(wx * mx) + complex32_to_64(wy * my.conj());
44        self.alignment +=
45            complex32_to_64((wx * mx.re) * xy) + complex32_to_64((wy * my.re) * yx.conj());
46    }
47}

49pub(super) struct Segment {
50    pub(super) angle: f64,
51    pub(super) channels: Vec<SegmentChannel>,
52}
53
54pub(super) enum FitError {
55    Singular,
56}

58pub(super) fn solve(
59    segments: &[Segment],
60    channels: usize,
61    circular: bool,
62    align_model: bool,
63) -> Result<(Vec<Option<Complex32>>, Complex32), FitError> {
64    let mut phases = Vec::with_capacity(channels);
65    let mut previous = Complex64::ONE;
66    let mut ambiguity = 1.0;
67    let mut alignment = 0.0;
68    for channel in 0..channels {
69        let rows = segments
70            .iter()
71            .map(|segment| &segment.channels[channel])
72            .filter(|row| row.weight > 0.0);
73        if rows.clone().next().is_none() {
74            phases.push(None);
75            previous = Complex64::ONE;
76            continue;
77        }
78        if !align_model && rows.clone().nth(1).is_none() {
79            phases.push(None);
80            continue;
81        }
82        let mut phase = if circular {
83            let slope = fit_complex(rows.clone().map(|row| {
84                (
85                    row.weight,
86                    row.model / row.weight,
87                    row.corrected / row.weight,
88                )
89            }))?;
90            let amplitude = slope.norm();
91            if amplitude == 0.0 || !amplitude.is_finite() {
92                phases.push(None);
93                previous = Complex64::ONE;
94                continue;
95            }
96            slope / amplitude
97        } else {
98            ambiguity * fit_phase(rows.clone())
99        };
100        if !circular && channel > 0 && (phase / previous).re < 0.0 {
101            phase = -phase;
102            ambiguity = -ambiguity;
103        }
104        alignment += rows.map(|row| (row.alignment / phase).re).sum::<f64>();
105        previous = phase;
106        phases.push(Some(phase));
107    }
108    if !circular && align_model && alignment < 0.0 {
109        phases
110            .iter_mut()
111            .flatten()
112            .for_each(|phase| *phase = -*phase);
113    }
114    let qu = if circular {
115        let rows = segments.iter().filter_map(|segment| {
116            let (weight, model, corrected) = segment.channels.iter().zip(&phases).fold(
117                (0.0, Complex64::ZERO, Complex64::ZERO),
118                |(weight, model, corrected), (row, phase)| {
119                    let Some(phase) = phase.filter(|_| row.weight > 0.0) else {
120                        return (weight, model, corrected);
121                    };
122                    let amplitude = row.model.norm();
123                    if amplitude == 0.0 || !amplitude.is_finite() {
124                        return (weight, model, corrected);
125                    }
126                    (
127                        weight + row.weight,
128                        model + row.weight * row.model / amplitude,
129                        corrected + row.corrected / phase,
130                    )
131                },
132            );
133            (weight > 0.0).then(|| (weight, model / weight, corrected / weight))
134        });
135        fit_complex(rows)?
136    } else {
137        fit_linear(segments, &phases)?
138    };
139    Ok((
140        phases
141            .into_iter()
142            .map(|phase| phase.map(to_complex32))
143            .collect(),
144        to_complex32(qu),
145    ))
146}

148fn fit_phase<'a>(rows: impl Iterator<Item = &'a SegmentChannel>) -> Complex64 {
149    let (mut weight, mut sum, mut rr, mut ii, mut ri) = (0.0, Complex64::ZERO, 0.0, 0.0, 0.0);
150    for row in rows {
151        let v = row.corrected / row.weight;
152        weight += row.weight;
153        sum += row.corrected;
154        rr += row.weight * v.re * v.re;
155        ii += row.weight * v.im * v.im;
156        ri += row.weight * v.re * v.im;
157    }
158    rr -= sum.re * sum.re / weight;
159    ii -= sum.im * sum.im / weight;
160    ri -= sum.re * sum.im / weight;
161    let shallow = ri / rr;
162    let steep = ri / ii;
163    let angle = if shallow.abs() < steep.abs() {
164        shallow.atan()
165    } else {
166        steep.recip().atan()
167    };
168    Complex64::from_polar(1.0, if angle.is_finite() { angle } else { 0.0 })
169}

171fn fit_complex(
172    rows: impl Iterator<Item = (f64, Complex64, Complex64)>,
173) -> Result<Complex64, FitError> {
174    let (mut weight, mut power) = (0.0, 0.0);
175    let (mut model, mut corrected, mut product) =
176        (Complex64::ZERO, Complex64::ZERO, Complex64::ZERO);
177    for (w, m, d) in rows {
178        weight += w;
179        power += w * m.norm_sqr();
180        model += w * m;
181        corrected += w * d;
182        product += w * m.conj() * d;
183    }
184    let (values, rank) = casa_numerics::solve_complex_normal_equations(
185        vec![weight.into(), model, model.conj(), power.into()],
186        vec![corrected, product],
187    )
188    .ok_or(FitError::Singular)?;
189    if rank < 2 {
190        return Err(FitError::Singular);
191    }
192    Ok(values[1])
193}

195fn fit_linear(segments: &[Segment], phases: &[Option<Complex64>]) -> Result<Complex64, FitError> {
196    let mut matrix = [[0.0; 3]; 3];
197    let mut values = [0.0; 3];
198    for segment in segments {
199        let (weight, corrected) = segment.channels.iter().zip(phases).fold(
200            (0.0, 0.0),
201            |(weight, corrected), (row, phase)| {
202                if let Some(phase) = phase {
203                    (weight + row.weight, corrected + (row.corrected / *phase).re)
204                } else {
205                    (weight, corrected)
206                }
207            },
208        );
209        if weight <= 0.0 {
210            continue;
211        }
212        let (sine, cosine) = (2.0 * segment.angle).sin_cos();
213        let basis = [-sine, cosine, 1.0];
214        for j in 0..3 {
215            values[j] += basis[j] * corrected;
216            for k in j..3 {
217                matrix[j][k] += weight * basis[j] * basis[k];
218            }
219        }
220    }
221    let [q, u, _] =
222        casa_numerics::solve_symmetric_ldlt_casacore(matrix, values).ok_or(FitError::Singular)?;
223    Ok(Complex64::new(q, u))
224}

226fn to_complex32(value: Complex64) -> Complex32 {
227    Complex32::new(value.re as f32, value.im as f32)
