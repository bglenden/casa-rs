69pub(super) fn prepare(
70    attached: &MeasurementSet,
71    donor: &MeasurementSet,
72    destination: &Table,
73    plan: &MetadataPlan<'_>,
74    scale: f32,
75) -> MsResult<Option<MainAppend>> {
76    let map = &plan.row_map;
77    let source = donor.main_table();
78    if source.row_count() == 0 {
79        return Ok(None);
80    }
81    let read = || scalar_columns(source, SCALARS);
82    let (scalars, attached_scan) = super::join_with_tracing(read, || {
83        if destination.row_count() == 0 {
84            Ok(None)
85        } else {
86            scalar_columns(destination, &["OBSERVATION_ID", "SCAN_NUMBER"]).map(Some)
87        }
88    });
89    let mut scalars = scalars?;
90    let order = row_orders(source.row_count(), &scalars, map, true)?;
91    let kernels = ArrayTransform::new(donor, &order, scale)?;
92    let array_names = array_names(attached.main_table(), source, destination, |_| true)?;
93    let channel_selection = if array_names.contains(&"MODEL_DATA") {
94        Some(model_channel_selection(attached, Some(plan))?)
95    } else {
96        None
97    };
98    let passthrough = array_names
99        .iter()
100        .copied()
101        .filter(|column| !kernels.changes(column))
102        .collect::<Vec<_>>();
103    let encoded = destination.plan_encoded_array_append(source, &passthrough)?;
104    let materialized = array_names
105        .into_iter()
106        .filter(|name| !encoded.columns().any(|column| name == &column))
107        .collect::<Vec<_>>();
108    let bounds = attached_scan?
109        .as_ref()
110        .map(ScanBounds::from_main)
111        .transpose()?
112        .unwrap_or_default();
113    let scans = bounds.offsets(&scalars, &map.observations)?;
114    let rows = (0..source.row_count()).collect::<Vec<_>>();
115    let dispatch = tracing::dispatcher::get_default(Clone::clone);
116    let arrays = materialized
117        .par_iter()
118        .map(|&name| {
119            tracing::dispatcher::with_default(&dispatch, || {
120                let cells = source
121                    .column_accessor(name)?
122                    .array_cells_owned_uncached(&rows)?;
123                Ok((
124                    VisibilityWriteDestination::for_append(destination, name)?,
125                    cells.into_iter(),
126                ))
127            })
128        })
129        .collect::<MsResult<Vec<_>>>()?;
130    let mut arrays = arrays;
131    let mut records = Vec::with_capacity(rows.len());
132    for (row, order) in order.into_iter().enumerate() {
133        let mut fields = Vec::with_capacity(SCALARS.len() + arrays.len());
134        append_scalars(&mut fields, &mut scalars, row, order, map, &scans)?;
135        for (column, cells) in &mut arrays {
136            if let Some(Some(array)) = cells.next() {
137                let array = kernels.apply(column.name(), order, array)?;
138                column.append_cell(array, &mut fields)?;
139            }
140        }
141        records.push(RecordValue::new(fields));
142    }
143    let reset_state =
144        (destination.row_count() > 0 && map.attached_has_state && !map.donor_has_state)
145            .then_some(ScalarValue::Int32(-1));
146    Ok(Some(MainAppend {
147        records,
148        encoded,
149        reset_state,
150        channel_selection,
151    }))
152}

160pub(super) fn row_orders(
161    rows: usize,
162    scalar: &ScalarColumns,
163    map: &MainRowMap,
164    reorder: bool,
165) -> MsResult<Vec<RowOrder>> {
166    (0..rows)
167        .map(|row| {
168            row_order(
169                map,
170                integer(scalar, "ANTENNA1", row)?,
171                integer(scalar, "ANTENNA2", row)?,
172                integer(scalar, "DATA_DESC_ID", row)?,
173                reorder,
174            )
175        })
176        .collect()
177}

179pub(super) fn row_order(
180    map: &MainRowMap,
181    first: i32,
182    second: i32,
183    description: i32,
184    reorder: bool,
185) -> MsResult<RowOrder> {
186    let a = row_id(*lookup(&map.antennas, first, "ANTENNA1")?)?;
187    let b = row_id(*lookup(&map.antennas, second, "ANTENNA2")?)?;
188    let description = lookup(&map.descriptions, description, "DATA_DESC_ID")?;
189    let flipped = reorder && a > b;
190    Ok(RowOrder {
191        first: if flipped { b } else { a },
192        second: if flipped { a } else { b },
193        description: description.id,
194        polarization: description.polarization,
195        flipped,
196        reverse: description.reverse,
197    })
198}

200fn append_scalars(
201    fields: &mut Vec<RecordField>,
202    scalar: &mut ScalarColumns,
203    row: usize,
204    order: RowOrder,
205    map: &MainRowMap,
206    scans: &ScanOffsets,
207) -> MsResult<()> {
208    let old_observation = integer(scalar, "OBSERVATION_ID", row)?;
209    let observation = sparse_id(&map.observations, old_observation)?;
210    let scan = scans.remap(
211        old_observation,
212        observation,
213        integer(scalar, "SCAN_NUMBER", row)?,
214    )?;
215    let (feed1, feed2) = if order.flipped {
216        ("FEED2", "FEED1")
217    } else {
218        ("FEED1", "FEED2")
219    };
220    let state = match (map.attached_has_state, map.donor_has_state) {
221        (false, false) => integer(scalar, "STATE_ID", row)?,
222        (true, true) => optional_id(&map.states, integer(scalar, "STATE_ID", row)?, "STATE_ID")?,
223        _ => -1,
224    };
225    let ids = [
226        ("ANTENNA1", order.first),
227        ("ANTENNA2", order.second),
228        ("DATA_DESC_ID", order.description),
229        ("FEED1", integer(scalar, feed1, row)?),
230        ("FEED2", integer(scalar, feed2, row)?),
231        (
232            "FIELD_ID",
233            *lookup(&map.fields, integer(scalar, "FIELD_ID", row)?, "FIELD_ID")?,
234        ),
235        ("OBSERVATION_ID", observation),
236        (
237            "PROCESSOR_ID",
238            sparse_id(&map.processors, integer(scalar, "PROCESSOR_ID", row)?)?,
239        ),
240        ("SCAN_NUMBER", scan),
241        ("STATE_ID", state),
242    ];
243    fields.extend(
244        ids.into_iter()
245            .map(|(name, id)| RecordField::new(name, Value::Scalar(ScalarValue::Int32(id)))),
246    );
247    for &name in &SCALARS[..6] {
248        let value = scalar
249            .get_mut(name)
250            .and_then(|cells| cells.get_mut(row))
251            .and_then(Option::take)
252            .ok_or_else(|| invalid(format!("missing MAIN {name} at row {row}")))?;
253        fields.push(RecordField::new(name, Value::Scalar(value)));
254    }
255    Ok(())
256}

258pub(super) fn integer(columns: &ScalarColumns, name: &str, row: usize) -> MsResult<i32> {
259    match columns.get(name).and_then(|cells| cells.get(row)) {
260        Some(Some(ScalarValue::Int32(value))) => Ok(*value),
261        _ => Err(invalid(format!(
262            "missing or non-int32 MAIN {name} at row {row}"
263        ))),
264    }
265}

267pub(super) fn lookup<'a, T>(indices: &'a [T], id: i32, column: &str) -> MsResult<&'a T> {
268    usize::try_from(id)
269        .ok()
270        .and_then(|id| indices.get(id))
271        .ok_or_else(|| invalid(format!("MAIN {column} index {id} is outside its subtable")))
272}

274pub(super) fn sparse_id(indices: &[usize], id: i32) -> MsResult<i32> {
275    // Older MeasurementSets can use zero with an empty PROCESSOR subtable.
276    // CASA remaps only IDs found in the observation and processor maps.
277    match usize::try_from(id)
278        .ok()
279        .and_then(|index| indices.get(index))
280    {
281        Some(&mapped) => row_id(mapped),
282        None => Ok(id),
283    }
284}
285
286pub(super) fn optional_id(indices: &[usize], id: i32, column: &str) -> MsResult<i32> {
287    if id == -1 {
288        Ok(id)
289    } else {
290        row_id(*lookup(indices, id, column)?)
291    }

294pub(super) fn model_channel_selection(
295    attached: &MeasurementSet,
296    plan: Option<&MetadataPlan<'_>>,
297) -> MsResult<Value> {
298    use casa_ms::subtables::{SubTable, get_i32};
299    let windows = attached.spectral_window()?;
300    let mut channel_count = (0..windows.table().row_count())
301        .map(|row| get_i32(windows.table(), row, "NUM_CHAN"))
302        .collect::<MsResult<Vec<_>>>()?;
303    if let Some(change) = plan.and_then(|plan| {
304        plan.changes
305            .iter()
306            .find(|change| change.table == SubtableId::SpectralWindow)
307    }) {
308        for row in &change.rows.appended {
309            let Some(Value::Scalar(ScalarValue::Int32(channels))) = row.get("NUM_CHAN") else {
310                return Err(invalid("SPECTRAL_WINDOW has no channel count"));
311            };
312            channel_count.push(*channels);
313        }
314    }
315    let selection = ArrayD::from_shape_fn(IxDyn(&[2, channel_count.len()]), |position| {
316        if position[0] == 0 {
317            0
318        } else {
319            channel_count[position[1]]
320        }
321    });
322    Ok(Value::Array(ArrayValue::Int32(selection)))
323}

325pub(super) fn array_names(
326    attached: &Table,
327    source: &Table,
328    destination: &Table,
329    needs_optional: impl Fn(&str) -> bool,
330) -> MsResult<Vec<&'static str>> {
331    let contains = |table: &Table, name| {
332        table
333            .schema()
334            .is_some_and(|schema| schema.contains_column(name))
335    };
336    let floats = contains(attached, "FLOAT_DATA");
337    let visibility = if floats { "FLOAT_DATA" } else { "DATA" };
338    let mut columns = vec![visibility, "FLAG", "WEIGHT", "SIGMA", "UVW"];
339    for name in ["MODEL_DATA", "CORRECTED_DATA"] {
340        if contains(attached, name) {
341            if !contains(source, name) {
342                return Err(invalid(format!(
343                    "incoming MeasurementSet is missing {name}"
344                )));
345            }
346            columns.push(name);
347        }
348    }
349    if !contains(source, visibility) || !contains(destination, visibility) {
350        return Err(invalid(format!("MeasurementSet is missing {visibility}")));
351    }
352    for name in ["FLAG_CATEGORY", "WEIGHT_SPECTRUM", "SIGMA_SPECTRUM"] {
353        if needs_optional(name)
354            && contains(destination, name)
355            && contains(source, name)
356            && (std::ptr::eq(source, destination)
357                || (first_output_cell_defined(destination, name)?
358                    && first_array_is_defined(source, name)?))
359        {
360            columns.push(name);
361        }
362    }
363    Ok(columns)
364}
365
366fn first_output_cell_defined(table: &Table, name: &str) -> MsResult<bool> {
367    if table.row_count() != 0 {
368        return first_array_is_defined(table, name);
369    }
370    // Fixed-shape cells are already defined after CASA allocates its
371    // destination rows, including a newly created named output.
372    Ok(matches!(
373        table
374            .schema()
375            .and_then(|schema| schema.column(name))
376            .map(|column| column.column_type()),
377        Some(ColumnType::Array(ArrayShapeContract::Fixed { .. }))
378    ))
379}
380
381fn first_array_is_defined(table: &Table, column: &str) -> MsResult<bool> {
382    let value = table
383        .column_accessor(column)?
384        .array_cells_owned_uncached(&[0])?;
385    Ok(value.first().is_some_and(Option::is_some))
386}

388fn polarization_swaps(
389    donor: &MeasurementSet,
390    order: &[RowOrder],
391) -> MsResult<BTreeMap<usize, Vec<usize>>> {
392    let table = required_table(donor, SubtableId::Polarization)?;
393    let ids = order
394        .iter()
395        .filter(|row| row.flipped)
396        .map(|row| row.polarization)
397        .collect::<BTreeSet<_>>();
398    ids.into_iter()
399        .map(|id| {
400            let ArrayValue::Int32(products) = get_array(table, id, "CORR_PRODUCT")? else {
401                return Err(invalid("POLARIZATION.CORR_PRODUCT must hold integers"));
402            };
403            if products.ndim() != 2 || products.shape()[0] != 2 {
404                return Err(invalid(
405                    "POLARIZATION.CORR_PRODUCT must be a two-receptor matrix",
406                ));
407            }
408            let n = products.shape()[1];
409            let partners = (0..n)
410                .map(|i| {
411                    (0..n)
412                        .find(|&j| {
413                            products[[0, i]] == products[[1, j]]
414                                && products[[1, i]] == products[[0, j]]
415                        })
416                        .ok_or_else(|| {
417                            invalid("baseline reversal has no matching correlation product")
418                        })
419                })
420                .collect::<MsResult<Vec<_>>>()?;
421            Ok((id, partners))
422        })
423        .collect()
424}

426pub(super) struct ArrayTransform {
427    polarizations: BTreeMap<usize, Vec<usize>>,
428    pub(super) any_flipped: bool,
429    any_reversed: bool,
430    scale: f32,
431}
432
433impl ArrayTransform {
434    pub(super) fn new(donor: &MeasurementSet, order: &[RowOrder], scale: f32) -> MsResult<Self> {
435        Ok(Self {
436            polarizations: polarization_swaps(donor, order)?,
437            any_flipped: order.iter().any(|row| row.flipped),
438            any_reversed: order.iter().any(|row| row.reverse),
439            scale: if scale > 0.0 { scale } else { 1.0 },
440        })
441    }
442
443    pub(super) fn changes(&self, name: &str) -> bool {
444        (self.any_flipped && name != "FLOAT_DATA")
445            || (self.any_reversed && channel_axis(name))
446            || (self.scale != 1.0 && weight_column(name))
447    }
448
449    pub(super) fn apply(
450        &self,
451        name: &str,
452        row: RowOrder,
453        mut value: ArrayValue,
454    ) -> MsResult<ArrayValue> {
455        if name == "UVW" {
456            if row.flipped {
457                let ArrayValue::Float64(uvw) = &mut value else {
458                    return Err(invalid("UVW must hold doubles"));
459                };
460                uvw.mapv_inplace(|pixel| -pixel);
461            }
462            return Ok(value);
463        }
464        let permute = if row.flipped && name != "FLOAT_DATA" {
465            Some(self.polarizations[&row.polarization].as_slice())
466        } else {
467            None
468        };
469        let reverse = row.reverse && channel_axis(name);
470        let scale = match name {
471            "WEIGHT" | "WEIGHT_SPECTRUM" => self.scale,
472            "SIGMA" | "SIGMA_SPECTRUM" => self.scale.sqrt().recip(),
473            _ => 1.0,
474        };
475        if permute.is_none() && !reverse && scale == 1.0 {
476            return Ok(value);
477        }
478        Ok(match value {
479            ArrayValue::Complex32(array) => {
480                let mut array = reorder(array, permute, reverse)?;
481                if row.flipped {
482                    array.mapv_inplace(|sample| sample.conj());
483                }
484                ArrayValue::Complex32(array)
485            }
486            ArrayValue::Float32(array) => {
487                let mut array = reorder(array, permute, reverse)?;
488                if scale != 1.0 {
489                    array.mapv_inplace(|sample| sample * scale);
490                }
491                ArrayValue::Float32(array)
492            }
493            ArrayValue::Bool(array) => ArrayValue::Bool(reorder(array, permute, reverse)?),
494            _ => {
495                return Err(invalid(format!(
496                    "unsupported {name} array for concatenation"
497                )));
498            }
499        })
500    }
501}
502
503fn reorder<T: Clone>(
504    mut cell: ArrayD<T>,
505    permutation: Option<&[usize]>,
506    reverse: bool,
507) -> MsResult<ArrayD<T>> {
508    if let Some(permutation) = permutation {
509        if cell.shape().first() != Some(&permutation.len()) {
510            return Err(invalid("correlation axis does not match POLARIZATION"));
511        }
512        if permutation.iter().copied().ne(0..permutation.len()) {
513            cell = cell.select(Axis(0), permutation);
514        }
515    }
516    if reverse {
517        if cell.ndim() < 2 {
518            return Err(invalid("visibility array has no channel axis"));
519        }
520        cell.invert_axis(Axis(1));
521    }
522    Ok(cell)
523}
524
525fn channel_axis(name: &str) -> bool {
526    !matches!(name, "UVW" | "WEIGHT" | "SIGMA")
527}
528
529fn weight_column(name: &str) -> bool {
530    matches!(
531        name,
532        "WEIGHT" | "SIGMA" | "WEIGHT_SPECTRUM" | "SIGMA_SPECTRUM"
533    )
534}
