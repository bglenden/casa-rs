14/// The data a solve selected: the sorted, distinct values each axis of its
15/// cross-correlation rows takes, and the intents the MeasurementSet names.
16///
17/// CASA's `gaincal` and `bandpass` return this as `selectvis`. Their
18/// selection always leaves out autocorrelations (`ANTENNA1!=ANTENNA2`), so a
19/// row whose antennas are equal does not count here either. Flags do not
20/// narrow it: an antenna whose data are all flagged is still selected.
21/// The intents are not narrowed by the selection at all: CASA 6.7.6 lists
22/// every `OBS_MODE` of the `STATE` table, split at its commas, sorted and
23/// distinct, and none when `STATE` has no rows. An empty `OBS_MODE` names
24/// none, though an empty item between two commas is kept.
25#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
26pub struct SelectedData {
27    /// The antennas of the selected baselines, `ANTENNA1` and `ANTENNA2`.
28    pub antennas: Vec<i32>,
29    /// The selected `FIELD_ID`s.
30    pub fields: Vec<i32>,
31    /// The intents the `STATE` table names.
32    pub intents: Vec<String>,
33    /// The selected `OBSERVATION_ID`s.
34    pub observations: Vec<i32>,
35    /// The selected `SCAN_NUMBER`s.
36    pub scans: Vec<i32>,
37    /// The selected spectral windows, through `DATA_DESCRIPTION`.
38    pub spectral_windows: Vec<i32>,
39}
40
41impl SelectedData {
42    /// The data `rows` of `ms` select.
43    pub(crate) fn of(
44        ms: &MeasurementSet,
45        rows: &[SelectedSolveRow],
46    ) -> Result<Self, GainSolveError> {
47        let mut antennas = BTreeSet::new();
48        let mut fields = BTreeSet::new();
49        let mut observations = BTreeSet::new();
50        let mut scans = BTreeSet::new();
51        let mut spectral_windows = BTreeSet::new();
52        for row in rows.iter().filter(|row| row.antenna1 != row.antenna2) {
53            antennas.extend([row.antenna1, row.antenna2]);
54            fields.insert(row.field_id);
55            observations.insert(row.observation_id);
56            scans.insert(row.scan_number);
57            spectral_windows.insert(row.data_spw_id);
58        }
59        Ok(Self {
60            antennas: antennas.into_iter().collect(),
61            fields: fields.into_iter().collect(),
62            intents: intents(ms)?,
63            observations: observations.into_iter().collect(),
64            scans: scans.into_iter().collect(),
65            spectral_windows: spectral_windows.into_iter().collect(),
66        })
67    }
68}

70/// Every intent `ms`'s `STATE` table names, sorted and distinct; none when
71/// the MeasurementSet has no `STATE` table.
72fn intents(ms: &MeasurementSet) -> Result<Vec<String>, GainSolveError> {
73    let Ok(state) = ms.state() else {
74        return Ok(Vec::new());
75    };
76    let failed = |source| GainSolveError::OpenMeasurementSet {
77        path: ms
78            .path()
79            .map(|path| path.display().to_string())
80            .unwrap_or_else(|| "<in-memory>".to_string()),
81        source,
82    };
83    let modes = (0..state.row_count())
84        .map(|row| state.string(row, "OBS_MODE").map_err(failed))
85        .collect::<Result<Vec<_>, _>>()?;
86    Ok(intents_of(modes.iter().map(String::as_str)))
87}
88
89/// The intents `modes` name, as CASA 6.7.6.14 lists them: each mode split
90/// at its commas, untrimmed, sorted and distinct. An empty mode names none,
91/// though an empty item between two commas is kept.
92fn intents_of<'a>(modes: impl IntoIterator<Item = &'a str>) -> Vec<String> {
93    modes
94        .into_iter()
95        .filter(|mode| !mode.is_empty())
96        .flat_map(|mode| mode.split(','))
97        .map(str::to_string)
98        .collect::<BTreeSet<_>>()
99        .into_iter()
100        .collect()
101}

103#[cfg(test)]
104mod tests {
105    use super::intents_of;
106
107    #[test]
108    fn intents_are_casas_split_sorted_and_distinct() {
109        // Each case as CASA 6.7.6.14's gaincal lists it in `selectvis`.
110        for (modes, expected) in [
111            (&["B,A", "C", "B"][..], &["A", "B", "C"][..]),
112            (&[""], &[]),
113            (&["", "A"], &["A"]),
114            (&["A, B"], &[" B", "A"]),
115            (&["A,,B"], &["", "A", "B"]),
116        ] {
117            assert_eq!(intents_of(modes.iter().copied()), expected, "{modes:?}");
118        }
119    }
