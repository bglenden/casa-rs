126/// Merge sparse `tclean` overrides over [`default_tclean_parameters`],
127/// applying conditional AW-CF defaults, canonicalizing `cell` (`arcsec`) and
128/// `threshold` (`Jy`), and normalizing numeric `phasecenter`/`start`/`width`
129/// scalars to strings.
130pub(crate) fn merge_tclean_defaults(overrides: &Map<String, Value>) -> Map<String, Value> {
131    let mut merged = default_tclean_parameters();
132    let gridder = overrides
133        .get("gridder")
134        .or_else(|| merged.get("gridder"))
135        .and_then(Value::as_str);
136    if gridder == Some("awproject") {
137        merged.insert("aw_cf_source".into(), json!("casa-import"));
138        merged.insert("cf_resident_mb".into(), json!("auto"));
139        let source = overrides
140            .get("aw_cf_source")
141            .and_then(Value::as_str)
142            .unwrap_or("casa-import");
143        if source == "native-evla" {
144            for (k, v) in [
145                ("native_cf_cache", "none"),
146                ("evla_surface", "none"),
147                ("native_cf_policy", "reuse-only"),
148                ("native_cf_working_size", "none"),
149                ("native_cf_oversampling", "none"),
150                ("native_cf_cache_bytes", "none"),
151                ("native_cf_maximum_cells", "none"),
152            ] {
153                merged.insert(k.into(), json!(v));
154            }
155        }
156    }
157    for (key, value) in overrides {
158        merged.insert(key.clone(), value.clone());
159    }
160    for text_or_number in ["phasecenter", "start", "width"] {
161        if let Some(value) = merged.get_mut(text_or_number)
162            && !value.is_string()
163            && !value.is_array()
164            && !value.is_object()
165            && !value.is_null()
166        {
167            *value = Value::String(value.to_string());
168        }
169    }
170    if let Some(cell) = merged.get_mut("cell") {
171        *cell = canonicalize_cell(cell);
172    }
173    if let Some(threshold) = merged.get_mut("threshold") {
174        *threshold = canonicalize_threshold(threshold);
175    }
176    merged
177}
178
179fn canonicalize_cell(value: &Value) -> Value {
180    let convert_one = |item: &Value| -> Value {
181        if let Some(text) = item.as_str()
182            && let Some(canonical) = convert_quantity(text, "arcsec")
183        {
184            return Value::String(canonical);
185        }
186        if let Some(number) = item.as_f64() {
187            return Value::String(format!("{number}arcsec"));
188        }
189        item.clone()
190    };
191    match value {
192        Value::Array(items) => Value::Array(items.iter().map(convert_one).collect()),
193        other => convert_one(other),
194    }
195}
196
197fn canonicalize_threshold(value: &Value) -> Value {
198    if let Some(number) = value.as_f64() {
199        return Value::String(format!("{number}Jy"));
200    }
201    if let Some(text) = value.as_str()
202        && let Some(canonical) = convert_quantity(text, "Jy")
203    {
204        return Value::String(canonical);
205    }
206    value.clone()
207}
208
209fn convert_quantity(text: &str, target_unit: &str) -> Option<String> {
210    let trimmed = text.trim();
211    if trimmed.is_empty() {
212        return None;
213    }
214    let quantity = Quantity::from_str(trimmed).ok()?;
215    let target = Unit::from_str(target_unit).ok()?;
216    if quantity.unit().name().is_empty() {
217        return Some(format!("{}{target_unit}", quantity.value()));
218    }
219    let converted = quantity.get_value_in(&target).ok()?;
220    if converted.is_finite() {
221        Some(format!("{converted}{target_unit}"))
222    } else {
223        None
224    }
225}
