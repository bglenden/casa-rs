24/// Common on-disk identity for one calibration-table family.
25pub(crate) struct CalibrationTableDescriptor<'a> {
26    pub output: &'a Path,
27    pub schema: TableSchema,
28    pub subtype: &'a str,
29    pub parameter_type: Option<&'a str>,
30    /// The MeasurementSet solved on, as a path or a name; the table records
31    /// its base name as `MSName`, as CASA's `NewCalTable::setMetaInfo`
32    /// does, and CASA appends only to a table whose `MSName` is the one it
33    /// would write.
34    pub measurement_set_name: String,
35    pub include_polarization_basis: bool,
36    pub time_extra_precision_column: Option<&'a str>,
37}

39/// Common calibration-table writer failure.
40#[derive(Debug, Error)]
41pub enum CalibrationTableWriteError {
42    #[error("failed to prepare output path {path}: {reason}")]
43    /// Could not prepare a calibration output path.
44    Prepare {
45        /// Requested output path.
46        path: String,
47        /// Diagnostic from staging preparation.
48        reason: String,
49    },
50    #[error("failed to append calibration row to {path}: {source}")]
51    /// A typed calibration row could not be appended.
52    Append {
53        /// Path of the calibration table being built.
54        path: String,
55        #[source]
56        /// Underlying row or schema error.
57        source: Box<TableError>,
58    },
59    #[error("failed to save calibration table {path}: {source}")]
60    /// The completed staged table could not be saved.
61    Save {
62        /// Path of the calibration table being saved.
63        path: String,
64        #[source]
65        /// Underlying table storage error.
66        source: Box<TableError>,
67    },
68    #[error("failed to copy {subtable} subtable into {path}: {source}")]
69    /// A required MeasurementSet subtable could not be copied.
70    CopySubtable {
71        /// CASA subtable name.
72        subtable: String,
73        /// Destination calibration table path.
74        path: String,
75        #[source]
76        /// Underlying table-copy error.
77        source: Box<TableError>,
78    },
79    #[error("failed to publish staged calibration table {staging} as {output}: {reason}")]
80    /// The staged table could not be made visible at its final path.
81    Publish {
82        /// Path of the fully staged calibration table.
83        staging: String,
84        /// Requested final output path.
85        output: String,
86        /// Diagnostic from the publication operation.
87        reason: String,
88    },
89}

91/// Typed owner of common calibration-table construction and final persistence.
92pub(crate) struct CalibrationTableWriter<'a> {
93    measurement_set: &'a MeasurementSet,
94    output: CalibrationOutput,
95    table: Table,
96    mode: CalibrationOutputMode,
97    sort_for_casa: bool,
98}
99
100impl<'a> CalibrationTableWriter<'a> {
101    pub(crate) fn create_simulated(
102        measurement_set: &'a MeasurementSet,
103        descriptor: CalibrationTableDescriptor<'_>,
104    ) -> Result<Self, CalibrationTableWriteError> {
105        let output = CalibrationOutput::prepare_simulated(descriptor.output)?;
106        let mut writer = Self::staged(
107            measurement_set,
108            descriptor,
109            output,
110            CalibrationOutputMode::Replace,
111        );
112        // Simulation supplies its established row order and budgets one
113        // materialized table. Do not duplicate it for task storeNCT sorting.
114        writer.preserve_row_order();
115        Ok(writer)
116    }
117
118    pub(crate) fn preserve_row_order(&mut self) {
119        self.sort_for_casa = false;
120    }
121
122    pub(crate) fn with_mode(
123        measurement_set: &'a MeasurementSet,
124        descriptor: CalibrationTableDescriptor<'_>,
125        mode: CalibrationOutputMode,
126    ) -> Result<Self, CalibrationTableWriteError> {
127        let output = CalibrationOutput::prepare(descriptor.output, mode)?;
128        Ok(Self::staged(measurement_set, descriptor, output, mode))
129    }
130
131    pub(crate) fn staged(
132        measurement_set: &'a MeasurementSet,
133        descriptor: CalibrationTableDescriptor<'_>,
134        output: CalibrationOutput,
135        mode: CalibrationOutputMode,
136    ) -> Self {
137        let mut table = Table::with_schema(descriptor.schema);
138        table.set_info(TableInfo {
139            table_type: TABLE_INFO_TYPE.to_string(),
140            sub_type: descriptor.subtype.to_string(),
141            readme: Vec::new(),
142        });
143        let measurement_set_name = Path::new(&descriptor.measurement_set_name)
144            .file_name()
145            .map_or(descriptor.measurement_set_name.clone(), |name| {
146                name.to_string_lossy().into_owned()
147            });
148        let producer = if descriptor.subtype == "G Jones" {
149            concat!("casa-rs ", env!("CARGO_PKG_VERSION"))
150        } else {
151            "casa-rs"
152        };
153        for (key, value) in [
154            (KEY_VIS_CAL, descriptor.subtype.to_string()),
155            (KEY_MS_NAME, measurement_set_name),
156            (KEY_CASA_VERSION, producer.to_string()),
157        ] {
158            table
159                .keywords_mut()
160                .upsert(key, Value::Scalar(ScalarValue::String(value)));
161        }
162        if let Some(parameter_type) = descriptor.parameter_type {
163            table.keywords_mut().upsert(
164                KEY_PAR_TYPE,
165                Value::Scalar(ScalarValue::String(parameter_type.to_string())),
166            );
167        }
168        if descriptor.include_polarization_basis {
169            table.keywords_mut().upsert(
170                KEY_POL_BASIS,
171                Value::Scalar(ScalarValue::String("unknown".to_string())),
172            );
173        }
174        set_fixed_unit_keyword(&mut table, COL_TIME, &["s"]);
175        set_measinfo_keyword(&mut table, COL_TIME, "epoch", Some("UTC"));
176        set_fixed_unit_keyword(&mut table, COL_INTERVAL, &["s"]);
177        if let Some(column) = descriptor.time_extra_precision_column {
178            set_fixed_unit_keyword(&mut table, column, &["s"]);
179        }
180        for name in STANDARD_SUBTABLE_KEYWORDS {
181            table.keywords_mut().upsert(
182                *name,
183                Value::table_ref(strip_directory(&output.staging.join(name), &output.staging)),
184            );
185        }
186        Self {
187            measurement_set,
188            output,
189            table,
190            mode,
191            sort_for_casa: true,
192        }
193    }
194
195    /// Staging root used for family-specific subtables before publication.
196    pub(crate) fn output_path(&self) -> &Path {
197        &self.output.staging
198    }
199
200    pub(crate) fn table_mut(&mut self) -> &mut Table {
201        &mut self.table
202    }
203
204    pub(crate) fn append(&mut self, row: RecordValue) -> Result<(), CalibrationTableWriteError> {
205        self.table
206            .add_row(row)
207            .map_err(|source| CalibrationTableWriteError::Append {
208                path: self.output.staging.display().to_string(),
209                source: Box::new(source),
210            })
211    }
212
213    pub(crate) fn copy_subtables(
214        &self,
215        subtables: &[(SubtableId, &'static str)],
216    ) -> Result<(), CalibrationTableWriteError> {
217        for &(id, name) in subtables {
218            self.measurement_set
219                .subtable(id)
220                .expect("required calibration subtable available")
221                .save(TableOptions::new(self.output.staging.join(name)))
222                .map_err(|source| CalibrationTableWriteError::CopySubtable {
223                    subtable: name.to_string(),
224                    path: self.output.staging.display().to_string(),
225                    source: Box::new(source),
226                })?;
227        }
228        Ok(())
229    }
230
231    pub(crate) fn save_main(mut self) -> Result<(), CalibrationTableWriteError> {
232        if self.mode == CalibrationOutputMode::Append && self.output.destination.exists() {
233            if matches!(
234                self.table.info().sub_type.as_str(),
235                "BPOLY" | "GSPLINE" | "EP Jones"
236            ) {
237                let mut old = output_merge::open(&self.output.destination)?;
238                output_merge::append_legacy(
239                    &mut old,
240                    &mut self.table,
241                    &self.output.destination,
242                    &self.output.staging,
243                )?;
244                self.table = old;
245            } else {
246                let spw = output_merge::merge_spw(&self.output.destination, &self.output.staging)?;
247                let mut old = output_merge::open(&self.output.destination)?;
248                output_merge::append(&mut old, &self.table, &self.output.destination)?;
249                output_merge::preserve_subtables(&self.output.destination, &self.output.staging)?;
250                spw.save(TableOptions::new(
251                    self.output.staging.join("SPECTRAL_WINDOW"),
252                ))
253                .map_err(|e| output_merge::invalid(&self.output.staging, e.to_string()))?;
254                self.table = old;
255            }
256        }
257        if self.sort_for_casa {
258            output_merge::save_sorted(&mut self.table, &self.output.staging)?;
259        } else {
260            self.table
261                .save(
262                    TableOptions::new(&self.output.staging)
263                        .with_data_manager(casa_tables::DataManagerKind::StandardStMan),
264                )
265                .map_err(|source| CalibrationTableWriteError::Save {
266                    path: self.output.staging.display().to_string(),
267                    source: Box::new(source),
268                })?;
269        }
270        self.output.commit()
271    }
272}

274pub(crate) fn set_fixed_unit_keyword(table: &mut Table, column: &str, units: &[&str]) {
275    let mut keywords = table.column_keywords(column).cloned().unwrap_or_default();
276    keywords.upsert(
277        "QuantumUnits",
278        Value::Array(ArrayValue::from_string_vec(
279            units.iter().map(|unit| (*unit).to_string()).collect(),
280        )),
281    );
282    table.set_column_keywords(column, keywords);
283}

285pub(crate) fn set_measinfo_keyword(
286    table: &mut Table,
287    column: &str,
288    measure_type: &str,
289    measure_ref: Option<&str>,
290) {
291    let mut keywords = table.column_keywords(column).cloned().unwrap_or_default();
292    let mut fields = vec![RecordField::new(
293        "type",
294        Value::Scalar(ScalarValue::String(measure_type.to_string())),
295    )];
296    if let Some(measure_ref) = measure_ref {
297        fields.push(RecordField::new(
298            "Ref",
299            Value::Scalar(ScalarValue::String(measure_ref.to_string())),
300        ));
301    }
302    keywords.upsert("MEASINFO", Value::Record(RecordValue::new(fields)));
303    table.set_column_keywords(column, keywords);
304}

306pub(crate) enum SingleChannelFrequency {
307    MiddleChannel,
308    Mean,
309}
310
311/// Reduce every spectral window of the SPECTRAL_WINDOW subtable at `path`
312/// to one channel, as CASA's `NewCalTable::makeSpwSingleChan` does for a
313/// term that does not depend on frequency: the middle channel's frequency
314/// (`CHAN_FREQ[n/2]`), and CHAN_WIDTH, EFFECTIVE_BW and RESOLUTION the
315/// window's TOTAL_BANDWIDTH. CASA's `applycal` refuses a table whose
316/// windows keep the MeasurementSet's channels.
317pub(crate) fn make_windows_single_channel(
318    path: &Path,
319    frequency: SingleChannelFrequency,
320) -> Result<(), CalibrationTableWriteError> {
321    let failed = |source: TableError| CalibrationTableWriteError::Save {
322        path: path.display().to_string(),
323        source: Box::new(source),
324    };
325    let mut table = Table::open(TableOptions::new(path)).map_err(failed)?;
326    for row in 0..table.row_count() {
327        let value = |column: &str| -> Result<Value, TableError> {
328            Ok(table
329                .cell_accessor(row, column)?
330                .value()?
331                .cloned()
332                .unwrap_or_else(|| Value::Scalar(ScalarValue::Int32(0))))
333        };
334        let frequencies = match value("CHAN_FREQ").map_err(failed)? {
335            Value::Array(ArrayValue::Float64(frequencies)) => frequencies,
336            _ => continue,
337        };
338        let channels = frequencies.len();
339        if channels <= 1 {
340            continue;
341        }
342        let bandwidth = match value("TOTAL_BANDWIDTH").map_err(failed)? {
343            Value::Scalar(ScalarValue::Float64(bandwidth)) => bandwidth,
344            _ => continue,
345        };
346        let one = |value: f64| {
347            Value::Array(ArrayValue::Float64(
348                ArrayD::from_shape_vec(IxDyn(&[1]), vec![value]).expect("one channel"),
349            ))
350        };
351        let frequency = match frequency {
352            SingleChannelFrequency::MiddleChannel => frequencies
353                .iter()
354                .nth(channels / 2)
355                .copied()
356                .unwrap_or_default(),
357            SingleChannelFrequency::Mean => frequencies.iter().sum::<f64>() / channels as f64,
358        };
359        for (column, cell) in [
360            ("NUM_CHAN", Value::Scalar(ScalarValue::Int32(1))),
361            ("CHAN_FREQ", one(frequency)),
362            ("CHAN_WIDTH", one(bandwidth)),
363            ("EFFECTIVE_BW", one(bandwidth)),
364            ("RESOLUTION", one(bandwidth)),
365        ] {
366            table
367                .cell_accessor_mut(row, column)
368                .and_then(|mut accessor| accessor.set(cell))
369                .map_err(failed)?;
370        }
371    }
372    table.flush().map_err(failed)
373}

375pub(crate) fn new_calibration_table_schema(complex: bool) -> TableSchema {
376    let (column, parameter) = if complex {
377        (COL_CPARAM, PrimitiveType::Complex32)
378    } else {
379        (COL_FPARAM, PrimitiveType::Float32)
380    };
381    TableSchema::new(vec![
382        ColumnSchema::scalar(COL_TIME, PrimitiveType::Float64),
383        ColumnSchema::scalar(COL_FIELD_ID, PrimitiveType::Int32),
384        ColumnSchema::scalar(COL_SPECTRAL_WINDOW_ID, PrimitiveType::Int32),
385        ColumnSchema::scalar(COL_ANTENNA1, PrimitiveType::Int32),
386        ColumnSchema::scalar(COL_ANTENNA2, PrimitiveType::Int32),
387        ColumnSchema::scalar(COL_INTERVAL, PrimitiveType::Float64),
388        ColumnSchema::scalar(COL_SCAN_NUMBER, PrimitiveType::Int32),
389        ColumnSchema::scalar(COL_OBSERVATION_ID, PrimitiveType::Int32),
390        ColumnSchema::array_variable(column, parameter, None),
391        ColumnSchema::array_variable(COL_PARAMERR, PrimitiveType::Float32, None),
392        ColumnSchema::array_variable(COL_FLAG, PrimitiveType::Bool, None),
393        ColumnSchema::array_variable(COL_SNR, PrimitiveType::Float32, None),
394        ColumnSchema::array_variable(COL_WEIGHT, PrimitiveType::Float32, None),
395    ])
396    .expect("valid simulated calibration schema")
397}

402    use tempfile::tempdir;
403
404    use crate::output::CalibrationOutput;
405    use crate::output::CalibrationOutputMode;
406    use std::fs;

408    #[test]
409    fn output_collision_is_rejected_without_modifying_existing_data() {
410        let temp = tempdir().expect("tempdir");
411        let output = temp.path().join("gain.cal");
412        fs::create_dir(&output).expect("create existing output");
413        let sentinel = output.join("sentinel");
414        fs::write(&sentinel, b"keep").expect("write sentinel");
415
416        let error = CalibrationOutput::prepare(&output, CalibrationOutputMode::Create)
417            .expect_err("collision must fail");
418
419        assert!(error.to_string().contains("output already exists"));
420        assert_eq!(fs::read(&sentinel).expect("sentinel remains"), b"keep");
421    }
422
423    #[test]
424    fn interrupted_staging_output_is_visible_and_blocks_reuse() {
425        let temp = tempdir().expect("tempdir");
426        let output = temp.path().join("bandpass.cal");
427        let staging = CalibrationOutput::prepare(&output, CalibrationOutputMode::Create)
428            .expect("reserve staging output");
429        fs::write(staging.staging.join("partial"), b"incomplete").expect("write partial output");
430
431        let error = CalibrationOutput::prepare(&output, CalibrationOutputMode::Create)
432            .expect_err("staging collision must fail");
433
434        assert!(!output.exists());
435        assert_eq!(
436            fs::read(staging.staging.join("partial")).expect("partial remains"),
437            b"incomplete"
438        );
439        assert!(
440            error
441                .to_string()
442                .contains("another writer or interrupted publication")
443        );
444    }
