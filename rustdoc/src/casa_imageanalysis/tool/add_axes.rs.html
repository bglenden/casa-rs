22const COPY_BLOCK_PIXELS: usize = 1_048_576;
23
24/// Parameters specifying which degenerate axes to append to an image.
25#[derive(Debug, Clone)]
26pub struct AddDegAxesRequest<'a> {
27    /// Output image path (empty creates a temporary in-memory image).
28    pub outfile: &'a str,
29    /// Append a degenerate direction coordinate (RA/Dec).
30    pub direction: bool,
31    /// Append a degenerate spectral axis.
32    pub spectral: bool,
33    /// Stokes polarization string to append when non-empty (e.g. `"I"`).
34    pub stokes: &'a str,
35    /// Append a degenerate linear axis.
36    pub linear: bool,
37    /// Append a degenerate tabular axis.
38    pub tabular: bool,
39    /// Overwrite `outfile` if it exists.
40    pub overwrite: bool,
41    /// Suppress erroring when requested coordinate already exists or no axes are added.
42    pub silent: bool,
43}
44
45impl<'a> AddDegAxesRequest<'a> {
46    /// Extends `record` with any requested missing coordinates and parses the resulting [`CoordinateSystem`].
47    pub fn build_coordinates(
48        &self,
49        record: &mut RecordValue,
50        ndim: usize,
51    ) -> ImageToolResult<CoordinateSystem> {
52        let expected = validate_coordinate_indices(record)?;
53        let mut coordinates = CoordinateSystem::from_record(record)?;
54        let original_count = coordinates.n_coordinates();
55        if original_count != expected || coordinates.n_pixel_axes() != ndim {
56            return Err(ImageToolError::new(
57                "Image coordinates do not describe all pixel axes",
58            ));
59        }
60        // New image records use CASA's typed field names. Translate the
61        // older generic spelling without rebuilding coordinate metadata.
62        for placement in CoordinateSystem::record_coordinates(record)? {
63            if let Some(index) = placement.field.strip_prefix("coordinate")
64                && let Some(Value::Record(axis)) = record.get(&placement.field)
65                && let Some(Value::Scalar(ScalarValue::String(kind))) = axis.get("coordinate_type")
66            {
67                let native_field = format!("{}{index}", kind.to_ascii_lowercase());
68                let axis = axis.clone();
69                record.remove(&placement.field);
70                record.upsert(native_field, axis.into());
71            }
72        }
73        let added: [Option<CoordinateModel>; 5] = [
74            self.direction
75                .then(|| defaults::direction_coordinate().into()),
76            self.spectral
77                .then(|| defaults::spectral_coordinate().into()),
78            (!self.stokes.is_empty()).then(|| {
79                let stokes = StokesType::from_name(self.stokes).unwrap_or(StokesType::Undefined);
80                StokesCoordinate::new(vec![stokes]).into()
81            }),
82            self.linear
83                .then(|| defaults::linear_coordinate(&["Axis1".into()], &[]).into()),
84            self.tabular
85                .then(|| TabularCoordinate::linear(0.0, 1.0, 0.0, "Tabular", "").into()),
86        ];
87        for coordinate in added.into_iter().flatten() {
88            let kind = coordinate.coordinate_type();
89            if coordinates.find_coordinate(kind).is_some() {
90                if !self.silent {
91                    return Err(ImageToolError::new(format!(
92                        "Image already contains a {kind}Coordinate"
93                    )));
94                }
95            } else {
96                append_coordinate(&mut coordinates, record, coordinate)?;
97            }
98        }
99        if coordinates.n_coordinates() == original_count && !self.silent {
100            return Err(ImageToolError::new(
101                "Exception: No degenerate axes specified.",
102            ));
103        }
104        Ok(coordinates)
105    }
106
107    fn history_command(&self) -> String {
108        format!(
109            "ia.adddegaxes(outfile={:?}, direction={}, spectral={}, stokes={:?}, linear={}, tabular={}, overwrite={}, silent={})",
110            self.outfile,
111            self.direction,
112            self.spectral,
113            self.stokes,
114            self.linear,
115            self.tabular,
116            self.overwrite,
117            self.silent
118        )
119    }
120}

122/// Appends requested degenerate axes to `attached`, returning a new [`AttachedImage`] and logs.
123pub fn add_degenerate_axes(
124    attached: &AttachedImage,
125    request: &AddDegAxesRequest<'_>,
126    record_history: bool,
127    is_external_open: ExternalPathCheck<'_>,
128) -> ImageToolResult<(AttachedImage, Vec<ToolLogLine>)> {
129    if !request.outfile.is_empty()
130        && !request.overwrite
131        && Path::new(request.outfile).symlink_metadata().is_ok()
132    {
133        return Err(ImageToolError::new(format!(
134            "File {} exists, and the user does not want to remove it.",
135            request.outfile
136        )));
137    }
138    let mut logs = Vec::new();
139    let handle = publish_image(
140        request.outfile,
141        request.overwrite,
142        is_external_open,
143        |path| {
144            let mut original = attached.lock();
145            if !request.outfile.is_empty() {
146                for source in original
147                    .path()
148                    .into_iter()
149                    .chain(attached.operand_paths().iter().map(PathBuf::as_path))
150                {
151                    ensure_outfile_does_not_alias_input(Path::new(request.outfile), source)?;
152                }
153            }
154            attached.require_pixel_read_lock(&mut original)?;
155            let mut record = original.coordinate_record();
156            let input_shape = original.shape().to_vec();
157            let coordinates = request.build_coordinates(&mut record, input_shape.len())?;
158            let mut image_info = image_info_record(&original).cloned().unwrap_or_default();
159            attached.beam_view().apply_to(&mut image_info);
160            let mut shape = input_shape.clone();
161            shape.resize(coordinates.n_pixel_axes(), 1);
162            let target = if request.outfile.is_empty() {
163                "(temp)image".to_owned()
164            } else {
165                format!("image '{}'", request.outfile)
166            };
167            logs.push(ToolLogLine::info(format!(
168                "Creating {target} of shape {shape:?}"
169            )));
170
171            let mut copy = match &*original {
172                ImageHandle::Float(source) => ImageHandle::Float(copy_storage(
173                    source,
174                    path,
175                    &input_shape,
176                    &shape,
177                    &coordinates,
178                    |src, start, lengths, stride| match original.fits() {
179                        Some(fits) => Ok(fits.read(start, lengths, stride)?),
180                        None => native_pixels(src, start, lengths, stride),
181                    },
182                )?),
183                ImageHandle::Double(source) => ImageHandle::Double(copy_storage(
184                    source,
185                    path,
186                    &input_shape,
187                    &shape,
188                    &coordinates,
189                    native_pixels,
190                )?),
191                ImageHandle::Complex(source) => ImageHandle::Complex(copy_storage(
192                    source,
193                    path,
194                    &input_shape,
195                    &shape,
196                    &coordinates,
197                    native_pixels,
198                )?),
199                ImageHandle::DComplex(source) => ImageHandle::DComplex(copy_storage(
200                    source,
201                    path,
202                    &input_shape,
203                    &shape,
204                    &coordinates,
205                    native_pixels,
206                )?),
207            };
208            copy.set_coordinate_record(record)?;
209            if !image_info.is_empty() {
210                copy.set_image_info_record(image_info)?;
211            }
212            if record_history {
213                copy.record_history("adddegaxes", request.history_command())?;
214            }
215            copy.save()?;
216            Ok(copy)
217        },
218    )?;
219    Ok((handle, logs))
220}

222fn native_pixels<T: ImagePixel>(
223    source: &dyn ImageInterface<T>,
224    start: &[usize],
225    lengths: &[usize],
226    stride: &[usize],
227) -> ImageToolResult<ArrayD<T>> {
228    Ok(source.get_slice(start, lengths, stride)?)
229}
230
231fn expand<T>(mut block: ArrayD<T>, ndim: usize) -> ArrayD<T> {
232    while block.ndim() < ndim {
233        let axis = block.ndim();
234        block = block.insert_axis(Axis(axis));
235    }
236    block
237}

239fn copy_storage<T: ImagePixel>(
240    source: &Storage<T>,
241    path: Option<&Path>,
242    input_shape: &[usize],
243    shape: &[usize],
244    coordinates: &CoordinateSystem,
245    read: impl Fn(&dyn ImageInterface<T>, &[usize], &[usize], &[usize]) -> ImageToolResult<ArrayD<T>>,
246) -> ImageToolResult<Storage<T>> {
247    if let Storage::Paged(image) = source
248        && let Some(groups) = image.table().keywords().get("ATTRGROUPS")
249        && !matches!(groups, Value::Record(groups) if groups.is_empty())
250    {
251        return Err(ImageToolError::not_implemented(
252            "image",
253            "adddegaxes with attribute groups",
254            1625,
255        ));
256    }
257    let mut output = Storage::create(shape.to_vec(), coordinates.clone(), path, None)?;
258    with_storage!(source, original => {
259        copy_pixels(original.as_ref(), input_shape, &mut output, shape.len(), read)?;
260        with_storage!(&mut output, image => {
261            image.set_units(original.units())?;
262            image.set_misc_info(original.misc_info())?;
263            Ok::<_, ImageToolError>(())
264        })?;
265        Ok::<_, ImageToolError>(())
266    })?;
267    let rows = match source {
268        Storage::Paged(image) => image_history_rows(image.name().expect("persistent image"))?,
269        Storage::Temp { image, .. } => image.history_rows()?.into_owned(),
270    };
271    with_storage!(&mut output, image => image.append_history_entries(&rows))?;
272    Ok(output)
273}

275fn copy_pixels<T: ImagePixel>(
276    source: &dyn ImageInterface<T>,
277    input_shape: &[usize],
278    output: &mut Storage<T>,
279    output_ndim: usize,
280    read: impl Fn(&dyn ImageInterface<T>, &[usize], &[usize], &[usize]) -> ImageToolResult<ArrayD<T>>,
281) -> ImageToolResult<()> {
282    let mut capacity = COPY_BLOCK_PIXELS;
283    let chunk_shape: Vec<_> = input_shape
284        .iter()
285        .map(|length| {
286            let extent = capacity.min(*length).max(1);
287            capacity /= extent;
288            extent
289        })
290        .collect();
291    let names = source.mask_names();
292    for name in &names {
293        validate_mask_name(name)?;
294    }
295    let stride = vec![1; input_shape.len()];
296    // Walk the first pixel axis fastest, matching the stored array order.
297    let grid: Vec<_> = input_shape
298        .iter()
299        .zip(&chunk_shape)
300        .rev()
301        .map(|(&length, &extent)| length.div_ceil(extent))
302        .collect();
303    for index in indices(IxDyn(&grid)) {
304        let start: Vec<_> = index
305            .slice()
306            .iter()
307            .rev()
308            .zip(&chunk_shape)
309            .map(|(index, extent)| index * extent)
310            .collect();
311        let lengths: Vec<_> = input_shape
312            .iter()
313            .zip(&start)
314            .zip(&chunk_shape)
315            .map(|((&length, &start), &extent)| extent.min(length - start))
316            .collect();
317        let block = read(source, &start, &lengths, &stride)?;
318        let mut position = start.clone();
319        position.resize(output_ndim, 0);
320        let pixels = expand(block, output_ndim);
321        with_storage!(output, image => image.put_slice(&pixels, &position))?;
322        for name in &names {
323            let block = source.get_named_mask_slice(name, &start, &lengths, &stride)?;
324            let block = expand(block, output_ndim);
325            with_storage!(output, image => image.put_mask_slice(name, &block, &position))?;
326        }
327    }
328    if let Some(default) = source.default_mask_name() {
329        with_storage!(output, image => image.set_default_mask(&default))?;
330    }
331    Ok(())
332}
