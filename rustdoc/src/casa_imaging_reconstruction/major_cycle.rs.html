3//! The major cycle: one pass forms the normal state with the final model.
4//!
5//! A [`MajorCycle`] owns its final model and the normal state its pass forms
6//! with that model, and releases them together as one
7//! [`MajorCycleCompletion`]: the final normal state, with its model-dependent
8//! residual, and the final model generation.

21/// Heap bound for one domain's channel-local window and its backing-access
22/// scratch. Covers residual, invariant dirty, PSF, sensitivity, optional PB,
23/// exact channel metadata, and an overlapping scalar conversion/I/O pair.
24#[doc(hidden)]
25pub fn normal_state_window_residency_bytes(
26    shape: [usize; 2],
27    polarizations: usize,
28    window_channels: usize,
29) -> Result<u64, SpectralOperatorError> {
30    let overflow = || SpectralOperatorError::ResidencyOverflow;
31    let planes = window_channels
32        .checked_mul(polarizations)
33        .ok_or_else(overflow)?;
34    let cells = shape[0].checked_mul(shape[1]).ok_or_else(overflow)?;
35    let image_bytes = cells
36        .checked_mul(planes)
37        .and_then(|values| {
38            values.checked_mul(5 * size_of::<num_complex::Complex64>() + 2 * size_of::<f64>())
39        })
40        .ok_or_else(overflow)?;
41    let metadata = planes
42        .checked_mul(2 * size_of::<f64>() + size_of::<crate::SpectralChannelValidity>())
43        .and_then(|bytes| {
44            bytes.checked_add(size_of::<crate::spectral_operator::SpectralDomainPrimitives>())
45        })
46        .ok_or_else(overflow)?;
47    u64::try_from(image_bytes.checked_add(metadata).ok_or_else(overflow)?).map_err(|_| overflow())
48}

50/// Versioned Normal State Generation catalog minted by a Major Cycle.
51#[derive(Debug, Clone, Copy, PartialEq, Eq)]
52pub enum NormalStateCatalog {
53    /// Unnormalized single-field Stokes-I constant-basis MFS normal state
54    /// whose residual follows the exact paired `A* W (d - A x)` composition.
55    UnnormalizedPlaneV1,
56    /// Unnormalized channel-major normal-state slab whose planes follow the
57    /// exact output-channel interval named by the paired spectral operator.
58    UnnormalizedChannelSlabV1,
59    /// Unnormalized Taylor residual terms and `2T-1` signed block-normal moments.
60    UnnormalizedTaylorBlockV1,
61}

63/// The normal state a major cycle's pass formed with its final model.
64///
65/// It owns the model-dependent unnormalized residual with the PSF,
66/// sensitivity and sum weights, and the sample and block counts of the
67/// pass; it makes no promise that the state is fully resident, dense, or
68/// shift-invariant. It is not a Product Graph artifact.
69///
70/// ```compile_fail
71/// use casa_imaging_reconstruction::FinalNormalState;
72///
73/// let _ = FinalNormalState {};
74/// ```
75#[doc(hidden)]
76#[derive(Debug)]
77pub struct FinalNormalState {
78    catalog: NormalStateCatalog,
79    sample_count: u64,
80    block_count: u64,
81    primitives: NormalStatePrimitives,
82}

252    /// Load a channel-local window, or borrow the complete coupled coefficient
253    /// family. The returned guard owns every channel-local payload it exposes.
254    pub fn read_window(
255        &self,
256        channels: std::ops::Range<usize>,
257    ) -> Result<FinalNormalStateWindow<'_>, SpectralOperatorError> {
258        let primitives = self.primitives.read_window(channels)?;
259        Ok(FinalNormalStateWindow {
260            owner: self,
261            primitives,
262        })
263    }

265    /// Select one absolute channel and polarization without loading image data.
266    ///
267    /// Field reads are independently bounded to this plane and borrow this
268    /// completion owner. Channel-local and resident constant-basis state are
269    /// supported; multi-term Taylor families use their complete-family path.
270    pub fn read_plane(
271        &self,
272        domain_ordinal: usize,
273        absolute_channel: usize,
274        polarization: usize,
275    ) -> Result<crate::FinalNormalPlaneReader<'_>, SpectralOperatorError> {
276        self.primitives
277            .read_plane(domain_ordinal, absolute_channel, polarization)
278    }
279
280    /// Read the residual and point-spread function of one plane, the inputs
281    /// of that plane's minor cycle.
282    ///
283    /// `domain_ordinal` names the image domain, `absolute_channel` the
284    /// output channel (in the slab's absolute numbering) and `polarization`
285    /// the polarization plane. Both fields are unnormalized, in their stored
286    /// precision: a paged state reads only this plane, a resident state
287    /// borrows it. The plane also carries its shape, validity and sum of
288    /// weights.
289    ///
290    /// # Errors
291    ///
292    /// When the plane is outside the state or its storage cannot be read.
293    pub fn read_reconstruction_plane(
294        &self,
295        domain_ordinal: usize,
296        absolute_channel: usize,
297        polarization: usize,
298    ) -> Result<FinalNormalStatePlane<'_>, SpectralOperatorError> {
299        let plane = self.read_plane(domain_ordinal, absolute_channel, polarization)?;
300        let residual_real = plane.read_residual_real()?;
301        let psf_real = plane.read_psf_real()?;
302        Ok(FinalNormalStatePlane {
303            owner: self,
304            domain_ordinal,
305            output_channel: absolute_channel,
306            polarization,
307            shape: plane.shape(),
308            validity: plane.validity(),
309            sum_weight: plane.sum_weight(),
310            residual: if let Some(real) = residual_real {
311                crate::normal_values::NormalPlane::Real(real)
312            } else {
313                crate::normal_values::NormalPlane::Complex(plane.read_residual()?)
314            },
315            psf: if let Some(real) = psf_real {
316                crate::normal_values::NormalPlane::Real(real)
317            } else {
318                crate::normal_values::NormalPlane::Complex(plane.read_psf()?)
319            },
320        })
321    }

324/// Explicitly loaded Normal State data retaining its global completion owner.
325/// Channel-local windows do not mint local completion or mask identities.
326#[derive(Debug)]
327pub struct FinalNormalStateWindow<'a> {
328    owner: &'a FinalNormalState,
329    primitives: NormalStateWindowPayload<'a>,
330}
331
332impl std::ops::Deref for FinalNormalStateWindow<'_> {
333    type Target = FinalNormalState;
334    fn deref(&self) -> &Self::Target {
335        self.owner
336    }
337}
338
339impl FinalNormalStateWindow<'_> {
340    /// Global completion owner; channel windows have no separate authority.
341    #[must_use]
342    pub fn owner(&self) -> &FinalNormalState {
343        self.owner
344    }
345
346    /// The exact loaded channel interval, not the global coverage interval.
347    #[must_use]
348    pub fn slab(&self) -> crate::SpectralSlabPlan {
349        self.primitives.slab()
350    }
351
352    /// Number of loaded channels. Coupled state retains its complete basis.
353    #[must_use]
354    pub fn channel_count(&self) -> usize {
355        self.slab().core_depth()
356    }
357
358    /// Image-sized heap capacity owned by this guard.
359    #[must_use]
360    pub fn owned_bytes(&self) -> usize {
361        match &self.primitives {
362            NormalStateWindowPayload::ChannelLocal(domains) => domains.owned_bytes(),
363            NormalStateWindowPayload::Coupled(_) => 0,
364        }
365    }
366
367    /// Borrow one canonical image-domain view by ordinal.
368    #[must_use]
369    pub fn domain(&self, ordinal: usize) -> Option<FinalNormalDomainState<'_>> {
370        self.primitives
371            .get(ordinal)
372            .map(|domain| FinalNormalDomainState {
373                owner: self.owner,
374                domain,
375            })
376    }
377
378    /// Iterate image-domain normal states in compiled geometry order.
379    pub fn domains(&self) -> impl ExactSizeIterator<Item = FinalNormalDomainState<'_>> {
380        self.primitives.iter().map(|domain| FinalNormalDomainState {
381            owner: self.owner,
382            domain,
383        })
384    }
385
386    /// Borrow the domain with this compiler-owned role.
387    #[must_use]
388    pub fn domain_by_role(&self, role: &ImageDomainRole) -> Option<FinalNormalDomainState<'_>> {
389        self.domains().find(|domain| domain.role() == role)
390    }
391
392    /// Return the authoritative model-dependent residual plane.
393    #[must_use]
394    pub fn residual(&self) -> crate::NormalValues<'_> {
395        self.primitives.dirty()
396    }
397
398    /// Return the normal approximation (the PSF) paired with the residual.
399    #[must_use]
400    pub fn normal_approximation(&self) -> crate::NormalValues<'_> {
401        self.primitives.psf()
402    }
403
404    /// Return sensitivity state in unnormalized normal-state units.
405    #[must_use]
406    pub fn sensitivity(&self) -> crate::SensitivityValues<'_> {
407        self.primitives.sensitivity()
408    }
409
410    /// Borrow one reconstruction-coefficient residual term.
411    #[must_use]
412    pub fn coefficient_term(
413        &self,
414        coefficient: usize,
415    ) -> Option<FinalNormalStateCoefficientTerm<'_>> {
416        if !matches!(self.catalog, NormalStateCatalog::UnnormalizedTaylorBlockV1) {
417            return None;
418        }
419        let cells = self.shape()[0].checked_mul(self.shape()[1])?;
420        let start = coefficient.checked_mul(cells)?;
421        let end = start.checked_add(cells)?;
422        Some(FinalNormalStateCoefficientTerm {
423            owner: self.owner,
424            coefficient,
425            residual: self.primitives.dirty().complex()?.get(start..end)?,
426        })
427    }
428
429    /// Borrow one retained normal moment.
430    #[must_use]
431    pub fn normal_moment(&self, moment: usize) -> Option<FinalNormalStateNormalMoment<'_>> {
432        if !matches!(self.catalog, NormalStateCatalog::UnnormalizedTaylorBlockV1) {
433            return None;
434        }
435        let cells = self.shape()[0].checked_mul(self.shape()[1])?;
436        let start = moment.checked_mul(cells)?;
437        let end = start.checked_add(cells)?;
438        Some(FinalNormalStateNormalMoment {
439            owner: self.owner,
440            moment,
441            normal_approximation: self.primitives.psf().complex()?.get(start..end)?,
442            sensitivity: self.primitives.sensitivity().dense()?.get(start..end)?,
443            sum_weight: *self.primitives.sum_weights().get(moment)?,
444        })
445    }
446
447    /// Borrow the Hankel normal block `H[row,column] = P[row+column]`.
448    #[must_use]
449    pub fn normal_block(
450        &self,
451        row: usize,
452        column: usize,
453    ) -> Option<FinalNormalStateNormalMoment<'_>> {
454        let moment = self.primitives.normal_moment_index(row, column)?;
455        self.normal_moment(moment)
456    }
457
458    /// Borrow one channel/polarization plane from this bounded Normal State slab.
459    #[must_use]
460    pub fn polarization_plane(
461        &self,
462        local_channel: usize,
463        polarization: usize,
464    ) -> Option<FinalNormalStatePlane<'_>> {
465        if matches!(self.catalog, NormalStateCatalog::UnnormalizedTaylorBlockV1) {
466            return None;
467        }
468        let cells = self.shape()[0].checked_mul(self.shape()[1])?;
469        let plane = local_channel
470            .checked_mul(self.primitives.polarization_count())?
471            .checked_add(polarization)?;
472        let start = plane.checked_mul(cells)?;
473        let end = start.checked_add(cells)?;
474        if local_channel >= self.channel_count()
475            || polarization >= self.primitives.polarization_count()
476        {
477            return None;
478        }
479        Some(FinalNormalStatePlane {
480            owner: self.owner,
481            domain_ordinal: self.primitives.get(0)?.domain_ordinal(),
482            output_channel: self.primitives.slab().core_range().start + local_channel,
483            polarization,
484            shape: self.shape(),
485            validity: *self.primitives.channel_validity().get(plane)?,
486            sum_weight: *self.primitives.sum_weights().get(plane)?,
487            residual: crate::normal_values::NormalPlane::borrowed(
488                self.primitives.dirty().slice(start..end)?,
489            ),
490            psf: crate::normal_values::NormalPlane::borrowed(
491                self.primitives.psf().slice(start..end)?,
492            ),
493        })
494    }
495}

750/// One major cycle's result: the final normal state and the final model its
751/// pass formed the residual with, released together.
752///
753/// A caller cannot assemble one from parts:
754///
755/// ```compile_fail
756/// use casa_imaging_reconstruction::MajorCycleCompletion;
757///
758/// let _ = MajorCycleCompletion {};
759/// ```
760#[derive(Debug)]
761pub struct MajorCycleCompletion {
762    normal_state: FinalNormalState,
763    final_model: ModelGeneration,
764}
765
766impl MajorCycleCompletion {
767    /// Borrow the final normal state.
768    #[must_use]
769    pub const fn normal_state(&self) -> &FinalNormalState {
770        &self.normal_state
771    }
772
773    /// Borrow the final model generation.
774    #[must_use]
775    pub const fn final_model(&self) -> &ModelGeneration {
776        &self.final_model
777    }
778
779    /// Release the normal state and the final model, which the next major
780    /// cycle refreshes and updates.
781    #[must_use]
782    pub fn into_parts(self) -> (FinalNormalState, ModelGeneration) {
783        (self.normal_state, self.final_model)
784    }
785}
786
787/// One major cycle: its final model and the normal state its pass forms
788/// with that model.
789///
790/// Owning both pairs the residual with the model that formed it: the pass
791/// reads the model and appends to the state through [`Self::parts`], and only
792/// [`Self::finish`] releases them, together. The pass cannot take the state
793/// out of its cycle, for example to exchange two cycles' states:
794///
795/// ```compile_fail,E0308
796/// use casa_imaging_reconstruction::MajorCycle;
797///
798/// fn exchange_states(first: &mut MajorCycle, second: &mut MajorCycle) {
799///     std::mem::swap(first.parts().1, second.parts().1);
800/// }
801/// ```
802pub struct MajorCycle {
803    model: PreparedFinalModel,
804    state: PassNormalState,
805}
806
807/// The pass's access to its [`MajorCycle`]'s normal state: it appends
808/// images and nothing else, so the state stays in its cycle, paired with
809/// the model that formed it.
810pub struct PassAppender<'a> {
811    state: &'a mut PassNormalState,
812}
813
814impl PassAppender<'_> {
815    /// Add one pass's images of one domain.
816    ///
817    /// A non-finite image value or `sumwt` is a generated non-finite value,
818    /// rejected under every [`casa_imaging_model::FiniteValuePolicy`]: the
819    /// policies differ only on non-finite inputs, which the source flags or
820    /// rejects before gridding. Nothing is changed when an append fails.
821    ///
822    /// # Errors
823    ///
824    /// Images whose domain, shape, polarizations, channels or planes do not
825    /// match the problem and the pass (PSF moments on a residual refresh, a
826    /// channel range out of order), and generated non-finite values.
827    pub fn append(&mut self, images: PassImages) -> Result<(), SpectralOperatorError> {
828        self.state.append(images)
829    }
830}
831
832impl MajorCycle {
833    /// The initial major cycle of `problem`: its pass forms the data and PSF
834    /// images with `model`, writing them as `storage` plans.
835    pub fn initial(
836        problem: &CompiledProblem,
837        model: PreparedFinalModel,
838        storage: NormalStoragePlan,
839    ) -> Result<Self, MajorCycleError> {
840        Ok(Self {
841            state: PassNormalState::initial(problem, storage).map_err(MajorCycleError::Residual)?,
842            model,
843        })
844    }
845
846    /// A residual refresh of `previous` with `model`: the PSF, `sumwt` and
847    /// validity carry over and every residual plane is formed again.
848    pub fn refresh(
849        problem: &CompiledProblem,
850        previous: FinalNormalState,
851        model: PreparedFinalModel,
852        storage: NormalStoragePlan,
853    ) -> Result<Self, MajorCycleError> {
854        Ok(Self {
855            state: PassNormalState::refresh(problem, previous, storage)
856                .map_err(MajorCycleError::Residual)?,
857            model,
858        })
859    }
860
861    /// The final model the pass predicts and the appender its images go to.
862    pub fn parts(&mut self) -> (&ModelGeneration, PassAppender<'_>) {
863        (
864            self.model.generation(),
865            PassAppender {
866                state: &mut self.state,
867            },
868        )
869    }
870
871    /// Finish the pass, which placed `samples` from `blocks` source blocks,
872    /// and release the final normal state with the final model.
873    ///
874    /// # Errors
875    ///
876    /// A pass that placed no sample, left an image incomplete or refreshed
877    /// fewer samples than the state it refreshed, a failed model update and
878    /// generated non-finite residuals all fail with no partial result.
879    pub fn finish(
880        self,
881        samples: u64,
882        blocks: u64,
883    ) -> Result<MajorCycleCompletion, MajorCycleError> {
884        if samples == 0 || blocks == 0 {
885            return Err(MajorCycleError::IncompleteCoverage);
886        }
887        let CompleteDataNormalState {
888            primitives,
889            completion,
890        } = self
891            .state
892            .finish(samples, blocks)
893            .map_err(MajorCycleError::Residual)?;
894        let final_model = self.model.complete()?;
895        Ok(MajorCycleCompletion {
896            normal_state: FinalNormalState {
897                catalog: match completion.primitive_catalog() {
898                    SpectralPrimitiveCatalog::UnnormalizedPlaneV1 => {
899                        NormalStateCatalog::UnnormalizedPlaneV1
900                    }
901                    SpectralPrimitiveCatalog::UnnormalizedChannelSlabV1 => {
902                        NormalStateCatalog::UnnormalizedChannelSlabV1
903                    }
904                    SpectralPrimitiveCatalog::UnnormalizedTaylorBlockV1 => {
905                        NormalStateCatalog::UnnormalizedTaylorBlockV1
906                    }
907                },
908                sample_count: completion.sample_count(),
909                block_count: completion.block_count(),
910                primitives,
911            },
912            final_model,
913        })
914    }
915}
916
917/// Exact reason a Major-Cycle reconciliation failed closed.
918#[derive(Debug, Clone, PartialEq, Eq)]
919pub enum MajorCycleError {
920    /// The pass placed no sample or read no source block.
921    IncompleteCoverage,
922    /// The final model's queued update could not be completed.
923    Model(ModelLifecycleError),
924    /// Reconciling the final model produced or consumed invalid numbers.
925    Residual(SpectralOperatorError),
926}
927
928impl fmt::Display for MajorCycleError {
929    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
930        match self {
931            Self::IncompleteCoverage => {
932                formatter.write_str("the major-cycle pass placed no sample")
933            }
934            Self::Model(error) => error.fmt(formatter),
935            Self::Residual(error) => error.fmt(formatter),
936        }
937    }
938}

940impl std::error::Error for MajorCycleError {}
941
942impl From<ModelLifecycleError> for MajorCycleError {
943    fn from(error: ModelLifecycleError) -> Self {
944        Self::Model(error)
945    }
