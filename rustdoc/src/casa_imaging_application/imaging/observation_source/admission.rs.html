2//! The batch route's memory: how many times each partition's batches hold,
3//! sized as the selected-observation route sizes its blocks
4//! ([`casa_imaging_runtime::finalize_source_access`]), and the charge of
5//! everything the source and the passes' blocks hold, admitted before the
6//! run opens the source.
7
8use casa_imaging_model::{CompiledProblem, GeometryPlan};
9use casa_imaging_runtime::pass::{

12use casa_imaging_runtime::{
13    Admitted, Demand, HostResources, ResourcePolicy, admit, bootstrap_source_budget, free_memory,
14    row_layout,
15};
16use casa_observation::{AccessRequest, Observation, PartitionEstimate};

23/// An upper bound of the bytes the survey's conversion table holds per
24/// (time, field) it converts at: its key, its factor and the table's share.
25const SURVEY_ENTRY_BYTES: usize = 64;
26
27/// The bytes a survey batch holds beyond one time, when one time holds
28/// less: the survey reads no payload, so its batches stay small.
29const SURVEY_BATCH_BYTES: usize = 1 << 18;
30
31/// How the batch route's traversals divide each partition: whole times
32/// per batch, and the most samples one batch holds.
33#[derive(Clone, Debug, PartialEq, Eq)]
34pub(crate) struct Batching {
35    times_per_batch: Vec<usize>,
36    rows: usize,
37    survey_bytes: usize,
38}
39
40impl Batching {
41    /// Whole times each batch of selected partition `index` holds.
42    pub(super) fn times_per_batch(&self, index: usize) -> usize {
43        self.times_per_batch[index]
44    }
45
46    /// The most samples (padding included) one batch holds: the passes'
47    /// block rows.
48    pub(super) fn rows(&self) -> usize {
49        self.rows
50    }
51
52    /// The most bytes one batch of the payload-free survey and its decode
53    /// scratch hold, in buffers of the survey's own: at least one time of
54    /// every partition.
55    pub(super) fn survey_bytes(&self) -> usize {
56        self.survey_bytes
57    }
58}
59
60/// One selected partition's traversal of its whole selection, from its
61/// estimate.
62struct PartitionCost {
63    /// Bytes one time holds in a batch and the cursor's scratch.
64    held_per_time: usize,
65    /// Bytes one time of its survey holds.
66    survey_per_time: usize,
67    baselines: usize,
68    /// Native channels a batch delivers: its selected channels' span.
69    channels: usize,
70    /// Times the traversal delivers.
71    times: usize,
72}
73
74/// `observation`'s estimate of `request`'s one partition.
75fn estimate(
76    observation: &dyn Observation,
77    request: &AccessRequest,
78) -> Result<PartitionEstimate, ImagingError> {
79    observation
80        .estimate(request)
81        .map_err(|error| ImagingError::Observation(Box::new(error)))?
82        .partitions
83        .into_iter()
84        .next()
85        .ok_or(ImagingError::Unsupported {
86            reason: "an estimate without its partition",
87        })
88}
89
90/// The batches of `selection` for `problem`, and the admitted charge of
91/// what the source holds for them beside the passes' blocks.
92///
93/// As the selected-observation route does for its rows, the batches hold
94/// at least one time of each partition, and grow beyond one time by at most
95/// the bootstrap budget ([`bootstrap_source_budget`]) or the source's share
96/// of a quarter of what `policy` leaves free on `host`, shared with the
97/// passes' per-row storage in proportion to their bytes per sample. They
98/// hold at most one part in [`OVERLAPPED_BLOCKS`] of a partition's selected
99/// times, so a pass streams in several batches and reads the next while it
100/// grids one. A row of [`row_layout`] produces at most `row` samples. The charge holds the largest batch's
101/// buffers, its geometry, the plan's parallactic cache at every selected
102/// (time, field), the pointing answers the source keeps within
103/// `pointing_budget` bytes when the geometry reads pointing, the survey's
104/// buffers and conversions, and the passes' blocks; the run keeps it while
105/// the source is open. What opening the store holds is the caller's to
106/// charge.
107///
108/// # Errors
109///
110/// When an estimate fails, or the charge does not fit the policy.
111pub(crate) fn admit_batches(
112    (problem, observation, selection, plan): (
113        &CompiledProblem,
114        &dyn Observation,
115        &PartitionSelection,
116        &GeometryPlan,
117    ),
118    row: RowSamples,
119    pointing_budget: usize,
120    (host, policy): (&HostResources, &ResourcePolicy),
121) -> Result<Admitted<Batching>, ImagingError> {
122    let costs = selection
123        .partitions
124        .iter()
125        .map(|partition| {
126            let entry = observation.catalogue().partition(partition.id);
127            let first = partition.channels[0] as usize;
128            let last = *partition.channels.last().expect("a selected channel") as usize;
129            let payload = estimate(
130                observation,
131                &requests::payload(
132                    (selection, plan),
133                    entry,
134                    partition,
135                    first..last + 1,
136                    true,
137                    usize::MAX,
138                ),
139            )?;
140            let survey = estimate(
141                observation,
142                &requests::survey((selection, plan), entry, partition, usize::MAX),
143            )?;
144            Ok(PartitionCost {
145                held_per_time: payload.held_bytes_per_time(),
146                survey_per_time: survey.held_bytes_per_time(),
147                baselines: payload.baselines.max(1),
148                channels: last + 1 - first,
149                times: payload.times,
150            })
151        })
152        .collect::<Result<Vec<_>, ImagingError>>()?;
153    let layout = row_layout(problem);
154    // Per sample: the batch's share and the cursor's scratch, its
155    // geometry, and the flag that keeps it.
156    let source_per_row = costs
157        .iter()
158        .map(|cost| cost.held_per_time.div_ceil(cost.baselines))
159        .max()
160        .unwrap_or(0)
161        + plan.geometry_bytes(1, 0, 0)
162        + 1;
163    let passes = |rows: usize| source_block_bytes(BlockShape { rows, ..layout });
164    let workers = policy.cores(host).streaming_workers;
165    let passes_per_row = passes(1) - passes(0) + chunk_bytes_per_row(layout, row, workers);
166    let minimum_rows = costs.iter().map(|cost| cost.baselines).max().unwrap_or(1);
167    let minimum = minimum_rows as u64 * (source_per_row as u64 + passes_per_row);
168    let free = free_memory(host, policy);
169    let quarter = free.saturating_sub(minimum) / 4;
170    let share = u128::from(source_per_row as u64) * u128::from(quarter)
171        / u128::from((source_per_row as u64 + passes_per_row).max(1));
172    let growth = bootstrap_source_budget()
173        .available_bytes()
174        .min(usize::try_from(share).unwrap_or(usize::MAX));
175    let target = (minimum_rows * source_per_row + growth) / source_per_row.max(1);
176    // A batch holds at most one part in OVERLAPPED_BLOCKS of its
177    // partition's selected times, and at least one time.
178    let times_per_batch = costs
179        .iter()
180        .map(|cost| {
181            (target / cost.baselines)
182                .min(cost.times.div_ceil(OVERLAPPED_BLOCKS))
183                .max(1)
184        })
185        .collect::<Vec<_>>();
186    let rows = costs
187        .iter()
188        .zip(&times_per_batch)
189        .map(|(cost, times)| times * cost.baselines)
190        .max()
191        .unwrap_or(0);
192    let longest = times_per_batch.iter().copied().max().unwrap_or(0);
193    let widest = costs.iter().map(|cost| cost.channels).max().unwrap_or(0);
194    let buffers = costs
195        .iter()
196        .zip(&times_per_batch)
197        .map(|(cost, times)| times * cost.held_per_time)
198        .max()
199        .unwrap_or(0);
200    let survey_bytes = costs
201        .iter()
202        .map(|cost| cost.survey_per_time)
203        .max()
204        .unwrap_or(0)
205        .max(SURVEY_BATCH_BYTES);
206    let keys = costs.iter().map(|cost| cost.times).sum::<usize>();
207    let pointing = if plan.reads_pointing() {
208        pointing_budget
209    } else {
210        0
211    };
212    let source = buffers
213        + plan.geometry_bytes(rows, longest, widest)
214        + rows
215        + plan.cache_bound(keys)
216        + pointing
217        + survey_bytes
218        + keys * SURVEY_ENTRY_BYTES
219        + PackScratch::bound(layout);
220    tracing::info!(
221        rows_per_block = rows,
222        source_bytes = source,
223        pass_bytes = passes(rows),
224        pointing_bytes = pointing,
225        survey_bytes,
226        "observation-batch source plan"
227    );
228    let reservation = admit(
229        host,
230        policy,
231        &Demand {
232            phase: "observation-batch source",
233            memory: source as u64 + passes(rows),
234        },
235    )?;
236    Ok(Admitted {
237        value: Batching {
238            times_per_batch,
239            rows,
240            survey_bytes,
241        },
242        reservation,
243    })
244}
