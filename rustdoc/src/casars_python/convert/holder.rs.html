2//! casacore's scalar `MeasureHolder`: epoch, direction, position, frequency,
3//! Doppler and radial-velocity values
4//! with any of its kind's reference codes, read from and written to the
5//! measure records CASA's tools exchange.
6//!
7//! A [`Measure`] keeps what casacore's measure value keeps: the epoch's MJD,
8//! the direction's cosines, the position's Cartesian vector and the
9//! frequency in Hz. It therefore holds codes casa-types does not convert
10//! (`SUN`, `B1950_VLA`, `REST`): they define, show and compare, and only a
11//! conversion that needs them fails.
12//!
13//! - **Reading** ([`Measure::from_record`]) follows
14//!   `MeasureHolder::fromRecord` and the measure values' `putValue`: the
15//!   record's `type` names the kind, `refer` names a code by unique
16//!   case-insensitive prefix (an unknown one is the kind's default, with
17//!   casacore's warning), and `m0`, `m1`, `m2` are quantity records in any
18//!   unit their kind accepts.
19//! - **Writing** ([`Measure::to_value`]) follows `MeasureHolder::toRecord`:
20//!   `{'type', 'refer', 'm0', ...}` with the epoch in days, angles in
21//!   radians, the position's length in metres and the frequency in Hz.
22//! - **Positions** keep casacore's encoding of a WGS84 height as the length
23//!   of the position vector (`MVPosition`): a zero height is a 1e-6 m
24//!   vector, and heights between 0 and −7000 km, or near 743.568 m, are
25//!   shifted, so `me.position('WGS84', '10deg', '20deg', '0m')` has a
26//!   height of 1e-6 m, as in CASA.
27//!
28//! The measures tool's record envelope handles vectors and reference offsets,
29//! reusing [`Measure::record_definition`] to parse metadata and units once.
30//! This scalar holder is also used by task parameter conversion.
31

96/// Resolve the optional R_ epoch prefix without changing listcodes output.
97pub(crate) fn resolve_epoch_code(input: &str) -> Option<&'static str> {
98    if let Some(code) = input
99        .strip_prefix("R_")
100        .or_else(|| input.strip_prefix("r_"))
101    {
102        let canonical = EPOCH_CODES.resolve(code)?;
103        let reference = EpochRef::from_str(canonical).ok()?;
104        Some(RAZED_EPOCH_CODES[reference.casacore_code() as usize])
105    } else {
106        EPOCH_CODES.resolve(input)
107    }
108}

110/// Interpret a canonical epoch code stored in a measure record.
111pub(crate) fn epoch_ref(code: &str) -> Result<(EpochRef, bool), casa_core::measures::MeasureError> {
112    let base = code.strip_prefix("R_");
113    Ok((EpochRef::from_str(base.unwrap_or(code))?, base.is_some()))

187impl Codes {
188    /// The code `input` names by unique case-insensitive prefix
189    /// (`MUString::minimaxNC`), as the code it stands for; `None` if it names
190    /// none.
191    pub(crate) fn resolve(&self, input: &str) -> Option<&'static str> {
192        let code = self.all[unique_prefix_index(input, self.all)?];
193        Some(
194            self.aliases
195                .iter()
196                .find(|(alias, _)| *alias == code)
197                .map_or(code, |(_, canonical)| canonical),
198        )
199    }
200
201    /// The code `input` names, else the kind's default (`setRefString`).
202    pub(crate) fn resolve_or_default(&self, input: &str) -> &'static str {
203        self.resolve(input).unwrap_or(self.default)
204    }
205
206    /// The kind's default code.
207    pub(crate) fn default_code(&self) -> &'static str {
208        self.default
209    }
210}

443/// Parsed reference and quantities, shared by scalar and vector records.
444pub(crate) struct MeasureDefinition {

447    pub(crate) quantities: Vec<Quantity>,
448    pub(crate) warning: Option<String>,

451impl Measure {
452    /// The measure's kind.
453    pub(crate) fn kind(&self) -> MeasureKind {
454        match self.value {
455            MeasureValue::Epoch(_) => MeasureKind::Epoch,
456            MeasureValue::Direction(_) => MeasureKind::Direction,
457            MeasureValue::Position(_) => MeasureKind::Position,
458            MeasureValue::Frequency(_) => MeasureKind::Frequency,
459            MeasureValue::Doppler(_) => MeasureKind::Doppler,
460            MeasureValue::RadialVelocity(_) => MeasureKind::RadialVelocity,
461            MeasureValue::Baseline(_) => MeasureKind::Baseline,
462            MeasureValue::Uvw(_) => MeasureKind::Uvw,
463            MeasureValue::EarthMagnetic(_) => MeasureKind::EarthMagnetic,
464        }
465    }
466
467    /// `MeasureHolder::fromRecord`.
468    pub(crate) fn from_record(
469        record: &BTreeMap<String, Variant>,
470    ) -> Result<ReadMeasure, HolderError> {
471        let definition = Self::record_definition(record)?;
472        Ok(ReadMeasure {
473            measure: Self::from_values(definition.kind, definition.refer, &definition.quantities)?,
474            warning: definition.warning,
475        })
476    }
477
478    /// Parse the record's kind, reference and units once, before evaluating values.
479    pub(crate) fn record_definition(
480        record: &BTreeMap<String, Variant>,
481    ) -> Result<MeasureDefinition, HolderError> {
482        let (Some(Variant::Text(kind)), Some(Variant::Text(refer))) =
483            (record.get("type"), record.get("refer"))
484        else {
485            return Err(HolderError::NotAMeasure);
486        };
487        let kind = match kind.to_ascii_lowercase().as_str() {
488            "epoch" => MeasureKind::Epoch,
489            "direction" => MeasureKind::Direction,
490            "position" => MeasureKind::Position,
491            "frequency" => MeasureKind::Frequency,
492            "doppler" => MeasureKind::Doppler,
493            "radialvelocity" => MeasureKind::RadialVelocity,
494            "baseline" => MeasureKind::Baseline,
495            "uvw" => MeasureKind::Uvw,
496            "earthmagnetic" => MeasureKind::EarthMagnetic,
497            _ => {
498                return Err(HolderError::Invalid(format!(
499                    "{kind} is an unknown measure typeUnknown Measure record in \
500                     MeasureHolder::fromRecord\n"
501                )));
502            }
503        };
504        let codes = kind.codes();
505        let (code, warning) = match kind.resolve_code(refer) {
506            Some(code) => (code, None),
507            None => (
508                codes.default,
509                (!refer.is_empty()).then(|| {
510                    format!(
511                        "Illegal or unknown reference type '{refer}' for {} definition. \
512                         DEFAULT ({}) assumed.",
513                        kind.record_type(),
514                        codes.default
515                    )
516                }),
517            ),
518        };
519        if record.contains_key("offset") {
520            return Err(HolderError::Invalid(
521                "measure offsets are not yet implemented in casa-rs".to_string(),
522            ));
523        }
524        Ok(MeasureDefinition {
525            kind,
526            refer: code,
527            quantities: record_values(record)?,
528            warning,
529        })
530    }

532    /// The measure from quantities, as the kind's `putValue` reads them.
533    pub(crate) fn from_values(
534        kind: MeasureKind,
535        refer: &'static str,
536        values: &[Quantity],
537    ) -> Result<Self, HolderError> {
538        let value = match kind {
539            MeasureKind::Baseline => MeasureValue::Baseline(spatial_value(values, false)?),
540            MeasureKind::Uvw => MeasureValue::Uvw(spatial_value(values, false)?),
541            MeasureKind::EarthMagnetic => MeasureValue::EarthMagnetic(spatial_value(values, true)?),
542            MeasureKind::Epoch => epoch_value(values)?,
543            MeasureKind::Direction => direction_value(values)?,
544            MeasureKind::Position => position_value(values)?,
545            MeasureKind::Frequency => frequency_value(values)?,
546            MeasureKind::Doppler => MeasureValue::Doppler(velocity_value(values, true)?),
547            MeasureKind::RadialVelocity => {
548                MeasureValue::RadialVelocity(velocity_value(values, false)?)
549            }
550        };
551        Ok(Self { refer, value })
552    }

554    /// The ordered numeric components of `MeasureHolder::toRecord`.
555    pub(crate) fn quantity_values(self) -> [f64; 3] {
556        match self.value {
557            MeasureValue::Epoch(mjd) => [mjd.as_mjd(), 0.0, 0.0],
558            MeasureValue::Direction(cosines) => {
559                let (lon, lat) = direction_angles(cosines);
560                [lon, lat, 0.0]
561            }
562            MeasureValue::Position(xyz) => {
563                let (lon, lat, length) = position_spherical(xyz);
564                [lon, lat, length]
565            }
566            MeasureValue::Baseline(xyz) | MeasureValue::Uvw(xyz) => {
567                let (lon, lat, length) = spatial_spherical(xyz);
568                [lon, lat, length]
569            }
570            MeasureValue::Frequency(hz) => [hz, 0.0, 0.0],
571            MeasureValue::Doppler(value) => [value * C_MPS, 0.0, 0.0],
572            MeasureValue::RadialVelocity(value) => [value, 0.0, 0.0],
573            MeasureValue::EarthMagnetic(xyz) => xyz,
574        }
575    }

577    /// Units of the ordered record components.
578    pub(crate) fn quantity_units(self) -> &'static [&'static str] {
579        match self.kind() {
580            MeasureKind::Epoch => &["d"],
581            MeasureKind::Direction => &["rad", "rad"],
582            MeasureKind::Position | MeasureKind::Baseline | MeasureKind::Uvw => {
583                &["rad", "rad", "m"]
584            }
585            MeasureKind::EarthMagnetic => &["nT", "nT", "nT"],
586            MeasureKind::Frequency => &["Hz"],
587            MeasureKind::Doppler | MeasureKind::RadialVelocity => &["m/s"],
588        }
589    }

591    /// `MeasureHolder::toRecord`.
592    pub(crate) fn to_value(self) -> CasaValue {
593        let values = self.quantity_values();
594        let mut fields = vec![
595            (
596                "type".to_owned(),
597                CasaValue::Text(self.kind().record_type().to_owned()),
598            ),
599            ("refer".to_owned(), CasaValue::Text(self.refer.to_owned())),
600        ];
601        for (i, &unit) in self.quantity_units().iter().enumerate() {
602            fields.push((format!("m{i}"), CasaValue::quantity(values[i], unit)));
603        }
604        CasaValue::record(fields)
605    }

607    /// The measure in casa-types' form, failing for a code casa-types does
608    /// not convert.
609    pub(crate) fn to_casa(self) -> Result<CasaMeasure, String> {
610        let unsupported = |kind: &str| {
611            format!(
612                "the {} {kind} frame is not yet implemented in casa-rs",
613                self.refer
614            )
615        };
616        Ok(match self.value {
617            MeasureValue::Baseline(_) | MeasureValue::Uvw(_) | MeasureValue::EarthMagnetic(_) => {
618                return Err(
619                    "geometry values use the native geometry conversion operator (#1252)".into(),
620                );
621            }
622            MeasureValue::Epoch(mjd) => {
623                let (refer, raze) = epoch_ref(self.refer).map_err(|_| unsupported("epoch"))?;
624                CasaMeasure::Epoch(MEpoch::new(mjd, refer).with_razing(raze))
625            }
626            MeasureValue::Direction(cosines) => CasaMeasure::Direction(MDirection::from_cosines(
627                cosines,
628                DirectionRef::from_str(self.refer).map_err(|_| unsupported("direction"))?,
629            )),
630            MeasureValue::Position(xyz) => CasaMeasure::Position(if self.refer == "WGS84" {
631                let (longitude, latitude, height) = position_spherical(xyz);
632                MPosition::new_wgs84(longitude, latitude, height)
633            } else {
634                MPosition::new_itrf(xyz[0], xyz[1], xyz[2])
635            }),
636            MeasureValue::Frequency(hz) => CasaMeasure::Frequency(MFrequency::new(
637                hz,
638                FrequencyRef::from_str(self.refer).map_err(|_| unsupported("frequency"))?,
639            )),
640            MeasureValue::Doppler(value) => {
641                CasaMeasure::Doppler(MDoppler::new(value, doppler_ref(self.refer)?))
642            }
643            MeasureValue::RadialVelocity(value) => {
644                CasaMeasure::RadialVelocity(MRadialVelocity::new(
645                    value,
646                    self.refer
647                        .parse()
648                        .map_err(|_| unsupported("radialvelocity"))?,
649                ))
650            }
651        })
652    }
653
654    /// A casa-types measure in casacore's form.
655    pub(crate) fn from_casa(measure: &CasaMeasure) -> Self {
656        match measure {
657            CasaMeasure::Epoch(epoch) => Self {
658                refer: if epoch.is_razing() {
659                    RAZED_EPOCH_CODES[epoch.refer().casacore_code() as usize]
660                } else {
661                    epoch.refer().casacore_name()
662                },
663                value: MeasureValue::Epoch(epoch.value()),
664            },
665            CasaMeasure::Direction(direction) => Self {
666                refer: DIRECTION_CODES.resolve_or_default(direction.refer().as_str()),
667                value: MeasureValue::Direction(direction.cosines()),
668            },
669            CasaMeasure::Position(position) => match position.refer() {
670                PositionRef::ITRF => Self {
671                    refer: "ITRF",
672                    value: MeasureValue::Position(position.values()),
673                },
674                PositionRef::WGS84 => {
675                    let [longitude, latitude, height] = position.values();
676                    Self {
677                        refer: "WGS84",
678                        value: MeasureValue::Position(position_vector(longitude, latitude, height)),
679                    }
680                }
681            },
682            CasaMeasure::Frequency(frequency) => Self {
683                refer: FREQUENCY_CODES.resolve_or_default(frequency.refer().as_str()),
684                value: MeasureValue::Frequency(frequency.hz()),
685            },
686            CasaMeasure::Doppler(value) => Self {
687                refer: DOPPLER_CODES.resolve_or_default(value.refer().as_str()),
688                value: MeasureValue::Doppler(value.value()),
689            },
690            CasaMeasure::RadialVelocity(value) => Self {
691                refer: RADIAL_VELOCITY_CODES.resolve_or_default(value.refer().as_str()),
692                value: MeasureValue::RadialVelocity(value.ms()),
693            },
694        }
695    }
696}

698// MVBaseline/MVuvw putValue, and coherent magnetic-field record quantities.
699fn spatial_value(values: &[Quantity], magnetic: bool) -> Result<[f64; 3], HolderError> {
700    let [a, b, c] = values else {
701        return Err(illegal_quantity());
702    };
703    let magnitude = if magnetic {
704        casa_core::quanta::Unit::new("T").expect("tesla").val().dim
705    } else {
706        UnitDim::basis(Dimension::Length)
707    };
708    let angle = UnitDim::basis(Dimension::Angle);
709    let is_magnitude = |q: &Quantity| q.unit().val().dim == magnitude;
710    let is_angle = |q: &Quantity| q.unit().val().dim == angle;
711    let scale = if magnetic { 1e9 } else { 1.0 };
712    if values.iter().all(is_magnitude) {
713        return Ok([
714            a.get_si_value() * scale,
715            b.get_si_value() * scale,
716            c.get_si_value() * scale,
717        ]);
718    }
719    let (length, lon, lat) = if is_magnitude(a) && is_angle(b) && is_angle(c) {
720        (a, b, c)
721    } else if is_angle(a) && is_angle(b) && is_magnitude(c) {
722        (c, a, b)
723    } else {
724        return Err(illegal_quantity());
725    };
726    let length = length.get_si_value() * scale;
727    let length = if length == 0.0 {
728        if magnetic { 1e-6 } else { 1e-12 }
729    } else {
730        length
731    };
732    let unit = direction_cosines(lon.get_si_value(), lat.get_si_value());
733    Ok(unit.map(|x| x * length))
734}

736fn velocity_value(values: &[Quantity], doppler: bool) -> Result<f64, HolderError> {
737    let value = match values {
738        [] => return Ok(0.0),
739        [value] => value,
740        _ => return Err(illegal_quantity()),
741    };
742    if doppler && value.unit().val().dim.is_dimensionless() {
743        return Ok(value.get_si_value());
744    }
745    let speed = UnitDim::basis(Dimension::Length) / UnitDim::basis(Dimension::Time);
746    if value.unit().val().dim != speed {
747        return Err(illegal_quantity());
748    }
749    Ok(value.get_si_value() / if doppler { C_MPS } else { 1.0 })
750}

752/// The `m0`, `m1`, `m2` quantities of a measure record, up to the first
753/// missing one. A quantity record may hold one value, or an array of one.
754fn record_values(record: &BTreeMap<String, Variant>) -> Result<Vec<Quantity>, HolderError> {
755    let mut values = Vec::new();
756    for key in ["m0", "m1", "m2"] {
757        let Some(Variant::Record(value)) = record.get(key) else {
758            break;
759        };
760        if let Some(Variant::Numbers(numbers)) = value.get("value")
761            && numbers.len() > 1
762        {
763            return Err(HolderError::Invalid(
764                "multi-valued measures are not yet implemented in casa-rs".to_string(),
765            ));
766        }
767        values.push(quantity_record(value).map_err(HolderError::Invalid)?);
768    }
769    Ok(values)
770}

772fn has(quantity: &Quantity, dimension: Dimension) -> bool {
773    quantity.unit().val().dim == UnitDim::basis(dimension)
774}

776/// `MVEpoch::putValue`: every value a time, summed.
777fn epoch_value(values: &[Quantity]) -> Result<MeasureValue, HolderError> {
778    if !values.iter().all(|value| has(value, Dimension::Time)) {
779        return Err(illegal_quantity());
780    }
781    let (mut day, mut fraction) = (0.0, 0.0);
782    for value in values {
783        let days = days_of(value);
784        let whole = days.floor();
785        day += whole;
786        fraction += days - whole;
787    }
788    Ok(MeasureValue::Epoch(MjdHighPrec::new(day, fraction)))
789}

791/// A time in days as casacore converts it (`Quantity::get("d")`).
792pub(crate) fn days_of(time: &Quantity) -> f64 {
793    time.value() * (time.unit().val().factor / 86_400.0)
794}

796/// `MVDirection::putValue`: three direction cosines, or up to two angles
797/// (none is the pole, one a longitude on the equator); normalised.
798fn direction_value(values: &[Quantity]) -> Result<MeasureValue, HolderError> {
799    let mut xyz = if values.len() == 3
800        && values
801            .iter()
802            .all(|value| value.unit().val().dim.is_dimensionless())
803    {
804        [values[0].value(), values[1].value(), values[2].value()]
805    } else {
806        if values.len() > 3 || !values.iter().all(|value| has(value, Dimension::Angle)) {
807            return Err(illegal_quantity());
808        }
809        let angles = values
810            .iter()
811            .map(Quantity::get_si_value)
812            .collect::<Vec<_>>();
813        match angles.as_slice() {
814            [] => [0.0, 0.0, 1.0],
815            [longitude] => [longitude.cos(), longitude.sin(), 0.0],
816            [longitude, latitude, ..] => direction_cosines(*longitude, *latitude),
817        }
818    };
819    let length = norm(xyz);
820    if length != 0.0 && length != 1.0 {
821        xyz = xyz.map(|value| value / length);
822    }
823    Ok(MeasureValue::Direction(xyz))
824}

826/// `MVPosition::putValue`: three lengths (x, y, z), a length then two
827/// angles, or two angles then a length.
828fn position_value(values: &[Quantity]) -> Result<MeasureValue, HolderError> {
829    let [first, second, third] = values else {
830        return Err(illegal_quantity());
831    };
832    let (length, angle) = (Dimension::Length, Dimension::Angle);
833    let xyz = if has(first, length) {
834        if has(second, length) && has(third, length) {
835            [
836                first.get_si_value(),
837                second.get_si_value(),
838                third.get_si_value(),
839            ]
840        } else if has(second, angle) && has(third, angle) {
841            position_vector(
842                second.get_si_value(),
843                third.get_si_value(),
844                first.get_si_value(),
845            )
846        } else {
847            return Err(illegal_quantity());
848        }
849    } else if has(third, length) && has(first, angle) && has(second, angle) {
850        position_vector(
851            first.get_si_value(),
852            second.get_si_value(),
853            third.get_si_value(),
854        )
855    } else {
856        return Err(illegal_quantity());
857    };
858    Ok(MeasureValue::Position(xyz))
859}

861/// `MVFrequency::putValue`: none is 0 Hz; one value is a frequency, a
862/// period, an angular rate, a wavelength, a wavenumber, an energy or a
863/// momentum (`MVFrequency::makeF`).
864fn frequency_value(values: &[Quantity]) -> Result<MeasureValue, HolderError> {

876fn norm(xyz: [f64; 3]) -> f64 {
877    (xyz[0] * xyz[0] + xyz[1] * xyz[1] + xyz[2] * xyz[2]).sqrt()
878}

880/// The cosines of a direction (`MVDirection(lon, lat)`).
881pub(crate) fn direction_cosines(longitude: f64, latitude: f64) -> [f64; 3] {
882    let cos_latitude = latitude.cos();
883    [
884        longitude.cos() * cos_latitude,
885        longitude.sin() * cos_latitude,
886        latitude.sin(),
887    ]
888}

890/// `MVDirection::get`: longitude in (−π, π] (0 at a pole) and latitude.
891pub(crate) fn direction_angles(cosines: [f64; 3]) -> (f64, f64) {
892    let longitude = if cosines[0] != 0.0 || cosines[1] != 0.0 {
893        cosines[1].atan2(cosines[0])
894    } else {
895        0.0
896    };
897    (longitude, cosines[2].asin())
898}

900/// `MVPosition(length, lon, lat)`: the vector, its length encoding a WGS84
901/// height as casacore encodes it.
902pub(crate) fn position_vector(longitude: f64, latitude: f64, length_m: f64) -> [f64; 3] {
903    let mut t = length_m;
904    if t < 0.0 && t > -7.0e6 {
905        t = t / 1.0e7 + HI_LIMIT_M;
906    } else if t > LO_LIMIT_M && t < HI_LIMIT_M {
907        t += 0.001;
908    }
909    let scale = if t == 0.0 { 1e-6 } else { t };
910    direction_cosines(longitude, latitude).map(|value| value * scale)
911}

913/// `MVPosition::get`: longitude, latitude and the (decoded) length.
914pub(crate) fn position_spherical(xyz: [f64; 3]) -> (f64, f64, f64) {
915    let (longitude, latitude, mut length) = spatial_spherical(xyz);
916    if length > LO_LIMIT_M && length < HI_LIMIT_M {
917        length = (length - HI_LIMIT_M) * 1e7;
918    }
919    (longitude, latitude, length)
920}
921
922/// Raw vector angles and magnitude without position height decoding.
923pub(crate) fn spatial_spherical(xyz: [f64; 3]) -> (f64, f64, f64) {
924    let length = norm(xyz);
925    let longitude = if xyz[0] != 0.0 || xyz[1] != 0.0 {
926        xyz[1].atan2(xyz[0])
927    } else {
928        0.0
929    };
930    let latitude = (xyz[2] / if length == 0.0 { 1.0 } else { length }).asin();
931    (longitude, latitude, length)
932}
