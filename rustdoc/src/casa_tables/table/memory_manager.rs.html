2//! A disk table whose schema persists and whose cells live in MemoryStMan.
3
4use std::fs;
5
6use crate::storage::table_control::{StagedTableDat, TableDatContents};
7use crate::storage::{DataManagerInfo, StorageError, TABLE_INFO_FILE};
8
9use super::*;
10
11const MANAGER: &str = "MemoryStMan";
12
13impl Table {
14    /// Bind this table's cells to a single named MemoryStMan, and save its
15    /// descriptor to disk. This table keeps the current cell values in
16    /// memory; reopening the path starts with default values, as in
17    /// casacore's `MSMBase::open64`.
18    ///
19    /// The output path must not already exist. This operation is intended
20    /// for a newly constructed table (e.g. a FITS import with memory storage).
21    #[cfg(unix)]
22    pub fn into_named_memory_manager(
23        mut self,
24        path: &Path,
25        manager_name: &str,
26        lock: LockOptions,
27    ) -> Result<Self, TableError> {
28        let io = |error| StorageError::Io(error);
29        fs::create_dir(path).map_err(io)?;
30        if let Err(error) = self.write_memory_layout(path, manager_name) {
31            let _ = fs::remove_dir_all(path);
32            return Err(error);
33        }
34        self.kind = TableKind::Memory;
35        self.source_path = Some(path.to_path_buf());
36        self.dm_info = vec![DataManagerInfo {
37            dm_type: MANAGER.to_string(),
38            name: manager_name.into(),
39            seq_nr: 0,
40            columns: self
41                .schema()
42                .expect("checked when writing")
43                .columns()
44                .iter()
45                .map(|column| column.name().to_owned())
46                .collect(),
47        }];
48        Self::open_with_lock_using(TableOptions::new(path), lock, |_| Ok(self))
49    }
50
51    fn write_memory_layout(&self, path: &Path, name: &str) -> Result<(), TableError> {
52        let schema = self.schema().ok_or_else(|| {
53            StorageError::FormatMismatch("a MemoryStMan descriptor requires a schema".into())
54        })?;
55        let mut contents = TableDatContents::from_snapshot(
56            schema,
57            self.keywords(),
58            self.inner.all_column_keywords(),
59            self.row_count() as u64,
60            MANAGER,
61            &[],
62            EndianFormat::default().is_big_endian(),
63        );
64        if schema.columns().is_empty() {
65            contents.column_set.data_managers.clear();
66        }
67        for column in &mut contents.table_desc.columns {
68            column.data_manager_type = MANAGER.into();
69            column.data_manager_group = name.into();
70        }
71        let control = StagedTableDat::in_table_directory(path)?;
72        self.note_persisted_write(path);
73        control.publish(&contents)?;
74        fs::write(path.join(TABLE_INFO_FILE), self.table_info.to_string())
75            .map_err(StorageError::Io)?;
76        Ok(())
77    }
78
79    /// Whether this table stores its cells in a memory manager while keeping
80    /// a descriptor and a lock file at a persistent path.
81    pub fn is_named_memory_manager(&self) -> bool {
82        self.kind == TableKind::Memory && self.source_path.is_some()
83    }
84
85    /// Finish a write to a named MemoryStMan table. Metadata changes write
86    /// the descriptor; cell changes stay on the current handle.
87    pub fn finish_memory_manager_write(&self, metadata: bool) -> Result<(), TableError> {
88        if self.is_named_memory_manager()
89            && let Some(path) = &self.source_path
90        {
91            if metadata {
92                let group = self
93                    .dm_info
94                    .first()
95                    .map_or(MANAGER, |manager| manager.name.as_str());
96                self.write_memory_layout(path, group)?;
97            }
98            self.mark_flushed();
99        }
100        Ok(())
101    }
102
103    pub(super) fn memory_manager_snapshot(snapshot: &StorageSnapshot) -> bool {
104        !snapshot.dm_info.is_empty()
105            && snapshot
106                .dm_info
107                .iter()
108                .all(|manager| manager.dm_type == MANAGER)
109    }
110
111    pub(super) fn from_memory_manager_snapshot(
112        path: PathBuf,
113        mut snapshot: StorageSnapshot,
114        rows: usize,
115    ) -> Self {
116        // MSMBase::flush writes no manager name. A reopened manager reports
117        // its type name as its group (even if a named handle created it).
118        for manager in &mut snapshot.dm_info {
119            manager.name = MANAGER.into();
120        }
121        let default = snapshot
122            .schema
123            .as_ref()
124            .expect("memory table has a schema")
125            .default_row();
126        let inner = TableImpl::with_rows_keywords_and_schema(
127            vec![default; rows],
128            Vec::new(),
129            snapshot.keywords,
130            snapshot.column_keywords,
131            snapshot.schema,
132        );
133        Self {
134            inner,
135            source_path: Some(path),
136            kind: TableKind::Memory,
137            virtual_columns: snapshot.virtual_columns,
138            virtual_bindings: Vec::new(),
139            table_info: snapshot.table_info,
140            dm_info: snapshot.dm_info,
141            external_sync: None,
142            measures: None,
143            marked_for_delete: false,
144            #[cfg(unix)]
145            lock_state: None,
146        }
147    }
148}
