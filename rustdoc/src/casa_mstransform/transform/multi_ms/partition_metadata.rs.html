2//! Consolidate SPW-scoped metadata before sharing it among Multi-MS members.
3
4use std::collections::BTreeSet;
5use std::path::{Path, PathBuf};
6
7use casa_core::{ScalarValue, Value};
8use casa_tables::{Table, TableOptions};
9
10use super::table_error;
11use crate::transform::selection_metadata::read_subtable_rows;
12use crate::transform::{MsTransformError, rewrite_selected_table_rows};
13use casa_ms::subtables::get_i32;

15pub(super) fn consolidate(
16    destination: &Path,
17    members: &[PathBuf],
18    groups: &[BTreeSet<i32>],
19) -> Result<(), MsTransformError> {
20    let master = destination.join(&members[0]);
21    merge_spectral_windows(&master, destination, members, groups)?;
22    for name in [
23        "FEED",
24        "SOURCE",
25        "FREQ_OFFSET",
26        "CALDEVICE",
27        "SYSPOWER",
28        "SYSCAL",
29    ] {
30        let path = master.join(name);
31        if !path.exists() {
32            continue;
33        }
34        let table =
35            Table::open(TableOptions::new(&path)).map_err(|error| table_error(&path, error))?;
36        let spws = (0..table.row_count())
37            .map(|row| {
38                get_i32(&table, row, "SPECTRAL_WINDOW_ID")
39                    .map_err(|error| MsTransformError::Spectrum(error.to_string()))
40            })
41            .collect::<Result<Vec<_>, _>>()?;
42        drop(table);
43        if spws.is_empty() {
44            continue;
45        }
46        let mut rows = Vec::new();
47        let mut output_ids = Vec::new();
48        for windows in groups {
49            let start = windows.first().copied().unwrap_or(0);
50            for (row, &id) in spws.iter().enumerate() {
51                if id < 0 || windows.contains(&id) {
52                    rows.push(row);
53                    // CASA offsets the wildcard rows from each new member,
54                    // along with that member's explicit SPW references.
55                    output_ids.push(if id < 0 { start + id } else { id });
56                }
57            }
58        }
59        if rows.iter().copied().eq(0..spws.len()) && output_ids == spws {
60            continue;
61        }
62        let original = master.join(format!(".{name}.partition-original"));
63        let io_error = |source: std::io::Error| MsTransformError::PrepareOutput {
64            path: path.display().to_string(),
65            reason: source.to_string(),
66        };
67        std::fs::rename(&path, &original).map_err(io_error)?;
68        let mut output_ids = output_ids.into_iter();
69        rewrite_selected_table_rows(&original, &path, &rows, |_, mut row| {
70            let id = output_ids.next().expect("each row has a mapped SPW");
71            row.upsert("SPECTRAL_WINDOW_ID", Value::Scalar(ScalarValue::Int32(id)));
72            Ok(row)
73        })?;
74        std::fs::remove_dir_all(original).map_err(io_error)?;
75    }
76    Ok(())
77}

79fn merge_spectral_windows(
80    master: &Path,
81    destination: &Path,
82    members: &[PathBuf],
83    groups: &[BTreeSet<i32>],
84) -> Result<(), MsTransformError> {
85    let name = "SPECTRAL_WINDOW";
86    let path = master.join(name);
87    let error = |source| table_error(&path, source);
88    let mut output = Table::open(TableOptions::new(&path)).map_err(error)?;
89    for (member, ids) in members.iter().zip(groups).skip(1) {
90        let source_path = destination.join(member).join(name);
91        let input = Table::open(TableOptions::new(&source_path))
92            .map_err(|source| table_error(&source_path, source))?;
93        for &id in ids {
94            let index = id as usize;
95            // Each member applies Doppler and phase-centre conversions at
96            // its own reference epoch. Publish the actual grid it wrote.
97            let mut rows = read_subtable_rows(&input, &[index])
98                .map_err(|source| table_error(&source_path, source))?;
99            *output.row_accessor_mut().row_mut(index).map_err(error)? =
100                rows.pop().expect("one selected spectral-window row");
101        }
102    }
103    output
104        .prepare_write()
105        .save(TableOptions::new(&path))
106        .map_err(error)
107}
