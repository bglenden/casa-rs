2//! Prediction: `predict`, `setoptions` and `setvp`, converted as casatools'
3//! `simulator_cmpt.cc` converts them and handed to the session.
4//!
5//! `setoptions` and `setvp` only record, and before `open` return `False`
6//! without converting their arguments, as CASA does. `predict` opens every
7//! named model image first, even before `open` (CASA's binding does, through
8//! `ImageFactory::fromFile`), so a missing image raises there; then it
9//! reads and prepares the model with the GIL released, and `close` writes
10//! the prediction.
11
12use std::path::PathBuf;

14use casa_simulation::{PredictSpec, PredictionOptions, VoltagePatternSpec};
15use pyo3::prelude::*;
16
17use super::{LOGGER, PySimulator, outcome, raised};
18use crate::convert::{ConvertResult, Variant, measures, quantity, value_in};
19use crate::tools::{LogLine, emit};

21/// `sm.predict(imagename, complist, incremental)`: predict the model into
22/// the rows observed so far.
23pub(super) fn predict(
24    tool: &PySimulator,
25    py: Python<'_>,
26    images: Vec<String>,
27    component_list: &str,
28    incremental: bool,
29) -> PyResult<bool> {
30    let spec = PredictSpec {
31        images: images.into_iter().map(PathBuf::from).collect(),
32        component_lists: vec![PathBuf::from(component_list)],
33        incremental,
34    };
35    py.detach(|| spec.check_images())
36        .map_err(|error| raised(&error))?;
37    let predicted = py.detach(|| {
38        tool.lock()
39            .as_mut()
40            .map(|session| session.simulator.predict(&spec))
41    });
42    let Some(result) = predicted else {
43        return Ok(false);
44    };
45    let mut lines = Vec::new();
46    let result = outcome(result, &mut lines);
47    emit(py, LOGGER, &lines)?;
48    result
49}

51/// The arguments of `sm.setoptions`.
52pub(super) struct OptionsRequest<'a> {
53    pub(super) ftmachine: &'a str,
54    pub(super) grid_function: &'a str,
55    pub(super) location: Variant,
56    pub(super) padding: f64,
57    pub(super) w_planes: i64,
58}
59
60/// `sm.setoptions(...)`: record the FTMachine `predict` degrids through,
61/// with `GridFT`'s padding and `WProjectFT`'s planes.
62///
63/// `location` is converted as CASA's binding converts it: one that does not
64/// convert is logged and the call goes on. `cache`, `tile`, `facets` and
65/// `maxdata`, which do not change what `GridFT` or `WProjectFT` predicts,
66/// are accepted as typed.
67pub(super) fn setoptions(
68    tool: &PySimulator,
69    py: Python<'_>,
70    request: &OptionsRequest<'_>,
71) -> PyResult<bool> {
72    let result = tool.on_session(py, |session, lines| {
73        if measures::position(&request.location, &session.measures)?.is_none()
74            && request.location != Variant::Empty
75        {
76            lines.push(LogLine::error(format!(
77                "Could not convert location {} to a Position Measures",
78                request.location.describe()
79            )));
80        }
81        let options = PredictionOptions {
82            ftmachine: request.ftmachine.to_string(),
83            grid_function: request.grid_function.to_string(),
84            padding: request.padding,
85            w_planes: request.w_planes,
86        };
87        outcome(session.simulator.set_options(&options), lines)
88    })?;
89    Ok(result.unwrap_or(false))
90}

92/// The arguments of `sm.setvp`.
93pub(super) struct VoltagePatternRequest<'a> {
94    pub(super) dovp: bool,
95    pub(super) use_default_vp: bool,
96    pub(super) vp_table: &'a str,
97    pub(super) do_squint: bool,
98    pub(super) parallactic_angle_increment: Variant,
99    pub(super) sky_position_threshold: Variant,
100}

102/// A `setvp` angle in degrees: CASA's binding keeps `default_deg` for `''`
103/// and `[]`, and converts anything else (raising when it cannot).
104fn angle_deg(value: &Variant, default_deg: f64) -> ConvertResult<f64> {
105    match value {
106        Variant::Empty => Ok(default_deg),
107        Variant::Text(text) if text.is_empty() => Ok(default_deg),
108        other => value_in(&quantity(other)?, "deg"),
109    }
110}

112/// `sm.setvp(...)`: record whether `predict` applies the primary beam.
113/// `pblimit`, which `GridFT` does not use, is accepted as typed.
114pub(super) fn setvp(
115    tool: &PySimulator,
116    py: Python<'_>,
117    request: &VoltagePatternRequest<'_>,
118) -> PyResult<bool> {
119    let result = tool.on_session(py, |session, lines| {
120        let parallactic_angle_increment_deg =
121            angle_deg(&request.parallactic_angle_increment, 360.0)?;
122        angle_deg(&request.sky_position_threshold, 180.0)?;
123        let spec = VoltagePatternSpec {
124            dovp: request.dovp,
125            use_default_vp: request.use_default_vp,
126            vp_table: request.vp_table.to_string(),
127            do_squint: request.do_squint,
128            parallactic_angle_increment_deg,
129        };
130        outcome(session.simulator.set_vp(&spec), lines)
131    })?;
132    Ok(result.unwrap_or(false))
133}
