22const ARCSEC: f64 = std::f64::consts::PI / (180. * 3600.);
23
24pub(super) fn image(
25    input: &Path,
26    p: &Parameters<'_>,
27    plan: &Plan,
28    resources: SessionResources,
29    cancel: &casa_imaging::Cancel,
30) -> Result<Vec<PathBuf>, SimobserveError> {
31    let model = PagedImage::<f32>::open(&plan.model).map_err(failure)?;
32    let CoordinateModel::Direction(direction) = model.coordinates().coordinate(0) else {
33        return Err(failure("sky model has no direction coordinate"));
34    };
35    let world = direction
36        .to_world(&[model.shape()[0] as f64 / 2., model.shape()[1] as f64 / 2.])
37        .map_err(failure)?;
38    let phasecenter = format!(
39        "{} {} {}",
40        direction.direction_ref().as_str(),
41        qa::hms(&Quantity::new(world[0], "rad").map_err(failure)?).map_err(failure)?,
42        qa::dms(&Quantity::new(world[1], "rad").map_err(failure)?).map_err(failure)?
43    );
44    let frequency = model.coordinates().coordinate(2).reference_value()[0];
45    let nchan = model.shape()[3];
46    let ms = MeasurementSet::open(input).map_err(failure)?;
47    let antennas = ms.antenna().map_err(failure)?;
48    let diameter = (0..antennas.row_count())
49        .map(|row| antennas.dish_diameter(row))
50        .collect::<Result<Vec<_>, _>>()
51        .map_err(failure)?
52        .iter()
53        .sum::<f64>()
54        / antennas.row_count() as f64;
55    let primary_beam = casa_imaging_singledish::primary_beam_fwhm_arcsec(
56        frequency,
57        diameter,
58        if diameter == 12. { 0.75 } else { 0. },
59        10.,
60    );
61    let sfcell = primary_beam / 9.;
62    let cell = plan.imaging["cell"]
63        .as_array()
64        .unwrap()
65        .iter()
66        .map(|value| qa::value_in(&qa::read(value.as_str().unwrap())?, "arcsec"))
67        .collect::<Result<Vec<_>, _>>()
68        .map_err(failure)?;
69    if cell.iter().any(|&value| value > sfcell) {
70        return Err(failure(format!(
71            "SF total-power imaging requires cell <= {sfcell}arcsec"
72        )));
73    }
74    let size = plan.imaging["imsize"].as_array().unwrap();
75    let size = [
76        size[0].as_u64().unwrap() as usize,
77        size[1].as_u64().unwrap() as usize,
78    ];
79    let temporary_size =
80        [0, 1].map(|axis| (cell[axis] / sfcell * size[axis] as f64).ceil() as usize);
81    let project = plan.directory.file_name().unwrap().to_string_lossy();
82    let output = plan.directory.join(format!("{project}.sd.image"));
83    if output.exists() && !p.flag("overwrite") {
84        return Err(failure("total-power image already exists"));
85    }
86    let temporary = suffixed(&output, "0");
87    let source = std::fs::canonicalize(&plan.model).map_err(failure)?;
88    for target in [
89        output.clone(),
90        temporary.clone(),
91        suffixed(&temporary, ".scaled"),
92        suffixed(&temporary, ".image"),
93        suffixed(&temporary, ".weight"),
94    ] {
95        if target.exists() {
96            if !p.flag("overwrite") {
97                return Err(failure(format!("{} already exists", target.display())));
98            }
99            if std::fs::canonicalize(&target).map_err(failure)? == source {
100                return Err(failure(
101                    "sky model would be overwritten by total-power imaging",
102                ));
103            }
104            PagedImage::<f32>::open(&target).map_err(failure)?;
105        }
106    }
107    let sparse = json!({"infiles": [input], "outfile": temporary, "overwrite": p.flag("overwrite"),
108            "phasecenter": phasecenter, "mode": "channel", "nchan": nchan, "start": 0, "width": 1,
109            "gridfunction": "SF", "convsupport": 6, "imsize": temporary_size,
110            "cell": [format!("{sfcell}arcsec"),format!("{sfcell}arcsec")]});
111    let gridded = casa_imaging::single_dish::image_from_sparse_parameters(
112        sparse.as_object().unwrap(),
113        &context(resources, cancel),
114    )
115    .map_err(failure)?;
116    let beam = gridded
117        .beam
118        .ok_or_else(|| failure("total-power raster has no measurable beam"))?;
119    let ratio = beam.major_in("arcsec").map_err(failure)?
120        * beam.minor_in("arcsec").map_err(failure)?
121        / primary_beam.powi(2);
122    // CASA formats this immath factor with %f (six decimal places).
123    let ratio: f32 = format!("{ratio:.6}").parse().map_err(failure)?;
124    let _memory = admit(
125        resources.host(),
126        resources.policy(),
127        &Demand {
128            phase: "simulation total-power scaling and regrid",
129            memory: total_power_memory(temporary_size, size, nchan)?,
130        },
131    )
132    .map_err(failure)?;
133    let image = PagedImage::<f32>::open(&gridded.image).map_err(failure)?;
134    let pixels = image.get().map_err(failure)?.mapv(|value| value * ratio);
135    let scaled = suffixed(&temporary, ".scaled");
136    let scaled_image = analysis::write_product(
137        &scaled,
138        &pixels,
139        image.coordinates().clone(),
140        image.units(),
141        &image.image_info().map_err(failure)?,
142        image.default_mask().map_err(failure)?.as_ref(),
143    )
144    .map_err(failure)?;
145    let mut template = RegridTemplate::of(&scaled_image);
146    let CoordinateModel::Direction(direction) = template.coordinates.coordinate(0) else {
147        unreachable!()
148    };
149    let direction = direction
150        .clone()
151        .with_reference_pixel([size[0] as f64 / 2., size[1] as f64 / 2.])
152        .with_increment([-cell[0] * ARCSEC, cell[1] * ARCSEC]);
153    template.coordinates.replace_coordinate(0, direction);
154    template.shape[..2].copy_from_slice(&size);
155    let regridded = regrid(&scaled_image, &template, RegridMethod::Cubic).map_err(failure)?;
156    let staging = tempfile::tempdir_in(&plan.directory).map_err(failure)?;
157    let staged = staging.path().join("image");
158    analysis::write_product(
159        &staged,
160        &regridded.data,
161        regridded.coordinates,
162        image.units(),
163        &image.image_info().map_err(failure)?,
164        Some(&regridded.mask),
165    )
166    .map_err(failure)?;
167    casa_imaging_products::publish_staged_image(&staged, &output).map_err(failure)?;
168    drop(image);
169    super::replace(&temporary, true)?;
170    std::fs::rename(&gridded.image, &temporary).map_err(failure)?;
171    Ok(vec![temporary, gridded.weight, scaled, output])
172}

174fn total_power_memory(
175    input: [usize; 2],
176    output: [usize; 2],
177    channels: usize,
178) -> Result<u64, SimobserveError> {
179    let grid = |shape: [usize; 2]| {
180        (shape[0] as u64)
181            .checked_mul(shape[1] as u64)
182            .and_then(|cells| cells.checked_mul(channels as u64))
183    };
184    grid(input)
185        .and_then(|cells| cells.checked_mul(32))
186        .and_then(|bytes| {
187            grid(output)
188                .and_then(|cells| cells.checked_mul(48))
189                .and_then(|output| bytes.checked_add(output))
190        })
191        .and_then(|bytes| bytes.checked_add(64 << 20))
192        .ok_or_else(|| failure("total-power regrid memory overflows"))
193}
