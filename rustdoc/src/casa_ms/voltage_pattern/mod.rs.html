17/// Failure to read a saved voltage-pattern table or interpret its record.
18#[derive(Debug, thiserror::Error)]
19#[error("{0}")]
20pub struct VoltagePatternError(String);
21
22impl VoltagePatternError {
23    fn read(error: impl ToString) -> Self {
24        Self(error.to_string())
25    }
26}
27
28/// The root of a CASA VPManager table and its telescope defaults subtable.
29pub struct VoltagePatternTable {
30    root: Table,
31    defaults: PathBuf,
32}
33
34impl VoltagePatternTable {
35    /// Estimate peak bytes needed to open one VPManager table. The caller
36    /// holds this reservation while its decoded records are in use.
37    ///
38    /// # Errors
39    /// The table header or directory entries could not be read.
40    pub fn reading_bytes(path: &Path) -> Result<u64, VoltagePatternError> {
41        let rows = Table::rows_on_disk(path).map_err(VoltagePatternError::read)? as u64;
42        let mut bytes = rows.saturating_mul(4096).saturating_add(64 << 10);
43        for entry in std::fs::read_dir(path).map_err(VoltagePatternError::read)? {
44            let metadata = entry
45                .and_then(|entry| entry.metadata())
46                .map_err(VoltagePatternError::read)?;
47            if metadata.is_file() {
48                bytes = bytes.saturating_add(metadata.len().saturating_mul(4));
49            }
50        }
51        Ok(bytes)
52    }
53
54    /// Open the descriptions and locate the defaults table via its CASA
55    /// keyword. This does not open the defaults table.
56    ///
57    /// # Errors
58    /// The root table is unreadable or has no `VPLIST_DEFAULTS` table keyword.
59    pub fn open(path: &Path) -> Result<Self, VoltagePatternError> {
60        let root = Table::open(TableOptions::new(path)).map_err(VoltagePatternError::read)?;
61        let Some(Value::TableRef(reference)) = root.keywords().get("VPLIST_DEFAULTS") else {
62            return Err(VoltagePatternError::read(
63                "missing VPLIST_DEFAULTS subtable",
64            ));
65        };
66        let defaults = path.join(reference);
67        Ok(Self { root, defaults })
68    }
69
70    /// Estimate peak bytes needed to read the defaults subtable.
71    ///
72    /// # Errors
73    /// The defaults subtable cannot be sized.
74    pub fn defaults_reading_bytes(&self) -> Result<u64, VoltagePatternError> {
75        Self::reading_bytes(&self.defaults)
76    }
77
78    /// Read all VP definitions after checking the saved default indices.
79    ///
80    /// # Errors
81    /// A definition or default row is unreadable, or a default index is invalid.
82    pub fn definitions(&self) -> Result<Vec<RecordValue>, VoltagePatternError> {
83        let defaults =
84            Table::open(TableOptions::new(&self.defaults)).map_err(VoltagePatternError::read)?;
85        for row in 0..defaults.row_count() {
86            let name = default_cell(&defaults, row, "tel_and_anttype")?;
87            if !matches!(name, ScalarValue::String(_)) {
88                return Err(VoltagePatternError::read(
89                    "VPLIST_DEFAULTS.tel_and_anttype is not a string",
90                ));
91            }
92            let index = default_index(&defaults, row)?;
93            if index < -1 || usize::try_from(index).is_ok_and(|i| i >= self.root.row_count()) {
94                return Err(VoltagePatternError::read(
95                    "VPLIST_DEFAULTS selects an absent beam",
96                ));
97            }
98        }
99        (0..self.root.row_count())
100            .map(|row| self.definition(row))
101            .collect()
102    }
103
104    /// Copy the root and the defaults subtable into an independently owned table.
105    /// The caller can update the returned copy and save it at `path`.
106    ///
107    /// # Errors
108    /// The table could not be read or copied.
109    pub fn copy_to(&self, path: &Path) -> Result<Table, VoltagePatternError> {
110        let defaults =
111            Table::open(TableOptions::new(&self.defaults)).map_err(VoltagePatternError::read)?;
112        let output_defaults =
113            std::path::absolute(path.join("VPLIST_DEFAULTS")).map_err(VoltagePatternError::read)?;
114        self.root
115            .deep_copy(TableOptions::new(path))
116            .map_err(VoltagePatternError::read)?;
117        defaults
118            .deep_copy(TableOptions::new(&output_defaults))
119            .map_err(VoltagePatternError::read)?;
120        let mut copy = Table::open(TableOptions::new(path)).map_err(VoltagePatternError::read)?;
121        copy.keywords_mut().upsert(
122            "VPLIST_DEFAULTS",
123            Value::TableRef(output_defaults.to_string_lossy().into_owned()),
124        );
125        Ok(copy)
126    }
127
128    /// Read the last matching default row, as `VPManager::loadfromtable`
129    /// updates the telescope map. A saved index of `-1` selects a built-in
130    /// pattern and returns `None`.
131    ///
132    /// # Errors
133    /// The telescope default is missing, invalid, or names an unreadable row.
134    pub fn default_record(
135        &self,
136        telescope: &str,
137    ) -> Result<Option<RecordValue>, VoltagePatternError> {
138        let defaults =
139            Table::open(TableOptions::new(&self.defaults)).map_err(VoltagePatternError::read)?;
140        let mut selected = None;
141        for row in 0..defaults.row_count() {
142            let name = default_cell(&defaults, row, "tel_and_anttype")?;
143            if matches!(name, ScalarValue::String(name) if name == telescope) {
144                selected = Some(default_index(&defaults, row)?);
145            }
146        }
147        let selected = selected.ok_or_else(|| {
148            VoltagePatternError::read(format!("no default for telescope {telescope}"))
149        })?;
150        if selected == -1 {
151            return Ok(None);
152        }
153        let index = usize::try_from(selected).map_err(VoltagePatternError::read)?;
154        if index >= self.root.row_count() {
155            return Err(VoltagePatternError::read(
156                "VPLIST_DEFAULTS selects an absent beam",
157            ));
158        }
159        self.definition(index).map(Some)
160    }
161
162    fn definition(&self, row: usize) -> Result<RecordValue, VoltagePatternError> {
163        self.root
164            .cell_accessor(row, "pbdescription")
165            .and_then(|cell| cell.record())
166            .map_err(VoltagePatternError::read)
167    }
168}
169
170fn default_cell<'a>(
171    table: &'a Table,
172    row: usize,
173    column: &str,
174) -> Result<&'a ScalarValue, VoltagePatternError> {
175    table
176        .cell_accessor(row, column)
177        .and_then(|cell| cell.scalar())
178        .map_err(VoltagePatternError::read)
179}
180
181fn default_index(table: &Table, row: usize) -> Result<i32, VoltagePatternError> {
182    match default_cell(table, row, "vplistnum")? {
183        ScalarValue::Int32(index) => Ok(*index),
184        _ => Err(VoltagePatternError::read(
185            "VPLIST_DEFAULTS.vplistnum is not an integer",
186        )),
187    }
188}

190/// Decode the circular Gaussian and Airy laws shared by imaging and
191/// simulation. CASA stores radii at the record's reference frequency.
192/// The result scales them to the arcmin-GHz argument of PBMath1D.
193///
194/// # Errors
195/// The record is invalid or names a law other than `GAUSS` or `AIRY`.
196pub fn radial_pattern(record: &RecordValue) -> Result<PrimaryBeam, VoltagePatternError> {
197    let Some(Value::Scalar(ScalarValue::String(name))) = record.get("name") else {
198        return Err(VoltagePatternError::read("missing string name"));
199    };
200    let (radius, ghz) = reference_radius(record)?;
201    match name.as_str() {
202        "AIRY" => {
203            let diameter = quantity(record, "dishdiam", "m")?;
204            let blockage = quantity(record, "blockagediam", "m")?;
205            if !diameter.is_finite()
206                || diameter <= 0.
207                || !blockage.is_finite()
208                || blockage < 0.
209                || blockage >= diameter
210            {
211                return Err(VoltagePatternError::read(
212                    "invalid dish or blockage diameter",
213                ));
214            }
215            Ok(PrimaryBeam::Airy {
216                dish_diameter_m: diameter,
217                blockage_diameter_m: blockage,
218                maximum_radius_arcmin_ghz: radius,
219            })
220        }
221        "GAUSS" => {
222            let width = quantity(record, "halfwidth", "arcmin")? * ghz;
223            if !width.is_finite() || width <= 0. {
224                return Err(VoltagePatternError::read(
225                    "Gaussian HWHM must be finite and positive",
226                ));
227            }
228            let Some(Value::Scalar(ScalarValue::Bool(is_voltage))) = record.get("isthisvp") else {
229                return Err(VoltagePatternError::read("missing boolean isthisvp"));
230            };
231            Ok(PrimaryBeam::Gaussian {
232                half_width_arcmin_ghz: width,
233                maximum_radius_arcmin_ghz: radius,
234                is_voltage: *is_voltage,
235            })
236        }
237        _ => Err(VoltagePatternError::read(format!(
238            "saved radial law {name} (#1307)"
239        ))),
240    }
241}

243fn reference_radius(record: &RecordValue) -> Result<(f64, f64), VoltagePatternError> {
244    let ghz = quantity(record, "reffreq", "GHz")?;
245    let radius = quantity(record, "maxrad", "arcmin")? * ghz;
246    if !ghz.is_finite() || ghz <= 0. || !radius.is_finite() || radius <= 0. {
247        return Err(VoltagePatternError::read(
248            "reference frequency and support must be finite and positive",
249        ));
250    }
251    Ok((radius, ghz))
252}

254/// Read a stored radial pattern's squint in tangent-angle radians and Hz.
255/// An omitted or zero direction leaves the beam centre unchanged.
256///
257/// # Errors
258/// A supplied direction or reference frequency is invalid.
259pub fn radial_squint(record: &RecordValue) -> Result<Option<([f64; 2], f64)>, VoltagePatternError> {
260    let Some(direction) = record.get("squintdir") else {
261        return Ok(None);
262    };
263    let Value::Record(direction) = direction else {
264        return Err(VoltagePatternError::read(
265            "squintdir must be a direction record",
266        ));
267    };
268    let direction = direction_from_record(direction).map_err(VoltagePatternError::read)?;
269    // PBMath reads a signed tangent offset, even when the direction was
270    // serialized with its longitude in [0, 2π).
271    let [x, y, _] = direction.cosines();
272    let offset = [y.atan2(x), direction.latitude_rad()];
273    if offset == [0.; 2] {
274        return Ok(None);
275    }
276    let frequency = quantity(record, "squintreffreq", "Hz")?;
277    if !offset.iter().all(|value| value.is_finite()) || !frequency.is_finite() || frequency <= 0. {
278        return Err(VoltagePatternError::read(
279            "invalid radial beam squint or reference frequency",
280        ));
281    }
282    Ok(Some((offset, frequency)))
283}

285fn quantity(record: &RecordValue, field: &str, unit: &str) -> Result<f64, VoltagePatternError> {
286    let Some(Value::Record(value)) = record.get(field) else {
287        return Err(VoltagePatternError::read(format!(
288            "missing quantity {field}"
289        )));
290    };
291    let Some(Value::Scalar(ScalarValue::Float64(number))) = value.get("value") else {
292        return Err(VoltagePatternError::read(format!(
293            "non-scalar quantity {field}"
294        )));
295    };
296    let Some(Value::Scalar(ScalarValue::String(saved_unit))) = value.get("unit") else {
297        return Err(VoltagePatternError::read("missing string unit"));
298    };
299    Quantity::new(*number, saved_unit)
300        .and_then(|q| q.get_value_in(&Unit::new(unit)?))
301        .map_err(VoltagePatternError::read)
302}
