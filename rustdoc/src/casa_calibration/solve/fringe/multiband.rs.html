2//! Fringe search with an FFT per spectral window and coherent band combination.

4use std::f64::consts::TAU;

13const PADDING: usize = 4;
14
15pub(super) struct MultibandSearch {
16    fft: Fft2<f32>,
17    bands: Vec<Band>,
18    grid: GridSpec,
19}
20
21struct Band {
22    description: i32,
23    offset: f64,
24    spectrum: Vec<Complex32>,
25}
26
27impl MultibandSearch {
28    pub(super) fn new<'a>(
29        mut grid: GridSpec,
30        windows: impl Iterator<Item = (i32, &'a Window)>,
31    ) -> Result<Self, GainSolveError> {
32        let mut windows = windows.collect::<Vec<_>>();
33        windows.sort_by_key(|(description, window)| (window.spw, *description));
34        let first = windows[0].1.selected_hz[0] as f32;
35        let last = &windows[windows.len() - 1].1.selected_hz;
36        grid.padding = PADDING;
37        grid.channels = windows
38            .iter()
39            .map(|(_, w)| w.selected_hz.len())
40            .max()
41            .unwrap_or(0);
42        grid.channel_spacing_hz = last
43            .get(1)
44            .map_or(grid.channel_spacing_hz, |&second| second - last[0]);
45        let (shape, elements) = grid.transform_shape(windows.len())?;
46        let fft = Fft2::with_threads(shape, 1)
47            .map_err(error)?
48            .with_estimated_plan();
49        let bandwidth = grid.channels as f64 * grid.channel_spacing_hz;
50        let bands = windows
51            .into_iter()
52            .map(|(description, window)| {
53                let mut spectrum = Vec::new();
54                spectrum.try_reserve_exact(elements).map_err(error)?;
55                spectrum.resize(elements, Complex32::default());
56                let delta = f64::from(window.selected_hz[0] as f32) - f64::from(first);
57                Ok(Band {
58                    description,
59                    offset: f64::from((delta / bandwidth) as f32),
60                    spectrum,
61                })
62            })
63            .collect::<Result<_, GainSolveError>>()?;
64        Ok(Self { fft, bands, grid })
65    }

67    pub(super) fn peak(
68        &mut self,
69        samples: &[FringeSample],
70        sign: f64,
71    ) -> Result<Option<FringePeak>, GainSolveError> {
72        let [channels, times] = self.fft.shape();
73        let mut weights = NoiseWeights::default();
74        for band in &mut self.bands {
75            band.spectrum.fill(Complex32::default());
76            for sample in samples
77                .iter()
78                .filter(|sample| sample.data_description == band.description)
79            {
80                band.spectrum[sample.local_channel * times + sample.integration] =
81                    sample.weighted_unit_visibility();
82                weights.add(sample.weight);
83            }
84            self.fft
85                .transform(&mut band.spectrum, false)
86                .map_err(error)?;
87        }
88        let [time, channel] = self.grid_peak(channels, times);
89        let evaluate = |[time, channel]: [f64; 2]| {
90            let time = if time < 0. { time + times as f64 } else { time };
91            let channel = if channel < 0. {
92                channel + channels as f64
93            } else {
94                channel
95            };
96            let time_weights = interpolation_weights(time, times);
97            let channel_weights = interpolation_weights(channel, channels);
98            self.bands
99                .iter()
100                .map(|band| {
101                    let rotation = phasor(-TAU * band.offset * channel.fract());
102                    let mut sum = Complex32::default();
103                    for (channel, &channel_weight) in channel_weights.iter().enumerate() {
104                        for (time, &time_weight) in time_weights.iter().enumerate() {
105                            sum += band.spectrum[channel * times + time]
106                                * ((rotation * time_weight) * channel_weight);
107                        }
108                    }
109                    sum
110                })
111                .sum::<Complex32>()
112        };
113        let point = refine_peak([time, channel], |point| -evaluate(point).norm());
114        let amplitude = evaluate(point);
115        if amplitude.norm_sqr() == 0. {
116            return Ok(None);
117        }
118        // CASA's separate-band search uses the entire FFT rather than the
119        // supplied windows, and folds the padded indices on the original grid.
120        let rate_period = self.grid.integration_seconds * self.grid.center_hz;
121        let delay_period = self.grid.channel_spacing_hz * 1e-9;
122        Ok(Some(FringePeak {
123            parameters: FringeParameters {
124                phase_rad: sign * f64::from(amplitude.arg()),
125                delay_ns: sign * folded_bin(point[1], self.grid.channels) / delay_period,
126                rate: sign * folded_bin(point[0], self.grid.timestamps) / rate_period,
127                dispersion: 0.,
128            },
129            snr: weights.snr(amplitude.norm()),
130            accepted: true,
131            active: true,
132        }))
133    }

135    fn grid_peak(&self, channels: usize, times: usize) -> [f64; 2] {
136        let extent = self.bands.iter().map(|band| band.offset).fold(0., f64::max);
137        let bins = (extent + 1.5) as usize;
138        let mut best = [0., 0.];
139        let mut largest = -1.;
140        for bin in 0..bins {
141            let fraction = (bin as f64 - (bins / 2) as f64) / bins as f64;
142            let rotations = self
143                .bands
144                .iter()
145                .map(|band| phasor(-TAU * band.offset * fraction))
146                .collect::<Vec<_>>();
147            for time in 0..times {
148                for channel in 0..channels {
149                    let index = channel * times + time;
150                    let combined = self
151                        .bands
152                        .iter()
153                        .zip(&rotations)
154                        .map(|(band, &rotation)| band.spectrum[index] * rotation)
155                        .sum::<Complex32>();
156                    let norm = combined.norm();
157                    if norm > largest {
158                        largest = norm;
159                        best = [time as f64, channel as f64 + fraction];
160                    }
161                }
162            }
163        }
164        best
165    }
166}

168fn phasor(phase: f64) -> Complex32 {
169    let (sin, cos) = (phase as f32).sin_cos();
170    Complex32::new(cos, sin)
171}
172
173fn interpolation_weights(position: f64, size: usize) -> Vec<Complex32> {
174    (0..size)
175        .map(|index| {
176            let delta = position - index as f64;
177            if position.fract().abs() < 1e-8 {
178                Complex32::new(f32::from(delta == 0.), 0.)
179            } else {
180                let one = Complex32::new(1., 0.);
181                (one - phasor(-TAU * delta))
182                    / (one - phasor(-TAU * delta / size as f64))
183                    / size as f32
184            }
185        })
186        .collect()
187}
188
189fn folded_bin(index: f64, extent: usize) -> f64 {
190    let fraction = index / extent as f64;
191    if fraction > 0.5 {
192        fraction - 1.
193    } else {
194        fraction
195    }
196}

198fn refine_peak(start: [f64; 2], cost: impl Fn([f64; 2]) -> f32) -> [f64; 2] {
199    // A short Nelder–Mead refinement of the fractional delay and rate bins.
200    let mut points = [
201        start,
202        [start[0] + 0.01, start[1]],
203        [start[0], start[1] + 0.01],
204    ]
205    .map(|point| (point, cost(point)));
206    for _ in 0..5 {
207        points.sort_by(|left, right| left.1.total_cmp(&right.1));
208        let [best, second, worst] = points;
209        let center = std::array::from_fn::<_, 2, _>(|i| (best.0[i] + second.0[i]) * 0.5);
210        let trial = |scale, point: [f64; 2]| {
211            let point = std::array::from_fn(|i| center[i] + scale * (point[i] - center[i]));
212            (point, cost(point))
213        };
214        let reflected = trial(-1., worst.0);
215        if reflected.1 < best.1 {
216            let expanded = trial(-2., worst.0);
217            points[2] = if expanded.1 < best.1 {
218                expanded
219            } else {
220                reflected
221            };
222        } else if reflected.1 > second.1 {
223            let closer = if reflected.1 < worst.1 {
224                reflected
225            } else {
226                worst
227            };
228            let contracted = trial(0.5, closer.0);
229            if contracted.1 <= closer.1 {
230                points[2] = contracted;
231            } else {
232                for vertex in &mut points[1..] {
233                    let midpoint = std::array::from_fn(|i| (best.0[i] + vertex.0[i]) * 0.5);
234                    *vertex = (midpoint, cost(midpoint));
235                }
236            }
237        } else {
238            points[2] = reflected;
239        }
240    }
241    points
242        .into_iter()
243        .min_by(|left, right| left.1.total_cmp(&right.1))
244        .unwrap()
245        .0
246}
