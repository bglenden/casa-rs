2//! Cubic B-spline knots and the four active basis functions at each integration.
3
4#[derive(Clone, Copy)]
5pub(super) struct SampleBasis {
6    pub(super) start: usize,
7    pub(super) weights: [f64; 4],
8}
9
10/// The CASA knot grid has four repeated end knots and equally spaced interior
11/// knots. An interval with fewer timestamps than control points cannot be fit.
12pub(super) fn knots(times: &[f64], seconds: f64) -> Option<Vec<f64>> {
13    let (&start, &end) = (times.first()?, times.last()?);
14    let duration = end - start;
15    if !duration.is_finite() || duration <= 0. || !seconds.is_finite() {
16        return None;
17    }
18    let segments = if seconds > 0. {
19        (duration / seconds).round().max(1.) as usize
20    } else {
21        1
22    };
23    if segments.checked_add(3)? > times.len() {
24        return None;
25    }
26    let step = duration / segments as f64;
27    let mut knots = vec![start; 4];
28    knots.extend((1..segments).map(|segment| start + segment as f64 * step));
29    knots.extend([end; 4]);
30    Some(knots)
31}
32
33pub(super) fn sample(knots: &[f64], time: f64) -> Option<SampleBasis> {
34    let count = knots.len().checked_sub(4)?;
35    if count < 4 || !time.is_finite() || time < knots[3] || time > knots[count] {
36        return None;
37    }
38    let span = knots
39        .partition_point(|&knot| knot <= time)
40        .saturating_sub(1)
41        .min(count - 1);
42    let start = span.checked_sub(3)?;
43    let mut weights = [1., 0., 0., 0.];
44    for degree in 1..=3 {
45        let mut saved = 0.;
46        for (offset, weight) in weights[..degree].iter_mut().enumerate() {
47            let first = knots[span + 1 + offset - degree];
48            let last = knots[span + offset + 1];
49            let width = last - first;
50            if width <= 0. {
51                return None;
52            }
53            let scaled = *weight / width;
54            *weight = saved + (last - time) * scaled;
55            saved = (time - first) * scaled;
56        }
57        weights[degree] = saved;
58    }
59    Some(SampleBasis { start, weights })
60}
