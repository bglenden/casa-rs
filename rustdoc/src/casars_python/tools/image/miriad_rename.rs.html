12use super::attachment::{

16use super::handle::{ImageHandle, raised, same_file_identity};

19pub(super) struct MiriadDirectory {
20    physical_path: PathBuf,
21    identity: Metadata,
22    #[cfg(unix)]
23    _directory: fs::File,
24}
25
26impl MiriadDirectory {
27    pub(super) fn open(path: &Path) -> PyResult<Self> {
28        let physical_path = path.canonicalize().map_err(raised)?;
29        #[cfg(unix)]
30        let directory =
31            casa_tables::publication::open_directory_identity(&physical_path).map_err(raised)?;
32        #[cfg(unix)]
33        let identity = directory.metadata().map_err(raised)?;
34        #[cfg(not(unix))]
35        let identity = physical_path.metadata().map_err(raised)?;
36        Ok(Self {
37            physical_path,
38            identity,
39            #[cfg(unix)]
40            _directory: directory,
41        })
42    }
43
44    pub(super) fn check(&self, name: &Path) -> PyResult<()> {
45        if name
46            .canonicalize()
47            .is_ok_and(|path| path == self.physical_path)
48            && self.physical_path.symlink_metadata().is_ok_and(|current| {
49                current.is_dir() && same_file_identity(&self.identity, &current)
50            })
51        {
52            return Ok(());
53        }
54        Err(raised(format!(
55            "Cannot rename image {} because the image at that path has changed",
56            name.display()
57        )))
58    }
59
60    pub(super) fn rename(
61        &self,
62        old_name: &Path,
63        name: &str,
64        expanded: &str,
65        overwrite: bool,
66        history: bool,
67    ) -> PyResult<AttachedImage> {
68        let same_name =
69            || raised("Exception: Specified output name is the same as the current image name.");
70        if OsStr::new(name) == old_name.as_os_str() {
71            return Err(same_name());
72        }
73        let source = &self.physical_path;
74        let destination = native_destination(expanded, name)?;
75        let target_snapshot = match destination.symlink_metadata() {
76            Ok(metadata) if overwrite => Some(metadata),
77            Ok(_) => {
78                return Err(raised(format!(
79                    "Exception: File {name} exists, and the user does not want to remove it.."
80                )));
81            }
82            Err(error) if error.kind() == ErrorKind::NotFound => None,
83            Err(error) => return Err(raised(error)),
84        };
85        if &destination == source {
86            return Err(same_name());
87        }
88        check_path_relationship(source, &destination, name)?;
89
90        let images = registry();
91        self.check(old_name)?;
92        let replace_alias = target_snapshot
93            .as_ref()
94            .is_some_and(|entry| entry.is_symlink());
95        if conflicting_images(&images, source, 0)
96            || conflicting_images(&images, &destination, 0)
97            || containing_tables(source).any(crate::tools::table::path_is_in_use)
98            || containing_tables(&destination).any(crate::tools::table::path_is_in_use)
99            || crate::tools::table::path_is_in_use(source)
100            || (!replace_alias && crate::tools::table::path_is_in_use(&destination))
101        {
102            return Err(raised(
103                "Cannot rename MIRIAD image while a table in either path is open",
104            ));
105        }
106        if target_snapshot.is_some() && check_write_access(&destination).is_err() {
107            return Err(raised(format!(
108                "RegularFile::copy/move: target file {} already exists and is not writable",
109                destination.display()
110            )));
111        }
112
113        let mut leases = BTreeMap::new();
114        lock_table_tree_for_read_only_move(source, &mut leases)?;
115        if target_snapshot.as_ref().is_some_and(Metadata::is_dir) {
116            lock_table_tree(&destination, &mut leases)?;
117        } else {
118            lock_table_ancestors(&destination, &mut leases)?;
119        }
120        self.check(old_name)?;
121        let moved = move_read_only_directory(
122            source,
123            &self.identity,
124            &destination,
125            target_snapshot.as_ref(),
126            |staged| ImageHandle::open_miriad(staged).map(drop).map_err(raised),
127        )?;
128        let reopen = || {
129            moved.check_target(&destination)?;
130            let attachment = AttachedImage::open_miriad(&destination)?;
131            moved.check_target(&destination)?;
132            if history {
133                attachment.lock().append_history_at(
134                    "::image::rename",
135                    &[
136                        "Ran ia.rename".into(),
137                        format!("ia.rename(name={name:?}, overwrite={overwrite})"),
138                        format!("original name was {}", old_name.display()),
139                    ],
140                )?;
141            }
142            Ok::<_, PyErr>(attachment)
143        };
144        let attachment = match reopen() {
145            Ok(attachment) => attachment,
146            Err(error) => return moved.restore(source, &destination, error),
147        };
148        moved.finish(&destination)?;
149        Ok(attachment)
150    }
151}
