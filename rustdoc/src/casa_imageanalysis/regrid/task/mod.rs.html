46impl TaskRegridOptions<'_> {
47    fn force(&self) -> bool {
48        self.image_tool.is_some_and(|options| options.force)
49    }
50
51    fn convert_reference(&self) -> bool {
52        self.image_tool
53            .is_none_or(|options| options.convert_reference)
54    }

57pub(crate) fn regrid_task<T: RegridPixel>(
58    input: &dyn ImageInterface<T>,
59    template: &RegridTemplate,
60    options: TaskRegridOptions<'_>,
61) -> Result<Regridded<T>, ImageError> {
62    let input_shape = input.shape();
63    let frame = frame::observation_frame(input.coordinates());
64    let resized = requested_shape(options.shape, input_shape)?;
65    let mut shape = resized.clone().unwrap_or_else(|| input_shape.to_vec());
66    let (all_axes, chosen, after_singleton) =
67        requested_axis_selection(input, template, &options, &shape, resized.is_none())?;
68    let ordered = index_coordinates(input.coordinates());
69    let target_order = index_coordinates(&template.coordinates);
70    let mut result = Regridded {
71        data: input.get()?,
72        mask: input
73            .get_mask_slice(
74                &vec![0; input_shape.len()],
75                input_shape,
76                &vec![1; input_shape.len()],
77            )?
78            .unwrap_or_else(|| ArrayD::from_elem(IxDyn(input_shape).f(), true)),
79        coordinates: input.coordinates().clone(),
80        template_coordinates: false,
81        scale: 1.0,
82    };
83    if !result.mask.iter().any(|&bit| bit) {
84        return Err(invalid("All selected pixels are masked"));
85    }
86    let mut found = false;
87    for (index, (placement, ours)) in ordered.iter().enumerate() {
88        let ours = *ours;
89        let axes: Vec<_> = placement.iter().copied().flatten().collect();
90        let n_axes = axes.len();
91        let Some(&first) = axes.first() else { continue };
92        let target = target_order.iter().find(|(other, model)| {
93            model.coordinate_type() == ours.coordinate_type()
94                && other
95                    .iter()
96                    .map(Option::is_some)
97                    .eq(placement.iter().map(Option::is_some))
98        });
99        let input_lengths: Vec<_> = axes.iter().map(|&axis| input_shape[axis]).collect();
100        let Some((other, theirs)) = target else {
101            regrid_omitted_coordinate(
102                &mut result,
103                input,
104                ours,
105                &axes,
106                &shape,
107                !all_axes && chosen[first],
108                &options,
109            )?;
110            continue;
111        };
112        let theirs = *theirs;
113        let target_lengths: Vec<_> = other
114            .iter()
115            .flatten()
116            .map(|&axis| template.shape[axis])
117            .collect();
118        if let (CoordinateModel::Stokes(ours), CoordinateModel::Stokes(theirs)) = (ours, theirs) {
119            let explicit_axis = !all_axes && chosen[first];
120            let selection =
121                stokes_regrid_request(&options, first, &shape, all_axes, explicit_axis)?;
122            let explicit_length = resized
123                .as_ref()
124                .filter(|_| options.image_tool.is_none_or(|tool| tool.shape_specified))
125                .map(|resized| resized[first]);
126            shape[first] = regrid_stokes(
127                &mut result,
128                (index, first),
129                (ours, theirs),
130                (shape[first], explicit_length),
131                selection,
132                explicit_axis,
133            )?;
134            continue;
135        }
136        let selected: Vec<_> = axes
137            .iter()
138            .copied()
139            .enumerate()
140            .map(|(i, axis)| {
141                let spectral = ours.coordinate_type() == CoordinateType::Spectral;
142                let selected = if all_axes || (after_singleton && !spectral) {
143                    input_lengths[i] > 1 && target_lengths[i] > 1
144                } else {
145                    chosen[axis]
146                };
147                selected && (resized.is_some() || !spectral || target_lengths[i] > 1)
148            })
149            .collect();
150        if resized.is_none() {
151            for (i, &axis) in axes.iter().enumerate() {
152                if selected[i] {
153                    shape[axis] = target_lengths[i];
154                }
155            }
156        }
157        if (options.image_tool.is_some() || !selected.contains(&true))
158            && axes
159                .iter()
160                .zip(&selected)
161                .any(|(&axis, &regrid)| !regrid && shape[axis] != input_shape[axis])
162        {
163            return Err(invalid(
164                "an axis not being regridded must keep its input length",
165            ));
166        }
167        if !selected.contains(&true) {
168            continue;
169        }
170        found = true;
171        let lengths: Vec<_> = axes.iter().map(|&axis| shape[axis]).collect();
172        if n_axes != ours.n_pixel_axes() && ours.coordinate_type() != CoordinateType::Linear {
173            return Err(invalid(
174                "cannot regrid a coordinate with a removed pixel axis",
175            ));
176        }
177        let coordinate = match (ours, theirs) {
178            (CoordinateModel::Direction(ours), CoordinateModel::Direction(theirs))
179                if n_axes == 2 =>
180            {
181                regrid_direction_coordinate(
182                    &mut result,
183                    [axes[0], axes[1]],
184                    input.units(),
185                    (ours, theirs),
186                    (
187                        [lengths[0], lengths[1]],
188                        &selected,
189                        all_axes || after_singleton,
190                    ),
191                    (&options, &frame),
192                )?
193            }
194            (CoordinateModel::Direction(ours), CoordinateModel::Direction(theirs))
195                if n_axes == 1 && input_lengths == lengths && ours == theirs =>
196            {
197                CoordinateModel::Direction(theirs.clone())
198            }
199            (CoordinateModel::Linear(ours), CoordinateModel::Linear(theirs)) => {
200                regrid_linear_coordinate(
201                    &mut result,
202                    input.units(),
203                    placement,
204                    (&input_lengths, &lengths),
205                    (ours, theirs),
206                    &selected,
207                    &options,
208                )?
209            }
210            (CoordinateModel::Spectral(ours), CoordinateModel::Spectral(theirs)) => {
211                regrid_spectral_coordinate(
212                    &mut result,
213                    input,
214                    first,
215                    (ours, theirs),
216                    [input_lengths[0], lengths[0]],
217                    &options,
218                    &frame,
219                )?
220            }
221            _ if n_axes == 1 => regrid_one_axis(
222                &mut result,
223                first,
224                [input_lengths[0], lengths[0]],
225                (ours, theirs),
226                &options,
227            ),
228            _ => {
229                return Err(invalid(
230                    "regridding this multi-axis coordinate is unsupported (#977)",
231                ));
232            }
233        };
234        result.coordinates.replace_coordinate(index, coordinate);
235    }
236    if options.image_tool.is_none() && all_axes && !found {
237        return Err(invalid("Found no axes to regrid"));
238    }
239    if !result.mask.iter().any(|&bit| bit) {
240        return Err(invalid("All output pixels are masked"));
241    }
242    Ok(result)
243}

245fn regrid_direction_coordinate<T: RegridPixel>(
246    result: &mut Regridded<T>,
247    axes: [usize; 2],
248    units: &str,
249    (input, target): (&DirectionCoordinate, &DirectionCoordinate),
250    (shape, selected, all_axes): ([usize; 2], &[bool], bool),
251    (options, frame): (&TaskRegridOptions<'_>, &MeasFrame),
252) -> Result<CoordinateModel, ImageError> {
253    if options.image_tool.is_some() && shape == [1, 1] {
254        return Err(invalid(
255            "cannot regrid a direction coordinate with shape [1, 1]",
256        ));
257    }
258    validate_decimation(options.decimate, shape, selected, all_axes)?;
259    if options.force()
260        || axes.map(|axis| result.data.shape()[axis]) != shape
261        || !same_direction_grid(input, target)
262    {
263        apply_direction(result, axes, units, (input, target), shape, options, frame)?;
264    }
265    Ok(CoordinateModel::Direction(target.clone()))
266}

268fn regrid_one_axis<T: RegridPixel>(
269    result: &mut Regridded<T>,
270    axis: usize,
271    [input_length, length]: [usize; 2],
272    (input, target): (&CoordinateModel, &CoordinateModel),
273    options: &TaskRegridOptions<'_>,
274) -> CoordinateModel {
275    let positions = (0..length)
276        .map(|pixel| {
277            if options.replicate {
278                Some(replicated_position(pixel, input_length, length))
279            } else {
280                target
281                    .to_world(&[pixel as f64])
282                    .and_then(|world| input.to_pixel(&world))
283                    .ok()
284                    .and_then(|pixel| pixel.first().copied())
285                    .filter(|pixel| pixel.is_finite())
286            }
287        })
288        .collect();
289    apply_one_axis(result, axis, length, positions, options.method);
290    target.clone()
291}

293fn regrid_omitted_coordinate<T: RegridPixel>(
294    result: &mut Regridded<T>,
295    input: &dyn ImageInterface<T>,
296    coordinate: &CoordinateModel,
297    axes: &[usize],
298    output_shape: &[usize],
299    explicitly_selected: bool,
300    options: &TaskRegridOptions<'_>,
301) -> Result<(), ImageError> {
302    let input_shape = input.shape();
303    if explicitly_selected {
304        match (options.image_tool, coordinate) {
305            (None, CoordinateModel::Spectral(_)) => {
306                return Err(invalid(
307                    "cannot regrid a spectral axis missing from the template",
308                ));
309            }
310            (Some(_), CoordinateModel::Stokes(_)) => {
311                return Err(invalid(
312                    "cannot regrid a Stokes axis missing from the template",
313                ));
314            }
315            (Some(_), CoordinateModel::Spectral(_)) => require_single_regrid_beam(input)?,
316            _ => {}
317        }
318    }
319    if axes
320        .iter()
321        .any(|&axis| output_shape[axis] != input_shape[axis])
322    {
323        return Err(invalid(
324            "an axis missing from the template must keep its input length",
325        ));
326    }
327    if explicitly_selected
328        && options.force()
329        && let CoordinateModel::Spectral(spectral) = coordinate
330    {
331        let axis = axes[0];
332        let length = input_shape[axis];
333        if length > 1 {
334            // The input grid and reference are used for forced sampling
335            // when no spectral template was supplied.
336            let positions = spectral_positions(
337                (spectral, spectral),
338                [length, length],
339                options.replicate,
340                None,
341                None,
342            )?;
343            apply_one_axis(result, axis, length, positions, options.method);
344        }
345    }
346    Ok(())
347}

349fn require_single_regrid_beam<T: ImagePixel>(
350    input: &dyn ImageInterface<T>,
351) -> Result<(), ImageError> {
352    if input.image_info()?.beam_set.is_multi() {
353        return Err(invalid(
354            "an image with multiple beams cannot be regridded along the spectral axis",
355        ));
356    }
357    Ok(())
358}

360fn use_default_axes<T: ImagePixel>(
361    source: &dyn ImageInterface<T>,
362    target: &RegridTemplate,
363    axes: &[i64],
364    infer_shape: bool,
365) -> bool {
366    if axes.first().is_none_or(|&axis| axis < 0) {
367        return true;
368    }
369    let Some(source_spectral) = source.find_axis(CoordinateType::Spectral) else {
370        return false;
371    };
372    let target_spectral = target
373        .coordinates
374        .find_coordinate(CoordinateType::Spectral)
375        .and_then(|index| target.coordinates.coordinate_pixel_axes(index))
376        .and_then(|axes| axes.first().copied().flatten());
377    infer_shape
378        && axes == [source_spectral as i64]
379        && target_spectral.is_some_and(|axis| target.shape[axis] == 1)
380}

382fn requested_axis_selection<T: ImagePixel>(
383    input: &dyn ImageInterface<T>,
384    template: &RegridTemplate,
385    options: &TaskRegridOptions<'_>,
386    shape: &[usize],
387    infer_shape: bool,
388) -> Result<(bool, Vec<bool>, bool), ImageError> {
389    let all_axes = if options.image_tool.is_some() {
390        options.axes.is_empty() || options.axes == [-1]
391    } else {
392        use_default_axes(input, template, options.axes, infer_shape)
393    };
394    let chosen = requested_axes(
395        options.axes,
396        all_axes,
397        input.ndim(),
398        options.image_tool.is_some(),
399    )?;
400    let after_singleton = input
401        .find_axis(CoordinateType::Spectral)
402        .is_some_and(|axis| {
403            options.axes == [axis as i64] && input.shape()[axis] == 1 && shape[axis] > 1
404        });
405    Ok((all_axes, chosen, after_singleton))
406}

408fn regrid_linear_coordinate<T: RegridPixel>(
409    result: &mut Regridded<T>,
410    units: &str,
411    placement: &[Option<usize>],
412    (input_lengths, lengths): (&[usize], &[usize]),
413    (input, target): (&LinearCoordinate, &LinearCoordinate),
414    selected: &[bool],
415    options: &TaskRegridOptions<'_>,
416) -> Result<CoordinateModel, ImageError> {
417    if let [Some(x), Some(y)] = placement {
418        if options.decimate < 0 {
419            return Err(invalid("decimate must be nonnegative"));
420        }
421        if options.force() || input_lengths != lengths || !same_linear_grid(input, target) {
422            apply_linear_plane(
423                result,
424                [*x, *y],
425                units,
426                (input, target),
427                [lengths[0], lengths[1]],
428                options,
429            )?;
430        }
431    } else {
432        if input_lengths
433            .iter()
434            .zip(lengths)
435            .zip(selected)
436            .any(|((old, new), selected)| !selected && old != new)
437        {
438            return Err(invalid(
439                "an axis not being regridded must keep its input length",
440            ));
441        }
442        apply_linear_axes(
443            result,
444            placement,
445            lengths,
446            selected,
447            (input, target),
448            options,
449        )?;
450    }
451    Ok(target.clone().into())
452}

454fn regrid_spectral_coordinate<T: RegridPixel>(
455    result: &mut Regridded<T>,
456    input: &dyn ImageInterface<T>,
457    axis: usize,
458    (ours, theirs): (&SpectralCoordinate, &SpectralCoordinate),
459    dimensions: [usize; 2],
460    options: &TaskRegridOptions<'_>,
461    frame: &MeasFrame,
462) -> Result<CoordinateModel, ImageError> {
463    let [old_length, length] = dimensions;
464    if old_length == 1 && length == 1 {
465        return Ok(ours.clone().into());
466    }
467    if length == 1 {
468        return Err(invalid(
469            "a spectral interpolation requires multiple output channels",
470        ));
471    }
472    require_single_regrid_beam(input)?;
473    let (coordinate, velocity) = if options.as_velocity && old_length > 1 {
474        let (coordinate, grids) = velocity_regrid(ours, theirs, dimensions)?;
475        (coordinate, Some(grids))
476    } else {
477        (theirs.clone(), None)
478    };
479    let source: CoordinateModel = ours.clone().into();
480    let target: CoordinateModel = coordinate.clone().into();
481    if options.force()
482        || old_length != length
483        || !same_axes(&source, &target, &[length])
484        || ours.world_frequency_ref() != theirs.world_frequency_ref()
485    {
486        let positions = spectral_positions(
487            (ours, &coordinate),
488            dimensions,
489            options.replicate,
490            velocity,
491            options.convert_reference().then_some(frame),
492        )?;
493        apply_one_axis(result, axis, length, positions, options.method);
494    }
495    Ok(target)
496}

498fn same_linear_grid(input: &LinearCoordinate, output: &LinearCoordinate) -> bool {
499    input.reference_pixel() == output.reference_pixel()
500        && input.reference_value() == output.reference_value()
501        && input.increment() == output.increment()
502        && input.axis_units() == output.axis_units()
503        && input.pc_matrix() == output.pc_matrix()
504}

506fn same_direction_grid(input: &DirectionCoordinate, output: &DirectionCoordinate) -> bool {
507    input.direction_ref() == output.direction_ref()
508        && input.conversion_direction_ref() == output.conversion_direction_ref()
509        && input.projection() == output.projection()
510        && input.pc_matrix() == output.pc_matrix()
511        && input.reference_pixel() == output.reference_pixel()
512        && input.reference_value() == output.reference_value()
513        && input.increment() == output.increment()
514        && input.longpole() == output.longpole()
515        && input.native_pole() == output.native_pole()
516}

518fn stokes_regrid_request(
519    options: &TaskRegridOptions<'_>,
520    axis: usize,
521    shape: &[usize],
522    all_axes: bool,
523    explicit_axis: bool,
524) -> Result<bool, ImageError> {
525    if options.image_tool.is_none() || all_axes || !explicit_axis {
526        return Ok(all_axes || explicit_axis);
527    }
528    let slot = options
529        .axes
530        .iter()
531        .position(|&candidate| candidate == axis as i64)
532        .expect("explicit Stokes axis");
533    // CASA checks the output Stokes slot in the requested axis order and
534    // enables Stokes selection only when that list covers the Stokes index.
535    if slot != axis && shape[slot] != shape[axis] {
536        return Err(invalid("Stokes axis and output shape do not match"));
537    }
538    Ok(options.axes.get(axis).is_some_and(|&value| value > 0))
539}

541fn regrid_stokes<T: RegridPixel>(
542    result: &mut Regridded<T>,
543    (coordinate, axis): (usize, usize),
544    (input, target): (&StokesCoordinate, &StokesCoordinate),
545    (length, explicit_length): (usize, Option<usize>),
546    select: bool,
547    explicit_axis: bool,
548) -> Result<usize, ImageError> {
549    if select && input.stokes().len() > 1 && target.stokes().len() > 1 {
550        select_stokes(result, coordinate, axis, input, target, explicit_length)?;
551    } else if length != result.data.shape()[axis] || explicit_axis {
552        return Err(invalid(
553            "an unchanged Stokes axis must keep its input length",
554        ));
555    }
556    Ok(result.data.shape()[axis])
557}

559fn index_coordinates(
560    system: &casa_coordinates::CoordinateSystem,
561) -> Vec<(Vec<Option<usize>>, &CoordinateModel)> {
562    (0..system.n_coordinates())
563        .map(|index| {
564            (
565                system
566                    .coordinate_pixel_axes(index)
567                    .expect("existing coordinate"),
568                system.coordinate(index),
569            )
570        })
571        .collect()
572}

574fn requested_axes(
575    axes: &[i64],
576    all: bool,
577    ndim: usize,
578    unique: bool,
579) -> Result<Vec<bool>, ImageError> {
580    let mut mask = vec![all; ndim];
581    if !all {
582        for &axis in axes {
583            let axis = usize::try_from(axis)
584                .ok()
585                .and_then(|i| mask.get_mut(i))
586                .ok_or_else(|| invalid("regridding axis is outside the input image"))?;
587            if unique && *axis {
588                return Err(invalid("Specified pixel axes are not unique"));
589            }
590            *axis = true;
591        }
592    }
593    Ok(mask)
594}
595
596fn requested_shape(shape: &[i64], input: &[usize]) -> Result<Option<Vec<usize>>, ImageError> {
597    if shape.is_empty() {
598        return Ok(Some(input.to_vec()));
599    }
600    if shape == [-1] {
601        return Ok(None);
602    }
603    if shape.len() != input.len() {
604        return Err(invalid(
605            "output shape must have the same number of axes as the input image",
606        ));
607    }
608    shape
609        .iter()
610        .map(|&length| {
611            usize::try_from(length)
612                .ok()
613                .filter(|&n| n > 0)
614                .ok_or_else(|| invalid("output shape lengths must be positive"))
615        })
616        .collect::<Result<Vec<_>, _>>()
617        .map(Some)
618}

620fn select_stokes<T: ImagePixel + Copy>(
621    result: &mut Regridded<T>,
622    index: usize,
623    axis: usize,
624    input: &StokesCoordinate,
625    template: &StokesCoordinate,
626    expected: Option<usize>,
627) -> Result<(), ImageError> {
628    let (indices, stokes): (Vec<_>, Vec<_>) = input
629        .stokes()
630        .iter()
631        .copied()
632        .enumerate()
633        .filter(|(_, value)| template.stokes().contains(value))
634        .unzip();
635    if stokes.is_empty() || expected.is_some_and(|n| n != stokes.len()) {
636        return Err(invalid(
637            "output Stokes length must match the common input and template Stokes",
638        ));
639    }
640    if stokes.len() != input.stokes().len() {
641        result.data = result.data.select(Axis(axis), &indices);
642        result.mask = result.mask.select(Axis(axis), &indices);
643        result
644            .coordinates
645            .replace_coordinate(index, StokesCoordinate::new(stokes));
646    }
647    Ok(())
648}

650#[derive(Clone, Copy)]
651struct VelocityGrid {
652    reference: f64,
653    increment: f64,
654    reference_pixel: f64,
655}
656
657fn velocity_regrid(
658    input: &SpectralCoordinate,
659    template: &SpectralCoordinate,
660    dimensions: [usize; 2],
661) -> Result<(SpectralCoordinate, [VelocityGrid; 2]), ImageError> {
662    if input.world_frequency_ref() != template.world_frequency_ref() {
663        return Err(invalid(
664            "velocity regridding requires matching spectral reference frames",
665        ));
666    }
667    let measure_frame = MeasFrame::new();
668    let grid = |coordinate: &SpectralCoordinate, count: usize| -> Result<_, ImageError> {
669        let rest = coordinate.rest_frequency_hz()?;
670        if rest == 0.0 {
671            return Err(invalid(
672                "velocity regridding requires nonzero rest frequencies",
673            ));
674        }
675        let velocity = |pixel| -> Result<f64, ImageError> {
676            let ratio = coordinate.world_frequency_hz(pixel)? / rest;
677            MDoppler::new(ratio, DopplerRef::RATIO)
678                .convert_to(coordinate.velocity_convention(), &measure_frame)
679                .map(|value| value.value())
680                .map_err(|error| invalid(error.to_string()))
681        };
682        let reference_pixel = coordinate.reference_pixel()[0];
683        let first = velocity(0.0)?;
684        let last = velocity((count - 1) as f64)?;
685        let grid = VelocityGrid {
686            reference: velocity(reference_pixel)?,
687            increment: velocity(1.0)? - first,
688            reference_pixel,
689        };
690        Ok((grid, [first.min(last), first.max(last)]))
691    };
692    let (source, input_extent) = grid(input, dimensions[0])?;
693    let (target, output_extent) = grid(template, dimensions[1])?;
694    if !overlapping_ranges(input_extent, output_extent) {
695        return Err(invalid(
696            "input and output velocity coordinates do not overlap",
697        ));
698    }
699    let rest = input.rest_frequency_hz()?;
700    let convention = input.velocity_convention();
701    if rest == template.rest_frequency_hz()?
702        && convention == DopplerRef::RADIO
703        && template.velocity_convention() == convention
704        && template.has_linear_mapping()
705        && template.frequency_ref() == template.world_frequency_ref()
706    {
707        // Radio velocity is affine here. Keep the exact template grid and
708        // the input's rest-frequency inventory.
709        let (pixel, reference, increment) = template.fits_linearized_axis_hz()?;
710        let coordinate = SpectralCoordinate::new(
711            input.world_frequency_ref(),
712            reference,
713            increment,
714            pixel,
715            rest,
716        )
717        .with_name(input.axis_names()[0].clone())
718        .with_linear_transform(template.linear_transform())?
719        .with_rest_frequencies(
720            input
721                .rest_frequencies()
722                .iter()
723                .map(|&rest| input.value_in_hz(rest))
724                .collect::<Result<Vec<_>, _>>()?,
725        )
726        .with_velocity_convention(convention);
727        return Ok((coordinate, [source, target]));
728    }
729    let frequency = |velocity| MDoppler::new(velocity, convention).shift_frequency_hz(rest);
730    let reference = frequency(target.reference);
731    let increment = frequency(target.reference + target.increment) - reference;
732    let coordinate = SpectralCoordinate::new(
733        input.world_frequency_ref(),
734        reference,
735        increment,
736        target.reference_pixel,
737        rest,
738    )
739    .with_velocity_convention(convention);
740    Ok((coordinate, [source, target]))
741}

743fn overlapping_ranges([lo_in, hi_in]: [f64; 2], [lo_out, hi_out]: [f64; 2]) -> bool {
744    let separated = |high: f64, low: f64| low - high > 1e-13 * high.abs().max(low.abs());
745    !separated(hi_in, lo_out) && !separated(hi_out, lo_in)
746}

748fn spectral_positions(
749    (input, target): (&SpectralCoordinate, &SpectralCoordinate),
750    [input_count, output_count]: [usize; 2],
751    replicate: bool,
752    velocity: Option<[VelocityGrid; 2]>,
753    frame: Option<&MeasFrame>,
754) -> Result<Vec<Option<f64>>, ImageError> {
755    if input_count > 1 && velocity.is_none() {
756        let bounds = |coordinate: &SpectralCoordinate, count: usize| {
757            let first = coordinate.world_frequency_hz(0.0)?;
758            let last = coordinate.world_frequency_hz((count - 1) as f64)?;
759            Ok::<_, ImageError>([first.min(last), first.max(last)])
760        };
761        if !overlapping_ranges(bounds(input, input_count)?, bounds(target, output_count)?) {
762            return Err(invalid(
763                "input and output spectral coordinates do not overlap",
764            ));
765        }
766    }
767    (0..output_count)
768        .map(|channel| {
769            if input_count == 1 {
770                return Ok(Some(0.0));
771            }
772            if replicate {
773                return Ok(Some(replicated_position(
774                    channel,
775                    input_count,
776                    output_count,
777                )));
778            }
779            if let Some([source, target]) = velocity {
780                let value =
781                    target.reference + (channel as f64 - target.reference_pixel) * target.increment;
782                let pixel = source.reference_pixel + (value - source.reference) / source.increment;
783                return Ok(pixel.is_finite().then_some(pixel));
784            }
785            let frequency = target.world_frequency_hz(channel as f64)?;
786            let input_frequency = match frame {
787                Some(frame) => MFrequency::new(frequency, target.world_frequency_ref())
788                    .convert_to(input.world_frequency_ref(), frame)
789                    .map_err(|error| invalid(error.to_string()))?
790                    .hz(),
791                None => frequency,
792            };
793            Ok(input
794                .pixel_at_frequency_hz(input_frequency)
795                .ok()
796                .filter(|pixel| pixel.is_finite()))
797        })
798        .collect()
799}

801fn apply_one_axis<T: RegridPixel>(
802    result: &mut Regridded<T>,
803    axis: usize,
804    length: usize,
805    positions: Vec<Option<f64>>,
806    method: RegridMethod,
807) {
808    let mut shape = result.data.shape().to_vec();
809    shape[axis] = length;
810    let plan = SpectralPlan::from_positions(axis, positions, method);
811    (result.data, result.mask) = plan.apply_precise(&result.data, &result.mask, &shape);
812}

814fn validate_decimation(
815    decimate: i64,
816    shape: [usize; 2],
817    selected: &[bool],
818    all: bool,
819) -> Result<(), ImageError> {
820    if decimate < 0 {
821        return Err(invalid("decimate must be nonnegative"));
822    }
823    if decimate > 1 && !all {
824        for (axis, length) in shape.into_iter().enumerate() {
825            if selected[axis] && length / 3 < decimate as usize {
826                return Err(invalid(format!(
827                    "maximum decimate for an axis of length {length} is {}",
828                    length / 3
829                )));
830            }
831        }
832    }
833    Ok(())
834}

836fn replicated_position(output: usize, input_count: usize, output_count: usize) -> f64 {
837    (output as f64 + 0.5) * (input_count as f64 / output_count as f64) - 0.5
838}

840fn apply_direction<T: RegridPixel>(
841    result: &mut Regridded<T>,
842    axes: [usize; 2],
843    units: &str,
844    (input, target): (&DirectionCoordinate, &DirectionCoordinate),
845    shape: [usize; 2],
846    options: &TaskRegridOptions<'_>,
847    frame: &MeasFrame,
848) -> Result<(), ImageError> {
849    let scale = jy_per_pixel_scale(units, input, target);
850    let input_shape = axes.map(|axis| result.data.shape()[axis]);
851    let grid = if options.replicate && options.image_tool.is_none() {
852        replicated_grid(input_shape, shape)
853    } else {
854        let projected = direction_grid(input, target, shape, options, frame)?;
855        if options.image_tool.is_some()
856            && !projected.iter().flatten().any(|pixel| {
857                pixel
858                    .iter()
859                    .zip(input_shape)
860                    .all(|(&value, length)| value > -1. && value < length as f64)
861            })
862        {
863            return Err(invalid(
864                "There is no overlap between the input and output direction grids",
865            ));
866        }
867        if options.replicate {
868            replicated_grid(input_shape, shape)
869        } else {
870            projected
871        }
872    };
873    apply_plane(result, axes, shape, &grid, options.method, scale);
874    Ok(())
875}

877fn direction_grid(
878    input: &DirectionCoordinate,
879    target: &DirectionCoordinate,
880    shape: [usize; 2],
881    options: &TaskRegridOptions<'_>,
882    frame: &MeasFrame,
883) -> Result<Vec<Option<[f64; 2]>>, ImageError> {
884    if options.image_tool.is_some_and(|tool| tool.half_turn) {
885        let reference = input.reference_pixel();
886        // The two grids have the same tangent point and opposite pixel
887        // offsets. Check the projection, then reflect about the exact pivot.
888        return Ok((0..shape[1])
889            .flat_map(|y| {
890                let reference = &reference;
891                (0..shape[0]).map(move |x| {
892                    let pixel = [x as f64, y as f64];
893                    target.pixel_to_world(pixel).ok()?;
894                    Some(std::array::from_fn(|axis| {
895                        2.0 * reference[axis] - pixel[axis]
896                    }))
897                })
898            })
899            .collect());
900    }
901    if options.image_tool.is_some()
902        && input.increment() == target.increment()
903        && input.reference_value() == target.reference_value()
904        && *input.pc_matrix() == *target.pc_matrix()
905        && let Some(affine) = direction_affine(input, target, shape)
906        && affine
907            .source_pixel([0.; 2])
908            .iter()
909            .all(|value| value.fract() == 0.)
910    {
911        // An integral crop or translation on the same tangent plane is exact.
912        // A sky round trip can move the first excluded row across the
913        // interpolation boundary and fill it by extrapolating the edge.
914        return Ok((0..shape[1])
915            .flat_map(|y| {
916                (0..shape[0]).map(move |x| Some(affine.source_pixel([x as f64, y as f64])))
917            })
918            .collect());
919    }
920    // Casacore's inter-coordinate conversion uses native frames, while
921    // toWorldMany and toPixelMany retain each coordinate's conversion layer.
922    let to = input.direction_ref();
923    let from = if options.convert_reference() {
924        target.direction_ref()
925    } else {
926        to
927    };
928    let decimate = options.decimate.max(0) as usize;
929    if decimate <= 1 && from == to {
930        return Ok(positions(input, target, shape));
931    }
932    let project = |pixel| {
933        let world = target.pixel_to_world(pixel).ok()?;
934        let world = frame::convert_angles(world, from, to, frame).ok()?;
935        input
936            .world_to_pixel(world)
937            .ok()
938            .filter(|p| p.iter().all(|x| x.is_finite()))
939    };
940    // Report a missing measures frame before treating projection failures as masks.
941    let world = target.reference_value();
942    frame::convert_angles([world[0], world[1]], from, to, frame)?;
943    Ok(coarse_positions(shape, decimate, project))
944}

946fn replicated_grid(input: [usize; 2], output: [usize; 2]) -> Vec<Option<[f64; 2]>> {
947    (0..output[1])
948        .flat_map(|y| {
949            (0..output[0]).map(move |x| {
950                Some([
951                    replicated_position(x, input[0], output[0]),
952                    replicated_position(y, input[1], output[1]),
953                ])
954            })
955        })
956        .collect()
957}

959fn linear_axis_unit_factor(input: &str, target: &str, axis: usize) -> Result<f64, ImageError> {
960    if input == target {
961        return Ok(1.0);
962    }
963    let factor = || Quantity::new(1.0, target)?.get_value_in(&Unit::new(input)?);
964    factor().map_err(|error| invalid(format!("linear regrid axis {axis}: {error}")))
965}

967fn apply_linear_axes<T: RegridPixel>(
968    result: &mut Regridded<T>,
969    placement: &[Option<usize>],
970    lengths: &[usize],
971    selected: &[bool],
972    (input, target): (&LinearCoordinate, &LinearCoordinate),
973    options: &TaskRegridOptions<'_>,
974) -> Result<(), ImageError> {
975    let mut passes: Vec<_> = placement
976        .iter()
977        .enumerate()
978        .filter_map(|(local, &axis)| axis.map(|axis| (local, axis)))
979        .zip(lengths.iter().zip(selected))
980        .filter_map(|((local, axis), (&length, &selected))| {
981            selected.then_some((local, axis, length))
982        })
983        .collect();
984    passes.sort_by_key(|&(_, axis, _)| {
985        options
986            .axes
987            .iter()
988            .position(|&selected| selected == axis as i64)
989            .unwrap_or(axis)
990    });
991    let input_units = input.axis_units();
992    let target_units = target.axis_units();
993    let mut factors = vec![1.0; placement.len()];
994    let mut target_pixel = target.reference_pixel();
995    let same_grid = same_linear_grid(input, target);
996    for (local, axis, length) in passes {
997        let old_length = result.data.shape()[axis];
998        // casacore aligns the selected axis's units and retains earlier conversions.
999        factors[local] = linear_axis_unit_factor(&input_units[local], &target_units[local], axis)?;
1000        if !options.force() && same_grid && length == old_length {
1001            continue;
1002        }
1003        // The coordinate transform uses the full local model. Keep its other
1004        // axes at their reference pixels for this one-axis pass.
1005        let reference_pixel = target_pixel[local];
1006        let positions = (0..length)
1007            .map(|pixel| {
1008                if options.replicate {
1009                    return Some(replicated_position(pixel, old_length, length));
1010                }
1011                target_pixel[local] = pixel as f64;
1012                let mut world = target.to_world(&target_pixel).ok()?;
1013                world
1014                    .iter_mut()
1015                    .zip(&factors)
1016                    .for_each(|(value, factor)| *value *= factor);
1017                let pixel = input.to_pixel(&world).ok()?[local];
1018                pixel.is_finite().then_some(pixel)
1019            })
1020            .collect();
1021        target_pixel[local] = reference_pixel;
1022        apply_one_axis(result, axis, length, positions, options.method);
1023    }
1024    Ok(())
1025}

1027fn apply_linear_plane<T: RegridPixel>(
1028    result: &mut Regridded<T>,
1029    axes: [usize; 2],
1030    brightness: &str,
1031    (input, target): (&LinearCoordinate, &LinearCoordinate),
1032    shape: [usize; 2],
1033    options: &TaskRegridOptions<'_>,
1034) -> Result<(), ImageError> {
1035    let input_units = input.axis_units();
1036    let target_units = target.axis_units();
1037    let factors = [
1038        linear_axis_unit_factor(&input_units[0], &target_units[0], axes[0])?,
1039        linear_axis_unit_factor(&input_units[1], &target_units[1], axes[1])?,
1040    ];
1041    let grid = if options.replicate {
1042        replicated_grid(axes.map(|axis| result.data.shape()[axis]), shape)
1043    } else {
1044        let project = |pixel: [f64; 2]| {
1045            let mut world = target.to_world(&pixel).ok()?;
1046            world
1047                .iter_mut()
1048                .zip(factors)
1049                .for_each(|(value, factor)| *value *= factor);
1050            let pixel = input.to_pixel(&world).ok()?;
1051            let [x, y] = pixel.as_slice() else {
1052                return None;
1053            };
1054            (x.is_finite() && y.is_finite()).then_some([*x, *y])
1055        };
1056        let decimate = options.decimate.max(0) as usize;
1057        if decimate > 1 {
1058            coarse_positions(shape, decimate, project)
1059        } else {
1060            (0..shape[1])
1061                .flat_map(|y| (0..shape[0]).map(move |x| project([x as f64, y as f64])))
1062                .collect()
1063        }
1064    };
1065    let scale = if brightness.eq_ignore_ascii_case("Jy/pixel") {
1066        let old = input.increment();
1067        let new = target.increment();
1068        (new[0] * factors[0] / old[0] * (new[1] * factors[1] / old[1])).abs()
1069    } else {
1070        1.0
1071    };
1072    apply_plane(result, axes, shape, &grid, options.method, scale);
1073    Ok(())
1074}

1076fn apply_plane<T: RegridPixel>(
1077    result: &mut Regridded<T>,
1078    axes: [usize; 2],
1079    shape: [usize; 2],
1080    positions: &[Option<[f64; 2]>],
1081    method: RegridMethod,
1082    scale: f64,
1083) {
1084    let sampling = DirectionSampling {
1085        axes: [axes; 2],
1086        shape,
1087        positions,
1088        method,
1089        scale,
1090    };
1091    (result.data, result.mask) = sample_direction_planes(
1092        &result.data,
1093        Some(&result.mask),
1094        sampling,
1095        T::COMPONENTS,
1096        T::component,
1097        T::with_component,
1098    );
1099    result.scale = scale;
1100}

1102fn coarse_positions(
1103    shape: [usize; 2],
1104    decimate: usize,
1105    project: impl Fn([f64; 2]) -> Option<[f64; 2]>,
1106) -> Vec<Option<[f64; 2]>> {
1107    let step = if decimate <= 1 {
1108        [1, 1]
1109    } else {
1110        shape
1111            .map(|length| {
1112                let samples = length / decimate;
1113                length
1114                    / if samples > 1 {
1115                        samples - 1
1116                    } else {
1117                        length.saturating_sub(1).max(1)
1118                    }
1119            })
1120            .map(|step| step.max(1))
1121    };
1122    if step == [1, 1] {
1123        return (0..shape[1])
1124            .flat_map(|y| (0..shape[0]).map(move |x| [x as f64, y as f64]))
1125            .map(project)
1126            .collect();
1127    }
1128    let sparse = [shape[0].div_ceil(step[0]), shape[1].div_ceil(step[1])];
1129    let mut projected_x = Vec::with_capacity(sparse[0] * sparse[1]);
1130    let mut projected_y = Vec::with_capacity(projected_x.capacity());
1131    let mut valid = Vec::with_capacity(projected_x.capacity());
1132    for y in (0..shape[1]).step_by(step[1]) {
1133        for x in (0..shape[0]).step_by(step[0]) {
1134            let pixel = project([x as f64, y as f64]);
1135            valid.push(pixel.is_some());
1136            let [x, y] = pixel.unwrap_or([0.0, 0.0]);
1137            projected_x.push(x);
1138            projected_y.push(y);
1139        }
1140    }
1141    let x = Plane {
1142        data: &projected_x,
1143        mask: Some(&valid),
1144        shape: sparse,
1145    };
1146    let y = Plane {
1147        data: &projected_y,
1148        mask: Some(&valid),
1149        shape: sparse,
1150    };
1151    (0..shape[1])
1152        .flat_map(|j| {
1153            (0..shape[0]).map(move |i| [i as f64 / step[0] as f64, j as f64 / step[1] as f64])
1154        })
1155        .map(|p| {
1156            Some([
1157                x.sample(RegridMethod::Linear, p)?,
1158                y.sample(RegridMethod::Linear, p)?,
1159            ])
1160        })
1161        .collect()
1162}
