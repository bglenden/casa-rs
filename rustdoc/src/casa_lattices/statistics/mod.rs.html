2//! Lattice statistics computation.
3//!
4//! Provides [`LatticeStatistics<T>`], the Rust equivalent of C++
5//! `LatticeStatistics<T>` from
6//! `casacore/lattices/LatticeMath/LatticeStatistics.h`.
7//!
8//! The implementation follows the C++ classical algorithm: a single pass
9//! accumulates `npts`, `sum`, `sumsq`, `min`, `max` and the centered
10//! second moment used for variance and standard deviation. Order
11//! statistics use quantile selection rather than full repeated sorts, while
12//! preserving the relevant casacore-style median and fractile semantics.
13//!
14//! # Execution policy
15//!
16//! Rust exposes an explicit execution-policy knob for large workloads via
17//! [`ExecutionPolicy`]. This is intentionally broader than the C++ casacore
18//! API, which largely hides iterator/cache/threading choices behind internal
19//! implementation details.
20//!
21//! The policy controls *how* the lattice is traversed and reduced, not *what*
22//! statistic is returned:
23//!
24//! - [`ExecutionPolicy::Auto`] keeps the existing behavior of choosing a
25//!   strategy from the lattice kind, traversal shape, and work size.
26//! - [`ExecutionPolicy::Serial`] forces a single-threaded pass. This is often
27//!   best for small in-memory arrays where chunking or extra threads would add
28//!   overhead.
29//! - [`ExecutionPolicy::Pipelined`] overlaps chunk reads with reduction work
30//!   using a bounded producer/consumer pipeline. This is mainly useful for
31//!   paged or persistent lattices where tile I/O is a visible cost.
32//! - [`ExecutionPolicy::Parallel`] uses the same producer/read overlap but
33//!   fans computation out across worker threads. This is usually the best
34//!   choice for large compute-heavy workloads.
35//!
36//! Policy selection only affects future cache builds. Once a statistic family
37//! is cached, subsequent calls reuse the cached results regardless of the
38//! policy that produced them.
39//!
40//! # Relationship to C++
41//!
42//! | Rust                                  | C++ equivalent                          |
43//! |---------------------------------------|-----------------------------------------|
44//! | `LatticeStatistics::new`              | `LatticeStatistics::LatticeStatistics`  |
45//! | `LatticeStatistics::set_axes`         | `LatticeStatistics::setAxes`            |
46//! | `LatticeStatistics::set_include_range`| `LatticeStatistics::setInExCludeRange(include, {})` |
47//! | `LatticeStatistics::set_exclude_range`| `LatticeStatistics::setInExCludeRange({}, exclude)` |
48//! | `LatticeStatistics::set_pixel_mask`   | `MaskedLattice::setPixelMask`           |
49//! | `LatticeStatistics::get_statistic`    | `LatticeStatistics::getStatistic`       |
50//! | `LatticeStatistics::get_min_max_pos`  | `LatticeStatistics::getMinMaxPos`       |
51//! | `Statistic` enum                      | `LatticeStatsBase::StatisticsTypes`     |

134/// Statistics quantities that [`LatticeStatistics`] can compute.
135///
136/// Mirrors C++ `LatticeStatsBase::StatisticsTypes`.
137#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
138pub enum Statistic {
139    /// Number of unmasked, unfiltered pixels.
140    Npts,
141    /// Sum of pixel values.
142    Sum,
143    /// Sum of squared pixel values.
144    SumSq,
145    /// Minimum pixel value.
146    Min,
147    /// Maximum pixel value.
148    Max,
149    /// Arithmetic mean.
150    Mean,
151    /// Root-mean-square: `sqrt(sumsq / npts)`.
152    Rms,
153    /// Bessel-corrected sample variance.
154    Variance,
155    /// Bessel-corrected sample standard deviation (`sqrt(variance)`).
156    Sigma,
157    /// Median, averaging the two middle values for even-sized samples.
158    Median,
159    /// First quartile (25th percentile).
160    Q1,
161    /// Third quartile (75th percentile).
162    Q3,
163    /// Interquartile range `Q3 − Q1`.
164    Quartile,
165    /// Median absolute deviation from the median.
166    MedAbsDevMed,
167}

169/// Execution policy for large lattice-statistics traversals.
170///
171/// This is a Rust-specific extension beyond C++ casacore's public
172/// `LatticeStatistics` API. It lets callers choose whether statistics should
173/// run serially, overlap tile I/O with reduction work, or also use worker
174/// threads for compute.
175///
176/// The policy does not change statistical semantics or result shapes; it only
177/// changes the execution strategy used to build the internal caches.
178///
179/// # Variants
180///
181/// - [`Auto`](crate::ExecutionPolicy::Auto): choose a strategy from the lattice type and work
182///   size. This is the default.
183/// - [`Serial`](crate::ExecutionPolicy::Serial): use one thread and the most direct traversal.
184/// - [`Pipelined`](crate::ExecutionPolicy::Pipelined): use one producer and one consumer thread
185///   with a bounded queue. This is mainly useful for paged lattices, where the
186///   next tile can be read while the previous tile is being reduced.
187/// - [`Parallel`](crate::ExecutionPolicy::Parallel): use one producer plus a worker pool. This
188///   combines the same I/O overlap as `Pipelined` with threaded compute.
189///
190/// Explicit policies are validated against the assigned resources and amount
191/// of work. Invalid worker or prefetch requests return an error rather than
192/// silently selecting another execution mode.
193///
194/// # Examples
195///
196/// Force threaded compute for a large in-memory lattice:
197///
198/// ```
199/// use casa_lattices::{ArrayLattice, ExecutionPolicy, LatticeStatistics, Statistic};
200/// use ndarray::{ArrayD, IxDyn};
201///
202/// let workers = std::thread::available_parallelism()
203///     .map(|parallelism| parallelism.get())
204///     .unwrap_or(1)
205///     .min(2);
206/// if workers < 2 {
207///     return;
208/// }
209/// let data = ArrayD::from_shape_fn(IxDyn(&[1024, 1024, 4]), |idx| {
210///     (idx[0] + idx[1] * 1024 + idx[2] * 1024 * 1024) as f32
211/// });
212/// let lattice = ArrayLattice::new(data);
213/// let mut stats = LatticeStatistics::new(&lattice);
214/// stats.set_axes(vec![0, 1]);
215/// stats.set_execution_policy(ExecutionPolicy::Parallel {
216///     workers,
217///     prefetch_depth: 2,
218/// });
219/// let mean = stats.get_statistic(Statistic::Mean).unwrap();
220/// assert_eq!(mean.shape(), &[4]);
221/// ```
222///
223/// For a large paged lattice, overlap-only execution can be a useful middle
224/// ground when tile I/O matters more than raw compute:
225///
226/// ```no_run
227/// use casa_lattices::{ExecutionPolicy, LatticeStatistics, PagedArray, Statistic};
228///
229/// # fn demo(path: &std::path::Path) -> Result<(), Box<dyn std::error::Error>> {
230/// let lattice = PagedArray::<f32>::open(path)?;
231/// let mut stats = LatticeStatistics::new(&lattice);
232/// stats.set_axes(vec![0, 1]);
233/// stats.set_execution_policy(ExecutionPolicy::Pipelined { prefetch_depth: 4 });
234/// let sigma = stats.get_statistic(Statistic::Sigma)?;
235/// println!("first plane sigma = {}", sigma[[0]]);
236/// # Ok(())
237/// # }
238/// ```
239/// Computes statistics over an N-dimensional lattice.
240///
241/// Mirrors C++ `LatticeStatistics<T>` from
242/// `casacore/lattices/LatticeMath/LatticeStatistics.h`.
243///
244/// By default statistics are computed over all pixels in the lattice.
245/// Call [`set_axes`](Self::set_axes) to restrict the collapse to a subset of
246/// dimensions; the result array is then indexed by the remaining axes.
247///
248/// Results are cached internally. The first request for any basic statistic
249/// computes the whole basic family (`npts`, `sum`, `sumsq`, `min`, `max`,
250/// `mean`, `rms`, `variance`, `sigma`) in one traversal. Likewise, the first
251/// request for any order statistic computes the whole order-stat family in one
252/// traversal and reuses it across subsequent calls.
253///
254/// The execution strategy used to build those caches is controlled by
255/// [`ExecutionPolicy`]. Changing the execution policy invalidates the existing
256/// caches so the next request rebuilds them with the new strategy.
257pub struct LatticeStatistics<'a, T> {
258    lattice: &'a dyn Lattice<T>,
259    axes: Option<Vec<usize>>,
260    include_range: Option<[f64; 2]>,
261    exclude_range: Option<[f64; 2]>,
262    mask: Option<ArrayD<bool>>,
263    execution_policy: ExecutionPolicy,
264    execution_resources: ExecutionResources,
265    cache_storage: TempStoragePolicy,
266    basic_cache: RefCell<Option<BasicStatsCache>>,
267    order_cache: RefCell<Option<OrderStatsCache>>,
268}

382    /// Returns an `ArrayD<f64>` with shape `[1]` for global statistics, or the

384    pub fn get_statistic(&self, stat: Statistic) -> Result<ArrayD<f64>, LatticeError> {
385        if stat.is_order_stat() {
386            self.ensure_order_cache()?;
387            self.order_cache

391                .get(stat)
392        } else {
393            self.ensure_basic_cache()?;
394            self.basic_cache
395                .borrow()
396                .as_ref()
397                .expect("basic cache initialized")
398                .get(stat)
399        }
400    }

478    fn build_global_basic_cache(&self) -> Result<BasicStatsCache, LatticeError> {
479        let mask_strides = self.mask.as_ref().and_then(|mask| {
480            mask.as_slice_memory_order()
481                .map(|_| c_order_strides(self.lattice.shape()))
482        });
483        let filters = self.filter_state(mask_strides.as_deref());
484
485        let strategy = self.basic_execution_strategy(std::mem::size_of::<GlobalBasicPartial>())?;
486        let partial = try_reduce_read_chunks(
487            self.lattice,
488            self.traversal_spec_for_strategy(strategy),
489            strategy,
490            GlobalBasicPartial::default,
491            |partial, chunk| accumulate_global_basic_chunk(partial, &chunk, &filters),
492            |partial, other| {
493                partial.merge(other);
494                Ok(())
495            },
496        )?;
497
498        if partial.accum.npts == 0 {
499            return BasicStatsCache::from_accumulators(&[0], &[], None, None, &self.cache_storage);
500        }
501
502        BasicStatsCache::from_accumulators(
503            &[1],
504            &[partial.accum],
505            partial.min_pos,
506            partial.max_pos,
507            &self.cache_storage,
508        )
509    }

511    fn build_axis_basic_cache(&self, layout: &AxisLayout) -> Result<BasicStatsCache, LatticeError> {
512        let mask_strides = self.mask.as_ref().and_then(|mask| {
513            mask.as_slice_memory_order()
514                .map(|_| c_order_strides(self.lattice.shape()))
515        });
516        let filters = self.filter_state(mask_strides.as_deref());
517        let per_worker_state_bytes = layout
518            .n_out
519            .saturating_mul(std::mem::size_of::<RunningStats>());
520        let strategy = self.basic_execution_strategy(per_worker_state_bytes)?;
521        let accumulators = try_reduce_read_chunks(
522            self.lattice,
523            self.traversal_spec_for_strategy(strategy),
524            strategy,
525            || vec![RunningStats::default(); layout.n_out],
526            |accumulators, chunk| {
527                accumulate_axis_basic_chunk(accumulators, layout, &chunk, &filters)
528            },
529            |accumulators, other| {
530                merge_running_stats(accumulators, other);
531                Ok(())
532            },
533        )?;
534
535        BasicStatsCache::from_accumulators(
536            &layout.out_shape,
537            &accumulators,
538            None,
539            None,
540            &self.cache_storage,
541        )
542    }

544    fn build_global_order_cache(&self) -> Result<OrderStatsCache, LatticeError> {
545        let mask_strides = self.mask.as_ref().and_then(|mask| {
546            mask.as_slice_memory_order()
547                .map(|_| c_order_strides(self.lattice.shape()))
548        });
549        let filters = self.filter_state(mask_strides.as_deref());
550        let reserve = global_order_reserve(self.lattice.nelements(), self.has_sparse_filter());
551        let strategy = self.order_execution_strategy(
552            reserve
553                .checked_mul(std::mem::size_of::<f64>())
554                .ok_or_else(|| {
555                    LatticeError::InvalidTraversal("statistics state byte overflow".into())
556                })?,
557        )?;
558        let mut values = try_reduce_read_chunks(
559            self.lattice,
560            self.traversal_spec_for_strategy(strategy),
561            strategy,
562            || Vec::with_capacity(reserve),
563            |values, chunk| accumulate_global_order_chunk(values, &chunk, &filters),
564            |values, other| {
565                values.extend(other);
566                Ok(())
567            },
568        )?;
569
570        if values.is_empty() {
571            return OrderStatsCache::from_summaries(&[0], &[], &self.cache_storage);
572        }
573
574        let summary = compute_order_summary(&mut values);
575        OrderStatsCache::from_summaries(&[1], &[summary], &self.cache_storage)
576    }
577
578    fn build_axis_order_cache(&self, layout: &AxisLayout) -> Result<OrderStatsCache, LatticeError> {
579        let mask_strides = self.mask.as_ref().and_then(|mask| {
580            mask.as_slice_memory_order()
581                .map(|_| c_order_strides(self.lattice.shape()))
582        });
583        let filters = self.filter_state(mask_strides.as_deref());
584        let reserve_per_bucket = axis_order_reserve(
585            layout.collapsed_size,
586            self.has_sparse_filter(),
587            layout.n_out,
588        );
589        let per_worker_state_bytes = layout
590            .n_out
591            .checked_mul(reserve_per_bucket)
592            .and_then(|value| value.checked_mul(std::mem::size_of::<f64>()))
593            .ok_or_else(|| {
594                LatticeError::InvalidTraversal("statistics state byte overflow".into())
595            })?;
596        let strategy = self.order_execution_strategy(per_worker_state_bytes)?;
597        let buckets = try_reduce_read_chunks(
598            self.lattice,
599            self.traversal_spec_for_strategy(strategy),
600            strategy,
601            || {
602                (0..layout.n_out)
603                    .map(|_| Vec::with_capacity(reserve_per_bucket))
604                    .collect::<Vec<_>>()
605            },
606            |buckets, chunk| accumulate_axis_order_chunk(buckets, layout, &chunk, &filters),
607            |buckets, other| {
608                merge_order_buckets(buckets, other);
609                Ok(())
610            },
611        )?;
612
613        let mut summaries = Vec::with_capacity(layout.n_out);
614        for mut values in buckets {
615            if values.is_empty() {
616                summaries.push(OrderStatsSummary::nan());
617            } else {
618                summaries.push(compute_order_summary(&mut values));
619            }
620        }
621
622        OrderStatsCache::from_summaries(&layout.out_shape, &summaries, &self.cache_storage)
623    }

625    fn execution_strategy(
626        &self,
627        per_worker_state_bytes: usize,
628    ) -> Result<ReadChunkExecutionStrategy, LatticeError> {
629        let available = thread_parallelism();
630        let task_count = self.chunked_task_count();
631        let cursor_shape = if self.lattice.is_paged() || self.lattice.is_persistent() {
632            self.lattice.nice_cursor_shape()
633        } else {
634            self.parallel_cursor_shape()
635        };
636        let chunk_elements = cursor_shape
637            .iter()
638            .try_fold(1usize, |product, &extent| product.checked_mul(extent));
639        let chunk_bytes = chunk_elements
640            .and_then(|elements| elements.checked_mul(std::mem::size_of::<T>()))
641            .ok_or_else(|| {
642                LatticeError::InvalidTraversal("statistics chunk byte overflow".into())
643            })?;
644        let persistent = self.lattice.is_paged() || self.lattice.is_persistent();
645        let plan = plan_execution(
646            self.execution_policy,
647            ExecutionInputs {
648                task_count,
649                chunk_bytes,
650                per_worker_state_bytes,
651                memory_budget_bytes: self.execution_resources.memory_budget_bytes,
652                available_workers: available,
653                requested_worker_limit: self.execution_resources.worker_limit,
654                source_residency: if persistent {
655                    SourceResidency::Persistent
656                } else {
657                    SourceResidency::Resident
658                },
659                prefetch_capability: true,
660                configured_prefetch_cap: self.execution_resources.prefetch_cap,
661            },
662        )
663        .map_err(|error| LatticeError::InvalidTraversal(error.to_string()))?;
664        Ok(match plan.mode {
665            ExecutionMode::Serial => ReadChunkExecutionStrategy::Serial,
666            ExecutionMode::Pipelined => {
667                ReadChunkExecutionStrategy::Pipelined(PipelinedReadChunkConfig {
668                    prefetch_depth: plan.prefetch_depth,
669                })
670            }
671            ExecutionMode::Parallel => {
672                ReadChunkExecutionStrategy::Parallel(ParallelReadChunkConfig {
673                    workers: plan.workers,
674                    prefetch_depth: plan.prefetch_depth,
675                })
676            }
677        })
678    }
679
680    fn traversal_spec_for_strategy(&self, strategy: ReadChunkExecutionStrategy) -> TraversalSpec {
681        if self.lattice.is_paged() || self.lattice.is_persistent() {
682            TraversalSpec::tiles()
683        } else if matches!(strategy, ReadChunkExecutionStrategy::Serial) {
684            TraversalSpec::chunks(self.lattice.shape().to_vec())
685        } else {
686            TraversalSpec::chunks(self.parallel_cursor_shape())
687        }
688    }

690    fn parallel_cursor_shape(&self) -> Vec<usize> {
691        let full_shape = self.lattice.shape();
692        self.lattice
693            .nice_cursor_shape()
694            .into_iter()
695            .zip(full_shape.iter())
696            .map(|(cursor, &extent)| cursor.clamp(1, extent.max(1)))
697            .collect()
698    }

700    fn chunked_task_count(&self) -> usize {
701        let full_shape = self.lattice.shape().to_vec();
702        let cursor_shape = if self.lattice.is_paged() || self.lattice.is_persistent() {
703            self.lattice.nice_cursor_shape()
704        } else {
705            self.parallel_cursor_shape()
706        };
707        let spec = if self.lattice.is_paged() || self.lattice.is_persistent() {
708            TraversalSpec::tiles()
709        } else {
710            TraversalSpec::chunks(cursor_shape.clone())
711        };
712        TraversalCursorIter::new(full_shape, cursor_shape, spec)
713            .size_hint()
714            .1
715            .unwrap_or(0)
716    }

718    fn axis_layout(&self, axes: &[usize]) -> Result<Option<AxisLayout>, LatticeError> {
719        let axes = self.validated_axes(axes)?;
720        let shape = self.lattice.shape();
721        let ndim = shape.len();
722        let out_axes: Vec<usize> = (0..ndim).filter(|axis| !axes.contains(axis)).collect();
723        if out_axes.is_empty() {
724            return Ok(None);
725        }
726
727        let out_shape: Vec<usize> = out_axes.iter().map(|&axis| shape[axis]).collect();
728        let collapsed_size = axes.iter().map(|&axis| shape[axis]).product();
729        Ok(Some(AxisLayout {
730            out_axes,
731            out_shape: out_shape.clone(),
732            out_strides: c_order_strides(&out_shape),
733            n_out: out_shape.iter().product(),
734            collapsed_size,
735        }))
736    }
737
738    fn validated_axes(&self, axes: &[usize]) -> Result<Vec<usize>, LatticeError> {
739        let ndim = self.lattice.ndim();
740        let mut seen = vec![false; ndim];
741        let mut normalized = Vec::with_capacity(axes.len());
742
743        for &axis in axes {
744            if axis >= ndim {
745                return Err(LatticeError::IndexOutOfBounds {
746                    index: vec![axis],
747                    shape: vec![ndim],
748                });
749            }
750            if seen[axis] {
751                return Err(LatticeError::InvalidTraversal(format!(
752                    "statistics axes contain duplicate axis {axis}"
753                )));
754            }
755            seen[axis] = true;
756            normalized.push(axis);
757        }
758
759        Ok(normalized)
760    }

776fn accumulate_global_basic_chunk<T: StatsElement>(
777    partial: &mut GlobalBasicPartial,
778    chunk: &ChunkTask<T>,
779    filters: &FilterState<'_>,
780) -> Result<(), LatticeError> {
781    let compare_ties = !chunk.data.view().reversed_axes().is_standard_layout();
782    for_each_accepted_value_in_chunk(chunk, filters, |idx, value| {
783        if partial.accum.npts == 0
784            || value < partial.accum.min
785            || (compare_ties
786                && value == partial.accum.min
787                && earlier_pixel(idx, partial.min_pos.as_deref()))
788        {
789            partial.min_pos = Some(idx.to_vec());
790        }
791        if partial.accum.npts == 0
792            || value > partial.accum.max
793            || (compare_ties
794                && value == partial.accum.max
795                && earlier_pixel(idx, partial.max_pos.as_deref()))
796        {
797            partial.max_pos = Some(idx.to_vec());
798        }
799        partial.accum.push(value);
800    })
801}

803fn earlier_pixel(pixel: &[usize], previous: Option<&[usize]>) -> bool {
804    // casacore visits pixels with axis zero varying fastest.
805    previous.is_none_or(|previous| pixel.iter().rev().lt(previous.iter().rev()))
806}

840fn for_each_accepted_value_in_chunk<T: StatsElement>(
841    chunk: &ChunkTask<T>,
842    filters: &FilterState<'_>,
843    mut f: impl FnMut(&[usize], f64),
844) -> Result<(), LatticeError> {
845    let origin = &chunk.cursor.position;
846
847    if let Some(values) = chunk.data.as_slice_memory_order() {
848        let axis_order = memory_order_axis_path(chunk.data.strides());
849        let mut full_idx = origin.clone();
850        let last = values.len().saturating_sub(1);
851        for (i, value) in values.iter().enumerate() {
852            let fv = value.to_f64_stats();
853            if accept_idx_fast(filters, fv, &full_idx) {
854                f(&full_idx, fv);
855            }
856            if i != last {
857                advance_position_in_axis_order(
858                    &mut full_idx,
859                    origin,
860                    &chunk.cursor.shape,
861                    &axis_order,
862                );
863            }
864        }
865        return Ok(());
866    }
867
868    let mut full_idx = origin.clone();
869    for (local_idx, value) in chunk.data.indexed_iter() {
870        write_global_idx(&mut full_idx, origin, local_idx.slice());
871        let fv = value.to_f64_stats();
872        if accept_idx_fast(filters, fv, &full_idx) {
873            f(&full_idx, fv);
874        }
875    }
876    Ok(())
877}
878
879fn accept_idx_fast(filters: &FilterState<'_>, value: f64, idx: &[usize]) -> bool {
880    if let Some(mask) = filters.mask {
881        let accept = if let (Some(mask_slice), Some(mask_strides)) =
882            (filters.mask_slice, filters.mask_strides)
883        {
884            let flat = flat_index(idx, mask_strides);
885            mask_slice[flat]
886        } else {
887            mask[IxDyn(idx)]
888        };
889        if !accept {
890            return false;
891        }
892    }
893
894    accept_value(filters, value)
895}
896
897fn accept_value(filters: &FilterState<'_>, value: f64) -> bool {
898    if let Some([lo, hi]) = filters.include_range
899        && !(lo <= value && value <= hi)
900    {
901        return false;
902    }
903    if let Some([lo, hi]) = filters.exclude_range
904        && value >= lo
905        && value <= hi
906    {
907        return false;
908    }
909    true
910}

930/// Classical sample moments and extrema accumulated without retaining data.
931#[derive(Clone, Copy, Debug)]
932pub struct RunningStats {
933    npts: usize,
934    sum: f64,
935    sumsq: f64,
936    origin: f64,
937    mean: f64,
938    centered_sumsq: f64,
939    min: f64,
940    max: f64,
941}
942
943impl Default for RunningStats {
944    fn default() -> Self {
945        Self {
946            npts: 0,
947            sum: 0.0,
948            sumsq: 0.0,
949            origin: 0.0,
950            mean: 0.0,
951            centered_sumsq: 0.0,
952            min: f64::INFINITY,
953            max: f64::NEG_INFINITY,
954        }
955    }
956}
957
958impl RunningStats {
959    /// Add one valid, unweighted sample.
960    pub fn push(&mut self, value: f64) {
961        if self.npts == 0 {
962            self.origin = value;
963        }
964        self.npts += 1;
965        self.sum += value;
966        self.sumsq += value * value;
967        // Keep the running mean relative to the first sample so adjustments
968        // smaller than the spacing of the original values are retained.
969        let shifted = value - self.origin;
970        let delta = shifted - self.mean;
971        self.mean += delta / self.npts as f64;
972        self.centered_sumsq += delta * (shifted - self.mean);
973        self.min = self.min.min(value);
974        self.max = self.max.max(value);
975    }
976
977    /// Return a moment, count, or extremum for a nonempty sample.
978    /// An empty sample or a statistic that depends on order returns `None`.
979    pub fn statistic(self, stat: Statistic) -> Option<f64> {
980        match stat {
981            Statistic::Median
982            | Statistic::Q1
983            | Statistic::Q3
984            | Statistic::Quartile
985            | Statistic::MedAbsDevMed => None,
986            _ => (self.npts > 0).then(|| self.finish(stat)),
987        }
988    }
989
990    fn merge(&mut self, other: Self) {
991        if other.npts == 0 {
992            return;
993        }
994        if self.npts == 0 {
995            *self = other;
996            return;
997        }
998        let first_count = self.npts as f64;
999        let second_count = other.npts as f64;
1000        self.npts += other.npts;
1001        let delta = (other.origin - self.origin) + (other.mean - self.mean);
1002        let second_weight = second_count / self.npts as f64;
1003        self.mean += delta * second_weight;
1004        self.centered_sumsq += other.centered_sumsq + delta * delta * first_count * second_weight;
1005        self.sum += other.sum;
1006        self.sumsq += other.sumsq;
1007        self.min = self.min.min(other.min);
1008        self.max = self.max.max(other.max);
1009    }
1010
1011    pub(crate) fn finish(self, stat: Statistic) -> f64 {
1012        debug_assert!(self.npts > 0);
1013        let n = self.npts as f64;
1014        match stat {
1015            Statistic::Npts => n,
1016            Statistic::Sum => self.sum,
1017            Statistic::SumSq => self.sumsq,
1018            Statistic::Min => self.min,
1019            Statistic::Max => self.max,
1020            Statistic::Mean => self.sum / n,
1021            Statistic::Rms => (self.sumsq / n).sqrt(),
1022            Statistic::Variance => {
1023                // casacore's `StatisticsUtilities::combine` gives a
1024                // zero variance, not an undefined one, below two
1025                // points.
1026                if self.npts < 2 {
1027                    0.0
1028                } else {
1029                    self.centered_sumsq / (n - 1.0)
1030                }
1031            }
1032            Statistic::Sigma => {
1033                if self.npts < 2 {
1034                    0.0
1035                } else {
1036                    self.finish(Statistic::Variance).sqrt()
1037                }
1038            }
1039            _ => unreachable!("order statistics require retained values"),
1040        }
1041    }
1042}

1103fn advance_position_in_axis_order(
1104    full_idx: &mut [usize],
1105    origin: &[usize],
1106    shape: &[usize],
1107    axis_order: &[usize],
1108) {
1109    for (order_pos, &axis) in axis_order.iter().enumerate() {
1110        let limit = origin[axis] + shape[axis];
1111        if full_idx[axis] + 1 < limit {
1112            full_idx[axis] += 1;
1113            for &reset_axis in axis_order.iter().take(order_pos) {
1114                full_idx[reset_axis] = origin[reset_axis];
1115            }
1116            return;
1117        }
1118    }
1119}

1133pub(crate) fn c_order_strides(shape: &[usize]) -> Vec<usize> {
1134    let mut strides = vec![1; shape.len()];
1135    let mut stride = 1usize;
1136    for axis in (0..shape.len()).rev() {

1168/// Compute a single statistic from a non-empty slice of accepted pixel values.
1169#[allow(dead_code)]
1170pub(crate) fn compute_statistic(values: &[f64], stat: Statistic) -> f64 {
1171    debug_assert!(!values.is_empty());
1172
1173    if stat.is_order_stat() {
1174        let summary = compute_order_summary(&mut values.to_vec());
1175        return match stat {
1176            Statistic::Median => summary.median,
1177            Statistic::Q1 => summary.q1,
1178            Statistic::Q3 => summary.q3,
1179            Statistic::Quartile => summary.quartile,
1180            Statistic::MedAbsDevMed => summary.mad,
1181            _ => unreachable!("order-stat dispatch"),
1182        };
1183    }
1184
1185    let mut accum = RunningStats::default();
1186    for &value in values {
1187        accum.push(value);
1188    }
1189    accum.finish(stat)
1190}

1208fn legacy_order_summary(values: &[f64]) -> OrderStatsSummary {

1211    let q3 = casacore_fractile(values, 0.75);

1221/// Median matching C++ `casacore::median(arr, takeEvenMean=true)`.
1222pub(crate) fn casacore_median(values: &[f64]) -> f64 {
1223    let mut sorted = values.to_vec();
1224    sorted.sort_by(nan_safe_cmp);
1225    let n = sorted.len();
1226    let n2 = (n - 1) / 2;
1227    if n.is_multiple_of(2) {
1228        (sorted[n2] + sorted[n2 + 1]) / 2.0
1229    } else {
1230        sorted[n2]
1231    }
1232}
1233
1234/// Fractile (percentile) matching C++ `casacore::fractile(arr, fraction)`.
1235pub(crate) fn casacore_fractile(values: &[f64], frac: f64) -> f64 {
1236    let mut sorted = values.to_vec();
1237    sorted.sort_by(nan_safe_cmp);
1238    let n = sorted.len();
1239    let n2 = ((n - 1) as f64 * frac + 0.01) as usize;
1240    sorted[n2.min(n - 1)]
1241}
1242
1243/// Median absolute deviation from the median, matching C++ `casacore::madfm`.
1244pub(crate) fn casacore_madfm(values: &[f64]) -> f64 {
1245    let med = casacore_median(values);
1246    let mut devs: Vec<f64> = values.iter().map(|value| (value - med).abs()).collect();
1247    devs.sort_by(nan_safe_cmp);
1248    let n = devs.len();
1249    let n2 = (n - 1) / 2;
1250    if n.is_multiple_of(2) {
1251        (devs[n2] + devs[n2 + 1]) / 2.0
1252    } else {
1253        devs[n2]
1254    }
1255}

1257/// NaN-safe comparator for f64 sorting (NaNs sort to the end).
1258fn nan_safe_cmp(a: &f64, b: &f64) -> Ordering {
1259    a.partial_cmp(b).unwrap_or(Ordering::Equal)
1260}
