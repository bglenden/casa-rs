2//! The rows a table holds on disk, which a partial in-place save must leave
3//! `table.dat` describing (#880).
4//!
5//! casacore rewrites `table.dat` only when a table's structure changed or
6//! the table is closed (`PlainTable::putFile`). Adding rows does not count
7//! as a change, so between those writes it publishes new rows in
8//! `table.lock`'s sync data alone, and `table.dat` trails them; a process
9//! that never closes the table, as CASA 6.7.6's simulator after a mosaic
10//! predict, leaves it so. casa-rs opens such a table with the published
11//! count. A partial save reads `table.dat` back from disk and writes it
12//! again, so it must bring the file's counts up to the table's: casacore
13//! checks every storage manager's rows against them.
14
15use super::*;
16use crate::storage::table_control::TableDatContents;
17
18/// The rows the table at `table_dir` holds as persisted, counted as
19/// [`Table::open`] counts them: the largest of `table.dat`'s two counts and
20/// the count the sync data publishes.
21pub(super) fn persisted_row_count(table_dir: &Path, table_dat: &TableDatContents) -> u64 {
22    table_dat
23        .nrrow
24        .max(table_dat.column_set.nrrow)
25        .max(published_row_count(table_dir))
26}
27
28/// The row count `table.lock`'s sync data publishes for the table at
29/// `table_dir`, 0 when there is none to read.
30pub(super) fn published_row_count(table_dir: &Path) -> u64 {
31    #[cfg(unix)]
32    {
33        crate::lock::read_sync_data_from_table_dir(table_dir)
34            .ok()
35            .flatten()
36            .map_or(0, |sync| sync.nrrow)
37    }
38    #[cfg(not(unix))]
39    {
40        let _ = table_dir;
41        0
42    }
43}
44
45impl Table {
46    /// The rows the plain table at `table_dir` holds on disk, counted as
47    /// [`Table::open`] counts them, read from `table.dat` and `table.lock`
48    /// without loading a row: for a caller that admits the memory an open
49    /// will take before it opens.
50    ///
51    /// # Errors
52    ///
53    /// The control file cannot be read, or the table is not a plain table.
54    pub fn rows_on_disk(table_dir: &Path) -> Result<usize, TableError> {
55        let control = table_dir.join(crate::storage::TABLE_CONTROL_FILE);
56        let table_dat = crate::storage::table_control::read_table_dat(&control)?;
57        Ok(persisted_row_count(table_dir, &table_dat) as usize)
58    }
59
60    /// Check that a partial in-place save may write the table at
61    /// `table_dir`, whose `table.dat` is `table_dat`, and return the rows the
62    /// table holds there.
63    ///
64    /// A partial save rewrites some data managers and leaves the others, so
65    /// this handle's rows must be the ones on disk: none added, inserted or
66    /// removed since the table was opened or last saved in full, and as many
67    /// as the disk holds. Otherwise the rewritten managers would disagree
68    /// with the others, or with `table.dat`, about the rows, and casacore
69    /// refuses the table. The check writes nothing, so a refused save leaves
70    /// the disk as it was.
71    pub(super) fn require_persisted_rows(
72        &self,
73        table_dir: &Path,
74        table_dat: &TableDatContents,
75    ) -> Result<u64, TableError> {
76        if self.inner.rows_reshaped() {
77            return Err(TableError::Storage(format!(
78                "a partial in-place save cannot write {}: rows were added or removed since \
79                 the table was opened or saved in full; save the whole table instead",
80                table_dir.display()
81            )));
82        }
83        let persisted = persisted_row_count(table_dir, table_dat);
84        if self.row_count() as u64 != persisted {
85            return Err(TableError::Storage(format!(
86                "a partial in-place save cannot write {}: it holds {persisted} rows on disk \
87                 and this handle {}; reopen the table",
88                table_dir.display(),
89                self.row_count()
90            )));
91        }
92        Ok(persisted)
93    }
94
95    /// Record a successful full save to `path`: when that is the table's own
96    /// directory, its rows on disk now match this handle's, and so does all
97    /// else it holds, so the release of a write lock has nothing to flush
98    /// (a flush after a save that removed rows would raise `table.dat`'s
99    /// counts back to the rows last published).
100    ///
101    /// A handle that holds the write lock also brings the row count in
102    /// `table.lock`'s sync data to the rows saved, now, and moves no counter
103    /// (other processes read the sync data only when they take a lock, so
104    /// they cannot see it early). Until the release publishes the write, a
105    /// metadata-only save, a partial save's admission and a reload all read
106    /// that count back (taking the largest of it and `table.dat`'s), and a
107    /// save that removed rows would have the old count raised again, leaving
108    /// `table.dat` claiming rows the storage managers do not hold. The
109    /// release still publishes, once, so the write is announced as before.
110    pub(super) fn note_full_save(&self, path: &Path) -> Result<(), TableError> {
111        if !self
112            .source_path
113            .as_deref()
114            .is_some_and(|source| locking::same_directory(source, path))
115        {
116            return Ok(());
117        }
118        self.inner.note_rows_persisted();
119        #[cfg(unix)]
120        if let Some(state) = self.lock_state.as_ref() {
121            state.flushed_generation.store(
122                self.inner.generation(),
123                std::sync::atomic::Ordering::Relaxed,
124            );
125            if state.lock_file.has_lock(LockType::Write) {
126                crate::lock::publish_persisted_rows(&state.lock_file, &state.path).map_err(
127                    |e| TableError::LockIo {
128                        path: state.path.display().to_string(),
129                        message: e.to_string(),
130                    },
131                )?;
132            }
133        }
134        Ok(())
135    }
136}
