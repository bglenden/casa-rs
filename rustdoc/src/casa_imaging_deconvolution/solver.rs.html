10/// What one minor cycle of one plane works on.
11///
12/// Planes are x-major ([`PlaneShape`]). The residual is in CASA's
13/// normalised units (Jy/beam) and the PSF is normalised so that term 0
14/// peaks at 1 at [`PsfSummary::peak`]; a Taylor view holds `N_t` residual
15/// terms and `2·N_t − 1` PSF terms, every other view one of each.
16#[derive(Clone, Copy, Debug)]
17pub struct MinorCycleView<'a> {
18    /// The plane shape.
19    pub shape: PlaneShape,
20    /// The residual terms.
21    pub residual: &'a [Vec<f64>],
22    /// The PSF terms.
23    pub psf: &'a [Vec<f64>],
24    /// What is known about PSF term 0.
25    pub summary: &'a PsfSummary,
26    /// Where components may be placed.
27    pub support: &'a Support,
28}
29
30/// One component a solver proposes.
31#[derive(Clone, Copy, Debug, PartialEq)]
32#[non_exhaustive]
33pub enum Candidate {
34    /// A pixel component at storage index `index` on scale `scale` (0 is a
35    /// point), with term-0 strength `strength` before the loop gain.
36    Pixel {
37        /// Storage index of the component centre.
38        index: usize,
39        /// Scale ordinal.
40        scale: usize,
41        /// Term-0 strength before the loop gain.
42        strength: f64,
43    },
44}
45
46/// The outcome of asking a solver for its next component.
47#[derive(Clone, Copy, Debug, PartialEq)]
48pub enum Next {
49    /// Clean this component.
50    Clean(Candidate),
51    /// The step ends by the solver's own rule: its threshold, its
52    /// divergence rule (multiscale's 50%) or no more work (Clark's ten
53    /// refreshes, no supported pixel). The plane controller judges the
54    /// plane from the step's peak and charge, as CASA's does.
55    Stop,
56}
57
58/// A plane solver: Högbom, Clark, multiscale or multi-term.
59///
60/// The driver runs one step as `initialize`, then `next`/`accept` until the
61/// budget is spent or `next` stops, then `finalize`. Each step starts from
62/// the residual the previous step left.
63pub trait Solver {
64    /// Working state of one step.
65    type State;
66
67    /// Iterations the solver may do beyond the step budget without charging
68    /// them: one for CASA's inclusive Högbom loop, otherwise none.
69    fn extra_iterations(&self) -> usize {
70        0
71    }
72
73    /// Whether a stop by the solver's own rule charges one iteration, as
74    /// CASA's `MatrixCleaner` counts the iteration that found it.
75    fn charges_stop(&self) -> bool {
76        false
77    }
78
79    /// Prepare one step from `residual`, cleaning down to `threshold`.
80    ///
81    /// # Errors
82    ///
83    /// When the PSF or the residual cannot support the solver.
84    fn initialize(
85        &self,
86        view: &MinorCycleView<'_>,
87        residual: &[Vec<f64>],
88        threshold: f64,
89    ) -> Result<Self::State, Error>;
90
91    /// The next component, tested against the step threshold.
92    ///
93    /// # Errors
94    ///
95    /// When the arithmetic leaves the finite numbers.
96    fn next(
97        &self,
98        state: &mut Self::State,
99        view: &MinorCycleView<'_>,
100        residual: &mut [Vec<f64>],
101    ) -> Result<Next, Error>;
102
103    /// The flux `candidate` adds to each model term at loop gain `gain`,
104    /// term 0 first, pushed onto `fluxes` before [`Solver::accept`] applies
105    /// it: the single term `gain × strength` unless the solver has more.
106    fn component_fluxes(
107        &self,
108        _state: &Self::State,
109        candidate: Candidate,
110        gain: f64,
111        fluxes: &mut Vec<f64>,
112    ) {
113        let Candidate::Pixel { strength, .. } = candidate;
114        fluxes.push(gain * strength);
115    }
116
117    /// Clean `candidate` with loop gain `gain`, adding the model update to
118    /// `delta`.
119    ///
120    /// # Errors
121    ///
122    /// When the arithmetic leaves the finite numbers.
123    fn accept(
124        &self,
125        state: &mut Self::State,
126        view: &MinorCycleView<'_>,
127        residual: &mut [Vec<f64>],
128        candidate: Candidate,
129        gain: f64,
130        delta: &mut Delta,
131    ) -> Result<(), Error>;
132
133    /// End the step: bring `residual` up to date with the step's `delta`
134    /// where the solver defers that, and report the peak residual CASA
135    /// reports for the step.
136    ///
137    /// # Errors
138    ///
139    /// When the arithmetic leaves the finite numbers.
140    fn finalize(
141        &self,
142        state: Self::State,
143        view: &MinorCycleView<'_>,
144        residual: &mut [Vec<f64>],
145        delta: &Delta,
146    ) -> Result<StepEnd, Error>;
147}

149/// How a step ended.
150#[derive(Clone, Copy, Debug, PartialEq)]
151pub struct StepEnd {
152    /// The peak residual magnitude CASA reports for the step.
153    pub peak: f64,
154    /// Exact whole-plane residual refreshes the step made (Clark's cycles,
155    /// multiscale's terminal convolution).
156    pub refreshes: usize,
157}

159/// The model update of a minor cycle: flux per storage index, per Taylor
160/// term.
161#[derive(Clone, Debug, Default, PartialEq)]
162pub struct Delta {
163    terms: Vec<BTreeMap<usize, f64>>,
164}
165
166impl Delta {
167    /// An empty update of `terms` terms.
168    #[must_use]
169    pub fn new(terms: usize) -> Self {
170        Self {
171            terms: vec![BTreeMap::new(); terms],
172        }
173    }
174
175    /// Add `flux` at storage index `index` of term `term`.
176    pub fn add(&mut self, term: usize, index: usize, flux: f64) {
177        *self.terms[term].entry(index).or_insert(0.0) += flux;
178    }
179
180    /// Add every flux of `other`.
181    pub fn merge(&mut self, other: &Self) {
182        for (term, fluxes) in other.terms.iter().enumerate() {
183            for (&index, &flux) in fluxes {
184                self.add(term, index, flux);
185            }
186        }
187    }
188
189    /// Number of terms.
190    #[must_use]
191    pub fn term_count(&self) -> usize {
192        self.terms.len()
193    }
194
195    /// The non-zero fluxes of term `term` in storage order.
196    pub fn term(&self, term: usize) -> impl Iterator<Item = (usize, f64)> + '_ {
197        self.terms[term]
198            .iter()
199            .filter(|(_, flux)| **flux != 0.0)
200            .map(|(index, flux)| (*index, *flux))
201    }
202
203    /// Whether the update changes nothing.
204    #[must_use]
205    pub fn is_empty(&self) -> bool {
206        self.terms
207            .iter()
208            .all(|fluxes| fluxes.values().all(|flux| *flux == 0.0))
209    }
210
211    /// Term `term` as a dense plane of `len` pixels.
212    #[must_use]
213    pub fn dense(&self, term: usize, len: usize) -> Vec<f64> {
214        let mut plane = vec![0.0; len];
215        for (index, flux) in self.term(term) {
216            plane[index] = flux;
217        }
218        plane
219    }
220}
