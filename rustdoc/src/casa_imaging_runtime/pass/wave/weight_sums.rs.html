2//! An owner's weight image, summed by contribution before it is gridded
3//! (#787).
4//!
5//! `Mode::Weight` places every sample's weight taps at the uv origin
6//! (`CpuBackend`'s gridding, the Metal records), so a sample's contribution
7//! is fixed by its kernel cell, plane, spectral factor, pointing gradient
8//! and the sign of its w, and scales with its weights alone. Samples that
9//! agree on those five add the same taps scaled by their weights: summing
10//! their weights in f64 first and gridding each group once gives the same
11//! image, to rounding, for a lookup and an add per sample rather than a
12//! kernel stamp.
13//!
14//! What it saves, in work and in precision, depends on how many groups the
15//! samples share:
16//! - **Fixed pointing.** A mosaic with fixed pointings shares a few groups
17//!   per pointing and plane: B10, the 1024² ALMA mosaic, sums 4.19 M
18//!   placements into 14 (#787). Its weight image then costs a lookup per
19//!   sample. On f32 grids (Metal, plan decision D2) it also keeps the grid's
20//!   own precision: millions of adds into the origin's cells had lost about
21//!   1e-3, and a few adds per group do not.
22//! - **A pointing that drifts row by row.** AW projection's pointing does,
23//!   which leaves about two samples a group: B3, the 63-field VLASS clean,
24//!   has 3.36 M groups for 6.42 M placements. The saving is then about
25//!   half, and f32 grids keep most of their per-sample drift.
26//!
27//! The row-work pass counts the groups the same way ([`GroupCount`]), so
28//! the strips charge the weight owner for the stamps it will make.
29

35/// Groups an owner's table holds before it grids them and starts again: a
36/// mosaic's first pass has a few per pointing and plane (14 on B10), so the
37/// table grids once a wave unless the pointing drifts row by row.
38pub(in crate::pass) const WEIGHT_GROUPS: usize = 1 << 14;
39
40/// Correlations a placement carries at most.
41const MAX_POL: usize = 4;
42
43/// What fixes one sample's weight-image contribution apart from its
44/// weights: every input `Mode::Weight` reads but the weights.
45///
46/// Both backends locate the weight taps at the uv origin, so the sample's
47/// `u` and `v` are not read. Only the sign of `w` is read (the Mueller table
48/// and the tap conjugation), and `phase` and `prediction_w_positive` are
49/// read only by prediction. Floats are kept as their bits, so two samples
50/// share a group only when their contributions are the same bits.
51#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
52struct Contribution {
53    cf: CfKey,
54    plane: u32,
55    spectral: u32,
56    gradient: [u32; 2],
57    w_positive: bool,
58}
59
60impl Contribution {
61    fn of(placement: &Placement) -> Self {
62        Self {
63            cf: placement.cf,
64            plane: placement.plane,
65            spectral: placement.spectral.to_bits(),
66            gradient: placement.gradient.map(f32::to_bits),
67            w_positive: placement.w > 0.0,
68        }
69    }
70
71    /// A placement with this contribution: at the uv origin, as
72    /// `Mode::Weight` grids every sample, with a w of the group's sign.
73    fn placement(self) -> Placement {
74        Placement {
75            u: 0.0,
76            v: 0.0,
77            w: if self.w_positive { 1.0 } else { -1.0 },
78            prediction_w_positive: self.w_positive,
79            phase: 0.0,
80            plane: self.plane,
81            spectral: f32::from_bits(self.spectral),
82            cf: self.cf,
83            gradient: self.gradient.map(f32::from_bits),
84        }
85    }
86}

88/// The groups a [`WeightSums`] of `capacity` groups grids for the
89/// placements it is shown, counted without summing: the row-work pass
90/// charges the weight owner a kernel stamp for each
91/// ([`ModeWork`](crate::pass::ModeWork)). Its set is allocated whole on the
92/// first placement, so a pass without a weight image holds nothing.
93pub(in crate::pass) struct GroupCount {
94    capacity: usize,
95    seen: HashSet<Contribution>,
96    last: Option<Contribution>,
97}
98
99impl GroupCount {
100    /// A counter that forgets every group when `capacity` are held, as a
101    /// full [`WeightSums`] grids and starts again.
102    pub(in crate::pass) fn new(capacity: usize) -> Self {
103        assert!(capacity > 0, "a group count holds at least one group");
104        Self {
105            capacity,
106            seen: HashSet::new(),
107            last: None,
108        }
109    }
110
111    /// Bytes a counter of `capacity` groups holds once it has seen a
112    /// placement: its hash set's buckets, a key and a control byte each.
113    #[must_use]
114    pub(in crate::pass) fn bytes(capacity: usize) -> u64 {
115        let buckets = (capacity * 8).div_ceil(7).next_power_of_two().max(4);
116        (buckets * (size_of::<Contribution>() + 1) + 16) as u64
117    }
118
119    /// Whether `placement` starts a group that [`WeightSums::add`] would
120    /// grid: its contribution is not among those held since the last time
121    /// the table filled.
122    pub(in crate::pass) fn starts(&mut self, placement: &Placement) -> bool {
123        let key = Contribution::of(placement);
124        if self.last == Some(key) {
125            return false;
126        }
127        self.last = Some(key);
128        if self.seen.capacity() == 0 {
129            self.seen.reserve(self.capacity);
130        }
131        if self.seen.contains(&key) {
132            return false;
133        }
134        if self.seen.len() == self.capacity {
135            self.seen.clear();
136        }
137        self.seen.insert(key);
138        true
139    }
140}

142/// One owner's weight-image contributions, summed in f64 by
143/// [`Contribution`] in the order they first appear, and gridded as one
144/// placement per group ([`Self::flush`]) when the table fills and when the
145/// wave ends.
146pub(in crate::pass) struct WeightSums {
147    npol: usize,
148    capacity: usize,
149    index: HashMap<Contribution, usize>,
150    groups: Vec<Contribution>,
151    /// `npol` weights per group.
152    sums: Vec<f64>,
153    /// The groups as placements, for the backend.
154    buffer: SampleBuffer,
155    /// Placements summed, groups gridded and grids made since the table
156    /// was made, for the wave's log ([`Self::totals`]).
157    totals: [u64; 3],
158}
159
160impl WeightSums {
161    /// An empty table of `capacity` groups of `npol` weights, allocated
162    /// whole ([`Self::bytes`]).
163    pub(in crate::pass) fn new(npol: usize, capacity: usize) -> Self {
164        assert!(
165            (1..=MAX_POL).contains(&npol),
166            "a placement carries one to four correlations"
167        );
168        assert!(capacity > 0, "a weight table holds at least one group");
169        Self {
170            npol,
171            capacity,
172            index: HashMap::with_capacity(capacity),
173            groups: Vec::with_capacity(capacity),
174            sums: Vec::with_capacity(capacity * npol),
175            buffer: SampleBuffer::with_capacity(npol, capacity),
176            totals: [0; 3],
177        }
178    }
179
180    /// Placements summed, groups gridded and grids made so far.
181    pub(in crate::pass) const fn totals(&self) -> [u64; 3] {
182        self.totals
183    }
184
185    /// Bytes a table of `capacity` groups of `npol` weights holds: the
186    /// hash index (its buckets, each a key, an index and a control byte,
187    /// at the standard library's 7/8 load), the groups, their sums and
188    /// the buffer they are gridded from.
189    #[must_use]
190    pub(in crate::pass) fn bytes(npol: usize, capacity: usize) -> u64 {
191        let buckets = (capacity * 8).div_ceil(7).next_power_of_two().max(4);
192        let index = buckets * (size_of::<(Contribution, usize)>() + 1) + 16;
193        (index + capacity * size_of::<Contribution>() + capacity * npol * size_of::<f64>()) as u64
194            + SampleBuffer::bytes(npol, capacity)
195    }
196
197    /// Bytes the table holds now, from its parts' capacities.
198    #[cfg(test)]
199    fn capacity_bytes(&self) -> u64 {
200        let index = self.index.capacity();
201        let buckets = (index * 8).div_ceil(7).next_power_of_two().max(4);
202        (buckets * (size_of::<(Contribution, usize)>() + 1)
203            + 16
204            + self.groups.capacity() * size_of::<Contribution>()
205            + self.sums.capacity() * size_of::<f64>()) as u64
206            + self.buffer.capacity_bytes()
207    }
208
209    /// Add the weights of every placement of `block` to its group, gridding
210    /// the groups through `grid` first whenever a new group finds the
211    /// table full.
212    pub(in crate::pass) fn add<E>(
213        &mut self,
214        block: &SampleBlock<'_>,
215        grid: &mut dyn FnMut(&SampleBlock<'_>) -> Result<(), E>,
216    ) -> Result<(), E> {
217        debug_assert_eq!(block.npol, self.npol, "the table's correlations");
218        self.totals[0] += block.len() as u64;
219        // Rows place their channels together, so successive samples share
220        // a group almost always: look the hash up only when it changes.
221        let mut last: Option<(Contribution, usize)> = None;
222        for (index, placement) in block.placements.iter().enumerate() {
223            let key = Contribution::of(placement);
224            let group = match last {
225                Some((held, group)) if held == key => group,
226                _ => {
227                    let group = match self.index.get(&key) {
228                        Some(&group) => group,
229                        None => {
230                            if self.groups.len() == self.capacity {
231                                self.flush(grid)?;
232                            }
233                            let group = self.groups.len();
234                            self.index.insert(key, group);
235                            self.groups.push(key);
236                            self.sums.resize(self.sums.len() + self.npol, 0.0);
237                            group
238                        }
239                    };
240                    last = Some((key, group));
241                    group
242                }
243            };
244            let sums = &mut self.sums[group * self.npol..(group + 1) * self.npol];
245            for (sum, weight) in sums.iter_mut().zip(block.weights_of(index)) {
246                *sum += f64::from(*weight);
247            }
248        }
249        Ok(())
250    }
251
252    /// Grid every group through `grid`, as one placement of its summed
253    /// weights each, and empty the table. Nothing when it is empty.
254    pub(in crate::pass) fn flush<E>(
255        &mut self,
256        grid: &mut dyn FnMut(&SampleBlock<'_>) -> Result<(), E>,
257    ) -> Result<(), E> {
258        if self.groups.is_empty() {
259            return Ok(());
260        }
261        self.buffer.clear();
262        let values = [Complex32::default(); MAX_POL];
263        let mut weights = [0.0_f32; MAX_POL];
264        for (group, key) in self.groups.iter().enumerate() {
265            let sums = &self.sums[group * self.npol..(group + 1) * self.npol];
266            for (weight, sum) in weights.iter_mut().zip(sums) {
267                *weight = *sum as f32;
268            }
269            self.buffer
270                .push(key.placement(), &values[..self.npol], &weights[..self.npol]);
271        }
272        self.totals[1] += self.groups.len() as u64;
273        self.totals[2] += 1;
274        self.index.clear();
275        self.groups.clear();
276        self.sums.clear();
277        grid(&self.buffer.block())
278    }
279}

299    /// The groups a table hands to `grid`, with their summed weights.
300    fn flushed(sums: &mut WeightSums) -> Vec<(Placement, Vec<f32>)> {
301        let mut out = Vec::new();
302        sums.flush(&mut |block: &SampleBlock<'_>| {
303            for (index, placement) in block.placements.iter().enumerate() {
304                out.push((*placement, block.weights_of(index).to_vec()));
305            }
306            Ok::<_, ()>(())
307        })
308        .expect("flush");
309        out
310    }

425    /// `samples` weight placements over three pointings, three spectral
426    /// factors and both signs of w, with random weights, in random order.
427    fn weight_samples(
428        operator: &casa_imaging_operator::MeasurementOperator,
429        samples: usize,
430    ) -> SampleBuffer {
431        let gradients = [[0.0, 0.0], [0.6, -0.2], [-0.3, 0.9]].map(|offset: [f64; 2]| {
432            operator
433                .geometry()
434                .pointing_gradient([offset[0] * 1.0e-4, offset[1] * 1.0e-4])
435        });
436        let spectral = [-0.01_f32, 0.0, 0.02];
437        let mut state = 0x9e37_79b9_7f4a_7c15_u64;
438        let mut next = || {
439            state ^= state << 13;
440            state ^= state >> 7;
441            state ^= state << 17;
442            state
443        };
444        let mut buffer = SampleBuffer::with_capacity(2, samples);
445        for _ in 0..samples {
446            let draw = next();
447            let weight = |shift: u32| 0.5 + ((draw >> shift) & 0xffff) as f32 / 65536.0;
448            buffer.push(
449                Placement {
450                    u: 0.0,
451                    v: 0.0,
452                    w: if draw & 1 == 0 { 25.0 } else { -25.0 },
453                    prediction_w_positive: true,
454                    phase: 0.0,
455                    plane: 0,
456                    spectral: spectral[(draw >> 1) as usize % 3],
457                    cf: CfKey::default(),
458                    gradient: gradients[(draw >> 3) as usize % 3],
459                },
460                &[Complex32::default(); 2],
461                &[weight(16), weight(32)],
462            );
463        }
464        buffer
465    }

467    /// The weight image `samples` grid on `operator`, one sample at a time
468    /// (`per_sample`) or summed by contribution, as its finished images.
469    fn weight_image(
470        operator: &casa_imaging_operator::MeasurementOperator,
471        samples: &SampleBuffer,
472        per_sample: bool,
473    ) -> Vec<f32> {
474        weight_image_and_sums(operator, samples, per_sample).0
475    }
476
477    /// [`weight_image`] and the finished images' `sumwt`.
478    fn weight_image_and_sums(
479        operator: &casa_imaging_operator::MeasurementOperator,
480        samples: &SampleBuffer,
481        per_sample: bool,
482    ) -> (Vec<f32>, Vec<f64>) {
483        use casa_imaging_operator::{CpuBackend, GridBackend, Mode, ModeSet, PlaneRange, Work};
484        let mut acc = operator.accumulator(PlaneRange::single(0), None, ModeSet::ALL);
485        let mut backend = CpuBackend::new();
486        let cf = operator.cf();
487        let mut grid = |block: &SampleBlock<'_>| {
488            backend.apply(
489                block,
490                cf,
491                Work::Grid {
492                    mode: Mode::Weight,
493                    acc: &mut acc,
494                },
495            )
496        };
497        if per_sample {
498            grid(&samples.block()).expect("grid");
499        } else {
500            let mut sums = WeightSums::new(2, WEIGHT_GROUPS);
501            sums.add(&samples.block(), &mut grid).expect("add");
502            sums.flush(&mut grid).expect("flush");
503        }
504        let images = operator.finish(acc).expect("finish");
505        let plane = &images.planes[0];
506        (
507            plane
508                .weight
509                .iter()
510                .flat_map(|image| image.iter().copied())
511                .collect(),
512            plane.sumwt.clone(),
513        )
514    }

516    /// The largest difference between `a` and `b`, relative to `b`'s peak.
517    fn relative(a: &[f32], b: &[f32]) -> f64 {
518        let peak = b
519            .iter()
520            .fold(0.0_f64, |peak, v| peak.max(f64::from(v.abs())));
521        a.iter()
522            .zip(b)
523            .map(|(a, b)| f64::from((a - b).abs()))
524            .fold(0.0, f64::max)
525            / peak
526    }

528    /// Summing the weight image by contribution grids, on the real mosaic
529    /// set with pointing ramps and Taylor terms, what gridding each sample
530    /// grids, to rounding in f64 grids (#787). On f32 grids it stays at the
531    /// f32 image's precision while the per-sample f32 weight image drifts
532    /// with the sample count.
533    #[test]
534    fn summed_weights_grid_what_each_sample_grids() {
535        use casa_imaging_operator::GridPrecision;
536        let f64_operator = mosaic_operator(GridPrecision::F64);
537        // Enough samples for the per-sample f32 drift, which grows with
538        // their count, to show: B10 grids 4.19 M (#787).
539        let samples = weight_samples(&f64_operator, 400_000);
540        let (reference, reference_sumwt) = weight_image_and_sums(&f64_operator, &samples, true);
541        let (summed, summed_sumwt) = weight_image_and_sums(&f64_operator, &samples, false);
542        let f64_error = relative(&summed, &reference);
543        assert!(f64_error <= 1.0e-6, "f64 grids: {f64_error}");
544        // The weight terms' sumwt takes each group's summed weight once, as
545        // the f32 weight its f64 sum rounds to (#792 review, Astra).
546        assert!(
547            reference_sumwt.iter().any(|sum| *sum > 0.0),
548            "a weight sumwt"
549        );
550        for (summed, each) in summed_sumwt.iter().zip(&reference_sumwt) {
551            assert!(
552                (summed - each).abs() <= 1.0e-7 * each.abs().max(1.0),
553                "sumwt: {summed} against {each}"
554            );
555        }
556        let f32_operator = mosaic_operator(GridPrecision::F32);
557        let summed_f32 = relative(&weight_image(&f32_operator, &samples, false), &reference);
558        let each_f32 = relative(&weight_image(&f32_operator, &samples, true), &reference);
559        assert!(summed_f32 <= 1.0e-6, "summed on f32 grids: {summed_f32}");
560        assert!(
561            each_f32 > 10.0 * summed_f32,
562            "per sample on f32 grids drifts: {each_f32} against {summed_f32}"
563        );
564    }

566    /// With no group shared, every pointing gradient distinct (AW's case
567    /// at its extreme, #792 review, Astra 2), the summed table grids on f32
568    /// grids exactly what each sample grids, bit for bit: groups of one
569    /// sample keep its f32 weights and their order, across the table's
570    /// flushes. It saves nothing then, but costs nothing in precision.
571    #[test]
572    fn with_no_group_shared_the_table_grids_each_sample_exactly() {
573        use casa_imaging_operator::GridPrecision;
574        let operator = mosaic_operator(GridPrecision::F32);
575        let mut samples = weight_samples(&operator, WEIGHT_GROUPS + 3000);
576        let block = samples.block();
577        let mut distinct = SampleBuffer::with_capacity(2, block.len());
578        for (index, placement) in block.placements.iter().enumerate() {
579            let step = index as f32 * 1.0e-7;
580            distinct.push(
581                Placement {
582                    gradient: [placement.gradient[0] + step, placement.gradient[1]],
583                    ..*placement
584                },
585                block.values_of(index),
586                block.weights_of(index),
