18pub(super) fn is_mosaic(record: &RecordValue) -> bool {
19    matches!(record_text(record, "name"), Ok("MosaicFT" | "MosaicFTNew"))
20}

22pub(super) struct MosaicLayout {
23    direction: DirectionCoordinate,
24    centre: [f64; 2],
25    increment: [f64; 2],
26    antenna_classes: Vec<u8>,
27}

29impl MosaicLayout {
30    pub(super) fn load(
31        record: &RecordValue,
32        row: &FourierRow<'_>,
33        direction: &DirectionCoordinate,
34        geometry: &GridGeometry,
35        polarization: &PolarizationRouting,
36        reference_hz: f64,
37        convert_frequencies: bool,
38    ) -> Result<(Box<dyn ConvolutionFunctionSet>, Self), GainSolveError> {
39        let Some(Value::Record(beam)) = record.get("pbconvfunc") else {
40            return Err(model_error("virtual mosaic has no beam convolution record"));
41        };
42        if record_text(beam, "name")? != "HetArrayConvFunc" {
43            return Err(model_error(
44                "virtual mosaic requires an array beam convolution",
45            ));
46        }
47        let telescope = row
48            .ms
49            .observation()
50            .map_err(model_error)?
51            .string(row.row.observation_id as usize, "TELESCOPE_NAME")
52            .map_err(model_error)?;
53        let (window, complete_window) = beam_window(row, convert_frequencies)?;
54        let mut loaded = saved::planes(record, beam, &window)?;
55        if loaded.is_none()
56            && let Some(complete) = complete_window
57            && window
58                .shares_beam_frequencies(&complete)
59                .map_err(model_error)?
60        {
61            loaded = saved::planes(record, beam, &complete)?;
62        }
63        let class = saved::integer(beam, "pbclass")?;
64        let custom = !matches!(class, 1 | 2);
65        let (dishes, antenna_classes) = match (custom, loaded.as_ref()) {
66            (true, Some(planes)) if planes.dishes == 1 => {
67                let path = record_text(beam, "vptable")?;
68                if path.is_empty() || !std::path::Path::new(path).is_dir() {
69                    return Err(model_error(
70                        "virtual mosaic voltage-pattern table is unavailable",
71                    ));
72                }
73                let antennas = row.ms.antenna().map_err(model_error)?;
74                (Vec::new(), vec![0; antennas.row_count()])
75            }
76            (true, _) => {
77                return Err(model_error(
78                    "virtual mosaic custom voltage-pattern selection requires a saved single-class convolution",
79                ));
80            }
81            (false, _) => antenna_beams(row, geometry, &telescope, class)?,
82        };
83        let restored = if let Some(planes) = loaded {
84            if planes.dishes != dishes.len() && !custom {
85                return Err(model_error(
86                    "virtual mosaic beam classes do not match the saved convolution",
87                ));
88            }
89            MosaicPb::from_saved_planes(geometry, polarization, &window, planes)
90                .map_err(model_error)?
91        } else {
92            None
93        };
94        let kernels = match restored {
95            Some(kernels) => kernels,
96            None if custom => {
97                return Err(model_error(
98                    "virtual mosaic has no saved convolution for the selected custom beam band",
99                ));
100            }
101            None => MosaicPb::new(
102                MosaicRules::Tclean,
103                geometry,
104                polarization,
105                reference_hz,
106                &dishes,
107                &[window],
108            )
109            .map_err(model_error)?,
110        };
111        let grid = geometry.image();
112        Ok((
113            Box::new(kernels),
114            Self {
115                direction: direction.clone(),
116                centre: grid.reference_pixel.map(|pixel| pixel as f64),
117                increment: grid.increment_rad,
118                antenna_classes,
119            },
120        ))
121    }

123    pub(super) fn pointing_offset(&self, row: &FourierRow<'_>) -> Result<[f64; 2], GainSolveError> {
124        // HetArrayConvFunc does not save its inherited POINTING selector.
125        // Reopened MosaicFT calls get without initializing a new predictor,
126        // so its convolution phases use the field direction.
127        let pointing = row
128            .field
129            .convert_to(self.direction.conversion_direction_ref(), row.frame)
130            .map_err(model_error)?;
131        let (longitude, latitude) = pointing.as_angles();
132        let pixel = self
133            .direction
134            .world_to_pixel([longitude, latitude])
135            .map_err(model_error)?;
136        Ok(std::array::from_fn(|axis| {
137            (pixel[axis] - self.centre[axis]) * self.increment[axis]
138        }))
139    }
140
141    pub(super) fn antenna_types(&self, row: &FourierRow<'_>) -> Result<[u8; 2], GainSolveError> {
142        let [Some(first), Some(second)] = [row.row.antenna1, row.row.antenna2]
143            .map(|antenna| self.antenna_classes.get(antenna as usize).copied())
144        else {
145            return Err(model_error(
146                "virtual mosaic refers to an unavailable antenna",
147            ));
148        };
149        Ok([first, second])
150    }
151}

153fn beam_window(
154    row: &FourierRow<'_>,
155    convert_frequencies: bool,
156) -> Result<(MosaicWindow, Option<MosaicWindow>), GainSolveError> {
157    let spw = row.row.data_spw_id as usize;
158    let windows = row.ms.spectral_window().map_err(model_error)?;
159    let channel_width_hz = windows
160        .chan_width(spw)
161        .map_err(model_error)?
162        .into_iter()
163        .next()
164        .ok_or_else(|| model_error("virtual mosaic window has no channel width"))?;
165    let ratio = if convert_frequencies && row.frequency_ref != FrequencyRef::LSRK {
166        let frequency = row.frequencies[0];
167        convert_frequency_to_frame(
168            row.frequency_ref,
169            FrequencyRef::LSRK,
170            frequency,
171            row.row.time_seconds,
172            row.row.field_id as usize,
173            row.engine,
174        )
175        .map_err(model_error)?
176            / frequency
177    } else {
178        1.0
179    };
180    let full = MosaicWindow {
181        spectral_window: spw as u32,
182        window_frequencies_hz: row.frequencies.to_vec(),
183        channel_width_hz,
184        selected_frequencies_hz: row
185            .frequencies
186            .iter()
187            .map(|frequency| frequency * ratio)
188            .collect(),
189        buffer_band: None,
190    };
191    if let Some(selection) = &row.row.channel_selection {
192        let selected = selection
193            .indices(row.frequencies.len())
194            .map_err(model_error)?;
195        let mut window = full.clone();
196        window.selected_frequencies_hz = selected
197            .into_iter()
198            .map(|index| full.selected_frequencies_hz[index])
199            .collect();
200        Ok((window, Some(full)))
201    } else {
202        Ok((full, None))
203    }
204}

206fn antenna_beams(
207    row: &FourierRow<'_>,
208    geometry: &GridGeometry,
209    telescope: &str,
210    class: i32,
211) -> Result<(Vec<PrimaryBeam>, Vec<u8>), GainSolveError> {
212    let antennas = row.ms.antenna().map_err(model_error)?;
213    if class == 1 {
214        let reference = row
215            .ms
216            .spectral_window()
217            .map_err(model_error)?
218            .ref_frequency(0)
219            .map_err(model_error)?;
220        if let Some(beam) = PrimaryBeam::for_telescope(telescope, reference) {
221            return Ok((vec![beam], vec![0; antennas.row_count()]));
222        }
223    }
224    let alma = matches!(telescope, "ALMA" | "ACA");
225    let mut dishes = Vec::new();
226    let mut antenna_classes = Vec::with_capacity(antennas.row_count());
227    for antenna in 0..antennas.row_count() {
228        let diameter = antennas.dish_diameter(antenna).map_err(model_error)?;
229        let dish = if alma {
230            AiryDish::casa_alma(diameter, geometry)
231        } else {
232            AiryDish::casa_airy(diameter)
233        };
234        let beam = PrimaryBeam::from(dish);
235        let class = dishes
236            .iter()
237            .position(|known| *known == beam)
238            .unwrap_or_else(|| {
239                dishes.push(beam);
240                dishes.len() - 1
241            });
242        antenna_classes.push(u8::try_from(class).map_err(model_error)?);
243    }
244    Ok((dishes, antenna_classes))
245}
