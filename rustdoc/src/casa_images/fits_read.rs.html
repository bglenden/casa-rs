15/// A FITS image read into memory ([`read_fits_image`]).
16#[derive(Debug, Clone)]
17pub struct FitsImage {
18    /// The axis lengths, `NAXIS1` first.
19    pub shape: Vec<usize>,
20    /// The coordinate system, as `importfits` makes it.
21    pub coordinates: CoordinateSystem,
22    /// The header keywords casa-rs reads.
23    pub header: FitsHeader,
24    /// The restoring beam, image type and object name.
25    pub image_info: ImageInfo,
26    /// `BUNIT`, or empty.
27    pub units: String,
28    /// The pixels in Fortran order, as single precision: casacore's
29    /// `FITSImage` reads every `BITPIX` as `Float`.
30    pub pixels: ArrayD<f32>,
31}
32
33/// Read the primary-HDU image of `fitsimage` into memory, as CASA's
34/// `ia.newimagefromfile` opens a FITS file.
35///
36/// # Errors
37///
38/// [`ImageError`] when the file cannot be read, its primary HDU is not an
39/// image, or its header does not make a coordinate system.
40pub fn read_fits_image(fitsimage: impl AsRef<Path>) -> Result<FitsImage, ImageError> {
41    let mut primary = PrimaryHdu::open(fitsimage.as_ref())?;
42    let data = primary
43        .hdu
44        .read_image::<Vec<f32>>(&mut primary.fits)
45        .map_err(|error| ImageError::Io(error.to_string()))?;
46    let pixels = ArrayD::from_shape_vec(IxDyn(&primary.shape).f(), data).map_err(|error| {
47        ImageError::InvalidMetadata(format!("invalid FITS image shape: {error}"))
48    })?;
49    Ok(FitsImage {
50        image_info: image_info_from_fits_header(&primary.header),
51        units: primary.header.get_string("BUNIT").unwrap_or("").to_string(),
52        shape: primary.shape,
53        coordinates: primary.coordinates,
54        header: primary.header,
55        pixels,
56    })
57}

59/// An open FITS file's primary HDU, with its shape, header and coordinates
60/// read.
61pub(crate) struct PrimaryHdu {
62    pub(crate) fits: FitsFile,
63    pub(crate) hdu: fitsio::hdu::FitsHdu,
64    pub(crate) image_type: FitsImageType,
65    pub(crate) shape: Vec<usize>,
66    pub(crate) header: FitsHeader,
67    pub(crate) coordinates: CoordinateSystem,
68}
69
70impl PrimaryHdu {
71    pub(crate) fn open(fitsimage: &Path) -> Result<Self, ImageError> {
72        let mut fits =
73            FitsFile::open(fitsimage).map_err(|error| ImageError::Io(error.to_string()))?;
74        let hdu = fits
75            .primary_hdu()
76            .map_err(|error| ImageError::Io(error.to_string()))?;
77        let image_type = match &hdu.info {
78            HduInfo::ImageInfo { image_type, .. } => *image_type,
79            HduInfo::TableInfo { .. } | HduInfo::AnyInfo => {
80                return Err(ImageError::InvalidMetadata(
81                    "primary HDU is not a FITS image".to_string(),
82                ));
83            }
84        };
85        let shape = read_fits_axis_shape(&mut fits, &hdu)?;
86        let header = read_primary_fits_header(&mut fits, &hdu, &shape)?;
87        let coordinates = from_fits_header(&header, &shape)?;
88        Ok(Self {
89            fits,
90            hdu,
91            image_type,
92            shape,
93            header,
94            coordinates,
95        })
96    }
97}

99pub(crate) fn image_info_from_fits_header(header: &FitsHeader) -> ImageInfo {
100    let object_name = header.get_string("OBJECT").unwrap_or("").to_string();
101    let beam_set = match (
102        header.get_float("BMAJ"),
103        header.get_float("BMIN"),
104        header.get_float("BPA"),
105    ) {
106        (Some(major_deg), Some(minor_deg), Some(position_angle_deg)) => {
107            ImageBeamSet::new(GaussianBeam::new(
108                major_deg.to_radians(),
109                minor_deg.to_radians(),
110                position_angle_deg.to_radians(),
111            ))
112        }
113        _ => ImageBeamSet::default(),
114    };
115    ImageInfo {
116        beam_set,
117        image_type: header
118            .get_int("IMTYPE")
119            .map(|value| ImageType::from_fits_value(value as i32))
120            .unwrap_or(ImageType::Undefined),
121        object_name,
122    }
123}

125fn read_primary_fits_header(
126    fits: &mut FitsFile,
127    hdu: &fitsio::hdu::FitsHdu,
128    shape: &[usize],
129) -> Result<FitsHeader, ImageError> {
130    let mut header = FitsHeader::new();
131    header.set("NAXIS", FitsValue::Integer(shape.len() as i64));
132    for (axis, axis_len) in shape.iter().enumerate() {
133        header.set(
134            format!("NAXIS{}", axis + 1),
135            FitsValue::Integer(*axis_len as i64),
136        );
137    }
138    for key in [
139        "RADESYS", "CUNIT1", "CUNIT2", "CUNIT3", "CUNIT4", "CTYPE1", "CTYPE2", "CTYPE3", "CTYPE4",
140        "SPECSYS", "BUNIT", "OBJECT", "TELESCOP", "OBSERVER",
141    ] {
142        if let Some(value) = read_optional_key::<String>(fits, hdu, key)? {
143            header.set(key, FitsValue::String(value.trim_end().to_string()));
144        }
145    }
146    for key in [
147        "EQUINOX", "CRVAL1", "CRVAL2", "CRVAL3", "CRVAL4", "CRPIX1", "CRPIX2", "CRPIX3", "CRPIX4",
148        "CDELT1", "CDELT2", "CDELT3", "CDELT4", "RESTFRQ", "BMAJ", "BMIN", "BPA", "IMTYPE",
149        "LONPOLE", "LATPOLE", "CROTA2",
150    ] {
151        if let Some(value) = read_optional_key::<f64>(fits, hdu, key)? {
152            header.set(key, FitsValue::Float(value));
153        }
154    }
155    for i in 1..=shape.len() {
156        for j in 1..=shape.len() {
157            let pc_key = format!("PC{i}_{j}");
158            if let Some(value) = read_optional_key::<f64>(fits, hdu, &pc_key)? {
159                header.set(pc_key, FitsValue::Float(value));
160            }
161            let cd_key = format!("CD{i}_{j}");
162            if let Some(value) = read_optional_key::<f64>(fits, hdu, &cd_key)? {
163                header.set(cd_key, FitsValue::Float(value));
164            }
165        }
166    }
167    Ok(header)
168}

170fn read_fits_axis_shape(
171    fits: &mut FitsFile,
172    hdu: &fitsio::hdu::FitsHdu,
173) -> Result<Vec<usize>, ImageError> {
174    let naxis = hdu
175        .read_key::<i64>(fits, "NAXIS")
176        .map_err(|error| ImageError::Io(error.to_string()))? as usize;
177    let mut shape = Vec::with_capacity(naxis);
178    for axis in 1..=naxis {
179        let axis_len = hdu
180            .read_key::<i64>(fits, &format!("NAXIS{axis}"))
181            .map_err(|error| ImageError::Io(error.to_string()))?;
182        shape.push(axis_len as usize);
183    }
184    Ok(shape)
185}

187fn read_optional_key<T>(
188    fits: &mut FitsFile,
189    hdu: &fitsio::hdu::FitsHdu,
190    key: &str,
191) -> Result<Option<T>, ImageError>
192where
193    T: fitsio::headers::ReadsKey,
194{
195    match hdu.read_key::<T>(fits, key) {
196        Ok(value) => Ok(Some(value)),
197        Err(error) => {
198            let text = error.to_string();
199            if text.contains("keyword not found") || text.contains("not found in header") {
200                Ok(None)
201            } else {
202                Err(ImageError::Io(text))
203            }
204        }
205    }
