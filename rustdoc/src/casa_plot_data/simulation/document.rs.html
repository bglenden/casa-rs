7    PlotPayloadStrategy, PlotTextStyle,

10pub(crate) fn panel(
11    id: &str,
12    title: &str,
13    labels: [&str; 2],
14    limits: [[f64; 2]; 2],
15    equal: bool,
16) -> PlotDocumentPanel {
17    PlotDocumentPanel {
18        id: id.into(),
19        title: title.into(),
20        title_font_size: None,
21        axes: ["x", "y"]
22            .into_iter()
23            .zip(labels)
24            .zip(limits)
25            .map(|((id, label), [lower, upper])| PlotDocumentAxis {
26                id: id.into(),
27                label: label.into(),
28                unit: String::new(),
29                lower,
30                upper,
31                scale: PlotAxisScale::Linear,
32                lane_labels: vec![],
33                draws_on_trailing_edge: false,
34            })
35            .collect(),
36        layers: vec![],
37        annotations: vec![],
38        aspect: Some(if equal {
39            PlotAspect::EqualDataLimits
40        } else {
41            PlotAspect::Auto
42        }),
43        frame_visible: None,
44    }
45}

47pub(crate) fn document(id: &str, panels: Vec<PlotDocumentPanel>, columns: u32) -> PlotDocument {
48    let rows = (panels.len() as u32).div_ceil(columns);
49    PlotDocument {
50        id: id.into(),
51        title: String::new(),
52        subtitle: String::new(),
53        header_lines: vec![],
54        axes: vec![],
55        layers: vec![],
56        annotations: vec![],
57        panels,
58        show_legend: false,
59        layout: Some(PlotLayout {
60            rows,
61            columns,
62            first_panel: None,
63            aspect: None,
64            margins: (columns > 1).then_some([0.05, 0.98, 0.09, 0.95, 0.2, 0.2]),
65            font_size: Some("x-small".into()),
66        }),
67    }
68}

70pub(crate) fn layer(
71    id: &str,
72    kind: PlotLayerKind,
73    data: PlotLayerData,
74    color: Option<&str>,
75    marker: Option<&str>,
76) -> PlotDocumentLayer {
77    let count = match &data {
78        PlotLayerData::Points { x, .. } | PlotLayerData::Circles { x, .. } => x.shape[0],
79        PlotLayerData::Image { pixels, .. } => pixels.shape.iter().product(),
80    };
81    PlotDocumentLayer {
82        id: id.into(),
83        title: String::new(),
84        kind,
85        x_axis_id: "x".into(),
86        y_axis_id: "y".into(),
87        x_values: vec![],
88        y_values: vec![],
89        point_labels: vec![],
90        point_symbol_sizes: vec![],
91        interval_x_start: vec![],
92        interval_x_end: vec![],
93        interval_y: vec![],
94        interval_height: vec![],
95        interval_labels: vec![],
96        provenance: vec![],
97        color_group: String::new(),
98        symbol_size: 9.0,
99        line_width: if kind == PlotLayerKind::Circles {
100            1.0
101        } else {
102            1.5
103        },
104        opacity: 1.0,
105        source_sample_count: count,
106        payload_strategy: PlotPayloadStrategy::BinaryArrays,
107        provenance_summary: "All native simulation samples; no thinning".into(),
108        data: Some(data),
109        style: Some(PlotLayerStyle {
110            color: color.map(str::to_string),
111            marker: marker.map(str::to_string),
112            line_style: None,
113        }),
114    }
115}

117pub(crate) fn annotation(
118    id: &str,
119    position: [f64; 2],
120    text: String,
121    relative: bool,
122) -> PlotDocumentAnnotation {
123    PlotDocumentAnnotation {
124        id: id.into(),
125        x: position[0],
126        y: position[1],
127        text,
128        style: Some(PlotTextStyle {
129            axes_relative: relative,
130            horizontal_alignment: Some(if relative { "left" } else { "center" }.into()),
131            vertical_alignment: Some(if relative { "top" } else { "center" }.into()),
132            font_size: relative.then(|| "x-small".into()),
133            background: relative.then(|| "white".into()),
134        }),
135    }
