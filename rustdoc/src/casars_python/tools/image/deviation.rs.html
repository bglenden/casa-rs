21struct LengthArgument {
22    value: Variant,
23    pixels: Option<i64>,
24    flags: Option<String>,
25}
26
27impl Default for LengthArgument {
28    fn default() -> Self {
29        Self {
30            value: Variant::Text("1pix".into()),
31            pixels: None,
32            flags: None,
33        }
34    }
35}
36
37impl<'a, 'py> FromPyObject<'a, 'py> for LengthArgument {
38    type Error = PyErr;
39
40    fn extract(value: Borrowed<'a, 'py, PyAny>) -> PyResult<Self> {
41        let pixels = if value.is_instance_of::<PyBool>() {
42            None
43        } else {
44            value.extract::<i64>().ok()
45        };
46        Ok(Self {
47            value: value.extract()?,
48            pixels,
49            flags: if value.is_instance_of::<PyString>() {
50                None
51            } else {
52                sequence_items(&value)?.as_deref().and_then(bool_history)
53            },
54        })
55    }
56}
57
58impl LengthArgument {
59    fn parse(&self, name: &str) -> PyResult<Quantity> {
60        if let Some(pixels) = self.pixels {
61            return Quantity::new(pixels as f64, "pix").map_err(raised);
62        }
63        match &self.value {
64            Variant::Text(_) | Variant::Number(_) => quantity(&self.value)
65                .map_err(|error| raised(format!("{name} is not a valid quantity: {error}"))),
66            _ => Err(raised(format!("{name} is not a valid quantity"))),
67        }
68    }
69
70    fn is_empty(&self) -> bool {
71        self.flags.is_some()
72            || self.value.is_empty_sequence()
73            || matches!(&self.value, Variant::Text(text) if text.is_empty())
74    }
75
76    fn history(&self) -> String {
77        self.flags.as_ref().map_or_else(
78            || history_variant(&self.value),
79            |flags| history_value(flags).to_owned(),
80        )
81    }
82}
83
84fn sequence_items<'py>(value: &Bound<'py, PyAny>) -> PyResult<Option<Vec<Bound<'py, PyAny>>>> {
85    value
86        .try_iter()
87        .ok()
88        .map(|items| items.collect())
89        .transpose()
90}
91
92fn bool_history(items: &[Bound<'_, PyAny>]) -> Option<String> {
93    let flags = items
94        .iter()
95        .map(|item| {
96            item.cast::<PyBool>()
97                .ok()
98                .map(|flag| if flag.is_true() { "true" } else { "false" })
99        })
100        .collect::<Option<Vec<_>>>()?;
101    Some(format!("[{}]", flags.join(",")))
102}
103
104fn history_value(value: &str) -> &str {
105    if value.len() > 300 { "..." } else { value }
106}
107
108fn history_string(value: &str) -> String {
109    format!("\"{}\"", history_value(value))
110}
111
112fn history_variant(value: &Variant) -> String {
113    match value {
114        Variant::Text(text) => history_string(text),
115        value => history_value(&value.describe()).to_owned(),
116    }
117}

119struct AnchorArgument {
120    history: String,
121    selection: Result<Option<[i64; 2]>, String>,
122}
123
124impl Default for AnchorArgument {
125    fn default() -> Self {
126        Self {
127            history: r#""ref""#.into(),
128            selection: Ok(None),
129        }
130    }
131}
132
133impl<'a, 'py> FromPyObject<'a, 'py> for AnchorArgument {
134    type Error = PyErr;
135
136    fn extract(value: Borrowed<'a, 'py, PyAny>) -> PyResult<Self> {
137        if let Ok(token) = value.extract::<String>() {
138            return Ok(Self {
139                history: history_string(&token),
140                selection: if token == "ref" {
141                    Ok(None)
142                } else {
143                    Err(format!("Unsupported value for anchor: {token}"))
144                },
145            });
146        }
147        let variant: Variant = value.extract()?;
148        if variant.is_empty_sequence() {
149            return Ok(Self {
150                history: "[]".into(),
151                selection: Ok(None),
152            });
153        }
154        let items = sequence_items(&value)?;
155        if let Some(flags) = items.as_deref().and_then(bool_history) {
156            return Ok(Self {
157                history: history_value(&flags).into(),
158                selection: Ok(None),
159            });
160        }
161        let integers = items.and_then(|items| {
162            items
163                .iter()
164                .map(|item| item.extract::<i64>().ok())
165                .collect::<Option<Vec<_>>>()
166        });
167        let selection: Result<Option<[i64; 2]>, String> = match (&variant, integers) {
168            (Variant::Numbers(_), Some(values)) => values
169                .try_into()
170                .map(Some)
171                .map_err(|_| "anchor must have exactly two integer values".to_owned()),
172            _ => Err("Unsupported type for anchor".to_owned()),
173        };
174        let history = match selection.as_ref() {
175            Ok(Some([x, y])) => format!("[{x},{y}]"),
176            _ => history_variant(&variant),
177        };
178        Ok(Self { history, selection })
179    }
180}

182#[pymethods]
183impl PyImageTool {
184    /// Create an image of local statistics sampled across each direction plane.
185    #[allow(clippy::too_many_arguments)]
186    #[pyo3(signature = (outfile="", region=RegionArgument::empty_text(), mask=None, overwrite=false, stretch=false, grid=IntList::of(&[1, 1]), anchor=AnchorArgument::default(), xlength=LengthArgument::default(), ylength=LengthArgument::default(), interp="cubic", stattype="sigma", statalg="classic", zscore=-1.0, maxiter=-1), text_signature = "($self, outfile='', region='', mask='', overwrite=False, stretch=False, grid=[1, 1], anchor='ref', xlength='1pix', ylength='1pix', interp='cubic', stattype='sigma', statalg='classic', zscore=-1.0, maxiter=-1)")]
187    fn deviation(
188        &self,
189        py: Python<'_>,
190        outfile: &str,
191        region: RegionArgument,
192        mask: Option<&Bound<'_, PyAny>>,
193        #[pyo3(from_py_with = require_bool)] overwrite: bool,
194        #[pyo3(from_py_with = require_bool)] stretch: bool,
195        grid: IntList,
196        anchor: AnchorArgument,
197        xlength: LengthArgument,
198        ylength: LengthArgument,
199        interp: &str,
200        stattype: &str,
201        statalg: &str,
202        zscore: f64,
203        maxiter: i64,
204    ) -> PyResult<Py<Self>> {
205        let mask = mask_argument(mask)?;
206        let output = py.detach(|| -> PyResult<_> {
207            let state = self.state();
208            let Some(attachment) = state.image.as_ref() else {
209                return Ok(None);
210            };
211            let grid_arr = parse_deviation_grid(&grid.0).map_err(raised)?;
212            let interpolation = parse_deviation_interpolation(interp).map_err(raised)?;
213            let options = DeviationOptions {
214                region: region.selection(),
215                mask: Some(&mask),
216                stretch,
217                destination: (!outfile.is_empty()).then(|| Path::new(outfile)),
218                grid: grid_arr,
219                anchor: anchor.selection.as_ref().copied().map_err(raised)?,
220                x_length: xlength.parse("xlength")?,
221                y_length: (!ylength.is_empty())
222                    .then(|| ylength.parse("ylength"))
223                    .transpose()?,
224                interpolation,
225                statistic: stattype,
226                algorithm: statalg,
227                zscore,
228                maxiter,
229            };
230            let history_command = state.history.then(|| {
231                let region_hist = match region.selection() {
232                    Some(SubimageRegion::Text(text)) => history_string(text),
233                    _ => region.history(),
234                };
235                let grid_str = grid
236                    .0
237                    .iter()
238                    .map(i64::to_string)
239                    .collect::<Vec<_>>()
240                    .join(",");
241                format!(
242                    "ia.deviation(outfile={}, region={region_hist}, mask={}, overwrite={overwrite}, stretch={stretch}, grid=[{grid_str}], anchor={}, xlength={}, ylength={}, interp={}, stattype={}, statalg={}, zscore={zscore:.6}, maxiter={maxiter})",
243                    history_string(outfile),
244                    history_string(&mask),
245                    anchor.history,
246                    xlength.history(),
247                    ylength.history(),
248                    history_string(interp),
249                    history_string(stattype),
250                    history_string(statalg),
251                )
252            });
253            deviation_image(
254                attachment,
255                outfile,
256                overwrite,
257                options,
258                history_command,
259                external_open_check(),
260            )
261            .map(Some)
262            .map_err(raised)
263        })?;
264        Py::new(py, output.map_or_else(Self::new, Self::attached))
265    }
