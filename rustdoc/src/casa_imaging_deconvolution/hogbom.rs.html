2//! Högbom CLEAN, as CASA's `SDAlgorithmHogbomClean` runs the Fortran
3//! `hclean` (casacore `scimath_f/hclean.f`).
4
5use crate::Error;
6use crate::patch::subtract_shifted;
7use crate::plane::{PlaneShape, first_peak, peak_magnitude};
8use crate::solver::{Candidate, Delta, MinorCycleView, Next, Solver, StepEnd};
9
10/// Högbom's point CLEAN on one plane.
11///
12/// Each iteration finds the supported pixel of largest magnitude (the first
13/// in x-fastest order on ties), stops when it is strictly below the
14/// threshold, and otherwise subtracts the gain-scaled PSF shifted to it.
15/// With `inclusive`, a step runs CASA's `do iter = 0, niter` loop: one more
16/// component than its budget, charged as the budget.
17#[derive(Clone, Copy, Debug, PartialEq, Eq)]
18pub struct Hogbom {
19    inclusive: bool,
20}
21
22impl Hogbom {
23    /// Högbom with CASA's inclusive iteration count when `inclusive`.
24    #[must_use]
25    pub const fn new(inclusive: bool) -> Self {
26        Self { inclusive }
27    }
28}
29
30impl Solver for Hogbom {
31    /// The step threshold.
32    type State = f64;
33
34    fn extra_iterations(&self) -> usize {
35        usize::from(self.inclusive)
36    }
37
38    fn initialize(
39        &self,
40        _: &MinorCycleView<'_>,
41        _: &[Vec<f64>],
42        threshold: f64,
43    ) -> Result<f64, Error> {
44        Ok(threshold)
45    }
46
47    fn next(
48        &self,
49        threshold: &mut f64,
50        view: &MinorCycleView<'_>,
51        residual: &mut [Vec<f64>],
52    ) -> Result<Next, Error> {
53        let Some((index, value)) = first_peak(&residual[0], view.support) else {
54            return Ok(Next::Stop);
55        };
56        if !value.is_finite() {
57            return Err(Error::NonFinite);
58        }
59        if value.abs() < *threshold {
60            return Ok(Next::Stop);
61        }
62        Ok(Next::Clean(Candidate::Pixel {
63            index,
64            scale: 0,
65            strength: value,
66        }))
67    }
68
69    fn accept(
70        &self,
71        _: &mut f64,
72        view: &MinorCycleView<'_>,
73        residual: &mut [Vec<f64>],
74        candidate: Candidate,
75        gain: f64,
76        delta: &mut Delta,
77    ) -> Result<(), Error> {
78        let Candidate::Pixel {
79            index, strength, ..
80        } = candidate;
81        let flux = gain * strength;
82        delta.add(0, index, flux);
83        // `hclean` takes the PSF at the plane's centre, whatever its peak.
84        let [x, y] = view.shape.centre();
85        subtract_shifted(
86            &mut residual[0],
87            &view.psf[0],
88            view.shape,
89            view.shape.index(x, y),
90            index,
91            flux,
92        )
93    }
94
95    fn finalize(
96        &self,
97        _: f64,
98        view: &MinorCycleView<'_>,
99        residual: &mut [Vec<f64>],
100        _: &Delta,
101    ) -> Result<StepEnd, Error> {
102        // SDAlgorithmHogbomClean::takeOneStep reports SDAlgorithmBase's
103        // findMaxAbsMask, a magnitude (unlike casacore's signed one).
104        Ok(StepEnd {
105            peak: peak_magnitude(&residual[0], view.support),
106            refreshes: 0,
107        })
108    }
109
110    /// Nothing: Högbom works in the driver's residual copy.
111    fn working_bytes(&self, _shape: PlaneShape, _terms: usize) -> u64 {
112        0
113    }
114
115    /// A point component.
116    fn component_cells(&self, _shape: PlaneShape) -> usize {
117        1
118    }
119}

121#[cfg(test)]
122mod hclean_tests {
123    use super::*;
124    use crate::{CycleControls, PlaneStatistics, PsfSummary, Support, run_plane};
125
126    /// `hclean` subtracts the PSF from the plane's centre even where a
127    /// higher point lies elsewhere in it: a unit lobe at the centre and a
128    /// slightly higher one four pixels from it in x (CASA 6.7.6.14 on channel
129    /// 0 of `refim_point_linRL.ms`, a PSF this shape).
130    #[test]
131    fn the_psf_is_taken_at_the_planes_centre() {
132        let shape = PlaneShape::new(64, 64);
133        let mut psf = vec![0.0; 64 * 64];
134        for x in 0..64_i32 {
135            for y in 0..64_i32 {
136                let lobe = |cx: i32, cy: i32, peak: f64| {
137                    let (dx, dy) = (f64::from(x - cx), f64::from(y - cy));
138                    peak * (-(dx * dx + dy * dy) / 2.0).exp()
139                };
140                psf[shape.index(x as usize, y as usize)] = lobe(32, 32, 0.9998) + lobe(36, 32, 1.0);
141            }
142        }
143        let residual = vec![psf.clone()];
144        let psfs = vec![psf];
145        let summary = PsfSummary::new(&psfs[0], shape).expect("a summary");
146        let support = Support::new(shape, vec![true; shape.len()]);
147        let view = MinorCycleView {
148            shape,
149            residual: &residual,
150            psf: &psfs,
151            summary: &summary,
152            support: &support,
153        };
154        let statistics = PlaneStatistics::measure(&residual[0], &support, &support, 0.0, false);
155        let controls = CycleControls {
156            iterations: 1,
157            threshold: 0.0,
158            threshold_reached: true,
159            gain: 1.0,
160            nsigma: 0.0,
161        };
162        let outcome = run_plane(&Hogbom::new(false), &view, &controls, &statistics, 0).unwrap();
163        let end = &outcome.end_residual[0];
164        // The peak residual is at the higher lobe. There the subtraction reads
165        // the PSF's centre, not its peak, which would leave almost nothing.
166        let at = shape.index(36, 32);
167        assert!(
168            (end[at] - (residual[0][at] - residual[0][at] * psfs[0][shape.index(32, 32)])).abs()
169                < 1e-9
170        );
171    }
