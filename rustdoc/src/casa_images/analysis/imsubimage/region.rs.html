120// CASA's implicit region selects a single pixel on two positional axes
121// when a multidimensional image has no direction coordinate.
122pub(super) fn implicit_positional_axes<T: ImagePixel>(
123    image: &dyn ImageInterface<T>,
124) -> Option<[usize; 2]> {
125    if image.find_axis(CoordinateType::Direction).is_some() {
126        return None;
127    }
128    let spectral = image.find_axis(CoordinateType::Spectral);
129    let stokes = image.find_axis(CoordinateType::Stokes);
130    let mut axes = (0..image.ndim()).filter(|&axis| Some(axis) != spectral && Some(axis) != stokes);
131    let leading = [axes.next()?, axes.next()?];
132    axes.next().map(|_| leading)
133}

135impl SubimageSelection {
136    pub(super) fn resolve_for_insert<T: ImagePixel>(
137        image: &dyn ImageInterface<T>,
138        target: &dyn ImageInterface<T>,
139        region: Option<SubimageRegion<'_>>,
140    ) -> Result<(Self, Vec<usize>), ImageError> {
141        if let Some(SubimageRegion::Record(record)) = region
142            && let Some(slicer) =
143                saved::lc_slicer_section(record, image.coordinates(), image.shape())?
144            && slicer.stride.iter().any(|&step| step != 1)
145        {
146            let mut selection = Self::resolve(image, None, None, None, None)?;
147            selection.start = slicer.start;
148            selection.shape = slicer.count;
149            return Ok((selection, slicer.stride));
150        }
151        let text_context = TextRegionContext {
152            coordinates: Some(target.coordinates()),
153            shape: None,
154        };
155        let mut selection =
156            Self::resolve_with_origin(image, None, region, None, None, Some(text_context))?;
157        selection.shape = selection.packed_shape();
158        let stride = selection
159            .stride
160            .take()
161            .unwrap_or_else(|| vec![1; image.ndim()]);
162        Ok((selection, stride))
163    }
164
165    pub(in crate::analysis) fn packed_shape(&self) -> Vec<usize> {
166        match &self.stride {
167            Some(stride) => self
168                .shape
169                .iter()
170                .zip(stride)
171                .map(|(&size, &step)| size.div_ceil(step))
172                .collect(),
173            None => self.shape.clone(),
174        }
175    }

177    pub(in crate::analysis) fn resolve<T: ImagePixel>(
178        image: &dyn ImageInterface<T>,
179        box_arg: Option<&str>,
180        region_arg: Option<SubimageRegion<'_>>,
181        chans_arg: Option<&str>,
182        stokes_arg: Option<&str>,
183    ) -> Result<Self, ImageError> {
184        let selection = Self::resolve_section(image, box_arg, region_arg, chans_arg, stokes_arg)?;
185        if let Some(strides) = &selection.stride
186            && let Some((axis, &step)) = strides.iter().enumerate().find(|(_, step)| **step != 1)
187        {
188            return Err(ImageError::InvalidMetadata(format!(
189                "saved region has non-unit stride {step} on axis {axis}"
190            )));
191        }
192        Ok(selection)
193    }

195    pub(in crate::analysis) fn resolve_section<T: ImagePixel>(
196        image: &dyn ImageInterface<T>,
197        box_arg: Option<&str>,
198        region_arg: Option<SubimageRegion<'_>>,
199        chans_arg: Option<&str>,
200        stokes_arg: Option<&str>,
201    ) -> Result<Self, ImageError> {
202        Self::resolve_with_origin(image, box_arg, region_arg, chans_arg, stokes_arg, None)
203    }

205    pub(super) fn resolve_with_origin<T: ImagePixel>(
206        image: &dyn ImageInterface<T>,
207        box_arg: Option<&str>,
208        region_arg: Option<SubimageRegion<'_>>,
209        chans_arg: Option<&str>,
210        stokes_arg: Option<&str>,
211        text_context: Option<TextRegionContext<'_>>,
212    ) -> Result<Self, ImageError> {
213        let ndim = image.ndim();
214        let full_shape = image.shape().to_vec();
215        let box_arg = box_arg.map(str::trim).filter(|s| !s.is_empty());
216        let plane = if box_arg.is_some() || region_arg.is_some() {
217            direction_plane_axes(image)?
218        } else {
219            [0, 1]
220        };
221
222        let mut regions: Vec<QualifiedShape> = Vec::new();
223        let mut saved_mask = None;
224        let mut saved_bounds: Option<(Vec<usize>, Vec<usize>)> = None;
225        let mut stride = None;
226
227        if let Some(box_text) = box_arg {
228            let PixelRegion::Boxes(boxes) = parse_box_region(box_text, &full_shape, plane)? else {
229                unreachable!("parse_box_region returns Boxes")
230            };
231            for b in boxes {
232                regions.push(SpatialShape::Box(b).into());
233            }
234        }
235
236        let mut is_compound = false;
237        let mut is_crtf = false;
238        if let Some(region) = region_arg {
239            let parsed = match region {
240                SubimageRegion::Text(text)
241                    if text_context.is_some_and(|context| context.shape.is_some()) =>
242                {
243                    parse_crtf_document(image, text, &full_shape, plane, text_context)?
244                }
245                SubimageRegion::Text(text) => {
246                    parse_region_argument(image, text, &full_shape, plane, text_context)?
247                }
248                SubimageRegion::Record(record) => {
249                    parse_saved_region("region", record, image.coordinates(), &full_shape, plane)?
250                }
251            };
252            if parsed.saved_bounds.is_some()
253                && (box_arg.is_some()
254                    || [chans_arg, stokes_arg]
255                        .into_iter()
256                        .flatten()
257                        .any(|v| !v.trim().is_empty()))
258            {
259                return Err(ImageError::CasaSelection(
260                    "saved region cannot be combined with box, chans, or stokes".into(),
261                ));
262            }
263            is_compound = parsed.is_compound;
264            is_crtf = parsed.is_crtf;
265            regions.extend(parsed.shapes);
266            saved_mask = parsed.saved_mask;
267            saved_bounds = parsed.saved_bounds;
268            stride = parsed.stride;
269        }
270
271        let explicit = |selection: Option<&str>| selection.is_some_and(|s| !s.trim().is_empty());
272        for region in &mut regions {
273            if explicit(chans_arg) {
274                region.planes.channels = None;
275            }
276            if explicit(stokes_arg) {
277                region.planes.stokes = None;
278            }
279        }
280        let channel_union = union_planes(&regions, |region| region.planes.channels.as_deref());
281        let stokes_union = union_planes(&regions, |region| region.planes.stokes.as_deref());
282        let spatial_shapes = regions
283            .iter()
284            .filter(|region| !region.excluded)
285            .map(|region| &region.spatial)
286            .collect::<Vec<_>>();
287        let SpatialResolution {
288            mut start,
289            mut shape,
290            spatial_shapes,
291        } = resolve_spatial_and_saved_bounds(
292            &full_shape,
293            ndim,
294            plane,
295            &spatial_shapes,
296            saved_bounds.as_ref(),
297            saved_mask.is_some(),
298        )?;
299
300        let (channel_axis, channel_spans) = resolve_channels(
301            image,
302            &full_shape,
303            chans_arg,
304            channel_union.as_deref(),
305            is_crtf,
306            &mut start,
307            &mut shape,
308        )?;
309
310        let StokesResolution {
311            axis: stokes_axis,
312            indices: stokes_indices,
313            selected: selected_stokes,
314        } = resolve_stokes(
315            image,
316            stokes_arg,
317            stokes_union.as_deref(),
318            &mut start,
319            &mut shape,
320        )?;
321
322        let has_multiple_regions = is_compound
323            || regions.len() > 1
324            || !channel_spans.is_empty()
325            || stokes_axis
326                .zip(stokes_indices.as_ref())
327                .is_some_and(|(a, v)| v.len() != shape[a]);
328        Ok(Self {
329            start,
330            shape,
331            stride,
332            plane,
333            spatial_shapes,
334            qualified_shapes: if regions
335                .iter()
336                .any(|r| r.excluded || r.planes.channels.is_some() || r.planes.stokes.is_some())
337            {
338                regions
339            } else {
340                Vec::new()
341            },
342            saved_mask,
343            channel_axis: channel_axis.or_else(|| image.find_axis(CoordinateType::Spectral)),
344            channel_spans,
345            stokes_axis: stokes_axis.or_else(|| image.find_axis(CoordinateType::Stokes)),
346            stokes_indices,
347            selected_stokes,
348            has_multiple_regions,
349        })
350    }
351}

353fn union_planes(
354    shapes: &[QualifiedShape],
355    indices: impl Fn(&QualifiedShape) -> Option<&[usize]>,
356) -> Option<Vec<usize>> {
357    if shapes.is_empty() {
358        return None;
359    }
360    let mut union = Vec::new();
361    for shape in shapes.iter().filter(|shape| !shape.excluded) {
362        union.extend_from_slice(indices(shape)?);
363    }
364    Some(union)
365}

384fn saved_region_source(text: &str) -> Option<(&Path, &str)> {
385    let (path, name) = text.rsplit_once(':')?;
386    let name = name.trim();
387    let path = Path::new(path);
388    (!path.as_os_str().is_empty() && !name.is_empty() && path.is_dir()).then_some((path, name))
389}

391pub(super) fn external_region_path(text: &str) -> Option<&Path> {
392    let text = text.trim();
393    if let Some((path, _)) = saved_region_source(text) {
394        return Some(path);
395    }
396    if text.starts_with("box:")
397        || (text.starts_with("pixelbox(") && text.ends_with(')'))
398        || looks_like_four_ints(text)
399    {
400        return None;
401    }
402    let path = Path::new(text);
403    path.is_file().then_some(path)
404}

406fn parse_region_argument<T: ImagePixel>(
407    image: &dyn ImageInterface<T>,
408    text: &str,
409    full_shape: &[usize],
410    plane: [usize; 2],
411    text_context: Option<TextRegionContext<'_>>,
412) -> Result<ParsedRegion, ImageError> {
413    if let Some((path, name)) = saved_region_source(text) {
414        return load_saved_region(path, name, image.coordinates(), full_shape, plane);
415    }
416
417    let box_text = text
418        .strip_prefix("box:")
419        .or_else(|| {
420            text.strip_prefix("pixelbox(")
421                .and_then(|value| value.strip_suffix(')'))
422        })
423        .or_else(|| looks_like_four_ints(text).then_some(text));
424    if let Some(box_text) = box_text {
425        let b = parse_single_box(box_text.trim(), full_shape, plane)?;
426        return Ok(ParsedRegion {
427            is_compound: false,
428            is_crtf: false,
429            shapes: vec![SpatialShape::Box(b).into()],
430            saved_mask: None,
431            saved_bounds: None,
432            stride: None,
433        });
434    }
435
436    let reg_path = Path::new(text);
437    let crtf_text = if reg_path.is_file() {
438        let bytes = fs::read(reg_path)
439            .map_err(|e| ImageError::Io(format!("read region file {}: {e}", reg_path.display())))?;
440        if bytes.starts_with(&[0xbe; 4]) {
441            let record = casa_tables::read_table_record_file(reg_path)
442                .map_err(|e| ImageError::Io(format!("read saved region {text}: {e}")))?;
443            return parse_saved_region(text, &record, image.coordinates(), full_shape, plane);
444        }
445        String::from_utf8(bytes)
446            .map_err(|e| ImageError::Io(format!("read CRTF file {text}: {e}")))?
447    } else if looks_like_inline_crtf(text, false) {
448        text.to_string()
449    } else if let Some(path) = image.name().filter(|path| path.is_dir()) {
450        return load_saved_region(path, text, image.coordinates(), full_shape, plane);
451    } else {
452        return Err(ImageError::CasaSelection(format!(
453            "Unable to open region file or region table description {text}."
454        )));
455    };
456
457    parse_crtf_document(image, &crtf_text, full_shape, plane, text_context)
458}

460fn looks_like_four_ints(text: &str) -> bool {
461    let parts: Vec<&str> = text.split(',').collect();
462    parts.len() == 4 && parts.iter().all(|p| p.trim().parse::<usize>().is_ok())
463}

465fn parse_single_box(
466    text: &str,
467    full_shape: &[usize],
468    plane: [usize; 2],
469) -> Result<[usize; 4], ImageError> {
470    let PixelRegion::Boxes(boxes) = parse_box_region(text, full_shape, plane)? else {
471        unreachable!("parse_box_region returns Boxes")
472    };
473    if boxes.len() != 1 {
474        return Err(ImageError::InvalidMetadata(format!(
475            "region box must hold 4 integers x0,y0,x1,y1: {text:?}"
476        )));
477    }
478    Ok(boxes[0])
479}

481fn parse_crtf_qualifiers(
482    tokens: &[&str],
483    coordinates: &CoordinateSystem,
484    full_shape: &[usize],
485    planes: &mut PlaneSelection,
486    text_origin: Option<&CoordinateSystem>,
487) -> Result<(), ImageError> {
488    let axis_for = |coordinates: &CoordinateSystem, kind| {
489        (0..coordinates.n_pixel_axes())
490            .find(|&axis| coordinates.pixel_axis_type(axis) == Some(kind))
491    };
492    for tok in tokens {
493        let trimmed = tok.trim();
494        let Some((key, val)) = trimmed.split_once('=') else {
495            continue;
496        };
497        let key = key.trim().to_ascii_lowercase();
498        let val = val.trim();
499        if key == "corr" {
500            let inner = val
501                .strip_prefix('[')
502                .and_then(|s| s.strip_suffix(']'))
503                .unwrap_or(val);
504            let joined = inner
505                .split(',')
506                .map(str::trim)
507                .filter(|s| !s.is_empty())
508                .collect::<Vec<_>>()
509                .join(",");
510            if !joined.is_empty() {
511                let selected = if let Some(origin) = text_origin {
512                    let missing =
513                        || ImageError::CasaSelection("Exception: No regions found.".into());
514                    let axis = axis_for(origin, CoordinateType::Stokes).ok_or_else(missing)?;
515                    let selected =
516                        parse_stokes_names(origin, axis, &joined).map_err(|_| missing())?;
517                    let origin_values = stokes_for_axis(origin, axis)?;
518                    let axis = axis_for(coordinates, CoordinateType::Stokes).ok_or_else(missing)?;
519                    let input_values = stokes_for_axis(coordinates, axis)?;
520                    let matched = selected
521                        .into_iter()
522                        .filter_map(|index| {
523                            input_values
524                                .iter()
525                                .position(|value| *value == origin_values[index])
526                        })
527                        .collect::<Vec<_>>();
528                    if matched.is_empty() {
529                        return Err(missing());
530                    }
531                    matched
532                } else {
533                    let axis = axis_for(coordinates, CoordinateType::Stokes).ok_or_else(|| {
534                        ImageError::InvalidMetadata(
535                            "Stokes selection requires a Stokes axis".into(),
536                        )
537                    })?;
538                    parse_stokes_names(coordinates, axis, &joined)?
539                };
540                planes.stokes = Some(selected);
541            }
542        } else if key == "range" {
543            let spec_ax = axis_for(coordinates, CoordinateType::Spectral).ok_or_else(|| {
544                ImageError::InvalidMetadata("CRTF range requires a spectral axis".into())
545            })?;
546            let indices = parse_crtf_spectral_range(
547                val,
548                coordinates,
549                spec_ax,
550                full_shape[spec_ax],
551                text_origin,
552            )?;
553            planes.channels = Some(indices);
554        }
555    }
556    Ok(())
557}

559fn parse_crtf_spectral_range(
560    val: &str,
561    coords: &CoordinateSystem,
562    spec_axis: usize,
563    axis_len: usize,
564    text_origin: Option<&CoordinateSystem>,
565) -> Result<Vec<usize>, ImageError> {
566    let [first, last] = crtf_spectral_bounds(val, coords, spec_axis, axis_len, text_origin)?;
567    Ok((first..=last).collect())
568}

570fn crtf_spectral_bounds(
571    val: &str,
572    coords: &CoordinateSystem,
573    spec_axis: usize,
574    axis_len: usize,
575    text_origin: Option<&CoordinateSystem>,
576) -> Result<[usize; 2], ImageError> {
577    let inner = val
578        .strip_prefix('[')
579        .and_then(|s| s.strip_suffix(']'))
580        .unwrap_or(val);
581    let parts: Vec<&str> = inner
582        .split(',')
583        .map(str::trim)
584        .filter(|s| !s.is_empty())
585        .collect();
586    if parts.len() != 2 {
587        return Err(ImageError::InvalidMetadata(format!(
588            "invalid CRTF spectral range [{inner}]"
589        )));
590    }
591    let c0 = parse_spectral_endpoint(parts[0], coords, spec_axis, text_origin)?;
592    let c1 = parse_spectral_endpoint(parts[1], coords, spec_axis, text_origin)?;
593    let lo = c0.min(c1).max(0);
594    let hi = c0.max(c1).min(axis_len as isize - 1);
595    if hi < lo || lo >= axis_len as isize {
596        return Err(ImageError::CasaSelection(format!(
597            "CRTF spectral range [{inner}] selects no channels in 0..{axis_len}"
598        )));
599    }
600    Ok([lo as usize, hi as usize])
601}

603fn channel_position(text: &str) -> Option<&str> {
604    ["channel", "chan", "pix"]
605        .into_iter()
606        .find_map(|unit| text.strip_suffix(unit))
607        .map(str::trim)

610fn nearest_channel(pixel: f64) -> isize {
611    // Match the LCBox intersection at the lower edge: conversion to a
612    // signed integer truncates the shifted position toward zero.
613    (pixel + 0.5) as isize
614}

616fn parse_spectral_endpoint(
617    text: &str,
618    coords: &CoordinateSystem,
619    spec_axis: usize,
620    text_origin: Option<&CoordinateSystem>,
621) -> Result<isize, ImageError> {
622    let lower = text.trim().to_ascii_lowercase();
623    let pixel =
624        if let Some(position) = channel_position(&lower) {
625            Some(position.parse::<f64>().map_err(|e| {
626                ImageError::InvalidMetadata(format!("invalid channel {text:?}: {e}"))
627            })?)
628        } else {
629            lower.parse::<isize>().ok().map(|channel| channel as f64)
630        };
631    if let Some(pixel) = pixel
632        && text_origin.is_none()
633    {
634        return Ok(nearest_channel(pixel));
635    }
636    if let Some(origin) = text_origin {
637        let origin = spectral_coordinate(origin)?;
638        let input = spectral_coordinate(coords)?;
639        let hz = match pixel {
640            Some(pixel) => origin.value_in_hz(origin.to_world(&[pixel])?[0])?,
641            None => Quantity::read(text.trim())
642                .map_err(|e| {
643                    ImageError::InvalidMetadata(format!("invalid CRTF range {text:?}: {e}"))
644                })?
645                .get_value_in(&casa_types::quanta::Unit::new("Hz").expect("known unit"))
646                .map_err(|error| ImageError::InvalidMetadata(error.to_string()))?,
647        };
648        let world = input.value_from_hz(hz)?;
649        return Ok(nearest_channel(input.to_pixel(&[world])?[0]));
650    }
651    let q = Quantity::read(text.trim())
652        .map_err(|e| ImageError::InvalidMetadata(format!("invalid CRTF range {text:?}: {e}")))?;
653    let (index, _) = coords
654        .pixel_axis_owner(spec_axis)
655        .ok_or_else(|| ImageError::InvalidMetadata("CRTF range requires a spectral axis".into()))?;
656    let CoordinateModel::Spectral(spectral) = coords.coordinate(index) else {
657        return Err(ImageError::InvalidMetadata(
658            "CRTF range requires a spectral axis".into(),
659        ));
660    };
661    let world = spectral.value_from_hz(q.get_si_value())?;
662    Ok(nearest_channel(spectral.to_pixel(&[world])?[0]))
663}

665fn spectral_coordinate(coordinates: &CoordinateSystem) -> Result<&SpectralCoordinate, ImageError> {
666    if let Some(index) = coordinates.find_coordinate(CoordinateType::Spectral)
667        && let CoordinateModel::Spectral(coordinate) = coordinates.coordinate(index)
668    {
669        return Ok(coordinate);
670    }
671    Err(ImageError::InvalidMetadata(
672        "CRTF range requires a spectral axis".into(),
673    ))
674}

676fn parse_crtf_shape(
677    coordinates: &CoordinateSystem,
678    stmt: &str,
679    plane: [usize; 2],
680    copy_text: bool,
681) -> Result<SpatialShape, ImageError> {
682    let trimmed = stmt.trim();
683    let Some(bracket) = trimmed.find('[') else {
684        return Err(ImageError::InvalidMetadata(format!(
685            "invalid CRTF region syntax: {trimmed:?}"
686        )));
687    };
688    let keyword = trimmed[..bracket].trim().to_ascii_lowercase();
689    let geometry = &trimmed[bracket..];
690    // CASA's coordinate-pair reader consumes one adjacent character.
691    let geometry = if matches!(keyword.as_str(), "box" | "centerbox" | "poly" | "polygon") {
692        geometry
693            .strip_suffix(')')
694            .filter(|body| body.ends_with(']'))
695            .unwrap_or(geometry)
696    } else {
697        geometry
698    };
699    let body = geometry
700        .strip_prefix('[')
701        .and_then(|s| s.strip_suffix(']'))
702        .ok_or_else(|| {
703            ImageError::InvalidMetadata(format!("unclosed CRTF brackets: {trimmed:?}"))
704        })?;
705    let args = split_top_level(body, ',');
706
707    match keyword.as_str() {
708        "box" => {
709            if args.len() != 2 {
710                return Err(ImageError::InvalidMetadata(format!(
711                    "invalid CRTF box {trimmed:?}"
712                )));
713            }
714            let [x0, y0] = crtf_point_to_pixel(coordinates, args[0], plane)?;
715            let [x1, y1] = crtf_point_to_pixel(coordinates, args[1], plane)?;
716            if copy_text {
717                return Ok(SpatialShape::Polygon(vec![
718                    [x0, y0],
719                    [x1, y0],
720                    [x1, y1],
721                    [x0, y1],
722                ]));
723            }
724            let lo_x = x0.min(x1).max(0.0).round() as usize;
725            let hi_x = x0.max(x1).max(0.0).round() as usize;
726            let lo_y = y0.min(y1).max(0.0).round() as usize;
727            let hi_y = y0.max(y1).max(0.0).round() as usize;
728            Ok(SpatialShape::Box([lo_x, lo_y, hi_x, hi_y]))
729        }
730        "centerbox" => {
731            if args.len() != 2 {
732                return Err(ImageError::InvalidMetadata(format!(
733                    "invalid CRTF centerbox {trimmed:?}"
734                )));
735            }
736            let [cx, cy] = crtf_point_to_pixel(coordinates, args[0], plane)?;
737            let [wx, wy] = crtf_extent_pair_to_pixels(coordinates, args[1])?;
738            Ok(SpatialShape::CenterBox {
739                cx,
740                cy,
741                half_w: wx * 0.5,
742                half_h: wy * 0.5,
743            })
744        }
745        "rotbox" => {
746            if args.len() != 3 {
747                return Err(ImageError::InvalidMetadata(format!(
748                    "invalid CRTF rotbox {trimmed:?}"
749                )));
750            }
751            let [cx, cy] = crtf_point_to_pixel(coordinates, args[0], plane)?;
752            let [wx, wy] = crtf_extent_pair_to_pixels(coordinates, args[1])?;
753            let pa = parse_world_angle(args[2])?;
754            Ok(SpatialShape::RotBox {
755                cx,
756                cy,
757                half_w: wx * 0.5,
758                half_h: wy * 0.5,
759                cos_pa: pa.cos(),
760                sin_pa: pa.sin(),
761            })
762        }
763        "circle" => {
764            if args.len() != 2 {
765                return Err(ImageError::InvalidMetadata(format!(
766                    "invalid CRTF circle {trimmed:?}"
767                )));
768            }
769            let [cx, cy] = crtf_point_to_pixel(coordinates, args[0], plane)?;
770            let [rx, ry] = crtf_radius_to_pixels(coordinates, args[1])?;
771            if copy_text {
772                if let Some(radius) = parse_crtf_pix(args[1])?
773                    && let Some([sx, sy]) = direction_pixel_scales(coordinates)
774                    && (sx / sy - 1.0).abs() > 1e-9
775                {
776                    // CASA's CRTF parser expresses a pixel-radius circle on a
777                    // non-square grid as an axis-aligned ellipse.
778                    return Ok(SpatialShape::Ellipse {
779                        cx,
780                        cy,
781                        semi_major: radius * sx / sy,
782                        semi_minor: radius * sy / sx,
783                        cos_pa: 0.0,
784                        sin_pa: 1.0,
785                    });
786                }
787                let radius = if plane[0] < plane[1] { rx } else { ry };
788                return Ok(SpatialShape::Circle {
789                    cx,
790                    cy,
791                    rx: radius,
792                    ry: radius,
793                });
794            }
795            Ok(SpatialShape::Circle { cx, cy, rx, ry })
796        }
797        "ellipse" => {
798            if args.len() != 3 {
799                return Err(ImageError::InvalidMetadata(format!(
800                    "invalid CRTF ellipse {trimmed:?}"
801                )));
802            }
803            let [cx, cy] = crtf_point_to_pixel(coordinates, args[0], plane)?;
804            let [a, b] = crtf_extent_pair_to_pixels(coordinates, args[1])?;
805            let pa = parse_world_angle(args[2])?;
806            Ok(SpatialShape::Ellipse {
807                cx,
808                cy,
809                semi_major: a,
810                semi_minor: b,
811                cos_pa: pa.cos(),
812                sin_pa: pa.sin(),
813            })
814        }
815        "poly" | "polygon" => {
816            if args.len() < 3 {
817                return Err(ImageError::InvalidMetadata(format!(
818                    "CRTF polygon requires at least 3 vertices: {trimmed:?}"
819                )));
820            }
821            let mut verts = Vec::with_capacity(args.len());
822            for arg in args {
823                verts.push(crtf_point_to_pixel(coordinates, arg, plane)?);
824            }
825            Ok(SpatialShape::Polygon(verts))
826        }
827        _ => Err(ImageError::InvalidMetadata(format!(
828            "unsupported CRTF region keyword {keyword:?}"
829        ))),
830    }
831}

833fn crtf_point_to_pixel(
834    coordinates: &CoordinateSystem,
835    pair_text: &str,
836    plane: [usize; 2],
837) -> Result<[f64; 2], ImageError> {
838    let inner = pair_text
839        .trim()
840        .strip_prefix('[')
841        .and_then(|s| s.strip_suffix(']'))
842        .ok_or_else(|| {
843            ImageError::InvalidMetadata(format!(
844                "expected [x, y] coordinate pair, got {pair_text:?}"
845            ))
846        })?;
847    let parts = split_top_level(inner, ',');
848    if parts.len() != 2 {
849        return Err(ImageError::InvalidMetadata(format!(
850            "expected [x, y] coordinate pair, got {pair_text:?}"
851        )));
852    }
853    let px = parse_crtf_pix(parts[0])?;
854    let py = parse_crtf_pix(parts[1])?;
855    match (px, py) {
856        (Some(x), Some(y)) => Ok([x, y]),
857        (None, None) => {
858            if coordinates.n_pixel_axes() < 2 || coordinates.n_world_axes() < 2 {
859                return Err(ImageError::InvalidMetadata(
860                    "world CRTF regions require at least 2 coordinate axes".into(),
861                ));
862            }
863            let [ax, ay] = plane;
864            let [longitude, latitude] =
865                super::super::axes::world_axes_for_plane(coordinates, plane)?;
866            let base = vec![0.0; coordinates.n_pixel_axes()];
867            let mut world = coordinates.to_world(&base)?;
868            world[longitude] = parse_world_angle(parts[0])?;
869            world[latitude] = parse_world_angle(parts[1])?;
870            let pix = coordinates.to_pixel(&world)?;
871            Ok([pix[ax], pix[ay]])
872        }
873        _ => Err(ImageError::InvalidMetadata(
874            "cannot mix pix and world coordinates in a CRTF point".into(),
875        )),
876    }
877}
878
879fn crtf_extent_pair_to_pixels(
880    coordinates: &CoordinateSystem,
881    pair_text: &str,
882) -> Result<[f64; 2], ImageError> {
883    let inner = pair_text
884        .trim()
885        .strip_prefix('[')
886        .and_then(|s| s.strip_suffix(']'))
887        .ok_or_else(|| {
888            ImageError::InvalidMetadata(format!("expected [wx, wy] pair, got {pair_text:?}"))
889        })?;
890    let parts = split_top_level(inner, ',');
891    if parts.len() != 2 {
892        return Err(ImageError::InvalidMetadata(format!(
893            "expected [wx, wy] pair, got {pair_text:?}"
894        )));
895    }
896    let scale = direction_pixel_scales(coordinates);
897    Ok([
898        parse_extent_single(parts[0], scale.map(|s| s[0]))?,
899        parse_extent_single(parts[1], scale.map(|s| s[1]))?,
900    ])
901}
902
903fn crtf_radius_to_pixels(
904    coordinates: &CoordinateSystem,
905    text: &str,
906) -> Result<[f64; 2], ImageError> {
907    if let Some(r) = parse_crtf_pix(text)? {
908        return Ok([r, r]);
909    }
910    let rad = parse_world_angle(text)?;
911    let [sx, sy] = direction_pixel_scales(coordinates).ok_or_else(|| {
912        ImageError::InvalidMetadata(
913            "angular CRTF radius requires a Direction or 2D linear coordinate".into(),
914        )
915    })?;
916    Ok([rad / sx, rad / sy])
917}
918
919fn parse_extent_single(text: &str, scale_rad_per_px: Option<f64>) -> Result<f64, ImageError> {
920    if let Some(px) = parse_crtf_pix(text)? {
921        return Ok(px);
922    }
923    let rad = parse_world_angle(text)?;
924    let scale = scale_rad_per_px.ok_or_else(|| {
925        ImageError::InvalidMetadata(
926            "angular CRTF size requires non-zero direction pixel scale".into(),
927        )
928    })?;
929    Ok(rad / scale)
930}
931
932fn direction_pixel_scales(coords: &CoordinateSystem) -> Option<[f64; 2]> {
933    for idx in 0..coords.n_coordinates() {
934        let coord = coords.coordinate(idx);
935        let np = coord.n_pixel_axes();
936        if coord.coordinate_type() == CoordinateType::Direction && np >= 2 {
937            let inc = coord.increment();
938            let sx = inc[0].abs();
939            let sy = inc[1].abs();
940            if sx > 0.0 && sy > 0.0 {
941                return Some([sx, sy]);
942            }
943        }
944    }
945    None
946}

948fn parse_crtf_pix(text: &str) -> Result<Option<f64>, ImageError> {
949    let lower = text.trim().to_ascii_lowercase();
950    let Some(num) = lower.strip_suffix("pix") else {
951        return Ok(None);
952    };
953    let val = num.trim().parse::<f64>().map_err(|e| {
954        ImageError::InvalidMetadata(format!("invalid CRTF pixel value {text:?}: {e}"))
955    })?;
956    if !val.is_finite() {
957        return Err(ImageError::InvalidMetadata(format!(
958            "non-finite CRTF pixel value {text:?}"
959        )));
960    }
961    Ok(Some(val))
962}

964pub(in crate::analysis) fn parse_world_angle(text: &str) -> Result<f64, ImageError> {
965    let trimmed = text.trim();
966    if trimmed.contains(':') {
967        return parse_colon_hms(trimmed);
968    }
969    if is_dot_dms(trimmed) {
970        return parse_dot_dms(trimmed);
971    }
972    if let Ok(rad) = trimmed.parse::<f64>() {
973        return if rad.is_finite() {
974            Ok(rad)
975        } else {
976            Err(ImageError::InvalidMetadata(format!(
977                "non-finite angle {text:?}"
978            )))
979        };
980    }
981    let q = Quantity::read(trimmed)
982        .map_err(|e| ImageError::InvalidMetadata(format!("invalid CRTF angle {text:?}: {e}")))?;
983    mvangle_radians(&q)
984        .map_err(|e| ImageError::InvalidMetadata(format!("invalid CRTF angle {text:?}: {e}")))
985}
986
987fn parse_colon_hms(text: &str) -> Result<f64, ImageError> {
988    let (sign, rest) = strip_sign(text);
989    let parts: Vec<&str> = rest.split(':').collect();
990    if !(2..=3).contains(&parts.len()) {
991        return Err(ImageError::InvalidMetadata(format!("invalid HMS {text:?}")));
992    }
993    let h = parts[0]
994        .parse::<f64>()
995        .map_err(|e| ImageError::InvalidMetadata(format!("invalid HMS hours in {text:?}: {e}")))?;
996    let m = parts[1].parse::<f64>().map_err(|e| {
997        ImageError::InvalidMetadata(format!("invalid HMS minutes in {text:?}: {e}"))
998    })?;
999    let s = if parts.len() == 3 {
1000        parts[2].parse::<f64>().map_err(|e| {
1001            ImageError::InvalidMetadata(format!("invalid HMS seconds in {text:?}: {e}"))
1002        })?
1003    } else {
1004        0.0
1005    };
1006    let hours = sign * (h + m / 60.0 + s / 3600.0);
1007    Ok((hours * 15.0).to_radians())
1008}

1010fn is_dot_dms(text: &str) -> bool {
1011    let (_, rest) = strip_sign(text);
1012    rest.matches('.').count() >= 2 && rest.chars().all(|c| c.is_ascii_digit() || c == '.')
1013}
1014
1015fn parse_dot_dms(text: &str) -> Result<f64, ImageError> {
1016    let (sign, rest) = strip_sign(text);
1017    let parts: Vec<&str> = rest.splitn(3, '.').collect();
1018    let d = parts[0].parse::<f64>().map_err(|e| {
1019        ImageError::InvalidMetadata(format!("invalid DMS degrees in {text:?}: {e}"))
1020    })?;
1021    let m = parts[1]
1022        .parse::<f64>()
1023        .map_err(|e| ImageError::InvalidMetadata(format!("invalid DMS arcmin in {text:?}: {e}")))?;
1024    let s = parts[2]
1025        .parse::<f64>()
1026        .map_err(|e| ImageError::InvalidMetadata(format!("invalid DMS arcsec in {text:?}: {e}")))?;
1027    let deg = sign * (d + m / 60.0 + s / 3600.0);
1028    Ok(deg.to_radians())
1029}
1030
1031fn strip_sign(text: &str) -> (f64, &str) {
1032    let t = text.trim();
1033    if let Some(rest) = t.strip_prefix('-') {
1034        (-1.0, rest.trim())
1035    } else if let Some(rest) = t.strip_prefix('+') {
1036        (1.0, rest.trim())
1037    } else {
1038        (1.0, t)
1039    }
1040}
1041
1042fn split_top_level(text: &str, sep: char) -> Vec<&str> {
1043    let mut parts = Vec::new();
1044    let mut depth = 0i32;
1045    let mut start = 0usize;
1046    for (idx, ch) in text.char_indices() {
1047        match ch {
1048            '[' | '(' => depth += 1,
1049            ']' | ')' => depth = (depth - 1).max(0),
1050            c if c == sep && depth == 0 => {
1051                let piece = text[start..idx].trim();
1052                if !piece.is_empty() {
1053                    parts.push(piece);
1054                }
1055                start = idx + ch.len_utf8();
1056            }
1057            _ => {}
1058        }
1059    }
1060    let tail = text[start..].trim();
1061    if !tail.is_empty() {
1062        parts.push(tail);
1063    }
1064    parts
1065}

1067struct SpatialResolution {
1068    start: Vec<usize>,
1069    shape: Vec<usize>,
1070    spatial_shapes: Vec<SpatialShape>,
1071}
1072
1073fn resolve_spatial_and_saved_bounds(
1074    full_shape: &[usize],
1075    ndim: usize,
1076    plane: [usize; 2],
1077    shapes: &[&SpatialShape],
1078    saved_bounds: Option<&(Vec<usize>, Vec<usize>)>,
1079    has_saved_mask: bool,
1080) -> Result<SpatialResolution, ImageError> {
1081    let mut start = vec![0usize; ndim];
1082    let mut shape = full_shape.to_vec();
1083
1084    if let Some((sb_start, sb_shape)) = saved_bounds {
1085        start = sb_start.clone();
1086        shape = sb_shape.clone();
1087    }
1088
1089    if shapes.is_empty() {
1090        return Ok(SpatialResolution {
1091            start,
1092            shape,
1093            spatial_shapes: Vec::new(),
1094        });
1095    }
1096
1097    if ndim < 2 {
1098        return Err(ImageError::ShapeMismatch {
1099            expected: full_shape.to_vec(),
1100            got: vec![2],
1101        });
1102    }
1103    let [px, py] = plane;
1104    let nx = full_shape[px];
1105    let ny = full_shape[py];
1106
1107    // Fast path for single box with no saved WCPolygon.
1108    if shapes.len() == 1
1109        && !has_saved_mask
1110        && let SpatialShape::Box([x0, y0, x1, y1]) = *shapes[0]
1111    {
1112        if x1 >= nx || y1 >= ny || x1 < x0 || y1 < y0 {
1113            return Err(ImageError::CasaSelection(
1114                "selected box exceeds image bounds".into(),
1115            ));
1116        }
1117        start[px] = x0;
1118        start[py] = y0;
1119        shape[px] = x1 - x0 + 1;
1120        shape[py] = y1 - y0 + 1;
1121        return Ok(SpatialResolution {
1122            start,
1123            shape,
1124            spatial_shapes: Vec::new(),
1125        });
1126    }
1127    if saved_bounds.is_some() && !has_saved_mask {
1128        return Ok(SpatialResolution {
1129            start,
1130            shape,
1131            spatial_shapes: shapes.iter().copied().cloned().collect(),
1132        });
1133    }
1134
1135    // Union bounding box + exact spatial mask over included integer pixels.
1136    let mut min_x = nx;
1137    let mut min_y = ny;
1138    let mut max_x = 0usize;
1139    let mut max_y = 0usize;
1140    let mut any_pixel = false;
1141
1142    for s in shapes {
1143        let (x0, y0, x1, y1) = shape_candidate_bounds(s, nx, ny);
1144        if let SpatialShape::Box(_) = s {
1145            if x0 <= x1 && y0 <= y1 && shape_contains(s, x0, y0) {
1146                min_x = min_x.min(x0);
1147                min_y = min_y.min(y0);
1148                max_x = max_x.max(x1);
1149                max_y = max_y.max(y1);
1150                any_pixel = true;
1151            }
1152            continue;
1153        }
1154        let mut has_pixels = false;
1155        for x in x0..=x1 {
1156            for y in y0..=y1 {
1157                if shape_contains(s, x, y) {
1158                    has_pixels = true;
1159                    min_x = min_x.min(x);
1160                    min_y = min_y.min(y);
1161                    max_x = max_x.max(x);
1162                    max_y = max_y.max(y);
1163                }
1164            }
1165        }
1166        if has_pixels {
1167            any_pixel = true;
1168            // LCEllipsoid uses a square enclosing box for a rotated ellipse,
1169            // including rows and columns that its mask excludes entirely.
1170            if let SpatialShape::Ellipse {
1171                cx,
1172                cy,
1173                semi_major,
1174                semi_minor,
1175                cos_pa,
1176                sin_pa,
1177            } = s
1178                && cos_pa.abs() > 1e-7
1179                && sin_pa.abs() > 1e-7
1180            {
1181                let r = semi_major.max(*semi_minor);
1182                let lower = |c: f64| (c - r - 1e-5).ceil().max(0.0) as usize;
1183                let upper = |c: f64, n: usize| ((c + r + 1e-5).floor() as usize).min(n - 1);
1184                min_x = min_x.min(lower(*cx));
1185                min_y = min_y.min(lower(*cy));
1186                max_x = max_x.max(upper(*cx, nx));
1187                max_y = max_y.max(upper(*cy, ny));
1188            }
1189        }
1190    }
1191
1192    if !any_pixel {
1193        return Err(empty_section());
1194    }
1195
1196    if has_saved_mask {
1197        // When both box and saved WCPolygon exist, form union over their planes.
1198        min_x = min_x.min(start[px]);
1199        min_y = min_y.min(start[py]);
1200        max_x = max_x.max(start[px] + shape[px] - 1);
1201        max_y = max_y.max(start[py] + shape[py] - 1);
1202    }
1203
1204    start[px] = min_x;
1205    start[py] = min_y;
1206    shape[px] = max_x - min_x + 1;
1207    shape[py] = max_y - min_y + 1;
1208
1209    Ok(SpatialResolution {
1210        start,
1211        shape,
1212        spatial_shapes: shapes.iter().copied().cloned().collect(),
1213    })
1214}

1216fn shape_candidate_bounds(
1217    shape: &SpatialShape,
1218    nx: usize,
1219    ny: usize,
1220) -> (usize, usize, usize, usize) {
1221    let clamp_x = |v: f64| v.clamp(0.0, nx.saturating_sub(1) as f64) as usize;
1222    let clamp_y = |v: f64| v.clamp(0.0, ny.saturating_sub(1) as f64) as usize;
1223    match shape {
1224        SpatialShape::Box([x0, y0, x1, y1]) => (
1225            (*x0).min(nx.saturating_sub(1)),
1226            (*y0).min(ny.saturating_sub(1)),
1227            (*x1).min(nx.saturating_sub(1)),
1228            (*y1).min(ny.saturating_sub(1)),
1229        ),
1230        SpatialShape::CenterBox {
1231            cx,
1232            cy,
1233            half_w,
1234            half_h,
1235        } => (
1236            clamp_x((cx - half_w - 1.0).floor()),
1237            clamp_y((cy - half_h - 1.0).floor()),
1238            clamp_x((cx + half_w + 1.0).ceil()),
1239            clamp_y((cy + half_h + 1.0).ceil()),
1240        ),
1241        SpatialShape::RotBox {
1242            cx,
1243            cy,
1244            half_w,
1245            half_h,
1246            cos_pa,
1247            sin_pa,
1248        } => {
1249            let dx = half_w * cos_pa.abs() + half_h * sin_pa.abs() + 1.0;
1250            let dy = half_w * sin_pa.abs() + half_h * cos_pa.abs() + 1.0;
1251            (
1252                clamp_x((cx - dx).floor()),
1253                clamp_y((cy - dy).floor()),
1254                clamp_x((cx + dx).ceil()),
1255                clamp_y((cy + dy).ceil()),
1256            )
1257        }
1258        SpatialShape::Circle { cx, cy, rx, ry } => (
1259            clamp_x((cx - rx - 1.0).floor()),
1260            clamp_y((cy - ry - 1.0).floor()),
1261            clamp_x((cx + rx + 1.0).ceil()),
1262            clamp_y((cy + ry + 1.0).ceil()),
1263        ),
1264        SpatialShape::Ellipse {
1265            cx,
1266            cy,
1267            semi_major,
1268            semi_minor,
1269            ..
1270        } => {
1271            let r = semi_major.max(*semi_minor) + 1.0;
1272            (
1273                clamp_x((cx - r).floor()),
1274                clamp_y((cy - r).floor()),
1275                clamp_x((cx + r).ceil()),
1276                clamp_y((cy + r).ceil()),
1277            )
1278        }
1279        SpatialShape::Polygon(verts) => {
1280            let min_x = verts.iter().map(|v| v[0]).fold(f64::INFINITY, f64::min) - 1.0;
1281            let min_y = verts.iter().map(|v| v[1]).fold(f64::INFINITY, f64::min) - 1.0;
1282            let max_x = verts.iter().map(|v| v[0]).fold(f64::NEG_INFINITY, f64::max) + 1.0;
1283            let max_y = verts.iter().map(|v| v[1]).fold(f64::NEG_INFINITY, f64::max) + 1.0;
1284            (
1285                clamp_x(min_x.floor()),
1286                clamp_y(min_y.floor()),
1287                clamp_x(max_x.ceil()),
1288                clamp_y(max_y.ceil()),
1289            )
1290        }
1291    }
1292}

1294pub(super) fn shape_contains(shape: &SpatialShape, x: usize, y: usize) -> bool {
1295    let xf = x as f64;
1296    let yf = y as f64;
1297    match shape {
1298        SpatialShape::Box([x0, y0, x1, y1]) => (*x0..=*x1).contains(&x) && (*y0..=*y1).contains(&y),
1299        SpatialShape::CenterBox {
1300            cx,
1301            cy,
1302            half_w,
1303            half_h,
1304        } => (xf - cx).abs() <= half_w + 1e-9 && (yf - cy).abs() <= half_h + 1e-9,
1305        SpatialShape::RotBox {
1306            cx,
1307            cy,
1308            half_w,
1309            half_h,
1310            cos_pa,
1311            sin_pa,
1312        } => {
1313            let dx = xf - cx;
1314            let dy = yf - cy;
1315            let u = dx * cos_pa + dy * sin_pa;
1316            let v = -dx * sin_pa + dy * cos_pa;
1317            u.abs() <= half_w + 1e-9 && v.abs() <= half_h + 1e-9
1318        }
1319        SpatialShape::Circle { cx, cy, rx, ry } => {
1320            if *rx <= 0.0 || *ry <= 0.0 {
1321                return false;
1322            }
1323            ((xf - cx) / rx).powi(2) + ((yf - cy) / ry).powi(2) <= 1.0 + 1e-9
1324        }
1325        SpatialShape::Ellipse {
1326            cx,
1327            cy,
1328            semi_major,
1329            semi_minor,
1330            cos_pa,
1331            sin_pa,
1332        } => {
1333            if *semi_major <= 0.0 || *semi_minor <= 0.0 {
1334                return false;
1335            }
1336            let dx = xf - cx;
1337            let dy = yf - cy;
1338            let along_major = -dx * sin_pa + dy * cos_pa;
1339            let along_minor = dx * cos_pa + dy * sin_pa;
1340            (along_major / semi_major).powi(2) + (along_minor / semi_minor).powi(2) <= 1.0 + 1e-12
1341        }
1342        SpatialShape::Polygon(verts) => point_in_polygon_f64(xf, yf, verts),
1343    }
1344}
1345
1346fn point_in_polygon_f64(x: f64, y: f64, vertices: &[[f64; 2]]) -> bool {
1347    let mut inside = false;
1348    let mut prev = vertices[vertices.len() - 1];
1349    for &cur in vertices {
1350        let (x0, y0) = (prev[0], prev[1]);
1351        let (x1, y1) = (cur[0], cur[1]);
1352        let cross = (x - x0) * (y1 - y0) - (y - y0) * (x1 - x0);
1353        if cross.abs() <= 1e-9
1354            && x >= x0.min(x1) - 1e-9
1355            && x <= x0.max(x1) + 1e-9
1356            && y >= y0.min(y1) - 1e-9
1357            && y <= y0.max(y1) + 1e-9
1358        {
1359            return true;
1360        }
1361        if (y0 > y) != (y1 > y) {
1362            let x_int = (x1 - x0) * (y - y0) / (y1 - y0) + x0;
1363            if x <= x_int {
1364                inside = !inside;
1365            }
1366        }
1367        prev = cur;
1368    }
1369    inside
1370}

1372struct StokesResolution {
1373    axis: Option<usize>,
1374    indices: Option<Vec<usize>>,
1375    selected: Option<Vec<StokesType>>,
1376}
1377
1378fn resolve_stokes(
1379    image: &dyn ImageInterface<impl ImagePixel>,
1380    stokes_arg: Option<&str>,
1381    crtf_stokes: Option<&[usize]>,
1382    start: &mut [usize],
1383    shape: &mut [usize],
1384) -> Result<StokesResolution, ImageError> {
1385    let arg_trimmed = stokes_arg.map(str::trim).filter(|s| !s.is_empty());
1386    if arg_trimmed.is_none() && crtf_stokes.is_none() {
1387        return Ok(StokesResolution {
1388            axis: None,
1389            indices: None,
1390            selected: None,
1391        });
1392    }
1393    let axis = image.find_axis(CoordinateType::Stokes).ok_or_else(|| {
1394        ImageError::InvalidMetadata("Stokes selection requires a Stokes axis".into())
1395    })?;
1396    let mut indices = if let Some(selection) = arg_trimmed {
1397        parse_stokes_selection(image, selection)?
1398    } else {
1399        crtf_stokes.expect("at least one Stokes selection").to_vec()
1400    };
1401    indices.sort_unstable();
1402    indices.dedup();
1403    let prev_lo = start[axis];
1404    let prev_hi = start[axis] + shape[axis] - 1;
1405    let bounded: Vec<usize> = indices
1406        .into_iter()
1407        .filter(|&s| s >= prev_lo && s <= prev_hi)
1408        .collect();
1409    if bounded.is_empty() {
1410        return Err(ImageError::CasaSelection(
1411            "selected Stokes planes lie outside saved region".into(),
1412        ));
1413    }
1414    start[axis] = bounded[0];
1415    shape[axis] = bounded[bounded.len() - 1] - bounded[0] + 1;
1416    let full_stokes = stokes_for_axis(image.coordinates(), axis)?;
1417    let kept = full_stokes[bounded[0]..=bounded[bounded.len() - 1]].to_vec();
1418    Ok(StokesResolution {
1419        axis: Some(axis),
1420        indices: Some(bounded),
1421        selected: Some(kept),
1422    })
1423}
