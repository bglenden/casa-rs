16macro_rules! native_array {
17    ($handle:expr, $image:ident => $read:expr) => {
18        match $handle {
19            ImageHandle::Float(storage) => {
20                with_storage!(storage, $image => $read.map(ArrayValue::Float32))
21            }
22            ImageHandle::Double(storage) => {
23                with_storage!(storage, $image => $read.map(ArrayValue::Float64))
24            }
25            ImageHandle::Complex(storage) => {
26                with_storage!(storage, $image => $read.map(ArrayValue::Complex32))
27            }
28            ImageHandle::DComplex(storage) => {
29                with_storage!(storage, $image => $read.map(ArrayValue::Complex64))
30            }
31        }
32    };
33}

35/// Opens a directory-backed native image directly using the attachment registry (`_data_open`).
36pub fn open_data_image(path: &Path, writable: bool) -> ImageToolResult<AttachedImage> {
37    if !path.is_dir() {
38        return Err(ImageToolError::new(format!(
39            "Cannot open native image {}",
40            path.display()
41        )));
42    }
43    let path = std::path::absolute(path)?;
44    AttachedImage::open_data(&mut attachment::registry(), &path, writable)
45}

80/// Computes direction WCS axes `(lon_axis, lat_axis)` and FITS header if the image has a direction axis matching pixel rank.
81pub fn data_direction_wcs(
82    attached: &AttachedImage,
83) -> ImageToolResult<Option<(usize, usize, FitsHeader)>> {
84    let image = attached.lock();
85    let longitude = with_pixels!(
86        &*image,
87        storage => with_storage!(storage, img => img.find_axis(CoordinateType::Direction))
88    );
89    let coordinates = image.coordinates();
90    let shape = image.shape();
91    let Some(longitude) = longitude.filter(|_| coordinates.n_pixel_axes() == shape.len()) else {
92        return Ok(None);
93    };
94    let header = try_to_fits_header(coordinates, shape)?;
95    Ok(Some((longitude, longitude + 1, header)))
96}

145/// Writes a typed pixel array into `attached` starting at `start` and saves (`_data_put_slice`).
146pub fn data_put_slice(
147    attached: &AttachedImage,
148    data: ArrayValue,
149    start: &[usize],
150) -> ImageToolResult<()> {
151    attached.check_directory()?;
152    let mut image = attached.lock();
153    match (&mut *image, data) {
154        (ImageHandle::Float(storage), ArrayValue::Float32(values)) => {
155            with_storage!(storage, img => img.put_slice(&values, start))
156        }
157        (ImageHandle::Double(storage), ArrayValue::Float64(values)) => {
158            with_storage!(storage, img => img.put_slice(&values, start))
159        }
160        (ImageHandle::Complex(storage), ArrayValue::Complex32(values)) => {
161            with_storage!(storage, img => img.put_slice(&values, start))
162        }
163        (ImageHandle::DComplex(storage), ArrayValue::Complex64(values)) => {
164            with_storage!(storage, img => img.put_slice(&values, start))
165        }
166        _ => {
167            return Err(ImageToolError::new(
168                "Image pixel type changed before the write",
169            ));
170        }
171    }?;
172    image.save()?;
173    Ok(())
174}

176/// Normalizes an optional stride vector against `shape` rank and positivity rules.
177pub fn normalize_stride(
178    shape: &[usize],
179    stride: Option<Vec<usize>>,
180) -> ImageToolResult<Vec<usize>> {
181    let stride = stride.unwrap_or_else(|| vec![1; shape.len()]);
182    if stride.len() != shape.len() {
183        return Err(ImageToolError::new(format!(
184            "stride rank {} does not match shape rank {}",
185            stride.len(),
186            shape.len()
187        )));
188    }
189    if stride.contains(&0) {
190        return Err(ImageToolError::new("stride values must be >= 1"));
191    }
192    Ok(stride)
193}
