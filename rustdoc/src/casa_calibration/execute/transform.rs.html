2//! Bounded, read-only application of a calibration chain during mstransform.
3
4use super::*;
5
6pub(crate) struct PreparedReadOnlyApply {
7    plan: ApplyPlan,
8    chain: ApplyChain,
9    row_positions: HashMap<usize, usize>,
10    correlation_types: HashMap<i32, Vec<i32>>,
11    observed_column: &'static str,
12    has_sigma_spectrum: bool,
13    has_weight_column: bool,
14}
15
16impl PreparedReadOnlyApply {
17    pub(crate) fn new(ms: &MeasurementSet, plan: ApplyPlan) -> Result<Self, ApplyExecutionError> {
18        let chain = ApplyChain::load(ms, &plan, false)?;
19        let correlation_types = load_correlation_types_by_ddid(ms).map_err(|source| {
20            ApplyExecutionError::MutateMeasurementSet {
21                path: display_ms_path(ms),
22                source,
23            }
24        })?;
25        let row_positions = plan
26            .selected_rows
27            .iter()
28            .enumerate()
29            .map(|(position, row)| (row.row_index, position))
30            .collect();
31        let first = plan.selected_rows.iter().map(|row| row.row_index).min();
32        let table = ms.main_table();
33        Ok(Self {
34            plan,
35            chain,
36            row_positions,
37            correlation_types,
38            observed_column: apply_observed_column(ms, false),
39            has_sigma_spectrum: weights::has_spectrum(table, first, "SIGMA_SPECTRUM")?,
40            has_weight_column: table
41                .schema()
42                .is_some_and(|schema| schema.contains_column("WEIGHT_SPECTRUM")),
43        })
44    }
45
46    pub(crate) fn visit_rows(
47        &self,
48        table: &Table,
49        rows: &[usize],
50        mut visit: impl FnMut(ArrayValue, ArrayValue, PlotWeights),
51    ) -> Result<(), ApplyExecutionError> {
52        let path = table
53            .path()
54            .map_or_else(|| "<memory>".into(), |path| path.display().to_string());
55        let arrays = SelectedArrayReader { table, path: &path };
56        let flags = arrays.read("FLAG", rows)?;
57        let data = arrays.read(self.observed_column, rows)?;
58        let sigmas = arrays.read("SIGMA", rows)?;
59        let mut geometry = RowGeometryCache::default();
60        for (((&row, flags), data), sigma) in rows.iter().zip(flags).zip(data).zip(sigmas) {
61            let row_plan = self
62                .row_positions
63                .get(&row)
64                .map(|&position| &self.plan.selected_rows[position])
65                .ok_or_else(|| ApplyExecutionError::MutateMeasurementSet {
66                    path: path.clone(),
67                    source: MsError::InvalidInput(format!(
68                        "calibration did not plan input row {row}"
69                    )),
70                })?;
71            let flags = required_prefetched_array(flags, row, "FLAG", &path)?;
72            let data = required_prefetched_array(data, row, self.observed_column, &path)?;
73            let data = match data {
74                ArrayValue::Float32(values) => complex_visibility_from_float(&values),
75                data => data,
76            };
77            let sigma = required_prefetched_array(sigma, row, "SIGMA", &path)?;
78            let (corrected, flags, weights) =
79                self.apply_row(row_plan, flags, &data, sigma, &arrays, &mut geometry)?;
80            visit(corrected, flags, weights);
81        }
82        Ok(())
83    }
84
85    fn apply_row(
86        &self,
87        row: &ApplyRowPlan,
88        flags: ArrayValue,
89        data: &ArrayValue,
90        sigma: ArrayValue,
91        arrays: &SelectedArrayReader<'_>,
92        geometry: &mut RowGeometryCache,
93    ) -> Result<(ArrayValue, ArrayValue, PlotWeights), ApplyExecutionError> {
94        let layout = self
95            .correlation_types
96            .get(&row.data_desc_id)
97            .ok_or_else(|| ApplyExecutionError::UnsupportedCorrelationLayout {
98                data_desc_id: row.data_desc_id,
99                correlation_types: Vec::new(),
100            })?;
101        let indices = self
102            .plan
103            .channel_selections
104            .get(&row.data_spw_id)
105            .map(|selection| selection.indices.as_slice());
106        let mut scalar = weights::from_sigma(sigma, "SIGMA")?;
107        let spectrum = self.weight_spectrum(arrays, row.row_index, &scalar, data.shape())?;
108        if let Some(weights) = &spectrum {
109            scalar = weights::scalar_medians(weights, indices).unwrap_or(scalar);
110        }
111        let result = apply_row(
112            row,
113            ExecutionRowInputs {
114                correlation_types: layout,
115                data,
116                original_flags: Some(&flags),
117                original_weight: Some(&scalar),
118                original_weight_spectrum: spectrum.as_ref(),
119                has_weight_spectrum: spectrum.is_some(),
120                preserved_corrected_data: None,
121                preserved_weight_spectrum: None,
122                channel_grid: None,
123                forward_model: false,
124                // VisCal::correct2 skips an unavailable term and continues
125                // with the rest of the callibrary, leaving flags to those terms.
126                skip_unavailable_terms: true,
127                trial: false,
128                parang_calwt: false,
129            },
130            &self.plan,
131            &self.chain,
132            Some(geometry),
133            None,
134            None,
135        )?;
136        let spectrum = result.updated_weight_spectrum.or(spectrum);
137        let mut scalar = result.updated_weight.unwrap_or(scalar);
138        if let Some(weights) = &spectrum {
139            scalar = weights::scalar_medians(weights, indices).unwrap_or(scalar);
140        }
141        Ok((
142            result.corrected_data,
143            result.updated_flags.unwrap_or(flags),
144            PlotWeights { scalar, spectrum },
145        ))
146    }
147
148    fn weight_spectrum(
149        &self,
150        arrays: &SelectedArrayReader<'_>,
151        row: usize,
152        scalar: &ArrayValue,
153        shape: &[usize],
154    ) -> Result<Option<ArrayValue>, ApplyExecutionError> {
155        if !self.has_weight_column
156            || arrays
157                .table
158                .array_shape_uncached(row, "WEIGHT_SPECTRUM")
159                .map_err(|source| arrays.error(source))?
160                .is_none()
161        {
162            return Ok(None);
163        }
164        let value = if self.has_sigma_spectrum {
165            let column = "SIGMA_SPECTRUM";
166            let selected = arrays.read(column, &[row])?.pop().flatten();
167            weights::from_sigma(
168                required_prefetched_array(selected, row, column, arrays.path)?,
169                column,
170            )?
171        } else {
172            weights::broadcast_spectrum(scalar, shape)?
173        };
174        Ok(Some(value))
175    }
176}
177
178struct SelectedArrayReader<'a> {
179    table: &'a Table,
180    path: &'a str,
181}
182
183impl SelectedArrayReader<'_> {
184    fn read(
185        &self,
186        column: &str,
187        rows: &[usize],
188    ) -> Result<Vec<Option<ArrayValue>>, ApplyExecutionError> {
189        self.table
190            .column_accessor(column)
191            .and_then(|accessor| accessor.array_cells_owned_uncached(rows))
192            .map_err(|source| self.error(source))
193    }
194
195    fn error(&self, source: TableError) -> ApplyExecutionError {
196        ApplyExecutionError::MutateMeasurementSet {
197            path: self.path.into(),
198            source: MsError::from(source),
199        }
200    }
