6pub(super) fn available(row: &ApplyRowPlan, plan: &ApplyPlan) -> bool {
7    plan.calibration_tables
8        .iter()
9        .filter(|table| table.applies_to(row))
10        .all(|table| {
11            if table
12                .unavailable_row_indices
13                .binary_search(&row.row_index)
14                .is_ok()
15            {
16                return false;
17            }
18            table
19                .spw_mapping
20                .iter()
21                .find(|mapping| mapping.data_spw_id == row.data_spw_id)
22                .is_some_and(|mapping| {
23                    table
24                        .calibration_spectral_windows
25                        .iter()
26                        .any(|spw| spw.spw_id == mapping.calibration_spw_id)
27                })
28        })
29}

31pub(super) fn execute(
32    ms: &mut MeasurementSet,
33    plan: ApplyPlan,
34    tracer: Option<&mut CalibrationPerfTracer>,
35) -> Result<(ApplyExecutionReport, ExecuteApplyPlanTraceSummary), ApplyExecutionError> {
36    let start = Instant::now();
37    let unavailable = plan
38        .selected_rows
39        .iter()
40        .filter(|row| !available(row, &plan))
41        .cloned()
42        .collect::<Vec<_>>();
43    if unavailable.is_empty() {
44        return execute_apply_available(ms, plan, tracer);
45    }
46    let mut usable = plan.clone();
47    usable.selected_rows.retain(|row| available(row, &plan));
48    usable.selected_row_count = usable.selected_rows.len();
49    let usable_count = usable.selected_row_count;
50    // An empty mutation still initializes a newly requested CORRECTED_DATA
51    // from DATA, as CASA's open(addcorr=True) does before checking availability.
52    let (mut report, mut trace) = execute_apply_available(ms, usable, None)?;
53    let flag_start = Instant::now();
54    if plan.apply_mode.is_strict() {
55        let (samples, rows) = flag_unavailable(ms, &plan, &unavailable)?;
56        report.flagged_sample_count += samples;
57        report.flagged_row_count += rows;
58        report.updated_row_count += unavailable.len();
59    }
60    let flag_ns = flag_start.elapsed().as_nanos() as u64;
61    report.wrote_measurement_set =
62        report.created_corrected_data_column || usable_count > 0 || plan.apply_mode.is_strict();
63    report.timings.row_writeback_ns += flag_ns;
64    trace.row_writeback_ns += flag_ns;
65    let total = start.elapsed().as_nanos() as u64;
66    trace.execute_apply_plan_unattributed_ns +=
67        total.saturating_sub(trace.execute_apply_plan_ns + flag_ns);
68    trace.execute_apply_plan_ns = total;
69    trace.selected_row_count = plan.selected_row_count;
70    trace.updated_row_count = report.updated_row_count;
71    trace.flagged_row_count = report.flagged_row_count;
72    trace.flagged_sample_count = report.flagged_sample_count;
73    report.plan = plan;
74    if let Some(tracer) = tracer {
75        tracer.emit_apply_plan_summary(&display_ms_path(ms), &report.plan, trace);
76    }
77    Ok((report, trace))
78}

80fn flag_unavailable(
81    ms: &mut MeasurementSet,
82    plan: &ApplyPlan,
83    rows: &[ApplyRowPlan],
84) -> Result<(usize, usize), ApplyExecutionError> {
85    let path = display_ms_path(ms);
86    let table_error = |source| ApplyExecutionError::MutateMeasurementSet {
87        path: path.clone(),
88        source: MsError::from(source),
89    };
90    let write_error =
91        |source: casa_ms::MeasurementSetWriteError| ApplyExecutionError::MutateMeasurementSet {
92            path: path.clone(),
93            source: MsError::InvalidInput(source.to_string()),
94        };
95    let mut maximum = 0;
96    for row in rows {
97        let shape = ms
98            .main_table()
99            .cell_accessor(row.row_index, "FLAG")
100            .and_then(|cell| cell.array_shape())
101            .map_err(table_error)?;
102        let shape = shape.ok_or_else(|| ApplyExecutionError::UnsupportedParameterShape {
103            path: "<measurement-set FLAG>".into(),
104            shape: vec![],
105        })?;
106        if shape.len() != 2 {
107            return Err(ApplyExecutionError::UnsupportedParameterShape {
108                path: "<measurement-set FLAG>".into(),
109                shape,
110            });
111        }
112        maximum = maximum.max(shape.iter().product::<usize>());
113    }
114    let write_plan = MeasurementSetWritePlan::selected_row_mutation(
115        rows.iter().map(|row| row.row_index).collect(),
116        vec![MeasurementSetWriteColumnPlan {
117            name: "FLAG".into(),
118            bytes_per_row: maximum,
119            mode: MeasurementSetColumnWriteMode::Replace,
120            storage_manager: MeasurementSetColumnStorage::Persisted,
121            tile_shape: None,
122            create_source_column: None,
123        }],
124        MeasurementSetWriteResources::from_system_memory(2).map_err(write_error)?,
125    )
126    .map_err(write_error)?;
127    let batch_rows = write_plan.batch_rows();
128    let mut writer = MeasurementSetWriteSession::start_selected_row_mutation(ms, write_plan)
129        .map_err(write_error)?;
130    let (mut samples, mut fully_flagged) = (0, 0);
131    for batch in rows.chunks(batch_rows) {
132        let indices = batch.iter().map(|row| row.row_index).collect::<Vec<_>>();
133        let values = ms
134            .main_table()
135            .column_accessor("FLAG")
136            .and_then(|column| column.array_cells_owned(&indices))
137            .map_err(table_error)?;
138        let mut output = Vec::with_capacity(batch.len());
139        for (row, value) in batch.iter().zip(values) {
140            let value = required_prefetched_array(value, row.row_index, "FLAG", &path)?;
141            let ArrayValue::Bool(mut flags) = value else {
142                return Err(ApplyExecutionError::UnsupportedParameterShape {
143                    path: "<measurement-set FLAG>".into(),
144                    shape: value.shape().to_vec(),
145                });
146            };
147            let mut changed = 0;
148            if let Some(selection) = plan.channel_selections.get(&row.data_spw_id) {
149                for &channel in &selection.indices {
150                    if channel >= flags.shape()[1] {
151                        return Err(ApplyExecutionError::UnsupportedParameterShape {
152                            path: "<measurement-set FLAG>".into(),
153                            shape: flags.shape().to_vec(),
154                        });
155                    }
156                    for flag in flags.index_axis_mut(ndarray::Axis(1), channel) {
157                        if !*flag {
158                            changed += 1;
159                        }
160                        *flag = true;
161                    }
162                }
163            } else {
164                for flag in &mut flags {
165                    if !*flag {
166                        changed += 1;
167                    }
168                    *flag = true;
169                }
170            }
171            samples += changed;
172            fully_flagged += usize::from(changed > 0 && flags.iter().all(|flag| *flag));
173            output.push(ArrayValue::Bool(flags));
174        }
175        writer
176            .write_mutation_batch(
177                ms,
178                MeasurementSetMutationBatch {
179                    row_indices: indices,
180                    columns: vec![MeasurementSetMutationColumnBatch {
181                        name: "FLAG".into(),
182                        values: MeasurementSetMutationColumnValues::Arrays(output),
183                    }],
184                },
185            )
186            .map_err(write_error)?;
187    }
188    writer.finish_mutation().map_err(write_error)?;
189    Ok((samples, fully_flagged))
190}
