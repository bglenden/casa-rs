3//! Atomic staging and installation for text log files written by image tools.
4
5use std::fs::{File, Metadata};
6use std::io::{self, Write};
7use std::path::{Path, PathBuf};
8
9use super::error::{ImageToolError, ImageToolResult};
10
11/// A text report written beside its destination and installed atomically.
12pub struct StagedTextOutput {
13    path: PathBuf,
14    previous: Option<Metadata>,
15    file: tempfile::NamedTempFile,
16}
17
18impl StagedTextOutput {
19    /// Stages `text` beside `path`, preserving existing permissions and optionally appending.
20    pub fn new(path: &Path, text: &str, append: bool) -> ImageToolResult<Self> {
21        let (path, previous) = match path.symlink_metadata() {
22            Ok(_) => {
23                let target = path.canonicalize()?;
24                let metadata = target.metadata()?;
25                if !metadata.is_file() {
26                    return Err(ImageToolError::new(format!(
27                        "{} is not a regular log file",
28                        path.display()
29                    )));
30                }
31                (target, Some(metadata))
32            }
33            Err(error) if error.kind() == io::ErrorKind::NotFound => (path.to_path_buf(), None),
34            Err(error) => return Err(error.into()),
35        };
36        let parent = path
37            .parent()
38            .filter(|parent| !parent.as_os_str().is_empty())
39            .unwrap_or_else(|| Path::new("."));
40        let mut file = tempfile::NamedTempFile::new_in(parent)?;
41        if let Some(metadata) = &previous {
42            file.as_file().set_permissions(metadata.permissions())?;
43            if append {
44                io::copy(&mut File::open(&path)?, file.as_file_mut())?;
45            }
46        }
47        file.write_all(text.as_bytes())?;
48        file.flush()?;
49        Ok(Self {
50            path,
51            previous,
52            file,
53        })
54    }
55
56    /// Atomically persists the staged text file over its destination.
57    pub fn install(self) -> ImageToolResult<()> {
58        if let Some(previous) = self.previous {
59            let current = self.path.symlink_metadata()?;
60            if !current.is_file()
61                || previous.len() != current.len()
62                || previous.modified().ok() != current.modified().ok()
63            {
64                return Err(ImageToolError::new(format!(
65                    "Log {} changed during fitting",
66                    self.path.display()
67                )));
68            }
69            self.file.persist(self.path)?;
70        } else {
71            self.file.persist_noclobber(self.path)?;
72        }
73        Ok(())
74    }
75}
