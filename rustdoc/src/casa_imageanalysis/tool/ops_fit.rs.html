32/// Profile result across real (`f32`) or complex (`Complex32`) images (`ia.getprofile`).
33pub enum ProfileData {
34    /// Single-precision real profile.
35    Float(ImageReadProfile<f32>),
36    /// Single-precision complex profile.
37    Complex(ImageReadProfile<Complex32>),
38}
39
40impl ProfileData {
41    /// Converts this profile into CASA's record representation plus the per-point pixel counts (`npix`).
42    pub fn into_record_and_counts(self) -> (RecordValue, Vec<usize>) {
43        match self {
44            Self::Float(profile) => profile_to_record(profile, |values| {
45                ArrayValue::Float64(profile_vector(values.into_iter().map(f64::from).collect()))
46            }),
47            Self::Complex(profile) => profile_to_record(profile, |values| {
48                ArrayValue::Complex64(profile_vector(
49                    values
50                        .into_iter()
51                        .map(|value| Complex64::new(value.re.into(), value.im.into()))
52                        .collect(),
53                ))
54            }),
55        }
56    }
57}

59fn profile_to_record<T: CollapsePixel>(
60    profile: ImageReadProfile<T>,
61    convert_values: impl FnOnce(Vec<T>) -> ArrayValue,
62) -> (RecordValue, Vec<usize>) {
63    let mut record = RecordValue::default();
64    for (field, values) in [
65        (
66            "coords",
67            ArrayValue::Float64(profile_vector(profile.coordinates)),
68        ),
69        ("mask", ArrayValue::Bool(profile_vector(profile.mask))),
70        (
71            "planes",
72            ArrayValue::Float64(profile_vector(profile.planes)),
73        ),
74        ("values", convert_values(profile.values)),
75    ] {
76        record.upsert(field, Value::Array(values));
77    }
78    record.upsert("xUnit", Value::Scalar(ScalarValue::String(profile.x_unit)));
79    record.upsert("yUnit", Value::Scalar(ScalarValue::String(profile.y_unit)));
80    (record, profile.counts)
81}
82
83fn profile_vector<T>(values: Vec<T>) -> ArrayD<T> {
84    ArrayD::from_shape_vec(IxDyn(&[values.len()]), values).expect("one-dimensional profile")
85}

106/// Resolves the pixel and world bounding box for a region (`ia.boundingbox`).
107pub fn bounding_box(
108    attached: &AttachedImage,
109    region: Option<SubimageRegion<'_>>,
110) -> ImageToolResult<ImageRegionBounds> {
111    let mut image = attached.lock();
112    attached.require_pixel_read_lock(&mut image)?;
113    let unsupported_precision = match &*image {
114        ImageHandle::Double(_) => Some("double"),
115        ImageHandle::DComplex(_) => Some("complex<double>"),
116        _ => None,
117    };
118    if let Some(precision) = unsupported_precision {
119        return Err(ImageToolError::new(format!(
120            "boundingbox does not support images with {precision} precision pixel values"
121        )));
122    }
123    match &*image {
124        ImageHandle::Float(storage) => {
125            Ok(storage.with_float_view(|img| ImageRegionBounds::resolve(img, region))?)
126        }
127        ImageHandle::Complex(storage) => {
128            with_storage!(storage, img => Ok(ImageRegionBounds::resolve(img.as_ref(), region)?))
129        }
130        _ => unreachable!("double precision checked above"),
131    }
132}

134/// Computes image pixel histograms (`ia.histograms`).
135pub fn histograms(
136    attached: &AttachedImage,
137    options: HistogramOptions<'_>,
138) -> ImageToolResult<RecordValue> {
139    if !matches!(
140        *attached.lock(),
141        ImageHandle::Float(_) | ImageHandle::Double(_)
142    ) {
143        return Err(ImageToolError::new(
144            "This method only supports real-valued images",
145        ));
146    }
147    read_locked(attached, |handle| match handle {
148        ImageHandle::Float(storage) => {
149            Ok(storage.with_float_view(|img| image_histogram(img, options))?)
150        }
151        ImageHandle::Double(storage) => {
152            with_storage!(storage, img => Ok(image_histogram(img.as_ref(), options)?))
153        }
154        _ => Err(ImageToolError::new(
155            "This method only supports real-valued images",
156        )),
157    })
158}

225/// Extracts a 1D collapsed profile along `options.axis` and optionally writes a text report (`ia.getprofile`).
226pub fn get_profile(
227    attached: &AttachedImage,
228    options: ReadProfileOptions<'_>,
229) -> ImageToolResult<ProfileData> {
230    check_profile_read_type(attached)?;
231    read_locked(attached, |source| match source {
232        ImageHandle::Float(storage) => storage.with_float_view(|image| {
233            read_profile_typed(attached, source, image, options).map(ProfileData::Float)
234        }),
235        ImageHandle::Complex(storage) => with_storage!(storage, image => {
236            read_profile_typed(attached, source, image.as_ref(), options).map(ProfileData::Complex)
237        }),
238        _ => Err(ImageToolError::new(PROFILE_READ_TYPE_ERROR)),
239    })
240}

258fn read_profile_typed<T: ProfileLogPixel>(
259    attached: &AttachedImage,
260    source: &ImageHandle,
261    image: &dyn ImageInterface<T>,
262    options: ReadProfileOptions<'_>,
263) -> ImageToolResult<ImageReadProfile<T>> {
264    if let Some(path) = options.logfile {
265        let region_file = options.region.and_then(SubimageRegion::external_path);
266        for input in source
267            .path()
268            .into_iter()
269            .chain(
270                attached
271                    .operand_paths()
272                    .iter()
273                    .map(std::path::PathBuf::as_path),
274            )
275            .chain(region_file)
276        {
277            ensure_artifacts_do_not_overlap(path, input)?;
278        }
279    }
280    let logfile = options.logfile;
281    let region_path = match &options.region {
282        Some(SubimageRegion::Text(text)) => Some(*text),
283        _ => None,
284    };
285    let mut beams = image.image_info()?.to_record();
286    attached.beam_view().apply_to(&mut beams);
287    let profile = read_image_profile(image, source.coordinate_record(), &beams, options)?;
288    if let Some(path) = logfile {
289        let name = source
290            .path()
291            .map(|p| p.to_string_lossy())
292            .unwrap_or_else(|| "Temporary_Image".into());
293        let text = format_profile_report(&profile, &name, region_path);
294        StagedTextOutput::new(path, &text, false)?.install()?;
295    }
296    Ok(profile)
297}
298
299fn format_profile_report<T: ProfileLogPixel>(
300    profile: &ImageReadProfile<T>,
301    image: &str,
302    region: Option<&str>,
303) -> String {
304    let mut text = format!("#title: {} profile - {image}\n", profile.axis_name);
305    if let Some(region) = region {
306        let _ = writeln!(text, "#region : {region}");
307    }
308    let _ = writeln!(
309        text,
310        "#xUnit {}\n#yUnit {}\n# {image}\n",
311        profile.x_unit, profile.y_unit
312    );
313    for ((&coordinate, &value), &valid) in profile
314        .coordinates
315        .iter()
316        .zip(&profile.values)
317        .zip(&profile.mask)
318    {
319        if valid {
320            let _ = writeln!(text, "{coordinate:.7} {}", value.format_report_value());
321        }
322    }
323    text
324}

388impl FitComponentsRequest<'_> {
389    fn history_call(&self) -> String {
390        let range = |values: &[f64]| {
391            let values = values
392                .iter()
393                .map(|value| format!("{value:.6}"))
394                .collect::<Vec<_>>()
395                .join(", ");
396            format!("[{values}]")
397        };
398        format!(
399            "ia.fitcomponents(box={:?}, region={}, chans={}, stokes={:?}, mask={:?}, includepix={}, excludepix={}, residual={:?}, model={:?}, estimates={:?}, logfile={:?}, append={}, newestimates={:?}, complist={:?}, dooff={}, offset={:.6}, fixoffset={}, stretch={}, rms={}, noisefwhm={})",
400            self.box_selection.unwrap_or_default(),
401            self.history.region,
402            self.history.channels,
403            self.stokes.unwrap_or_default(),
404            self.mask.unwrap_or_default(),
405            range(self.includepix),
406            range(self.excludepix),
407            self.residual,
408            self.model,
409            self.estimates
410                .map_or_else(|| "".into(), Path::to_string_lossy),
411            self.logfile,
412            self.append,
413            self.newestimates,
414            self.complist,
415            self.fit_zero_level,
416            self.zero_level.0,
417            self.zero_level.1,
418            self.stretch,
419            self.history.rms,
420            self.history.noise_fwhm,
421        )
422    }
423}

425/// Fits 2D Gaussian components on selected sky planes (`ia.fitcomponents`).
426pub fn fit_components(
427    attached: &AttachedImage,
428    request: FitComponentsRequest<'_>,
429    is_external_open: ExternalPathCheck<'_>,
430) -> ImageToolResult<RecordValue> {
431    if !matches!(
432        *attached.lock(),
433        ImageHandle::Float(_) | ImageHandle::Double(_)
434    ) {
435        return Err(ImageToolError::new(
436            "This method only supports real valued images",
437        ));
438    }
439    let history_call = request.history_call();
440    let paths = [
441        request.model,
442        request.residual,
443        request.complist,
444        request.logfile,
445        request.newestimates,
446        request.summary,
447    ]
448    .into_iter()
449    .filter(|path| !path.is_empty())
450    .map(Path::new)
451    .collect::<Vec<_>>();
452    for (index, path) in paths.iter().enumerate() {
453        for peer in &paths[..index] {
454            ensure_artifacts_do_not_overlap(path, peer)?;
455        }
456    }
457    let fit = with_mask_output_paths(paths.iter().copied(), || {
458        publish_image_set(
459            [request.model, request.residual],
460            true,
461            is_external_open,
462            |staged| {
463                read_locked(attached, |source| {
464                    for target in &paths {
465                        for input in source.path().into_iter().chain(
466                            attached
467                                .operand_paths()
468                                .iter()
469                                .map(std::path::PathBuf::as_path),
470                        ) {
471                            ensure_artifacts_do_not_overlap(target, input)?;
472                        }
473                    }
474                    match source {
475                        ImageHandle::Float(storage) => storage.with_float_view(|img| {
476                            fit_components_typed(
477                                attached,
478                                source,
479                                img,
480                                &request,
481                                &history_call,
482                                &paths,
483                                staged,
484                            )
485                        }),
486                        ImageHandle::Double(storage) => with_storage!(storage, img => {
487                            fit_components_typed(attached, source, img.as_ref(), &request, &history_call, &paths, staged)
488                        }),
489                        _ => Err(ImageToolError::new(
490                            "This method only supports real valued images",
491                        )),
492                    }
493                })
494            },
495        )
496    })?;
497    if !fit.components.is_empty() {
498        if !request.complist.is_empty()
499            && (request.overwrite || !Path::new(request.complist).exists())
500        {
501            publish_image_set(
502                [request.complist],
503                request.overwrite,
504                is_external_open,
505                |[path]| {
506                    fit.components
507                        .write_table(path.expect("component list destination"))
508                        .map_err(|err| ImageToolError::new(err.to_string()))
509                },
510            )?;
511        }
512        if !request.newestimates.is_empty() {
513            StagedTextOutput::new(
514                Path::new(request.newestimates),
515                &fit.estimates_text(),
516                false,
517            )?
518            .install()?;
519        }
520        if !request.summary.is_empty() {
521            StagedTextOutput::new(Path::new(request.summary), &fit.summary_text(), false)?
522                .install()?;
523        }
524    }
525    if !request.logfile.is_empty() {
526        let text = format!(
527            "{}\n{} fitted components\n{}",
528            history_call,
529            fit.components.len(),
530            fit.log_text()
531        );
532        StagedTextOutput::new(Path::new(request.logfile), &text, request.append)?.install()?;
533    }
534    Ok(fit.record)
535}

537fn fit_components_typed<T: ComponentFitPixel>(
538    attached: &AttachedImage,
539    source: &ImageHandle,
540    image: &dyn ImageInterface<T>,
541    request: &FitComponentsRequest<'_>,
542    history_call: &str,
543    paths: &[&Path],
544    staged: [Option<&Path>; 2],
545) -> ImageToolResult<ImageComponentFit> {
546    let beam_view = attached.beam_view();
547    let beams = beam_view.beam_set()?;
548    let fit = fit_image_components(
549        image,
550        FitComponentsOptions {
551            pixel_box: request.box_selection,
552            region: request.region,
553            channels: request.chans,
554            stokes: request.stokes,
555            mask: request.mask,
556            destinations: paths,
557            stretch: request.stretch,
558            include_pixels: request.includepix,
559            exclude_pixels: request.excludepix,
560            estimates: request.estimates,
561            zero_level: request.fit_zero_level.then_some(request.zero_level),
562            beams: Some(&beams),
563            rms: request.rms.as_ref().map(|q| match q {
564                ComponentFitQuantity::Numeric(v) => ComponentFitQuantity::Numeric(*v),
565                ComponentFitQuantity::Quantity(r) => ComponentFitQuantity::Quantity(r),
566            }),
567            report_log: !request.logfile.is_empty(),
568            noise_fwhm: request.noise_fwhm.as_ref().map(|q| match q {
569                ComponentFitQuantity::Numeric(v) => ComponentFitQuantity::Numeric(*v),
570                ComponentFitQuantity::Quantity(r) => ComponentFitQuantity::Quantity(r),
571            }),
572        },
573    )?;
574    let [model, residual] = staged.map(|path| {
575        path.map(|path| {
576            let shape = image.shape().to_vec();
577            let tiled = TiledShape::new(shape.clone(), std::mem::size_of::<f32>())?;
578            PagedImage::<f32>::create_with_tile_shape_and_cache(
579                shape,
580                tiled.tile_shape().to_vec(),
581                image.coordinates().clone(),
582                path,
583                16 * 1024 * 1024,
584            )
585            .map_err(ImageToolError::from)
586        })
587        .transpose()
588    });
589    let mut model = model?;
590    let mut residual = residual?;
591    fit.write_images(
592        image,
593        model
594            .as_mut()
595            .map(|img| img as &mut dyn MutableImageInterface<f32>),
596        residual
597            .as_mut()
598            .map(|img| img as &mut dyn MutableImageInterface<f32>),
599    )?;
600    if model.is_some() || residual.is_some() {
601        let coordinate_record = source.coordinate_record();
602        let mut image_info = image.image_info()?.to_record();
603        beam_view.apply_to(&mut image_info);
604        let source_history = source.history_entries()?;
605        for mut output in [model, residual].into_iter().flatten() {
606            output.set_coordinate_record(coordinate_record.clone())?;
607            output.set_image_info_record(image_info.clone())?;
608            let mut handle = ImageHandle::Float(Storage::Paged(Box::new(output)));
609            handle.append_history_entries(&source_history)?;
610            if request.record_history {
611                handle.record_history("fitcomponents", history_call.to_owned())?;
612            }
613            handle.save()?;
614        }
615    }
616    Ok(fit)
617}

657    /// Include full fitted parameter report lines in returned `ToolLogLine`s.
658    pub log_details: bool,
659    /// Region argument as displayed in CASA history and reports.
660    pub region_history: &'a str,
661    /// Write history into output spectra/sigma images.
662    pub record_history: bool,
663}

665impl FitProfileRequest<'_> {
666    fn history_call(&self) -> String {
667        let [pixel_box, chans, stokes] = self.selection;
668        let [model, residual] = self.outputs.spectra;
669        format!(
670            "ia.fitprofile(box={pixel_box:?}, region={}, chans={chans:?}, stokes={stokes:?}, axis={}, mask={:?}, ngauss={}, poly={}, estimates={:?}, minpts={}, multifit={}, model={model:?}, residual={residual:?})",
671            self.region_history,
672            self.axis,
673            self.mask,
674            self.ngauss,
675            self.poly,
676            self.estimates_file,
677            self.min_points,
678            self.multifit
679        )
680    }
681}

685/// Validates the pixel type before interpreting `ia.fitprofile` arguments.
686pub fn check_profile_fit_type(attached: &AttachedImage) -> ImageToolResult<()> {
687    if matches!(*attached.lock(), ImageHandle::Float(_)) {
688        Ok(())
689    } else {
690        Err(ImageToolError::new(PROFILE_FIT_TYPE_ERROR))
691    }
692}

694/// Fits 1D spectral/axis profiles (`ia.fitprofile`).
695pub fn fit_profile(
696    attached: &AttachedImage,
697    request: FitProfileRequest<'_>,
698    is_external_open: ExternalPathCheck<'_>,
699) -> ImageToolResult<(RecordValue, Vec<ToolLogLine>)> {
700    check_profile_fit_type(attached)?;
701    let sigma_img = match request.sigma {
702        ProfileSigmaInput::Path(path) => Some(Image::open(path)?),
703        _ => None,
704    };
705    validate_profile_log(request.logfile)?;
706    let history_call = request.history_call();
707    let (stages, fit) = read_locked(attached, |source| match source {
708        ImageHandle::Float(storage) => storage.with_float_view(|image| {
709            compute_profile_fit(
710                attached,
711                source,
712                image,
713                &request,
714                sigma_img.as_ref(),
715                &history_call,
716            )
717        }),
718        _ => Err(ImageToolError::new(PROFILE_FIT_TYPE_ERROR)),
719    })?;
720    let mut logs: Vec<_> = fit.summary.iter().cloned().map(ToolLogLine::info).collect();
721    if request.ngauss < 0 {
722        logs.push(ToolLogLine::warning(
723            "ngauss < 0 is meaningless. Setting ngauss = 0",
724        ));
725    }
726    if !request.outputs.sigma.is_empty() && matches!(request.sigma, ProfileSigmaInput::None) {
727        logs.push(ToolLogLine::warning(
728            "Ignoring outsigma because no sigma image or array was specified",
729        ));
730    }
731    let details = (request.log_details || !request.logfile.is_empty())
732        .then(|| profile_log_text(&history_call, &fit));
733    let staged_log = if let Some(details) = &details
734        && !request.logfile.is_empty()
735    {
736        let text = format!("{}\n\n{details}", fit.summary.join("\n"));
737        Some(StagedTextOutput::new(
738            Path::new(request.logfile),
739            &text,
740            request.append,
741        )?)
742    } else {
743        None
744    };
745    stages.publish_with(true, is_external_open, || {
746        staged_log.map_or(Ok(()), StagedTextOutput::install)
747    })?;
748    if let Some(details) = details
749        && request.log_details
750    {
751        logs.push(ToolLogLine::info(details));
752    }
753    Ok((fit.record, logs))
754}

756fn compute_profile_fit(
757    attachment: &AttachedImage,
758    source: &ImageHandle,
759    image: &dyn ImageInterface<f32>,
760    request: &FitProfileRequest<'_>,
761    sigma_img: Option<&Image>,
762    history_call: &str,
763) -> ImageToolResult<(NamedImageStages, ImageProfileFitResult)> {
764    let mut beam_record = image.image_info()?.to_record();
765    attachment.beam_view().apply_to(&mut beam_record);
766    let [pixel_box, channels, stokes] = request.selection;
767    let region_file = request.region.and_then(SubimageRegion::external_path);
768    let fit = ImageProfileFit::new(
769        image,
770        source.coordinate_record(),
771        ProfileFitOptions {
772            selection: ProfileRegion {
773                pixel_box: nonempty(pixel_box),
774                region: request.region,
775                chans: nonempty(channels),
776                stokes: nonempty(stokes),
777            },
778            axis: request.axis,
779            mask: nonempty(request.mask),
780            stretch: request.stretch,
781            multifit: request.multifit,
782            min_points: request.min_points.max(0) as usize,
783            gaussians: request.ngauss.max(0) as usize,
784            polynomial: usize::try_from(request.poly).ok(),
785            estimates_file: nonempty(request.estimates_file).map(Path::new),
786            estimates: request.estimates,
787            good_ranges: request.good_ranges,
788            planes: request.planes,
789            divisor: request.divisor.clone(),
790            sigma: match request.sigma {
791                ProfileSigmaInput::Values(values) => Some(ProfileSigma::Values(values)),
792                ProfileSigmaInput::Path(_) => {
793                    sigma_img.map(|img| ProfileSigma::Image(img as &dyn ImageInterface<f32>))
794                }
795                ProfileSigmaInput::None => None,
796            },
797            other_artifact: nonempty(request.logfile).map(Path::new),
798            beam_record: Some(&beam_record),
799            log_details: request.log_details || !request.logfile.is_empty(),
800        },
801        request.outputs,
802    )?;
803    validate_fit_paths(
804        fit.destinations().chain(nonempty(request.logfile)),
805        source
806            .path()
807            .into_iter()
808            .chain(
809                attachment
810                    .operand_paths()
811                    .iter()
812                    .map(std::path::PathBuf::as_path),
813            )
814            .chain(sigma_img.and_then(ImageInterface::name))
815            .chain(nonempty(request.estimates_file).map(Path::new))
816            .chain(region_file),
817    )?;
818    let output_is_spectrum: Vec<_> = fit
819        .destinations()
820        .map(|name| request.outputs.spectra.contains(&name) || request.outputs.sigma == name)
821        .collect();
822    let stages = NamedImageStages::with_replacements(
823        fit.destinations()
824            .map(|name| (name, name == request.outputs.sigma)),
825    )?;
826    let paths = stages.paths();
827    let fit = fit.fit(&paths)?;
828    {
829        let mut paths = paths
830            .into_iter()
831            .zip(output_is_spectrum)
832            .filter_map(|(path, with_history)| path.filter(|_| with_history))
833            .peekable();
834        let prior = if paths.peek().is_some() {
835            source.history_entries()?
836        } else {
837            Vec::new()
838        };
839        for path in paths {
840            let mut out = Image::open(path)?;
841            if out.history()?.is_empty() {
842                out.append_history_entries(&prior)?;
843            }
844            if request.record_history {
845                out.append_history_at(
846                    "::image::fitprofile",
847                    &["Ran ia.fitprofile".into(), history_call.to_owned()],
848                )?;
849            }
850            out.append_history_at("::ImageProfileFitter::fit", &fit.summary)?;
851            out.save()?;
852        }
853    }
854    Ok((stages, fit))
855}

857fn profile_log_text(history_call: &str, fit: &ImageProfileFitResult) -> String {
858    let mut text = format!("Input parameters ---\n{history_call}\n");
859    let record = &fit.record;
860    let Some(Value::Array(ArrayValue::Bool(valid))) = record.get("valid") else {
861        return text;
862    };
863    let unit = |name| match record.get(name) {
864        Some(Value::Scalar(ScalarValue::String(unit))) => unit.as_str(),
865        _ => "",
866    };
867    let (x_unit, y_unit) = (unit("xUnit"), unit("yUnit"));
868    let integral_unit = if x_unit.is_empty() || y_unit.is_empty() {
869        format!("{x_unit}{y_unit}")
870    } else {
871        format!("{x_unit}.{y_unit}")
872    };
873    for (pixel, &ok) in valid.indexed_iter() {
874        let position = pixel.slice();
875        let Some(report) = fit.profile_reports.get(position) else {
876            continue;
877        };
878        let coordinates = report
879            .position
880            .iter()
881            .map(|coordinate| coordinate.map_or_else(|| "*".into(), |c| c.to_string()))
882            .collect::<Vec<_>>();
883        let _ = writeln!(text, "Fit\n    Pixel        : [{}]", coordinates.join(", "));
884        if let Some(Value::Array(ArrayValue::String(direction))) = record.get("direction") {
885            let _ = writeln!(text, "    Direction    : {}", direction[IxDyn(position)]);
886        }
887        for (key, label) in [("attempted", "Attempted"), ("converged", "Converged")] {
888            if let Some(Value::Array(ArrayValue::Bool(flags))) = record.get(key) {
889                let status = if flags[IxDyn(position)] { "YES" } else { "NO" };
890                let _ = writeln!(text, "    {label:12} : {status}");
891            }
892        }
893        if let Some(iterations) = report.iterations {
894            let _ = writeln!(text, "    Iterations   : {iterations}");
895        }
896        let _ = writeln!(text, "    Valid        : {}", if ok { "YES" } else { "NO" });
897        if !ok {
898            continue;
899        }
900        for line in &report.lines {
901            write_profile_line(&mut text, line, (x_unit, y_unit, &integral_unit));
902        }
903        if let Some(polynomial) = fit.polynomials.get(position) {
904            let _ = writeln!(text, "    Type: POLYNOMIAL");
905            for (kind, terms) in [("pixel", &polynomial.pixel), ("world", &polynomial.world)] {
906                let _ = writeln!(text, "        Coefficients in {kind} coordinates");
907                for (index, &[value, error]) in terms.iter().enumerate() {
908                    let _ = writeln!(text, "        c{index} : {value:.12e} +/- {error:.8e}");
909                }
910            }
911        }
912        for field in record.fields() {
913            if !matches!(field.name.as_str(), "plp" | "ltp") {
914                continue;
915            }
916            let Value::Record(model) = &field.value else {
917                continue;
918            };
919            let _ = writeln!(text, "    {}", field.name);
920            for field in model.fields() {
921                let Value::Array(ArrayValue::Float64(values)) = &field.value else {
922                    continue;
923                };
924                let mut index = position.to_vec();
925                index.push(0);
926                let count = values.shape()[values.ndim() - 1];
927                let _ = write!(text, "        {}", field.name);
928                for component in 0..count {
929                    index[position.len()] = component;
930                    let _ = write!(text, "  {:.8e}", values[IxDyn(&index)]);
931                }
932                text.push('\n');
933            }
934        }
935    }
936    text
937}

939fn write_profile_line(
940    text: &mut String,
941    line: &ProfileLineReport,
942    (x_unit, y_unit, integral_unit): (&str, &str, &str),
943) {
944    if line.subcomponent.is_none_or(|index| index == 0) {
945        let _ = writeln!(text, "    Results for component {}:", line.component);
946        let _ = writeln!(text, "        Type     : {}", line.kind);
947    }
948    if let Some(index) = line.subcomponent {
949        let _ = writeln!(text, "        Results for subcomponent {index}:");
950    }
951    let [peak_fixed, center_fixed, width_fixed] = line.fixed;
952    for (label, estimate, unit, fixed) in [
953        ("Peak", Some(line.peak), y_unit, peak_fixed),
954        ("Center", Some(line.center), x_unit, center_fixed),
955        ("Center", line.pixel_center, "pixel", center_fixed),
956        ("FWHM", Some(line.fwhm), x_unit, width_fixed),
957        ("FWHM", line.pixel_fwhm, "pixel", width_fixed),
958        (
959            "Integral",
960            Some(line.integral),
961            integral_unit,
962            peak_fixed && width_fixed,
963        ),
964    ] {
965        let fixed = if fixed { " (fixed)" } else { "" };
966        let Some([value, error]) = estimate else {
967            let _ = writeln!(
968                text,
969                "                   Could not convert {label} to pixel"
970            );
971            continue;
972        };
973        if unit == "pixel" {
974            let _ = writeln!(
975                text,
976                "                   {value:.12e} +/- {error:.12e} {unit}{fixed}"
977            );
978        } else {
979            let _ = writeln!(
980                text,
981                "        {label:8} : {value:.12e} +/- {error:.12e} {unit}{fixed}"
982            );
983        }
984    }
985    if line.channel_fwhm <= 3. {
986        let _ = writeln!(
987            text,
988            "        WARNING: The FWHM is at most three channel widths."
989        );
990    }
991}

993fn validate_fit_paths<'a, 'b>(
994    outputs: impl IntoIterator<Item = &'a str>,
995    inputs: impl IntoIterator<Item = &'b Path>,
996) -> ImageToolResult<()> {
997    let outputs: Vec<_> = outputs.into_iter().map(Path::new).collect();
998    let inputs: Vec<_> = inputs.into_iter().collect();
999    for (index, path) in outputs.iter().enumerate() {
1000        for other in inputs
1001            .iter()
1002            .copied()
1003            .chain(outputs[..index].iter().copied())
1004        {
1005            ensure_artifacts_do_not_overlap(path, other)?;
1006        }
1007    }
1008    Ok(())
1009}

1011fn validate_profile_log(path: &str) -> ImageToolResult<()> {
1012    if let Some(path) = nonempty(path).map(Path::new) {
1013        if path.is_dir() {
1014            return Err(ImageToolError::new(format!(
1015                "Log destination {} is a directory",
1016                path.display()
1017            )));
1018        }
1019        if let Some(parent) = path.parent()
1020            && !parent.as_os_str().is_empty()
1021            && !parent.is_dir()
1022        {
1023            return Err(ImageToolError::new(format!(
1024                "Log directory {} does not exist",
1025                parent.display()
1026            )));
1027        }
1028    }
1029    Ok(())
1030}
