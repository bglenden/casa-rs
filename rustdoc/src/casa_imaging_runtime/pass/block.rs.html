7/// Where a row lives in its MeasurementSet, for the model-column writer.

26/// A row's baseline in the frame of one image domain: CASA rotates each

51/// The largest block a source delivers ([`super::BoundedSource::maximum_block`]):
52/// what each pass sizes its per-block storage by, and admits before it
53/// allocates it.
54#[derive(Clone, Copy, Debug, Default, PartialEq)]
55pub struct BlockShape {
56    /// Rows.
57    pub rows: usize,
58    /// Selected channels per row and their frequencies' extent, which bound
59    /// the samples a row places ([`RowSpectrum`]).
60    pub spectrum: RowSpectrum,
61    /// Selected correlations per row.
62    pub correlations: usize,
63    /// Image domains each row is projected on.
64    pub domains: usize,
65}
66
67impl BlockShape {
68    /// Selected channels per row.
69    #[must_use]
70    pub const fn channels(self) -> usize {
71        self.spectrum.channels
72    }
73
74    /// Bytes a [`NativeBlock`] of this shape holds
75    /// ([`NativeBlock::with_capacity`]).
76    #[must_use]
77    pub const fn bytes(self) -> u64 {
78        let channels = self.channels();
79        let cells = channels * self.correlations;
80        let per_row = size_of::<NativeRowHeader>()
81            + size_of::<u32>()
82            + self.domains * size_of::<DomainProjection>()
83            + channels * size_of::<f64>()
84            + cells * (size_of::<Complex32>() + size_of::<f32>() + 2 * size_of::<bool>());
85        (self.rows * per_row + (channels + self.correlations) * size_of::<u32>()) as u64
86    }
87
88    /// Bytes a [`NativeBlock`] of this shape holds when its rows share one
89    /// spectrum and their samples ([`NativeBlock::shared_with_capacity`]):
90    /// each row's header, order entry and projections, and one row's
91    /// frequencies, values, weights and flags.
92    #[must_use]
93    pub const fn shared_bytes(self) -> u64 {
94        let channels = self.channels();
95        let cells = channels * self.correlations;
96        let per_row = size_of::<NativeRowHeader>()
97            + size_of::<u32>()
98            + self.domains * size_of::<DomainProjection>();
99        let samples = channels * size_of::<f64>()
100            + cells * (size_of::<Complex32>() + size_of::<f32>() + size_of::<bool>());
101        (self.rows * per_row + samples + (channels + self.correlations) * size_of::<u32>()) as u64
102    }
103}

105/// One block of native rows sharing a channel and correlation layout,
106/// projected on every image domain of the pass.
107///
108/// Arrays are `[row][channel][correlation]`, correlation fastest, over the
109/// selected channels and correlations in the operator's routing order;
110/// projections are `[row][domain]`. Frequencies are per row and channel
111/// because Doppler tracking moves them. Capacity survives
112/// [`NativeBlock::reset`], so a source allocates once.
113///
114/// Rows that all have one spectrum and, in every sample, a zero value, a
115/// unit weight and no flag, as a simulator's rows to predict do, can share
116/// them ([`NativeBlock::reset_shared`]): the block keeps them once, and a
117/// row adds only its header and projections.
118///
119/// The rows also have a locality order ([`Self::order_rows_by_baseline`]):
120/// a baseline's successive integrations fall on nearly the same uv cells,
121/// so a pass that reads them together reuses the grid cells it just read
122/// and wrote.
123#[derive(Clone, Debug, Default)]
124pub struct NativeBlock {
125    domains: usize,
126    rows: Vec<NativeRowHeader>,
127    /// Row indices in locality order; empty in arrival order.
128    order: Vec<u32>,
129    projections: Vec<DomainProjection>,
130    channel_indices: Vec<u32>,
131    correlation_indices: Vec<u32>,
132    frequencies_hz: Vec<f64>,
133    values: Vec<Complex32>,
134    weights: Vec<f32>,
135    flags: Vec<bool>,
136    /// Original flags, with the parent selection, for an untuned model write.
137    prediction_flags: Vec<bool>,
138    /// Whether every row reads the first row's frequencies, values, weights
139    /// and flags ([`Self::reset_shared`]).
140    shared: bool,
141}
142
143impl NativeBlock {
144    /// An empty block with room for every block of `shape`, so filling it
145    /// never grows it beyond [`BlockShape::bytes`].
146    #[must_use]
147    pub fn with_capacity(shape: BlockShape) -> Self {
148        Self::sized(shape, shape.rows)
149    }
150
151    /// An empty block with room for every block of `shape` whose rows share
152    /// their spectrum and samples ([`Self::reset_shared`]), so filling it
153    /// never grows it beyond [`BlockShape::shared_bytes`].
154    #[must_use]
155    pub fn shared_with_capacity(shape: BlockShape) -> Self {
156        Self::sized(shape, 1)
157    }
158
159    /// An empty block with room for `shape`'s rows and the spectra and
160    /// samples of `sample_rows` of them.
161    fn sized(shape: BlockShape, sample_rows: usize) -> Self {
162        let channels = shape.channels();
163        let cells = sample_rows * channels * shape.correlations;
164        Self {
165            domains: shape.domains,
166            rows: Vec::with_capacity(shape.rows),
167            order: Vec::with_capacity(shape.rows),
168            projections: Vec::with_capacity(shape.rows * shape.domains),
169            channel_indices: Vec::with_capacity(channels),
170            correlation_indices: Vec::with_capacity(shape.correlations),
171            frequencies_hz: Vec::with_capacity(sample_rows * channels),
172            values: Vec::with_capacity(cells),
173            weights: Vec::with_capacity(cells),
174            flags: Vec::with_capacity(cells),
175            prediction_flags: Vec::new(),
176            shared: false,
177        }
178    }
179
180    /// Bytes the block holds now, from its capacities: at most
181    /// [`BlockShape::bytes`] for a block [`Self::with_capacity`] made and
182    /// filled within its shape.
183    #[must_use]
184    pub fn capacity_bytes(&self) -> u64 {
185        (self.rows.capacity() * size_of::<NativeRowHeader>()
186            + self.projections.capacity() * size_of::<DomainProjection>()
187            + (self.order.capacity()
188                + self.channel_indices.capacity()
189                + self.correlation_indices.capacity())
190                * size_of::<u32>()
191            + self.frequencies_hz.capacity() * size_of::<f64>()
192            + self.values.capacity() * size_of::<Complex32>()
193            + self.weights.capacity() * size_of::<f32>()
194            + (self.flags.capacity() + self.prediction_flags.capacity()) * size_of::<bool>())
195            as u64
196    }

198    /// Empty the block and set the layout of the rows that follow: the
199    /// number of image domains each row is projected on, and the native
200    /// channel and correlation indices every row selects.
201    pub fn reset(&mut self, domains: usize, channel_indices: &[u32], correlation_indices: &[u32]) {
202        assert!(
203            domains > 0 && !channel_indices.is_empty() && !correlation_indices.is_empty(),
204            "a native block projects on one domain and selects one channel and correlation"
205        );
206        self.domains = domains;
207        self.shared = false;
208        self.rows.clear();
209        self.order.clear();
210        self.projections.clear();
211        self.frequencies_hz.clear();
212        self.values.clear();
213        self.weights.clear();
214        self.flags.clear();
215        self.prediction_flags.clear();
216        self.channel_indices.clear();
217        self.channel_indices.extend_from_slice(channel_indices);
218        self.correlation_indices.clear();
219        self.correlation_indices
220            .extend_from_slice(correlation_indices);
221    }
222
223    /// [`Self::reset`] for rows that all have `frequencies_hz` and, in
224    /// every channel and correlation, a zero value, a unit weight and no
225    /// flag: the block keeps them once, and the rows that follow come from
226    /// [`Self::push_shared_row`]. Each row then reads exactly as
227    /// [`Self::push_row`] would have made it with those samples.
228    pub fn reset_shared(
229        &mut self,
230        domains: usize,
231        channel_indices: &[u32],
232        correlation_indices: &[u32],
233        frequencies_hz: &[f64],
234    ) {
235        self.reset(domains, channel_indices, correlation_indices);
236        assert_eq!(
237            frequencies_hz.len(),
238            channel_indices.len(),
239            "frequencies per row"
240        );
241        let cells = channel_indices.len() * correlation_indices.len();
242        self.frequencies_hz.extend_from_slice(frequencies_hz);
243        self.values.resize(cells, Complex32::default());
244        self.weights.resize(cells, 1.0);
245        self.flags.resize(cells, false);
246        self.shared = true;
247    }
248
249    /// Append one row of the shared spectrum and samples
250    /// ([`Self::reset_shared`]): its header and its projection on every
251    /// domain.
252    pub fn push_shared_row(&mut self, header: NativeRowHeader, projections: &[DomainProjection]) {
253        assert!(self.shared, "shared rows follow reset_shared");
254        assert_eq!(projections.len(), self.domains, "one projection per domain");
255        self.rows.push(header);
256        self.order.clear();
257        self.projections.extend_from_slice(projections);
258    }
259
260    /// Append one row: its projection on every domain, `frequencies_hz` per
261    /// channel, `values`, `weights` and `flags` per channel and correlation.
262    pub fn push_row(
263        &mut self,
264        header: NativeRowHeader,
265        projections: &[DomainProjection],
266        frequencies_hz: &[f64],
267        values: &[Complex32],
268        weights: &[f32],
269        flags: &[bool],
270    ) {
271        let cells = self.channels() * self.correlations();
272        assert!(
273            values.len() == cells && weights.len() == cells && flags.len() == cells,
274            "values, weights and flags per row must be channels × correlations"
275        );
276        self.push_row_with(header, projections, frequencies_hz, |v, w, f| {
277            v.copy_from_slice(values);
278            w.copy_from_slice(weights);
279            f.copy_from_slice(flags);
280            Ok::<_, std::convert::Infallible>(())
281        })
282        .unwrap_or_else(|never| match never {});
283    }
284
285    /// [`Self::push_row`], with `fill` writing the row's values, weights and
286    /// flags in place: it receives them zeroed, `channels × correlations`
287    /// each, so a source copies each sample once. When `fill` fails the
288    /// block is left as it was and the error returned.
289    pub fn push_row_with<E>(
290        &mut self,
291        header: NativeRowHeader,
292        projections: &[DomainProjection],
293        frequencies_hz: &[f64],
294        fill: impl FnOnce(&mut [Complex32], &mut [f32], &mut [bool]) -> Result<(), E>,
295    ) -> Result<(), E> {
296        assert!(!self.shared, "a block of shared rows takes push_shared_row");
297        let cells = self.channels() * self.correlations();
298        assert_eq!(projections.len(), self.domains, "one projection per domain");
299        assert_eq!(frequencies_hz.len(), self.channels(), "frequencies per row");
300        let start = self.values.len();
301        self.values.resize(start + cells, Complex32::default());
302        self.weights.resize(start + cells, 0.0);
303        self.flags.resize(start + cells, false);
304        if let Err(error) = fill(
305            &mut self.values[start..],
306            &mut self.weights[start..],
307            &mut self.flags[start..],
308        ) {
309            self.values.truncate(start);
310            self.weights.truncate(start);
311            self.flags.truncate(start);
312            return Err(error);
313        }
314        self.rows.push(header);
315        self.order.clear();
316        self.projections.extend_from_slice(projections);
317        self.frequencies_hz.extend_from_slice(frequencies_hz);
318        Ok(())
319    }

321    /// Give the most recently appended row separate per-channel grid and
322    /// prediction selections. Each mask is true for a selected channel.
323    pub fn separate_last_row_selection(&mut self, grid: &[bool], prediction: &[bool]) {
324        let channels = self.channels();
325        let products = self.correlations();
326        assert!(!self.shared && grid.len() == channels && prediction.len() == channels);
327        let cells = channels * products;
328        let start = self.flags.len() - cells;
329        assert_eq!(self.prediction_flags.len(), start);
330        if self.prediction_flags.capacity() == 0 {
331            self.prediction_flags.reserve_exact(self.flags.capacity());
332        }
333        self.prediction_flags
334            .extend_from_slice(&self.flags[start..]);
335        for channel in 0..channels {
336            if !grid[channel] {
337                self.flags[start + channel * products..start + (channel + 1) * products].fill(true);
338            }
339            if !prediction[channel] {
340                self.prediction_flags[start + channel * products..start + (channel + 1) * products]
341                    .fill(true);
342            }
343        }
344    }

346    /// The row as the final model predictor reads it.
347    #[must_use]
348    pub fn prediction_row(&self, domain: usize, index: usize) -> NativeRow<'_> {
349        let mut row = self.row(domain, index);
350        if !self.prediction_flags.is_empty() {
351            let cells = self.channels() * self.correlations();
352            row.flags = &self.prediction_flags[index * cells..(index + 1) * cells];
353        }
354        row
355    }

357    /// Order the rows for locality: by antenna pair, a baseline's rows in
358    /// arrival order (time order for a MeasurementSet). Appending a row
359    /// returns the block to arrival order.
360    pub fn order_rows_by_baseline(&mut self) {
361        let rows = &self.rows;
362        self.order.clear();
363        self.order.extend(0..rows.len() as u32);
364        self.order
365            .sort_unstable_by_key(|&index| (rows[index as usize].context.antennas, index));
366    }

425    /// Row `index` as the operator of image domain `domain` reads it.
426    #[must_use]
427    pub fn row(&self, domain: usize, index: usize) -> NativeRow<'_> {
428        assert!(
429            domain < self.domains,
430            "the block projects on {} domains",
431            self.domains
432        );
433        let header = &self.rows[index];
434        let projection = &self.projections[index * self.domains + domain];
435        let channels = self.channels();
436        let cells = channels * self.correlations();
437        // Shared rows all read the one row of samples the block keeps.
438        let samples = if self.shared { 0 } else { index };
439        NativeRow {
440            uvw_m: projection.uvw_m,
441            phase_shift_m: projection.phase_shift_m,
442            pointing_offset_rad: projection.pointing_offset_rad,
443            frequencies_hz: &self.frequencies_hz[samples * channels..(samples + 1) * channels],
444            values: &self.values[samples * cells..(samples + 1) * cells],
445            weights: &self.weights[samples * cells..(samples + 1) * cells],
446            flags: &self.flags[samples * cells..(samples + 1) * cells],
447            row_flag: header.row_flag,
448            context: header.context,
449        }
450    }

625    #[test]
626    fn shared_rows_read_as_rows_pushed_whole() {
627        let (channels, correlations, domains, rows) = (5, 2, 2, 7);
628        let frequencies = [1.0e9, 1.1e9, 1.2e9, 1.3e9, 1.4e9];
629        let channel_indices = [0, 1, 2, 3, 4];
630        let correlation_indices = [0, 3];
631        let cells = channels * correlations;
632        let shape = BlockShape {
633            rows,
634            spectrum: RowSpectrum::evenly_spaced(channels),
635            correlations,
636            domains,
637        };
638        let projections = |row: usize| {
639            [0, 1].map(|domain| DomainProjection {
640                uvw_m: [row as f64, domain as f64, 0.5],
641                phase_shift_m: 0.25 * domain as f64,
642                pointing_offset_rad: [0.0; 2],
643            })
644        };
645        let mut whole = NativeBlock::with_capacity(shape);
646        whole.reset(domains, &channel_indices, &correlation_indices);
647        let mut shared = NativeBlock::shared_with_capacity(shape);
648        for _ in 0..2 {
649            shared.reset_shared(
650                domains,
651                &channel_indices,
652                &correlation_indices,
653                &frequencies,
654            );
655            for row in 0..rows {
656                shared.push_shared_row(header([0, row as u32 + 1], row as u64), &projections(row));
657            }
658        }
659        for row in 0..rows {
660            whole.push_row(
661                header([0, row as u32 + 1], row as u64),
662                &projections(row),
663                &frequencies,
664                &vec![Complex32::default(); cells],
665                &vec![1.0; cells],
666                &vec![false; cells],
667            );
668        }
669        assert_eq!(shared.len(), rows);
670        for row in 0..rows {
671            for domain in 0..domains {
672                let (a, b) = (whole.row(domain, row), shared.row(domain, row));
673                assert_eq!(a.uvw_m, b.uvw_m, "row {row}, domain {domain}");
674                assert_eq!(a.phase_shift_m, b.phase_shift_m);
675                assert_eq!(a.frequencies_hz, b.frequencies_hz);
676                assert_eq!(a.values, b.values);
677                assert_eq!(a.weights, b.weights);
678                assert_eq!(a.flags, b.flags);
679                assert_eq!(a.context.antennas, b.context.antennas);
680                assert_eq!(a.row_flag, b.row_flag);
681            }
682        }
683        assert!(
684            shared.capacity_bytes() <= shape.shared_bytes(),
685            "{} against {}",
686            shared.capacity_bytes(),
687            shape.shared_bytes()
688        );
689        assert!(shape.shared_bytes() < shape.bytes());
690    }
