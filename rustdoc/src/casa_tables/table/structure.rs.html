2//! Structural changes made in place on disk: the change is written to
3//! `table.dat` (and, where a storage manager goes away, its files are
4//! deleted); every other storage manager file is left as it is, so a
5//! MeasurementSet keeps its tiled, incremental and standard managers.
6//!
7//! Each operation edits what `table.dat` holds, then reloads the table from
8//! the disk ([`Table::resync_or_discard`]), so the handle is what the disk
9//! says without the change having loaded a row, and holds nothing a release
10//! of the write lock would write in full. An operation refused before it
11//! wrote leaves the handle as clean as it found it. The in-memory edits of
12//! [`Table::rename_column`] and [`Table::remove_column`] load every row and
13//! are for a table written in full afterwards.
14//!
15//! # C++ equivalent
16//!
17//! `Table::renameColumn` and `Table::removeColumn`, which change the
18//! table description and the column set and leave a storage manager that
19//! keeps other columns where it is.
20

28/// Prefix of the files of the storage manager with sequence number `seq_nr`:
29/// `table.f<seq_nr>`, the `_TSM<n>` files of a tiled manager, and the `i`
30/// file of a standard manager's indirect arrays. A manager whose number
31/// merely starts with the same digits is another manager.
32fn belongs_to_manager(file_name: &str, seq_nr: u32) -> bool {
33    let Some(rest) = file_name.strip_prefix(&format!("table.f{seq_nr}")) else {
34        return false;
35    };
36    rest.is_empty() || rest == "i" || rest.starts_with('_')
37}
38
39/// Deletes the files of the storage manager with sequence number `seq_nr`.
40fn remove_manager_files(table_dir: &Path, seq_nr: u32) -> std::io::Result<()> {
41    for entry in std::fs::read_dir(table_dir)? {
42        let entry = entry?;
43        if entry
44            .file_name()
45            .to_str()
46            .is_some_and(|name| belongs_to_manager(name, seq_nr))
47        {
48            std::fs::remove_file(entry.path())?;
49        }
50    }
51    Ok(())
52}

54/// Whether a hypercolumn definition of the table description lists `name`
55/// as a coordinate or an id column (the private keyword `Hypercolumn_<name>`
56/// lists its `data`, `coord` and `id` columns).
57fn is_id_or_coordinate_column(private_keywords: &RecordValue, name: &str) -> bool {
58    private_keywords
59        .fields()
60        .iter()
61        .filter(|field| field.name.starts_with("Hypercolumn_"))
62        .filter_map(|field| match &field.value {
63            Value::Record(definition) => Some(definition),
64            _ => None,
65        })
66        .flat_map(|definition| ["coord", "id"].map(|list| definition.get(list)))
67        .any(|list| match list {
68            Some(Value::Array(ArrayValue::String(names))) => names.iter().any(|n| n == name),
69            _ => false,
70        })
71}

73/// Renames `old` to `new` in the hypercolumn definitions of the table
74/// description, as `TableDesc::renameColumn` does through
75/// `adjustHypercolumns` (casacore's own handling of coordinate and id
76/// columns there is not followed: they are refused, not renamed).
77fn rename_in_hypercolumns(private_keywords: &mut RecordValue, old: &str, new: &str) {
78    edit_hypercolumns(private_keywords, |names| {
79        for name in names {
80            if name == old {
81                *name = new.to_string();
82            }
83        }
84    });
85}

87/// Applies `edit` to each list of column names of each hypercolumn
88/// definition in `private_keywords`.
89fn edit_hypercolumns(private_keywords: &mut RecordValue, edit: impl Fn(&mut Vec<String>)) {
90    for field in private_keywords.fields_mut() {
91        if !field.name.starts_with("Hypercolumn_") {
92            continue;
93        }
94        let Value::Record(definition) = &mut field.value else {
95            continue;
96        };
97        for list in definition.fields_mut() {
98            let Value::Array(ArrayValue::String(array)) = &mut list.value else {
99                continue;
100            };
101            let mut names: Vec<String> = array.iter().cloned().collect();
102            edit(&mut names);
103            if names.len() != array.len() || names.iter().zip(array.iter()).any(|(a, b)| a != b) {
104                *array = ndarray::ArrayD::from_shape_vec(ndarray::IxDyn(&[names.len()]), names)
105                    .expect("a vector's shape is its length");
106            }
107        }
108    }
109}

205    /// Renames column `old` to `new` in the table on disk, without loading
206    /// its rows or touching a storage manager's file.
207    ///
208    /// Only `table.dat` changes: the column's description, and its name in
209    /// the hypercolumn definitions. The name the column's storage manager
210    /// knows it by (casacore's `originalName`) stays as it was. Fails if
211    /// `old` is not a column or `new` already is one. The table is reloaded
212    /// from the disk afterwards, so changes this handle has not saved are
213    /// discarded.
214    ///
215    /// # C++ equivalent
216    ///
217    /// `Table::renameColumn`.
218    pub fn rename_column_in_place(&mut self, old: &str, new: &str) -> Result<(), TableError> {
219        self.edit_structure_in_place("rename_column_in_place", false, |_, dat, _| {
220            let index = column_index(dat, old)?;
221            if column_index(dat, new).is_ok() {
222                return Err(SchemaError::DuplicateColumn(new.to_string()).into());
223            }
224            // The name of an id or coordinate column is also kept in the
225            // hypercube values of its tiled manager's file, which an edit of
226            // `table.dat` alone leaves behind: the renamed column would be
227            // unreadable. (casacore's own `adjustHypercolumns` does not carry
228            // these names over either.)
229            if is_id_or_coordinate_column(&dat.table_desc.private_keywords, old) {
230                return Err(TableError::Storage(format!(
231                    "column \"{old}\" is an id or coordinate column of a hypercolumn: its name \
232                     is also held in the hypercube values of its storage manager's file, which \
233                     cannot be renamed in place"
234                )));
235            }
236            dat.table_desc.columns[index].col_name = new.to_string();
237            dat.column_set.columns[index].original_name = new.to_string();
238            let on_file = dat
239                .names_on_file
240                .remove(old)
241                .unwrap_or_else(|| old.to_string());
242            if on_file != new {
243                dat.names_on_file.insert(new.to_string(), on_file);
244            }
245            rename_in_hypercolumns(&mut dat.table_desc.private_keywords, old, new);
246            Ok(Vec::new())
247        })
248    }

250    /// Removes the columns `names` from the table on disk, without loading
251    /// its rows. Every storage manager that held any of them must hold only
252    /// them: its files are deleted, and the other managers' files are not
253    /// touched. A column whose manager keeps other columns cannot be taken
254    /// out in place (the manager's file holds its data), and is refused
255    /// before anything changes. The table is reloaded from the disk
256    /// afterwards, so changes this handle has not saved are discarded.
257    ///
258    /// # C++ equivalent
259    ///
260    /// `Table::removeColumn`.
261    pub fn remove_columns_in_place(&mut self, names: &[&str]) -> Result<(), TableError> {
262        self.remove_persisted_columns(names, false)
263    }

265    /// Remove persisted columns, allowing shared StandardStMan managers.
266    /// A shared StandardStMan is rebuilt from its surviving columns while
267    /// other managers keep their existing files and bindings. A compressed
268    /// virtual column and its backing columns may be removed together.
269    /// Shared groups whose storage managers cannot remove individual
270    /// columns are refused before the table is changed.
271    pub fn remove_columns_preserving_managers(&mut self, names: &[&str]) -> Result<(), TableError> {
272        self.remove_persisted_columns(names, true)
273    }

275    fn remove_persisted_columns(
276        &mut self,
277        names: &[&str],
278        allow_shared_standard: bool,
279    ) -> Result<(), TableError> {
280        self.edit_structure_in_place(
281            "remove_columns_in_place",
282            allow_shared_standard,
283            |table, dat, path| {
284                let mut indices = names
285                    .iter()
286                    .map(|name| column_index(dat, name))
287                    .collect::<Result<Vec<_>, _>>()?;
288                indices.sort_unstable();
289                indices.dedup();
290                let (mut gone, shared) = removal_groups(dat, &indices, allow_shared_standard)?;
291                if allow_shared_standard {
292                    check_virtual_dependencies(table, dat, names)?;
293                    gone.extend(rewrite_standard_groups(table, dat, path, &shared)?);
294                }
295                detach_columns(dat, indices, names, &gone);
296                Ok(gone)
297            },
298        )
299    }

301    /// Applies `edit` to the table's `table.dat` and writes it, then deletes
302    /// the files of the storage managers `edit` returns the numbers of, and
303    /// reloads the table from the disk. `edit` refuses before anything is
304    /// written.
305    fn edit_structure_in_place(
306        &mut self,
307        operation: &str,
308        allow_persisted_virtual: bool,
309        edit: impl FnOnce(&Self, &mut TableDatContents, &Path) -> Result<Vec<u32>, TableError>,
310    ) -> Result<(), TableError> {
311        if self.kind != TableKind::Plain {
312            return Err(TableError::Storage(format!(
313                "{operation} requires a plain disk-backed table"
314            )));
315        }
316        if !self.virtual_bindings.is_empty()
317            || (!allow_persisted_virtual && !self.virtual_columns.is_empty())
318        {
319            return Err(TableError::Storage(format!(
320                "{operation} does not support virtual columns"
321            )));
322        }
323        let path = self
324            .source_path
325            .clone()
326            .ok_or_else(|| TableError::Storage("table has no source path".to_string()))?;
327        // Every admitted write operation counts as a change of the handle,
328        // and a change the disk lacks is what releasing the write lock writes
329        // in full (as `StManAipsIO`, whatever the table's layout). This
330        // operation writes to the disk itself, so it must leave the handle
331        // as clean as it found it.
332        let dirty_before = self.holds_unflushed_changes();
333        let auto_unlock = self.begin_write_operation(operation)?;
334        let mut wrote = false;
335        let result = (|| {
336            let control_path = path.join(crate::storage::TABLE_CONTROL_FILE);
337            let TableDatResult::Plain(mut table_dat) =
338                crate::storage::table_control::read_table_dat_dispatch(&control_path)?
339            else {
340                return Err(TableError::Storage(format!(
341                    "{operation} only supports plain tables"
342                )));
343            };
344            if table_dat.uses_multifile() {
345                return Err(TableError::Storage(format!(
346                    "{operation} cannot change MultiFile storage-manager bindings"
347                )));
348            }
349            let persisted_rows = self.require_persisted_rows(&path, &table_dat)?;
350            table_dat.raise_row_count(persisted_rows);
351            let removed_managers = edit(self, &mut table_dat, &path)?;
352            self.note_persisted_write(&path);
353            wrote = true;
354            crate::storage::table_control::write_table_dat(&control_path, &table_dat)?;
355            for seq_nr in removed_managers {
356                remove_manager_files(&path, seq_nr).map_err(StorageError::from)?;
357            }
358            crate::storage::tiled_stman::invalidate_shared_tile_cache_for_table(&path);
359            Ok(())
360        })();
361        if !wrote && !dirty_before {
362            // Refused before anything reached the disk.
363            self.mark_flushed();
364        }
365        let result = if wrote {
366            // The disk changed, wholly or in part: the handle takes what it
367            // holds from the disk, and is clean whether or not that works.
368            let reloaded = self.resync_or_discard();
369            result.and(reloaded)
370        } else {
371            result
372        };
373        self.finish_write_operation(auto_unlock, result)
374    }
375}

377type SharedStandardGroup = (u32, Vec<usize>);
378
379fn removal_groups(
380    dat: &TableDatContents,
381    removed: &[usize],
382    allow_shared_standard: bool,
383) -> Result<(Vec<u32>, Vec<SharedStandardGroup>), TableError> {
384    let mut drop_managers = Vec::new();
385    let mut shared = Vec::new();
386    for manager in &dat.column_set.data_managers {
387        let columns = dat
388            .column_set
389            .columns
390            .iter()
391            .enumerate()
392            .filter_map(|(index, column)| (column.dm_seq_nr == manager.seq_nr).then_some(index));
393        let (retain, discard): (Vec<_>, Vec<_>) =
394            columns.partition(|index| removed.binary_search(index).is_err());
395        if discard.is_empty() {
396            continue;
397        }
398        if retain.is_empty() {
399            drop_managers.push(manager.seq_nr);
400        } else if allow_shared_standard && manager.type_name == "StandardStMan" {
401            shared.push((manager.seq_nr, retain));
402        } else {
403            let name = &dat.table_desc.columns[discard[0]].col_name;
404            return Err(TableError::Storage(format!(
405                "column {name:?} shares storage manager {} ({}) with {} other columns",
406                manager.seq_nr,
407                manager.type_name,
408                retain.len()
409            )));
410        }
411    }
412    Ok((drop_managers, shared))
413}

415fn check_virtual_dependencies(
416    table: &Table,
417    dat: &TableDatContents,
418    removed: &[&str],
419) -> Result<(), TableError> {
420    for column in &dat.table_desc.columns {
421        if !table.virtual_columns.contains(&column.col_name)
422            || removed.contains(&column.col_name.as_str())
423        {
424            continue;
425        }
426        let depends_on_removed = column
427            .keywords
428            .fields()
429            .iter()
430            .any(|field| match &field.value {
431                Value::Scalar(ScalarValue::String(value)) if field.name.starts_with('_') => {
432                    removed.iter().any(|name| value.contains(name))
433                }
434                _ => false,
435            });
436        if depends_on_removed {
437            return Err(TableError::Storage(format!(
438                "removing columns required by virtual column {:?}",
439                column.col_name
440            )));
441        }
442    }
443    Ok(())
444}

446fn detach_columns(dat: &mut TableDatContents, indices: Vec<usize>, names: &[&str], gone: &[u32]) {
447    for index in indices.into_iter().rev() {
448        dat.table_desc.columns.remove(index);
449        dat.column_set.columns.remove(index);
450    }
451    for name in names {
452        dat.names_on_file.remove(*name);
453    }
454    dat.column_set
455        .data_managers
456        .retain(|manager| !gone.contains(&manager.seq_nr));
457    edit_hypercolumns(&mut dat.table_desc.private_keywords, |list| {
458        list.retain(|name| !names.contains(&name.as_str()));
459    });
460    let keywords = &mut dat.table_desc.private_keywords;
461    let emptied: Vec<_> = keywords
462        .fields()
463        .iter()
464        .filter(|field| field.name.starts_with("Hypercolumn_"))
465        .filter(|field| match &field.value {
466            Value::Record(definition) => matches!(definition.get("data"),
467                Some(Value::Array(array)) if array.shape().iter().product::<usize>() == 0),
468            _ => false,
469        })
470        .map(|field| field.name.clone())
471        .collect();
472    for name in emptied {
473        keywords.remove(&name);
474    }
475}

477fn rewrite_standard_groups(
478    table: &Table,
479    dat: &mut TableDatContents,
480    path: &Path,
481    groups: &[SharedStandardGroup],
482) -> Result<Vec<u32>, TableError> {
483    let mut next_seq = dat
484        .column_set
485        .data_managers
486        .iter()
487        .map(|entry| entry.seq_nr)
488        .max()
489        .map_or(0, |seq| seq + 1);
490    let mut new_managers = Vec::new();
491    for (seq_nr, indices) in groups {
492        let columns: Vec<_> = indices
493            .iter()
494            .map(|&index| dat.table_desc.columns[index].clone())
495            .collect();
496        let manager = dat
497            .column_set
498            .data_managers
499            .iter()
500            .find(|manager| manager.seq_nr == *seq_nr)
501            .expect("shared StandardStMan is present");
502        let name = dm_name_from_blob(&manager.data)
503            .unwrap_or_else(|| columns[0].data_manager_group.clone());
504        let staged = path.join(format!("table.f{next_seq}"));
505        let source = path.join(format!("table.f{seq_nr}"));
506        let cache = RefCell::new((usize::MAX, 0, Vec::new()));
507        let result = rewrite_ssm_columns(
508            &source,
509            &staged,
510            &columns,
511            table.row_count(),
512            &name,
513            |column, row| {
514                let mut slot = cache.borrow_mut();
515                if slot.0 != column || row < slot.1 || row - slot.1 >= slot.2.len() {
516                    let values = structural_cells(table, &columns[column], row)
517                        .map_err(|error| StorageError::FormatMismatch(error.to_string()))?;
518                    *slot = (column, row, values);
519                }
520                let offset = row - slot.1;
521                Ok(slot.2[offset].take())
522            },
523        );
524        let data = match result {
525            Ok(data) => data,
526            Err(error) => {
527                for (entry, _) in &new_managers {
528                    let _ = remove_manager_files(path, *entry);
529                }
530                let _ = remove_manager_files(path, next_seq);
531                return Err(error.into());
532            }
533        };
534        new_managers.push((
535            next_seq,
536            crate::storage::table_control::DataManagerEntry {
537                seq_nr: next_seq,
538                type_name: manager.type_name.clone(),
539                data,
540            },
541        ));
542        for &index in indices {
543            dat.column_set.columns[index].dm_seq_nr = next_seq;
544        }
545        next_seq += 1;
546    }
547    dat.column_set
548        .data_managers
549        .extend(new_managers.into_iter().map(|(_, manager)| manager));
550    dat.column_set.seq_count = next_seq.max(dat.column_set.seq_count);
551    Ok(groups.iter().map(|(seq, _)| *seq).collect())
552}

554fn structural_cells(
555    table: &Table,
556    column: &ColumnDescContents,
557    row: usize,
558) -> Result<Vec<Option<Value>>, TableError> {
559    let reader = table.column_accessor(&column.col_name)?;
560    // Keep at most one array cell, or a small block of scalar cells.
561    if column.is_record() {
562        return Ok(vec![Some(Value::Record(reader.record_cell(row)?))]);
563    }
564    if column.is_array {
565        return Ok(reader
566            .array_cells_owned_uncached(&[row])?
567            .into_iter()
568            .map(|value| value.map(Value::Array))
569            .collect());
570    }
571    let rows: Vec<_> = (row..table.row_count().min(row.saturating_add(512))).collect();
572    Ok(reader
573        .scalar_cells_owned_for_rows(&rows)?
574        .into_iter()
575        .map(|value| value.map(Value::Scalar))
576        .collect())
577}

579/// The position of column `name` in the table description (the column set
580/// lists its columns in the same order).
581fn column_index(dat: &TableDatContents, name: &str) -> Result<usize, TableError> {
582    dat.table_desc
583        .columns
584        .iter()
585        .position(|column| column.col_name == name)
586        .ok_or_else(|| TableError::SchemaColumnUnknown {
587            column: name.to_string(),
588        })
589}
