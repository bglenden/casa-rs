2//! Payload-free array-cell metadata through the ordinary manager bindings.
3use super::table_control::{ColumnDescContents, DataManagerEntry};
4use super::*;
5
6struct Binding<'a> {
7    manager: &'a DataManagerEntry,
8    columns: Vec<&'a ColumnDescContents>,
9    column: usize,
10}
11
12impl<'a> Binding<'a> {
13    fn of(table: &'a TableDatContents, name: &str) -> Result<Self, StorageError> {
14        let descriptor = table
15            .table_desc
16            .columns
17            .iter()
18            .position(|column| column.col_name == name)
19            .ok_or_else(|| {
20                StorageError::FormatMismatch(format!("array column '{name}' not found"))
21            })?;
22        if !table.table_desc.columns[descriptor].is_array {
23            return Err(StorageError::FormatMismatch(format!(
24                "column '{name}' is not an array column"
25            )));
26        }
27        let sequence = table
28            .column_set
29            .columns
30            .iter()
31            .find(|column| column.original_name == name)
32            .ok_or_else(|| {
33                StorageError::FormatMismatch(format!(
34                    "array column '{name}' missing ColumnSet binding"
35                ))
36            })?
37            .dm_seq_nr;
38        let manager = table
39            .column_set
40            .data_managers
41            .iter()
42            .find(|manager| manager.seq_nr == sequence)
43            .ok_or_else(|| {
44                StorageError::FormatMismatch(format!(
45                    "array column '{name}' missing data manager {sequence}"
46                ))
47            })?;
48        let bindings = table
49            .column_set
50            .columns
51            .iter()
52            .filter(|binding| binding.dm_seq_nr == sequence)
53            .collect::<Vec<_>>();
54        let column = bindings
55            .iter()
56            .position(|binding| binding.original_name == name)
57            .ok_or_else(|| {
58                StorageError::FormatMismatch(format!(
59                    "array column '{name}' missing data-manager column binding"
60                ))
61            })?;
62        let columns = bindings
63            .into_iter()
64            .map(|binding| {
65                table
66                    .table_desc
67                    .columns
68                    .iter()
69                    .find(|column| column.col_name == binding.original_name)
70                    .ok_or_else(|| {
71                        StorageError::FormatMismatch(format!(
72                            "array binding '{}' has no descriptor",
73                            binding.original_name
74                        ))
75                    })
76            })
77            .collect::<Result<Vec<_>, _>>()?;
78        Ok(Self {
79            manager,
80            columns,
81            column,
82        })
83    }
84}

86impl CompositeStorage {
87    pub(crate) fn plain_array_cell_shapes_uncached(
88        &self,
89        table_path: &Path,
90        metadata: &RetainedTableReadMetadata,
91        column: &str,
92        rows: &[usize],
93    ) -> Result<Vec<Option<Vec<usize>>>, StorageError> {
94        if rows.is_empty() {
95            return Ok(Vec::new());
96        }
97        let binding = Binding::of(&metadata.table_dat, column)?;
98        let manager = binding.manager;
99        match manager.type_name.as_str() {
100            "StManAipsIO" => {
101                let data = table_path.join(format!("{TABLE_DATA_FILE_PREFIX}{}", manager.seq_nr));
102                let descriptors = binding
103                    .columns
104                    .iter()
105                    .map(|column| (*column).clone())
106                    .collect::<Vec<_>>();
107                stman_aipsio::read_stman_array_column_shapes(
108                    &data,
109                    &descriptors,
110                    binding.column,
111                    rows,
112                    ByteOrder::BigEndian,
113                )?
114                .ok_or_else(|| {
115                    StorageError::FormatMismatch(format!(
116                        "metadata-only StManAipsIO shapes unavailable for '{column}'"
117                    ))
118                })
119            }
120            "IncrementalStMan" => {
121                let data = table_path.join(format!("{TABLE_DATA_FILE_PREFIX}{}", manager.seq_nr));
122                incremental_stman::read_ism_array_column_shapes(
123                    &data,
124                    &binding.columns,
125                    binding.column,
126                    rows,
127                )?
128                .ok_or_else(|| {
129                    StorageError::FormatMismatch(format!(
130                        "metadata-only ISM shapes unavailable for '{column}'"
131                    ))
132                })
133            }
134            "StandardStMan" => {
135                let data = table_path.join(format!("{TABLE_DATA_FILE_PREFIX}{}", manager.seq_nr));
136                standard_stman::read_ssm_array_column_shapes(
137                    &data,
138                    &manager.data,
139                    &binding.columns,
140                    binding.column,
141                    rows,
142                )?
143                .ok_or_else(|| {
144                    StorageError::FormatMismatch(format!(
145                        "metadata-only shapes unavailable for SSM column '{column}'"
146                    ))
147                })
148            }
149            "TiledColumnStMan" | "TiledShapeStMan" | "TiledCellStMan" | "TiledDataStMan" => {
150                metadata.tiled.array_shapes(
151                    manager.seq_nr,
152                    &binding.columns,
153                    &metadata.table_dat.table_desc,
154                    binding.column,
155                    rows,
156                )
157            }
158            other => Err(StorageError::UnsupportedDataManager(format!(
159                "metadata-only array shapes for {other}"
160            ))),
161        }
162    }
163
164    pub(crate) fn plain_array_cell_is_defined_uncached(
165        &self,
166        table: &TableDatContents,
167        metadata: &TiledReadMetadata,
168        column: &str,
169        row: usize,
170    ) -> Result<bool, StorageError> {
171        let binding = Binding::of(table, column)?;
172        if binding.manager.type_name != "TiledShapeStMan" {
173            return Err(StorageError::FormatMismatch(format!(
174                "metadata-only array-cell definedness for column '{column}' requires TiledShapeStMan, found {}",
175                binding.manager.type_name
176            )));
177        }
178        metadata.shape_cell_is_defined(binding.manager.seq_nr, binding.column, row)
179    }
