3//! The task surface of native CASA-RS imaging, on the shared task front end
4//! (#931).
5//!
6//! The front end resolves `tclean` parameters through the contracts catalog:
7//! command-line flags and `--json-run` requests become the same resolved
8//! values, and those values, and nothing else, make the [`ImagingRequest`]

31/// Per-invocation execution options for the `tclean` imaging task adapter.
32#[derive(Clone, Debug)]
33pub struct ImagingExecutionOptions {
34    /// Cooperative cancellation token observed by the imaging run.
35    pub cancel: Cancel,
36    /// Whether to write `<imagename>.summary.json` beside the products.
37    pub write_run_summary: bool,
38    /// Optional path to the EVLA dish surface table used by AWP2.
39    pub evla_surface: Option<PathBuf>,
40}
41
42impl ImagingExecutionOptions {
43    /// Read CLI environment configuration (`CASA_RS_IMAGER_RUN_SUMMARY` and
44    /// `HOME`) with a fresh cancellation token.
45    pub fn from_env() -> Self {
46        Self {
47            cancel: Cancel::new(),
48            write_run_summary: run_summary_requested(),
49            evla_surface: default_evla_surface(),
50        }
51    }
52}
53
54impl Default for ImagingExecutionOptions {
55    fn default() -> Self {
56        Self {
57            cancel: Cancel::new(),
58            write_run_summary: false,
59            evla_surface: default_evla_surface(),
60        }
61    }
62}

64fn default_evla_surface() -> Option<PathBuf> {
65    std::env::var_os("HOME").map(|home| PathBuf::from(home).join(".casa/data/nrao/VLA/VLA.surface"))
66}
67
68fn imager_surface() -> Result<SurfaceContractBundle, String> {
69    casa_contracts::builtin_surface_bundle("tclean")
70}
71
72fn workspace() -> Result<PathBuf, String> {
73    std::env::current_dir().map_err(|error| format!("the working directory: {error}"))
74}

76/// The request `resolved`, every tclean parameter, makes. A value the
77/// catalog accepts but casa-rs cannot honour in the request's modes, or that
78/// the request's own read refuses (`stokes = "IX"`), is refused at its
79/// parameter.
80fn request_from(
81    resolved: Map<String, Value>,
82    evla_surface_fallback: Option<&PathBuf>,
83) -> Result<(ImagingParameters, ImagingRequest), TaskFailure> {
84    let mut request = ImagingRequest::from_tclean(&resolved).map_err(|refusal| {
85        let failure = TaskFailure::validation(refusal.message);
86        if refusal.parameter.is_empty() {
87            failure
88        } else {
89            failure.at(refusal.parameter)
90        }
91    })?;
92    // AWP2 reads the same physical dish data as CASA when the host has
93    // the local data tree. The imaging library takes an explicit path.
94    if let casa_imaging_application::Gridder::Awp2 { evla_surface, .. } = &mut request.gridder
95        && evla_surface.is_none()
96    {
97        *evla_surface = evla_surface_fallback.cloned();
98    }
99    Ok((ImagingParameters(resolved), request))
100}

102/// Resolve `parameters`, catalog-named and sparse, over the catalog's
103/// defaults: the resolved parameters and the request they make.
104pub fn resolve_request(
105    parameters: &ImagingParameters,
106) -> Result<(ImagingParameters, ImagingRequest), String> {
107    let resolved = resolve_plain_values(imager_surface()?, workspace()?, &parameters.0)
108        .map_err(|error| format!("resolve tclean parameters: {error}"))?;
109    let default_surface = default_evla_surface();
110    request_from(resolved, default_surface.as_ref()).map_err(|failure| failure.message)
111}

113/// The environment variable that asks the `casars-imager` binary for its
114/// run summary. CASA's `tclean` writes nothing beside its image products, so
115/// neither does a tclean run here (owner decision of 2026-11-02, #980); set
116/// to `1`, the binary also writes `<imagename>.summary.json`, the run's
117/// status, resolved parameters, backend and each phase's time and memory,
118/// for performance tooling ([`casa_imaging_application::RunSummary`]).
119pub const RUN_SUMMARY_VARIABLE: &str = "CASA_RS_IMAGER_RUN_SUMMARY";
120
121/// Whether this process was asked for run summaries
122/// ([`RUN_SUMMARY_VARIABLE`]).
123fn run_summary_requested() -> bool {
124    std::env::var_os(RUN_SUMMARY_VARIABLE).is_some_and(|value| value == "1")
125}

127/// Run `request`, resolved as `parameters`, as CASA's tclean does: it
128/// writes the image products and nothing else beside them.
129pub fn run(
130    parameters: ImagingParameters,
131    request: &ImagingRequest,
132) -> Result<ImagerRunTaskResult, String> {
133    run_with_options(parameters, request, &ImagingExecutionOptions::default())
134        .map(|(result, _)| result)
135}
136
137/// Keep what the CASA return needs beyond the report in memory, without
138/// adding fields to the serialized task report or the run summary, which is
139/// written beside the products only when `options.write_run_summary` asks
140/// for it.
141fn run_with_options(
142    parameters: ImagingParameters,
143    request: &ImagingRequest,
144    options: &ImagingExecutionOptions,
145) -> Result<(ImagerRunTaskResult, RunReturn), String> {
146    // Task continuation is distinct from internal atomic output replacement.
147    let restart = parameters
148        .0
149        .get("restart")
150        .and_then(Value::as_bool)
151        .ok_or_else(|| "resolved tclean restart must be boolean".to_owned())?;
152    let request = request
153        .with_tclean_output_policy(restart)
154        .map_err(|error| error.to_string())?;
155    let request = &request;
156    let started = Instant::now();
157    let summary = options.write_run_summary.then(|| {
158        let mut path = request.imagename.clone().into_os_string();
159        path.push(".summary.json");
160        SummaryTarget {
161            path: path.into(),
162            request: Value::Object(parameters.0.clone()),
163        }
164    });
165    let context = RunContext {
166        host: HostResources::detect().map_err(|error| error.to_string())?,
167        policy: request.resource_policy(),
168        cancel: options.cancel.clone(),
169        summary,
170    };
171    let outcome = execute(request, context).map_err(|error| error.to_string())?;
172    let result =
173        ImagerRunTaskResult::from_outcome(parameters, request, &outcome, started.elapsed());
174    Ok((
175        result,
176        RunReturn {
177            statistics: outcome.final_residual_statistics,
178            plane_rows: outcome.plane_rows,
179        },
180    ))
181}

183/// The front end of the `casars-imager` binary, which hosts `tclean`, with
184/// caller-provided execution options and cancellation token.
185pub fn task_front_end_with_options(
186    options: ImagingExecutionOptions,
187) -> TaskFrontEnd<impl Fn(&TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure>> {
188    TaskFrontEnd::new_with_deferred_schema(
189        "casars-imager",
190        imager_task_schema_bundle(),
191        &["tclean"],
192        move |task| run_task_with_options(task, &options),
193    )
194    .expect("the imager contract carries its surface")
195}
196
197/// The front end of the `casars-imager` binary, using default per-invocation
198/// execution options.
199pub fn task_front_end()
200-> TaskFrontEnd<impl Fn(&TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure>> {
201    task_front_end_with_options(ImagingExecutionOptions::from_env())
202}

204/// The resolved parameters of `task`, a tclean run, and the request they
205/// make; a request they cannot make is refused.
206pub fn task_request(
207    task: &TaskInvocation<'_>,
208) -> Result<(ImagingParameters, ImagingRequest), TaskFailure> {
209    let default_surface = default_evla_surface();
210    request_from(task.plain_values(), default_surface.as_ref())
211}

213/// Run `task`, one resolved imager request, with default execution options.
214pub fn run_task(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
215    run_task_with_options(task, &ImagingExecutionOptions::from_env())
216}
217
218/// Run `task`, one resolved imager request, observing `options.cancel`: if
219/// cancelled, the run stops at its next safe point, publishes nothing, and
220/// replies that it was interrupted.
221pub fn run_task_with_options(
222    task: &TaskInvocation<'_>,
223    options: &ImagingExecutionOptions,
224) -> Result<TaskSuccess, TaskFailure> {
225    let (parameters, request) = request_from(task.plain_values(), options.evla_surface.as_ref())?;
226    // The record echoes some of tclean's values: a dirty run's threshold as
227    // the caller gave it, a number as a number and text as text, and
228    // tclean's default 0.0 where it was not given.
229    let mut values = parameters.0.clone();
230    values.insert(
231        "threshold".to_string(),
232        task.session()
233            .given("threshold")
234            .map_or(Value::from(0.0), ParameterValue::to_plain_json),
235    );
236    let (result, held) = run_with_options(parameters, &request, options).map_err(|error| {
237        if options.cancel.is_cancelled() {
238            TaskFailure::interrupted(format!("{error}; nothing was published"))
239        } else {
240            TaskFailure::execution(error)
241        }
242    })?;
243    let text = format!(
244        "Wrote CASA-compatible products at prefix {} ({} gridded samples, {} major cycles, \
245         {} reported minor iterations, {} actual components, stop={:?})\n",
246        request.imagename.display(),
247        result.run.gridded_samples,
248        result.run.major_cycles,
249        result.run.minor_iterations,
250        result.run.actual_minor_iterations,
251        result.run.clean_stop_reason,
252    );
253    let value = tclean_record(&request, &result.run, &values, &held);
254    history::write_tclean_history(&request.imagename, task);
255    Ok(TaskSuccess::new(value, &result)?.with_text(text))
256}

259mod tests {
260    use super::*;
261    use casa_imaging_application::{default_tclean_parameters, single_dish};

263
264    #[test]
265    fn tclean_domain_defaults_match_catalog_resolution() {
266        let overrides = json!({
267            "vis": ["unused.ms"],
268            "imagename": "unused",
269            "cell": ["1arcsec"],
270        });
271        let overrides = overrides.as_object().expect("object");
272        let catalog_defaults = resolve_plain_values(
273            casa_contracts::builtin_surface_bundle("tclean").expect("tclean surface"),
274            PathBuf::from("."),
275            overrides,
276        )
277        .expect("tclean defaults resolve");
278        let mut expected = default_tclean_parameters();
279        for (key, value) in overrides {
280            expected.insert(key.clone(), value.clone());
281        }
282        assert_eq!(expected, catalog_defaults);
283    }
