25fn copy<T: ImageExprValue>(
26    input: &dyn ImageInterface<T>,
27    coordinates: RecordValue,
28    options: ImageRebinOptions<'_>,
29    beams: &BeamView,
30    path: Option<&Path>,
31) -> PyResult<Storage<T>> {
32    let rebin = ImageRebin::new(input, coordinates, options, beams.record()).map_err(raised)?;
33    let section = rebin.section();
34    let mut info = section.image_info().to_record();
35    let source_beams = beams.beam_set().map_err(raised)?;
36    if source_beams.is_multi() {
37        let channels = rebin.beam_planes(
38            input.find_axis(CoordinateType::Spectral),
39            source_beams.n_channels(),
40        );
41        let stokes = rebin.beam_planes(
42            input.find_axis(CoordinateType::Stokes),
43            source_beams.n_stokes(),
44        );
45        subset_beam_record(
46            &mut info,
47            Some(beams.record()),
48            &source_beams,
49            &channels,
50            &stokes,
51        )
52        .map_err(raised)?;
53    } else {
54        beams.apply_to(&mut info);
55    }
56    let mut product = Storage::create(
57        section.shape.clone(),
58        section.coordinates.clone(),
59        path,
60        None,
61    )
62    .map_err(raised)?;
63    with_storage!(&mut product, output => rebin.copy_to(input, output.as_mut())).map_err(raised)?;
64    with_storage!(&mut product, output => output.set_image_info_record(info)).map_err(raised)?;
65    if let Some(record) = &section.coordinate_record {
66        match &mut product {
67            Storage::Paged(image) => image
68                .set_coordinate_record(record.clone())
69                .map_err(raised)?,
70            Storage::Temp {
71                record: coordinates,
72                ..
73            } => coordinates.clone_from(record),
74        }
75    }
76    Ok(product)
77}

79fn rebin_storage<T: ImageExprValue>(
80    source: &Storage<T>,
81    coordinates: RecordValue,
82    options: ImageRebinOptions<'_>,
83    beams: &BeamView,
84    path: Option<&Path>,
85) -> PyResult<Storage<T>> {
86    with_storage!(source, image => copy(image.as_ref(), coordinates, options, beams, path))
87}

89#[pymethods]
90impl PyImageTool {
91    /// Average selected pixels in bins and return the rebinned image.
92    #[allow(clippy::too_many_arguments)]
93    #[pyo3(signature = (outfile, bin, region=None, mask=None, dropdeg=false, overwrite=false, stretch=false, crop=false), text_signature = "($self, outfile, bin, region='', mask='', dropdeg=False, overwrite=False, stretch=False, crop=False)")]
94    fn rebin(
95        &self,
96        py: Python<'_>,
97        outfile: &str,
98        bin: IntList,
99        region: Option<&Bound<'_, PyAny>>,
100        mask: Option<&Bound<'_, PyAny>>,
101        dropdeg: bool,
102        overwrite: bool,
103        stretch: bool,
104        crop: bool,
105    ) -> PyResult<Py<Self>> {
106        let region = Region::capture(py, region);
107        let mask = mask_argument(mask);
108        let product = py.detach(|| {
109            let state = self.state();
110            let attachment = state.image.as_ref().ok_or_else(|| raised("Unable to create image"))?;
111            let region = region?;
112            let mask = mask?;
113            let mut live_coordinates = None;
114            let mut product = publish_image(outfile, overwrite, |path| {
115                let mut handle = attachment.lock();
116                match &*handle {
117                    ImageHandle::Double(_) => return Err(raised("rebin does not support images with double precision pixel values.")),
118                    ImageHandle::DComplex(_) => return Err(raised("rebin does not support images with complex<double> precision pixel values.")),
119                    _ => {}
120                }
121                let factors = bin.0.iter().map(|&n| {
122                    usize::try_from(n).map_err(|_| raised("All factors must be positive"))
123                }).collect::<PyResult<Vec<_>>>()?;
124                if !outfile.is_empty() {
125                    for source in handle.path().into_iter().chain(attachment.operand_paths().iter().map(|path| path.as_path())) {
126                        ensure_outfile_does_not_alias_input(Path::new(outfile), source).map_err(raised)?;
127                    }
128                }
129                attachment.require_pixel_read_lock(&mut handle)?;
130                let beams = attachment.beam_view().clone();
131                let coordinates = handle.coordinate_record();
132                let options = ImageRebinOptions {
133                    factors: &factors,
134                    region: region.as_ref().map(Region::as_region),
135                    mask: (!mask.is_empty()).then_some(mask.as_str()),
136                    stretch,
137                    dropdeg,
138                    crop,
139                };
140                let mut output = match &*handle {
141                    ImageHandle::Float(Storage::Temp { image, fits: Some(fits), .. }) => {
142                        ImageHandle::Float(copy(&fits.image_view(image), coordinates, options, &beams, path)?)
143                    }
144                    ImageHandle::Float(source) => ImageHandle::Float(rebin_storage(source, coordinates, options, &beams, path)?),
145                    ImageHandle::Complex(source) => ImageHandle::Complex(rebin_storage(source, coordinates, options, &beams, path)?),
146                    _ => unreachable!("double precision is checked before rebinning"),
147                };
148                let history = with_pixels!(&*handle, source => match source {
149                    Storage::Paged(image) => image_history_rows(image.name().expect("native image path")).map_err(raised)?,
150                    Storage::Temp { image, .. } => image.history_rows().map_err(raised)?.into_owned(),
151                });
152                with_pixels!(&mut output, dest => with_storage!(dest, image => image.append_history_entries(&history))).map_err(raised)?;
153                if state.history {
154                    let region = region.as_ref().map_or_else(|| "\"\"".into(), Region::history);
155                    let factors = factors.iter().map(usize::to_string).collect::<Vec<_>>().join(",");
156                    output.record_history("rebin", format!("ia.rebin(outfile={outfile:?}, bin=[{factors}], region={region}, mask={mask:?}, dropdeg={dropdeg}, overwrite={overwrite}, stretch={stretch}, crop={crop})"))?;
157                }
158                if path.is_some() {
159                    live_coordinates = AttachedImage::capture_live_coordinates(&output)?;
160                }
161                output.save().map_err(raised)?;
162                attachment.release_requested_lock(&mut handle)?;
163                Ok(output)
164            })?;
165            if let Some((record, system)) = live_coordinates {
166                product.keep_live_coordinates(record, system);
167            }
168            Ok::<_, PyErr>(product)
169        })?;
170        Py::new(py, Self::attached(product))
171    }
