2//! Host resources, the run's resource policy and the admission of each
3//! phase's memory against them (plan section 5.4).
4//!
5//! The host is detected once per process. A policy turns it into the
6//! workers and memory a run may use; each phase then admits the memory its
7//! owners say it holds ([`admit`]) and keeps it until the returned
8//! [`Reservation`] drops. Reservations of every run in the process count
9//! against each run's policy.
10
11#[cfg(target_os = "macos")]
12use std::process::Command;
13use std::sync::atomic::{AtomicU64, Ordering};
14use std::sync::{Mutex, MutexGuard, OnceLock, PoisonError};
15
16/// The host's capacity, detected once per process by
17/// [`HostResources::detect`]; tests construct it with pinned values.
18#[derive(Clone, Copy, Debug, PartialEq, Eq)]
19pub struct HostResources {
20    /// Logical CPU threads.
21    pub threads: usize,
22    /// Performance-class cores; `threads` on a host that reports no core
23    /// classes.
24    pub performance_cores: usize,
25    /// Bytes of memory free for new allocations when the host was detected:
26    /// free, inactive and speculative pages on macOS, `MemAvailable` on
27    /// Linux.
28    pub available_memory: u64,
29    /// Whether a unified-memory Metal 3 device is present.
30    pub metal: bool,
31}
32
33/// Why the host could not be detected.
34#[derive(Clone, Debug, thiserror::Error)]
35pub enum HostError {
36    /// A system query failed or reported something unreadable.
37    #[error("host detection: {0}")]
38    Detection(String),
39}
40
41impl HostResources {
42    /// The host, detected on the first call and the same thereafter.
43    ///
44    /// # Errors
45    ///
46    /// Fails when the thread count or the free memory cannot be read.
47    pub fn detect() -> Result<Self, HostError> {
48        static HOST: OnceLock<Result<HostResources, HostError>> = OnceLock::new();
49        HOST.get_or_init(|| {
50            let threads = std::thread::available_parallelism()
51                .map_err(|error| HostError::Detection(error.to_string()))?
52                .get();
53            Ok(Self {
54                threads,
55                performance_cores: performance_cores().map_or(threads, |cores| {
56                    usize::try_from(cores).unwrap_or(threads).clamp(1, threads)
57                }),
58                available_memory: available_memory()?,
59                metal: casa_imaging_metal::available(),
60            })
61        })
62        .clone()
63    }
64}

66/// How much of the host one run may use.
67#[derive(Clone, Copy, Debug, PartialEq, Eq)]
68pub enum ResourcePolicy {
69    /// Half the threads, rounded up, and half the free memory.
70    Interactive,
71    /// Three quarters of the threads, rounded up and at most the
72    /// performance cores, and three quarters of the free memory.
73    Balanced,
74    /// Every thread and all the free memory.
75    Exclusive,
76    /// At most `workers` threads and `memory` bytes, each capped at what the
77    /// host has.
78    Explicit {
79        /// Worker ceiling.
80        workers: usize,
81        /// Memory ceiling in bytes.
82        memory: u64,
83    },
84}
85
86impl ResourcePolicy {
87    /// Workers a run may use on `host`; at least one.
88    #[must_use]
89    pub fn workers(&self, host: &HostResources) -> usize {
90        let threads = host.threads;
91        match *self {
92            Self::Interactive => threads.div_ceil(2),
93            Self::Balanced => (threads * 3).div_ceil(4).min(host.performance_cores),
94            Self::Exclusive => threads,
95            Self::Explicit { workers, .. } => workers.min(threads),
96        }
97        .max(1)
98    }
99
100    /// Bytes of memory a run may use on `host`, before reservations.
101    #[must_use]
102    pub fn memory(&self, host: &HostResources) -> u64 {
103        let free = u128::from(host.available_memory);
104        let scaled = |numerator: u128, denominator: u128| {
105            u64::try_from(free * numerator / denominator).expect("a fraction of a u64")
106        };
107        match *self {
108            Self::Interactive => scaled(1, 2),
109            Self::Balanced => scaled(3, 4),
110            Self::Exclusive => host.available_memory,
111            Self::Explicit { memory, .. } => memory.min(host.available_memory),
112        }
113    }
114}

116/// The memory one phase holds while it runs, as its owners report it.
117#[derive(Clone, Copy, Debug, PartialEq, Eq)]
118pub struct Demand {
119    /// The phase, for the rejection message.
120    pub phase: &'static str,
121    /// Bytes held.
122    pub memory: u64,
123}
124
125/// A phase's demand did not fit what the policy leaves free.
126#[derive(Clone, Copy, Debug, PartialEq, Eq, thiserror::Error)]
127#[error("{phase} needs {required} bytes but the resource policy leaves {available} free")]
128pub struct Admission {
129    /// The rejected phase.
130    pub phase: &'static str,
131    /// Bytes the phase needs.
132    pub required: u64,
133    /// Bytes the policy leaves free.
134    pub available: u64,
135}
136
137/// Memory admitted to one phase; it is released when this drops.
138#[must_use = "the memory is released when the reservation drops"]
139#[derive(Debug)]
140pub struct Reservation {
141    memory: u64,
142}
143
144impl Reservation {
145    /// A reservation holding nothing, for state not yet formed.
146    pub const fn none() -> Self {
147        Self { memory: 0 }
148    }
149
150    /// Bytes held.
151    #[must_use]
152    pub const fn memory(&self) -> u64 {
153        self.memory
154    }
155
156    /// Keep `memory` of the held bytes and release the rest: what a phase
157    /// hands on stays charged with that state when its scratch is released.
158    ///
159    /// # Panics
160    ///
161    /// When `memory` exceeds the bytes held.
162    pub fn retain(&mut self, memory: u64) {
163        assert!(
164            memory <= self.memory,
165            "a reservation retains at most what it holds"
166        );
167        RESERVED.fetch_sub(self.memory - memory, Ordering::AcqRel);
168        self.memory = memory;
169    }
170
171    /// Move `memory` of the held bytes into a reservation of their own, for
172    /// state that outlives the phase that admitted it.
173    ///
174    /// # Panics
175    ///
176    /// When `memory` exceeds the bytes held.
177    pub fn split(&mut self, memory: u64) -> Self {
178        assert!(
179            memory <= self.memory,
180            "a reservation splits off at most what it holds"
181        );
182        self.memory -= memory;
183        Self { memory }
184    }
185
186    /// Hold `other`'s bytes too, until this reservation drops.
187    pub fn join(&mut self, mut other: Self) {
188        self.memory += std::mem::take(&mut other.memory);
189    }
190}
191
192impl Drop for Reservation {
193    fn drop(&mut self) {
194        RESERVED.fetch_sub(self.memory, Ordering::AcqRel);
195    }
196}

198/// Bytes held by the live reservations of this process. Only [`admit`]
199/// raises it, holding [`WINDOWS`]; releases lower it without the lock.
200static RESERVED: AtomicU64 = AtomicU64::new(0);
201
202/// The open [`ReservationWindow`]s, each with the most [`RESERVED`] has
203/// held since it last closed.
204static WINDOWS: Mutex<Windows> = Mutex::new(Windows {
205    next: 0,
206    open: Vec::new(),
207});
208
209struct Windows {
210    next: u64,
211    open: Vec<(u64, u64)>,
212}
213
214fn windows() -> MutexGuard<'static, Windows> {
215    WINDOWS.lock().unwrap_or_else(PoisonError::into_inner)
216}
217
218/// One observer's view of the process's reservations: the most they held
219/// at once over each interval it closes. Every run summary has its own, so
220/// runs sharing a process neither reset nor take each other's intervals.
221/// Admission raises every open window under the lock that a window closes
222/// under, so an admission falls wholly within one interval.
223#[derive(Debug)]
224pub struct ReservationWindow {
225    id: u64,
226}
227
228impl ReservationWindow {
229    /// A window whose first interval starts at what the reservations hold
230    /// now.
231    #[must_use]
232    pub fn open() -> Self {
233        let mut windows = windows();
234        let id = windows.next;
235        windows.next += 1;
236        windows.open.push((id, RESERVED.load(Ordering::Acquire)));
237        Self { id }
238    }
239
240    /// The most the reservations held at once in the interval this call
241    /// closes; the next starts at what they hold now, so a reservation that
242    /// outlives an interval counts in the next too.
243    pub fn close(&self) -> u64 {
244        let mut windows = windows();
245        let now = RESERVED.load(Ordering::Acquire);
246        let (_, peak) = windows
247            .open
248            .iter_mut()
249            .find(|(id, _)| *id == self.id)
250            .expect("a window is open until it drops");
251        std::mem::replace(peak, now).max(now)
252    }
253}
254
255impl Drop for ReservationWindow {
256    fn drop(&mut self) {
257        windows().open.retain(|(id, _)| *id != self.id);
258    }
259}

261/// Bytes `policy` leaves free on `host` beside the live reservations.
262#[must_use]
263pub fn free_memory(host: &HostResources, policy: &ResourcePolicy) -> u64 {
264    policy
265        .memory(host)
266        .saturating_sub(RESERVED.load(Ordering::Acquire))
267}
268
269/// Bytes the live reservations of this process hold.
270#[must_use]
271pub fn reserved_memory() -> u64 {
272    RESERVED.load(Ordering::Acquire)
273}
274
275/// Admit `demand` if it fits what `policy` leaves free on `host`, and raise
276/// every open [`ReservationWindow`] to what the process then holds.
277///
278/// # Errors
279///
280/// [`Admission`] when the demand exceeds the free memory; nothing is held.
281pub fn admit(
282    host: &HostResources,
283    policy: &ResourcePolicy,
284    demand: &Demand,
285) -> Result<Reservation, Admission> {
286    let capacity = policy.memory(host);
287    let mut windows = windows();
288    // Only admissions raise the total, and they hold the lock, so releases
289    // between this read and the add only leave more free.
290    let available = capacity.saturating_sub(RESERVED.load(Ordering::Acquire));
291    if demand.memory > available {
292        return Err(Admission {
293            phase: demand.phase,
294            required: demand.memory,
295            available,
296        });
297    }
298    let held = RESERVED.fetch_add(demand.memory, Ordering::AcqRel) + demand.memory;
299    for (_, peak) in &mut windows.open {
300        *peak = (*peak).max(held);
301    }
302    Ok(Reservation {
303        memory: demand.memory,
304    })
305}

317#[cfg(target_os = "macos")]
318fn available_memory() -> Result<u64, HostError> {
319    let output = Command::new("/usr/bin/vm_stat")
320        .output()
321        .map_err(|error| HostError::Detection(error.to_string()))?;
322    if !output.status.success() {
323        return Err(HostError::Detection("vm_stat failed".to_string()));
324    }
325    let text = String::from_utf8(output.stdout)
326        .map_err(|error| HostError::Detection(error.to_string()))?;
327    let page_size = text
328        .lines()
329        .next()
330        .and_then(|line| line.split("page size of ").nth(1))
331        .and_then(|value| value.split_ascii_whitespace().next())
332        .and_then(|value| value.parse::<u64>().ok())
333        .ok_or_else(|| HostError::Detection("vm_stat reports no page size".to_string()))?;
334    let pages = ["Pages free", "Pages inactive", "Pages speculative"]
335        .into_iter()
336        .map(|name| {
337            text.lines()
338                .find_map(|line| {
339                    let value = line.strip_prefix(name)?.strip_prefix(':')?;
340                    value.trim().trim_end_matches('.').parse::<u64>().ok()
341                })
342                .unwrap_or(0)
343        })
344        .sum::<u64>();
345    Ok(pages.saturating_mul(page_size))
346}

348#[cfg(target_os = "linux")]
349fn available_memory() -> Result<u64, HostError> {
350    let contents = std::fs::read_to_string("/proc/meminfo")
351        .map_err(|error| HostError::Detection(error.to_string()))?;
352    contents
353        .lines()
354        .find_map(|line| {
355            line.strip_prefix("MemAvailable:")?
356                .split_ascii_whitespace()
357                .next()?
358                .parse::<u64>()
359                .ok()
360        })
361        .map(|kib| kib.saturating_mul(1024))
362        .ok_or_else(|| HostError::Detection("/proc/meminfo has no MemAvailable".to_string()))
363}
364
365#[cfg(not(any(target_os = "linux", target_os = "macos")))]
366fn available_memory() -> Result<u64, HostError> {
367    Err(HostError::Detection(
368        "free memory is unknown on this platform".to_string(),
369    ))
370}

372#[cfg(target_os = "macos")]
373fn sysctl_u64(name: &str) -> Result<u64, HostError> {
374    let output = Command::new("/usr/sbin/sysctl")
375        .args(["-n", name])
376        .output()
377        .map_err(|error| HostError::Detection(error.to_string()))?;
378    if !output.status.success() {
379        return Err(HostError::Detection(format!("sysctl {name} failed")));
380    }
381    String::from_utf8_lossy(&output.stdout)
382        .trim()
383        .parse()
384        .map_err(|_| HostError::Detection(format!("sysctl {name} is not a count")))
385}

388mod tests {
389    use super::*;
390
391    const HOST: HostResources = HostResources {
392        threads: 10,
393        performance_cores: 6,
394        available_memory: 1 << 30,
395        metal: false,

398    #[test]
399    fn policies_scale_the_host() {
400        let cases = [
401            (ResourcePolicy::Interactive, 5, 1 << 29),
402            (ResourcePolicy::Balanced, 6, 3 << 28),
403            (ResourcePolicy::Exclusive, 10, 1 << 30),
404            (
405                ResourcePolicy::Explicit {
406                    workers: 64,
407                    memory: u64::MAX,
408                },
409                10,
410                1 << 30,
411            ),
412            (
413                ResourcePolicy::Explicit {
414                    workers: 0,
415                    memory: 1 << 20,
416                },
417                1,
418                1 << 20,
419            ),
420        ];
421        for (policy, workers, memory) in cases {
422            assert_eq!(policy.workers(&HOST), workers, "{policy:?}");
423            assert_eq!(policy.memory(&HOST), memory, "{policy:?}");
424        }
425    }
426
427    #[test]
428    fn a_reservation_holds_its_memory_until_it_drops() {
429        let policy = ResourcePolicy::Explicit {
430            workers: 1,
431            memory: 1 << 20,
432        };
433        let held = admit(
434            &HOST,
435            &policy,
436            &Demand {
437                phase: "first",
438                memory: 3 << 18,
439            },
440        )
441        .expect("fits");
442        assert_eq!(held.memory(), 3 << 18);
443        let rejected = admit(
444            &HOST,
445            &policy,
446            &Demand {
447                phase: "second",
448                memory: 1 << 19,
449            },
450        )
451        .expect_err("only a quarter is free");
452        assert_eq!(
453            rejected,
454            Admission {
455                phase: "second",
456                required: 1 << 19,
457                available: 1 << 18,
458            }
459        );
460        drop(held);
461        assert_eq!(free_memory(&HOST, &policy), 1 << 20);
462    }
463
464    fn admit_bytes(memory: u64) -> Reservation {
465        let policy = ResourcePolicy::Explicit {
466            workers: 1,
467            memory: 1 << 30,
468        };
469        admit(
470            &HOST,
471            &policy,
472            &Demand {
473                phase: "transient",
474                memory,
475            },
476        )
477        .expect("fits")
478    }

480    // Other tests of this process may hold reservations too, so these laws
481    // bound windows from below.
482
483    #[test]
484    fn a_window_keeps_its_peak_whatever_other_windows_do() {
485        let first = ReservationWindow::open();
486        drop(admit_bytes(5 << 20));
487        // Another run begins and ends a phase, then finishes, before the
488        // first run's phase ends.
489        let second = ReservationWindow::open();
490        let _ = second.close();
491        drop(second);
492        assert!(first.close() >= 5 << 20);
493    }

495    #[test]
496    fn a_reservation_held_across_a_close_counts_in_the_next_interval() {
497        let window = ReservationWindow::open();
498        let held = admit_bytes(3 << 20);
499        assert!(window.close() >= 3 << 20);
500        assert!(window.close() >= 3 << 20, "still held");
501        drop(held);
502        assert!(window.close() >= 3 << 20, "held when the interval began");
503    }
