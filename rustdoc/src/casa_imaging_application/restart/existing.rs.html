2//! The PSF, sums of weights and residual an earlier run left, read in place
3//! of the major cycle that would form them (`calcpsf=False`,
4//! `calcres=False`).
5//!
6//! CASA leaves its images standing when it does not form them again
7//! (`task_tclean.py`; `SIImageStore` opens what exists, and
8//! `FTMachine::finalizeToSkyNew` keeps a `.sumwt` already filled). The
9//! images on disk are normalized: the PSF to its peak, the residual by its
10//! sum of weights. The normal state holds them unnormalized, as a pass
11//! grids them, and products and minor cycles normalize them again; so the
12//! reader returns what a pass would have returned: the PSF times its
13//! principal sum of weights (the PSF peak of a direction-independent
14//! gridder) and the residual times the sum of weights that normalized it.
15//! Normalizing them again gives back the images read, to the rounding of
16//! one `f32` product and quotient.
17//!
18//! Only direction-independent gridders (standard and W-projection, one
19//! facet) are read: their sensitivity is the sum of weights alone
20//! ([`super::check_inputs`] refuses the rest).
21

32/// The tile cache each image is read with.
33const CACHE_BYTES: usize = 64 * 1024;

35/// What a run reads of an earlier run's products at its prefix.
36#[derive(Clone, Debug)]
37pub(crate) struct ExistingProducts {
38    prefix: PathBuf,
39    family: Family,
40    /// `[nx, ny]`.
41    shape: [usize; 2],
42    polarizations: usize,
43    /// The largest tile of the images read, in bytes.
44    tile_bytes: u64,
45    /// The PSF and its sums of weights (`calcpsf=False`).
46    psf: bool,
47    /// The residual (`calcres=False` with one at the prefix).
48    residual: bool,
49    /// A complete PSF beam set read from the earlier image, channel-major.
50    psf_beams: Option<Arc<[RestoringBeam]>>,
51    /// The same beam grid in pixels for minor-cycle preparation.
52    psf_pixel_beams: Option<Arc<[RestoringBeam]>>,
53    /// The residual beam grid in pixels, used for automatic masking.
54    mask_beams: Option<Arc<[RestoringBeam]>>,
55}

57impl ExistingProducts {
58    /// What a run of `family` images of `shape` and `polarizations`, over
59    /// `channels` output channels, reads at `prefix`: the PSF and sums of
60    /// weights when `psf`, the residual when `residual`. The images' grids
61    /// were checked; their sums of weights are checked here to hold one
62    /// value per Stokes plane and channel.
63    pub(crate) fn new(
64        prefix: &Path,
65        family: Family,
66        (shape, polarizations, channels): ([usize; 2], usize, usize),
67        (psf, residual): (bool, bool),
68    ) -> Result<Self, PrepareError> {
69        let planes = match family {
70            Family::Single => channels,
71            Family::Taylor { .. } => 1,
72        };
73        let mut tile_bytes = 0;
74        let mut bases = Vec::new();
75        if psf {
76            bases.extend(["psf", "sumwt"]);
77        }
78        if residual {
79            bases.push("residual");
80        }
81        for base in bases {
82            for name in family.names(base) {
83                let path = product_path(prefix, &name);
84                let image = open(&path)?;
85                tile_bytes = image
86                    .tile_shape()
87                    .iter()
88                    .fold(4_u64, |n, &side| n.saturating_mul(side as u64))
89                    .max(tile_bytes);
90                if base == "sumwt" && image.shape() != [1, 1, polarizations, planes] {
91                    return Err(PrepareError::Restart {
92                        path,
93                        reason: format!(
94                            "its sums of weights have shape {:?}, not one per Stokes plane and \
95                             channel [1, 1, {polarizations}, {planes}]",
96                            image.shape()
97                        ),
98                    });
99                }
100            }
101        }
102        Ok(Self {
103            prefix: prefix.to_path_buf(),
104            family,
105            shape,
106            polarizations,
107            tile_bytes,
108            psf,
109            residual,
110            psf_beams: None,
111            psf_pixel_beams: None,
112            mask_beams: None,
113        })
114    }

116    /// Read the earlier PSF's complete beam grid, as `makeImageBeamSet`
117    /// reuses it before looking at the new run's cutoff.
118    pub(crate) fn read_psf_beams(
119        &mut self,
120        channels: usize,
121        cell_rad: f64,
122    ) -> Result<(), PrepareError> {
123        let name = self
124            .family
125            .names("psf")
126            .into_iter()
127            .next()
128            .expect("a PSF name");
129        let path = product_path(&self.prefix, &name);
130        let image = open(&path)?;
131        let info = image.image_info().map_err(|error| PrepareError::Restart {
132            path: path.clone(),
133            reason: error.to_string(),
134        })?;
135        let planes = match self.family {
136            Family::Single => channels,
137            Family::Taylor { .. } => 1,
138        };
139        self.psf_beams =
140            complete_psf_beams(&info.beam_set, planes, self.polarizations).map_err(|error| {
141                PrepareError::Restart {
142                    path: path.clone(),
143                    reason: error.to_string(),
144                }
145            })?;
146        if let Some(beams) = &self.psf_beams {
147            let pixels = beams
148                .iter()
149                .map(|beam| {
150                    RestoringBeam::new(
151                        beam.major_fwhm_rad() / cell_rad,
152                        beam.minor_fwhm_rad() / cell_rad,
153                        beam.position_angle_rad(),
154                    )
155                })
156                .collect::<Result<Vec<_>, _>>()
157                .map_err(|error| PrepareError::Restart {
158                    path,
159                    reason: error.to_string(),
160                })?;
161            self.psf_pixel_beams = Some(pixels.into());
162        }
163        Ok(())
164    }

209    /// Retained heap of the shared beam grid, including the `Arc` counters.
210    pub(crate) fn beam_resident_bytes(&self) -> u64 {
211        [&self.psf_beams, &self.psf_pixel_beams, &self.mask_beams]
212            .into_iter()
213            .flatten()
214            .map(|beams| (std::mem::size_of_val(beams.as_ref()) + 2 * size_of::<usize>()) as u64)
215            .sum()
216    }

218    /// Bound an image beam header and beam conversion read together.
219    /// Each beam is three quantity records (one field and two nested fields
220    /// each), plus the channel and Stokes records. The table keyword tree,
221    /// the decoded beam grid, and the converted grid may coexist.
222    pub(crate) fn beam_reading_bytes(channels: usize, polarizations: usize) -> u64 {
223        let planes = channels.saturating_mul(polarizations);
224        let fields = 3 * 3 + 2;
225        let record = fields * (size_of::<casa_types::RecordField>() + 3 * size_of::<String>());
226        let per_plane = 3 * record
227            + size_of::<casa_images::GaussianBeam>()
228            + 4 * size_of::<RestoringBeam>()
229            + size_of::<Vec<casa_images::GaussianBeam>>();
230        (planes.saturating_mul(per_plane) + 4 * CACHE_BYTES + 2 * size_of::<usize>()) as u64
231    }

238    /// The `(terms, moments)` of each channel: one of each, or a Taylor
239    /// family's coefficients and PSF moments.
240    const fn terms(&self) -> (usize, usize) {
241        match self.family {
242            Family::Single => (1, 1),
243            Family::Taylor { terms } => (terms, 2 * terms - 1),
244        }
245    }

247    /// Bytes reading `channels` channels holds besides the pass images it
248    /// returns ([`Self::pass_images`], [`Self::images_bytes`]): one image's
249    /// slice of those channels in `f32`, the tile cache and two tile
250    /// buffers.
251    pub(crate) fn reading_bytes(&self, channels: usize) -> u64 {
252        let planes = match self.family {
253            Family::Single => channels,
254            Family::Taylor { .. } => 1,
255        };
256        (self.shape[0] * self.shape[1] * self.polarizations * planes * 4) as u64
257            + 2 * self.tile_bytes
258            + CACHE_BYTES as u64
259    }

261    /// Bytes the pass images of `channels` channels hold: the residual
262    /// terms and PSF moments in `f32` and two sums of weights per moment.
263    pub(crate) fn images_bytes(&self, channels: usize) -> u64 {
264        let (terms, moments) = self.terms();
265        let planes = channels * self.polarizations;
266        let cells = self.shape[0] * self.shape[1];
267        (planes * (cells * (terms + moments) * 4 + moments * 2 * 8)) as u64
268    }

270    /// The images of domain 0 a pass forming its PSF and residual would
271    /// return for output channels `channels`, from the PSF, sums of weights
272    /// and residual read: the residual is zero when it is not read, for a
273    /// pass to form.
274    pub(crate) fn pass_images(&self, channels: Range<usize>) -> Result<PassImages, ImageError> {
275        assert!(self.psf, "the PSF is read");
276        let (_, moments) = self.terms();
277        let (pols, cells) = (self.polarizations, self.shape[0] * self.shape[1]);
278        let mut sums = vec![0.0; channels.len() * moments * pols];
279        for (moment, name) in self.family.names("sumwt").iter().enumerate() {
280            let read = self.read(name, channels.clone())?;
281            for channel in 0..channels.len() {
282                for pol in 0..pols {
283                    sums[(channel * moments + moment) * pols + pol] =
284                        f64::from(read[&[0, 0, pol, self.plane(channel)][..]]);
285                }
286            }
287        }
288        // A pass's PSF peaks at its principal sum of weights; every moment
289        // shares that one divisor (`SIImageStore::normPSF`).
290        let principal = |channel: usize, pol: usize| sums[channel * moments * pols + pol];
291        let mut psf = vec![0.0; channels.len() * moments * pols * cells];
292        for (moment, name) in self.family.names("psf").iter().enumerate() {
293            let read = self.read(name, channels.clone())?;
294            for channel in 0..channels.len() {
295                for pol in 0..pols {
296                    let plane = (channel * moments + moment) * pols + pol;
297                    self.write_plane(
298                        &mut psf[plane * cells..(plane + 1) * cells],
299                        &read,
300                        (channel, pol),
301                        principal(channel, pol),
302                    );
303                }
304            }
305        }
306        let mut images = PassImages {
307            domain: 0,
308            shape: self.shape,
309            channels,
310            polarizations: pols,
311            residual: Vec::new(),
312            psf: Some(psf),
313            sum_weights: sums.clone(),
314            published_sum_weights: sums,
315            weight: None,
316        };
317        self.fill_residual(&mut images)?;
318        Ok(images)
319    }

321    /// Set the residual of `images`, a pass's images of domain 0 with
322    /// their published sums of weights and no residual (a pass that formed
323    /// the PSF alone), to the residual read times the sum of weights that
324    /// normalizes it: each plane's own, or a Taylor family's principal one
325    /// for every term (`SIImageStoreMultiTerm::divideResidualByWeight`).
326    /// Without a residual read it is zero.
327    pub(crate) fn fill_residual(&self, images: &mut PassImages) -> Result<(), ImageError> {
328        let (_, moments) = self.terms();
329        let pols = images.polarizations;
330        let published = &images.published_sum_weights;
331        images.residual = self.residual(images.channels.clone(), |channel, pol| {
332            published[channel * moments * pols + pol]
333        })?;
334        Ok(())
335    }

337    /// The images of domain 0 that refresh a normal state's residual with
338    /// the residual read ([`Self::fill_residual`]); `principal` is the
339    /// state's published sum of weights of window channel `(channel, pol)`.
340    pub(crate) fn residual_images(
341        &self,
342        channels: Range<usize>,
343        principal: impl Fn(usize, usize) -> f64,
344    ) -> Result<PassImages, ImageError> {
345        Ok(PassImages {
346            domain: 0,
347            shape: self.shape,
348            channels: channels.clone(),
349            polarizations: self.polarizations,
350            residual: self.residual(channels, principal)?,
351            psf: None,
352            sum_weights: Vec::new(),
353            published_sum_weights: Vec::new(),
354            weight: None,
355        })
356    }

358    /// The residual planes of `channels`, `[channel][term][pol]` x-major,
359    /// read and times `principal(channel, pol)`, which each polarization's
360    /// Taylor family takes at its principal moment, `(0, pol)`, for every
361    /// term; zero without a residual read.
362    fn residual(
363        &self,
364        channels: Range<usize>,
365        principal: impl Fn(usize, usize) -> f64,
366    ) -> Result<Vec<f32>, ImageError> {
367        let (terms, _) = self.terms();
368        let pols = self.polarizations;
369        let cells = self.shape[0] * self.shape[1];
370        let mut residual = vec![0.0; channels.len() * terms * pols * cells];
371        if !self.residual {
372            return Ok(residual);
373        }
374        let scale = |channel: usize, pol: usize| match self.family {
375            Family::Single => principal(channel, pol),
376            Family::Taylor { .. } => principal(0, pol),
377        };
378        for (term, name) in self.family.names("residual").iter().enumerate() {
379            let read = self.read(name, channels.clone())?;
380            for channel in 0..channels.len() {
381                for pol in 0..pols {
382                    let plane = (channel * terms + term) * pols + pol;
383                    self.write_plane(
384                        &mut residual[plane * cells..(plane + 1) * cells],
385                        &read,
386                        (channel, pol),
387                        scale(channel, pol),
388                    );
389                }
390            }
391        }
392        Ok(residual)
393    }

395    /// The image plane of window channel `channel`: the channel itself, or
396    /// a Taylor image's one plane.
397    const fn plane(&self, channel: usize) -> usize {
398        match self.family {
399            Family::Single => channel,
400            Family::Taylor { .. } => 0,
401        }
402    }

404    /// Write `read`'s plane `(channel, pol)` times `scale` to `out`,
405    /// x-major (y fastest) as the normal state holds it.
406    fn write_plane(
407        &self,
408        out: &mut [f32],
409        read: &ndarray::ArrayD<f32>,
410        (channel, pol): (usize, usize),
411        scale: f64,
412    ) {
413        let plane = self.plane(channel);
414        let ny = self.shape[1];
415        for (index, value) in out.iter_mut().enumerate() {
416            let (x, y) = (index / ny, index % ny);
417            *value = (f64::from(read[&[x, y, pol, plane][..]]) * scale) as f32;
418        }
419    }

421    /// The pixels of `name` over `channels`: their slice of a single image,
422    /// or a Taylor image's one plane.
423    fn read(&self, name: &str, channels: Range<usize>) -> Result<ndarray::ArrayD<f32>, ImageError> {
424        let path = product_path(&self.prefix, name);
425        let image = PagedImage::<f32>::open_with_cache(&path, CACHE_BYTES)?;
426        let shape = image.shape().to_vec();
427        let (start, count) = match self.family {
428            Family::Single => (channels.start, channels.len()),
429            Family::Taylor { .. } => (0, 1),
430        };
431        image.get_slice(&[0, 0, 0, start], &[shape[0], shape[1], shape[2], count])
432    }
433}

435/// CASA reuses the PSF header only when its complete beam matrix has the
436/// image's channel and polarization shape (`makeImageBeamSet`).
437fn complete_psf_beams(
438    set: &ImageBeamSet,
439    channels: usize,
440    polarizations: usize,
441) -> Result<Option<Arc<[RestoringBeam]>>, casa_imaging_deconvolution::PsfBeamFitError> {
442    if set.shape() != (channels, polarizations) {
443        return Ok(None);
444    }
445    let mut beams = Vec::with_capacity(channels * polarizations);
446    for channel in 0..channels {
447        for polarization in 0..polarizations {
448            let beam = set.beam(channel, polarization);
449            beams.push(RestoringBeam::new(
450                beam.major,
451                beam.minor,
452                beam.position_angle,
453            )?);
454        }
455    }
456    Ok(Some(beams.into()))
457}

459fn residual_pixel_beams(
460    set: &ImageBeamSet,
461    channels: usize,
462    polarizations: usize,
463    cell_rad: f64,
464) -> Result<Option<Arc<[RestoringBeam]>>, String> {
465    if set.is_empty() {
466        return Ok(None);
467    }
468    let (nc, np) = set.shape();
469    if (nc != 1 && nc != channels) || (np != 1 && np != polarizations) {
470        return Err(format!(
471            "residual beam grid has shape ({nc}, {np}), expected ({channels}, {polarizations}) or an axis of length one"
472        ));
473    }
474    let mut beams = Vec::with_capacity(channels * polarizations);
475    for channel in 0..channels {
476        for polarization in 0..polarizations {
477            let beam = set.beam(channel, polarization);
478            beams.push(
479                RestoringBeam::new(
480                    beam.major / cell_rad,
481                    beam.minor / cell_rad,
482                    beam.position_angle,
483                )
484                .map_err(|error| error.to_string())?,
485            );
486        }
487    }
488    Ok(Some(beams.into()))
489}

491fn open(path: &Path) -> Result<PagedImage<f32>, PrepareError> {
492    PagedImage::<f32>::open_with_cache(path, CACHE_BYTES).map_err(|error| PrepareError::Restart {
493        path: path.to_path_buf(),
494        reason: error.to_string(),
495    })
496}

511    /// CASA accepts only a complete PSF beam grid, in channel/polarization
512    /// order; a missing or differently shaped header is refitted.
513    #[test]
514    fn a_restart_reuses_only_a_complete_psf_beam_grid() {
515        let set =
516            ImageBeamSet::from_grid(vec![vec![beam(1.0), beam(2.0)], vec![beam(3.0), beam(4.0)]]);
517        let result = complete_psf_beams(&set, 2, 2).unwrap().unwrap();
518        assert_eq!(
519            result
520                .iter()
521                .map(|b| b.major_fwhm_rad())
522                .collect::<Vec<_>>(),
523            [1.0, 2.0, 3.0, 4.0]
524        );
525        assert!(complete_psf_beams(&set, 4, 1).unwrap().is_none());
526        assert!(
527            complete_psf_beams(&ImageBeamSet::empty(), 2, 2)
528                .unwrap()
529                .is_none()
530        );
531    }
532    #[test]
533    fn residual_beams_expand_single_axes_in_channel_major_order() {
534        let set = ImageBeamSet::from_grid(vec![vec![beam(2.0), beam(4.0)]]);
535        let pixels = residual_pixel_beams(&set, 3, 2, 2.0).unwrap().unwrap();
536        assert_eq!(
537            pixels
538                .iter()
539                .map(|b| b.major_fwhm_rad())
540                .collect::<Vec<_>>(),
541            [1.0, 2.0, 1.0, 2.0, 1.0, 2.0]
542        );
543        let grid =
544            ImageBeamSet::from_grid(vec![vec![beam(2.0), beam(4.0)], vec![beam(6.0), beam(8.0)]]);
545        let pixels = residual_pixel_beams(&grid, 2, 2, 2.0).unwrap().unwrap();
546        assert_eq!(
547            pixels
548                .iter()
549                .map(|b| b.major_fwhm_rad())
550                .collect::<Vec<_>>(),
551            [1.0, 2.0, 3.0, 4.0]

555            residual_pixel_beams(&ImageBeamSet::empty(), 3, 2, 2.0)
