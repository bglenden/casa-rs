13fn i32_vec_value(values: &[impl Copy + TryInto<i32>]) -> Result<Value, StorageError> {
14    let mut converted = Vec::with_capacity(values.len());
15    for &value in values {
16        converted.push(value.try_into().map_err(|_| {
17            StorageError::FormatMismatch("tiled spec dimension exceeds i32".into())
18        })?);
19    }
20    let len = converted.len();
21    ArrayD::from_shape_vec(IxDyn(&[len]), converted)
22        .map(ArrayValue::Int32)
23        .map(Value::Array)
24        .map_err(|error| StorageError::FormatMismatch(error.to_string()))
25}

27/// The `SPEC` record of a `TiledStMan` manager (`TiledStMan::dataManagerSpec`).
28pub(crate) fn tiled_spec(
29    table_path: &std::path::Path,
30    dm_seq_nr: u32,
31    table_dat: &TableDatContents,
32) -> Result<RecordValue, StorageError> {
33    let header_path = table_path.join(format!(
34        "{}{dm_seq_nr}",
35        super::super::TABLE_DATA_FILE_PREFIX
36    ));
37    let (variant, header) = read_tiled_header(&header_path)?;
38    let bound_cols: Vec<&ColumnDescContents> = table_dat
39        .column_set
40        .columns
41        .iter()
42        .enumerate()
43        .filter(|(_, pc)| pc.dm_seq_nr == dm_seq_nr)
44        .filter_map(|(index, _)| table_dat.table_desc.columns.get(index))
45        .collect();
46    let tiles = tile_columns(
47        &variant,
48        Cow::Owned(header),
49        &bound_cols,
50        &table_dat.table_desc,
51        0,
52    )?;
53    let header = &*tiles.header;
54    let first_data_col = bound_cols
55        .iter()
56        .zip(&tiles.tile_index)
57        .find_map(|(col, index)| index.map(|_| *col));
58
59    let mut hypercubes = Vec::with_capacity(header.cubes.len());
60    for (index, cube) in header
61        .cubes
62        .iter()
63        .filter(|cube| !cube.cube_shape.is_empty())
64        .enumerate()
65    {
66        let bucket_size = compute_tile_layout(&header.col_data_types, &cube.tile_shape).0;
67        let bucket_size = i32::try_from(bucket_size)
68            .map_err(|_| StorageError::FormatMismatch("tiled bucket size exceeds i32".into()))?;
69        let cell_ndim = match first_data_col {
70            Some(col) if !col.is_array => 0,
71            Some(col) if col.nrdim > 0 => col.nrdim as usize,
72            _ if matches!(variant, TiledVariant::Cell { .. }) => cube.cube_shape.len(),
73            _ => cube.cube_shape.len().saturating_sub(1),
74        };
75        let cell_shape = &cube.cube_shape[..cell_ndim.min(cube.cube_shape.len())];
76        hypercubes.push(RecordField::new(
77            format!("*{}", index + 1),
78            Value::Record(RecordValue::new(vec![
79                RecordField::new("BucketSize", Value::Scalar(ScalarValue::Int32(bucket_size))),
80                RecordField::new("CellShape", i32_vec_value(cell_shape)?),
81                RecordField::new("CubeShape", i32_vec_value(&cube.cube_shape)?),
82                RecordField::new("ID", Value::Record(cube.values.clone())),
83                RecordField::new("TileShape", i32_vec_value(&cube.tile_shape)?),
84            ])),
85        ));
86    }
87
88    let max_cache_size = i32::try_from(header.max_cache_size)
89        .map_err(|_| StorageError::FormatMismatch("tiled cache size exceeds i32".into()))?;
90    let seq_nr = i32::try_from(dm_seq_nr)
91        .map_err(|_| StorageError::FormatMismatch("tiled manager seq_nr exceeds i32".into()))?;
92    let mut fields = vec![
93        RecordField::new(
94            "DEFAULTTILESHAPE",
95            i32_vec_value(variant.default_tile_shape())?,
96        ),
97        RecordField::new("HYPERCUBES", Value::Record(RecordValue::new(hypercubes))),
98    ];
99    if let TiledVariant::Shape {
100        nr_used_row_map, ..
101    } = variant
102    {
103        let index_size = i32::try_from(nr_used_row_map)
104            .map_err(|_| StorageError::FormatMismatch("tiled index size exceeds i32".into()))?;
105        fields.push(RecordField::new(
106            "IndexSize",
107            Value::Scalar(ScalarValue::Int32(index_size)),
108        ));
109    }
110    fields.push(RecordField::new(
111        "MAXIMUMCACHESIZE",
112        Value::Scalar(ScalarValue::Int32(max_cache_size)),
113    ));
114    fields.push(RecordField::new(
115        "MaxCacheSize",
116        Value::Scalar(ScalarValue::Int32(max_cache_size)),
117    ));
118    fields.push(RecordField::new(
119        "SEQNR",
120        Value::Scalar(ScalarValue::Int32(seq_nr)),
121    ));
122    Ok(RecordValue::new(fields))
123}
