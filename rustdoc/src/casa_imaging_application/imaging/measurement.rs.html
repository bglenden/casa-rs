24/// The operator of one image domain and the resampler that places rows on
25/// its planes.
26pub(crate) struct DomainOperator {
27    pub(crate) operator: MeasurementOperator,
28    pub(crate) resampler: SpectralResampler,
29    /// Whether the kernel set grids a sensitivity (weight) image
30    /// (`ConvolutionFunctionSet::weight_taps`): mosaic and AW.
31    pub(crate) weight_image: bool,
32    /// The charge of the kernel set, held while the operator lives.
33    kernels: Kernels,
34}
35
36/// The charge of a kernel set: admitted before the set was built or, for
37/// an AW catalog whose holders are the passes' workers, admitted once the
38/// run has chosen its workers (#737).
39enum Kernels {
40    Admitted {
41        _charge: Reservation,
42    },
43    /// An AW catalog's charge at its cache's bound, for at most `ceiling`
44    /// holders. An `automatic` bound (`cf_resident_mb = auto`) rises, as the
45    /// charge is admitted, into the memory the chosen workers leave (#449).
46    Catalog {
47        charge: CacheCharge,
48        bound: CacheBound,
49        ceiling: usize,
50        automatic: bool,
51    },
52}
53
54impl DomainOperator {
55    /// Bytes the kernel set still to be admitted holds for passes on
56    /// `workers` streaming workers, at most the ceiling it was opened for,
57    /// at its cache's bound now; zero for a set already admitted.
58    pub(crate) fn pending_kernel_bytes(&self, workers: usize) -> u64 {
59        match &self.kernels {
60            Kernels::Admitted { .. } => 0,
61            Kernels::Catalog {
62                charge,
63                bound,
64                ceiling,
65                ..
66            } => charge.bytes(workers.clamp(1, *ceiling), bound.bytes()),
67        }
68    }
69
70    /// Admit the kernel set's charge for passes on `workers` streaming
71    /// workers, if it is still to be admitted. An automatic cell cache first
72    /// takes what it can of `spare`, the bytes the run's chosen workers
73    /// leave free: every cell when they fit, otherwise as much more bound as
74    /// `spare` allows. What the raise costs comes off `spare`, and what it
75    /// saves goes back for the next catalog ([`CacheCharge::raise_within`]).
76    pub(crate) fn admit_kernels(
77        &mut self,
78        admission: KernelAdmission<'_>,
79        workers: usize,
80        spare: &mut u64,
81    ) -> Result<(), ImagingError> {
82        if let Kernels::Catalog {
83            charge,
84            bound,
85            ceiling,
86            automatic,
87        } = &self.kernels
88        {
89            let holders = workers.clamp(1, *ceiling);
90            if *automatic {
91                charge.raise_within(holders, bound, spare);
92                tracing::info!(
93                    "AW cell cache: {:.0} MB of {:.0} MB of cells resident at most (auto)",
94                    bound.bytes().min(charge.every_bytes() as usize) as f64 / 1e6,
95                    charge.every_bytes() as f64 / 1e6
96                );
97            }
98            let memory = charge.bytes(holders, bound.bytes());
99            self.kernels = Kernels::Admitted {
100                _charge: admission.admit("convolution functions", memory)?,
101            };
102        }
103        Ok(())
104    }
105}

107/// Correlation types every block delivers, in block order: the first
108/// source's selection, which every source must share.
109pub(crate) fn selected_correlations(
110    problem: &CompiledProblem,
111) -> Result<Vec<Correlation>, ImagingError> {
112    let mut layouts = problem
113        .observation_transaction()
114        .read_set()
115        .sources()
116        .iter()
117        .flat_map(|source| source.selection().correlations())
118        .map(|selection| {
119            selection
120                .products()
121                .iter()
122                .map(|product| product.correlation_type())
123                .collect::<Vec<_>>()
124        });
125    let first = layouts.next().ok_or(ImagingError::Unsupported {
126        reason: "the selection names no correlations",
127    })?;
128    if layouts.any(|layout| layout != first) {
129        return Err(ImagingError::Unsupported {
130            reason: "selected polarization setups differ in their correlations",
131        });
132    }
133    Ok(first)
134}

136/// The most samples a row of `problem`'s source places and predicts on any
137/// of its image domains, which share one resampler
138/// ([`SpectralResampler::samples_per_row`],
139/// [`SpectralResampler::prediction_samples_per_row`]).
140pub(crate) fn row_samples(problem: &CompiledProblem) -> Result<RowSamples, ImagingError> {
141    let resampler = resampler(problem, basis(problem)?)?;
142    let spectrum = row_layout(problem).spectrum;
143    Ok(RowSamples {
144        placed: resampler.samples_per_row(spectrum),
145        predicted: resampler.prediction_samples_per_row(spectrum.channels),
146    })
147}

149/// The grid basis of the compiled reconstruction basis. A Taylor basis
150/// expands about the image's reference frequency.
151pub(crate) fn basis(problem: &CompiledProblem) -> Result<Basis, ImagingError> {
152    Ok(match problem.reconstruction().basis() {
153        ReconstructionBasis::Constant => Basis::Constant,
154        ReconstructionBasis::Taylor { terms } => Basis::Taylor {
155            terms: terms as u32,
156            reference_hz: reference_frequency_hz(problem)?,
157        },
158        ReconstructionBasis::ChannelLocal { channels } => Basis::ChannelLocal {
159            planes: channels as u32,
160        },
161    })
162}

164/// The image's spectral reference frequency (CASA `reffreq`).
165pub(crate) fn reference_frequency_hz(problem: &CompiledProblem) -> Result<f64, ImagingError> {
166    match problem.geometry().spectral().wcs() {
167        SpectralWcs::Linear {
168            reference_frequency_hz,
169            ..
170        } => Ok(*reference_frequency_hz),
171        SpectralWcs::Tabular { .. } => Err(ImagingError::Unsupported {
172            reason: "a tabular spectral axis has no uniform channel width",
173        }),
174    }
175}

177/// The grid precision: the requested one, else plan decision D2's rule as
178/// the owner amended it on 2026-10-15 (`gridprecision = auto`): f64 on the
179/// CPU for every basis, since CASA accumulates its grids in double
180/// (`useDoublePrec`, `SynthesisUtilMethods.cc:3722`), and f32 on Metal,
181/// which accumulates in `f32` only.
182///
183/// A cube is no exception: an f32 running sum drifts with the number of
184/// samples a cell takes, and a mosaic's weight image takes every sample of
185/// a channel at the uv origin (#659).
186pub(crate) const fn precision(
187    backend: BackendChoice,
188    requested: Option<GridPrecision>,
189) -> GridPrecision {
190    if let Some(precision) = requested {
191        return precision;
192    }
193    match backend {
194        BackendChoice::Cpu => GridPrecision::F64,
195        BackendChoice::Metal => GridPrecision::F32,
196    }
197}

199/// Whether the problem grids with the standard kernel set, the one facets
200/// take.
201pub(crate) fn standard_kernel_set(problem: &CompiledProblem) -> bool {
202    kernel_set_kind(problem) == KernelSetKind::Standard
203}

205/// Whether the Metal backend grids the problem's kernel set: the standard
206/// spheroidal set, and the mosaic primary beams (#787). The mosaic's PSF
207/// and prediction cells are its imaging cells, and its weight cells share
208/// their support, so Metal's one tap route per mode serves it. Its weight
209/// image keeps f32 precision because the weight owner sums it by
210/// contribution first (#792). Each Metal backend of a pass copies the
211/// cells it meets to the device, which the wave demand admits per backend
212/// (`casa_imaging_metal::kernel_bytes`, #793 review). W-planes and the AW
213/// catalog run on the CPU until #653's rows for them.
214pub(crate) fn metal_kernel_set(problem: &CompiledProblem) -> bool {
215    matches!(
216        kernel_set_kind(problem),
217        KernelSetKind::Standard | KernelSetKind::Mosaic
218    )
219}

221/// Which kernel set the compiled problem's measurement equation names.
222#[derive(Clone, Copy, Debug, PartialEq, Eq)]
223enum KernelSetKind {
224    /// The standard spheroidal set (`GridFT`).
225    Standard,
226    /// W-projection planes (`WProjectFT`).
227    WPlanes,
228    /// The heterogeneous-array mosaic beams (`MosaicFT` with
229    /// `HetArrayConvFunc`).
230    Mosaic,
231    /// The AW catalog (`AWProjectFT`).
232    Aw,
233}
234
235fn kernel_set_kind(problem: &CompiledProblem) -> KernelSetKind {
236    let equation = problem.science().measurement_equation();
237    if equation.aw_projection().is_some() {
238        KernelSetKind::Aw
239    } else if equation.w_projection().is_some() {
240        KernelSetKind::WPlanes
241    } else if problem.science().instrument_model()
242        == Some(InstrumentModel::CasaAlmaAcaHeterogeneousInterferometricResponseV1)
243    {
244        KernelSetKind::Mosaic
245    } else {
246        KernelSetKind::Standard
247    }
248}

250/// The operator of a chart of `extent` (an image domain, or one of its
251/// facets) with the kernel set the problem names: the
252/// standard spheroidal set, W-projection planes sized by the W contract,
253/// the mosaic primary beams of the selected windows, or the AW catalog of
254/// `aw_catalog`, in the requested precision or the one `backend` grids at.
255/// The mosaic set takes one dish per selected aperture class in
256/// `dish_classes` (`HetArrayConvFunc::findAntennaSizes`), the order the
257/// rows' antenna types index. Mosaic and AW grid without padding (CASA
258/// `MosaicFT`, `AWProjectFT`); the others on CASA's composite-padded grid.
259/// The kernel set is admitted as [`kernel_set`] says, and its charge lives
260/// with the operator.
261#[allow(clippy::too_many_arguments)]
262pub(crate) fn domain_operator(
263    problem: &CompiledProblem,
264    extent: ImageExtent,
265    correlations: &[Correlation],
266    backend: BackendChoice,
267    requested_precision: Option<GridPrecision>,
268    aw_catalog: Option<&AwCatalogDeployment>,
269    dish_classes: &[AntennaResponseClass],
270    admission: KernelAdmission<'_>,
271) -> Result<DomainOperator, ImagingError> {
272    let kind = kernel_set_kind(problem);
273    // Each CASA facet's GridFT pads its own facet grid.
274    let padding = match kind {
275        KernelSetKind::Standard | KernelSetKind::WPlanes => {
276            GridPadding::CasaComposite(PaddingFactor::TCLEAN)
277        }
278        KernelSetKind::Mosaic | KernelSetKind::Aw => GridPadding::None,
279    };
280    let geometry = GridGeometry::new(extent, padding)?;
281    let polarization = PolarizationRouting::compile(
282        correlations,
283        problem.reconstruction().polarization().coordinates(),
284    )?;
285    // An AW cube is refused by the gate (`Unsupported::CubeAwProjection`).
286    let basis = basis(problem)?;
287    let resampler = resampler(problem, basis)?;
288    let (cf, kernels) = kernel_set(
289        problem,
290        kind,
291        (&geometry, &polarization),
292        aw_catalog,
293        dish_classes,
294        admission,
295    )?;
296    Ok(DomainOperator {
297        operator: MeasurementOperator::new(
298            geometry,
299            basis,
300            polarization,
301            cf,
302            precision(backend, requested_precision),
303        ),
304        resampler,
305        weight_image: matches!(kind, KernelSetKind::Mosaic | KernelSetKind::Aw),
306        kernels,
307    })
308}

310/// Where a run admits its kernel sets: its host and policy, and the most
311/// workers that fetch cells from a paged set, the policy's; the run admits
312/// a paged set's charge once it has chosen how many do
313/// ([`DomainOperator::admit_kernels`]).
314#[derive(Clone, Copy)]
315pub(crate) struct KernelAdmission<'a> {
316    pub(crate) host: &'a HostResources,
317    pub(crate) policy: &'a ResourcePolicy,
318    pub(crate) workers: usize,
319}
320
321impl KernelAdmission<'_> {
322    fn admit(self, phase: &'static str, memory: u64) -> Result<Reservation, ImagingError> {
323        Ok(admit(self.host, self.policy, &Demand { phase, memory })?)
324    }
325}

327/// The kernel set of `kind` on `geometry` routing `polarization`, with the
328/// charge of what it holds, admitted before it is built. W-projection and
329/// mosaic sets are built in two stages, their screens and then the kernels
330/// cut from them, each admitted before it allocates; the screens' charge
331/// is released once the kernels are built. An AW catalog's charge, its
332/// cache bound and what its workers can hold beyond it
333/// ([`AwCatalog::charge_bytes`]), is known once its index is read, for
334/// each count of workers up to `admission`'s, and is admitted when the run
335/// has chosen its workers.
336fn kernel_set(
337    problem: &CompiledProblem,
338    kind: KernelSetKind,
339    (geometry, polarization): (&GridGeometry, &PolarizationRouting),
340    aw_catalog: Option<&AwCatalogDeployment>,
341    dish_classes: &[AntennaResponseClass],
342    admission: KernelAdmission<'_>,
343) -> Result<(Box<dyn ConvolutionFunctionSet>, Kernels), ImagingError> {
344    const PHASE: &str = "convolution functions";
345    Ok(match kind {
346        KernelSetKind::Standard => {
347            let kernels = admission.admit(PHASE, Spheroidal::bytes(geometry))?;
348            (
349                Box::new(Spheroidal::new(geometry, polarization)),
350                Kernels::Admitted { _charge: kernels },
351            )
352        }
353        KernelSetKind::WPlanes => {
354            let contract = problem
355                .science()
356                .measurement_equation()
357                .w_projection()
358                .expect("the kind names a W contract");
359            let count = match (contract.planes(), contract.statistics()) {
360                (Some(planes), _) => WPlaneCount::Fixed(planes.get() as u32),
361                (None, Some(statistics)) => WPlaneCount::Auto {
362                    min_w: statistics.minimum_abs_w_lambda(),
363                    max_w: contract.maximum_abs_w_lambda(),
364                    rms_w: statistics.rms_w_lambda(),
365                },
366                (None, None) => {
367                    return Err(ImagingError::Unsupported {
368                        reason: "an automatic W-plane count needs the selection's w statistics",
369                    });
370                }
371            };
372            let screening = admission.admit(
373                "W-projection screens",
374                WPlanes::screens_bytes(geometry, count)?,
375            )?;
376            let screens = WPlanes::screens(geometry, polarization, count)?;
377            let mut kernels = admission.admit(PHASE, screens.kernel_bytes())?;
378            let set = screens.finish()?;
379            drop(screening);
380            kernels.retain(set.resident_bytes());
381            (Box::new(set), Kernels::Admitted { _charge: kernels })
382        }
383        KernelSetKind::Mosaic => {
384            if dish_classes.is_empty() {
385                return Err(ImagingError::Unsupported {
386                    reason: "the mosaic set needs ALMA or ACA dishes in the selection",
387                });
388            }
389            let dishes = dish_classes
390                .iter()
391                .map(|class| {
392                    AiryDish::casa_alma(
393                        match class {
394                            AntennaResponseClass::CasaAlma12m => 12.0,
395                            AntennaResponseClass::CasaAca7m => 7.0,
396                        },
397                        geometry,
398                    )
399                })
400                .collect::<Vec<_>>();
401            let frequency_hz = reference_frequency_hz(problem)?;
402            let windows = mosaic_windows(problem)?;
403            let rules = MosaicRules::Tclean;
404            let screening = admission.admit(
405                "mosaic screens",
406                MosaicPb::screens_bytes(rules, geometry, frequency_hz, &dishes, &windows)?,
407            )?;
408            let screens = MosaicPb::screens(
409                rules,
410                geometry,
411                polarization,
412                frequency_hz,
413                &dishes,
414                &windows,
415            )?;
416            let mut kernels = admission.admit(PHASE, screens.kernel_bytes())?;
417            let set = screens.finish()?;
418            drop(screening);
419            kernels.retain(set.resident_bytes());
420            (Box::new(set), Kernels::Admitted { _charge: kernels })
421        }
422        KernelSetKind::Aw => {
423            let deployment = aw_catalog.ok_or(ImagingError::Unsupported {
424                reason: "an A-projection run needs its convolution-function catalog",
425            })?;
426            let catalog = AwCatalog::open_casa(
427                &deployment.root,
428                deployment.indexing,
429                geometry,
430                polarization,
431                deployment.resident_bytes,
432            )?;
433            let kernels = Kernels::Catalog {
434                charge: catalog.cache_charge(),
435                bound: catalog.cache_bound(),
436                ceiling: admission.workers.max(1),
437                automatic: deployment.automatic,
438            };
439            (Box::new(catalog), kernels)
440        }
441    })
442}

444/// The selected spectral windows as the mosaic set keys them: every
445/// channel of the window, its first channel width and the selected
446/// channels' frequencies (`HetArrayConvFunc` beams per window).
447fn mosaic_windows(problem: &CompiledProblem) -> Result<Vec<MosaicWindow>, ImagingError> {
448    let mut windows = Vec::new();
449    for source in problem.observation_transaction().read_set().sources() {
450        for window in source.selection().spectral_windows() {
451            if windows
452                .iter()
453                .any(|known: &MosaicWindow| known.spectral_window == window.spectral_window_id())
454            {
455                continue;
456            }
457            let catalog = window
458                .coordinate_catalog()
459                .ok_or(ImagingError::Unsupported {
460                    reason: "the mosaic set needs every window's channel frequencies",
461                })?;
462            let selected = window
463                .channel_indices()
464                .iter()
465                .map(|index| {
466                    catalog
467                        .channel_frequency_hz(*index as usize)
468                        .ok_or(ImagingError::Unsupported {
469                            reason: "a selected channel lies outside its window",
470                        })
471                })
472                .collect::<Result<Vec<_>, _>>()?;
473            windows.push(MosaicWindow {
474                spectral_window: window.spectral_window_id(),
475                window_frequencies_hz: catalog.channel_frequencies_hz().to_vec(),
476                channel_width_hz: catalog.first_channel_width_hz(),
477                selected_frequencies_hz: selected,
478            });
479        }
480    }
481    Ok(windows)
482}

484pub(crate) fn image_extent(domain: &CompiledImageDomain) -> Result<ImageExtent, ImagingError> {
485    let direction = domain.direction();
486    let reference = direction.reference_pixel();
487    if reference
488        .iter()
489        .any(|pixel| pixel.fract() != 0.0 || *pixel < 0.0)
490    {
491        return Err(ImagingError::Unsupported {
492            reason: "the image reference pixel must be a whole pixel",
493        });
494    }
495    Ok(ImageExtent {
496        shape: domain.shape().pixels(),
497        increment_rad: direction.increment_rad(),
498        reference_pixel: [reference[0] as usize, reference[1] as usize],
499    })
500}

502/// CASA `FTMachine` channel mapping for a channel-local basis; every native
503/// channel on plane 0 otherwise.
504fn resampler(problem: &CompiledProblem, basis: Basis) -> Result<SpectralResampler, ImagingError> {
505    let Basis::ChannelLocal { planes } = basis else {
506        return Ok(SpectralResampler::direct(basis)?);
507    };
508    let SpectralWcs::Linear {
509        reference_pixel,
510        reference_frequency_hz,
511        increment_hz,
512        ..
513    } = problem.geometry().spectral().wcs()
514    else {
515        return Err(ImagingError::Unsupported {
516            reason: "a tabular spectral axis has no uniform channel width",
517        });
518    };
519    let kernel = match problem.science().spectral().sampling().kernel() {
520        SpectralKernel::Nearest | SpectralKernel::Identity => {
521            casa_imaging_operator::SpectralKernel::Nearest
522        }
523        SpectralKernel::Linear => casa_imaging_operator::SpectralKernel::Linear,
524        SpectralKernel::Cubic | SpectralKernel::ChannelIntegration { .. } => {
525            return Err(ImagingError::Unsupported {
526                reason: "cube interpolation is nearest or linear",
527            });
528        }
529    };
530    let first_hz = reference_frequency_hz - reference_pixel * increment_hz;
531    Ok(SpectralResampler::channel_local(
532        SpectralAxis::new(first_hz, *increment_hz, planes)?,
533        kernel,
534    ))
535}

537/// The widest spacing between adjacent selected native channels of any
538/// selected spectral window, from the storage owner's `CHAN_FREQ` catalog,
539/// widened by [`FRAME_MARGIN`]. A window without a catalog makes it
540/// infinite, so every wave holds the whole model.
541pub(crate) fn native_spacing_hz(problem: &CompiledProblem) -> f64 {
542    let mut widest = 0.0_f64;
543    for window in problem
544        .observation_transaction()
545        .read_set()
546        .sources()
547        .iter()
548        .flat_map(|source| source.selection().spectral_windows())
549    {
550        let Some(catalog) = window.coordinate_catalog() else {
551            return f64::INFINITY;
552        };
553        for pair in window.channel_indices().windows(2) {
554            let (Some(first), Some(second)) = (
555                catalog.channel_frequency_hz(pair[0] as usize),
556                catalog.channel_frequency_hz(pair[1] as usize),
557            ) else {
558                return f64::INFINITY;
559            };
560            widest = widest.max((second - first).abs());
561        }
562    }
563    widest * (1.0 + FRAME_MARGIN)
564}

566/// Shape of the weight-density grid of `domain`: one global plane on CASA's
567/// `VisImagingWeight` cells, or one plane per output channel, plus the
568/// compiled padding planes on each side, on `BriggsCubeWeightor` cells.
569pub(crate) fn density_shape(
570    problem: &CompiledProblem,
571    domain: &CompiledImageDomain,
572) -> Result<DensityGridShape, ImagingError> {
573    let [width, height] = domain.shape().pixels();
574    let (planes, padding, rule) = match problem.weighting().density_scope() {
575        WeightDensityScope::PerOutputChannel => {
576            // CASA always pads the cube density axis
577            // (`BriggsCubeWeightor::estimateSwingChanPad`); a contract without
578            // the padding cannot reproduce it.
579            let padding = problem.weighting().casa_cube_density_padding().ok_or(
580                ImagingError::Unsupported {
581                    reason: "a per-channel weight density needs its compiled padding",
582                },
583            )?;
584            let planes = padding
585                .checked_mul(2)
586                .and_then(|padding| {
587                    padding.checked_add(problem.geometry().spectral().output_channels())
588                })
589                .ok_or(ImagingError::Unsupported {
590                    reason: "the padded density axis overflows",
591                })?;
592            let padding = u32::try_from(padding).map_err(|_| ImagingError::Unsupported {
593                reason: "the padded density axis overflows",
594            })?;
595            (planes, padding, DensityCellRule::Cube)
596        }
597        WeightDensityScope::NotApplicable | WeightDensityScope::GlobalSelection => {
598            (1, 0, DensityCellRule::Standard)
599        }
600    };
601    Ok(DensityGridShape {
602        width,
603        height,
604        planes,
605        padding,
606        increment_rad: domain.direction().increment_rad(),
607        rule,
608    })
