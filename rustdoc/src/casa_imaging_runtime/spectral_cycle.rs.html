41/// Bounded consumer of final visibility samples produced inside the paired
42/// terminal replay. Implementations may write selected visibility columns in
43/// place or retain residual-visibility products, but receive no publication
44/// authority from this interface.
45pub trait FinalVisibilitySink: Send {
46    /// Bind the exact final model before any replay sample is consumed.
47    fn bind(
48        &mut self,
49        problem: casa_imaging_model::CompiledProblemId,
50        final_model: casa_imaging_reconstruction::ModelGenerationId,
51    ) -> io::Result<()>;
52
53    /// Start the one scheduled terminal replay.
54    fn begin_replay(&mut self) -> io::Result<()>;
55
56    /// Consume one bounded canonical selected-visibility block.
57    fn consume(
58        &mut self,
59        samples: &[casa_imaging_reconstruction::runtime_adapter::FinalVisibilitySample],
60    ) -> io::Result<()>;
61
62    /// Close the stream against the terminal selected/weighting replay proof.
63    fn finish(&mut self, replay: &WeightingReplayCompletion) -> io::Result<()>;
64}

94impl FinalVisibilityReplay {
95    /// Create an empty product stream and its runtime sink capability.
96    #[must_use]
97    pub fn new() -> (Self, Box<dyn FinalVisibilitySink>) {
98        let state = Arc::new(Mutex::new(FinalVisibilityReplayState::Unbound));
99        (
100            Self {
101                state: Arc::clone(&state),
102                visibility_write: None,
103            },
104            Box::new(FinalVisibilityReplay {
105                state,
106                visibility_write: None,
107            }),
108        )
109    }
110
111    /// Create a product stream paired with one storage-owner visibility writeback.
112    pub fn with_visibility_write(
113        path: PathBuf,
114        expected: casa_imaging_model::ObservationSourceState,
115        selection: Arc<casa_imaging_model::ObservationSelection>,
116        targets: SelectedVisibilityWriteTargets,
117    ) -> io::Result<(Self, Box<dyn FinalVisibilitySink>)> {
118        let state = Arc::new(Mutex::new(FinalVisibilityReplayState::Unbound));
119        let visibility_write = Arc::new(VisibilityWriteBinding {
120            path,
121            expected,
122            selection,
123            targets,
124            state: Mutex::new(VisibilityWriteState::Idle),
125        });
126        Ok((
127            Self {
128                state: Arc::clone(&state),
129                visibility_write: Some(visibility_write.clone()),
130            },
131            Box::new(FinalVisibilityReplay {
132                state,
133                visibility_write: Some(visibility_write),
134            }),
135        ))
136    }
137
138    /// Return the closed product completion after terminal replay.
139    pub fn completion(&self) -> io::Result<casa_imaging_products::VisibilityProductCompletion> {
140        let state = self
141            .state
142            .lock()
143            .map_err(|_| io::Error::other("final-visibility replay poisoned"))?;
144        match &*state {
145            FinalVisibilityReplayState::Finished(completion) => Ok(*completion),
146            _ => Err(io::Error::other("final-visibility replay is incomplete")),
147        }
148    }
149
150    /// Return whether this handle writes `MODEL_DATA`.
151    #[must_use]
152    pub fn has_model_column(&self) -> bool {
153        self.visibility_write
154            .as_ref()
155            .is_some_and(|write| write.targets.model_data())
156    }
157
158    /// Return whether this handle writes transformed observations to `CORRECTED_DATA`.
159    #[must_use]
160    pub fn has_corrected_data(&self) -> bool {
161        self.visibility_write
162            .as_ref()
163            .is_some_and(|write| write.targets.corrected_data())
164    }
165
166    /// Return whether this handle owns any selected visibility writeback.
167    #[must_use]
168    pub fn has_visibility_write(&self) -> bool {
169        self.visibility_write.is_some()
170    }
171}

173impl FinalVisibilitySink for FinalVisibilityReplay {
174    fn bind(
175        &mut self,
176        problem: casa_imaging_model::CompiledProblemId,
177        final_model: casa_imaging_reconstruction::ModelGenerationId,
178    ) -> io::Result<()> {
179        let mut state = self
180            .state
181            .lock()
182            .map_err(|_| io::Error::other("final-visibility replay poisoned"))?;
183        match &*state {
184            FinalVisibilityReplayState::Unbound => {
185                *state = FinalVisibilityReplayState::Bound(
186                    casa_imaging_products::VisibilityProductAuthority::new(problem, final_model),
187                );
188                Ok(())
189            }
190            FinalVisibilityReplayState::Bound(_) => Ok(()),
191            FinalVisibilityReplayState::Finished(completion)
192                if completion.problem_id() == problem
193                    && completion.final_model() == final_model =>
194            {
195                Ok(())
196            }
197            FinalVisibilityReplayState::Finished(_) => Err(io::Error::other(
198                "visibility products belong to another final model",
199            )),
200        }
201    }
202
203    fn begin_replay(&mut self) -> io::Result<()> {
204        let Some(write) = &self.visibility_write else {
205            return Ok(());
206        };
207        let mut state = write
208            .state
209            .lock()
210            .map_err(|_| io::Error::other("visibility write state poisoned"))?;
211        match &*state {
212            VisibilityWriteState::Idle => {
213                *state = VisibilityWriteState::Writing(VisibilityWriteWorker::spawn(
214                    write.path.clone(),
215                    write.expected.clone(),
216                    Arc::clone(&write.selection),
217                    write.targets,
218                )?);
219                Ok(())
220            }
221            VisibilityWriteState::Writing(_) => Ok(()),
222            VisibilityWriteState::Complete => {
223                Err(io::Error::other("visibility write already finished"))
224            }
225        }
226    }
227
228    fn consume(
229        &mut self,
230        samples: &[casa_imaging_reconstruction::runtime_adapter::FinalVisibilitySample],
231    ) -> io::Result<()> {
232        let mut state = self
233            .state
234            .lock()
235            .map_err(|_| io::Error::other("final-visibility replay poisoned"))?;
236        let FinalVisibilityReplayState::Bound(authority) = &mut *state else {
237            return Err(io::Error::other("final-visibility replay is not bound"));
238        };
239        authority.consume(samples).map_err(io::Error::other)?;
240        if let Some(write) = &self.visibility_write {
241            let state = write
242                .state
243                .lock()
244                .map_err(|_| io::Error::other("visibility write state poisoned"))?;
245            let VisibilityWriteState::Writing(worker) = &*state else {
246                return Err(io::Error::other("visibility writer was not plan-prepared"));
247            };
248            worker.write(samples)?;
249        }
250        Ok(())
251    }
252
253    fn finish(&mut self, replay: &WeightingReplayCompletion) -> io::Result<()> {
254        let mut state = self
255            .state
256            .lock()
257            .map_err(|_| io::Error::other("final-visibility replay poisoned"))?;
258        let authority = match std::mem::replace(&mut *state, FinalVisibilityReplayState::Unbound) {
259            FinalVisibilityReplayState::Bound(authority) => authority,
260            other => {
261                *state = other;
262                return Err(io::Error::other("final-visibility replay is not bound"));
263            }
264        };
265        let completion = authority.finish(
266            replay.selected_generation(),
267            replay
268                .continuum_transform()
269                .map(|completion| completion.generation_id()),
270            replay.weighting_generation(),
271        );
272        if let Some(transform) = replay.continuum_transform()
273            && completion.sample_count() != transform.output_sample_count()
274        {
275            return Err(io::Error::other(format!(
276                "final visibility emitted {} samples after the continuum transform admitted {} output-role samples and weighting retained {} spectral-support samples",
277                completion.sample_count(),
278                transform.output_sample_count(),
279                replay.spectral_support_sample_count(),
280            )));
281        }
282        if let Some(write) = &self.visibility_write {
283            let mut write_state = write
284                .state
285                .lock()
286                .map_err(|_| io::Error::other("visibility write state poisoned"))?;
287            let worker = match std::mem::replace(&mut *write_state, VisibilityWriteState::Idle) {
288                VisibilityWriteState::Writing(worker) => worker,
289                other => {
290                    *write_state = other;
291                    return Err(io::Error::other(
292                        "visibility writer was not active at replay completion",
293                    ));
294                }
295            };
296            drop(write_state);
297            let corrected_data = write
298                .targets
299                .corrected_data()
300                .then(|| {
301                    replay
302                        .continuum_transform()
303                        .map(|transform| {
304                            LogicalIdentity::from_sha256(transform.generation_id().as_bytes())
305                        })
306                        .ok_or_else(|| {
307                            io::Error::other("CORRECTED_DATA write lacks continuum generation")
308                        })
309                })
310                .transpose()?;
311            worker.finish(
312                completion.sample_count(),
313                SelectedVisibilityWriteGenerations {
314                    model_data: write
315                        .targets
316                        .model_data()
317                        .then_some(completion.model_product().identity()),
318                    corrected_data,
319                },
320            )?;
321            *write
322                .state
323                .lock()
324                .map_err(|_| io::Error::other("visibility write state poisoned"))? =
325                VisibilityWriteState::Complete;
326        }
327        *state = FinalVisibilityReplayState::Finished(completion);
328        Ok(())
329    }
330}

343struct SelectedVisibilityCoverage {
344    selection: Arc<casa_imaging_model::ObservationSelection>,
345    row: usize,
346    channel: usize,
347    correlation: usize,
348    written: u64,
349}
350
351struct ExpectedVisibilityAddress {
352    physical_row: u64,
353    data_description_id: u32,
354    spectral_window_id: u32,
355    polarization_id: u32,
356    channel_index: u32,
357    correlation_index: u32,
358}
359
360impl SelectedVisibilityCoverage {
361    fn new(selection: Arc<casa_imaging_model::ObservationSelection>) -> Self {
362        Self {
363            selection,
364            row: 0,
365            channel: 0,
366            correlation: 0,
367            written: 0,
368        }
369    }
370
371    fn push(&mut self, address: casa_imaging_model::SelectedSampleAddress) -> io::Result<()> {
372        let Some(expected) = self.expected()? else {
373            return Err(io::Error::other(
374                "visibility replay exceeded the exact selected write set",
375            ));
376        };
377        if address.physical_row != expected.physical_row
378            || u32::try_from(address.data_description_id).ok() != Some(expected.data_description_id)
379            || address.spectral_window_id != expected.spectral_window_id
380            || address.polarization_id != expected.polarization_id
381            || address.channel_index != expected.channel_index
382            || address.correlation_index != expected.correlation_index
383        {
384            return Err(io::Error::other(format!(
385                "visibility replay address ({}, {}, {}) did not match selected address ({}, {}, {})",
386                address.physical_row,
387                address.channel_index,
388                address.correlation_index,
389                expected.physical_row,
390                expected.channel_index,
391                expected.correlation_index,
392            )));
393        }
394        self.written = self
395            .written
396            .checked_add(1)
397            .ok_or_else(|| io::Error::other("visibility write sample count overflowed"))?;
398        self.advance()?;
399        Ok(())
400    }
401
402    fn finish(&self, expected_samples: u64) -> io::Result<()> {
403        if self.written != expected_samples || self.expected()?.is_some() {
404            return Err(io::Error::other(format!(
405                "visibility writer wrote {} samples without exhausting the exact selected write set of {expected_samples}",
406                self.written
407            )));
408        }
409        Ok(())
410    }
411
412    fn expected(&self) -> io::Result<Option<ExpectedVisibilityAddress>> {
413        let Some(row) = self.selection.rows().ordered_main_rows().get(self.row) else {
414            return Ok(None);
415        };
416        let data_description = self
417            .selection
418            .data_descriptions()
419            .iter()
420            .find(|selection| selection.data_description_id() == row.data_description_id())
421            .ok_or_else(|| io::Error::other("selected visibility DDID is absent"))?;
422        let spectral_window = self
423            .selection
424            .spectral_windows()
425            .iter()
426            .find(|selection| {
427                selection.spectral_window_id() == data_description.spectral_window_id()
428            })
429            .ok_or_else(|| io::Error::other("selected visibility SPW is absent"))?;
430        let correlations = self
431            .selection
432            .correlations()
433            .iter()
434            .find(|selection| selection.polarization_id() == data_description.polarization_id())
435            .ok_or_else(|| io::Error::other("selected visibility polarization is absent"))?;
436        let channel = spectral_window
437            .channel_indices()
438            .get(self.channel)
439            .ok_or_else(|| io::Error::other("selected visibility channel is absent"))?;
440        let correlation = correlations
441            .products()
442            .get(self.correlation)
443            .ok_or_else(|| io::Error::other("selected visibility correlation is absent"))?;
444        Ok(Some(ExpectedVisibilityAddress {
445            physical_row: row.physical_row(),
446            data_description_id: data_description.data_description_id(),
447            spectral_window_id: data_description.spectral_window_id(),
448            polarization_id: data_description.polarization_id(),
449            channel_index: *channel,
450            correlation_index: correlation.correlation_index(),
451        }))
452    }
453
454    fn advance(&mut self) -> io::Result<()> {
455        let row = self
456            .selection
457            .rows()
458            .ordered_main_rows()
459            .get(self.row)
460            .ok_or_else(|| io::Error::other("selected visibility row is absent"))?;
461        let data_description = self
462            .selection
463            .data_descriptions()
464            .iter()
465            .find(|selection| selection.data_description_id() == row.data_description_id())
466            .ok_or_else(|| io::Error::other("selected visibility DDID is absent"))?;
467        let spectral_window = self
468            .selection
469            .spectral_windows()
470            .iter()
471            .find(|selection| {
472                selection.spectral_window_id() == data_description.spectral_window_id()
473            })
474            .ok_or_else(|| io::Error::other("selected visibility SPW is absent"))?;
475        let correlations = self
476            .selection
477            .correlations()
478            .iter()
479            .find(|selection| selection.polarization_id() == data_description.polarization_id())
480            .ok_or_else(|| io::Error::other("selected visibility polarization is absent"))?;
481        self.correlation += 1;
482        if self.correlation == correlations.products().len() {
483            self.correlation = 0;
484            self.channel += 1;
485            if self.channel == spectral_window.channel_indices().len() {
486                self.channel = 0;
487                self.row += 1;
488            }
489        }
490        Ok(())
491    }
492}

494enum VisibilityWriteCommand {
495    Write {
496        values: Vec<SelectedVisibilityCellWrite>,
497        reply: SyncSender<Result<(), String>>,
498    },
499    Finish {
500        expected_samples: u64,
501        generations: SelectedVisibilityWriteGenerations,
502    },
503}
504
505impl VisibilityWriteWorker {
506    fn spawn(
507        path: PathBuf,
508        expected: casa_imaging_model::ObservationSourceState,
509        selection: Arc<casa_imaging_model::ObservationSelection>,
510        targets: SelectedVisibilityWriteTargets,
511    ) -> io::Result<Self> {
512        // A rendezvous channel keeps exactly one copied replay block resident:
513        // the scheduler-accounted producer cannot queue a second block while
514        // the storage worker owns the first.
515        let (sender, receiver) = std::sync::mpsc::sync_channel(0);
516        let (ready_sender, ready_receiver) = std::sync::mpsc::sync_channel(0);
517        let join = std::thread::Builder::new()
518            .name("selected-visibility-write".to_string())
519            .stack_size(VISIBILITY_WRITE_WORKER_STACK_BYTES)
520            .spawn(move || {
521                let mut coverage = SelectedVisibilityCoverage::new(Arc::clone(&selection));
522                let mut writer =
523                    match SelectedVisibilityWrite::begin(path, &expected, &selection, targets) {
524                        Ok(writer) => {
525                            let _ = ready_sender.send(Ok(()));
526                            writer
527                        }
528                        Err(error) => {
529                            let message = error.to_string();
530                            let _ = ready_sender.send(Err(message.clone()));
531                            return Err(io::Error::other(message));
532                        }
533                    };
534                let mut written_samples = 0_u64;
535                let completed_samples = loop {
536                    let command = receiver
537                        .recv()
538                        .map_err(|_| io::Error::other("visibility write controller stopped"))?;
539                    match command {
540                        VisibilityWriteCommand::Write { values, reply } => {
541                            let result = (|| {
542                                written_samples = written_samples
543                                    .checked_add(values.len() as u64)
544                                    .ok_or_else(|| {
545                                        io::Error::other("visibility write sample count overflowed")
546                                    })?;
547                                for cell in values {
548                                    coverage.push(cell.address)?;
549                                    if targets.model_data() {
550                                        writer
551                                            .write(
552                                                casa_imaging_model::MsColumnKind::ModelData,
553                                                cell.address.physical_row,
554                                                cell.address.channel_index,
555                                                cell.address.correlation_index,
556                                                cell.predicted,
557                                            )
558                                            .map_err(io::Error::other)?;
559                                    }
560                                    if targets.corrected_data() {
561                                        writer
562                                            .write(
563                                                casa_imaging_model::MsColumnKind::CorrectedData,
564                                                cell.address.physical_row,
565                                                cell.address.channel_index,
566                                                cell.address.correlation_index,
567                                                cell.observed,
568                                            )
569                                            .map_err(io::Error::other)?;
570                                    }
571                                }
572                                Ok::<(), io::Error>(())
573                            })();
574                            match result {
575                                Ok(()) => {
576                                    let _ = reply.send(Ok(()));
577                                }
578                                Err(error) => {
579                                    let message = error.to_string();
580                                    let _ = reply.send(Err(message.clone()));
581                                    return Err(io::Error::other(message));
582                                }
583                            }
584                        }
585                        VisibilityWriteCommand::Finish {
586                            expected_samples,
587                            generations,
588                        } => {
589                            coverage.finish(expected_samples)?;
590                            writer.complete(generations).map_err(io::Error::other)?;
591                            break written_samples;
592                        }
593                    }
594                };
595                Ok(completed_samples)
596            })
597            .map_err(io::Error::other)?;
598        ready_receiver
599            .recv()
600            .map_err(|_| io::Error::other("visibility writer stopped during startup"))?
601            .map_err(io::Error::other)?;
602        Ok(Self { sender, join })
603    }
604
605    fn write(
606        &self,
607        samples: &[casa_imaging_reconstruction::runtime_adapter::FinalVisibilitySample],
608    ) -> io::Result<()> {
609        let values = samples
610            .iter()
611            .map(|sample| {
612                let address = sample.address();
613                let predicted = sample.predicted();
614                let observed = sample.observed();
615                SelectedVisibilityCellWrite {
616                    address,
617                    predicted: num_complex::Complex32::new(
618                        predicted.re as f32,
619                        predicted.im as f32,
620                    ),
621                    observed: num_complex::Complex32::new(observed.re as f32, observed.im as f32),
622                }
623            })
624            .collect();
625        let (reply, response) = std::sync::mpsc::sync_channel(0);
626        self.sender
627            .send(VisibilityWriteCommand::Write { values, reply })
628            .map_err(|_| io::Error::other("visibility writer stopped"))?;
629        response
630            .recv()
631            .map_err(|_| io::Error::other("visibility writer stopped"))?
632            .map_err(io::Error::other)
633    }
634
635    fn finish(
636        self,
637        expected_samples: u64,
638        generations: SelectedVisibilityWriteGenerations,
639    ) -> io::Result<u64> {
640        self.sender
641            .send(VisibilityWriteCommand::Finish {
642                expected_samples,
643                generations,
644            })
645            .map_err(|_| io::Error::other("visibility writer stopped"))?;
646        self.join
647            .join()
648            .map_err(|_| io::Error::other("visibility writer panicked"))?
649    }
650}

652/// Runtime-owned immutable registry for one serial CPU implementation bundle.
653pub struct SpectralCycleRegistry<I> {
654    id: ImplementationRegistryId,
655    implementation_id: WorkImplementationId,
656    metadata: ImplementationContractMetadata,
657    implementation: I,
658}
659
660impl<I> SpectralCycleRegistry<I> {
661    /// Bind one implementation and its exact compiled science contract.
662    #[must_use]
663    pub fn new(
664        id: ImplementationRegistryId,
665        implementation_id: WorkImplementationId,
666        problem: &CompiledProblem,
667        implementation: I,
668    ) -> Self {
669        Self {
670            id,
671            implementation_id,
672            metadata: ImplementationContractMetadata::new(
673                problem.problem_id(),
674                problem.numerics_id(),
675                problem.required_capabilities().clone(),
676            ),
677            implementation,
678        }
679    }
680
681    /// Return the sole implementation identity owned by this bundle.
682    #[must_use]
683    pub const fn implementation_id(&self) -> &WorkImplementationId {
684        &self.implementation_id
685    }
686
687    /// Borrow the stateful implementation owned by this registry.
688    #[must_use]
689    pub const fn implementation(&self) -> &I {
690        &self.implementation
691    }
692}
693
694impl<I: WorkImplementation> ImplementationRegistry for SpectralCycleRegistry<I> {
695    type Implementation = I;
696
697    fn registry_id(&self) -> ImplementationRegistryId {
698        self.id
699    }
700
701    fn resolve(&self, id: &WorkImplementationId) -> Option<&Self::Implementation> {
702        (id == &self.implementation_id).then_some(&self.implementation)
703    }
704
705    fn implementation_contract(
706        &self,
707        id: &WorkImplementationId,
708    ) -> Option<ImplementationContractMetadata> {
709        (id == &self.implementation_id).then(|| self.metadata.clone())
710    }
711}

715    id: WorkImplementationId,
716    problem: CompiledProblem,

745/// and lease epoch that prepared it.

747    owner_node: WorkNodeId,
748    attempt: crate::ExecutionAttemptId,
749    lease_epoch: u64,

753impl PreparedFinalModel {
754    fn new(context: WorkExecutionContext<'_>, preparation: MajorCyclePreparation) -> Self {
755        Self {
756            owner_node: context.node().id.clone(),
757            attempt: context.attempt_id(),
758            lease_epoch: context.lease_epoch(),
759            preparation,
760        }
761    }
762
763    fn for_replay(
764        &self,
765        context: WorkExecutionContext<'_>,
766    ) -> Result<&MajorCyclePreparation, io::Error> {
767        let direct_predecessor = context
768            .node()
769            .dependencies
770            .contains(&WorkDependency::Work(self.owner_node.clone()));
771        if !direct_predecessor
772            || context.attempt_id() != self.attempt
773            || context.lease_epoch() != self.lease_epoch
774        {
775            return Err(io::Error::other(
776                "terminal replay is not bound to its final-model preparation capability",
777            ));
778        }
779        Ok(&self.preparation)
780    }
781
782    fn into_reconciliation(
783        self,
784        context: WorkExecutionContext<'_>,
785    ) -> Result<MajorCyclePreparation, io::Error> {
786        if context.attempt_id() != self.attempt || context.lease_epoch() != self.lease_epoch {
787            return Err(io::Error::other(
788                "post-replay reconciliation changed final-model preparation authority",
789            ));
790        }
791        Ok(self.preparation)
792    }
793}

812    #[must_use]
813    pub fn identity(&self) -> crate::ArtifactIdentity {
814        let mut hash = Sha256::new();
815        hash.update(b"casa-rs-spectral-cycle-final-major-input-v1");
816        hash.update(self.evidence.reconstruction_cycle.evidence_id().as_bytes());

823        }

837    }

997    fn prepare_final_model(
998        state: &mut SpectralCycleExecutorState,
999        context: WorkExecutionContext<'_>,
1000    ) -> Result<(), io::Error> {
1001        if state.lifecycle.is_some() || state.prepared_model.is_some() {
1002            return Err(io::Error::other(
1003                "final-model preparation executed more than once",
1004            ));
1005        }
1006        let executable = state
1007            .executable
1008            .take()
1009            .ok_or_else(|| io::Error::other("executable model input missing"))?;
1010        let input = state
1011            .pass_input
1012            .take()
1013            .ok_or_else(|| io::Error::other("spectral cycle pass input missing"))?;
1014        let attempt = ModelExecutionAttemptId::new(LogicalIdentity::from_sha256(
1015            context.attempt_id().as_bytes(),
1016        ));
1017        let epoch = context.lease_epoch();
1018        let (lifecycle, named, terms) = match input {
1019            SpectralCyclePassInput::Initial => {
1020                let mut lifecycle =
1021                    ModelLifecycle::bind(executable, attempt, epoch).map_err(io::Error::other)?;
1022                let named = match lifecycle.contract().input() {
1023                    ModelInputCommitment::Empty => lifecycle.initial_empty(),
1024                    ModelInputCommitment::ReprojectedSeed(_) => lifecycle.initial_reprojected(),
1025                    ModelInputCommitment::AlignedSeed { .. }
1026                    | ModelInputCommitment::Generation(_) => {
1027                        return Err(io::Error::other(
1028                            "spectral cycle execution requires an owner-prepared direct model input",
1029                        ));
1030                    }
1031                }
1032                .map_err(io::Error::other)?;
1033                (lifecycle, named, None)
1034            }
1035            SpectralCyclePassInput::FinalMajor(input) => {
1036                let (terms, continuation) = input.into_execution_parts();
1037                let (lifecycle, named) =
1038                    ModelLifecycle::continue_from(executable, attempt, epoch, continuation)
1039                        .map_err(io::Error::other)?;
1040                (lifecycle, named, Some(terms))
1041            }
1042        };
1043        let delta = match terms {
1044            Some(terms) if !terms.is_empty() => Some(
1045                lifecycle
1046                    .compile_delta(&named, terms.iter().copied())
1047                    .map_err(io::Error::other)?,
1048            ),
1049            Some(_) | None => None,
1050        };
1051        let preparation =
1052            MajorCyclePreparation::prepare(&lifecycle, named, delta).map_err(io::Error::other)?;
1053        state.prepared_model = Some(PreparedFinalModel::new(context, preparation));
1054        state.lifecycle = Some(lifecycle);
1055        Ok(())
1056    }

1058    fn run_stream(
1059        &self,
1060        state: &mut SpectralCycleExecutorState,
1061        context: WorkExecutionContext<'_>,
1062        fragment: &WeightingPlanFragment<'_>,
1063        selected: Option<BoundSelectedObservation>,
1064    ) -> Result<(), io::Error> {
1065        if let Some(sink) = &self.final_visibility_sink {
1066            sink.lock()
1067                .map_err(|_| io::Error::other("final visibility sink poisoned"))?
1068                .begin_replay()?;
1069        }
1070        let prepared = state
1071            .prepared
1072            .take()
1073            .ok_or_else(|| io::Error::other("FFT preparation did not run"))?;
1074        let mut operator = prepared
1075            .begin_streaming(context, &self.problem, &self.complete_data)
1076            .map_err(io::Error::other)?;
1077        if self.final_visibility_sink.is_some() {
1078            operator.enable_final_visibility_samples();
1079        }
1080        operator
1081            .bind_major_cycle_model(
1082                state
1083                    .prepared_model
1084                    .as_ref()
1085                    .ok_or_else(|| io::Error::other("final-model preparation missing"))?
1086                    .for_replay(context)?,
1087            )
1088            .map_err(io::Error::other)?;
1089        state.operator = Some(operator);
1090        let SpectralCycleExecutorState {
1091            weighting,
1092            operator,
1093            ..
1094        } = state;
1095        let mut consume = |block: &casa_imaging_reconstruction::WeightingReplayChunk| {
1096            let predicted = operator
1097                .as_mut()
1098                .ok_or_else(|| io::Error::other("complete-data operator missing"))?
1099                .consume_streaming_block(block)
1100                .map_err(io::Error::other)?;
1101            if !predicted.is_empty()
1102                && let Some(sink) = &self.final_visibility_sink
1103            {
1104                sink.lock()
1105                    .map_err(|_| io::Error::other("final visibility sink poisoned"))?
1106                    .consume(predicted)?;
1107            }
1108            Ok::<(), io::Error>(())
1109        };
1110        match fragment.streaming_mode() {
1111            Some(crate::WeightingStreamingMode::NaturalInitial)
1112            | Some(crate::WeightingStreamingMode::DensityInitial) => weighting
1113                .traverse_initial_stream(context, fragment, &self.problem, selected, &mut consume)
1114                .map_err(io::Error::other),
1115            Some(crate::WeightingStreamingMode::Reuse) => {
1116                let selected = selected
1117                    .ok_or_else(|| io::Error::other("later-major selected observation missing"))?;
1118                weighting
1119                    .traverse_reuse_stream(context, fragment, selected, &self.problem, &mut consume)
1120                    .map_err(io::Error::other)
1121            }
1122            None => Err(io::Error::other("streaming weighting mode missing")),
1123        }
1124    }
1125}

1134    fn execute(&self, context: WorkExecutionContext<'_>) -> Result<WorkMeasurements, Self::Error> {
1135        let fragment = self.fragment();
1136        let mut state = self
1137            .state
1138            .lock()
1139            .map_err(|_| io::Error::other("spectral cycle state poisoned"))?;
1140        let final_model_preparation =
1141            crate::spectral_cycle_plan::pass_node("final-model-preparation", self.pass);
1142        if context.node().id == final_model_preparation {
1143            Self::prepare_final_model(&mut state, context)?;
1144            if let (Some(sink), Some(prepared_model)) =
1145                (&self.final_visibility_sink, state.prepared_model.as_ref())
1146            {
1147                sink.lock()
1148                    .map_err(|_| io::Error::other("final visibility sink poisoned"))?
1149                    .bind(
1150                        self.problem.problem_id(),
1151                        prepared_model.preparation.final_model().generation_id(),
1152                    )?;
1153            }
1154        } else if context.node().id == *self.complete_data.preparation_node() {
1155            state.prepared = Some(
1156                self.complete_data
1157                    .prepare(context)
1158                    .map_err(io::Error::other)?,
1159            );
1160        } else if context.node().id == *fragment.source_read_node() {
1161            let selected = state
1162                .selected
1163                .take()
1164                .ok_or_else(|| io::Error::other("selected observation already consumed"))?;
1165            match fragment.streaming_mode() {
1166                Some(crate::WeightingStreamingMode::NaturalInitial) => {
1167                    self.run_stream(&mut state, context, &fragment, Some(selected))?;
1168                }
1169                Some(crate::WeightingStreamingMode::DensityInitial) => {
1170                    state.selected_completion = Some(
1171                        state
1172                            .weighting
1173                            .traverse_density_source(context, &fragment, selected, &self.problem)
1174                            .map_err(io::Error::other)?,
1175                    );
1176                }
1177                Some(crate::WeightingStreamingMode::Reuse) => {
1178                    self.run_stream(&mut state, context, &fragment, Some(selected))?;
1179                }
1180                None => return Err(io::Error::other("streaming weighting mode missing")),
1181            }
1182        } else if context.node().id == *fragment.generation_node()
1183            && fragment.streaming_mode() == Some(crate::WeightingStreamingMode::DensityInitial)
1184        {
1185            self.run_stream(&mut state, context, &fragment, None)?;
1186        } else if self
1187            .complete_data
1188            .reconciliation_node()
1189            .is_some_and(|node| context.node().id == *node)
1190        {
1191            let complete = state
1192                .complete_data
1193                .take()
1194                .ok_or_else(|| io::Error::other("complete-data evidence missing"))?;
1195            let preparation = state
1196                .prepared_model
1197                .take()
1198                .ok_or_else(|| io::Error::other("final-model preparation missing"))?
1199                .into_reconciliation(context)?;
1200            let owner =
1201                MajorCycleOperatorState::begin(complete, preparation).map_err(io::Error::other)?;
1202            let mut lifecycle = state
1203                .lifecycle
1204                .take()
1205                .ok_or_else(|| io::Error::other("model lifecycle missing"))?;
1206            state.result = Some(
1207                owner
1208                    .reconcile(context, &mut lifecycle)
1209                    .map_err(io::Error::other)?,
1210            );
1211            state.lifecycle = Some(lifecycle);
1212        } else if self
1213            .reconstruction_cycle
1214            .as_ref()
1215            .is_some_and(|cycle| context.node().id == cycle.node)
1216        {
1217            let cycle = self
1218                .reconstruction_cycle
1219                .as_ref()
1220                .expect("reconstruction-cycle node matched");
1221            let result = state
1222                .result
1223                .take()
1224                .ok_or_else(|| io::Error::other("initial major completion missing"))?;
1225            let lifecycle = state
1226                .lifecycle
1227                .as_ref()
1228                .ok_or_else(|| io::Error::other("model lifecycle missing"))?;
1229            state.reconstruction_cycle_completion = Some(
1230                InitialMajorPhaseCompletion::new(result)
1231                    .run_reconstruction_cycle(lifecycle, &cycle.mask, cycle.program.clone())
1232                    .map_err(io::Error::other)?,
1233            );
1234        } else if context.node().id == *fragment.release_node() {
1235            state
1236                .weighting
1237                .release(context, &fragment)
1238                .map_err(io::Error::other)?;
1239        }
1240        let resources = context
1241            .node()
1242            .claims
1243            .iter()
1244            .map(|claim| {
1245                ResourceMeasurement::new(
1246                    claim.resource.clone(),
1247                    claim.lifetime.clone(),
1248                    claim.amount,
1249                )
1250            })
1251            .collect();
1252        let io = context
1253            .node()
1254            .claims
1255            .iter()
1256            .filter_map(|claim| match claim.resource {
1257                LeaseResource::IoBuffer(kind) => Some(IoMeasurement::unobserved(kind)),
1258                _ => None,
1259            })
1260            .collect();
1261        let artifacts = self
1262            .phase_input_artifact
1263            .filter(|_| context.node().id == final_model_preparation)
1264            .map(|(identity, bytes)| {
1265                crate::ArtifactMeasurement::new(
1266                    identity,
1267                    Some(identity),
1268                    crate::ArtifactDisposition::Loaded,
1269                    bytes,
1270                    None,
1271                )
1272                .expect("Loaded input evidence is implementation-owned")
1273            })
1274            .into_iter()
1275            .collect();
1276        Ok(WorkMeasurements::new(resources, io, artifacts))
1277    }

1294    fn complete_observation_read(
1295        &self,
1296        completion: ObservationReadCompletionContext,
1297    ) -> Result<AttemptBoundObservationCompletion, Self::Error> {
1298        let fragment = self.fragment();
1299        let mut state = self
1300            .state
1301            .lock()
1302            .map_err(|_| io::Error::other("spectral cycle state poisoned"))?;
1303        if completion.owner_node() == fragment.streaming_node() {
1304            let predecessor = state
1305                .weighting
1306                .complete_replay(completion)
1307                .map_err(io::Error::other)?;
1308            let operator = state
1309                .operator
1310                .take()
1311                .ok_or_else(|| io::Error::other("complete-data operator missing"))?;
1312            let frozen_weighting = state.weighting.frozen_artifact().and_then(|artifact| {
1313                if let Some(reservation) = state.pending_frozen_reservation.take() {
1314                    Some(artifact.with_cross_plan_reservation(reservation))
1315                } else {
1316                    artifact.has_cross_plan_reservation().then_some(artifact)
1317                }
1318            });
1319            let replay = state
1320                .weighting
1321                .replay_completion()
1322                .ok_or_else(|| io::Error::other("replay completion missing"))?;
1323            if let Some(sink) = &self.final_visibility_sink {
1324                sink.lock()
1325                    .map_err(|_| io::Error::other("final visibility sink poisoned"))?
1326                    .finish(replay)?;
1327            }
1328            let complete_data = operator.complete(replay).map_err(io::Error::other)?;
1329            state.frozen_weighting = frozen_weighting;
1330            state.complete_data = Some(complete_data);
1331            return Ok(predecessor);
1332        }
1333        let selected = state
1334            .selected_completion
1335            .take()
1336            .ok_or_else(|| io::Error::other("selected-observation completion missing"))?;
1337        completion.bind(selected).map_err(io::Error::other)
1338    }

1360    }

1362    /// Run one resource-admitted independent cycle over the complete channel slab.
1363    pub fn run_reconstruction_cycle(
1364        self,
1365        lifecycle: &ModelLifecycle,
1366        mask_plan: &ReconstructionMaskPlan,
1367        program: MinorCycleProgram,
1368    ) -> Result<ReconstructionCyclePhaseCompletion, ReconstructionCycleError> {
1369        let completion = self.result.into_completion();
1370        let (normal_state, continuation) = completion.into_continuation();
1371        let (mask, auto_mask) = mask_plan
1372            .materialize(continuation.generation(), &normal_state)
1373            .map_err(|error| ReconstructionCycleError::Minor(error.into()))?;
1374        let cycle = ReconstructionCycle::new(ChannelCyclePolicy::Independent, program).run(
1375            lifecycle,
1376            continuation.generation(),
1377            &normal_state,
1378            &mask,
1379        )?;
1380        let (delta, evidence) = cycle.into_parts();
1381        Ok(ReconstructionCyclePhaseCompletion {
1382            normal_state,
1383            continuation,
1384            mask,
1385            delta,
1386            evidence,
1387            auto_mask,
1388        })
1389    }
1390}
