2//! `sm.openfromms`: a session on a MeasurementSet that already exists.
3//!
4//! CASA's `Simulator(MeasurementSet&)` reads the array, the spectral
5//! windows, the fields and the feed mode back from the MeasurementSet
6//! (`NewMSSimulator::getAnt`, `getSpWindows`, `getFields`, `getFeedMode`);
7//! its `predict` and `corrupt` then rewrite the rows already there. The
8//! session reads what its steps take:
9//! - ANTENNA, whose dishes the beams and the mosaic machine take;
10//! - OBSERVATION, whose first telescope picks the beams (`VPSkyJones`). The
11//!   session's own telescope stays `Unknown`, as `NewMSSimulator`'s does:
12//!   only `setconfig` sets it;
13//! - each data description's spectral window and correlations. casa-rs's
14//!   window `w` is data description `w`, so the data descriptions must name
15//!   the windows in order, as both simulators write them;
16//! - each field's name, code and PHASE_DIR, the phase centre CASA's
17//!   `VisBuffer` predicts at;
18//! - MAIN's rows as scans: each run of consecutive rows with one
19//!   SCAN_NUMBER, FIELD_ID and DATA_DESC_ID is a scan whose rows are on disk
20//!   ([`ScanRows::OnDisk`]). They were observed before any step, so every
21//!   later `predict` and `corrupt` covers them.
22//!
23//! `close` then patches DATA in place (`crate::write::patch`).
24//!
25//! What casa-rs does not read yet is refused, here or by the step needing
26//! it, rather than predicted wrongly: a PHASE_DIR polynomial, channel
27//! frequencies off a linear axis, several feeds per antenna, data
28//! descriptions out of the windows' order, and a primary beam, the mosaic
29//! machine or total-power sampling when POINTING does not point at the
30//! scans' phase centres. Adding to the MeasurementSet after `openfromms`
31//! (`setconfig`, `setfeed`, `setspwindow`, `setfield`, `setmosaicfield`,
32//! `observe`, `observemany`) is not yet implemented.
33

52/// MAIN rows read at a time while finding the scans.
53const SCAN_READ_ROWS: usize = 1 << 16;

55/// Two directions closer than this, radians, are one: a POINTING row at a
56/// scan's phase centre (about 0.2 mas).
57const SAME_DIRECTION_RAD: f64 = 1e-9;

59/// The MeasurementSet a session opened with `openfromms`.
60#[derive(Debug, Clone, PartialEq)]
61pub(crate) struct OpenedMs {
62    /// Whether MAIN has MODEL_DATA, which CASA's `predict` sets to 1.
63    pub(crate) model_data: bool,
64    /// Whether MAIN has CORRECTED_DATA, which CASA's `predict` and
65    /// `corrupt` set to DATA. casa-rs keeps the scratch columns there are
66    /// in step with DATA and creates none (plan decision 6).
67    pub(crate) corrected_data: bool,
68    /// Each scan's first and last TIME, seconds: a POINTING row valid
69    /// during that span points the scan.
70    pub(crate) scan_times_s: Vec<[f64; 2]>,
71    /// Each scan's distinct TIMEs in order, seconds: its integrations,
72    /// which `setgain` solves one by one.
73    pub(crate) integration_times_s: Vec<Vec<f64>>,
74}

77    move |error| SimulatorError::raised(format!("openfromms: {what}: {error}"))

80impl Simulator {
81    /// `sm.openfromms(ms)`: a session on the MeasurementSet at `path`, whose
82    /// rows the following `predict` and `corrupt` rewrite at
83    /// [`close`](Self::close).
84    ///
85    /// # Errors
86    ///
87    /// The MeasurementSet cannot be read (raised, as CASA's table open
88    /// raises), or it holds what casa-rs does not read yet (see the
89    /// module).
90    pub fn open_from_ms(
91        path: impl AsRef<Path>,
92        options: SimulatorOptions,
93    ) -> SimulatorResult<Self> {
94        let raised = |error: &dyn std::fmt::Display| {
95            SimulatorError::raised(format!(
96                "Exception Reported: {}: {error}",
97                path.as_ref().display()
98            ))
99        };
100        // The session holds the MeasurementSet it read: `close` and the
101        // POINTING reads find it whatever the working directory is by then.
102        let path = std::path::absolute(path.as_ref()).map_err(|error| raised(&error))?;
103        let ms = MeasurementSet::open(&path).map_err(|error| raised(&error))?;
104        let mut session = Self::open(&path, options);
105        session.antennas = read_antennas(&ms)?;
106        session.observations = read_observations(&ms)?;
107        session.spectral_windows = read_windows(&ms)?;
108        refuse_several_feeds(&ms)?;
109        // CASA's `getFeedMode` reads the feeds back; `observe`, which takes
110        // them, is refused after `openfromms`.
111        session.feeds_set = true;
112        session.fields = read_fields(&session, &ms)?;
113        let ReadScans {
114            scans,
115            times_s: scan_times_s,
116            integrations,
117        } = read_scans(&ms, &session.fields, session.spectral_windows.len())?;
118        session.scans = scans;
119        let main = ms.main_table();
120        let has = |column: &str| {
121            main.schema()
122                .is_some_and(|schema| schema.contains_column(column))
123        };
124        session.opened = Some(OpenedMs {
125            model_data: has("MODEL_DATA"),
126            corrected_data: has("CORRECTED_DATA"),
127            scan_times_s,
128            integration_times_s: integrations,
129        });
130        tracing::info!(
131            rows = ms.row_count(),
132            scans = session.scans.len(),
133            antennas = session.antennas.len(),
134            spectral_windows = session.spectral_windows.len(),
135            fields = session.fields.len(),
136            "Opened MeasurementSet {}",
137            path.display()
138        );
139        Ok(session)
140    }

142    /// Refuse `call`, which adds to the MeasurementSet, after `openfromms`.
143    pub(super) fn refuse_after_openfromms(&self, call: &str) -> SimulatorResult<()> {
144        if self.opened.is_some() {
145            return Err(SimulatorError::not_yet_implemented(format!(
146                "{call} after openfromms is not yet implemented in casa-rs (#841)"
147            )));
148        }
149        Ok(())

152    /// Refuse `what` (a primary beam, the mosaic machine, total-power
153    /// sampling), which points each scan at its phase centre, on an opened
154    /// MeasurementSet whose POINTING points a scan elsewhere: a pointing
155    /// `observemany` moved off its field, or an offset or tracking error.
156    /// POINTING is read here, when a step needs it.
157    pub(super) fn require_pointings_at_phase_centres(&self, what: &str) -> SimulatorResult<()> {
158        let Some(opened) = &self.opened else {
159            return Ok(());
160        };
161        match pointing_away_from_phase_centre(self.path(), &self.scans, &opened.scan_times_s)? {
162            Some(scan) => Err(SimulatorError::not_yet_implemented(format!(
163                "{what} on a MeasurementSet whose POINTING points scan {} away from its \
164                 field's PHASE_DIR is not yet implemented in casa-rs (#841)",
165                self.scans[scan].scan_number
166            ))),
167            None => Ok(()),
168        }
169    }

178fn read_antennas(ms: &MeasurementSet) -> SimulatorResult<Vec<Antenna>> {
179    let error = read_error("ANTENNA");
180    let antenna = ms.antenna().map_err(&error)?;
181    (0..antenna.row_count())
182        .map(|row| {
183            Ok(Antenna {
184                name: antenna.name(row).map_err(&error)?,
185                station: antenna.station(row).map_err(&error)?,
186                mount: antenna.mount(row).map_err(&error)?,
187                diameter_m: antenna.dish_diameter(row).map_err(&error)?,
188                position_itrf_m: antenna.position(row).map_err(&error)?,
189            })
190        })
191        .collect()
192}

194fn read_observations(ms: &MeasurementSet) -> SimulatorResult<Vec<Observation>> {
195    let error = read_error("OBSERVATION");
196    let observation = ms.observation().map_err(&error)?;
197    (0..observation.row_count())
198        .map(|row| {
199            let time_range = match observation.array(row, "TIME_RANGE").map_err(&error)? {
200                ArrayValue::Float64(range) if range.len() == 2 => [range[[0]], range[[1]]],
201                _ => {
202                    return Err(SimulatorError::raised(
203                        "openfromms: OBSERVATION TIME_RANGE is not two Float64 values",
204                    ));
205                }
206            };
207            Ok(Observation {
208                telescope: observation.string(row, "TELESCOPE_NAME").map_err(&error)?,
209                time_range_s: time_range,
210                observer: observation.string(row, "OBSERVER").map_err(&error)?,
211                project: observation.string(row, "PROJECT").map_err(&error)?,
212            })
213        })
214        .collect()
215}

217/// The windows, one per data description: casa-rs's window `w` is data
218/// description `w` and spectral window `w`, with data description `w`'s
219/// correlations.
220fn read_windows(ms: &MeasurementSet) -> SimulatorResult<Vec<SpectralWindow>> {
221    let error = read_error("SPECTRAL_WINDOW");
222    let windows = ms.spectral_window().map_err(&error)?;
223    let descriptions = ms
224        .data_description()
225        .map_err(read_error("DATA_DESCRIPTION"))?;
226    let polarizations = ms.polarization().map_err(read_error("POLARIZATION"))?;
227    let in_order = descriptions.row_count() == windows.row_count()
228        && (0..descriptions.row_count())
229            .all(|row| descriptions.spectral_window_id(row).ok() == Some(row as i32));
230    if !in_order {
231        return Err(SimulatorError::not_yet_implemented(
232            "openfromms on a MeasurementSet whose data descriptions are not its spectral \
233             windows in order is not yet implemented in casa-rs (#841)",
234        ));
235    }
236    (0..windows.row_count())
237        .map(|row| {
238            let polarization = descriptions
239                .polarization_id(row)
240                .map_err(read_error("DATA_DESCRIPTION"))?;
241            let correlations = usize::try_from(polarization)
242                .ok()
243                .filter(|&id| id < polarizations.row_count())
244                .ok_or_else(|| {
245                    SimulatorError::raised(format!(
246                        "openfromms: data description {row} names POLARIZATION row \
247                         {polarization}, which does not exist"
248                    ))
249                })
250                .and_then(|id| {
251                    polarizations
252                        .corr_type(id)
253                        .map_err(read_error("POLARIZATION"))
254                })?;
255            read_window(&windows, row, correlations)
256        })
257        .collect()
258}

260fn read_window(
261    windows: &casa_ms::subtables::spectral_window::MsSpectralWindow<'_>,
262    row: usize,
263    correlations: Vec<i32>,
264) -> SimulatorResult<SpectralWindow> {
265    let error = read_error("SPECTRAL_WINDOW");
266    let frequencies = windows.chan_freq(row).map_err(&error)?;
267    let widths = windows.chan_width(row).map_err(&error)?;
268    let (Some(&start_frequency_hz), Some(&channel_width_hz)) =
269        (frequencies.first(), widths.first())
270    else {
271        return Err(SimulatorError::raised(format!(
272            "openfromms: spectral window {row} has no channels"
273        )));
274    };
275    let code = windows.meas_freq_ref(row).map_err(&error)?;
276    let frame = FrequencyRef::from_casacore_code(code).ok_or_else(|| {
277        SimulatorError::raised(format!(
278            "openfromms: spectral window {row} has an unknown MEAS_FREQ_REF {code}"
279        ))
280    })?;
281    let window = SpectralWindow {
282        name: windows.name(row).map_err(&error)?,
283        channel_count: frequencies.len(),
284        start_frequency_hz,
285        channel_width_hz,
286        reference_frequency_hz: windows.ref_frequency(row).map_err(&error)?,
287        total_bandwidth_hz: windows.total_bandwidth(row).map_err(&error)?,
288        frame,
289        correlations,
290    };
291    // The prediction takes the channels as the start plus whole channel
292    // widths, as both simulators write them.
293    let linear = widths.iter().all(|&width| width == channel_width_hz)
294        && window
295            .channel_frequencies_hz()
296            .iter()
297            .zip(&frequencies)
298            .all(|(axis, stored)| (axis - stored).abs() <= 4.0 * f64::EPSILON * stored.abs());
299    if !linear {
300        return Err(SimulatorError::not_yet_implemented(format!(
301            "openfromms on spectral window {row}, whose channels are not evenly spaced by their \
302             width, is not yet implemented in casa-rs (#841)"
303        )));
304    }
305    Ok(window)
306}

308/// Refuse a FEED table with several feeds per antenna, which `setfeed`
309/// cannot write either.
310fn refuse_several_feeds(ms: &MeasurementSet) -> SimulatorResult<()> {
311    let error = read_error("FEED");
312    let feed = ms.feed().map_err(&error)?;
313    for row in 0..feed.row_count() {
314        if feed.i32(row, "FEED_ID").map_err(&error)? != 0 {
315            return Err(SimulatorError::not_yet_implemented(
316                "openfromms on a MeasurementSet with several feeds per antenna is not yet \
317                 implemented in casa-rs (#841)",
318            ));
319        }
320    }
321    Ok(())
322}

324fn read_fields(session: &Simulator, ms: &MeasurementSet) -> SimulatorResult<Vec<Field>> {
325    let error = read_error("FIELD");
326    let field = ms.field().map_err(&error)?;
327    (0..field.row_count())
328        .map(|row| {
329            if field.num_poly(row).map_err(&error)? != 0 {
330                return Err(SimulatorError::not_yet_implemented(format!(
331                    "openfromms on field {row}, whose PHASE_DIR is a polynomial, is not yet \
332                     implemented in casa-rs (#841)"
333                )));
334            }
335            let direction = raw_field_phase_direction(ms, row).map_err(&error)?;
336            Ok(Field {
337                name: field.name(row).map_err(&error)?,
338                code: field.code(row).map_err(&error)?,
339                direction_rad: session.j2000_rad(&direction, "the field's PHASE_DIR")?,
340                distance_m: 0.0,
341                source_time_s: field.time(row).map_err(&error)?,
342                rest_frequency_hz: None,
343            })
344        })
345        .collect()
346}

348/// MAIN's rows as scans: each run of consecutive rows with one SCAN_NUMBER,
349/// FIELD_ID and DATA_DESC_ID, phased on its field's PHASE_DIR, as CASA's
350/// `VisBuffer` phases it; each scan's first and last TIME; and each scan's
351/// integrations, its distinct TIMEs in order.
352///
353/// A pointing `observemany` moved off its field keeps its field's FIELD
354/// row, so it is phased on the field: CASA predicts a source at the field's
355/// direction with zero phase there although the pointing's UVW were
356/// computed at the moved direction (measured on CASA 6.7.6;
357/// `docs/CASA (C++) bugs.md`).
358fn read_scans(ms: &MeasurementSet, fields: &[Field], windows: usize) -> SimulatorResult<ReadScans> {
359    let main = ms.main_table();
360    let rows = main.row_count();
361    let mut scans: Vec<Scan> = Vec::new();
362    let mut times_s: Vec<[f64; 2]> = Vec::new();
363    let mut integrations: Vec<Vec<f64>> = Vec::new();
364    let (mut numbers, mut field_ids, mut descriptions) = (Vec::new(), Vec::new(), Vec::new());
365    let mut row_times = Vec::new();
366    let mut indices = Vec::with_capacity(SCAN_READ_ROWS.min(rows));
367    for first in (0..rows).step_by(SCAN_READ_ROWS) {
368        indices.clear();
369        indices.extend(first..rows.min(first + SCAN_READ_ROWS));
370        for column in [&mut numbers, &mut field_ids, &mut descriptions] {
371            column.clear();
372        }
373        row_times.clear();
374        main.required_scalar_columns_for_rows_into(
375            &indices,
376            &mut [
377                RequiredScalarColumnDestination::new(
378                    "SCAN_NUMBER",
379                    RequiredScalarColumnValuesMut::Int32(&mut numbers),
380                ),
381                RequiredScalarColumnDestination::new(
382                    "FIELD_ID",
383                    RequiredScalarColumnValuesMut::Int32(&mut field_ids),
384                ),
385                RequiredScalarColumnDestination::new(
386                    "DATA_DESC_ID",
387                    RequiredScalarColumnValuesMut::Int32(&mut descriptions),
388                ),
389                RequiredScalarColumnDestination::new(
390                    "TIME",
391                    RequiredScalarColumnValuesMut::Float64(&mut row_times),
392                ),
393            ],
394        )
395        .map_err(|error| SimulatorError::raised(format!("openfromms: MAIN: {error}")))?;
396        for (offset, (((&number, &field), &description), &time_s)) in numbers
397            .iter()
398            .zip(&field_ids)
399            .zip(&descriptions)
400            .zip(&row_times)
401            .enumerate()
402        {
403            let row = first + offset;
404            let field_id = index(field, fields.len(), row, "FIELD_ID")?;
405            let window = index(description, windows, row, "DATA_DESC_ID")?;
406            if let Some(scan) = scans.last_mut()
407                && (scan.scan_number, scan.field_id, scan.spectral_window_id)
408                    == (number, field_id, window)
409                && let ScanRows::OnDisk(scan_rows) = &mut scan.rows
410                && let Some(span) = times_s.last_mut()
411                && let Some(times) = integrations.last_mut()
412            {
413                scan_rows.end = row + 1;
414                *span = [span[0].min(time_s), span[1].max(time_s)];
415                if times.last() != Some(&time_s) {
416                    times.push(time_s);
417                }
418                continue;
419            }
420            scans.push(Scan {
421                scan_number: number,
422                field_id,
423                spectral_window_id: window,
424                phase_center_rad: fields[field_id].direction_rad,
425                // POINTING, read when a step needs it, must point there
426                // (`require_pointings_at_phase_centres`).
427                pointing_rad: fields[field_id].direction_rad,
428                rows: ScanRows::OnDisk(row..row + 1),
429            });
430            times_s.push([time_s, time_s]);
431            integrations.push(vec![time_s]);
432        }
433    }
434    for times in &mut integrations {
435        times.sort_by(f64::total_cmp);
436        times.dedup();
437    }
438    Ok(ReadScans {
439        scans,
440        times_s,
441        integrations,
442    })
443}

454/// `value`, MAIN row `row`'s `column`, as an index below `count`.
455fn index(value: i32, count: usize, row: usize, column: &str) -> SimulatorResult<usize> {
456    usize::try_from(value)
457        .ok()
458        .filter(|&index| index < count)
459        .ok_or_else(|| {
460            SimulatorError::raised(format!(
461                "openfromms: MAIN row {row} has {column} {value}, which does not exist"
462            ))
463        })
464}

466/// The first scan a POINTING row valid within its time span points away
467/// from its phase centre, `None` when every row points at the scans it is
468/// valid for; without POINTING rows CASA points at the fields.
469///
470/// A row is valid for the times casacore's `MSPointingColumns::
471/// pointingIndex` finds it at, which CASA's `VisBuffer` reads each row's
472/// pointing with ([`Validity`]). A scan's span is from its first to its
473/// last TIME, so a row valid between two of its integrations counts too:
474/// that refuses what CASA might not use, rather than predicting a scan at
475/// a pointing it does not have.
476fn pointing_away_from_phase_centre(
477    path: &Path,
478    scans: &[Scan],
479    times_s: &[[f64; 2]],
480) -> SimulatorResult<Option<usize>> {
481    let error =
482        |error: casa_tables::TableError| SimulatorError::raised(format!("read POINTING: {error}"));
483    let pointing_path = path.join("POINTING");
484    if !pointing_path.exists() {
485        return Ok(None);
486    }
487    let pointing = Table::open(TableOptions::new(&pointing_path)).map_err(error)?;
488    let by_time = ScansByTime::new(times_s);
489    let mut validity = Validity::default();
490    let (mut antennas, mut times, mut intervals, mut rows) =
491        (Vec::new(), Vec::new(), Vec::new(), Vec::new());
492    let mut valid = Vec::new();
493    let count = pointing.row_count();
494    for first in (0..count).step_by(SCAN_READ_ROWS) {
495        rows.clear();
496        rows.extend(first..count.min(first + SCAN_READ_ROWS));
497        antennas.clear();
498        times.clear();
499        intervals.clear();
500        pointing
501            .required_scalar_columns_for_rows_into(
502                &rows,
503                &mut [
504                    RequiredScalarColumnDestination::new(
505                        "ANTENNA_ID",
506                        RequiredScalarColumnValuesMut::Int32(&mut antennas),
507                    ),
508                    RequiredScalarColumnDestination::new(
509                        "TIME",
510                        RequiredScalarColumnValuesMut::Float64(&mut times),
511                    ),
512                    RequiredScalarColumnDestination::new(
513                        "INTERVAL",
514                        RequiredScalarColumnValuesMut::Float64(&mut intervals),
515                    ),
516                ],
517            )
518            .map_err(error)?;
519        // Cell by cell, which every data manager serves: CASA keeps
520        // DIRECTION in a StandardStMan.
521        let directions = pointing
522            .column_accessor("DIRECTION")
523            .and_then(|column| column.array_cells_owned_uncached(&rows))
524            .map_err(error)?;
525        for (((&antenna, &time_s), &interval_s), direction) in
526            antennas.iter().zip(&times).zip(&intervals).zip(&directions)
527        {
528            let direction = fixed_direction(direction.as_ref());
529            validity.push((antenna, time_s, interval_s), direction, &mut valid);
530            if let Some(scan) = by_time.away(&mut valid, scans) {
531                return Ok(Some(scan));
532            }
533        }
534    }
535    validity.finish(&mut valid);
536    Ok(by_time.away(&mut valid, scans))
537}

539/// A POINTING row's DIRECTION, as (longitude, latitude), radians: its
540/// first polynomial term, `None` when a later term moves it or the cell is
541/// not a direction, which no scan's phase centre matches.
542fn fixed_direction(cell: Option<&ArrayValue>) -> Option<[f64; 2]> {
543    match cell {
544        Some(ArrayValue::Float64(direction))
545            if direction.ndim() == 2 && direction.shape()[0] == 2 && direction.shape()[1] > 0 =>
546        {
547            let moves = direction
548                .indexed_iter()
549                .any(|(index, &term)| index[1] > 0 && term != 0.0);
550            (!moves).then(|| [direction[[0, 0]], direction[[1, 0]]])
551        }
552        _ => None,
553    }
554}

556/// A POINTING row's direction ([`fixed_direction`]) and the times it is
557/// valid for, seconds.
558type ValidPointing = (Option<[f64; 2]>, [f64; 2]);

560/// The times POINTING rows are valid for, found in table order as
561/// casacore's `MSPointingColumns::pointingIndex` finds them. A row is
562/// valid within half its INTERVAL of its TIME; with INTERVAL 0, within half
563/// the gap to its antenna's next row at another TIME, else to its previous
564/// one, else at every time; and with a negative INTERVAL at every time.
565#[derive(Debug, Default)]
566struct Validity {
567    antennas: HashMap<i32, AntennaPointings>,
568}

570/// One antenna's rows so far.
571#[derive(Debug)]
572struct AntennaPointings {
573    /// The TIME of its last row.
574    last_s: f64,
575    /// The TIME of its last row at another TIME before that.
576    earlier_s: Option<f64>,
577    /// The directions of its rows of INTERVAL 0 at `last_s`, which the next
578    /// row at another TIME bounds.
579    waiting: Vec<Option<[f64; 2]>>,
580}
581
582impl Validity {
583    /// Take the next row, of `antenna` at `time_s` with `interval_s`,
584    /// pointing at `direction`: onto `valid` go the rows whose times are
585    /// now known.
586    fn push(
587        &mut self,
588        (antenna, time_s, interval_s): (i32, f64, f64),
589        direction: Option<[f64; 2]>,
590        valid: &mut Vec<ValidPointing>,
591    ) {
592        let rows = self
593            .antennas
594            .entry(antenna)
595            .or_insert_with(|| AntennaPointings {
596                last_s: time_s,
597                earlier_s: None,
598                waiting: Vec::new(),
599            });
600        if rows.last_s != time_s {
601            let half = (time_s - rows.last_s).abs() / 2.0;
602            let span = [rows.last_s - half, rows.last_s + half];
603            valid.extend(rows.waiting.drain(..).map(|direction| (direction, span)));
604            rows.earlier_s = Some(rows.last_s);
605            rows.last_s = time_s;
606        }
607        if interval_s == 0.0 {
608            rows.waiting.push(direction);
609        } else if interval_s > 0.0 {
610            let half = interval_s / 2.0;
611            valid.push((direction, [time_s - half, time_s + half]));
612        } else {
613            valid.push((direction, [f64::NEG_INFINITY, f64::INFINITY]));
614        }
615    }
616
617    /// Onto `valid` go the rows of INTERVAL 0 no later row bounds: by the
618    /// antenna's earlier TIME, else at every time.
619    fn finish(&mut self, valid: &mut Vec<ValidPointing>) {
620        for rows in self.antennas.values_mut() {
621            let span = rows
622                .earlier_s
623                .map_or([f64::NEG_INFINITY, f64::INFINITY], |earlier| {
624                    let half = (rows.last_s - earlier).abs() / 2.0;
625                    [rows.last_s - half, rows.last_s + half]
626                });
627            valid.extend(rows.waiting.drain(..).map(|direction| (direction, span)));
628        }
629    }
630}

632/// The scans by their first TIME, with the latest last TIME so far, to
633/// find those a span of time overlaps.
634struct ScansByTime<'t> {
635    times_s: &'t [[f64; 2]],
636    by_start: Vec<usize>,
637    latest_end: Vec<f64>,
638}
639
640impl<'t> ScansByTime<'t> {
641    fn new(times_s: &'t [[f64; 2]]) -> Self {
642        let mut by_start: Vec<usize> = (0..times_s.len()).collect();
643        by_start.sort_by(|&a, &b| times_s[a][0].total_cmp(&times_s[b][0]));
644        let mut latest_end = Vec::with_capacity(by_start.len());
645        for &scan in &by_start {
646            let end = latest_end
647                .last()
648                .map_or(f64::NEG_INFINITY, |&end: &f64| end);
649            latest_end.push(end.max(times_s[scan][1]));
650        }
651        Self {
652            times_s,
653            by_start,
654            latest_end,
655        }
656    }
657
658    /// The first scan of `scans` a row of `valid`, which this empties, is
659    /// valid during and points away from.
660    fn away(&self, valid: &mut Vec<ValidPointing>, scans: &[Scan]) -> Option<usize> {
661        for (direction, [from_s, to_s]) in valid.drain(..) {
662            let started = self
663                .by_start
664                .partition_point(|&scan| self.times_s[scan][0] <= to_s);
665            for position in (0..started).rev() {
666                if self.latest_end[position] < from_s {
667                    break;
668                }
669                let scan = self.by_start[position];
670                let away = direction.is_none_or(|direction| {
671                    separation_rad(scans[scan].phase_center_rad, direction) >= SAME_DIRECTION_RAD
672                });
673                if self.times_s[scan][1] >= from_s && away {
674                    return Some(scan);
675                }
676            }
677        }
678        None
679    }
680}

682/// The angle between two directions given as (longitude, latitude),
683/// radians.
684fn separation_rad([lon1, lat1]: [f64; 2], [lon2, lat2]: [f64; 2]) -> f64 {
685    let haversine = ((lat2 - lat1) / 2.0).sin().powi(2)
686        + lat1.cos() * lat2.cos() * ((lon2 - lon1) / 2.0).sin().powi(2);
687    2.0 * haversine.sqrt().min(1.0).asin()
688}
