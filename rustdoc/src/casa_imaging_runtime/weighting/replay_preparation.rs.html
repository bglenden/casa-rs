24#[derive(Clone, Copy, Debug)]
25pub(crate) struct ReplayPreparationPlan {
26    index: SelectedObservationBlockIndexPlan,
27    workers: usize,
28    samples_per_worker: usize,
29    runs_per_batch: usize,
30    numeric: Option<super::bulk_source::NumericGeometryPlan>,
31    bytes: u64,
32}
33
34impl ReplayPreparationPlan {
35    pub(crate) fn new(
36        problem: &CompiledProblem,
37        weighting: &WeightingPlan,
38        workers: usize,
39        source: &super::SelectedObservationSourceResources,
40    ) -> Result<Self, WeightingError> {
41        let correlations = problem
42            .selected_observation()
43            .read_set()
44            .sources()
45            .iter()
46            .flat_map(|source| source.selection().correlations())
47            .map(|selection| selection.products().len())
48            .max()
49            .unwrap_or(0);
50        if workers == 0 || correlations == 0 {
51            return Err(WeightingError::ResidencyOverflow);
52        }
53        let runs = weighting.limits().max_block_samples() / correlations;
54        if runs == 0 {
55            return Err(WeightingError::ResidencyOverflow);
56        }
57        let index = SelectedObservationBlockIndexPlan::new(problem, runs)
58            .map_err(|_| WeightingError::ResidencyOverflow)?;
59        let samples_per_worker = runs
60            .div_ceil(workers)
61            .checked_mul(correlations)
62            .ok_or(WeightingError::ResidencyOverflow)?;
63        let numeric = if problem.visibility_transform().is_none()
64            && matches!(
65                problem.reconstruction().basis(),
66                casa_imaging_model::ReconstructionBasis::Constant
67            )
68            && casa_imaging_reconstruction::SpectralOperatorSpecification::new(problem)
69                .map_err(|_| WeightingError::ProblemMismatch)?
70                .supports_bulk_mfs()
71        {
72            Some(super::bulk_source::NumericGeometryPlan::for_source(
73                problem,
74                &source.residency,
75            )?)
76        } else {
77            None
78        };
79        let residency = weighting.planned_residency();
80        let sample_bytes =
81            residency.weighted_block_bytes() / weighting.limits().max_block_samples();
82        let worker_bytes = samples_per_worker
83            .checked_mul(sample_bytes)
84            .and_then(|bytes| {
85                bytes.checked_add(
86                    residency
87                        .spectral_cache_bytes()
88                        .checked_sub(size_of::<WeightingSpectralCache<'_>>())?,
89                )
90            })
91            .and_then(|bytes| {
92                bytes.checked_add(
93                    SelectedObservationProjector::required_bytes(problem)?
94                        .checked_sub(size_of::<SelectedObservationProjector>())?,
95                )
96            })
97            .and_then(|bytes| bytes.checked_add(size_of::<PreparationWorker<'_>>()))
98            .ok_or(WeightingError::ResidencyOverflow)?;
99        let bytes = workers
100            .checked_mul(worker_bytes)
101            .and_then(|bytes| {
102                bytes.checked_add(
103                    index
104                        .capacity_bytes()
105                        .checked_sub(size_of::<SelectedObservationBlockIndex>())?,
106                )
107            })
108            .and_then(|bytes| bytes.checked_add(size_of::<ReplayPreparation<'_>>()))
109            .and_then(|bytes| bytes.checked_add(numeric.map_or(0, |plan| plan.bytes)))
110            .and_then(|bytes| {
111                bytes.checked_add(if numeric.is_some() && workers > 1 {
112                    residency.weighted_block_bytes()
113                } else {
114                    0
115                })
116            })
117            .and_then(|bytes| u64::try_from(bytes).ok())
118            .ok_or(WeightingError::ResidencyOverflow)?;
119        Ok(Self {
120            index,
121            workers,
122            samples_per_worker,
123            runs_per_batch: runs,
124            numeric,
125            bytes,
126        })
127    }
128
129    pub(crate) const fn workers(self) -> usize {
130        self.workers
131    }
132    pub(crate) fn admitted_heap_bytes(self) -> Result<u64, WeightingError> {
133        let stacks = if self.workers > 1 {
134            self.workers as u64 * crate::bounded_stream::BOUNDED_WORKER_STACK_BYTES as u64
135        } else {
136            0
137        };
138        crate::bounded_stream::BoundedKernelPlan::new::<(), ()>(self.workers, 1, 0)
139            .ok()
140            .and_then(|plan| plan.capacity_bytes().checked_sub(stacks))
141            .and_then(|bytes| bytes.checked_add(self.bytes))
142            .ok_or(WeightingError::ResidencyOverflow)
143    }
144}

146struct PreparationWorker<'a> {
147    projector: SelectedObservationProjector,
148    spectral: WeightingSpectralCache<'a>,
149    prepared: Vec<ReconstructionWeightedSample>,
150    samples: u64,
151}
152
153struct ActivePreparation<'a>(&'a AtomicUsize);
154impl Drop for ActivePreparation<'_> {
155    fn drop(&mut self) {
156        self.0.fetch_sub(1, Ordering::Relaxed);
157    }
158}

160pub(super) struct ReplayPreparation<'a> {
161    plan: ReplayPreparationPlan,
162    index: Option<SelectedObservationBlockIndex>,
163    geometry: Option<casa_ms::SelectedObservationNumericGeometry>,
164    workers: Vec<PreparationWorker<'a>>,
165    active: AtomicUsize,
166    peak_active: AtomicUsize,
167    runs: u64,
168    index_payload_bytes: u64,
169    preparation_nanos: u128,
170    ordered_commit_nanos: u128,
171}
172
173impl<'a> ReplayPreparation<'a> {
174    pub(super) fn new(
175        problem: &'a CompiledProblem,
176        plan: ReplayPreparationPlan,
177    ) -> Result<Self, WeightingError> {
178        let mut workers = Vec::with_capacity(plan.workers);
179        for _ in 0..plan.workers {
180            workers.push(PreparationWorker {
181                projector: SelectedObservationProjector::new(problem),
182                spectral: WeightingSpectralCache::new(problem)?,
183                prepared: Vec::with_capacity(plan.samples_per_worker),
184                samples: 0,
185            });
186        }
187        Ok(Self {
188            plan,
189            index: plan.numeric.is_none().then(|| plan.index.create_index()),
190            geometry: plan
191                .numeric
192                .map(|plan| plan.create())
193                .transpose()
194                .map_err(|_| WeightingError::ResidencyOverflow)?,
195            workers,
196            active: AtomicUsize::new(0),
197            peak_active: AtomicUsize::new(0),
198            runs: 0,
199            index_payload_bytes: 0,
200            preparation_nanos: 0,
201            ordered_commit_nanos: 0,
202        })
203    }
204
205    pub(super) fn sample_count(&self) -> u64 {
206        self.workers.iter().map(|worker| worker.samples).sum()
207    }
208
209    pub(super) fn log(&self) {
210        eprintln!(
211            "imaging_replay_preparation_summary workers={} peak_active_jobs={} indexed_runs={} prepared_samples={} planned_workspace_bytes={} index_payload_bytes={} preparation_nanos={} ordered_commit_nanos={} source_kind={}",
212            self.plan.workers,
213            self.peak_active.load(Ordering::Relaxed),
214            self.runs,
215            self.sample_count(),
216            self.plan.bytes,
217            self.index_payload_bytes,
218            self.preparation_nanos,
219            self.ordered_commit_nanos,
220            if self.plan.numeric.is_some() {
221                "numeric_columns"
222            } else {
223                "indexed_samples"
224            }
225        );
226    }
227
228    pub(super) fn consume<W, F, E>(
229        &mut self,
230        problem: &CompiledProblem,
231        consumer: &mut SelectedObservationBlockConsumer<'_>,
232        weights: &mut W,
233        storage: &SelectedObservationBlock,
234        execution: crate::bounded_stream::BoundedExecution<'_>,
235        emit: &mut F,
236    ) -> Result<(), WeightingBlockKernelError<E>>
237    where
238        W: StreamingWeightPhase + Sync,
239        F: FnMut(
240                &ReconstructionWeightedBlock,
241                crate::bounded_stream::BoundedExecution<'_>,
242            ) -> Result<(), E>
243            + Send,
244        E: Error + Send + 'static,
245    {
246        if self.plan.numeric.is_some() {
247            return self.consume_numeric(problem, consumer, weights, storage, execution, emit);
248        }
249        let index = self.index.as_mut().expect("indexed preparation plan");
250        let count = storage.selected_run_count().map_err(|error| {
251            WeightingBlockKernelError::Traversal(SelectedObservationTraversalError::Source(error))
252        })?;
253        let mut start = 0;
254        while start < count {
255            let range = self
256                .plan
257                .index
258                .next_range(storage, start)
259                .map_err(|error| {
260                    WeightingBlockKernelError::Traversal(SelectedObservationTraversalError::Source(
261                        error,
262                    ))
263                })?;
264            let end = range.end;
265            consumer
266                .index_block_range(storage, start..end, index)
267                .map_err(|error| {
268                    WeightingBlockKernelError::Traversal(widen_terminal_traversal_error(error))
269                })?;
270            self.runs += (end - start) as u64;
271            self.index_payload_bytes +=
272                (index.current_bytes().map_err(|_| {
273                    WeightingBlockKernelError::Owner(WeightingError::ResidencyOverflow)
274                })? - size_of::<SelectedObservationBlockIndex>()) as u64;
275            let indexed = index.view(storage, problem).map_err(|error| {
276                WeightingBlockKernelError::Traversal(SelectedObservationTraversalError::Source(
277                    error,
278                ))
279            })?;
280            let runs_per_worker = indexed.run_count().div_ceil(self.plan.workers);
281            let weights = &mut *weights;
282            let (active, peak_active) = (&self.active, &self.peak_active);
283            let started = Instant::now();
284            execution.for_each_mut(&mut self.workers, |ordinal, worker| {
285                worker.prepared.clear();
286                let first = (ordinal * runs_per_worker).min(indexed.run_count());
287                let last = (first + runs_per_worker).min(indexed.run_count());
288                if first == last {
289                    return Ok(());
290                }
291                let active_jobs = active.fetch_add(1, Ordering::Relaxed) + 1;
292                peak_active.fetch_max(active_jobs, Ordering::Relaxed);
293                let _active = ActivePreparation(active);
294                indexed
295                    .visit_range(&mut worker.projector, first..last, |run| {
296                        for reported in run.samples() {
297                            if worker.prepared.len() == worker.prepared.capacity() {
298                                return Err(ReplayCallbackError::Owner(
299                                    WeightingError::ResidencyOverflow,
300                                ));
301                            }
302                            let contributions = worker
303                                .spectral
304                                .compile(reported.selected(), reported.spectral_evaluation())
305                                .map_err(ReplayCallbackError::Owner)?;
306                            worker.prepared.push(
307                                weights
308                                    .prepare_sample(
309                                        problem,
310                                        reported.selected(),
311                                        reported.spectral_evaluation().output_frame().centre_hz(),
312                                        contributions,
313                                    )
314                                    .map_err(ReplayCallbackError::Owner)?,
315                            );
316                        }
317                        Ok::<(), ReplayCallbackError<E>>(())
318                    })
319                    .map_err(WeightingBlockKernelError::Traversal)?;
320                worker.samples += worker.prepared.len() as u64;
321                Ok(())
322            })?;
323            self.preparation_nanos += started.elapsed().as_nanos();
324            let started = Instant::now();
325            for worker in &mut self.workers {
326                while !worker.prepared.is_empty() {
327                    if let Some(block) = weights
328                        .commit_prepared(problem, &mut worker.prepared)
329                        .map_err(WeightingBlockKernelError::Owner)?
330                    {
331                        emit(&block, execution).map_err(WeightingBlockKernelError::Consumer)?;
332                        weights
333                            .reuse_emitted_block(block)
334                            .map_err(WeightingBlockKernelError::Owner)?;
335                    }
336                }
337            }
338            self.ordered_commit_nanos += started.elapsed().as_nanos();
339            start = end;
340        }
341        index.clear();
342        Ok(())
343    }
