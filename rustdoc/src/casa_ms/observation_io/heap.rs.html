2//! The heap a catalogue and its entities hold: every string and array at
3//! its capacity, a map's entries twice (for its nodes' spare slots) and
4//! each ephemeris once. Each entity is taken apart whole, so a member added
5//! to one is counted here or the build fails. Openings bound what they will
6//! hold with these before they build it, and creation plans charge their
7//! catalogues by them.
8
9use casa_observation::{
10    AntennaEntry, AntennaId, Catalogue, DataGroupEntry, FieldEntry, ObservationEntry,
11    PartitionEntry, ProcessorEntry, Receptors, ScanEntry, SourceEntry, SourceRecord, StateEntry,
12    VisibilityUnits, WindowEntry,
13};
14use casa_types::quanta::Quantity;

16/// The heap `catalogue` holds ([`Catalogue::ephemeris_bytes`] for its
17/// ephemerides).
18pub(crate) fn catalogue_bytes(catalogue: &Catalogue) -> usize {
19    let Catalogue {
20        antennas,
21        fields,
22        scans,
23        observations,
24        partitions,
25        windows,
26        sources,
27        states,
28        processors,
29    } = catalogue;
30    catalogue.ephemeris_bytes()
31        + entries(antennas, antenna_bytes)
32        + entries(fields, field_bytes)
33        + entries(scans, |scan| {
34            let ScanEntry {
35                name,
36                number: _,
37                intents,
38            } = scan;
39            name.capacity() + strings(intents)
40        })
41        + entries(observations, observation_bytes)
42        + entries(partitions, partition_bytes)
43        + entries(windows, window_bytes)
44        + entries(sources, |source| {
45            let SourceEntry { name, records } = source;
46            name.capacity() + entries(records, record_bytes)
47        })
48        + entries(states, |state| {
49            let StateEntry {
50                obs_mode,
51                cal_k: _,
52                load_k: _,
53                signal: _,
54                reference: _,
55                sub_scan: _,
56                flagged: _,
57            } = state;
58            obs_mode.capacity()
59        })
60        + entries(processors, |processor| {
61            let ProcessorEntry {
62                kind,
63                sub_type,
64                type_id: _,
65                mode_id: _,
66                pass_id: _,
67                flagged: _,
68            } = processor;
69            optional(kind) + optional(sub_type)
70        })
71}

73/// A vector's slots, at its capacity, and what each entry holds beyond.
74pub(crate) fn entries<T>(values: &Vec<T>, held: impl Fn(&T) -> usize) -> usize {
75    values.capacity() * size_of::<T>() + values.iter().map(held).sum::<usize>()
76}
77
78/// The heap a vector of plain values holds.
79pub(crate) fn plain<T>(values: &Vec<T>) -> usize {
80    values.capacity() * size_of::<T>()
81}
82
83pub(crate) fn strings(values: &Vec<String>) -> usize {
84    entries(values, String::capacity)
85}
86
87fn optional(value: &Option<String>) -> usize {
88    value.as_ref().map_or(0, String::capacity)
89}
90
91/// A quantity's heap: its unit's name.
92fn quantity(value: &Quantity) -> usize {
93    value.unit().name().len()
94}

96fn antenna_bytes(antenna: &AntennaEntry) -> usize {
97    let AntennaEntry {
98        name,
99        station,
100        position: _,
101        mount,
102        dish_diameter,
103        flagged: _,
104    } = antenna;
105    name.capacity() + station.capacity() + mount.capacity() + quantity(dish_diameter)
106}

108fn field_bytes(field: &FieldEntry) -> usize {
109    // The ephemeris, which fields may share, is counted once for the
110    // catalogue.
111    let FieldEntry {
112        name,
113        phase_centre: _,
114        reference_epoch: _,
115        source_name,
116        source: _,
117        code,
118        delay_centre: _,
119        reference_centre: _,
120        flagged: _,
121        ephemeris: _,
122    } = field;
123    name.capacity() + optional(source_name) + optional(code)
124}

126fn observation_bytes(observation: &ObservationEntry) -> usize {
127    let ObservationEntry {
128        telescope,
129        observers,
130        project,
131        execution_block,
132        session_reference,
133        scheduling_block,
134        time_range: _,
135        release_date: _,
136        flagged: _,
137    } = observation;
138    telescope.capacity()
139        + strings(observers)
140        + project.capacity()
141        + optional(execution_block)
142        + optional(session_reference)
143        + optional(scheduling_block)
144}

146fn window_bytes(window: &WindowEntry) -> usize {
147    let WindowEntry {
148        name,
149        reference_frequency: _,
150        total_bandwidth_hz: _,
151        if_chain: _,
152        net_sideband: _,
153        frequency_group: _,
154        frequency_group_name,
155        effective_bandwidths_hz,
156        resolutions_hz,
157        flagged: _,
158    } = window;
159    name.capacity()
160        + optional(frequency_group_name)
161        + effective_bandwidths_hz.as_ref().map_or(0, plain)
162        + resolutions_hz.as_ref().map_or(0, plain)
163}

165fn record_bytes(record: &SourceRecord) -> usize {
166    let SourceRecord {
167        window: _,
168        time: _,
169        interval_s: _,
170        code,
171        calibration_group: _,
172        direction: _,
173        proper_motion_rad_s: _,
174        position: _,
175        pulsar_id: _,
176        num_lines: _,
177        transitions,
178        rest_frequencies,
179        systemic_velocities,
180    } = record;
181    optional(code)
182        + transitions.as_ref().map_or(0, strings)
183        + rest_frequencies.as_ref().map_or(0, plain)
184        + systemic_velocities.as_ref().map_or(0, plain)
185}

187/// The heap a partition entry holds.
188pub(crate) fn partition_bytes(partition: &PartitionEntry) -> usize {
189    let PartitionEntry {
190        spectral_window,
191        window: _,
192        frequencies_hz,
193        frequency_frame: _,
194        channel_widths_hz,
195        correlations,
196        intents,
197        observation: _,
198        feeds: _,
199        fields,
200        scans,
201        baselines,
202        time_scale: _,
203        time_origin: _,
204        data_groups,
205        receptors,
206        states,
207        processor: _,
208    } = partition;
209    spectral_window.capacity()
210        + plain(frequencies_hz)
211        + plain(channel_widths_hz)
212        + plain(correlations)
213        + strings(intents)
214        + plain(fields)
215        + plain(scans)
216        + plain(baselines)
217        + entries(data_groups, group_bytes)
218        + receptors.values().map(receptors_bytes).sum::<usize>()
219        + plain(states)
220}

222/// The heap one antenna's receptors hold in a partition's map, its entry
223/// included.
224pub(crate) fn receptors_bytes(receptors: &Receptors) -> usize {
225    let Receptors { types, angles } = receptors;
226    2 * size_of::<(AntennaId, Receptors)>() + strings(types) + entries(angles, quantity)
227}

229/// The heap a data group's entry holds.
230pub(crate) fn group_bytes(group: &DataGroupEntry) -> usize {
231    let DataGroupEntry {
232        name,
233        visibilities,
234        flags,
235        weights,
236        uvw,
237        units,
238        uvw_frame: _,
239    } = group;
240    name.capacity()
241        + visibilities.capacity()
242        + flags.capacity()
243        + weights.capacity()
244        + uvw.capacity()
245        + units_bytes(units)
246}
247
248pub(crate) fn units_bytes(units: &VisibilityUnits) -> usize {
249    match units {
250        VisibilityUnits::Declared(units) => units.capacity(),
251        _ => 0,
252    }
