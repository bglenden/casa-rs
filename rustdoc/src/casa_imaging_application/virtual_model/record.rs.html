2//! The serialized `FTMachine` a virtual model entry holds, field for field
3//! as CASA 6.7.6 writes it: `refim::GridFT::toRecord` (over
4//! `FTMachine::toRecord` and `storeMovingSourceState`) and
5//! `MultiTermFTNew::toRecord`, measured in the records `tclean` left in
6//! `refim_twochan.ms` and `refim_point.ms`.
7//!
8//! CASA reads the record back with `GridFT(const RecordInterface&)`, which
9//! takes the grid from the image `diskimage` names and recomputes its
10//! sizes, scales and channel and polarization maps from that image and
11//! each visibility buffer (`GridFT::init`, `FTMachine::initMaps`). What it
12//! keeps is the phase centre (`mimage_rec`), the observatory
13//! (`mlocation_rec`) and the image's coordinates; the rest is the state the
14//! machine had after degridding the first buffer, which casa-rs writes as
15//! CASA would for its selection's first buffer.
16

22/// What CASA's machine knew of the first visibility buffer it degridded:
23/// its field, the selected channels' LSRK frequencies, their image
24/// channels, its correlations and their grid polarizations.
25#[derive(Clone, Debug, PartialEq)]
26pub(crate) struct FirstBuffer {
27    pub(crate) field: i32,
28    pub(crate) frequencies_lsrk_hz: Vec<f64>,
29    pub(crate) channel_map: Vec<i32>,
30    /// The correlations' `CORR_TYPE` codes.
31    pub(crate) correlations: Vec<i32>,
32    pub(crate) polarization_map: Vec<i32>,
33}

35/// One `GridFT` of a virtual model: where its grid image is and what CASA
36/// recorded beside it.
37pub(crate) struct GridFt<'a> {
38    pub(crate) disk_image: &'a Path,
39    /// The padded grid `[nx, ny]`, the image's shape.
40    pub(crate) grid: [usize; 2],
41    /// Grid polarizations and image channels.
42    pub(crate) polarizations: usize,
43    pub(crate) image_frequencies_hz: &'a [f64],
44    /// Signed radians per pixel.
45    pub(crate) increment_rad: [f64; 2],
46    /// The image's phase centre and its frame (`J2000`, ...).
47    pub(crate) phase_centre_rad: [f64; 2],
48    pub(crate) direction_frame: &'a str,
49    /// The observatory, ITRF longitude, latitude (geocentric) and radius.
50    pub(crate) observatory: [f64; 3],
51    /// The image's spectral coordinate record.
52    pub(crate) spectral: RecordValue,
53    pub(crate) buffer: &'a FirstBuffer,
54    /// `InterpolateArray1D`'s method: nearest 0, linear 1, cubic 2.
55    pub(crate) interpolation: i32,
56    /// Whether the machine converted its buffers' frequencies itself: a
57    /// Taylor term's machine, which `MultiTermFTNew` drives, records not.
58    pub(crate) frequency_frame_valid: bool,
59}
60
61impl GridFt<'_> {
62    /// The record, `GridFT::toRecord` with its image on disk: CASA writes
63    /// padding 1 since the image is the padded grid.
64    pub(crate) fn record(&self) -> RecordValue {
65        let [nx, ny] = self.grid;
66        let channels = self.image_frequencies_hz.len();
67        let buffer = self.buffer;
68        let used = buffer
69            .polarization_map
70            .iter()
71            .filter(|&&pol| pol >= 0)
72            .count();
73        let mut record = RecordValue::default();
74        let mut define = |name: &str, value: Value| record.push(RecordField::new(name, value));
75        define("name", text("GridFT"));
76        define("diskimage", text(&self.disk_image.display().to_string()));
77        define("nantenna", Value::Scalar(ScalarValue::UInt32(0)));
78        define("distance", double(0.0));
79        define("lastfieldid", int(buffer.field));
80        define("lastmsid", int(0));
81        define("tangentspecified", boolean(false));
82        define(
83            "mtangent_rec",
84            direction([0.0, std::f64::consts::FRAC_PI_2], "J2000"),
85        );
86        define(
87            "mimage_rec",
88            direction(self.phase_centre_rad, self.direction_frame),
89        );
90        define("nx", int(nx as i32));
91        define("ny", int(ny as i32));
92        define("npol", int(self.polarizations as i32));
93        define("nchan", int(channels as i32));
94        define("nvischan", int(buffer.frequencies_lsrk_hz.len() as i32));
95        define("nvispol", int(buffer.correlations.len() as i32));
96        define("douvwrotation", boolean(true));
97        define("freqinterpmethod", int(self.interpolation));
98        define("spwchanselflag", ints(&[0, 0, 0], &[]));
99        define("freqframevalid", boolean(self.frequency_frame_valid));
100        define("imagefreq", doubles(self.image_frequencies_hz));
101        define("lsrfreq", doubles(&buffer.frequencies_lsrk_hz));
102        // A continuum machine interpolates onto the buffer's own
103        // frequencies; a cube's onto its image channels.
104        define(
105            "interpvisfreq",
106            doubles(if channels == 1 {
107                &buffer.frequencies_lsrk_hz
108            } else {
109                self.image_frequencies_hz
110            }),
111        );
112        define(
113            "chanmap",
114            ints(&[buffer.channel_map.len()], &buffer.channel_map),
115        );
116        define(
117            "polmap",
118            ints(&[buffer.polarization_map.len()], &buffer.polarization_map),
119        );
120        define("nvischanmulti", ints(&[0], &[]));
121        define("mlocation_rec", position(self.observatory));
122        define("spectralcoord", Value::Record(self.spectral.clone()));
123        define(
124            "movingdir_rec",
125            direction([0.0, std::f64::consts::FRAC_PI_2], "J2000"),
126        );
127        define("fixmovingsource", boolean(false));
128        define(
129            "firstmovingdir_rec",
130            direction([0.0, std::f64::consts::FRAC_PI_2], "J2000"),
131        );
132        define("pointingdircol", text("DIRECTION"));
133        define("usedoublegrid", boolean(true));
134        define(
135            "cfstokes",
136            ints(&[buffer.correlations.len()], &buffer.correlations),
137        );
138        define("polinuse", int(used as i32));
139        define("tovis", boolean(true));
140        define(
141            "sumweight",
142            Value::Array(ArrayValue::Float64(ArrayD::zeros(IxDyn(&[
143                self.polarizations,
144                channels,
145            ])))),
146        );
147        define("numthreads", int(-1));
148        define("phasecentertime", double(-1.0));
149        define(
150            "cachesize",
151            Value::Scalar(ScalarValue::Int64(1_000_000_000)),
152        );
153        define("tilesize", int(16));
154        define("convtype", text("SF"));
155        define(
156            "uvscale",
157            doubles(&[
158                nx as f64 * self.increment_rad[0],
159                ny as f64 * self.increment_rad[1],
160            ]),
161        );
162        define("uvoffset", doubles(&[(nx / 2) as f64, (ny / 2) as f64]));
163        define("istiled", boolean(false));
164        define("maxabsdata", Value::Scalar(ScalarValue::Float32(0.0)));
165        define("centerloc", ints(&[4], &[0, 0, 0, 0]));
166        define("offsetloc", ints(&[4], &[0, 0, 0, 0]));
167        define("padding", Value::Scalar(ScalarValue::Float32(1.0)));
168        define("usezero", boolean(false));
169        define("nopadding", boolean(false));
170        record
171    }
172}

174/// `MultiTermFTNew::toRecord` of a prediction machine: its Taylor terms'
175/// `GridFT` records, each naming its own grid image.
176pub(crate) fn multi_term(terms: Vec<RecordValue>, reference_hz: f64) -> RecordValue {
177    let count = terms.len();
178    let mut record = RecordValue::new(vec![
179        RecordField::new("name", text("MultiTermFTNew")),
180        RecordField::new("nterms", Value::Scalar(ScalarValue::UInt32(count as u32))),
181        RecordField::new("reffreq", double(reference_hz)),
182        RecordField::new("machinename", text("MultiTermFTNew")),
183        RecordField::new(
184            "psfnterms",
185            Value::Scalar(ScalarValue::UInt32(count as u32)),
186        ),
187        RecordField::new("numfts", int(count as i32)),
188    ]);
189    for (term, machine) in terms.into_iter().enumerate() {
190        record.push(RecordField::new(
191            format!("subftm_{term}"),
192            Value::Record(machine),
193        ));
194    }
195    record
196}

198/// The fields of CASA's spectral coordinate record, in the order
199/// `SpectralCoordinate::save` writes them; `conversion` follows.
200const SPECTRAL_FIELDS: [&str; 12] = [
201    "version",
202    "system",
203    "restfreq",
204    "restfreqs",
205    "velType",
206    "nativeType",
207    "velUnit",
208    "waveUnit",
209    "formatUnit",
210    "wcs",
211    "unit",
212    "name",
213];
214
215/// A spectral coordinate's record as CASA's `SpectralCoordinate::save`
216/// leaves it in a machine record: its fields in CASA's order, a linear
217/// axis's WCS in `crval, crpix, cdelt, pc, ctype` with the type as wcslib's
218/// nine-character `FREQ`, and, for a coordinate with no conversion layer,
219/// CASA's default one (the default J2000 direction, ITRF origin and LAST
220/// epoch, to the coordinate's own system).
221pub(crate) fn casa_spectral(record: &RecordValue) -> RecordValue {
222    let mut out = RecordValue::default();
223    for name in SPECTRAL_FIELDS {
224        let Some(value) = record.get(name) else {
225            continue;
226        };
227        let value = match (name, value) {
228            ("wcs", Value::Record(wcs)) => Value::Record(casa_wcs(wcs)),
229            _ => value.clone(),
230        };
231        out.push(RecordField::new(name, value));
232    }
233    for field in record.fields() {
234        if !SPECTRAL_FIELDS.contains(&field.name.as_str()) && field.name != "conversion" {
235            out.push(field.clone());
236        }
237    }
238    let conversion = match record.get("conversion") {
239        Some(Value::Record(conversion)) => {
240            let mut ordered = RecordValue::default();
241            for name in ["direction", "position", "epoch", "system"] {
242                if let Some(value) = conversion.get(name) {
243                    ordered.push(RecordField::new(name, value.clone()));
244                }
245            }
246            ordered
247        }
248        _ => {
249            let system = record
250                .get("system")
251                .cloned()
252                .unwrap_or_else(|| text("LSRK"));
253            RecordValue::new(vec![
254                RecordField::new(
255                    "direction",
256                    direction([0.0, std::f64::consts::FRAC_PI_2], "J2000"),
257                ),
258                RecordField::new("position", position([0.0, 0.0, 0.0])),
259                RecordField::new(
260                    "epoch",
261                    Value::Record(RecordValue::new(vec![
262                        RecordField::new("type", text("epoch")),
263                        RecordField::new("refer", text("LAST")),
264                        RecordField::new("m0", quantity(0.0, "d", false)),
265                    ])),
266                ),
267                RecordField::new("system", system),
268            ])
269        }
270    };
271    out.push(RecordField::new("conversion", Value::Record(conversion)));
272    out
273}

275/// A linear spectral axis's WCS in CASA's order, its `FREQ` type padded
276/// with NULs to wcslib's nine characters as CASA records it.
277fn casa_wcs(wcs: &RecordValue) -> RecordValue {
278    let mut out = RecordValue::default();
279    for name in ["crval", "crpix", "cdelt", "pc", "ctype"] {
280        if let Some(value) = wcs.get(name) {
281            let value = match value {
282                Value::Scalar(ScalarValue::String(kind)) if kind == "FREQ" => {
283                    text("FREQ\0\0\0\0\0")
284                }
285                value => value.clone(),
286            };
287            out.push(RecordField::new(name, value));
288        }
289    }
290    for field in wcs.fields() {
291        if out.get(&field.name).is_none() {
292            out.push(field.clone());
293        }
294    }
295    out
296}

298fn text(value: &str) -> Value {
299    Value::Scalar(ScalarValue::String(value.to_string()))
300}
301
302fn int(value: i32) -> Value {
303    Value::Scalar(ScalarValue::Int32(value))
304}
305
306fn double(value: f64) -> Value {
307    Value::Scalar(ScalarValue::Float64(value))

310fn boolean(value: bool) -> Value {
311    Value::Scalar(ScalarValue::Bool(value))

314fn doubles(values: &[f64]) -> Value {
315    Value::Array(ArrayValue::Float64(
316        ArrayD::from_shape_vec(IxDyn(&[values.len()]), values.to_vec()).expect("one axis"),
317    ))
318}

320fn ints(shape: &[usize], values: &[i32]) -> Value {
321    Value::Array(ArrayValue::Int32(
322        ArrayD::from_shape_vec(IxDyn(shape), values.to_vec()).expect("the shape holds the values"),
323    ))
324}

326/// A quantity `{value, unit}` or, as CASA writes the first axis of a
327/// measure, `{unit, value}`.
328fn quantity(value: f64, unit: &str, unit_first: bool) -> Value {
329    let value = RecordField::new("value", double(value));
330    let unit = RecordField::new("unit", text(unit));
331    Value::Record(RecordValue::new(if unit_first {
332        vec![unit, value]
333    } else {
334        vec![value, unit]
335    }))
336}

338/// A direction measure as `MeasureHolder::toRecord` leaves it in CASA's
339/// machine records: `type`, `refer`, then `m1` and `m0`, the latitude's
340/// value before its unit and the longitude's unit before its value. The
341/// longitude is CASA's `MVDirection` angle, the `atan2` of its direction
342/// cosines, in (−π, π].
343fn direction([longitude, latitude]: [f64; 2], frame: &str) -> Value {
344    let cosine = latitude.cos();
345    let longitude = (cosine * longitude.sin()).atan2(cosine * longitude.cos());
346    Value::Record(RecordValue::new(vec![
347        RecordField::new("type", text("direction")),
348        RecordField::new("refer", text(frame)),
349        RecordField::new("m1", quantity(latitude, "rad", false)),
350        RecordField::new("m0", quantity(longitude, "rad", true)),
351    ]))
352}

354/// An ITRF position measure, as CASA leaves it: `m2` (the radius), `m1`
355/// and `m0`.
356fn position([longitude, latitude, radius]: [f64; 3]) -> Value {
357    Value::Record(RecordValue::new(vec![
358        RecordField::new("type", text("position")),
359        RecordField::new("refer", text("ITRF")),
360        RecordField::new("m2", quantity(radius, "m", false)),
361        RecordField::new("m1", quantity(latitude, "rad", true)),
362        RecordField::new("m0", quantity(longitude, "rad", true)),
363    ]))
364}
