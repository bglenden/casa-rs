31/// The compiled spectral axis and the source channels it reads.
32pub(super) struct PreparedSpectralAxis {
33    pub(super) selected_source_channels: WindowChannels,
34    pub(super) source_frame: FrequencyFrame,
35    pub(super) output_frequency_reference: FrequencyRef,
36    pub(super) output_frame: FrequencyFrame,
37    pub(super) anchor: SpectralFrameAnchor,
38    pub(super) wcs: SpectralWcs,
39    pub(super) rest_frequency: RestFrequency,
40    /// The spectral axis the image header records: the gridded axis, but
41    /// for an MFS image's frame and rest frequencies.
42    pub(super) image: ImageSpectralCoordinate,
43    pub(super) doppler: DopplerConvention,
44    pub(super) sampling: SpectralSamplingLaw,
45    pub(super) basis: ReconstructionBasis,
46    pub(super) output_channels: usize,
47    /// Centre of the first output channel.
48    pub(super) reference_frequency_hz: f64,
49    /// Channel increment; a continuum plane's full band.

56        SpectralCoordinateSpec::new(
57            self.source_frame,
58            self.output_frame,
59            self.anchor,
60            self.wcs.clone(),
61            self.rest_frequency,

98/// The request's spectral axis over the surveyed selection.
99pub(super) fn prepare_axis(
100    request: &ImagingRequest,
101    (msv2, catalogue): (Option<&MeasurementSet>, &Catalogue),
102    survey: &Survey,
103    frame: &FrameContext<'_>,
104    moving_rest_frame: Option<&MeasFrame>,
105) -> Result<PreparedSpectralAxis, PrepareError> {
106    let source_reference = survey.source_frequency_reference;
107    let law = match request.specmode {
108        SpecMode::Mfs => continuum_law(request, (msv2, catalogue), survey)?,
109        SpecMode::Cube | SpecMode::Cubedata => cube_law(request, catalogue, survey, frame)?,
110        SpecMode::Cubesource => {
111            let frame_with_velocity = moving_rest_frame
112                .expect("prepare resolves the moving rest frame of every source-frame cube");
113            source_frame_cube_law(request, catalogue, survey, frame, frame_with_velocity)?
114        }
115    };
116    let source_frame = imaging_frequency_frame(source_reference)?;
117    let output_frame = imaging_frequency_frame(law.output_frequency_reference)?;
118    Ok(PreparedSpectralAxis {
119        selected_source_channels: law.selected_source_channels,
120        source_frame,
121        output_frequency_reference: law.output_frequency_reference,
122        output_frame,
123        anchor: spectral_frame_anchor(source_frame, output_frame, request.specmode, frame)?,
124        wcs: SpectralWcs::Linear {
125            channels: law.output_channels,
126            reference_pixel: 0.0,
127            reference_frequency_hz: law.reference_frequency_hz,
128            increment_hz: law.increment_hz,
129        },
130        rest_frequency: law.rest_frequency,
131        // CASA lists tclean's further rest frequencies after the image's.
132        image: ImageSpectralCoordinate {
133            additional_rest_frequencies_hz: request.restfreq.iter().skip(1).copied().collect(),
134            ..law.image
135        },
136        doppler: law.doppler,
137        sampling: law.sampling,
138        basis: law.basis,
139        output_channels: law.output_channels,
140        reference_frequency_hz: law.reference_frequency_hz,
141        increment_hz: law.increment_hz,
142    })
143}

145/// One plane over the selected band's envelope, gridded in LSRK as CASA's
146/// `FTMachine::matchChannel` grids it whatever the image's `outframe`.
147/// Its header records the envelope in `outframe`
148/// (`buildCoordinateSystem`'s `MSUtil::getFreqRangeInSpw`) and `restfreq`'s
149/// first rest frequency, else the first visibility buffer's source line
150/// ([`continuum_rest_frequency_hz`] on an MSv2, whose `SOURCE` and
151/// `DOPPLER` tables CASA reads; the catalogue's first line on an MSv4),
152/// else the recorded band's midpoint (`getImFreq`).
153fn continuum_law(
154    request: &ImagingRequest,
155    (msv2, catalogue): (Option<&MeasurementSet>, &Catalogue),
156    survey: &Survey,
157) -> Result<AxisLaw, PrepareError> {
158    let (channels, envelope, recorded) = survey
159        .continuum
160        .as_ref()
161        .expect("the survey reduces the spectral envelope of every continuum run");
162    let [lower_hz, upper_hz] = envelope.edges_hz();
163    let [recorded_lower_hz, recorded_upper_hz] = recorded.edges_hz();
164    // `getImFreq`: a given reffreq is the reference at pixel 0, with the
165    // band widened by twice its offset from the band's centre (CAS-9518).
166    let reference = |midpoint_hz: f64, width_hz: f64| match request.reffreq {
167        Some(reference_hz) => (
168            reference_hz,
169            width_hz + 2.0 * (reference_hz - midpoint_hz).abs(),
170        ),
171        None => (midpoint_hz, width_hz),
172    };
173    if request.reffreq.is_some() && request.outframe != FrequencyRef::LSRK {
174        return Err(PrepareError::ReferenceFrequencyOutsideLsrk {
175            outframe: request.outframe,
176        });
177    }
178    let (reference_frequency_hz, increment_hz) =
179        reference(envelope.midpoint_hz(), upper_hz - lower_hz);
180    let (image_reference_hz, image_increment_hz) = reference(
181        recorded.midpoint_hz(),
182        recorded_upper_hz - recorded_lower_hz,
183    );
184    let field_id = survey.first_visibility_field_id;
185    let spw_id = survey.spectral_windows[0].spw_id;
186    let image_rest_frequency_hz = match (request.restfreq.first(), msv2) {
187        (Some(&hertz), _) => hertz,
188        (None, Some(ms)) => {
189            continuum_rest_frequency_hz(ms, field_id, spw_id, recorded.midpoint_hz())?
190        }
191        (None, None) => catalogue
192            .source_records(
193                FieldId(u32::try_from(field_id).expect("a FIELD row is an u32 id")),
194                WindowId(u32::try_from(spw_id).expect("a window id fits u32")),
195            )
196            .first()
197            .and_then(|record| record.rest_frequencies.as_ref()?.first())
198            .map(casa_types::measures::frequency::MFrequency::hz)
199            .filter(|value| *value != 0.0)
200            .unwrap_or(recorded.midpoint_hz()),
201    };
202    Ok(AxisLaw {
203        selected_source_channels: channels.clone(),
204        output_frequency_reference: FrequencyRef::LSRK,
205        reference_frequency_hz,
206        increment_hz,
207        output_channels: 1,
208        rest_frequency: RestFrequency::NotApplicable,
209        image: ImageSpectralCoordinate {
210            frequency_reference: request.outframe,
211            reference_frequency_hz: image_reference_hz,
212            increment_hz: image_increment_hz,
213            rest_frequency_hz: image_rest_frequency_hz,
214            additional_rest_frequencies_hz: Vec::new(),
215            undefined_frame: false,
216        },
217        doppler: DopplerConvention::NotApplicable,
218        sampling: SpectralSamplingLaw::IDENTITY,
219        basis: ReconstructionBasis::Constant,
220    })
221}

223/// CASA `SynthesisParamsImage::buildCoordinateSystemCore` asks
224/// `MSDopplerUtil::dopplerInfo` for the first selected SPW and field. Its
225/// first rest frequency wins; an absent or zero value uses the MFS band's
226/// midpoint. This header label does not change the continuum/Taylor basis.
227fn continuum_rest_frequency_hz(
228    ms: &MeasurementSet,
229    field_id: usize,
230    spw_id: usize,
231    band_midpoint_hz: f64,
232) -> Result<f64, PrepareError> {
233    let source_id = ms.field()?.source_id(field_id)?;
234    let window = ms
235        .subtable(SubtableId::SpectralWindow)
236        .expect("required SPW table");
237    let doppler_id = if window
238        .schema()
239        .is_some_and(|schema| schema.contains_column("DOPPLER_ID"))
240    {
241        let scalar = window
242            .cell_accessor(spw_id, "DOPPLER_ID")
243            .map_err(casa_ms::MsError::from)?
244            .scalar()
245            .map_err(casa_ms::MsError::from)?;
246        match scalar {
247            casa_types::ScalarValue::Int32(value) => *value,
248            _ => {
249                return Err(
250                    casa_ms::MsError::InvalidInput("DOPPLER_ID must be Int32".into()).into(),
251                );
252            }
253        }
254    } else {
255        -1
256    };
257    if doppler_id >= 0 && ms.subtable(SubtableId::Doppler).is_some() {
258        return Err(casa_ms::MsError::InvalidInput(
259            "MFS rest-frequency metadata with a DOPPLER table is not supported".into(),
260        )
261        .into());
262    }
263    if ms.subtable(SubtableId::Source).is_none() {
264        return Ok(band_midpoint_hz);
265    }
266    let resolved = continuum_source_line(ms, source_id, spw_id)?;
267    Ok(resolved
268        .filter(|value| *value != 0.0)
269        .unwrap_or(band_midpoint_hz))
270}

272/// The first SOURCE line within a matching key group; -1 is a wildcard.
273/// CASA binary-searches sorted unique keys using its wildcard comparator.
274/// With multiple key groups, wildcards can select or hide a matching group;
275/// refuse that unresolved case instead of merging physical row order.
276fn continuum_source_line(
277    ms: &MeasurementSet,
278    source_id: i32,
279    spw_id: usize,
280) -> Result<Option<f64>, PrepareError> {
281    let source = ms.source()?;
282    let spw_id = i32::try_from(spw_id).expect("SPW ids are stored i32 values");
283    let mut first_key = None;
284    let mut multiple_keys = false;
285    let mut wildcard = false;
286    let mut matching_key = None;
287    let mut resolved = None;
288    for row in 0..source.row_count() {
289        let row_source = source.i32(row, "SOURCE_ID")?;
290        let row_spw = source.i32(row, "SPECTRAL_WINDOW_ID")?;
291        let key = (row_source, row_spw);
292        multiple_keys |= first_key.is_some_and(|prior| prior != key);
293        first_key.get_or_insert(key);
294        wildcard |= row_source == -1 || row_spw == -1;
295        if (row_source != source_id && row_source != -1) || (row_spw != spw_id && row_spw != -1) {
296            continue;
297        }
298        if matching_key.is_some_and(|prior| prior != key) {
299            return Err(casa_ms::MsError::InvalidInput(
300                "overlapping SOURCE key groups for MFS rest-frequency metadata are not supported"
301                    .into(),
302            )
303            .into());
304        }
305        matching_key = Some(key);
306        if resolved.is_some() {
307            continue;
308        }
309        let Some(ArrayValue::Float64(values)) = source.optional_array(row, "REST_FREQUENCY")?
310        else {
311            continue;
312        };
313        if values.is_empty() {
314            continue;
315        }
316        resolved = values.first().copied();
317    }
318    if wildcard && multiple_keys {
319        return Err(casa_ms::MsError::InvalidInput(
320            "multiple SOURCE key groups with wildcard ids for MFS rest-frequency metadata are not supported".into(),
321        ).into());
322    }
323    Ok(resolved)
324}

326/// Whether CASA reads a `cube`'s data by the image's band rather than by
327/// the channels `spw` names (`SynthesisImagerVi2::tuneSelectData`): for a
328/// window of 30 or more channels, or when the model is saved to
329/// `MODEL_DATA`, `SynthesisImager::tuneSelectData` replaces the channel
330/// selection with the LSRK frequencies from pixel −1.5 to `nchan + 0.5` of
331/// the image axis, so every channel whose samples reach a plane is read;
332/// a `cubedata` or `cubesource` cube keeps the selection. The channels the
333/// axis reaches are those [`cube_setup`]'s support finds. casa-rs's
334/// visibility-domain continuum fit (`fitspw`, which CASA's tclean lacks)
335/// keeps the named channels as its output channels.
336fn tunes_selection(request: &ImagingRequest, window: &SourceSpectralWindow) -> bool {
337    request.specmode == SpecMode::Cube
338        && request.fitspw.is_none()
339        && (window.frequencies_hz.len() >= 30 || request.savemodel == SaveModel::ModelColumn)
340}

342/// A cube in its output frame (`cube`) or the data's (`cubedata`).
343fn cube_law(
344    request: &ImagingRequest,
345    catalogue: &Catalogue,
346    survey: &Survey,
347    frame: &FrameContext<'_>,
348) -> Result<AxisLaw, PrepareError> {
349    let window = one_window(&survey.spectral_windows, "native cube imaging")?;
350    let axis = cube_axis(request, survey, window)?;
351    let output_channels = request.channel_count.unwrap_or(window.frequencies_hz.len());
352    let (setup, support) = cube_setup(survey, window, output_channels, &axis, frame)?;
353    let mut selected = support.indices;
354    let explicit = survey
355        .spw_channels
356        .get(&window.spw_id)
357        .map(|channels| channels.iter().copied().collect::<BTreeSet<_>>());
358    if let Some(explicit) = &explicit
359        && !tunes_selection(request, window)
360    {
361        selected.retain(|channel| explicit.contains(channel));
362    }
363    if selected.is_empty() {
364        return Err(PrepareError::NoCommonCubeChannels);
365    }
366    let (reference_frequency_hz, increment_hz) = first_and_increment(&setup, output_channels, 1.0)?;
367    let source_rest_hz =
368        source_rest_frequency(catalogue, frame.field_id, &survey.spectral_windows)?;
369    let (resolved_rest_hz, image_rest_frequency_hz) = cube_rest_frequency_hz(
370        request.restfreq.first().copied(),
371        source_rest_hz,
372        window,
373        explicit.as_ref(),
374    );
375    let (rest_frequency, doppler) = match resolved_rest_hz {
376        None => (
377            RestFrequency::NotApplicable,
378            DopplerConvention::NotApplicable,
379        ),
380        Some(hertz) => (
381            RestFrequency::Line { hertz },
382            doppler_convention(request.veltype)?,
383        ),
384    };
385    Ok(AxisLaw {
386        selected_source_channels: BTreeMap::from([(window.spw_id, selected)]),
387        output_frequency_reference: setup.output_freq_ref,
388        reference_frequency_hz,
389        increment_hz,
390        output_channels,
391        rest_frequency,
392        image: ImageSpectralCoordinate {
393            frequency_reference: setup.output_freq_ref,
394            reference_frequency_hz,
395            increment_hz,
396            rest_frequency_hz: image_rest_frequency_hz,
397            additional_rest_frequencies_hz: Vec::new(),
398            undefined_frame: request.specmode == SpecMode::Cubedata
399                && setup.output_freq_ref != FrequencyRef::REST,
400        },
401        doppler,
402        sampling: sampling_law(setup.interpolation),
403        basis: ReconstructionBasis::ChannelLocal {
404            channels: output_channels,
405        },
406    })
407}

409/// A cube in the rest frame of a moving source: the data-frame axis
410/// scaled into the source's rest frame (`cubesource`).
411fn source_frame_cube_law(
412    request: &ImagingRequest,
413    catalogue: &Catalogue,
414    survey: &Survey,
415    frame: &FrameContext<'_>,
416    moving_rest_frame: &MeasFrame,
417) -> Result<AxisLaw, PrepareError> {
418    let window = one_window(&survey.spectral_windows, "source-frame cube imaging")?;
419    let source_reference = survey.source_frequency_reference;
420    let mut axis = cube_axis(request, survey, window)?;
421    axis.specmode = CubeSpecMode::Cubedata;
422    axis.outframe = source_reference;
423    let output_channels = request.channel_count.unwrap_or(window.frequencies_hz.len());
424    let (setup, support) = cube_setup(survey, window, output_channels, &axis, frame)?;
425    let factor = casa_ms::convert_frequency_to_frame_with_frame(
426        source_reference,
427        FrequencyRef::REST,
428        1.0,
429        Some(moving_rest_frame),
430    )?;
431    let (reference_frequency_hz, increment_hz) =
432        first_and_increment(&setup, output_channels, factor)?;
433    let rest_frequency_hz = request
434        .restfreq
435        .first()
436        .copied()
437        .or(source_rest_frequency(
438            catalogue,
439            frame.field_id,
440            &survey.spectral_windows,
441        )?)
442        .ok_or(PrepareError::RestFrequencyMissing)?;
443    Ok(AxisLaw {
444        selected_source_channels: BTreeMap::from([(window.spw_id, support.indices)]),
445        output_frequency_reference: FrequencyRef::REST,
446        reference_frequency_hz,
447        increment_hz,
448        output_channels,
449        rest_frequency: RestFrequency::Line {
450            hertz: rest_frequency_hz,
451        },
452        image: ImageSpectralCoordinate {
453            frequency_reference: FrequencyRef::REST,
454            reference_frequency_hz,
455            increment_hz,
456            rest_frequency_hz,
457            additional_rest_frequencies_hz: Vec::new(),
458            undefined_frame: false,
459        },
460        doppler: doppler_convention(request.veltype)?,
461        sampling: sampling_law(request.interpolation),
462        basis: ReconstructionBasis::ChannelLocal {
463            channels: output_channels,
464        },
465    })
466}

468/// CASA's cube axis of the request over `window` of `survey`: `start`
469/// (else `channel_start`) and `width` in the request's velocity
470/// convention, and the window's channels its `spw` names.
471fn cube_axis(
472    request: &ImagingRequest,
473    survey: &Survey,
474    window: &SourceSpectralWindow,
475) -> Result<CubeAxisConfig, PrepareError> {
476    let start = match (&request.start, request.channel_start) {
477        (Some(start), _) => Some(CubeAxisValue::parse(start, request.veltype)?),
478        (None, Some(channel)) => Some(CubeAxisValue::Channel(
479            i32::try_from(channel).map_err(|_| PrepareError::CubeStartChannel { channel })?,
480        )),
481        (None, None) => None,
482    };
483    let width = request
484        .width
485        .as_deref()
486        .map(|width| CubeAxisValue::parse(width, request.veltype))
487        .transpose()?;
488    Ok(CubeAxisConfig {
489        specmode: if request.specmode == SpecMode::Cubedata {
490            CubeSpecMode::Cubedata
491        } else {
492            CubeSpecMode::Cube
493        },
494        outframe: request.outframe,
495        veltype: request.veltype,
496        interpolation: request.interpolation,
497        rest_frequency_hz: request.restfreq.first().copied(),
498        start,
499        width,
500        selected_channels: survey.spw_channels.get(&window.spw_id).cloned(),
501    })
502}

525/// The first output channel's centre and the channel increment, scaled by
526/// `factor`; a one-channel cube's increment is its width.
527fn first_and_increment(
528    setup: &CubeSpectralSetup,
529    output_channels: usize,
530    factor: f64,
531) -> Result<(f64, f64), PrepareError> {
532    let first_hz = setup.output_channel_frequencies_hz[0];
533    let increment_hz = if output_channels > 1 {
534        setup.output_channel_frequencies_hz[1] - first_hz
535    } else {
536        setup.output_channel_widths_hz[0]
537    } * factor;
538    if !increment_hz.is_finite() || increment_hz == 0.0 {
539        return Err(PrepareError::CubeIncrement { increment_hz });
540    }
541    Ok((first_hz * factor, increment_hz))
542}

544/// The model's Doppler convention of CASA `veltype`.
545fn doppler_convention(veltype: DopplerRef) -> Result<DopplerConvention, PrepareError> {
546    match veltype {
547        DopplerRef::RADIO => Ok(DopplerConvention::Radio),
548        DopplerRef::Z => Ok(DopplerConvention::Optical),
549        DopplerRef::BETA => Ok(DopplerConvention::Relativistic),
550        DopplerRef::RATIO | DopplerRef::GAMMA => Err(PrepareError::UnsupportedDoppler {
551            convention: veltype,
552        }),
553    }
554}

556const fn sampling_law(interpolation: CubeInterpolation) -> SpectralSamplingLaw {
557    match interpolation {
558        CubeInterpolation::Nearest => SpectralSamplingLaw::NEAREST,
559        CubeInterpolation::Linear => SpectralSamplingLaw::LINEAR,
560        CubeInterpolation::Cubic => SpectralSamplingLaw::CUBIC,
561    }
562}

564/// The image's epoch, reference direction and observatory, where the
565/// rows' frame and the output frame differ, or where a `cube` is gridded
566/// through LSRK ([`FrequencyFrame::gridded_through_lsrk`]): CASA grids a
567/// `cube` (not `cubedata`, whose rows keep their frequencies) in LSRK
568/// whatever its `outframe`, so a TOPO cube of TOPO rows is fixed in LSRK at
569/// the first time and each row is Doppler tracked from it.
570fn spectral_frame_anchor(
571    source_frame: FrequencyFrame,
572    output_frame: FrequencyFrame,
573    specmode: SpecMode,
574    frame: &FrameContext<'_>,
575) -> Result<SpectralFrameAnchor, PrepareError> {
576    let tracked = specmode == SpecMode::Cube && output_frame.gridded_through_lsrk();
577    if source_frame == output_frame && !tracked {
578        return Ok(SpectralFrameAnchor::NotApplicable);
579    }
580    let [x_metres, y_metres, z_metres] = frame.engine.observatory_position().as_itrf();
581    Ok(SpectralFrameAnchor::Conversion {
582        // As `MEpoch::from_mjd_seconds` reads a row's time, so a row at the
583        // first time is at the anchor's epoch exactly.
584        epoch: Epoch::new(
585            frame.anchor_time_mjd_seconds * (1.0 / 86_400.0),
586            imaging_time_scale(frame.engine.time_reference())?,
587        ),
588        direction: frame.direction.reference_direction(),
589        observatory_position: ItrfPosition::new(x_metres, y_metres, z_metres),
590    })
591}

593fn imaging_frequency_frame(reference: FrequencyRef) -> Result<FrequencyFrame, PrepareError> {
594    match reference {
595        FrequencyRef::REST => Ok(FrequencyFrame::Rest),
596        FrequencyRef::TOPO => Ok(FrequencyFrame::Topocentric),
597        FrequencyRef::BARY => Ok(FrequencyFrame::Barycentric),
598        FrequencyRef::LSRK => Ok(FrequencyFrame::Lsrk),
599        _ => Err(PrepareError::UnsupportedFrequencyFrame { reference }),
600    }
601}
602
603pub(super) fn imaging_time_scale(reference: EpochRef) -> Result<TimeScale, PrepareError> {
604    match reference {
605        EpochRef::UTC => Ok(TimeScale::Utc),
606        EpochRef::TAI => Ok(TimeScale::Tai),
607        EpochRef::TT => Ok(TimeScale::Tt),
608        EpochRef::TDB => Ok(TimeScale::Tdb),
609        _ => Err(PrepareError::UnsupportedEpochReference { reference }),
610    }
611}

613/// The rest frequency the field's source states over the selected windows
614/// (MSv2 `SOURCE::REST_FREQUENCY`, MSv4 `LINE_REST_FREQUENCY`), if any: the
615/// first line of each record for a selected window or for every window
616/// ([`Catalogue::source_records`]), which must agree.
617fn source_rest_frequency(
618    catalogue: &Catalogue,
619    field_id: usize,
620    spectral_windows: &[SourceSpectralWindow],
621) -> Result<Option<f64>, PrepareError> {
622    let field = FieldId(u32::try_from(field_id).expect("a FIELD row is an u32 id"));
623    let mut resolved = None;
624    for window in spectral_windows {
625        let window = WindowId(u32::try_from(window.spw_id).expect("a window id fits u32"));
626        for record in catalogue.source_records(field, window) {
627            let Some(value) = record
628                .rest_frequencies
629                .as_ref()
630                .and_then(|lines| lines.first())
631                .map(casa_types::measures::frequency::MFrequency::hz)
632                .filter(|value| *value > 0.0)
633            else {
634                continue;
635            };
636            if resolved.is_some_and(|prior: f64| prior.to_bits() != value.to_bits()) {
637                return Err(PrepareError::RestFrequencyConflict);
638            }
639            resolved = Some(value);
640        }
641    }
642    Ok(resolved)
643}

645/// CASA's precedence for a cube's rest frequency: the request's, else the
646/// source's; the header records the selected band's centre when neither
647/// exists.
648fn cube_rest_frequency_hz(
649    explicit_hz: Option<f64>,
650    source_hz: Option<f64>,
651    spectral_window: &SourceSpectralWindow,
652    selected_channels: Option<&BTreeSet<usize>>,
653) -> (Option<f64>, f64) {
654    let resolved = explicit_hz.or(source_hz);
655    let image_hz = resolved.unwrap_or_else(|| {
656        let (lower, upper) = spectral_window
657            .frequencies_hz
658            .iter()
659            .zip(&spectral_window.channel_widths_hz)
660            .enumerate()
661            .filter(|(index, _)| selected_channels.is_none_or(|channels| channels.contains(index)))
662            .map(|(_, (&centre, &width))| (centre - width.abs() / 2.0, centre + width.abs() / 2.0))
663            .fold(
664                (f64::INFINITY, f64::NEG_INFINITY),
665                |(lower, upper), (lo, hi)| (lower.min(lo), upper.max(hi)),
666            );
667        lower + (upper - lower) / 2.0
668    });
669    (resolved, image_hz)
670}
