38    /// Apply the configured chain to the attached, selected visibilities.
39    /// Returns false if a selected spectral window has no calibration.
40    ///
41    /// # Errors
42    ///
43    /// Returns an error without an attachment, or if calibration cannot be
44    /// applied. A failed apply clears the prior terms without replaying writes.
45    pub fn correct_visibilities(&mut self, mode: &str) -> Result<bool, CalibraterLifecycleError> {
46        self.apply_equation(ApplyOperation::Correct(mode))
47    }

49    /// Apply the selected chain forward to the attached physical model.
50    /// # Errors
51    /// Returns an error without an attachment or when MODEL_DATA cannot be
52    /// read, corrupted or published. A failed run clears the apply terms.
53    pub fn corrupt_model(&mut self) -> Result<bool, CalibraterLifecycleError> {
54        self.apply_equation(ApplyOperation::Corrupt)

57    fn apply_equation(
58        &mut self,
59        operation: ApplyOperation<'_>,
60    ) -> Result<bool, CalibraterLifecycleError> {
61        let using_callibrary = self.using_callibrary;
62        let legacy = self.legacy_iterator;
63        let equation = self.equation_mut()?;
64        let origin = match operation {
65            ApplyOperation::Correct(_) => "Calibrater::correct",
66            ApplyOperation::Corrupt if legacy => "Calibrater::corrupt",
67            ApplyOperation::Corrupt => "Calibrater::corrupt2",
68        };
69        let mode = match operation {
70            ApplyOperation::Correct("") => String::from("CALFLAG"),
71            ApplyOperation::Correct(mode) => mode.to_ascii_uppercase(),
72            ApplyOperation::Corrupt => String::new(),
73        };
74        let trial = mode.contains("TRIAL") || mode.contains("FLAGONLY");
75        let apply_mode = match (mode.contains("CAL"), mode.contains("FLAG")) {
76            (true, true) => ApplyMode::CalFlag,
77            (true, false) => ApplyMode::CalOnly,
78            (false, true) => ApplyMode::FlagOnly,
79            (false, false) => ApplyMode::Trial,
80        };
81        let request = ApplyPlanRequest {
82            selection: equation.attachment.selection().clone(),
83            apply_mode,
84            parang: equation.apply.parang_terms > 0,
85            calibration_tables: equation.apply.tables.clone(),
86        };
87        let correction = ToolCorrection {
88            has_library_registration: !equation.apply.library_tables.is_empty(),
89            strict: mode.contains("STRICT"),
90            trial,
91            parang_terms: equation.apply.parang_terms,
92            parang_calwt: equation.apply.parang_calwt,
93            tableless_opacity: !equation.apply.tableless_opacity.is_empty(),
94        };
95        let names = equation
96            .apply
97            .tables
98            .iter()
99            .map(|table| equation.apply.source_name(table))
100            .collect::<Vec<_>>();
101        let opacity = equation
102            .apply
103            .tableless_opacity
104            .iter()
105            .map(|term| (term.subtype, term.index))
106            .collect::<Vec<_>>();
107        let tracking = ToolActivityInput {
108            names: &names,
109            libraries: &equation.apply.library_tables,
110            opacity: &opacity,
111            previous: equation.apply.counts,
112            additive: &equation.apply.additive,
113        };
114        let result = equation.attachment.with_write(|ms, selected, lock| {
115            let applied = match operation {
116                ApplyOperation::Correct(_) => {
117                    execute_tool_apply(ms, selected, lock, &request, correction, tracking)
118                }
119                ApplyOperation::Corrupt => execute_tool_corrupt(
120                    ms,
121                    selected,
122                    lock,
123                    &request,
124                    correction,
125                    tracking,
126                    crate::execute::ModelCorruption {
127                        legacy,
128                        using_callibrary,
129                    },
130                ),
131            };
132            applied.map_err(|error| casa_ms::MsError::InvalidInput(error.to_string()).into())
133        });
134        let (all_calibrated, activity) = result.map_err(|error| {
135            equation.apply = super::ApplyTerms::default();
136            tracing::error!(%error, origin, "could not apply calibration to selected visibilities");
137            CalibraterLifecycleError::ApplyConfiguration(format!("Error in {origin}."))
138        })?;
139        equation.apply.counts = activity.single_term_counts();
140        if !legacy || matches!(operation, ApplyOperation::Correct(_)) {
141            self.last_activity = Some(activity.into_record(origin));
142        }
143        Ok(all_calibrated)
144    }

146    /// Append the tables in a calibration-library record to the attached
147    /// equation. A missing table prevents registration of the library.
148    /// Each successfully ingested table remains registered if a subsequent
149    /// readable table fails to configure.
150    ///
151    /// # Errors
152    ///
153    /// There is no attached MeasurementSet, a table is unreadable, or one of
154    /// the table records cannot configure an apply term.
155    pub fn set_callibrary(
156        &mut self,
157        library: &RecordValue,
158    ) -> Result<(), CalibraterLifecycleError> {
159        let equation = self.equation_mut()?;
160        let failure = CalibraterLifecycleError::ApplyConfiguration;
161        let cwd = std::env::current_dir().map_err(|error| failure(error.to_string()))?;
162        // CASA preflights the whole set of keys before reading any slices.
163        let tables = library
164            .fields()
165            .iter()
166            .map(|field| {
167                open_apply_table(Path::new(&field.name))
168                    .map_err(|_| failure(format!("Caltable {} does not exist.", field.name)))
169            })
170            .collect::<Result<Vec<_>, _>>()?;
171        let mut ms = None;
172        for (field, table) in library.fields().iter().zip(tables) {
173            let info = table.info();
174            if info.table_type != TABLE_INFO_TYPE {
175                return Err(non_calibration_table(&field.name, &info.table_type));
176            }
177            let slices = tool_table_record(field).map_err(failure)?;
178            let invalid_library = |reason: String| {
179                tracing::error!(
180                    table = field.name,
181                    reason,
182                    "could not register calibration library table"
183                );
184                failure("Error in Calibrater::callib2.".into())
185            };
186            if !calibration_factory_type(&info.sub_type)
187                .is_some_and(|name| !matches!(name, "BPOLY" | "GSPLINE"))
188            {
189                return Err(invalid_library(format!(
190                    "unsupported library type {}",
191                    info.sub_type
192                )));
193            }
194            let mut specs =
195                tool_library_specs(&field.name, slices, &cwd).map_err(invalid_library)?;
196            if specs.len() > 1
197                || specs.iter().any(|spec| {
198                    spec.interp == ApplyInterpolationMode::Invalid
199                        || spec.gainfield.is_some()
200                        || spec
201                            .callibrary
202                            .as_ref()
203                            .is_some_and(|options| options.selection != MsSelection::new())
204                })
205            {
206                let ms = if let Some(ms) = &ms {
207                    ms
208                } else {
209                    ms.insert(
210                        MeasurementSet::open(equation.attachment.path())
211                            .map_err(|error| invalid_library(error.to_string()))?,
212                    )
213                };
214                let request = ApplyPlanRequest {
215                    selection: equation.attachment.selection().clone(),
216                    apply_mode: ApplyMode::Trial,
217                    parang: false,
218                    calibration_tables: specs.clone(),
219                };
220                plan_with_visibility_metadata(ms, &request, None)
221                    .map_err(|error| invalid_library(error.to_string()))?;
222            }
223            if let Some(first) = specs.first() {
224                let (path, snapshot) = snapshot_table(&first.path, &table)
225                    .map_err(|error| invalid_library(error.to_string()))?;
226                for spec in &mut specs {
227                    spec.path.clone_from(&path);
228                }
229                equation.apply.table_snapshots.insert(path, snapshot);
230            }
231            let start = equation.apply.tables.len();
232            equation.apply.tables.extend(specs);
233            equation
234                .apply
235                .library_tables
236                .push(LibraryApplyRegistration {
237                    name: field.name.clone(),
238                    subtype: info.sub_type.clone(),
239                    tables: start..equation.apply.tables.len(),
240                });
241        }
242        self.using_callibrary = true;
243        Ok(())
244    }

246    /// Sorted distinct paths of configured calibration tables, as the
247    /// calibrater returns them in its `apply_tables` record field.
248    #[must_use]
249    pub fn applied_table_names(&self) -> Vec<String> {
250        let Ok(equation) = self.equation() else {
251            return Vec::new();
252        };
253        let mut names = equation
254            .apply
255            .tables
256            .iter()
257            .map(|table| equation.apply.source_name(table))
258            .chain(
259                equation
260                    .apply
261                    .library_tables
262                    .iter()
263                    .map(|library| library.name.clone()),
264            )
265            .collect::<Vec<_>>();
266        names.sort_unstable();
267        names.dedup();
268        names
269    }

271    /// Current selections and configured calibration table names.
272    ///
273    /// # Errors
274    ///
275    /// The tool has no attached dataset, or the selected MeasurementSet
276    /// metadata cannot be read.
277    pub fn configuration_record(&self) -> Result<RecordValue, CalibraterLifecycleError> {
278        let equation = self.equation()?;
279        let ms = MeasurementSet::open(equation.attachment.path()).map_err(AttachmentError::from)?;
280        let rows = match collect_selected_rows(&ms, equation.attachment.selection(), true) {
281            Ok(rows) => rows,
282            Err(GainSolveError::EmptySelection) => Vec::new(),
283            Err(error) => return Err(CalibrationError::Gain(Box::new(error)).into()),
284        };
285        let selected = SelectedData::of_rows(&ms, rows.iter())
286            .map_err(|error| CalibrationError::Gain(Box::new(error)))?;
287        let output = equation.solve.as_ref().and_then(|configured| {
288            let path = match configured {
289                ConfiguredSolve::LegacyEp(ep) => ep.output_path(),
290                ConfiguredSolve::SingleDish(solver) => solver.recorded_output()?,
291                ConfiguredSolve::Polarization { request, .. } => &request.output,
292                ConfiguredSolve::GatherOnly { request, .. } => &request.output_table,
293                ConfiguredSolve::Task { request, .. } => match request.as_ref() {
294                    CalibrationSolveRequest::Gain(gain)
295                    | CalibrationSolveRequest::GainSpline { gain, .. }
296                    | CalibrationSolveRequest::Fringe { gain, .. }
297                    | CalibrationSolveRequest::FringeTool { gain, .. }
298                    | CalibrationSolveRequest::Accor(gain)
299                    | CalibrationSolveRequest::AdditiveMueller(gain)
300                    | CalibrationSolveRequest::AdditiveMuellerPolynomial { gain, .. }
301                    | CalibrationSolveRequest::Baseline { gain, .. } => &gain.output_table,
302                    CalibrationSolveRequest::Bandpass(request) => &request.output_table,
303                },
304            };
305            Some(path.display().to_string())
306        });
307        let mut entries = vec![
308            RecordField::new(
309                "apply_tables",
310                Value::Array(ArrayValue::from_string_vec(self.applied_table_names())),
311            ),
312            RecordField::new(
313                "selectvis",
314                Value::Record(selectvis_record_value(Some(&selected))),
315            ),
316            RecordField::new(
317                "solve_tables",
318                Value::Array(ArrayValue::from_string_vec(output.into_iter().collect())),
319            ),
320        ];
321        if let Some(stats) = &self.last_solve_stats {
322            entries.push(RecordField::new("solvestats", Value::Record(stats.clone())));
323        }
324        Ok(RecordValue::new(entries))
325    }

334impl super::ApplyTerms {
335    fn source_name(&self, table: &crate::apply::ApplyCalibrationTableSpec) -> String {
336        if let Some(library) = &table.callibrary {
337            library.table_name.clone()
338        } else {
339            self.table_snapshots.get(&table.path).map_or_else(
340                || table.path.display().to_string(),
341                |snapshot| snapshot.name.clone(),
342            )
343        }
344    }
