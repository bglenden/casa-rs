19/// Pixel section enclosing a selected region in the source image.
20pub struct ImageRegionSection {
21    /// First source pixel on each axis.
22    pub start: Vec<usize>,
23    /// Number of sampled pixels on each axis.
24    pub count: Vec<usize>,
25    /// Distance between adjacent samples.
26    pub stride: Vec<usize>,
27    /// Membership of the sampled pixels when requested for a nonrectangular region.
28    pub membership: Option<ArrayD<bool>>,
29    selection: Option<SubimageSelection>,
30}
31
32impl ImageRegionSection {
33    /// Resolve an image region using the shared subimage geometry.
34    ///
35    /// # Errors
36    /// The region cannot be resolved on the supplied image.
37    pub fn resolve<T: ImagePixel>(
38        image: &dyn ImageInterface<T>,
39        region: Option<SubimageRegion<'_>>,
40        include_membership: bool,
41    ) -> Result<Self, ImageError> {
42        // Pixel-only regions also work when the image retains only one axis
43        // of a direction coordinate and has no complete direction plane.
44        if let Some(SubimageRegion::Record(record)) = region
45            && let Some(section) =
46                lattice_pixel_section(record, image.coordinates(), image.shape())?
47        {
48            return Ok(Self {
49                start: section.start,
50                count: section.count,
51                stride: section.stride,
52                membership: None,
53                selection: None,
54            });
55        }
56        let section = SubimageSelection::resolve_section(image, None, region, None, None)?;
57        Ok(Self::from_selection(
58            section,
59            image.ndim(),
60            include_membership,
61        ))
62    }
63
64    /// Resolve inline CRTF, a region file, or a named table region for a copy.
65    ///
66    /// Use the optional origin coordinate system and lattice shape to interpret
67    /// CRTF before mapping it onto the image being copied. Without that origin,
68    /// CRTF is in the copied image's own coordinates. A named table region is
69    /// always looked up in the supplied table.
70    ///
71    /// # Errors
72    /// The text, region file, geometry, or named region is invalid.
73    pub fn resolve_copy_text<T: ImagePixel>(
74        image: &dyn ImageInterface<T>,
75        text: &str,
76        origin: Option<(&CoordinateSystem, &[usize])>,
77        table: Option<&Path>,
78    ) -> Result<Self, ImageError> {
79        let (text, file) = match copy_text_input(text, table)? {
80            CopyTextInput::Record(record) => {
81                return Self::resolve(image, Some(SubimageRegion::Record(&record)), false);
82            }
83            CopyTextInput::Document { text, file } => (text, file),
84        };
85        let context = TextRegionContext {
86            coordinates: origin.map(|(coordinates, _)| coordinates),
87            shape: Some(origin.map_or_else(|| image.shape(), |(_, shape)| shape)),
88        };
89        let section = SubimageSelection::resolve_with_origin(
90            image,
91            None,
92            Some(SubimageRegion::Text(&text)),
93            None,
94            None,
95            Some(context),
96        )
97        .map_err(|error| file.map_or(error, invalid_region_file))?;
98        Ok(Self::from_selection(section, image.ndim(), false))
99    }
100
101    fn from_selection(section: SubimageSelection, ndim: usize, include_membership: bool) -> Self {
102        let mut membership = None;
103        if include_membership && section.stride.is_none() {
104            section.apply_selection_masks(&mut membership);
105        }
106        let count = section.packed_shape();
107        Self {
108            start: section.start.clone(),
109            count,
110            stride: section.stride.clone().unwrap_or_else(|| vec![1; ndim]),
111            membership,
112            selection: section.stride.is_none().then_some(section),
113        }
114    }
115
116    /// Compute region membership in a window of the packed section.
117    /// `start` is relative to the selected section's first output pixel.
118    ///
119    /// # Errors
120    /// The window is outside the selected section.
121    pub fn mask_window(
122        &self,
123        start: &[usize],
124        count: &[usize],
125    ) -> Result<Option<ArrayD<bool>>, ImageError> {
126        validate_slice_request(&self.count, start, count, &vec![1; self.count.len()])?;
127        let mut mask = None;
128        if let Some(selection) = &self.selection {
129            let source = self
130                .start
131                .iter()
132                .zip(start)
133                .map(|(&base, &offset)| base + offset)
134                .collect::<Vec<_>>();
135            selection.apply_window_masks(&mut mask, &source, count);
136        }
137        Ok(mask)
138    }
139}

141#[cfg(test)]
142mod tests {
143    use ndarray::IxDyn;
144
145    use super::{ImageRegionSection, SubimageRegion};
146    use casa_coordinates::{CoordinateSystem, LinearCoordinate};
147    use casa_images::{TempImage, TempStoragePolicy};
148
149    #[test]
150    fn membership_windows_use_source_offsets_on_both_spatial_axes() {
151        let mut coordinates = CoordinateSystem::new();
152        coordinates.add_coordinate(LinearCoordinate::new(2, Vec::new(), Vec::new()));
153        let image =
154            TempImage::<f32>::new(vec![18, 20], coordinates, TempStoragePolicy::Memory).unwrap();
155        let region = SubimageRegion::Text("poly[[4pix,5pix],[10pix,5pix],[4pix,12pix]]");
156        let selection = ImageRegionSection::resolve(&image, Some(region), false).unwrap();
157        assert_eq!(selection.start, [4, 5]);
158        assert_eq!(selection.count, [7, 8]);
159        for (start, count) in [([0, 2], [3, 4]), ([3, 2], [4, 4])] {
160            let expected = ndarray::ArrayD::from_shape_fn(IxDyn(&count), |index| {
161                7 * (start[0] + index[0]) + 6 * (start[1] + index[1]) <= 42
162            });
163            assert_eq!(
164                selection.mask_window(&start, &count).unwrap(),
165                Some(expected)
166            );
167        }
168        assert!(selection.mask_window(&[4, 4], &[4, 4]).is_err());
169    }
