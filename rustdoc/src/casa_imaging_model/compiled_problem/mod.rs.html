84    UnorderedWithinBudget,

87/// Treatment of non-finite input and generated values.
88#[derive(Debug, Clone, Copy, PartialEq, Eq)]
89pub enum FiniteValuePolicy {
90    /// Reject every non-finite value, flagged or not.
91    RejectAll,
92    /// Reject a non-finite input in an unflagged sample, and every
93    /// generated non-finite value; ignore one in a flagged sample.
94    ///
95    /// A flagged sample is never imaged, so its values do not matter, as in
96    /// CASA. An unflagged non-finite visibility is a different matter: CASA
97    /// grids it, and one such sample turns the whole residual and image
98    /// into NaN without a message. Failing the run says so at the first
99    /// pass that reads it, so the data can be flagged first.
100    RejectUnflaggedInput,
101}

168/// Complete numerical behavior permitted for one problem.
169#[derive(Debug, Clone, PartialEq)]
170pub struct NumericsContract {
171    permitted_precisions: Vec<NumericPrecision>,
172    reduction: ReductionPolicy,
173    finite_values: FiniteValuePolicy,
174    stage_error_budgets: Vec<(NumericalStage, StageErrorBudget)>,
175}
176
177impl NumericsContract {
178    /// Construct numerical requirements. Compilation canonicalizes ordering.
179    #[must_use]
180    pub fn new(
181        permitted_precisions: Vec<NumericPrecision>,
182        reduction: ReductionPolicy,
183        finite_values: FiniteValuePolicy,
184        stage_error_budgets: Vec<(NumericalStage, StageErrorBudget)>,
185    ) -> Self {
186        Self {
187            permitted_precisions,
188            reduction,
189            finite_values,
190            stage_error_budgets,
191        }
192    }
193
194    /// Return permitted arithmetic precisions in canonical order after compilation.
195    #[must_use]
196    pub fn permitted_precisions(&self) -> &[NumericPrecision] {
197        &self.permitted_precisions
198    }
199
200    /// Return permitted reduction semantics.
201    #[must_use]
202    pub const fn reduction(&self) -> ReductionPolicy {
203        self.reduction
204    }
205
206    /// Return finite-value behavior.
207    #[must_use]
208    pub const fn finite_values(&self) -> FiniteValuePolicy {
209        self.finite_values
210    }
211
212    /// Return complete stage budgets in canonical stage order after compilation.
213    #[must_use]
214    pub fn stage_error_budgets(&self) -> &[(NumericalStage, StageErrorBudget)] {
215        &self.stage_error_budgets
216    }
217
218    fn canonicalize(mut self) -> Result<Self, CompileProblemError> {
219        self.permitted_precisions.sort_unstable();
220        self.permitted_precisions.dedup();
221        if self.permitted_precisions.is_empty() {
222            return Err(CompileProblemError::InvalidNumerics {
223                reason: "at least one arithmetic precision must be permitted",
224            });
225        }
226        self.stage_error_budgets
227            .sort_unstable_by_key(|(stage, _)| *stage);
228        if self
229            .stage_error_budgets
230            .windows(2)
231            .any(|pair| pair[0].0 == pair[1].0)
232        {
233            return Err(CompileProblemError::InvalidNumerics {
234                reason: "a numerical stage has more than one error budget",
235            });
236        }
237        if self.stage_error_budgets.len() != NumericalStage::ALL.len()
238            || NumericalStage::ALL
239                .iter()
240                .zip(&self.stage_error_budgets)
241                .any(|(required, (actual, _))| required != actual)
242        {
243            return Err(CompileProblemError::InvalidNumerics {
244                reason: "every numerical stage must have exactly one error budget",
245            });
246        }
247        if self.stage_error_budgets.iter().any(|(_, budget)| {
248            !(budget.absolute.is_finite()
249                && budget.absolute >= 0.0
250                && budget.relative.is_finite()
251                && budget.relative >= 0.0)
252        }) {
253            return Err(CompileProblemError::InvalidNumerics {
254                reason: "stage error budgets must be finite and non-negative",
255            });
256        }
257        Ok(self)
258    }
259}

653    /// Numerical requirements are incomplete or invalid.

656        /// Stable human-readable reason.

658    },

667/// Compile and validate one immutable backend-independent problem.
668pub fn compile(input: ProblemInput) -> Result<CompiledProblem, CompileProblemError> {
669    let ProblemInput {
670        specification,
671        geometry,
672        observation,
673        model_lifecycle,
674    } = input;
675    let geometry = compile_geometry(geometry)?;
676    let visibility_transform = specification.visibility_transform;
677    let science = specification.science;
678    let products = specification.products.canonicalize();
679    let observation_transaction = compile_observation_transaction(
680        &observation,
681        specification.observation_transaction,
682        visibility_transform.as_ref(),
683    )?;
684    let numerics = specification.numerics.canonicalize()?;
685    validate_science(&science)?;
686    validate_reconstruction(&specification.reconstruction, &geometry)?;
687    if visibility_transform.is_some()
688        && !matches!(
689            specification.reconstruction.basis(),
690            ReconstructionBasis::ChannelLocal { .. }
691        )
692    {
693        return Err(CompileProblemError::InvalidScientificContract {
694            reason: "sequential visibility transforms require channel-local reconstruction",
695        });
696    }
697    let reconstruction = specification.reconstruction.canonicalize()?;
698    specification
699        .weighting
700        .validate()
701        .map_err(|reason| CompileProblemError::InvalidWeighting { reason })?;
702    validate_products(&science, &reconstruction, &products)?;
703    let normal_equation = compile_normal_equation(
704        &geometry,
705        &observation,
706        &science,
707        &reconstruction,
708        specification.weighting,
709    );
710    let mut required_capabilities = derive_capabilities(
711        &geometry,
712        &science,
713        &reconstruction,
714        normal_equation.weighting(),
715        &products,
716    );
717    if visibility_transform.is_some() {
718        required_capabilities.insert(RequiredCapability::SequentialContinuumTransform);
719    }
720    let product_list = compile_product_list(&geometry, &reconstruction, &products);
721    let model_lifecycle = compile_model_lifecycle_contract(
722        &geometry,
723        normal_equation.measurement_operator().domain(),
724        &numerics,
725        model_lifecycle,
726    )?;
727    Ok(CompiledProblem {
728        model_lifecycle,
729        observation,
730        geometry,
731        science,
732        reconstruction,
733        normal_equation,
734        products,
735        product_list,
736        observation_transaction,
737        numerics,
738        required_capabilities,
739        visibility_transform,
740    })
741}

743fn validate_science(science: &ScientificContract) -> Result<(), CompileProblemError> {
744    if let SpectralKernel::ChannelIntegration { maximum_terms: 0 } =
745        science.spectral.sampling.kernel()
746    {
747        return Err(CompileProblemError::InvalidScientificContract {
748            reason: "spectral channel averaging requires a positive bin width",
749        });
750    }
751    if !matches!(
752        (
753            science.measurement_equation.instrument_response,
754            science.instrument_model,
755        ),
756        (InstrumentResponse::Scalar, None)
757            | (
758                InstrumentResponse::PrimaryBeam,
759                Some(
760                    InstrumentModel::CasaAca7mInterferometricDirectPbV1
761                        | InstrumentModel::CasaAlmaAcaHeterogeneousInterferometricResponseV1
762                        | InstrumentModel::CasaEvlaCommonInterferometricResponseV1
763                        | InstrumentModel::CasaAiryHeterogeneousInterferometricResponseV1
764                        | InstrumentModel::CasaEvlaWidebandAwV1
765                        | InstrumentModel::CasaVlaAwplpg
766                        | InstrumentModel::CasaVlaAwplpgSquint
767                )
768            )
769    ) {
770        return Err(CompileProblemError::InvalidScientificContract {
771            reason: "instrument response and instrument model must form one supported exact pair",
772        });
773    }
774    if science.measurement_equation.w_projection.is_some()
775        && science.measurement_equation.aw_projection.is_some()
776    {
777        return Err(CompileProblemError::InvalidScientificContract {
778            reason: "W projection and AW projection are distinct paired operators",
779        });
780    }
781    if science.measurement_equation.aw_projection.is_some()
782        && science.instrument_model != Some(InstrumentModel::CasaEvlaWidebandAwV1)
783    {
784        return Err(CompileProblemError::InvalidScientificContract {
785            reason: "the paired AW contract requires the exact EVLA wideband instrument model",
786        });
787    }
788    if science.measurement_equation.mosaic_conjugate_beams
789        && !matches!(
790            science.instrument_model,
791            Some(
792                InstrumentModel::CasaAlmaAcaHeterogeneousInterferometricResponseV1
793                    | InstrumentModel::CasaEvlaCommonInterferometricResponseV1
794                    | InstrumentModel::CasaAiryHeterogeneousInterferometricResponseV1
795            )
796        )
797    {
798        return Err(CompileProblemError::InvalidScientificContract {
799            reason: "mosaic conjugate beams require a mosaic instrument model",
800        });
801    }
802    Ok(())
803}

805fn validate_reconstruction(
806    contract: &ReconstructionContract,
807    geometry: &CompiledGeometry,
808) -> Result<(), CompileProblemError> {
809    match contract.basis {
810        // CASA's `mtmfs` runs one Taylor term as a multi-term family
811        // (`SIImageStoreMultiTerm` with `itsNTerms = 1`); every other
812        // single-term MFS deconvolver uses the constant basis, which the
813        // MT-MFS pairing check below keeps apart.
814        ReconstructionBasis::Taylor { terms: 0 } => {
815            return Err(CompileProblemError::InvalidCapabilityCombination {
816                reason: "a Taylor basis requires at least one term",
817            });
818        }
819        ReconstructionBasis::ChannelLocal { channels: 0 } => {
820            return Err(CompileProblemError::InvalidCapabilityCombination {
821                reason: "a channel-local basis requires at least one channel",
822            });
823        }
824        ReconstructionBasis::Constant
825        | ReconstructionBasis::Taylor { .. }
826        | ReconstructionBasis::ChannelLocal { .. } => {}
827    }
828    if let ReconstructionBasis::ChannelLocal { channels } = contract.basis
829        && channels != geometry.spectral().output_channels()
830    {
831        return Err(CompileProblemError::SpectralChannelCountMismatch {
832            geometry_channels: geometry.spectral().output_channels(),
833            reconstruction_channels: channels,
834        });
835    }
836    if matches!(contract.algorithm, ReconstructionAlgorithm::Mtmfs { .. })
837        != matches!(contract.basis, ReconstructionBasis::Taylor { .. })
838    {
839        return Err(CompileProblemError::InvalidCapabilityCombination {
840            reason: "MT-MFS and Taylor-basis reconstruction must be requested together",
841        });
842    }
843    if matches!(contract.algorithm, ReconstructionAlgorithm::Dirty)
844        && contract.controls.max_minor_iterations != 0
845    {
846        return Err(CompileProblemError::InvalidCapabilityCombination {
847            reason: "dirty reconstruction cannot request minor-cycle iterations",
848        });
849    }
850    if matches!(contract.algorithm, ReconstructionAlgorithm::Dirty)
851        && (contract.controls.gain != 1.0 || contract.controls.threshold_jy_per_beam != 0.0)
852    {
853        return Err(CompileProblemError::InvalidCapabilityCombination {
854            reason: "dirty reconstruction requires canonical inactive controls: gain 1 and threshold 0",
855        });
856    }
857    if !matches!(contract.algorithm, ReconstructionAlgorithm::Hogbom)
858        && contract.controls.hogbom_iteration_accounting != HogbomIterationAccounting::Strict
859    {
860        return Err(CompileProblemError::InvalidCapabilityCombination {
861            reason: "CASA-inclusive iteration accounting is specific to Högbom reconstruction",
862        });
863    }
864    // CASA bounds neither the loop gain nor its sign: a negative gain makes
865    // the residual grow until the divergence stop (`test_iterbot_divergence`).
866    if !(contract.controls.gain.is_finite()
867        && contract.controls.threshold_jy_per_beam.is_finite()
868        && contract.controls.threshold_jy_per_beam >= 0.0)
869    {
870        return Err(CompileProblemError::InvalidCapabilityCombination {
871            reason: "reconstruction gain and threshold must be finite and in their valid domains",
872        });
873    }
874    if contract.controls.cycle_iteration_limit == Some(0)
875        || contract
876            .controls
877            .noise_sigma
878            .is_some_and(|sigma| !sigma.is_finite() || sigma < 0.0)
879    {
880        return Err(CompileProblemError::InvalidCapabilityCombination {
881            reason: "minor-cycle limit must be positive and nsigma must be finite and non-negative",
882        });
883    }
884    if contract.controls.cycle_threshold.is_some_and(|rule| {
885        !(rule.factor.is_finite()
886            && rule.factor > 0.0
887            && rule.minimum_psf_fraction.is_finite()
888            && rule.minimum_psf_fraction >= 0.0
889            && rule.maximum_psf_fraction.is_finite()
890            && rule.maximum_psf_fraction > 0.0)
891    }) {
892        // The fractions need not be ordered: CASA raises the fraction to
893        // the minimum and then caps it at the maximum
894        // (`grpcInteractiveCleanManager::updateCycleThreshold`), so the
895        // maximum wins when they cross.
896        return Err(CompileProblemError::InvalidCapabilityCombination {
897            reason: "cycle threshold requires a positive factor and finite PSF fractions",
898        });
899    }
900    if let ReconstructionAlgorithm::Multiscale {
901        scales_px,
902        small_scale_bias,
903    }
904    | ReconstructionAlgorithm::Mtmfs {
905        scales_px,
906        small_scale_bias,
907    } = &contract.algorithm
908    {
909        if scales_px.is_empty()
910            || scales_px
911                .iter()
912                .any(|scale| !(scale.is_finite() && *scale >= 0.0))
913        {
914            return Err(CompileProblemError::InvalidCapabilityCombination {
915                reason: "scale-aware reconstruction requires finite non-negative explicit scales",
916            });
917        }
918        if !small_scale_bias.is_finite() || !(0.0..=1.0).contains(small_scale_bias) {
919            return Err(CompileProblemError::InvalidCapabilityCombination {
920                reason: "scale-aware small-scale bias must be finite and in [0, 1]",
921            });
922        }
923    }
924    Ok(())
925}

927fn validate_products(
928    science: &ScientificContract,
929    reconstruction: &ReconstructionContract,
930    products: &ProductRequirements,
931) -> Result<(), CompileProblemError> {
932    if products.products.is_empty() {
933        return Err(CompileProblemError::InvalidProductCombination {
934            reason: "at least one product must be requested",
935        });
936    }
937    let restored_image_requested = products.contains(ProductKind::RestoredImage);
938    let restoring_beam_requested = !matches!(products.restoring_beam, RestoringBeamPolicy::None);
939    if restored_image_requested != restoring_beam_requested {
940        return Err(CompileProblemError::InvalidProductCombination {
941            reason: "restored-image and restoring-beam requirements must be requested together",
942        });
943    }
944    if restored_image_requested
945        && !(products.contains(ProductKind::Residual) && products.contains(ProductKind::Model))
946    {
947        return Err(CompileProblemError::InvalidProductCombination {
948            reason: "a restored image requires residual and model products",
949        });
950    }
951    let common_spectral_beam = science.spectral.coupling == SpectralCoupling::CommonRestoringBeam;
952    let common_product_beam = products.restoring_beam == RestoringBeamPolicy::Common;
953    if common_spectral_beam != common_product_beam {
954        return Err(CompileProblemError::InvalidProductCombination {
955            reason: "common spectral coupling and common restoring-beam publication must be requested together",
956        });
957    }
958    if products.contains(ProductKind::PbCorrectedImage)
959        && !(products.contains(ProductKind::RestoredImage)
960            && products.contains(ProductKind::PrimaryBeam))
961    {
962        return Err(CompileProblemError::InvalidProductCombination {
963            reason: "a PB-corrected image requires restored-image and primary-beam products",
964        });
965    }
966    let taylor_terms = match reconstruction.basis {
967        ReconstructionBasis::Taylor { terms } => terms,
968        ReconstructionBasis::Constant | ReconstructionBasis::ChannelLocal { .. } => 0,
969    };
970    if products.contains(ProductKind::TaylorTerms) && taylor_terms == 0 {
971        return Err(CompileProblemError::InvalidProductCombination {
972            reason: "Taylor products require a Taylor reconstruction basis",
973        });
974    }
975    if products.contains(ProductKind::TaylorTerms)
976        && ![
977            ProductKind::Psf,
978            ProductKind::Residual,
979            ProductKind::Model,
980            ProductKind::RestoredImage,
981            ProductKind::SumWeights,
982            ProductKind::Weight,
983            ProductKind::PrimaryBeam,
984            ProductKind::PbCorrectedImage,
985        ]
986        .into_iter()
987        .any(|product| products.contains(product))
988    {
989        return Err(CompileProblemError::InvalidProductCombination {
990            reason: "a Taylor coefficient set requires at least one Taylor image product",
991        });
992    }
993    // CASA forms a spectral index only for a Stokes I image
994    // (`SIImageStoreMultiTerm::calculateAlphaBeta`'s `isOnlyStokesI`).
995    if products.contains(ProductKind::SpectralIndex)
996        && !(taylor_terms >= 2
997            && reconstruction.polarization.coordinates() == [PolarizationCoordinate::StokesI]
998            && products.contains(ProductKind::TaylorTerms)
999            && products.contains(ProductKind::Residual)
1000            && products.contains(ProductKind::RestoredImage))
1001    {
1002        return Err(CompileProblemError::InvalidProductCombination {
1003            reason: "spectral index requires at least two Taylor terms of Stokes I alone plus Taylor, residual, and restored-image products",
1004        });
1005    }
1006    if products.contains(ProductKind::SpectralIndexError)
1007        && !products.contains(ProductKind::SpectralIndex)
1008    {
1009        return Err(CompileProblemError::InvalidProductCombination {
1010            reason: "spectral-index uncertainty requires a spectral-index product",
1011        });
1012    }
1013    if products.contains(ProductKind::SpectralCurvature)
1014        && !(taylor_terms >= 3 && products.contains(ProductKind::SpectralIndex))
1015    {
1016        return Err(CompileProblemError::InvalidProductCombination {
1017            reason: "spectral curvature requires at least three Taylor terms and a spectral-index product",
1018        });
1019    }
1020    if products.contains(ProductKind::PbCorrectedSpectralIndex)
1021        && !(products.contains(ProductKind::SpectralIndex)
1022            && products.contains(ProductKind::PrimaryBeam))
1023    {
1024        return Err(CompileProblemError::InvalidProductCombination {
1025            reason: "PB-corrected spectral index requires spectral-index and primary-beam products",
1026        });
1027    }
1028    if products.contains(ProductKind::Beam)
1029        && ![
1030            ProductKind::Psf,
1031            ProductKind::Residual,
1032            ProductKind::RestoredImage,
1033        ]
1034        .into_iter()
1035        .any(|product| products.contains(product))
1036    {
1037        return Err(CompileProblemError::InvalidProductCombination {
1038            reason: "beam metadata requires a PSF, residual, or restored-image product",
1039        });
1040    }
1041    Ok(())
1042}

1044fn derive_capabilities(
1045    geometry: &CompiledGeometry,
1046    science: &ScientificContract,
1047    reconstruction: &ReconstructionContract,
1048    weighting: &WeightingOperatorContract,
1049    products: &ProductRequirements,
1050) -> BTreeSet<RequiredCapability> {
1051    let mut capabilities = BTreeSet::new();
1052    if geometry.domains().len() > 1 {
1053        capabilities.insert(RequiredCapability::MultiDomainGeometry);
1054    }
1055    if geometry
1056        .domains()
1057        .iter()
1058        .any(|domain| domain.facets().len() > 1)
1059    {
1060        capabilities.insert(RequiredCapability::FacetedGeometry);
1061    }
1062    if geometry.spectral().source_frame() != geometry.spectral().output_frame() {
1063        capabilities.insert(RequiredCapability::SpectralFrameTransform);
1064    }
1065    if science.spectral.sampling != SpectralSamplingLaw::IDENTITY {
1066        capabilities.insert(RequiredCapability::SpectralResampling);
1067    }
1068    if science.measurement_equation.w_projection.is_some() {
1069        capabilities.insert(RequiredCapability::WProjection);
1070    }
1071    if science.measurement_equation.aw_projection.is_some() {
1072        capabilities.insert(RequiredCapability::AwProjection);
1073    }
1074    if science.spectral.coupling == SpectralCoupling::CommonRestoringBeam {
1075        capabilities.insert(RequiredCapability::CommonBeamSpectralCoupling);
1076    }
1077    capabilities.extend(
1078        reconstruction
1079            .polarization
1080            .coordinates
1081            .iter()
1082            .copied()
1083            .map(RequiredCapability::Polarization),
1084    );
1085    match science.measurement_equation.instrument_response {
1086        InstrumentResponse::Scalar => {}
1087        InstrumentResponse::PrimaryBeam => {
1088            capabilities.insert(RequiredCapability::PrimaryBeamResponse);
1089        }
1090        InstrumentResponse::FullMueller => {
1091            capabilities.insert(RequiredCapability::FullMuellerResponse);
1092        }
1093    }
1094    if weighting.uv_taper().is_some() {
1095        capabilities.insert(RequiredCapability::UvTaper);
1096    }
1097    capabilities.insert(match reconstruction.basis {
1098        ReconstructionBasis::Constant => RequiredCapability::ConstantBasis,
1099        ReconstructionBasis::Taylor { .. } => RequiredCapability::TaylorBasis,
1100        ReconstructionBasis::ChannelLocal { .. } => RequiredCapability::ChannelLocalBasis,
1101    });
1102    capabilities.insert(match reconstruction.algorithm {
1103        ReconstructionAlgorithm::Dirty => RequiredCapability::DirtyReconstruction,
1104        ReconstructionAlgorithm::Hogbom => RequiredCapability::HogbomReconstruction,
1105        ReconstructionAlgorithm::Clark | ReconstructionAlgorithm::ClarkStokes => {
1106            RequiredCapability::ClarkReconstruction
1107        }
1108        ReconstructionAlgorithm::Mem => RequiredCapability::MemReconstruction,
1109        ReconstructionAlgorithm::Multiscale { .. } => RequiredCapability::MultiscaleReconstruction,
1110        ReconstructionAlgorithm::Mtmfs { .. } => RequiredCapability::MtmfsReconstruction,
1111    });
1112    capabilities.insert(match weighting.scheme() {
1113        WeightingScheme::Natural => RequiredCapability::NaturalWeighting,
1114        WeightingScheme::Uniform => RequiredCapability::UniformWeighting,
1115        WeightingScheme::Briggs { .. } => RequiredCapability::BriggsWeighting,
1116        WeightingScheme::BriggsBandwidthTaper { .. } => {
1117            RequiredCapability::BriggsBandwidthTaperWeighting
1118        }
1119        WeightingScheme::Radial => RequiredCapability::RadialWeighting,
1120        WeightingScheme::BriggsAbsolute { .. } => RequiredCapability::BriggsAbsoluteWeighting,
1121    });
1122    capabilities.insert(match products.normalization {
1123        ProductNormalization::UnitResponse => RequiredCapability::UnitResponseNormalization,
1124        ProductNormalization::FlatNoise => RequiredCapability::FlatNoiseNormalization,
1125        ProductNormalization::FlatSky => RequiredCapability::FlatSkyNormalization,
1126        ProductNormalization::PbSquare => RequiredCapability::PbSquareNormalization,
1127    });
1128    capabilities.extend(
1129        products
1130            .products
1131            .iter()
1132            .copied()
1133            .map(RequiredCapability::Product),
1134    );
1135    capabilities
1136}
