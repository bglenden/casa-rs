3use casa_components::{ShapeType, SkyComponent};
4use casa_core::{ArrayValue, RecordValue, ScalarValue, Value};
5
6use casa_images::{GaussianBeam, ImageError};
7
8pub(super) struct Measurement {
9    pub(super) channel: usize,
10    pub(super) pixels: [f64; 2],
11    pub(super) peak: [f64; 2],
12    pub(super) sum: f64,
13    pub(super) brightness: String,
14    pub(super) beam: Option<GaussianBeam>,
15    pub(super) beam_in_pixels: f64,
16    pub(super) is_point: bool,
17    pub(super) deconvolved_peak: Option<[f64; 2]>,
18}
19
20impl Measurement {
21    pub(super) fn record(
22        &self,
23        component: &SkyComponent,
24        deconvolved: bool,
25    ) -> Result<RecordValue, ImageError> {
26        let mut record = component
27            .to_record()
28            .map_err(|error| ImageError::CasaSelection(error.to_string()))?;
29        let peak = if deconvolved {
30            self.deconvolved_peak
31        } else {
32            Some(self.peak)
33        };
34        if let Some([value, error]) = peak {
35            let mut peak = quantity(value, &self.brightness);
36            peak.upsert("error", real(error));
37            record.upsert("peak", Value::Record(peak));
38        }
39        record.upsert("sum", Value::Record(quantity(self.sum, &self.brightness)));
40        if !deconvolved {
41            record.upsert(
42                "pixelcoords",
43                Value::Array(ArrayValue::from_f64_vec(self.pixels.to_vec())),
44            );
45        }
46        if let Some(beam) = self.beam {
47            record.upsert("ispoint", Value::Scalar(ScalarValue::Bool(self.is_point)));
48            if !deconvolved || component.shape().shape_type() == ShapeType::Gaussian {
49                let mut metadata = RecordValue::default();
50                metadata.upsert("beamarcsec", Value::Record(beam.to_record()));
51                metadata.upsert("beampixels", real(self.beam_in_pixels));
52                metadata.upsert("beamster", real(beam.area()));
53                record.upsert("beam", Value::Record(metadata));
54            }
55        }
56        if let Some(Value::Record(spectrum)) = record.get_mut("spectrum") {
57            spectrum.upsert(
58                "channel",
59                Value::Scalar(ScalarValue::Int32(self.channel as i32)),
60            );
61        }
62        if let Some(Value::Record(shape)) = record.get_mut("shape") {
63            display_gaussian(shape);
64        }
65        Ok(record)
66    }
67}
68
69pub(super) fn quantity(value: f64, unit: &str) -> RecordValue {
70    let mut record = RecordValue::default();
71    record.upsert("value", real(value));
72    record.upsert("unit", Value::Scalar(ScalarValue::String(unit.to_owned())));
73    record
74}
75
76fn real(value: f64) -> Value {
77    Value::Scalar(ScalarValue::Float64(value))
78}
79
80fn display_gaussian(shape: &mut RecordValue) {
81    const DEGREES_PER_RADIAN: f64 = 180. / std::f64::consts::PI;
82    for (key, unit, conversion) in [
83        ("majoraxis", "arcsec", 60.),
84        ("minoraxis", "arcsec", 60.),
85        ("majoraxiserror", "arcsec", 3600. * DEGREES_PER_RADIAN),
86        ("minoraxiserror", "arcsec", 3600. * DEGREES_PER_RADIAN),
87        ("positionangleerror", "deg", DEGREES_PER_RADIAN),
88    ] {
89        if let Some(Value::Record(quantity)) = shape.get_mut(key) {
90            if let Some(Value::Scalar(ScalarValue::Float64(value))) = quantity.get_mut("value") {
91                *value *= conversion;
92            }
93            quantity.upsert("unit", Value::Scalar(ScalarValue::String(unit.to_owned())));
94        }
95    }
