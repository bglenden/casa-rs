16/// The task's structured result, sharing the existing run/artifact wire format.
17#[derive(Serialize, JsonSchema)]
18pub struct TotalPowerTaskResult {
19    /// Resolved CASA parameters.
20    pub request: ImagingParameters,
21    /// Run identity and timing.
22    pub run: TotalPowerRun,
23    /// Both published CASA images.
24    pub artifacts: Vec<ImagerArtifact>,
25}
26/// Metadata for a completed total-power run.
27#[derive(Serialize, JsonSchema)]
28pub struct TotalPowerRun {
29    /// Output stem.
30    pub stem: String,
31    /// Output directory.
32    pub directory: String,
33    /// Scientific warnings, when applicable.
34    pub warnings: Vec<String>,
35    /// Wall time in nanoseconds.
36    pub elapsed_ns: u64,
37}
38/// The native task contract, with CASA's ordered 33-parameter surface.
39pub fn tsdimaging_task_schema_bundle() -> TaskProviderContract {
40    let mut bundle = imager_task_schema_bundle();
41    bundle.protocol = ProviderProtocolDescriptor::new(
42        "casa_tsdimaging_task",
43        1,
44        ProviderSurfaceKind::Task,
45        env!("CARGO_PKG_VERSION"),
46    );
47    let result = schema_for!(TotalPowerTaskResult);
48    bundle.semantic.result_schema = result.clone();
49    bundle.domain_schemas.result_schema = result;
50    bundle.components = merged_components([
51        &bundle.domain_schemas.request_schema,
52        &bundle.domain_schemas.result_schema,
53    ]);
54    bundle.parameter_surfaces =
55        vec![builtin_surface_bundle("tsdimaging").expect("built-in tsdimaging surface")];
56    bundle
57}
58/// The CLI and Python task host on the shared task envelope, observing
59/// `cancel`.
60pub fn tsdimaging_front_end_with_cancel(
61    cancel: Cancel,
62) -> TaskFrontEnd<impl Fn(&TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure>> {
63    TaskFrontEnd::new_with_deferred_schema(
64        "tsdimaging",
65        tsdimaging_task_schema_bundle(),
66        &["tsdimaging"],
67        move |task| run(task, cancel.clone()),
68    )
69    .expect("tsdimaging contract")
70}
71/// The CLI and Python task host on the shared task envelope.
72pub fn tsdimaging_front_end()
73-> TaskFrontEnd<impl Fn(&TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure>> {
74    tsdimaging_front_end_with_cancel(Cancel::new())
75}
76pub(crate) fn run(task: &TaskInvocation<'_>, cancel: Cancel) -> Result<TaskSuccess, TaskFailure> {
77    let started = Instant::now();
78    let parameters = task.plain_values();
79    let mut execution_parameters = parameters.clone();
80    task.resolve_file_values(&mut execution_parameters, &["infile", "outfile"]);
81    let context = RunContext {
82        host: HostResources::detect().map_err(|e| TaskFailure::execution(e.to_string()))?,
83        policy: ResourcePolicy::Balanced,
84        cancel,
85        summary: None,
86    };
87    let output = image_from_parameters(&execution_parameters, &context).map_err(|error| {
88        if context.cancel.is_cancelled() {
89            TaskFailure::interrupted(error.to_string())
90        } else {
91            TaskFailure::execution(error.to_string())
92        }
93    })?;
94    let artifacts = [
95        (ImagerArtifactKind::Image, &output.image),
96        (ImagerArtifactKind::Weight, &output.weight),
97    ]
98    .into_iter()
99    .map(|(kind, path)| ImagerArtifact {
100        kind,
101        label: path
102            .file_name()
103            .unwrap_or_default()
104            .to_string_lossy()
105            .into_owned(),
106        path: path.display().to_string(),
107        exists: path.exists(),
108    })
109    .collect();
110    let result = TotalPowerTaskResult {
111        run: TotalPowerRun {
112            stem: parameters["outfile"].as_str().unwrap_or_default().into(),
113            directory: output
114                .image
115                .parent()
116                .unwrap_or_else(|| std::path::Path::new("."))
117                .display()
118                .to_string(),
119            warnings: Vec::new(),
120            elapsed_ns: started.elapsed().as_nanos().try_into().unwrap_or(u64::MAX),
121        },
122        request: ImagingParameters(parameters),
123        artifacts,
124    };
125    TaskSuccess::new(crate::CasaValue::None, &result)
126}
