2//! Feeds: `sm.setfeed`, and the feeds `observe` creates when none are set.
3//!
4//! `NewMSSimulator::initFeeds` appends one FEED row per antenna (and per
5//! feed in list mode) for the antennas defined when it is called.
6
7use super::error::{SimulatorError, SimulatorResult};
8
9/// The feed modes `Simulator::setfeed` accepts.
10const MODES: [&str; 3] = ["perfect R L", "perfect X Y", "list"];
11
12/// One FEED row.
13#[derive(Debug, Clone, PartialEq)]
14pub(crate) struct Feed {
15    pub(crate) antenna_id: i32,
16    pub(crate) feed_id: i32,
17    /// POLARIZATION_TYPE of the two receptors.
18    pub(crate) polarization: [&'static str; 2],
19}
20
21/// The FEED rows of `setfeed(mode, x, y, pol)` for `antenna_count`
22/// antennas.
23///
24/// Only one feed per antenna is supported: CASA chooses list mode, one feed
25/// per position, whenever `x` has more than one element, and the simulator
26/// writes such multi-feed observations from Phase 2 on.
27pub(crate) fn feeds(
28    mode: &str,
29    x: &[f64],
30    y: &[f64],
31    pol: &[String],
32    antenna_count: usize,
33) -> SimulatorResult<Vec<Feed>> {
34    if !MODES.contains(&mode) {
35        return Err(SimulatorError::refused(
36            "Currently, only perfect R L or perfect X Y feeds or list are recognized",
37        ));
38    }
39    if antenna_count == 0 {
40        // CASA logs this, then aborts building an empty table slice.
41        return Err(SimulatorError::raised(
42            "NewMSSimulator::initFeeds: must call initAnt() first",
43        ));
44    }
45    if x.len() > 1 {
46        if y.len() != x.len() {
47            return Err(SimulatorError::raised(
48                "Feed x and y must be the same length",
49            ));
50        }
51        if pol.len() != x.len() {
52            return Err(SimulatorError::raised(
53                "Feed polarization list must be same length as the number of positions",
54            ));
55        }
56        return Err(SimulatorError::raised(
57            "setfeed: more than one feed per antenna (list mode) is not yet implemented in casa-rs",
58        ));
59    }
60    Ok(perfect_feeds(mode, antenna_count))
61}
62
63/// One perfect feed per antenna: X and Y receptors if `mode` contains an
64/// uppercase X, otherwise R and L.
65pub(crate) fn perfect_feeds(mode: &str, antenna_count: usize) -> Vec<Feed> {
66    let polarization = if mode.contains('X') {
67        ["X", "Y"]
68    } else {
69        ["R", "L"]
70    };
71    (0..antenna_count)
72        .map(|antenna| Feed {
73            antenna_id: antenna as i32,
74            feed_id: 0,
75            polarization,
76        })
77        .collect()
78}

80#[cfg(test)]
81mod tests {
82    use super::*;
83
84    #[test]
85    fn perfect_modes_choose_the_receptors() {
86        let linear = feeds("perfect X Y", &[0.0], &[0.0], &["R".to_string()], 2).unwrap();
87        assert_eq!(linear.len(), 2);
88        assert_eq!(linear[1].antenna_id, 1);
89        assert_eq!(linear[0].polarization, ["X", "Y"]);
90        let circular = feeds("perfect R L", &[0.0], &[0.0], &["R".to_string()], 1).unwrap();
91        assert_eq!(circular[0].polarization, ["R", "L"]);
92        // A one-position list is a perfect feed.
93        let list = feeds("list", &[0.0], &[0.0], &["X".to_string()], 1).unwrap();
94        assert_eq!(list[0].polarization, ["R", "L"]);
95    }
96
97    #[test]
98    fn unknown_modes_and_missing_antennas_are_refused() {
99        let error = feeds("", &[0.0], &[0.0], &[], 3).unwrap_err();
100        assert_eq!(
101            error.outcome(),
102            super::super::error::CasaOutcome::ReturnsFalse
103        );
104        assert!(feeds("perfect R L", &[0.0], &[0.0], &[], 0).is_err());
105    }
106}
