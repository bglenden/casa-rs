8pub(super) fn modern_record(
9    mut moments: Moments,
10    masked: bool,
11    weighted: bool,
12    quantiles: bool,
13    boolean_column: bool,
14) -> RecordValue {
15    let populated = moments.count > 0;
16    let scalar = |value: f64| {
17        if populated && boolean_column {
18            ScalarValue::Int64(value as i64)
19        } else {
20            ScalarValue::Float64(value)
21        }
22    };
23    let missing = if quantiles { f64::NAN } else { 0.0 };
24    let variance = moments.variance();
25    let mut fields = RecordValue::new(vec![
26        field("isMasked", ScalarValue::Bool(masked && populated)),
27        field("isWeighted", ScalarValue::Bool(weighted && populated)),
28        field("npts", ScalarValue::Float64(moments.count as f64)),
29    ]);
30    for (name, value) in [
31        ("mean", moments.mean),
32        ("rms", moments.rms()),
33        ("stddev", variance.sqrt()),
34        ("sum", moments.sum),
35        ("sumOfWeights", moments.sum_weights),
36        ("sumsq", moments.sum_squares),
37        ("variance", variance),
38    ] {
39        fields.push(field(name, scalar(if populated { value } else { missing })));
40    }
41    if let Some(min) = moments.min {
42        fields.push(field("min", scalar(min.value)));
43        fields.push(field(
44            "minDatasetIndex",
45            ScalarValue::Int64(min.dataset as i64),
46        ));
47        fields.push(field("minIndex", ScalarValue::Int64(min.index as i64)));
48    }
49    if let Some(max) = moments.max {
50        fields.push(field("max", scalar(max.value)));
51        fields.push(field(
52            "maxDatasetIndex",
53            ScalarValue::Int64(max.dataset as i64),
54        ));
55        fields.push(field("maxIndex", ScalarValue::Int64(max.index as i64)));
56    }
57    if quantiles {
58        let observed = moments.quantiles();
59        for (name, value) in [
60            ("firstquartile", observed.as_ref().map(|q| q.first)),
61            ("thirdquartile", observed.as_ref().map(|q| q.third)),
62            ("median", observed.as_ref().map(|q| q.median)),
63            (
64                "medabsdevmed",
65                observed.as_ref().map(|q| q.median_deviation),
66            ),
67        ] {
68            fields.push(field(name, scalar(value.unwrap_or(f64::NAN))));
69        }
70    }
71    fields
72        .fields_mut()
73        .sort_unstable_by(|a, b| a.name.cmp(&b.name));
74    fields
75}

77pub(super) fn legacy_record(mut moments: Moments) -> MsResult<RecordValue> {
78    let quantiles = moments
79        .quantiles()
80        .ok_or_else(|| MsError::InvalidInput("All selected rows are flagged.".into()))?;
81    let variance = moments.variance();
82    Ok(RecordValue::new(
83        [
84            ("max", moments.max.unwrap().value),
85            ("mean", moments.mean),
86            ("medabsdevmed", quantiles.median_deviation),
87            ("median", quantiles.median),
88            ("min", moments.min.unwrap().value),
89            ("npts", moments.count as f64),
90            ("quartile", quantiles.third - quantiles.first),
91            ("rms", moments.rms()),
92            ("stddev", variance.sqrt()),
93            ("sum", moments.sum),
94            ("sumsq", moments.sum_squares),
95            ("var", variance),
96        ]
97        .into_iter()
98        .map(|(name, value)| field(name, ScalarValue::Float64(value)))
99        .collect(),
100    ))
101}

103fn field(name: &str, value: ScalarValue) -> RecordField {
104    RecordField::new(name, Value::Scalar(value))
