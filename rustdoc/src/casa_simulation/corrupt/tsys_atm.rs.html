2//! `sm.setnoise(mode='tsys-atm')`: thermal noise whose σ comes from a
3//! system temperature the atmosphere model gives, channel by channel (CASA's
4//! `AtmosCorruptor` with ATM, #841 Phase 4c).
5//!
6//! # What CASA computes
7//!
8//! As for `tsys-manual` ([`super::tsys`]), `Simulator::setnoise` adds a
9//! unit `ANoise` and a `T` term flagged `NoiseScale`, here a `Tf Jones`
10//! (type "TF NOISE"), which depends on frequency: `SolvableVisCal` asks the
11//! corruptor for every channel of the window (`setFocusChan`), so σ is
12//! per channel. Antenna `i`'s `T` in channel `c` is `D_i / f / √Tsys(A_i, c)`
13//! with `f` as for `tsys-manual`, so a row of antennas `i` and `j` gets
14//! σ_ij(c) = c·a/√(Δν t) · √(Tsys(A_i, c) Tsys(A_j, c)) / (D_i D_j), and
15//!
16//! ```text
17//! Tsys(A, c) = (hν/k) · e^(τ A) · (1/(e^(hν/k / T_ebb(A)) − 1) + T_rx/(hν/k)) · (1 + g)
18//! ```
19//!
20//! (`AtmosCorruptor::tsys` in its ATM branch), with τ the zenith opacity
21//! (ATM's dry opacity plus its wet opacity for `pwv`) and `T_ebb` the sky's
22//! equivalent blackbody temperature at airmass `A`, the receiver coupled to
23//! the sky by `spillefficiency` with the rest of its beam on the ground at
24//! `tground` (`SkyStatus::getTebbSky(…, pwv, A, spilleff, tground)`). hν/k
25//! is 4.80145·10⁻¹¹ K/Hz times the channel's frequency.
26//!
27//! The atmosphere is ATM's (`casa-atmosphere`) for CASA's simulator site
28//! defaults: `altitude` 5000 m, `pground` 560 hPa, `tground` 270 K,
29//! `relhum` 20 %, `waterheight` 2 km, a −5.6 K/km lapse rate, 10 hPa steps,
30//! top 48 km and ATM's tropical reference atmosphere; `setnoise` replaces an
31//! unset `pwv` (≤ 0) with 1 mm, `pground` ≤ 100 hPa with 560, `relhum` ≤ 0
32//! with 20, `altitude` ≤ 0 with 5000 m and `waterheight` ≤ 100 m with 2 km.
33//!
34//! Measured against CASA 6.7.6 (probe of 64-channel windows over the
35//! 183 GHz and 325 GHz water lines, `tools/simulation`), every channel's
36//! noise variance agrees with this formula through `casa-atmosphere` to the
37//! probe's statistics (1 %), once CASA's frequency mapping below is used.
38//!
39//! # Where casa-rs departs
40//!
41//! Recorded in `docs/CASA (C++) bugs.md`:
42//! - CASA's ATM grid for a window is centred on its REF_FREQUENCY, the first
43//!   channel's frequency: ATM channel `i` sits at `REF + (i − n/2)·Δν`
44//!   (`n/2 + 1` for windows after the first), so Tsys(c) is the atmosphere's
45//!   half a band below channel `c`. casa-rs evaluates the atmosphere at each
46//!   channel's own frequency (CHAN_FREQ).
47//! - For a window of one channel after the first, CASA reads its channel map
48//!   past its end, which is undefined; in CASA 6.7.6 the term's `T` came out
49//!   bad and those rows got unscaled, 1 Jy noise. casa-rs evaluates the
50//!   atmosphere at the channel's frequency as for the first window.
51//! - As for `tsys-manual`, casa-rs takes each row's own airmasses and
52//!   EXPOSURE, where CASA caches airmasses for 100 s and uses the first
53//!   row's EXPOSURE.
54

63/// h/k, K/Hz, as `AtmosCorruptor::tsys` writes it.
64const PLANCK_OVER_BOLTZMANN: f64 = 4.801_45e-11;
65
66/// The sky background ATM's `SkyStatus` takes, K.
67const BACKGROUND_K: f64 = 2.73;
68
69/// The site and atmosphere arguments of `setnoise(mode='tsys-atm')`, as
70/// the user gives them: unset values (at or below CASA's thresholds) take
71/// CASA's defaults, and every value is held in single precision, as CASA's
72/// corruptor holds it.
73#[derive(Debug, Clone, Copy, PartialEq)]
74pub struct AtmosphereSpec {
75    /// `pground`: the ground pressure, hPa (mbar); 560 when at most 100.
76    pub ground_pressure_hpa: f64,
77    /// `relhum`: the ground relative humidity, %; 20 when not positive.
78    pub relative_humidity_percent: f64,
79    /// `altitude`: the site's altitude, m; 5000 when not positive.
80    pub altitude_m: f64,
81    /// `waterheight`: the water vapour scale height, m; 2 km when at most
82    /// 100 m.
83    pub water_scale_height_m: f64,
84}
85
86impl Default for AtmosphereSpec {
87    /// Every value unset, as `sm.setnoise`'s defaults leave them.
88    fn default() -> Self {
89        Self {
90            ground_pressure_hpa: 0.0,
91            relative_humidity_percent: 0.0,
92            altitude_m: 0.0,
93            water_scale_height_m: 0.0,
94        }
95    }
96}

98impl AtmosphereSpec {
99    /// The ATM profile parameters `Simulator::setnoise` and
100    /// `AtmosCorruptor::initAtm` make of these values, the ground at
101    /// `ground_k` (`tground`): each in single precision, as the corruptor
102    /// reads them (`asFloat`).
103    pub(crate) fn resolved(&self, ground_k: f64) -> ProfileParameters {
104        let pick = |value: f64, unset: bool, default: f64| {
105            if unset { default } else { single(value) }
106        };
107        ProfileParameters {
108            altitude_m: pick(self.altitude_m, self.altitude_m <= 0.0, 5000.0),
109            ground_pressure_hpa: pick(
110                self.ground_pressure_hpa,
111                self.ground_pressure_hpa <= 100.0,
112                560.0,
113            ),
114            ground_temperature_k: single(ground_k),
115            lapse_rate_k_per_km: -5.6,
116            relative_humidity_percent: pick(
117                self.relative_humidity_percent,
118                self.relative_humidity_percent <= 0.0,
119                20.0,
120            ),
121            water_vapour_scale_height_km: pick(
122                self.water_scale_height_m / 1000.0,
123                self.water_scale_height_m <= 100.0,
124                2.0,
125            ),
126            pressure_step_hpa: 10.0,
127            top_altitude_km: 48.0,
128            atmosphere: AtmosphereType::Tropical,
129        }
130    }
131}

133/// `value` rounded to single precision, as CASA's corruptor holds its

135pub(crate) fn single(value: f64) -> f64 {

139/// A spectral window's atmosphere for the noise: each channel's line of
140/// sight, zenith opacity (single precision, as CASA holds it) and hν/k.
141#[derive(Debug, Clone, PartialEq)]
142pub(crate) struct WindowAtmosphere {
143    columns: Vec<SkyColumn>,
144    opacity: Vec<f32>,
145    h_nu_over_k: Vec<f64>,
146}
147
148impl WindowAtmosphere {
149    /// The heap a window of `channels` channels keeps for an atmosphere of
150    /// `layers` layers.
151    pub(crate) fn bytes(channels: usize, layers: usize) -> u64 {
152        channels as u64
153            * (size_of::<SkyColumn>() as u64
154                + SkyColumn::heap_bytes(layers)
155                + size_of::<f32>() as u64
156                + size_of::<f64>() as u64)
157    }
158
159    /// The window of `profile`'s channels, the sky seen through `pwv_m` of
160    /// water with `spillover` of the beam on the sky and the rest on the
161    /// ground at `ground_k`, each as the caller holds it ([`single`]).
162    pub(crate) fn new(
163        profile: &AbsorptionProfile<ChannelTotals>,
164        pwv_m: f64,
165        (spillover, ground_k): (f64, f64),
166    ) -> Result<Self, casa_atmosphere::AtmosphereError> {
167        let sky = Sky {
168            water_column_m: pwv_m,
169            airmass: 1.0,
170            sky_coupling: spillover,
171            spill_temperature_k: ground_k,
172            background_temperature_k: BACKGROUND_K,
173        };
174        let channels = profile.channels().len();
175        let mut columns = Vec::with_capacity(channels);
176        let mut opacity = Vec::with_capacity(channels);
177        let mut h_nu_over_k = Vec::with_capacity(channels);
178        for (channel, frequency_hz) in profile.grid().frequencies_hz().enumerate() {
179            columns.push(profile.sky_column(channel, &sky)?);
180            // `AtmosCorruptor::opac`: dry plus wet in double precision, the
181            // sum held as a `Float`.
182            opacity
183                .push((profile.dry_opacity(channel) + profile.wet_opacity(channel, pwv_m)) as f32);
184            h_nu_over_k.push(PLANCK_OVER_BOLTZMANN * frequency_hz);
185        }
186        Ok(Self {
187            columns,
188            opacity,
189            h_nu_over_k,
190        })
191    }
192
193    /// The window's channels.
194    pub(crate) fn channels(&self) -> usize {
195        self.columns.len()
196    }
197
198    /// Channel `channel`'s zenith opacity.
199    pub(crate) fn zenith_opacity(&self, channel: usize) -> f32 {
200        self.opacity[channel]
201    }
202}

204/// The `tsys-atm` model as CASA's corruptor holds it: single-precision
205/// parameters, and each spectral window's atmosphere, with the memory
206/// admitted for what it keeps.
207#[derive(Debug)]
208pub(crate) struct TsysAtm {
209    /// `senscoeff() · amp()` in single precision: c · a, Jy m² per kelvin.
210    jansky_per_kelvin: f32,
211    receiver_k: f32,
212    /// 1 + g.
213    sidebands: f64,
214    /// By spectral window id; `None` for a window defined after `setnoise`.
215    windows: Vec<Option<WindowAtmosphere>>,
216    _reservation: Reservation,
217}
218
219impl TsysAtm {
220    /// The model of `parameters` over `windows` (by spectral window id),
221    /// holding `reservation`.
222    pub(crate) fn new(
223        parameters: &TsysParameters,
224        windows: Vec<Option<WindowAtmosphere>>,
225        reservation: Reservation,
226    ) -> Self {
227        let manual = super::tsys::TsysManual::new(parameters);
228        Self {
229            jansky_per_kelvin: manual.jansky_per_kelvin(),
230            receiver_k: parameters.receiver_k as f32,
231            sidebands: manual.sidebands(),
232            windows,
233            _reservation: reservation,
234        }
235    }
236
237    /// Whether the model has spectral window `window`'s atmosphere.
238    pub(crate) fn covers(&self, window: usize) -> bool {
239        self.windows.get(window).is_some_and(Option::is_some)
240    }
241
242    /// Tsys in channel `channel` of `window` at airmass `airmass`, K, in
243    /// single precision (`AtmosCorruptor::tsys`).
244    pub(crate) fn tsys(&self, window: &WindowAtmosphere, channel: usize, airmass: f32) -> f32 {
245        let h_nu_over_k = window.h_nu_over_k[channel];
246        let tau = f64::from(window.opacity[channel] * airmass);
247        let tebb = window.columns[channel].tebb(f64::from(airmass));
248        let radiance = tau.exp()
249            * (1.0 / ((h_nu_over_k / tebb).exp() - 1.0) + f64::from(self.receiver_k) / h_nu_over_k);
250        (h_nu_over_k * radiance * self.sidebands) as f32
251    }
252
253    /// Spectral window `window`'s atmosphere.
254    ///
255    /// # Panics
256    ///
257    /// When the model does not cover the window ([`Self::covers`]).
258    pub(crate) fn window(&self, window: usize) -> &WindowAtmosphere {
259        self.windows[window]
260            .as_ref()
261            .expect("corrupt refuses a window the tsys-atm term does not cover")
262    }
263
264    /// Each channel's `1/T` for an antenna of diameter `diameter_m` at
265    /// airmass `airmass`, in a window whose channels are
266    /// `channel_bandwidth_hz` wide, for an integration of `exposure_s`,
267    /// into `scales` (one a channel): the antenna's factor of σ (`s` in
268    /// [`super::tsys`]), 1 where CASA applies a bad `T` as 1.
269    pub(crate) fn antenna_scales(
270        &self,
271        window: &WindowAtmosphere,
272        (diameter_m, airmass): (f64, f32),
273        (channel_bandwidth_hz, exposure_s): (f64, f64),
274        scales: &mut [f32],
275    ) {
276        let factor =
277            (f64::from(self.jansky_per_kelvin) / (channel_bandwidth_hz * exposure_s).sqrt()).sqrt();
278        for (channel, scale) in scales.iter_mut().enumerate() {
279            let tsys = self.tsys(window, channel, airmass);
280            let jones = (diameter_m / factor / f64::from(tsys.sqrt())) as f32;
281            *scale = if jones.is_nan() || jones == 0.0 {
282                1.0
283            } else {
284                1.0 / jones
285            };
286        }
287    }
288}

290/// A grid of one window of the channels at `frequencies_hz`.
291pub(crate) fn window_grid(
292    frequencies_hz: &[f64],
293) -> Result<SpectralGrid, casa_atmosphere::AtmosphereError> {
294    let mut grid = SpectralGrid::new();
295    grid.add_channels(frequencies_hz, 0)?;
296    Ok(grid)
297}

299/// Each channel's totals of `atmosphere` at `frequencies_hz`: one
300/// independent computation per channel, which a caller spreads over its
301/// workers.
302pub(crate) fn channel_totals(
303    atmosphere: &AtmosphereProfile,
304    frequencies_hz: &[f64],
305) -> Vec<ChannelTotals> {
306    frequencies_hz
307        .iter()
308        .map(|&frequency_hz| casa_atmosphere::absorption::channel_totals(atmosphere, frequency_hz))
309        .collect()
310}

312#[cfg(test)]
313mod tests {
314    use casa_atmosphere::ProfileParameters;
315
316    use super::AtmosphereSpec;
317
318    /// `sm.setnoise`'s Python defaults (`relhum=20`, the quantities `[]`)
319    /// resolve to the site CASA logs for them ("altitude=5000m,
320    /// Pground=560mb, Tground=270K, humidity= 20%, water scale
321    /// height=2000m"), which `casa-atmosphere`'s defaults are and the
322    /// Phase 4c probe ran with.
323    #[test]
324    fn setnoise_defaults_resolve_to_casas_site() {
325        let python = AtmosphereSpec {
326            relative_humidity_percent: 20.0,
327            ..AtmosphereSpec::default()
328        };
329        assert_eq!(python.resolved(270.0), ProfileParameters::default());
330        assert_eq!(
331            AtmosphereSpec::default().resolved(270.0),
332            ProfileParameters::default()
333        );
334    }
335
336    /// Values at CASA's thresholds are unset; values past them are held in
337    /// single precision, as the corruptor reads them (`asFloat`), the water
338    /// scale height in kilometres.
339    #[test]
340    fn site_values_past_casas_thresholds_are_taken_in_single_precision() {
341        let thresholds = AtmosphereSpec {
342            ground_pressure_hpa: 100.0,
343            relative_humidity_percent: 0.0,
344            altitude_m: 0.0,
345            water_scale_height_m: 100.0,
346        };
347        assert_eq!(thresholds.resolved(270.0), ProfileParameters::default());
348        let given = AtmosphereSpec {
349            ground_pressure_hpa: 555.1,
350            relative_humidity_percent: 33.3,
351            altitude_m: 5058.7,
352            water_scale_height_m: 1234.5,
353        };
354        let resolved = given.resolved(269.9);
355        let single = |value: f64| f64::from(value as f32);
356        assert_eq!(resolved.ground_pressure_hpa, single(555.1));
357        assert_eq!(resolved.relative_humidity_percent, single(33.3));
358        assert_eq!(resolved.altitude_m, single(5058.7));
359        assert_eq!(resolved.water_vapour_scale_height_km, single(1.2345));
360        assert_eq!(resolved.ground_temperature_k, single(269.9));
361        assert_ne!(resolved.ground_pressure_hpa, 555.1);
362    }
