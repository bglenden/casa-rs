2//! `casars-imager` — native CASA-compatible imaging CLI shell.
3
4use std::process::ExitCode;
5
6use casa_tasks::imaging::{
7    ImagingExecutionOptions, RUN_SUMMARY_VARIABLE, task_front_end_with_options,
8};
9
10/// Freed working buffers leave the process at once, so a run's footprint
11/// follows the memory its phases admit.
12#[global_allocator]
13static ALLOCATOR: casa_imaging_application::ReturningAllocator =
14    casa_imaging_application::ReturningAllocator;
15
16fn main() -> ExitCode {
17    if let Err(error) = casars_cli::interrupt::install() {
18        eprintln!("casars-imager: install interrupt handler: {error}");
19        return ExitCode::FAILURE;
20    }
21    let mut options = ImagingExecutionOptions::from_env();
22    options.cancel = casars_cli::interrupt::token().clone();
23    options.write_run_summary =
24        std::env::var_os(RUN_SUMMARY_VARIABLE).is_some_and(|value| value == "1");
25    let front_end = task_front_end_with_options(options);
26    casars_cli::logging::run_front_end(
27        &front_end,
28        std::env::args_os().skip(1),
29        casars_cli::logging::CasaLoggingConfig::from_env(),
30        std::env::set_current_dir,
31        &mut std::io::stdout(),
32        std::io::stderr(),
33    )
34}
