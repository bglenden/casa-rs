2//! Change row counts in an existing directory without rebinding columns.
3//! Stage modified files using their existing manager sequence numbers. A
4//! manager that only extends an index, header or data file needs no copy
5//! of its existing payload.
6

21pub(crate) enum Rows<'a> {
22    Append {
23        count: usize,
24        defaults: &'a RecordValue,
25    },
26    Keep(&'a [usize]),
27}
28
29impl Rows<'_> {
30    fn len(&self, old: u64) -> Result<u64, StorageError> {
31        match *self {
32            Rows::Append { count, .. } => old
33                .checked_add(count as u64)
34                .ok_or_else(|| StorageError::FormatMismatch("table row count overflow".into())),
35            Rows::Keep(rows) => Ok(rows.len() as u64),
36        }
37    }
38}
39
40enum Extension {
41    Index {
42        path: PathBuf,
43        offset: u64,
44        bytes: Vec<u8>,
45    },
46    Payload {
47        path: PathBuf,
48        len: u64,
49    },
50}
51
52pub(crate) struct StagedEdit<'a> {
53    pub(super) table: &'a Path,
54    pub(super) staged: tempfile::TempDir,
55    extensions: Vec<Extension>,
56}
57
58impl<'a> StagedEdit<'a> {
59    pub(crate) fn stage(
60        table: &'a Path,
61        mut dat: TableDatContents,
62        old_rows: u64,
63        rows: Rows<'_>,
64    ) -> Result<Self, StorageError> {
65        let new_rows = rows.len(old_rows)?;
66        let mut edit = Self::new(table)?;
67        let order = if dat.big_endian {
68            ByteOrder::BigEndian
69        } else {
70            ByteOrder::LittleEndian
71        };
72        let private_keywords = &dat.table_desc.private_keywords;
73        for manager in &mut dat.column_set.data_managers {
74            let columns: Vec<_> = dat
75                .table_desc
76                .columns
77                .iter()
78                .zip(&dat.column_set.columns)
79                .filter(|(_, entry)| entry.dm_seq_nr == manager.seq_nr)
80                .map(|(column, _)| column.clone())
81                .collect();
82            edit.stage_manager(
83                manager,
84                &columns,
85                &rows,
86                old_rows,
87                new_rows,
88                (order, private_keywords),
89            )?;
90        }
91        dat.nrrow = new_rows;
92        dat.column_set.nrrow = new_rows;
93        edit.stage_control(&dat)?;
94        Ok(edit)
95    }
96
97    pub(super) fn new(table: &'a Path) -> Result<Self, StorageError> {
98        Ok(Self {
99            table,
100            staged: tempfile::Builder::new()
101                .prefix(".casa-rs-edit-")
102                .tempdir_in(table)?,
103            extensions: Vec::new(),
104        })
105    }
106
107    pub(super) fn stage_control(&self, dat: &TableDatContents) -> Result<(), StorageError> {
108        let dat = crate::table_path::stored_table_dat_keywords(dat, self.table);
109        fs::write(
110            self.staged.path().join(TABLE_CONTROL_FILE),
111            table_control::serialize_table_dat(&dat)?,
112        )?;
113        Ok(())
114    }
115
116    fn stage_manager(
117        &mut self,
118        manager: &mut DataManagerEntry,
119        columns: &[ColumnDescContents],
120        rows: &Rows<'_>,
121        old_rows: u64,
122        new_rows: u64,
123        layout: (ByteOrder, &RecordValue),
124    ) -> Result<(), StorageError> {
125        let (order, private_keywords) = layout;
126        let name = format!("table.f{}", manager.seq_nr);
127        let original = self.table.join(&name);
128        let destination = self.staged.path().join(&name);
129        match manager.type_name.as_str() {
130            "IncrementalStMan" if matches!(rows, Rows::Append { .. }) && old_rows > 0 => {
131                let (offset, bytes) =
132                    incremental_stman::appended_ism_index(&original, old_rows, new_rows)?;
133                self.extensions.push(Extension::Index {
134                    path: original,
135                    offset,
136                    bytes,
137                });
138            }
139            "TiledColumnStMan" | "TiledShapeStMan" | "TiledCellStMan" | "TiledDataStMan"
140                if matches!(rows, Rows::Append { .. }) =>
141            {
142                let fixed = columns
143                    .iter()
144                    .any(|column| column.is_array && !column.shape.is_empty());
145                if fixed
146                    && matches!(
147                        manager.type_name.as_str(),
148                        "TiledShapeStMan" | "TiledCellStMan"
149                    )
150                {
151                    let values = self.manager_rows(columns, rows, old_rows)?;
152                    let tile = tiled_stman::tiled_tile_shape(self.table, manager.seq_nr)?;
153                    let name = tiled_stman::append::manager_name(&original)?;
154                    tiled_stman::save_tiled_columns(
155                        self.staged.path(),
156                        manager.seq_nr,
157                        &manager.type_name,
158                        columns,
159                        &values,
160                        order == ByteOrder::BigEndian,
161                        tile.as_deref(),
162                        &name,
163                        private_keywords,
164                    )?;
165                    tiled_stman::append::set_cache(
166                        &destination,
167                        tiled_stman::append::cache_size(&original)?,
168                    )?;
169                    return Ok(());
170                }
171                if let Some((path, len)) = tiled_stman::append::stage_append(
172                    self.table,
173                    self.staged.path(),
174                    manager.seq_nr,
175                    old_rows,
176                    new_rows,
177                    (columns, private_keywords),
178                )? {
179                    self.extensions.push(Extension::Payload { path, len });
180                }
181            }
182            "IncrementalStMan" if matches!(rows, Rows::Append { .. }) => {
183                let values = self.manager_rows(columns, rows, old_rows)?;
184                let (endian, size, cache) = incremental_stman::ism_layout(&original)?;
185                incremental_stman::write_ism_with_layout(
186                    &destination,
187                    columns,
188                    &values,
189                    endian,
190                    Some(size),
191                    cache,
192                )?;
193            }
194            "StandardStMan" | "StManAipsIO" => {
195                let values = self.manager_rows(columns, rows, old_rows)?;
196                if manager.type_name == "StandardStMan" {
197                    let name = standard_stman::dm_name_from_blob(&manager.data).unwrap_or_default();
198                    manager.data = standard_stman::rewrite_ssm_rows(
199                        &original,
200                        &destination,
201                        columns,
202                        &values,
203                        &name,
204                    )?;
205                } else {
206                    let name = stman_aipsio::name_from_file(&original).unwrap_or_default();
207                    stman_aipsio::write_named_stman_file(
208                        &destination,
209                        manager.seq_nr,
210                        columns,
211                        &values,
212                        ByteOrder::BigEndian,
213                        &name,
214                    )?;
215                }
216            }
217            // Virtual columns derive their row count from their sources.
218            other if super::is_virtual_engine(other) => {}
219            other => return Err(StorageError::UnsupportedDataManager(other.to_string())),
220        }
221        Ok(())
222    }
223
224    fn manager_rows(
225        &self,
226        columns: &[ColumnDescContents],
227        rows: &Rows<'_>,
228        old_rows: u64,
229    ) -> Result<Vec<RecordValue>, StorageError> {
230        let names: Vec<_> = columns
231            .iter()
232            .map(|column| column.col_name.as_str())
233            .collect();
234        let mut existing = CompositeStorage
235            .load_selected_columns_with_row_hint(self.table, &names, Some(old_rows))?
236            .rows;
237        match *rows {
238            Rows::Append { count, defaults } => {
239                let row = RecordValue::new(
240                    names
241                        .iter()
242                        .filter_map(|name| {
243                            defaults
244                                .get(name)
245                                .map(|value| RecordField::new(*name, value.clone()))
246                        })
247                        .collect(),
248                );
249                existing.extend(std::iter::repeat_n(row, count));
250                Ok(existing)
251            }
252            Rows::Keep(kept) => {
253                let mut original = existing.into_iter();
254                let mut next = 0;
255                let mut result = Vec::with_capacity(kept.len());
256                for &row in kept {
257                    result.push(original.nth(row - next).ok_or_else(|| {
258                        StorageError::FormatMismatch(format!("missing row {row} during remove"))
259                    })?);
260                    next = row + 1;
261                }
262                Ok(result)
263            }
264        }
265    }
