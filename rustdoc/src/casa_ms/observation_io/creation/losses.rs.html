2//! What a source holds that a conversion cannot carry, found before
3//! anything is written.
4//!
5//! The inventory is of whole tables and columns: an MSv2's `SIGMA`,
6//! `FLAG_ROW` flags, `FLAG_CATEGORY` cells, `ARRAY_ID`s other than 0,
7//! `FEED` receptors that vary in time (of which the catalogue keeps the
8//! last row, as casacore's `MSIter` does), MAIN columns outside the
9//! boundary's roles and subtables the catalogue does not carry (system
10//! calibration, weather, flag commands, history, Doppler, frequency
11//! offsets, and any further subtable a keyword names), and into an MSv4,
12//! which holds less of them than the catalogue, its `SOURCE` rows,
13//! `PROCESSOR_ID`s other than 0, a `PROCESSOR` of more than one row and
14//! `STATE` rows that say more than their intents; an MSv4's sub-datasets
15//! and main-dataset variables the writer does not write; and in either the
16//! data groups not selected; and, into an MSv4, antennas no partition's
17//! baselines name. What an MSv2 destination's subtables write as MSv2's
18//! default is found by writing them (`super::msv2_tables::rehearse`).
19//! What pointing carries is
20//! decided by [`super::pointing_plan`]. Columns of the subtables the
21//! catalogue does carry (an antenna's offset, a field's delay direction, a
22//! spectral window's resolution) are not inventoried, but an ephemeris
23//! table's are (`ephemeris_tables`), since the catalogue carries only the
24//! body's directions, distances and velocities. Fields whose direction is a
25//! polynomial in time are refused: the catalogue would hold a constant term
26//! as the phase centre.
27

38/// The data groups to convert into an MSv4: `requested`, each present in
39/// every partition, or every group of the source with `base` first.
40#[cfg(feature = "msv4")]
41pub fn selected_groups(
42    catalogue: &Catalogue,
43    requested: Option<&[String]>,
44) -> Result<Vec<String>, ObservationError> {
45    let groups: Vec<String> = match requested {
46        Some(requested) => requested.to_vec(),
47        None => {
48            let mut groups = vec!["base".to_string()];
49            for entry in &catalogue.partitions {
50                for group in &entry.data_groups {
51                    if !groups.contains(&group.name) {
52                        groups.push(group.name.clone());
53                    }
54                }
55            }
56            groups
57        }
58    };
59    if groups.is_empty() {
60        return Err(request_error(
61            "a conversion selects at least one data group",
62        ));
63    }
64    if let Some(group) = groups
65        .iter()
66        .enumerate()
67        .find_map(|(index, group)| groups[..index].contains(group).then_some(group))
68    {
69        return Err(request_error(format!(
70            "data group {group} is selected twice"
71        )));
72    }
73    for entry in &catalogue.partitions {
74        if let Some(missing) = groups
75            .iter()
76            .find(|group| entry.data_group(group).is_none())
77        {
78            return Err(request_error(format!(
79                "partition {} has no data group {missing}",
80                entry.spectral_window
81            )));
82        }
83    }
84    Ok(groups)
85}

87/// The store a conversion creates, which decides what it must refuse.
88#[derive(Clone, Copy, Debug, PartialEq, Eq)]
89pub enum Store {
90    /// An MSv2, which writes every catalogue field.
91    Msv2,
92    /// An MSv4, which writes the fields its datasets name.
93    Msv4,
94}

96/// Everything `source` holds that converting `groups` into `destination`
97/// would not carry.
98pub fn inventory(
99    source: &Path,
100    catalogue: &Catalogue,
101    groups: &[String],
102    destination: Store,
103) -> Result<Vec<Loss>, ObservationError> {
104    let mut losses = Vec::new();
105    if destination == Store::Msv4
106        && catalogue
107            .sources
108            .iter()
109            .flat_map(|source| &source.records)
110            .any(|record| record.model.is_some())
111    {
112        losses.push(Loss::Column("SOURCE SOURCE_MODEL".into()));
113    }
114    for entry in &catalogue.partitions {
115        for group in &entry.data_groups {
116            let loss = Loss::DataGroup(group.name.clone());
117            if !groups.contains(&group.name) && !losses.contains(&loss) {
118                losses.push(loss);
119            }
120        }
121    }
122    if source.join("zarr.json").is_file() {
123        msv4(source, &mut losses)?;
124        // An MSv4 holds no sigma: an MSv2 destination states one.
125        if destination == Store::Msv2 {
126            losses.push(Loss::SigmaDerived);
127        }
128    } else {
129        msv2(source, catalogue, destination, &mut losses)?;
130    }
131    losses.sort();
132    losses.dedup();
133    Ok(losses)
134}

136/// The most [`inventory`] holds reading an MSv2 of `catalogue`: a block of
137/// MAIN's row keys, and a block of `FLAG_CATEGORY` cells of three
138/// categories (casacore's `FLAG_CMD`, `ORIGINAL` and `USER`) of its widest
139/// partition.
140pub(super) fn scan_bytes(catalogue: &Catalogue) -> usize {
141    let keys = ROW_CHUNK * (size_of::<usize>() + size_of::<bool>() + 2 * size_of::<i32>());
142    let widest = |each: fn(&casa_observation::PartitionEntry) -> usize| {
143        catalogue.partitions.iter().map(each).max().unwrap_or(0)
144    };
145    let cell = widest(|entry| entry.frequencies_hz.len() * entry.correlations.len());
146    let categories = CATEGORY_CHUNK * (3 * cell + size_of::<Option<casa_core::ArrayValue>>());
147    keys.max(categories)
148}

150/// The antennas of `catalogue` that no partition's baselines name, as a
151/// loss when there are any: an MSv4 dataset holds only its baselines'
152/// antennas.
153#[cfg(feature = "msv4")]
154pub fn unused_antennas(catalogue: &Catalogue) -> Option<Loss> {
155    let mut named = vec![false; catalogue.antennas.len()];
156    for baseline in catalogue
157        .partitions
158        .iter()
159        .flat_map(|entry| &entry.baselines)
160    {
161        for antenna in [baseline.antenna1, baseline.antenna2] {
162            if let Some(named) = named.get_mut(antenna.0 as usize) {
163                *named = true;
164            }
165        }
166    }
167    let unused = named.iter().filter(|named| !**named).count();
168    (unused > 0).then_some(Loss::UnusedAntennas(unused))
169}

171/// That the last samples of `ephemeris` are extrapolated rather than the
172/// source's ([`FieldEphemeris::extrapolated`]), as a loss when they are: the
173/// destination holds them as samples like any other.
174pub(super) fn extrapolated(ephemeris: &FieldEphemeris) -> Option<Loss> {
175    let count = ephemeris.extrapolated();
176    (count > 0).then(|| {
177        Loss::Column(format!(
178            "the ephemeris of {}: that its last {} extrapolated, not the source's (continuing \
179             the line through the two before to reach the last time's TDB)",
180            ephemeris.name(),
181            if count == 1 {
182                "sample is".to_string()
183            } else {
184                format!("{count} samples are")
185            }
186        ))
187    })
188}

190/// Refuse the first of `losses` whose kind is not in `accept`.
191pub fn refuse_unaccepted(losses: &[Loss], accept: &[LossKind]) -> Result<(), ObservationError> {
192    let refused: Vec<&Loss> = losses
193        .iter()
194        .filter(|loss| !accept.contains(&loss.kind()))
195        .collect();
196    if refused.is_empty() {
197        return Ok(());
198    }
199    let mut kinds: Vec<LossKind> = refused.iter().map(|loss| loss.kind()).collect();
200    kinds.sort_unstable();
201    kinds.dedup();
202    let described: Vec<String> = refused.iter().map(ToString::to_string).collect();
203    let named: Vec<&str> = kinds.iter().map(|kind| kind.name()).collect();
204    Err(request_error(format!(
205        "the conversion would lose information: {}; accept {} to convert anyway",
206        described.join("; "),
207        named.join(", ")
208    )))
209}

251/// Rows of `FLAG_ROW` read at a time.
252const ROW_CHUNK: usize = 1 << 18;
253
254fn msv2(
255    source: &Path,
256    catalogue: &Catalogue,
257    destination: Store,
258    losses: &mut Vec<Loss>,
259) -> Result<(), ObservationError> {
260    let ms = crate::MeasurementSet::open_retained_read(source).map_err(storage)?;
261    refuse_polynomial_fields(&ms, catalogue, destination)?;
262    ephemeris_tables(source, &ms, catalogue, destination, losses)?;
263    let main = ms.main_table();
264    let columns: Vec<String> = main
265        .schema()
266        .map(|schema| {
267            schema
268                .columns()
269                .iter()
270                .map(|column| column.name().to_string())
271                .collect()
272        })
273        .unwrap_or_default();
274    if columns.iter().any(|name| name == "FLOAT_DATA") && !columns.iter().any(|name| name == "DATA")
275    {
276        return Err(ObservationError::Unsupported {
277            what: "converting single-dish data (FLOAT_DATA without DATA)",
278        });
279    }
280    for column in &columns {
281        match column.as_str() {
282            "SIGMA" | "SIGMA_SPECTRUM" if destination == Store::Msv2 => {}
283            "SIGMA" | "SIGMA_SPECTRUM" => losses.push(Loss::Sigma),
284            "FLAG_CATEGORY" => {
285                if flag_categories_held(main)? {
286                    losses.push(Loss::FlagCategories);
287                }
288            }
289            name if !CARRIED_COLUMNS.contains(&name) => losses.push(Loss::Column(name.to_string())),
290            _ => {}
291        }
292    }
293    if destination == Store::Msv2 && !columns.iter().any(|name| name == "SIGMA") {
294        losses.push(Loss::SigmaDerived);
295    }
296    if destination == Store::Msv4 && columns.iter().any(|name| name == "WEIGHT_SPECTRUM") {
297        losses.push(Loss::Column(
298            "WEIGHT (the row summary beside WEIGHT_SPECTRUM)".to_string(),
299        ));
300    }
301    main_keys(main, destination, losses)?;
302    let reads = crate::observation_io::msv2::catalogue::feed_reads(&catalogue.partitions);
303    if crate::observation_io::msv2::catalogue::read_feeds(&ms)?
304        .varying(&reads)
305        .is_some()
306    {
307        losses.push(Loss::Column(
308            "FEED receptors' time dependence (the last row of each antenna, feed and window \
309             is kept)"
310                .to_string(),
311        ));
312    }
313    subtables(&ms, destination, losses)?;
314    if destination == Store::Msv2 {
315        super::preservation::inventory(&ms, losses);
316    }
317    Ok(())
318}

320/// The fields `destination` writes: every `FIELD` row into an MSv2, and
321/// into an MSv4 those `catalogue`'s partitions name (an MSv2 catalogue's
322/// fields are its `FIELD` rows).
323fn written_fields(catalogue: &Catalogue, destination: Store, rows: usize) -> Vec<bool> {
324    let mut named = vec![destination == Store::Msv2; rows];
325    for field in catalogue.partitions.iter().flat_map(|entry| &entry.fields) {
326        if let Some(named) = named.get_mut(field.0 as usize) {
327            *named = true;
328        }
329    }
330    named
331}

333/// Refuse fields whose directions are polynomials in time (`NUM_POLY` above
334/// 0) among those `destination` writes, as XRADIO's converter refuses the
335/// fields it converts: the catalogue holds one constant direction.
336fn refuse_polynomial_fields(
337    ms: &crate::MeasurementSet,
338    catalogue: &Catalogue,
339    destination: Store,
340) -> Result<(), ObservationError> {
341    let fields = ms.field().map_err(storage)?;
342    let named = written_fields(catalogue, destination, fields.row_count());
343    for row in (0..fields.row_count()).filter(|row| named[*row]) {
344        if fields.num_poly(row).map_err(storage)? > 0 {
345            return Err(ObservationError::Unsupported {
346                what: "converting fields whose direction is a polynomial in time (FIELD NUM_POLY)",
347            });
348        }
349    }
350    Ok(())
351}

353/// What the ephemeris tables of the fields `destination` writes hold that
354/// the catalogue does not carry: columns other than `MJD`, `RA`, `DEC`,
355/// `Rho` and `RadVel`, and keywords other than those an MSv2 destination
356/// writes. Into an MSv4, also the tables' `NAME` (an MSv4 ephemeris is
357/// named by its source) and the fields' `TIME`, at which an image without a
358/// phase centre is centred on the body (#987).
359fn ephemeris_tables(
360    source: &Path,
361    ms: &crate::MeasurementSet,
362    catalogue: &Catalogue,
363    destination: Store,
364    losses: &mut Vec<Loss>,
365) -> Result<(), ObservationError> {
366    let ids = crate::ephemeris::field_ephemeris_ids(ms).map_err(storage)?;
367    let named = written_fields(catalogue, destination, ids.len());
368    let mut followed: Vec<i32> = ids
369        .iter()
370        .zip(&named)
371        .filter_map(|(id, named)| id.filter(|_| *named))
372        .collect();
373    followed.sort_unstable();
374    followed.dedup();
375    if !followed.is_empty() && destination == Store::Msv4 {
376        losses.push(Loss::Column(
377            "FIELD TIME of fields that follow an ephemeris (when an image without a phase \
378             centre is centred on the body)"
379                .to_string(),
380        ));
381    }
382    for id in followed {
383        let path = crate::ephemeris::find_field_ephemeris(source, id).map_err(storage)?;
384        let table =
385            casa_tables::Table::open(casa_tables::TableOptions::new(&path)).map_err(storage)?;
386        let name = path
387            .strip_prefix(source)
388            .unwrap_or(&path)
389            .display()
390            .to_string();
391        let columns: Vec<String> = table
392            .schema()
393            .map(|schema| {
394                schema
395                    .columns()
396                    .iter()
397                    .map(|column| column.name().to_string())
398                    .filter(|column| {
399                        !["MJD", "RA", "DEC", "Rho", "RadVel"].contains(&column.as_str())
400                    })
401                    .collect()
402            })
403            .unwrap_or_default();
404        if !columns.is_empty() {
405            losses.push(Loss::Column(format!("{name} {}", columns.join(", "))));
406        }
407        let mut keywords: Vec<String> = table
408            .keywords()
409            .fields()
410            .iter()
411            .map(|field| field.name.clone())
412            .filter(|keyword| {
413                !super::msv2_ephemeris::KEYWORDS.contains(&keyword.as_str())
414                    || (destination == Store::Msv4 && keyword == "NAME")
415            })
416            .collect();
417        keywords.sort();
418        if !keywords.is_empty() {
419            losses.push(Loss::Column(format!(
420                "{name} keywords {}",
421                keywords.join(", ")
422            )));
423        }
424    }
425    Ok(())
426}

428/// MAIN's keys the destination does not hold: a set `FLAG_ROW`, an
429/// `ARRAY_ID` above the 0 an MSv2 destination writes, and into an MSv4,
430/// which has neither, a `PROCESSOR_ID` above 0 (an MSv2 destination writes
431/// each partition's), read in row chunks. A negative id names nothing:
432/// casacore's UVFITS filler writes `PROCESSOR_ID` -1 in every row.
433fn main_keys(
434    main: &casa_tables::Table,
435    destination: Store,
436    losses: &mut Vec<Loss>,
437) -> Result<(), ObservationError> {
438    let rows = main.row_count();
439    let (mut flags, mut arrays, mut processors) = (Vec::new(), Vec::new(), Vec::new());
440    let (mut flagged, mut array, mut processor) = (false, false, false);
441    for start in (0..rows).step_by(ROW_CHUNK) {
442        let chunk: Vec<usize> = (start..rows.min(start + ROW_CHUNK)).collect();
443        flags.clear();
444        arrays.clear();
445        processors.clear();
446        let mut destinations = [
447            RequiredScalarColumnDestination::new(
448                "FLAG_ROW",
449                RequiredScalarColumnValuesMut::Bool(&mut flags),
450            ),
451            RequiredScalarColumnDestination::new(
452                "ARRAY_ID",
453                RequiredScalarColumnValuesMut::Int32(&mut arrays),
454            ),
455            RequiredScalarColumnDestination::new(
456                "PROCESSOR_ID",
457                RequiredScalarColumnValuesMut::Int32(&mut processors),
458            ),
459        ];
460        main.required_scalar_columns_for_rows_into(&chunk, &mut destinations)
461            .map_err(storage)?;
462        flagged |= flags.iter().any(|flag| *flag);
463        array |= arrays.iter().any(|id| *id > 0);
464        processor |= processors.iter().any(|id| *id > 0);
465        if flagged && array && processor {
466            break;
467        }
468    }
469    if flagged && destination == Store::Msv4 {
470        losses.push(Loss::RowFlags);
471    }
472    if array {
473        losses.push(Loss::Column("ARRAY_ID".to_string()));
474    }
475    if processor && destination == Store::Msv4 {
476        losses.push(Loss::Column("PROCESSOR_ID".to_string()));
477    }
478    Ok(())
479}

481/// Subtables, and subtable content, the destination does not hold: the
482/// uncarried subtables with rows, any further table a keyword names, and
483/// into an MSv4 (an MSv2 destination writes them from the catalogue)
484/// `SOURCE` rows, a `PROCESSOR` of more than one row or of another type
485/// than a correlator, and `STATE` rows that say more than their `OBS_MODE`.
486fn subtables(
487    ms: &crate::MeasurementSet,
488    destination: Store,
489    losses: &mut Vec<Loss>,
490) -> Result<(), ObservationError> {
491    for id in UNCARRIED_SUBTABLES {
492        if ms.subtable(*id).is_some_and(|table| table.row_count() > 0) {
493            losses.push(Loss::Table(id.name().to_string()));
494        }
495    }
496    let main = ms.main_table();
497    let standard: Vec<&str> = SubtableId::ALL_REQUIRED
498        .iter()
499        .chain(UNCARRIED_SUBTABLES)
500        .chain(&[SubtableId::Pointing, SubtableId::Source])
501        .map(SubtableId::name)
502        .collect();
503    for field in main.keywords().fields() {
504        if matches!(field.value, Value::TableRef(_)) && !standard.contains(&field.name.as_str()) {
505            losses.push(Loss::Table(field.name.clone()));
506        }
507    }
508    if destination == Store::Msv2 {
509        return Ok(());
510    }
511    if ms
512        .subtable(SubtableId::Source)
513        .is_some_and(|table| table.row_count() > 0)
514    {
515        losses.push(Loss::Table(SubtableId::Source.name().to_string()));
516    }
517    let processors = ms.processor().map_err(storage)?;
518    let correlator = |row| {
519        processors
520            .string(row, "TYPE")
521            .map(|kind| kind == "CORRELATOR")
522            .map_err(storage)
523    };
524    if processors.row_count() > 1 || (processors.row_count() == 1 && !correlator(0)?) {
525        losses.push(Loss::Table("PROCESSOR".to_string()));
526    }
527    let state = ms.state().map_err(storage)?;
528    for row in 0..state.row_count() {
529        let plain = state.i32(row, "SUB_SCAN").map_err(storage)? == 0
530            && state.f64(row, "CAL").map_err(storage)? == 0.0
531            && state.f64(row, "LOAD").map_err(storage)? == 0.0
532            && state.bool(row, "SIG").map_err(storage)?
533            && !state.bool(row, "REF").map_err(storage)?;
534        if !plain {
535            losses.push(Loss::Column(
536                "STATE SUB_SCAN, CAL, LOAD, SIG and REF".to_string(),
537            ));
538            break;
539        }
540    }
541    Ok(())
542}

544/// Rows of `FLAG_CATEGORY` read at once: an undefined or empty cell costs
545/// nothing to read, and a held one ends the search, so this bounds the
546/// cells read with the first held one (256 cells of three categories of
547/// 32,768 channels and 4 correlations are about 100 MB).
548const CATEGORY_CHUNK: usize = 256;
549
550/// Whether MAIN's `FLAG_CATEGORY` holds categories: a defined cell with
551/// elements in any row. The first and last rows are read first, since
552/// writers usually fill the column whole or leave it undefined throughout.
553fn flag_categories_held(main: &casa_tables::Table) -> Result<bool, ObservationError> {
554    let rows = main.row_count();
555    if rows == 0 {
556        return Ok(false);
557    }
558    let column = main.column_accessor("FLAG_CATEGORY").map_err(storage)?;
559    // Uncached: the inventory then reads FLAG_ROW from the lazy table.
560    let held = |chunk: &[usize]| {
561        column
562            .array_cells_owned_uncached(chunk)
563            .map(|cells| cells.iter().flatten().any(|cell| !cell.is_empty()))
564            .map_err(storage)
565    };
566    if held(&[0, rows - 1])? {
567        return Ok(true);
568    }
569    for start in (1..rows.saturating_sub(1)).step_by(CATEGORY_CHUNK) {
570        let chunk: Vec<usize> = (start..(rows - 1).min(start + CATEGORY_CHUNK)).collect();
571        if held(&chunk)? {
572            return Ok(true);
573        }
574    }
575    Ok(false)
576}

596fn msv4(source: &Path, losses: &mut Vec<Loss>) -> Result<(), ObservationError> {
597    let datasets = std::fs::read_dir(source).map_err(storage)?;
598    for dataset in datasets {
599        let dataset = dataset.map_err(storage)?.path();
600        if !dataset.join("zarr.json").is_file() {
601            continue;
602        }
603        for child in std::fs::read_dir(&dataset).map_err(storage)? {
604            let child = child.map_err(storage)?.path();
605            let Some(name) = child
606                .file_name()
607                .map(|name| name.to_string_lossy().into_owned())
608            else {
609                continue;
610            };
611            if !child.join("zarr.json").is_file() {
612                continue;
613            }
614            let is_group = std::fs::read_to_string(child.join("zarr.json")).is_ok_and(|metadata| {
615                metadata.contains("\"node_type\": \"group\"")
616                    || metadata.contains("\"node_type\":\"group\"")
617            });
618            if is_group {
619                if name == "pointing_xds" {
620                    pointing_variables(&child, losses)?;
621                } else if name.starts_with("field_and_source_") {
622                    ephemeris_variables(&child, losses)?;
623                } else if name != "antenna_xds" {
624                    losses.push(Loss::Table(name));
625                }
626            } else if !CARRIED_VARIABLES.contains(&name.as_str()) && !name.starts_with("VISIBILITY")
627            {
628                losses.push(Loss::Column(name));
629            }
630        }
631    }
632    Ok(())
633}

635/// The optional variables of XRADIO's `field_and_source_ephemeris`
636/// (from the table's `NP_ang`, `NP_dist`, `DiskLong`, `DiskLat`, `Sl_lon`,
637/// `Sl_lat`, `r`, `rdot` and `phang`), which the catalogue does not carry.
638const EPHEMERIS_EXTRAS: [&str; 7] = [
639    "NORTH_POLE_POSITION_ANGLE",
640    "NORTH_POLE_ANGULAR_DISTANCE",
641    "SUB_OBSERVER_DIRECTION",
642    "SUB_SOLAR_DIRECTION",
643    "SUB_SOLAR_DISTANCE",
644    "HELIOCENTRIC_RADIAL_VELOCITY",
645    "OBSERVER_PHASE_ANGLE",
646];
647
648/// A field dataset's ephemeris variables a conversion does not carry
649/// ([`EPHEMERIS_EXTRAS`]), as one column loss; and an ephemeris in XRADIO's
650/// interpolated layout (no `time_ephemeris`), whose samples start at the
651/// dataset's first time, so that an MSv2 written from them cannot be
652/// evaluated there (`MeasComet::fillMeas` puts a time on a table's first
653/// row at row −1 for some roundings). Their end, the last time, the
654/// reader extends ([`extrapolated`]).
655fn ephemeris_variables(path: &Path, losses: &mut Vec<Loss>) -> Result<(), ObservationError> {
656    let name = |path: &Path| {
657        path.file_name()
658            .map(|name| name.to_string_lossy().into_owned())
659            .unwrap_or_default()
660    };
661    let label = format!(
662        "{}/{}",
663        path.parent().map(name).unwrap_or_default(),
664        name(path)
665    );
666    let moving = std::fs::read_to_string(path.join("zarr.json"))
667        .is_ok_and(|metadata| metadata.contains("field_and_source_ephemeris"));
668    if moving && !path.join("time_ephemeris").join("zarr.json").is_file() {
669        losses.push(Loss::Column(format!(
670            "{label} ephemeris samples (interpolated by XRADIO onto the visibility times; the \
671             table's own samples are not in the store, nor any before the first time, so \
672             casacore's MeasComet cannot evaluate the first time)"
673        )));
674    }
675    let mut lost = Vec::new();
676    for child in std::fs::read_dir(path).map_err(storage)? {
677        let child = child
678            .map_err(storage)?
679            .file_name()
680            .to_string_lossy()
681            .into_owned();
682        if EPHEMERIS_EXTRAS.contains(&child.as_str()) {
683            lost.push(child);
684        }
685    }
686    if !lost.is_empty() {
687        lost.sort();
688        losses.push(Loss::Column(format!("{label} {}", lost.join(", "))));
689    }
690    Ok(())
691}

693/// A `pointing_xds`'s variables a conversion does not carry, as table
694/// losses: it carries `POINTING_BEAM` and its coordinates.
695fn pointing_variables(path: &Path, losses: &mut Vec<Loss>) -> Result<(), ObservationError> {
696    const CARRIED: [&str; 6] = [
697        "POINTING_BEAM",
698        "time_pointing",
699        "time",
700        "antenna_name",
701        "local_sky_dir_label",
702        "n_polynomial",
703    ];
704    for child in std::fs::read_dir(path).map_err(storage)? {
705        let name = child
706            .map_err(storage)?
707            .file_name()
708            .to_string_lossy()
709            .into_owned();
710        if name != "zarr.json" && !CARRIED.contains(&name.as_str()) {
711            losses.push(Loss::Table(format!("pointing_xds {name}")));
712        }
713    }
714    Ok(())
715}
