17/// One snapshot source and its bounded-content budget.
18///
19/// The content budget is the sole physical blocking authority.
20#[derive(Debug, Clone, PartialEq, Eq)]
21pub struct ObservationSourceBinding {
22    measurement_set: usize,
23    content_budget: SelectedObservationContentBudget,
24    ephemeris: Option<Arc<crate::SelectedObservationEphemeris>>,
25    pointing_query_domain: Option<SelectedPointingQueryDomain>,
26}

28/// Check that a binding carries the ephemeris the compiled geometry tracks,
29/// and that its reference data fits its content budget.
30fn check_reference_data(
31    problem: &CompiledProblem,
32    binding: &ObservationSourceBinding,
33) -> Result<(), BoundSelectedObservationError> {
34    let expected = problem.geometry().ephemeris_reference();
35    let actual = binding.ephemeris_identity();
36    if actual != expected {
37        return Err(BoundSelectedObservationError::EphemerisReferenceMismatch {
38            measurement_set: binding.measurement_set,
39            expected,
40            actual,
41        });
42    }
43    let available_bytes = binding.content_budget.available_bytes();
44    let required_bytes = binding.reference_data_bytes();
45    if required_bytes > available_bytes {
46        return Err(BoundSelectedObservationError::ReferenceDataBudgetExceeded {
47            measurement_set: binding.measurement_set,
48            required_bytes,
49            available_bytes,
50        });
51    }
52    Ok(())
53}

55/// Find the one binding for a snapshot source.
56fn source_binding(
57    bindings: &[ObservationSourceBinding],
58    measurement_set: usize,
59) -> Result<&ObservationSourceBinding, BoundSelectedObservationError> {
60    let mut matches = bindings
61        .iter()
62        .filter(|binding| binding.measurement_set == measurement_set);
63    let binding = matches
64        .next()
65        .ok_or(BoundSelectedObservationError::MissingSourceBinding { measurement_set })?;
66    if matches.next().is_some() {
67        return Err(BoundSelectedObservationError::DuplicateSourceBinding { measurement_set });
68    }
69    Ok(binding)
70}

171/// Retained read-locked access to every source in one compiled selected observation.
172///
173/// Each traversal is one ordered stream of row blocks; completing the stream
174/// returns this access for the next traversal. Storage buffers and POINTING
175/// lookup primitives are owner-internal.
176///
177/// ```compile_fail
178/// use casa_ms::{SelectedObservationBuffer, SelectedObservationBufferRequest};
179///
180/// let _ = std::mem::size_of::<SelectedObservationBuffer>();
181/// let _ = std::mem::size_of::<SelectedObservationBufferRequest>();
182/// ```
183pub struct BoundSelectedObservation {
184    sources: Vec<BoundObservationSource>,
185}

187impl BoundSelectedObservation {
188    /// The distinct CASA aperture classes of the selected sources' antennas
189    /// in class order (`HetArrayConvFunc::findAntennaSizes`: one class per
190    /// distinct dish), empty when no antenna is an ALMA or ACA dish.
191    #[must_use]
192    pub fn antenna_response_classes(&self) -> Vec<casa_imaging_model::AntennaResponseClass> {
193        let mut classes = self
194            .sources
195            .iter()
196            .flat_map(|source| source.antenna_response_classes().iter().flatten().copied())
197            .collect::<Vec<_>>();
198        classes.sort_unstable();
199        classes.dedup();
200        classes
201    }

203    #[cfg(unix)]
204    pub(crate) fn single_source_content_requirements(
205        problem: &CompiledProblem,
206        measures: &SelectedObservationMeasures,
207        binding: &ObservationSourceBinding,
208    ) -> Result<SelectedObservationContentRequirements, BoundSelectedObservationError> {
209        let expected = problem.inputs().observation_snapshot().sources();
210        if expected.len() != 1 {
211            return Err(BoundSelectedObservationError::BindingSetMismatch);
212        }
213        let source = &expected[0];
214        if source.input_ordinal() != binding.measurement_set() {
215            return Err(BoundSelectedObservationError::MissingSourceBinding {
216                measurement_set: source.input_ordinal(),
217            });
218        }
219        let shared = Self::shared_bytes(measures, std::slice::from_ref(binding))?;
220        BoundObservationSource::content_requirements(problem, source, binding, measures, shared)
221            .map_err(|error| BoundSelectedObservationError::Source {
222                measurement_set: source.input_ordinal(),
223                error: Box::new(error),
224            })
225    }
226
227    /// The Measures provider and every binding's reference data, charged
228    /// once, to the first source.
229    fn shared_bytes(
230        measures: &SelectedObservationMeasures,
231        bindings: &[ObservationSourceBinding],
232    ) -> Result<SelectedObservationSharedBytes, BoundSelectedObservationError> {
233        let reference_data_retained_bytes =
234            bindings.iter().try_fold(0_usize, |bytes, binding| {
235                bytes
236                    .checked_add(binding.reference_data_bytes())
237                    .ok_or(BoundSelectedObservationError::ReferenceDataByteOverflow)
238            })?;
239        Ok(SelectedObservationSharedBytes::new(
240            measures.retained_bytes(),
241            reference_data_retained_bytes,
242        ))
243    }

245    /// Open every compiled source under retained read locks and its content budget.
246    ///
247    /// Caller binding order is irrelevant. Sources are retained and streamed only in the
248    /// compiler's canonical read-set order.
249    #[cfg(unix)]
250    pub fn open(
251        problem: &CompiledProblem,
252        measures: SelectedObservationMeasures,
253        bindings: Vec<ObservationSourceBinding>,
254    ) -> Result<Self, BoundSelectedObservationError> {
255        measures.validate_problem(problem)?;
256        let expected = problem.inputs().observation_snapshot().sources();
257        if bindings.len() != expected.len() {
258            return Err(BoundSelectedObservationError::BindingSetMismatch);
259        }
260        for source in expected {
261            check_reference_data(problem, source_binding(&bindings, source.input_ordinal())?)?;
262        }
263        let mut sources = Vec::with_capacity(expected.len());
264        let first_source_shared_bytes = Self::shared_bytes(&measures, &bindings)?;
265        for (source_index, source) in expected.iter().enumerate() {
266            let measurement_set = source.input_ordinal();
267            let binding = source_binding(&bindings, measurement_set)?;
268            let shared_bytes = if source_index == 0 {
269                first_source_shared_bytes
270            } else {
271                SelectedObservationSharedBytes::NONE
272            };
273            sources.push(
274                BoundObservationSource::open_with_measures(
275                    problem,
276                    source,
277                    &measures,
278                    shared_bytes,
279                    binding.content_budget,
280                    BoundObservationReferenceData::new(
281                        binding.ephemeris.as_ref(),
282                        binding.pointing_query_domain(),
283                    ),
284                )
285                .map_err(|error| BoundSelectedObservationError::Source {
286                    measurement_set,
287                    error: Box::new(error),
288                })?,
289            );
290        }
291        Ok(Self { sources })
292    }

309    /// Stream every selected row of every source, in canonical compiler order,
310    /// as refillable blocks.
311    #[must_use]
312    pub fn into_block_stream(
313        self,
314        problem: &CompiledProblem,
315    ) -> SelectedObservationBlockSource<'_> {
316        self.into_block_stream_with_window(problem, None)
317    }
318
319    /// Stream every selected row, reading only the channels whose
320    /// output-frame frequencies reach `frequency_bounds_hz`, with the
321    /// straddling channel kept at each edge so interpolation keeps both
322    /// partners. A row block with no such channel is skipped without reading
323    /// its payload.
324    #[must_use]
325    pub fn into_windowed_block_stream(
326        self,
327        problem: &CompiledProblem,
328        frequency_bounds_hz: [f64; 2],
329    ) -> SelectedObservationBlockSource<'_> {
330        self.into_block_stream_with_window(problem, Some(frequency_bounds_hz))
331    }
332
333    fn into_block_stream_with_window(
334        self,
335        problem: &CompiledProblem,
336        window: Option<[f64; 2]>,
337    ) -> SelectedObservationBlockSource<'_> {
338        let maximum_rows = self
339            .sources
340            .iter()
341            .map(BoundObservationSource::rows_per_block)
342            .max()
343            .unwrap_or(0);
344        SelectedObservationBlockSource {
345            problem,
346            observation: self,
347            source_index: 0,
348            row_replay: None,
349            selected_rows: 0,
350            exhausted: false,
351            maximum_rows,
352            window,
353        }
354    }

357/// Ordered refillable source for one retained selected-observation pass.
358pub struct SelectedObservationBlockSource<'a> {
359    problem: &'a CompiledProblem,
360    observation: BoundSelectedObservation,
361    source_index: usize,
362    row_replay: Option<SelectedRowReplay>,
363    /// Rows the predicate selected in the sources already walked.
364    selected_rows: u64,
365    exhausted: bool,
366    maximum_rows: usize,
367    window: Option<[f64; 2]>,
368}
369
370impl SelectedObservationBlockSource<'_> {
371    /// Exact row ceiling from the freshly opened source's physical content plan.
372    /// Consumers may reduce an already admitted scratch bound to this size.
373    pub const fn maximum_rows_per_block(&self) -> usize {
374        self.maximum_rows
375    }
376
377    /// Create one empty block sized for this stream.
378    #[must_use]
379    pub fn create_storage(&self) -> SelectedObservationBlock {
380        SelectedObservationBlock::new(self.maximum_rows)
381    }
382
383    /// Fill `block` with the next canonical row block and borrow it as filled;
384    /// `None` once every source is exhausted.
385    pub fn fill_next<'b>(
386        &mut self,
387        block: &'b mut SelectedObservationBlock,
388    ) -> Result<Option<FilledObservationBlock<'b>>, BoundObservationSourceError> {
389        if self.exhausted {
390            return Ok(None);
391        }
392        loop {
393            let Some(source) = self.observation.sources.get(self.source_index) else {
394                self.exhausted = true;
395                return Ok(None);
396            };
397            let logical_source = self
398                .problem
399                .selected_observation()
400                .read_set()
401                .sources()
402                .get(self.source_index)
403                .ok_or(BoundObservationSourceError::ProblemSourceMismatch)?;
404            if self.row_replay.is_none() {
405                self.row_replay = Some(source.selected_row_replay()?);
406            }
407            let replay = self
408                .row_replay
409                .as_mut()
410                .expect("selected-row replay initialized for current source");
411            if let Some(binding) = source.fill_next_selected_block(
412                self.problem,
413                logical_source,
414                replay,
415                block,
416                self.window,
417            )? {
418                return Ok(Some(FilledObservationBlock::new(block, binding)));
419            }
420            self.selected_rows += replay.selected_rows();
421            self.source_index += 1;
422            self.row_replay = None;
423        }
424    }
425
426    /// Return the retained access once every block has been read.
427    ///
428    /// The stream must be exhausted, and its walk of MAIN must have selected
429    /// exactly the rows the compiled selection counted: the content plan and
430    /// the model were sized from that count, so a MAIN that has since gained
431    /// or lost selected rows is refused. A channel window skips payload, not
432    /// rows, so a windowed stream is held to the same count.
433    pub fn complete(self) -> Result<BoundSelectedObservation, BoundObservationSourceError> {
434        if !self.exhausted {
435            return Err(BoundObservationSourceError::IncompleteBlockTraversal);
436        }
437        let expected = self
438            .problem
439            .selected_observation()
440            .read_set()
441            .sources()
442            .iter()
443            .map(|source| source.selection().rows().selected_row_count())
444            .sum();
445        if self.selected_rows != expected {
446            return Err(BoundObservationSourceError::SelectedRowCountMismatch {
447                expected,
448                delivered: self.selected_rows,
449            });
450        }
451        Ok(self.observation)
452    }
453}

455/// Failure to bind a complete compiled selected observation.
456#[derive(Debug, Error)]
457pub enum BoundSelectedObservationError {
458    /// The injected Measures provider is missing, stale, or unaccounted.
459    #[error(transparent)]
460    Measures(#[from] SelectedObservationMeasuresError),
461    /// The supplied binding count or membership differs from the compiled source set.
462    #[error("source binding set does not match the compiled selected observation")]
463    BindingSetMismatch,
464    /// One compiled source has no current state and budget binding.
465    #[error("compiled source {measurement_set} has no retained-access binding")]
466    MissingSourceBinding {
467        /// Source missing a binding.
468        measurement_set: usize,
469    },
470    /// One compiled source was assigned more than one binding.
471    #[error("compiled source {measurement_set} has duplicate retained-access bindings")]
472    DuplicateSourceBinding {
473        /// Source with duplicate bindings.
474        measurement_set: usize,
475    },
476    /// A source binding omitted, substituted, or unexpectedly supplied ephemeris data.
477    #[error(
478        "compiled source {measurement_set} ephemeris reference mismatch: expected {expected:?}, actual {actual:?}"
479    )]
480    EphemerisReferenceMismatch {
481        /// Source whose reference-data binding differs from compiled geometry.
482        measurement_set: usize,
483        /// Compiler-owned ephemeris identity, or absence for fixed geometry.
484        expected: Option<LogicalIdentity>,
485        /// Supplied source-binding identity, or absence when none was supplied.
486        actual: Option<LogicalIdentity>,
487    },
488    /// A source's retained reference data exceeds its selected-content ceiling.
489    #[error(
490        "compiled source {measurement_set} retains {required_bytes} reference-data bytes but its content budget has {available_bytes} bytes"
491    )]
492    ReferenceDataBudgetExceeded {
493        /// Source whose exact reference-data charge exceeds its budget.
494        measurement_set: usize,
495        /// Exact retained reference-data allocation.
496        required_bytes: usize,
497        /// Bytes authorized by the source content budget.
498        available_bytes: usize,
499    },
500    /// One retained source could not be bound under its state and budget.
501    #[error("bind compiled source {measurement_set}: {error}")]
502    Source {
503        /// Source whose binding failed.
504        measurement_set: usize,
505        /// Exact source-level failure.
506        #[source]
507        error: Box<BoundObservationSourceError>,
508    },
509    /// Aggregate retained reference data exceeded the host byte domain.
510    #[error("selected-observation reference-data residency projection overflowed")]
511    ReferenceDataByteOverflow,
