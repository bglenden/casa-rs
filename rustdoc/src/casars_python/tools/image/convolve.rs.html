26const INVALID_KERNEL: &str = "kernel is not understood, try using an array or an image";
27const KERNEL_PRECISION: &str =
28    "Invalid Table data type when accessing column in ArrayColumn ctor for column map";

30#[derive(Default)]
31enum KernelArgument {
32    File {
33        path: String,
34        history: String,
35    },
36    Array {
37        pixels: ArrayD<f64>,
38        history: String,
39    },
40    #[default]
41    Unsupported,
42}
43
44impl KernelArgument {
45    fn capture(py: Python<'_>, value: Option<&Bound<'_, PyAny>>) -> PyResult<Self> {
46        let Some(value) = value else {
47            return Ok(Self::Unsupported);
48        };
49        let converted = if value.is_instance_of::<PyList>() || value.is_instance_of::<PyTuple>() {
50            if value.len()? == 0 {
51                return Ok(Self::Unsupported);
52            }
53            match py.import("numpy")?.call_method1("asarray", (value,)) {
54                Ok(array) => Some(array),
55                Err(_) => return Ok(Self::Unsupported),
56            }
57        } else {
58            None
59        };
60        let raw = converted.as_ref().unwrap_or(value);
61        if let Ok(array) = raw.cast::<PyUntypedArray>()
62            && array.ndim() != 0
63        {
64            let dtype = array.dtype();
65            match (dtype.kind(), dtype.itemsize()) {
66                (b'f', 4 | 8) => {
67                    let pixels = snapshot_array(py, array)?;
68                    let history = if pixels.len() > 100 {
69                        "\"(...)\"".into()
70                    } else {
71                        let values = pixels
72                            .view()
73                            .reversed_axes()
74                            .iter()
75                            .map(|value| format!("{value:.6}"))
76                            .collect::<Vec<_>>();
77                        format!("[{}]", values.join(","))
78                    };
79                    return Ok(Self::Array { pixels, history });
80                }
81                (b'i' | b'u', _) => {
82                    let integers = snapshot_array::<i64>(py, array)?;
83                    let history = if integers.len() > 100 {
84                        "\"(...)\"".into()
85                    } else {
86                        let values = integers
87                            .view()
88                            .reversed_axes()
89                            .iter()
90                            .map(i64::to_string)
91                            .collect::<Vec<_>>();
92                        format!("[{}]", values.join(","))
93                    };
94                    let pixels = integers.mapv(|value| f64::from(value as i32));
95                    return Ok(Self::Array { pixels, history });
96                }
97                _ => {}
98            }
99        }
100        Ok(match value.extract::<Variant>()? {
101            Variant::Text(path) => {
102                let history = format!("{path:?}");
103                Self::File { path, history }
104            }
105            argument @ Variant::Texts(_) => {
106                let path = argument.describe();
107                Self::File {
108                    history: path.clone(),
109                    path,
110                }
111            }
112            _ => Self::Unsupported,
113        })
114    }
115
116    fn history(&self) -> &str {
117        match self {
118            Self::File { history, .. } | Self::Array { history, .. } => history,
119            Self::Unsupported => unreachable!("validated convolution kernel"),
120        }
121    }
122}
123
124struct Request<'a> {
125    outfile: &'a str,
126    kernel: &'a KernelArgument,
127    scale: f64,
128    region: &'a RegionArgument,
129    mask: &'a str,
130    overwrite: bool,
131    stretch: bool,
132    history: bool,
133}
134
135impl Request<'_> {
136    fn run(&self, source: &AttachedImage) -> PyResult<AttachedImage> {
137        if !matches!(
138            &*source.lock(),
139            ImageHandle::Float(_) | ImageHandle::Double(_)
140        ) {
141            return Err(raised("This method only supports real-valued images"));
142        }
143        let kernel_path = match self.kernel {
144            KernelArgument::File { path, .. } => {
145                let path = Path::new(path);
146                if !path.is_dir() {
147                    return Err(raised(format!("Table {} does not exist", path.display())));
148                }
149                Some(path)
150            }
151            KernelArgument::Array { .. } => None,
152            KernelArgument::Unsupported => return Err(raised(INVALID_KERNEL)),
153        };
154        let same = kernel_path
155            .and_then(|path| path.canonicalize().ok())
156            .is_some_and(|path| source.native_path() == Some(&path));
157        let kernel = kernel_path
158            .filter(|_| !same)
159            .map(AttachedImage::open_for_ordered_read)
160            .transpose()?;
161        let mut coordinates = None;
162        let mut output = publish_image(self.outfile, self.overwrite, |destination| {
163            let build = |image: &ImageHandle, kernel: Option<&ImageHandle>| {
164                self.build(source, image, kernel, destination)
165            };
166            let product = if let Some(kernel) = &kernel {
167                let source_first = source
168                    .native_path()
169                    .zip(kernel.native_path())
170                    .is_none_or(|(left, right)| left < right);
171                if source_first {
172                    read_locked(source, |image| {
173                        read_locked(kernel, |kernel| build(image, Some(kernel)))
174                    })?
175                } else {
176                    read_locked(kernel, |kernel| {
177                        read_locked(source, |image| build(image, Some(kernel)))
178                    })?
179                }
180            } else {
181                read_locked(source, |image| build(image, same.then_some(image)))?
182            };
183            coordinates = AttachedImage::capture_live_coordinates(&product)?;
184            Ok(product)
185        })?;
186        if let Some((record, system)) = coordinates {
187            output.keep_live_coordinates(record, system);
188        }
189        Ok(output)
190    }
191
192    fn build(
193        &self,
194        attachment: &AttachedImage,
195        image: &ImageHandle,
196        kernel: Option<&ImageHandle>,
197        destination: Option<&Path>,
198    ) -> PyResult<ImageHandle> {
199        if !self.outfile.is_empty() {
200            let destination = Path::new(self.outfile);
201            for input in image
202                .path()
203                .into_iter()
204                .chain(kernel.and_then(ImageHandle::path))
205                .chain(attachment.operand_paths().iter().map(|path| path.as_path()))
206            {
207                ensure_outfile_does_not_alias_input(destination, input).map_err(raised)?;
208            }
209        }
210        let record = image.coordinate_record();
211        let (mut product, changed, dimensions) = match image {
212            ImageHandle::Float(storage) => {
213                let native = match kernel {
214                    Some(ImageHandle::Float(Storage::Paged(native))) => {
215                        Some(native.as_ref() as &dyn ImageInterface<f32>)
216                    }
217                    Some(_) => return Err(raised(KERNEL_PRECISION)),
218                    None => None,
219                };
220                let (out, changed, dimensions) = match storage {
221                    Storage::Temp {
222                        image,
223                        fits: Some(fits),
224                        ..
225                    } => self.convolve(&fits.image_view(image), record, native, destination),
226                    _ => {
227                        with_storage!(storage, source => self.convolve(source.as_ref(), record, native, destination))
228                    }
229                }?;
230                (ImageHandle::Float(out), changed, dimensions)
231            }
232            ImageHandle::Double(storage) => {
233                let native = match kernel {
234                    Some(ImageHandle::Double(Storage::Paged(native))) => {
235                        Some(native.as_ref() as &dyn ImageInterface<f64>)
236                    }
237                    Some(_) => return Err(raised(KERNEL_PRECISION)),
238                    None => None,
239                };
240                let (out, changed, dimensions) = with_storage!(storage, source => self.convolve(source.as_ref(), record, native, destination))?;
241                (ImageHandle::Double(out), changed, dimensions)
242            }
243            _ => return Err(raised("This method only supports real-valued images")),
244        };
245        if let Some(record) = changed {
246            product.set_coordinate_record(record)?;
247        }
248        product.append_history_entries(&image.history_entries()?)?;
249        if self.history {
250            let Self {
251                outfile,
252                kernel,
253                scale,
254                region,
255                mask,
256                overwrite,
257                stretch,
258                ..
259            } = self;
260            let kernel = kernel.history();
261            let region = region.history();
262            product.record_history("convolve", format!("ia.convolve(outfile={outfile:?}, kernel={kernel}, scale={scale:.6}, region={region}, vmask={mask:?}, overwrite={overwrite}, stretch={stretch})"))?;
263        }
264        if dimensions < image.shape().len() {
265            product.append_history_at("ImageConvolverTask::convolve", &["NOTE: kernel with fewer dimensions than the input image applied. Degenerate axes added to the kernel".to_owned()])?;
266        }
267        product.save().map_err(raised)?;
268        Ok(product)
269    }
270
271    fn convolve<T: ConvolutionPixel>(
272        &self,
273        source: &dyn ImageInterface<T>,
274        coordinates: RecordValue,
275        native: Option<&dyn ImageInterface<T>>,
276        path: Option<&Path>,
277    ) -> PyResult<(Storage<T>, Option<RecordValue>, usize)> {
278        let data;
279        let (kernel, rank) = if let Some(native) = native {
280            (ConvolutionKernel::Image(native), native.ndim())
281        } else {
282            let KernelArgument::Array { pixels, .. } = self.kernel else {
283                unreachable!("validated convolution kernel")
284            };
285            data = pixels.mapv(T::from_f64);
286            (ConvolutionKernel::Array(&data), data.ndim())
287        };
288        let plan =
289            ImageConvolution::new(source, coordinates, self.region.selection()).map_err(raised)?;
290        let mut output = Storage::create(
291            plan.section.shape.clone(),
292            plan.section.coordinates.clone(),
293            path,
294            None,
295        )
296        .map_err(raised)?;
297        let options = ConvolutionOptions {
298            scale: self.scale,
299            mask: Some(self.mask),
300            stretch: self.stretch,
301            destination: (!self.outfile.is_empty()).then(|| Path::new(self.outfile)),
302        };
303        with_storage!(&mut output, dest => plan.copy_to(source, kernel, options, dest.as_mut()))
304            .map_err(raised)?;
305        Ok((output, plan.section.coordinate_record, rank))
306    }
307}

309#[pymethods]
310impl PyImageTool {
311    /// Linearly convolve a selected real image with an array or native image.
312    #[allow(clippy::too_many_arguments)]
313    #[pyo3(signature = (outfile="", kernel=None, scale=-1.0, region=RegionArgument::default(), mask=None, overwrite=false, stretch=false), text_signature = "($self, outfile='', kernel=[], scale=-1.0, region={}, mask='', overwrite=False, stretch=False)")]
314    fn convolve(
315        &self,
316        py: Python<'_>,
317        outfile: &str,
318        kernel: Option<&Bound<'_, PyAny>>,
319        scale: f64,
320        region: RegionArgument,
321        mask: Option<&Bound<'_, PyAny>>,
322        #[pyo3(from_py_with = require_bool)] overwrite: bool,
323        #[pyo3(from_py_with = require_bool)] stretch: bool,
324    ) -> PyResult<Py<Self>> {
325        let kernel = KernelArgument::capture(py, kernel)?;
326        let mask = mask_argument(mask)?;
327        let output = py.detach(|| {
328            let state = self.state();
329            let Some(source) = state.image.as_ref() else {
330                return Ok(None);
331            };
332            Request {
333                outfile,
334                kernel: &kernel,
335                scale,
336                region: &region,
337                mask: &mask,
338                overwrite,
339                stretch,
340                history: state.history,
341            }
342            .run(source)
343            .map(Some)
344        })?;
345        Py::new(py, output.map_or_else(Self::new, Self::attached))
346    }
