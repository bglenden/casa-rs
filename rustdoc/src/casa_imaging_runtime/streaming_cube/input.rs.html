3//! One run-private store for a homogeneous selected native-channel layout.
4//!
5//! Row blocks contain one metadata frame followed by native-channel tiles.
6//! Every offset is derived from the shape; no resident per-row/tile directory
7//! or per-sample object graph is needed. The writer transfers its original open
8//! file directly to readers. There is no reopen/import or publication authority.

29const ROW_BYTES: usize = 7 * 8;
30const SAMPLE_BYTES: usize = 2 * 8 + 8 + 2;
31const CRC_BYTES: usize = 4;
32
33fn invalid_input(message: &'static str) -> io::Error {
34    io::Error::new(io::ErrorKind::InvalidInput, message)
35}
36
37fn overflow() -> io::Error {
38    invalid_input("native cube store size overflow")
39}

41/// Byte arithmetic for a selected native layout, not an imaging scheduler.
42/// `tile_channels` comes from the parent band's useful native window. The row
43/// count is derived from the admitted preparation-buffer budget, never tuned
44/// to a particular dataset. Source blocks/kernel/owner storage are additional
45/// live allocations that the parent execution plan must count.
46#[derive(Clone, Copy, Debug, PartialEq, Eq)]
47pub(super) struct StorePlan {
48    pub(super) rows: u64,
49    pub(super) channels: usize,
50    pub(super) correlations: usize,
51    pub(super) block_rows: usize,
52    pub(super) tile_channels: usize,
53    tiles: usize,
54    channel_bytes: usize,
55    row_bytes: usize,
56    frame_bytes: usize,
57    preparation_bytes: usize,
58    page_cache_bytes: u64,
59    pub(super) artifact_bytes: u64,
60}
61
62impl StorePlan {
63    /// Batch across source/worker callback boundaries using the selected source's
64    /// I/O-buffer envelope. Reserve at least one complete native row; the parent
65    /// separately admits this arena alongside the still-live selected source.
66    pub(super) fn for_source_buffer(
67        rows: u64,
68        channels: usize,
69        correlations: usize,
70        tile_channels: usize,
71        source_buffer_bytes: usize,
72    ) -> io::Result<Self> {
73        if source_buffer_bytes == 0 {
74            return Err(invalid_input("native source buffer is empty"));
75        }
76        let (row_bytes, encoded_per_row) = Self::row_layout(channels, correlations, tile_channels)?;
77        let working_bytes = row_bytes
78            .checked_add(encoded_per_row)
79            .and_then(|bytes| {
80                bytes.checked_add(size_of::<NativeBlock>() + size_of::<Vec<u8>>() + CRC_BYTES)
81            })
82            .ok_or_else(overflow)?
83            .max(source_buffer_bytes);
84        Self::new(
85            rows,
86            channels,
87            correlations,
88            tile_channels,
89            working_bytes,
90            u64::MAX,
91        )
92    }
93
94    fn row_layout(
95        channels: usize,
96        correlations: usize,
97        tile_channels: usize,
98    ) -> io::Result<(usize, usize)> {
99        if channels == 0
100            || !(1..=4).contains(&correlations)
101            || tile_channels == 0
102            || tile_channels > channels
103        {
104            return Err(invalid_input("invalid native cube store shape"));
105        }
106        let channel_bytes = SAMPLE_BYTES
107            .checked_mul(correlations)
108            .and_then(|bytes| bytes.checked_add(8))
109            .ok_or_else(overflow)?;
110        let row_bytes = channels
111            .checked_mul(channel_bytes)
112            .and_then(|bytes| bytes.checked_add(ROW_BYTES))
113            .ok_or_else(overflow)?;
114        let encoded_per_row = tile_channels
115            .checked_mul(channel_bytes)
116            .ok_or_else(overflow)?
117            .max(ROW_BYTES);
118        Ok((row_bytes, encoded_per_row))
119    }
120
121    pub(super) fn new(
122        rows: u64,
123        channels: usize,
124        correlations: usize,
125        tile_channels: usize,
126        working_bytes: usize,
127        storage_bytes: u64,
128    ) -> io::Result<Self> {
129        if rows == 0 {
130            return Err(invalid_input("invalid native cube store shape"));
131        }
132        let (row_bytes, encoded_per_row) = Self::row_layout(channels, correlations, tile_channels)?;
133        let channel_bytes = SAMPLE_BYTES * correlations + 8;
134        let fixed_bytes = size_of::<NativeBlock>() + size_of::<Vec<u8>>() + CRC_BYTES;
135        let available = working_bytes
136            .checked_sub(fixed_bytes)
137            .ok_or_else(overflow)?;
138        let block_rows = (available
139            / row_bytes
140                .checked_add(encoded_per_row)
141                .ok_or_else(overflow)?)
142        .min(usize::try_from(rows).unwrap_or(usize::MAX));
143        if block_rows == 0 {
144            return Err(invalid_input(
145                "native store budget cannot hold one row and encoded tile",
146            ));
147        }
148        let frame_bytes = block_rows
149            .checked_mul(encoded_per_row)
150            .and_then(|bytes| bytes.checked_add(CRC_BYTES))
151            .ok_or_else(overflow)?;
152        let preparation_bytes = block_rows
153            .checked_mul(row_bytes)
154            .and_then(|bytes| bytes.checked_add(frame_bytes))
155            .and_then(|bytes| bytes.checked_add(fixed_bytes - CRC_BYTES))
156            .ok_or_else(overflow)?;
157        if frame_bytes > isize::MAX as usize || preparation_bytes > isize::MAX as usize {
158            return Err(overflow());
159        }
160        let tiles = channels.div_ceil(tile_channels);
161        let blocks = rows.div_ceil(block_rows as u64);
162        let frames = blocks
163            .checked_mul((tiles as u64).checked_add(1).ok_or_else(overflow)?)
164            .ok_or_else(overflow)?;
165        let artifact_bytes = rows
166            .checked_mul(row_bytes as u64)
167            .and_then(|bytes| bytes.checked_add(frames.checked_mul(CRC_BYTES as u64)?))
168            .ok_or_else(overflow)?;
169        if artifact_bytes > storage_bytes {
170            return Err(invalid_input("native store exceeds admitted storage"));
171        }
172        // A frame can begin partway through a page. The parent accounts this
173        // one bounded I/O window separately from the userspace preparation arena.
174        let page_cache_bytes = page_cache_window_bytes(frame_bytes as u64, 1)
175            .and_then(|bytes| page_cache_window_bytes(1, 1).map(|page| bytes + page))
176            .map_err(io::Error::other)?;
177        Ok(Self {
178            rows,
179            channels,
180            correlations,
181            block_rows,
182            tile_channels,
183            tiles,
184            channel_bytes,
185            row_bytes,
186            frame_bytes,
187            preparation_bytes,
188            page_cache_bytes,
189            artifact_bytes,
190        })
191    }
192
193    pub(super) fn blocks(self) -> u64 {
194        self.rows.div_ceil(self.block_rows as u64)
195    }
196
197    /// Writer-owned bytes when native row arrays are admitted separately in
198    /// worker-local blocks. No full native block is included in this charge.
199    pub(super) fn writer_residency(self) -> io::Result<u64> {
200        (size_of::<NativeStoreWriter>() as u64)
201            .checked_add(self.frame_bytes as u64)
202            .and_then(|bytes| bytes.checked_add(self.page_cache_bytes))
203            .ok_or_else(overflow)
204    }
205
206    fn maximum_cache_slots(self) -> usize {
207        // Reuse at most the preparation arena's byte envelope, regardless of
208        // dataset size. Its source rows are no longer live during band replay.
209        (self.preparation_bytes / (self.frame_bytes + size_of::<CachedFrame>())).max(1)
210    }
211
212    pub(super) fn reader_cache_slots(
213        self,
214        workers: usize,
215        native_tiles: usize,
216    ) -> io::Result<usize> {
217        if workers == 0 || native_tiles == 0 || native_tiles > self.tiles {
218            return Err(invalid_input("invalid native reader cache geometry"));
219        }
220        // One metadata frame per row block and the active bands' tile windows.
221        let tiles = workers
222            .checked_mul(native_tiles)
223            .ok_or_else(overflow)?
224            .min(self.tiles);
225        let desired = self
226            .blocks()
227            .checked_mul(tiles as u64 + 1)
228            .ok_or_else(overflow)?;
229        Ok(desired.min(self.maximum_cache_slots() as u64) as usize)
230    }
231
232    pub(super) fn reader_capacity_bytes(self, slots: usize) -> io::Result<u64> {
233        if slots == 0 || slots > self.maximum_cache_slots() {
234            return Err(invalid_input(
235                "native frame cache exceeds its byte envelope",
236            ));
237        }
238        let payload = slots
239            .checked_mul(self.frame_bytes + size_of::<CachedFrame>())
240            .ok_or_else(overflow)?;
241        (payload as u64)
242            .checked_add(size_of::<NativeStoreReader>() as u64)
243            .and_then(|bytes| bytes.checked_add(self.page_cache_bytes))
244            .ok_or_else(overflow)
245    }
246
247    pub(super) fn rows_in(self, block: u64) -> io::Result<usize> {
248        if block >= self.blocks() {
249            return Err(invalid_input("native row block out of range"));
250        }
251        Ok((self.rows - block * self.block_rows as u64).min(self.block_rows as u64) as usize)
252    }
253
254    fn tile(self, tile: usize) -> Range<usize> {
255        let start = tile * self.tile_channels;
256        start..(start + self.tile_channels).min(self.channels)
257    }
258
259    fn frame(self, block: u64, tile: Option<usize>) -> io::Result<(u64, u64, usize)> {
260        let rows = self.rows_in(block)?;
261        let block_bytes = (self.block_rows as u64 * self.row_bytes as u64)
262            + (self.tiles as u64 + 1) * CRC_BYTES as u64;
263        let offset = block * block_bytes;
264        let ordinal = block * (self.tiles as u64 + 1);
265        match tile {
266            None => Ok((ordinal, offset, rows * ROW_BYTES + CRC_BYTES)),
267            Some(tile) if tile < self.tiles => {
268                let channels = self.tile(tile);
269                Ok((
270                    ordinal + tile as u64 + 1,
271                    offset
272                        + (rows * ROW_BYTES + CRC_BYTES) as u64
273                        + rows as u64 * channels.start as u64 * self.channel_bytes as u64
274                        + tile as u64 * CRC_BYTES as u64,
275                    rows * channels.len() * self.channel_bytes + CRC_BYTES,
276                ))
277            }
278            _ => Err(invalid_input("native channel tile out of range")),
279        }
280    }
281}

306impl NativeStoreWriter {
307    pub(super) fn create(storage: &ManagedSpillStorage, plan: StorePlan) -> io::Result<Self> {
308        let file = tempfile::Builder::new()
309            .prefix(".casa-rs-native-cube-")
310            .tempfile_in(storage.directory())?;
311        configure_bounded_page_cache(file.as_file()).map_err(io::Error::other)?;
312        Ok(Self {
313            file,
314            plan,
315            encoded: Vec::with_capacity(plan.frame_bytes),
316            next_block: 0,
317            poisoned: false,
318            io: StoreIo::default(),
319        })
320    }
321
322    #[cfg(test)]
323    pub(super) fn append(&mut self, block: &NativeBlock) -> io::Result<()> {
324        self.append_parts(&[block])
325    }
326
327    /// Write one planned row block from ordered worker-owned row partitions.
328    /// Only the encoded frame is assembled; native payload arrays stay borrowed.
329    pub(super) fn append_parts(&mut self, parts: &[&NativeBlock]) -> io::Result<()> {
330        if self.poisoned {
331            return Err(io::Error::other("native store writer failed earlier"));
332        }
333        let result = self.append_parts_inner(parts);
334        self.poisoned = result.is_err();
335        result
336    }
337
338    fn append_parts_inner(&mut self, parts: &[&NativeBlock]) -> io::Result<()> {
339        let rows = self.plan.rows_in(self.next_block)?;
340        let mut supplied_rows = 0_usize;
341        for block in parts {
342            let part_rows = block.metadata.len();
343            if part_rows == 0 {
344                return Err(invalid_input("native store row partition is empty"));
345            }
346            block.validate_shape(part_rows, self.plan.channels, self.plan.correlations)?;
347            supplied_rows = supplied_rows.checked_add(part_rows).ok_or_else(overflow)?;
348        }
349        if supplied_rows != rows {
350            return Err(invalid_input(
351                "native store row partitions do not match planned rows",
352            ));
353        }
354        self.encoded.clear();
355        for row in parts.iter().flat_map(|block| &block.metadata) {
356            self.encoded
357                .extend_from_slice(&row.physical_row.to_le_bytes());
358            for value in row
359                .uvw_m
360                .into_iter()
361                .chain([row.phase_shift_m])
362                .chain(row.original_pair_hz)
363            {
364                self.encoded.extend_from_slice(&value.to_le_bytes());
365            }
366        }
367        self.write_frame(None)?;
368        for tile in 0..self.plan.tiles {
369            self.encoded.clear();
370            for block in parts {
371                for row in 0..block.metadata.len() {
372                    for channel in self.plan.tile(tile) {
373                        let cell = row * self.plan.channels + channel;
374                        self.encoded
375                            .extend_from_slice(&block.frequencies_hz[cell].to_le_bytes());
376                        for corr in 0..self.plan.correlations {
377                            let sample = cell * self.plan.correlations + corr;
378                            for value in [
379                                block.values[sample].re,
380                                block.values[sample].im,
381                                block.weights[sample],
382                            ] {
383                                self.encoded.extend_from_slice(&value.to_le_bytes());
384                            }
385                            self.encoded.push(u8::from(block.flags[sample]));
386                            self.encoded.push(u8::from(block.weight_flags[sample]));
387                        }
388                    }
389                }
390            }
391            self.write_frame(Some(tile))?;
392        }
393        self.next_block += 1;
394        Ok(())
395    }
396
397    fn write_frame(&mut self, tile: Option<usize>) -> io::Result<()> {
398        let (ordinal, offset, bytes) = self.plan.frame(self.next_block, tile)?;
399        if offset != self.io.bytes || self.encoded.len() + CRC_BYTES != bytes {
400            return Err(io::Error::other("native store encoder/offset mismatch"));
401        }
402        let crc = frame_crc(ordinal, &self.encoded);
403        self.encoded.extend_from_slice(&crc.to_le_bytes());
404        self.file.as_file_mut().write_all(&self.encoded)?;
405        release_page_cache(
406            self.file.as_file(),
407            offset,
408            bytes,
409            true,
410            &mut self.io.cache_operations,
411        )
412        .map_err(io::Error::other)?;
413        self.io.bytes += bytes as u64;
414        self.io.operations += 1;
415        self.io.checksum_bytes += (bytes - CRC_BYTES + size_of::<u64>()) as u64;
416        Ok(())
417    }
418
419    pub(super) fn finish(self) -> io::Result<NativeStore> {
420        if self.poisoned
421            || self.next_block != self.plan.blocks()
422            || self.io.bytes != self.plan.artifact_bytes
423            || self.file.as_file().metadata()?.len() != self.plan.artifact_bytes
424        {
425            return Err(io::Error::other("native store is incomplete"));
426        }
427        // Ownership transfer only: no payload reread, digest or completion token.
428        Ok(NativeStore {
429            file: self.file,
430            plan: self.plan,
431            #[cfg(test)]
432            written: self.io,
433            retention: None,
434            metadata_retention: None,
435        })
436    }
437}

465impl NativeStore {
466    pub(super) fn retain(&mut self, permit: crate::RetainedArtifactPermit) -> io::Result<()> {
467        if self.retention.is_some() {
468            return Err(io::Error::other("native store capacity retained twice"));
469        }
470        self.retention = Some(permit);
471        Ok(())
472    }
473
474    /// One source producer owns file I/O and fans borrowed input windows out to
475    /// band workers. This also preserves the existing strict Linux cache-release
476    /// policy: another read cannot repopulate a page while release is verified.
477    pub(super) fn reader(&mut self, cache_slots: usize) -> io::Result<NativeStoreReader<'_>> {
478        self.plan.reader_capacity_bytes(cache_slots)?;
479        let frame_bytes = self.plan.frame_bytes;
480        Ok(NativeStoreReader {
481            store: self,
482            encoded: vec![0; frame_bytes * cache_slots],
483            frames: vec![CachedFrame::default(); cache_slots],
484            clock: 0,
485            io: StoreIo::default(),
486            profile: std::env::var_os("CASA_RS_PROFILE_CUBE").is_some(),
487        })
488    }
489}
490
491impl NativeStoreReader<'_> {
492    fn read_frame(&mut self, block: u64, tile: Option<usize>) -> io::Result<&[u8]> {
493        let (ordinal, offset, bytes) = self.store.plan.frame(block, tile)?;
494        self.clock = self.clock.checked_add(1).ok_or_else(overflow)?;
495        let hit = self
496            .frames
497            .iter()
498            .position(|frame| frame.ordinal == Some(ordinal));
499        let slot = hit.unwrap_or_else(|| {
500            self.frames
501                .iter()
502                .enumerate()
503                .min_by_key(|(_, frame)| frame.used)
504                .unwrap()
505                .0
506        });
507        let start = slot * self.store.plan.frame_bytes;
508        let payload_bytes = bytes - CRC_BYTES;
509        if hit.is_some() {
510            self.frames[slot].used = self.clock;
511            self.io.cache_hits = self.io.cache_hits.checked_add(1).ok_or_else(overflow)?;
512            return Ok(&self.encoded[start..start + payload_bytes]);
513        }
514        // Failed reads/checks must not leave the evicted identity on overwritten
515        // bytes. A valid entry is installed only after CRC and cache release.
516        self.frames[slot] = CachedFrame::default();
517        let encoded = &mut self.encoded[start..start + bytes];
518        let started = self.profile.then(Instant::now);
519        self.store.file.as_file().read_exact_at(encoded, offset)?;
520        if let Some(started) = started {
521            self.io.read_nanos += started.elapsed().as_nanos();
522        }
523        let started = self.profile.then(Instant::now);
524        let expected = u32::from_le_bytes(encoded[payload_bytes..].try_into().unwrap());
525        if frame_crc(ordinal, &encoded[..payload_bytes]) != expected {
526            return Err(io::Error::new(
527                io::ErrorKind::InvalidData,
528                "native store frame checksum mismatch",
529            ));
530        }
531        if let Some(started) = started {
532            self.io.crc_nanos += started.elapsed().as_nanos();
533        }
534        let started = self.profile.then(Instant::now);
535        release_page_cache(
536            self.store.file.as_file(),
537            offset,
538            bytes,
539            false,
540            &mut self.io.cache_operations,
541        )
542        .map_err(io::Error::other)?;
543        if let Some(started) = started {
544            self.io.cache_release_nanos += started.elapsed().as_nanos();
545        }
546        self.io.bytes += bytes as u64;
547        self.io.operations += 1;
548        self.io.checksum_bytes += (payload_bytes + size_of::<u64>()) as u64;
549        self.frames[slot] = CachedFrame {
550            ordinal: Some(ordinal),
551            used: self.clock,
552        };
553        Ok(&encoded[..payload_bytes])
554    }
555
556    pub(super) fn read_block(
557        &mut self,
558        block: u64,
559        channels: Range<usize>,
560        output: &mut NativeBlock,
561    ) -> io::Result<()> {
562        let plan = self.store.plan;
563        if channels.is_empty()
564            || channels.end > plan.channels
565            || output.correlations != plan.correlations
566        {
567            return Err(invalid_input("invalid native store read window"));
568        }
569        let rows = plan.rows_in(block)?;
570        output.set_shape(rows, channels.len())?;
571        let metadata = self.read_frame(block, None)?;
572        for (row, bytes) in output
573            .metadata
574            .iter_mut()
575            .zip(metadata.chunks_exact(ROW_BYTES))
576        {
577            row.physical_row = u64::from_le_bytes(bytes[..8].try_into().unwrap());
578            row.uvw_m = std::array::from_fn(|axis| decode_f64(bytes, (axis + 1) * 8));
579            row.phase_shift_m = decode_f64(bytes, 32);
580            row.original_pair_hz = [decode_f64(bytes, 40), decode_f64(bytes, 48)];
581        }
582        for tile in channels.start / plan.tile_channels..=(channels.end - 1) / plan.tile_channels {
583            let tile_channels = plan.tile(tile);
584            let selected =
585                channels.start.max(tile_channels.start)..channels.end.min(tile_channels.end);
586            let encoded = self.read_frame(block, Some(tile))?;
587            for row in 0..rows {
588                for channel in selected.clone() {
589                    let offset = (row * tile_channels.len() + channel - tile_channels.start)
590                        * plan.channel_bytes;
591                    let cell = row * channels.len() + channel - channels.start;
592                    output.frequencies_hz[cell] = decode_f64(encoded, offset);
593                    for corr in 0..plan.correlations {
594                        let position = offset + 8 + corr * SAMPLE_BYTES;
595                        let sample = cell * plan.correlations + corr;
596                        output.values[sample] = Complex64::new(
597                            decode_f64(encoded, position),
598                            decode_f64(encoded, position + 8),
599                        );
600                        output.weights[sample] = decode_f64(encoded, position + 16);
601                        output.flags[sample] = decode_flag(encoded[position + 24])?;
602                        output.weight_flags[sample] = decode_flag(encoded[position + 25])?;
603                    }
604                }
605            }
606        }
607        Ok(())
608    }

610    pub(super) fn complete(self) -> StoreIo {
611        self.io
612    }
613}

615/// The existing bounded executor owns source slots and its worker team. This
616/// producer decodes the union needed by one admitted band wave exactly once
617/// per row block; all partitions borrow the resulting arrays until the join.
618pub(super) struct NativeSource<'a> {
619    reader: NativeStoreReader<'a>,
620    channels: Range<usize>,
621    next_block: u64,
622    exhausted: bool,
623    poisoned: bool,
624}
625
626impl<'a> NativeSource<'a> {
627    #[cfg(test)]
628    pub(super) fn new(store: &'a mut NativeStore, channels: Range<usize>) -> io::Result<Self> {
629        if channels.is_empty() || channels.end > store.plan.channels {
630            return Err(invalid_input("invalid native source window"));
631        }
632        Ok(Self {
633            reader: store.reader(1)?,
634            channels,
635            next_block: 0,
636            exhausted: false,
637            poisoned: false,
638        })
639    }
640
641    /// Charge once, in addition to the store/file owner and each source slot.
642    #[cfg(test)]
643    fn shared_capacity_bytes(&self) -> u64 {
644        size_of::<Self>() as u64
645            + self.reader.encoded.capacity() as u64
646            + (self.reader.frames.capacity() * size_of::<CachedFrame>()) as u64
647            + self.reader.store.plan.page_cache_bytes
648    }
649
650    /// Per slot; BoundedStreamPlan's source capacity is the sum over its slots.
651    #[cfg(test)]
652    fn slot_capacity_bytes(&self) -> u64 {
653        self.block_bytes(self.reader.store.plan.block_rows)
654    }
655
656    fn block_bytes(&self, rows: usize) -> u64 {
657        // StorePlan already checked the larger full-channel preparation block.
658        (size_of::<NativeBlock>()
659            + rows * (ROW_BYTES + self.channels.len() * self.reader.store.plan.channel_bytes))
660            as u64
661    }
662
663    /// Payload-free source projection before its decoder and slots allocate.
664    pub(super) fn memory(plan: StorePlan, channels: Range<usize>) -> io::Result<(u64, u64)> {
665        if channels.is_empty() || channels.end > plan.channels {
666            return Err(invalid_input("invalid native source window"));
667        }
668        let slot = (size_of::<NativeBlock>()
669            + plan.block_rows * (ROW_BYTES + channels.len() * plan.channel_bytes))
670            as u64;
671        let shared = (size_of::<Self>() - size_of::<NativeStoreReader>()) as u64
672            + plan.reader_capacity_bytes(1)?;
673        Ok((shared, slot))
674    }
675}
676
677impl OrderedBlockSource for NativeSource<'_> {
678    type Storage = Option<NativeBlock>;
679    type Completion = StoreIo;
680    type Error = io::Error;
681
682    fn create_storage(&self, _slot: usize) -> Self::Storage {
683        let plan = self.reader.store.plan;
684        Some(
685            NativeBlock::new(plan.block_rows, self.channels.len(), plan.correlations)
686                .expect("native store plan bounds every source window"),
687        )
688    }
689
690    fn fill(
691        &mut self,
692        block_ordinal: u64,
693        storage: &mut Self::Storage,
694        cancellation: SourceFillCancellation<'_>,
695    ) -> io::Result<SourcePoll> {
696        if self.poisoned || block_ordinal != self.next_block {
697            self.poisoned = true;
698            return Err(invalid_input("native source block order or failed state"));
699        }
700        if cancellation.is_cancelled() {
701            self.poisoned = true;
702            return Err(io::Error::new(
703                io::ErrorKind::Interrupted,
704                "native source cancelled",
705            ));
706        }
707        if self.next_block == self.reader.store.plan.blocks() {
708            self.exhausted = true;
709            return Ok(SourcePoll::Exhausted);
710        }
711        let storage = storage.as_mut().ok_or_else(|| {
712            self.poisoned = true;
713            invalid_input("native source slot is absent")
714        })?;
715        let before = self.reader.io;
716        if let Err(error) = self
717            .reader
718            .read_block(block_ordinal, self.channels.clone(), storage)
719        {
720            self.poisoned = true;
721            return Err(error);
722        }
723        self.next_block += 1;
724        Ok(SourcePoll::Ready {
725            source_ordinal: 0,
726            logical_units: storage.metadata.len(),
727            logical_bytes: self.reader.io.bytes - before.bytes,
728            source_read_operations: self.reader.io.operations - before.operations,
729            resident_current_bytes: self.block_bytes(storage.metadata.len()),
730            resident_capacity_bytes: storage.capacity_bytes() as u64,
731        })
732    }
733
734    fn complete(self) -> io::Result<StoreIo> {
735        if self.poisoned || !self.exhausted || self.next_block != self.reader.store.plan.blocks() {
736            return Err(io::Error::other("native source incomplete"));
737        }
738        Ok(self.reader.io)
739    }
740}

742fn decode_f64(bytes: &[u8], offset: usize) -> f64 {
743    f64::from_le_bytes(bytes[offset..offset + 8].try_into().unwrap())
744}
745
746fn decode_flag(byte: u8) -> io::Result<bool> {
747    match byte {
748        0 => Ok(false),
749        1 => Ok(true),
750        _ => Err(io::Error::new(
751            io::ErrorKind::InvalidData,
752            "invalid native store flag",
753        )),
754    }
755}

757/// The ordinal detects misplaced frames. CRC is solely a scratch-file integrity
758/// check, consumed on actual reads; it is never used to authorize publication.
759fn frame_crc(ordinal: u64, payload: &[u8]) -> u32 {
760    crc32c::crc32c_append(crc32c::crc32c(&ordinal.to_le_bytes()), payload)
761}
