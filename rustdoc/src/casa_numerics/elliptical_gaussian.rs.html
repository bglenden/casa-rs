3//! Unit-consistent algebra for two-dimensional elliptical Gaussians.
4//!
5//! Axes may use any common angular unit. Position angles use radians in the
6//! CASA convention: east from north, modulo pi.
7
8use std::{error::Error, fmt};
9
10/// One elliptical Gaussian described by full-width-at-half-maximum axes.
11#[derive(Debug, Clone, Copy, PartialEq)]
12pub struct EllipticalGaussian {
13    /// Major-axis FWHM in a caller-chosen angular unit.
14    pub major: f64,
15    /// Minor-axis FWHM in the same unit as [`Self::major`].
16    pub minor: f64,
17    /// CASA position angle, east from north, in radians.
18    pub position_angle: f64,
19}
20
21impl EllipticalGaussian {
22    /// Construct one numerical Gaussian descriptor.
23    #[must_use]
24    pub const fn new(major: f64, minor: f64, position_angle: f64) -> Self {
25        Self {
26            major,
27            minor,
28            position_angle,
29        }
30    }
31
32    /// Return the Gaussian area in squared axis units.
33    #[must_use]
34    pub fn area(self) -> f64 {
35        std::f64::consts::PI / (4.0 * 2.0_f64.ln()) * self.major * self.minor
36    }
37}

39/// Failure of elliptical-Gaussian enclosure or deconvolution algebra.
40#[derive(Debug, Clone, Copy, PartialEq, Eq)]
41pub enum EllipticalGaussianError {
42    /// No usable Gaussian was supplied.
43    EmptySet,
44    /// A Gaussian had non-finite, non-positive, or inverted axes.
45    InvalidGaussian,
46    /// The requested target cannot be produced by convolving the source.
47    TargetSmallerThanSource,
48}
49
50impl fmt::Display for EllipticalGaussianError {
51    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
52        formatter.write_str(match self {
53            Self::EmptySet => "elliptical Gaussian set is empty",
54            Self::InvalidGaussian => {
55                "elliptical Gaussian requires finite positive major >= minor axes"
56            }
57            Self::TargetSmallerThanSource => "target Gaussian is smaller than the source Gaussian",
58        })
59    }
60}

62impl Error for EllipticalGaussianError {}

64/// Wrap a CASA beam position angle into `(-pi/2, pi/2]`.
65#[must_use]
66pub fn normalize_position_angle(angle: f64) -> f64 {
67    // GaussianBeam::_unwrap returns an already canonical angle unchanged.
68    if angle > -std::f64::consts::FRAC_PI_2 && angle <= std::f64::consts::FRAC_PI_2 {
69        return angle;
70    }
71    let mut wrapped = normalize_angle_pi(angle);
72    if wrapped <= -std::f64::consts::FRAC_PI_2 {
73        wrapped += std::f64::consts::PI;
74    } else if wrapped > std::f64::consts::FRAC_PI_2 {
75        wrapped -= std::f64::consts::PI;
76    }
77    wrapped
78}

80/// Compute CASA's minimum-area common enclosing Gaussian.
81///
82/// This is the numerical core of `CasaImageBeamSet::getCommonBeam()`.
83/// `radians_per_axis_unit` converts each FWHM axis to radians; the returned
84/// axes use the caller's unit. CASA constructs each enclosing ellipse in
85/// arcseconds and tests it in radians.
86///
87/// # Errors
88///
89/// Returns [`EllipticalGaussianError::EmptySet`] for no beams and
90/// [`EllipticalGaussianError::InvalidGaussian`] for invalid beam parameters
91/// or a non-finite or non-positive axis-unit conversion.
92pub fn common_enclosing_gaussian(
93    gaussians: &[EllipticalGaussian],
94    radians_per_axis_unit: f64,
95) -> Result<EllipticalGaussian, EllipticalGaussianError> {
96    if gaussians.is_empty() {
97        return Err(EllipticalGaussianError::EmptySet);
98    }
99    if !radians_per_axis_unit.is_finite()
100        || radians_per_axis_unit <= 0.0
101        || gaussians.iter().any(|gaussian| !valid(*gaussian))
102    {
103        return Err(EllipticalGaussianError::InvalidGaussian);
104    }
105    if gaussians.iter().all(|gaussian| *gaussian == gaussians[0]) {
106        return Ok(gaussians[0]);
107    }
108    let mut radians = gaussians.to_vec();
109    if radians_per_axis_unit != 1.0 {
110        for beam in &mut radians {
111            beam.major *= radians_per_axis_unit;
112            beam.minor *= radians_per_axis_unit;
113        }
114    }
115    let mut common = common_enclosing_iterative(&mut radians)?;
116    if radians_per_axis_unit != 1.0 {
117        common.major /= radians_per_axis_unit;
118        common.minor /= radians_per_axis_unit;
119    }
120    Ok(common)
121}

123/// Compute the Gaussian that convolves `source` into `target`.
124///
125/// Returns `Ok(None)` when target and source are effectively identical.
126///
127/// # Errors
128///
129/// Returns [`EllipticalGaussianError::TargetSmallerThanSource`] when the
130/// covariance difference is not positive semidefinite.
131pub fn deconvolving_gaussian(
132    target: EllipticalGaussian,
133    source: EllipticalGaussian,
134) -> Result<Option<EllipticalGaussian>, EllipticalGaussianError> {
135    if !valid(target) || !valid(source) {
136        return Err(EllipticalGaussianError::InvalidGaussian);
137    }
138    let lhs = gaussian_covariance(target);
139    let rhs = gaussian_covariance(source);
140    let delta = [
141        [lhs[0][0] - rhs[0][0], lhs[0][1] - rhs[0][1]],
142        [lhs[1][0] - rhs[1][0], lhs[1][1] - rhs[1][1]],
143    ];
144    let trace = delta[0][0] + delta[1][1];
145    let determinant = delta[0][0] * delta[1][1] - delta[0][1] * delta[1][0];
146    let discriminant = ((trace * trace) / 4.0 - determinant).max(0.0).sqrt();
147    let lambda_major = trace / 2.0 + discriminant;
148    let lambda_minor = trace / 2.0 - discriminant;
149
150    // These are the existing CASA-compatible absolute covariance tolerances
151    // for radian beam axes: one treats numerical zero as a point source and
152    // the other rejects a genuinely negative covariance eigenvalue.
153    if lambda_minor < -1.0e-12 || lambda_major < -1.0e-12 {
154        return Err(EllipticalGaussianError::TargetSmallerThanSource);
155    }
156    if lambda_major <= 1.0e-24 && lambda_minor <= 1.0e-24 {
157        return Ok(None);
158    }
159    let x_axis_angle =
160        if delta[0][1].abs() <= 1.0e-18 && (delta[0][0] - delta[1][1]).abs() <= 1.0e-18 {
161            0.0
162        } else {
163            0.5 * (2.0 * delta[0][1]).atan2(delta[0][0] - delta[1][1])
164        };
165    Ok(Some(from_x_axis(
166        lambda_major.max(0.0).sqrt(),
167        lambda_minor.max(0.0).sqrt(),
168        x_axis_angle,
169    )))
170}

172fn valid(gaussian: EllipticalGaussian) -> bool {
173    gaussian.major.is_finite()
174        && gaussian.minor.is_finite()
175        && gaussian.position_angle.is_finite()
176        && gaussian.major > 0.0
177        && gaussian.minor > 0.0
178        && gaussian.major >= gaussian.minor
179}

181/// `ImageBeamSet` keeps the first beam when two have the same area.
182fn largest_area(gaussians: &[EllipticalGaussian]) -> Option<(usize, &EllipticalGaussian)> {
183    gaussians
184        .iter()
185        .enumerate()
186        .max_by(|(left_index, lhs), (right_index, rhs)| {
187            lhs.area()
188                .total_cmp(&rhs.area())
189                .then_with(|| right_index.cmp(left_index))
190        })
191}

193fn common_enclosing_iterative(
194    gaussians: &mut [EllipticalGaussian],
195) -> Result<EllipticalGaussian, EllipticalGaussianError> {
196    loop {
197        let (max_index, &max_gaussian) =
198            largest_area(gaussians).ok_or(EllipticalGaussianError::EmptySet)?;
199
200        // CASA skips every beam equal to the largest (`*iBeam != maxBeam`),
201        // not only the largest itself: a duplicate (Stokes planes share a
202        // beam) can fail the enclosure test by rounding, and enclosing a
203        // beam with itself gives no ellipse.
204        let problem = gaussians
205            .iter()
206            .copied()
207            .enumerate()
208            .filter(|(index, gaussian)| *index != max_index && *gaussian != max_gaussian)
209            .map(|(_, gaussian)| gaussian)
210            .rfind(|gaussian| !encloses(max_gaussian, *gaussian));
211        let Some(problem) = problem else {
212            return Ok(max_gaussian);
213        };
214
215        let relative_angle = normalize_position_angle(problem.position_angle)
216            - normalize_position_angle(max_gaussian.position_angle);
217        if (normalize_angle_pi(relative_angle).abs() - std::f64::consts::FRAC_PI_2).abs() <= 1.0e-12
218        {
219            let max_has_major = max_gaussian.major >= problem.major;
220            return Ok(EllipticalGaussian::new(
221                if max_has_major {
222                    max_gaussian.major
223                } else {
224                    problem.major
225                },
226                if max_has_major {
227                    problem.major
228                } else {
229                    max_gaussian.major
230                },
231                normalize_position_angle(if max_has_major {
232                    max_gaussian.position_angle
233                } else {
234                    problem.position_angle
235                }),
236            ));
237        }
238
239        // CASA constructs the enclosing ellipse in arcseconds before its
240        // Gaussian deconvolution checks convert those axes back to radians.
241        let arcseconds_per_radian = 180.0 * 3_600.0 / std::f64::consts::PI;
242        let radians_per_arcsecond = std::f64::consts::PI / (180.0 * 3_600.0);
243        // Quantum::getValue divides by the ratio of destination and source
244        // unit factors. Multiplying by the reciprocal can round differently
245        // at an enclosure boundary and change the next beam chosen.
246        let max_major = max_gaussian.major / radians_per_arcsecond;
247        let max_minor = max_gaussian.minor / radians_per_arcsecond;
248        let equal_area_axis = (max_major * max_minor).sqrt();
249        let x_scale = equal_area_axis / max_major;
250        let y_scale = equal_area_axis / max_minor;
251        let (problem_major, _, problem_angle) = transform_ellipse_by_scaling(
252            problem.major / radians_per_arcsecond,
253            problem.minor / radians_per_arcsecond,
254            relative_angle,
255            x_scale,
256            y_scale,
257        );
258        let (mut major, mut minor, common_angle) = transform_ellipse_by_scaling(
259            problem_major,
260            equal_area_axis,
261            problem_angle,
262            1.0 / x_scale,
263            1.0 / y_scale,
264        );
265        let position_angle = common_angle + normalize_position_angle(max_gaussian.position_angle);
266        // A degenerate construction would never enclose the pair: refuse it
267        // rather than grow it for ever.
268        if !(major.is_finite() && minor.is_finite() && position_angle.is_finite())
269            || major <= 0.0
270            || minor <= 0.0
271        {
272            return Err(EllipticalGaussianError::InvalidGaussian);
273        }
274        let gaussian = |major, minor| {
275            EllipticalGaussian::new(
276                major / arcseconds_per_radian,
277                minor / arcseconds_per_radian,
278                normalize_position_angle(position_angle),
279            )
280        };
281        let mut enclosing = gaussian(major, minor);
282        while !(encloses(enclosing, max_gaussian) && encloses(enclosing, problem)) {
283            major *= 1.001;
284            minor *= 1.001;
285            enclosing = gaussian(major, minor);
286        }
287
288        gaussians[max_index] = enclosing;
289    }
290}

292fn from_x_axis(major: f64, minor: f64, x_axis_angle: f64) -> EllipticalGaussian {
293    EllipticalGaussian::new(
294        major,
295        minor,
296        normalize_position_angle(x_axis_angle - std::f64::consts::FRAC_PI_2),
297    )

300fn normalize_angle_pi(angle: f64) -> f64 {
301    let mut wrapped = angle.rem_euclid(2.0 * std::f64::consts::PI);
302    if wrapped > std::f64::consts::PI {
303        wrapped -= 2.0 * std::f64::consts::PI;
304    }
305    wrapped
306}

308fn encloses(enclosing: EllipticalGaussian, other: EllipticalGaussian) -> bool {
309    let theta_source = normalize_position_angle(enclosing.position_angle);
310    let theta_beam = normalize_position_angle(other.position_angle);
311    let alpha = (enclosing.major * theta_source.cos()).powi(2)
312        + (enclosing.minor * theta_source.sin()).powi(2)
313        - (other.major * theta_beam.cos()).powi(2)
314        - (other.minor * theta_beam.sin()).powi(2);
315    let beta = (enclosing.major * theta_source.sin()).powi(2)
316        + (enclosing.minor * theta_source.cos()).powi(2)
317        - (other.major * theta_beam.sin()).powi(2)
318        - (other.minor * theta_beam.cos()).powi(2);
319    let gamma = 2.0
320        * (((enclosing.minor * enclosing.minor) - (enclosing.major * enclosing.major))
321            * theta_source.sin()
322            * theta_source.cos()
323            - ((other.minor * other.minor) - (other.major * other.major))
324                * theta_beam.sin()
325                * theta_beam.cos());
326    let sum = alpha + beta;
327    let difference = ((alpha - beta).powi(2) + gamma.powi(2)).sqrt();
328    alpha >= 0.0 && beta >= 0.0 && sum >= difference
329}

331/// Return the symmetric FWHM-squared covariance form in image x/y axes.
332///
333/// The common Gaussian-to-standard-deviation factor is deliberately omitted:
334/// it cancels in enclosure and deconvolution and preserves CASA arithmetic.
335#[must_use]
336pub fn gaussian_covariance(gaussian: EllipticalGaussian) -> [[f64; 2]; 2] {
337    let angle = normalize_position_angle(gaussian.position_angle) + std::f64::consts::FRAC_PI_2;
338    let cos = angle.cos();
339    let sin = angle.sin();
340    let major2 = gaussian.major * gaussian.major;
341    let minor2 = gaussian.minor * gaussian.minor;
342    [
343        [
344            cos * cos * major2 + sin * sin * minor2,
345            cos * sin * (major2 - minor2),
346        ],
347        [
348            cos * sin * (major2 - minor2),
349            sin * sin * major2 + cos * cos * minor2,
350        ],
351    ]
352}

354fn transform_ellipse_by_scaling(
355    major: f64,
356    minor: f64,
357    position_angle: f64,
358    x_scale: f64,
359    y_scale: f64,
360) -> (f64, f64, f64) {
361    let cos = position_angle.cos();
362    let sin = position_angle.sin();
363    let cos2 = cos * cos;
364    let sin2 = sin * sin;
365    let major2 = major * major;
366    let minor2 = minor * minor;
367    let a = cos2 / major2 + sin2 / minor2;
368    let b = -2.0 * cos * sin * (1.0 / major2 - 1.0 / minor2);
369    let c = sin2 / major2 + cos2 / minor2;
370    let x_scale2 = x_scale * x_scale;
371    let y_scale2 = y_scale * y_scale;
372    let r = a / x_scale2;
373    let s = b * b / (4.0 * x_scale2 * y_scale2);
374    let t = c / y_scale2;
375    let difference = r - t;
376    let discriminant = difference * difference + 4.0 * s;
377    let signed = discriminant.sqrt() * difference.abs();
378    let j1 = (signed + discriminant) / discriminant / 2.0;
379    let j2 = (-signed + discriminant) / discriminant / 2.0;
380    let k1 = (j1 * r + j1 * t - t) / (2.0 * j1 - 1.0);
381    let k2 = (j2 * r + j2 * t - t) / (2.0 * j2 - 1.0);
382    let axis1 = (1.0 / k1).sqrt();
383    let axis2 = (1.0 / k2).sqrt();
384
385    // CASA's `_transformEllipseByScaling` takes the ellipse for a circle
386    // only when its axes are exactly equal, whatever their unit. (Its circle
387    // has the reciprocal of the axis, `sqrt(k1)`, against its evident intent;
388    // the circle has the axis.)
389    if axis1 == axis2 {
390        return (axis1, axis1, 0.0);
391    }
392    if axis1 > axis2 {
393        (
394            axis1,
395            axis2,
396            if position_angle >= 0.0 {
397                j1.sqrt().acos()
398            } else {
399                -j1.sqrt().acos()
400            },
401        )
402    } else {
403        (
404            axis2,
405            axis1,
406            if position_angle >= 0.0 {
407                j2.sqrt().acos()
408            } else {
409                -j2.sqrt().acos()
410            },
411        )
412    }
413}
