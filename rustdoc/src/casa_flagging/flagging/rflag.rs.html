5pub(super) fn apply_rflag_flags(
6    ms: &mut MeasurementSet,
7    selected_rows: &[usize],
8    channel_selections: Option<&BTreeMap<i32, ChannelSelection>>,
9    request: &FlagDataRequest,
10    calculate: bool,
11    model_factory: Option<FlagDataModelFactory>,
12    averaging: AveragedAgentInput<'_>,
13) -> Result<ChangeSet, FlaggingError> {
14    let AveragedAgentInput {
15        time: averaged_time,
16        uv: averaged_uv,
17        mut view,
18    } = averaging;
19    let started = Instant::now();
20    let column = request.data_column;
21    let calibration = is_calibration_table(ms);
22    let public_ms = request.casa_row_selection && !calibration;
23    let casa_rflag = request.casa_row_selection || calibration;
24    let time_average = request
25        .clip_time_bin
26        .as_deref()
27        .filter(|_| public_ms)
28        .map(time_bin_seconds);
29    let widths = request.clip_channel_bins.as_deref().filter(|_| public_ms);
30    let average = view.is_some()
31        || time_average.flatten().is_some()
32        || widths.is_some_and(|widths| widths.iter().any(|&width| width != 0));
33    let correlations = public_ms
34        .then(|| correlation::RflagCorrelations::read(ms, request.correlation.as_deref(), column))
35        .transpose()?;
36    let corr_admitted = correlations.as_ref().is_none_or(|c| c.admitted);
37    let active = request.rflag_window.is_some() && time_average != Some(None) && corr_admitted;
38    let (mut columns, configures_averaging) = if public_ms {
39        autoflag_columns(
40            ms,
41            column,
42            view.as_deref()
43                .map_or(selected_rows, SharedFlagView::input_rows),
44            model_factory,
45            active,
46            average,
47            FlagDataMode::Rflag,
48        )?
49    } else {
50        (None, false)
51    };
52    let active = active && (!public_ms || columns.is_some());
53    if (public_ms && !configures_averaging) || (!active && !request.extendflags) {
54        return Ok(ChangeSet {
55            agent_skipped: true,
56            ..ChangeSet::default()
57        });
58    }
59    let configures_averaging = configures_averaging && corr_admitted && time_average != Some(None);
60    let widths = widths.filter(|_| configures_averaging);
61    if let Some(widths) = widths {
62        check_channel_bins(ms, widths, &display_ms_path(ms))?;
63    }
64    let loaded = load_automatic_samples(
65        ms,
66        selected_rows,
67        channel_selections,
68        request,
69        AutomaticSampleSource {
70            columns: columns.as_mut(),
71            fit: active,
72            seconds: time_average.flatten().filter(|_| configures_averaging),
73            widths,
74            input: AveragedAgentInput {
75                time: averaged_time,
76                uv: averaged_uv,
77                view: view.as_deref_mut(),
78            },
79        },
80    )?;
81    let samples = loaded.samples;
82    let projection = calibration
83        .then(|| calibration::automatic_projection(request, &samples))
84        .transpose()?;
85    let context = RflagContext {
86        casa_arithmetic: casa_rflag,
87        correlations: correlations.as_ref(),
88        calibration: projection.as_ref(),
89        model_roundoff: &loaded.model_roundoff,
90    };
91    let loaded_ms = started.elapsed().as_millis();
92    let groups = if let Some(view) = view.as_deref() {
93        view.groups(&samples)
94    } else if casa_rflag {
95        ordered_autoflag_groups(ms, selected_rows, &samples, request)?
96    } else {
97        group_samples(&samples)
98            .into_values()
99            .map(|samples| (AutoflagChunk::default(), samples))
100            .collect()
101    };
102    let fit_groups = time_selected_autoflag_groups(&groups, &loaded.axes);
103    if !calculate && active && casa_rflag {
104        validate_rflag_matrix_scales(&fit_groups, request, context)?;
105    }
106    if calculate {
107        let thresholds = if active {
108            rflag_threshold_maps_for_groups(&fit_groups, request, context)?
109        } else {
110            RFlagThresholds::default()
111        };
112        return Ok(ChangeSet {
113            selected_rows: loaded.axes.time_selected_rows(averaged_time),
114            rflag_report: (active || request.extendflags)
115                .then(|| thresholds.public_report(request, active)),
116            ..ChangeSet::default()
117        });
118    }
119
120    let mut modified = vec![false; samples.len()];
121    let mut time_flags = vec![false; samples.len()];
122    let mut freq_flags = vec![false; samples.len()];
123    let RflagFitOutcome {
124        thresholds,
125        buffer_cutoffs,
126        threshold_ms,
127    } = if active {
128        fit_rflag_chunks(
129            &fit_groups,
130            request,
131            context,
132            &mut TfcropFitMasks {
133                modified: &mut modified,
134                freq_flags: &mut freq_flags,
135                time_flags: &mut time_flags,
136            },
137        )?
138    } else {
139        RflagFitOutcome::default()
140    };
141    if request.extendflags && view.is_none() {
142        if casa_rflag && channel_selections.is_some() && loaded.axes.channels_by_row.is_empty() {
143            extend_rflag_selected_channels(
144                &groups,
145                &loaded.axes,
146                &loaded.flags_by_row,
147                &mut modified,
148                request,
149            )?;
150        } else {
151            extend_autoflag_grouped(
152                groups.iter().map(|(_, samples)| samples.as_slice()),
153                &samples,
154                &mut modified,
155            );
156        }
157    }
158    let union_candidates = count_mask(&modified);
159    let planned_ms = started.elapsed().as_millis();
160    trace_flagdata(|| {
161        json!({
162            "mode": "rflag",
163            "implementation": "casa-rs-casa-shaped-rflag",
164            "selected_rows": selected_rows.len(),
165            "samples": samples.len(),
166            "groups": groups.len(),
167            "timedev_by_field_spw": format_threshold_map(&thresholds.timedev),
168            "freqdev_by_field_spw": format_threshold_map(&thresholds.freqdev),
169            "time_candidate_samples": count_mask(&time_flags),
170            "freq_candidate_samples": count_mask(&freq_flags),
171            "union_candidate_samples": union_candidates,
172            "union_by_spw_corr": flag_mask_counts_by_spw_corr(&modified, &samples),
173            "timing_ms": {
174                "load": loaded_ms,
175                "threshold": threshold_ms,
176                "plan": planned_ms.saturating_sub(loaded_ms + threshold_ms),
177            },
178        })
179    });
180    let mut changes = if let Some(view) = view {
181        view.stage(&samples, &modified, request.action.value())
182    } else {
183        apply_tfcrop_mask(
184            &samples,
185            &modified,
186            loaded.flags_by_row,
187            &loaded.axes,
188            ms,
189            !request.casa_row_selection,
190            request.action.value(),
191        )?
192    };
193    changes.selected_rows = loaded.axes.time_selected_rows(averaged_time);
194    let applied_ms = started.elapsed().as_millis();
195    changes.thresholds = if casa_rflag {
196        buffer_cutoffs
197            .first()
198            .copied()
199            .filter(|cutoff| buffer_cutoffs.iter().all(|candidate| candidate == cutoff))
200    } else {
201        Some(thresholds.representative_pair())
202    };
203    trace_flagdata(|| {
204        json!({
205            "mode": "rflag",
206            "implementation": "casa-rs-casa-shaped-rflag",
207            "timing_ms": {
208                "apply": applied_ms.saturating_sub(planned_ms),
209                "total_before_save": applied_ms,
210            },
211        })
212    });
213    Ok(changes)
214}

216fn extend_rflag_selected_channels(
217    groups: &[(AutoflagChunk, Vec<Sample>)],
218    axes: &AveragedSampleAxes,
219    flags_by_row: &BTreeMap<usize, ArrayD<bool>>,
220    modified: &mut [bool],
221    request: &FlagDataRequest,
222) -> Result<(), FlaggingError> {
223    let fit_flags = axes.flags_by_bin.as_ref().unwrap_or(flags_by_row);
224    for (_, group) in groups {
225        request.check_cancelled()?;
226        extend_tfcrop_selected_channels(group, fit_flags, modified);
227    }
228    Ok(())
229}

239pub(super) fn load_automatic_samples(
240    ms: &MeasurementSet,
241    rows: &[usize],
242    channels: Option<&BTreeMap<i32, ChannelSelection>>,
243    request: &FlagDataRequest,
244    source: AutomaticSampleSource<'_>,
245) -> Result<LoadedSamples, FlaggingError> {
246    let AutomaticSampleSource {
247        mut columns,
248        fit,
249        seconds,
250        widths,
251        input,
252    } = source;
253    let column = request.data_column;
254    if let Some(view) = input.view {
255        let path = display_ms_path(ms);
256        if let Some(columns) = &mut columns {
257            view.prepare_column(ms, column, columns, &path)?;
258        }
259        return view.load_samples(
260            SharedAgentSelection {
261                rows: Some(rows),
262                channels,
263                correlations: None,
264                time: input.time,
265                uv: input.uv,
266            },
267            columns.map(|_| column),
268            request,
269            &path,
270        );
271    }
272    if let Some(seconds) = seconds {
273        return load_time_averaged_tfcrop(
274            ms,
275            rows,
276            channels,
277            columns,
278            request,
279            clip_time::TimeAveraging {
280                seconds,
281                filter: input.time,
282            },
283            widths,
284        );
285    }
286    load_samples(
287        ms,
288        rows,
289        channels,
290        fit.then(|| {
291            columns.map_or(SampleData::Stored(column), |operands| {
292                SampleData::Visibility(column, operands)
293            })
294        }),
295        None,
296        SampleLoadOptions {
297            widths,
298            cancellation: request.cancellation.as_ref(),
299            ..SampleLoadOptions::default()
300        },
301    )
302}

304pub(super) fn time_selected_autoflag_groups<'a, K: Copy>(
305    groups: &'a [(K, Vec<Sample>)],
306    axes: &AveragedSampleAxes,
307) -> Cow<'a, [(K, Vec<Sample>)]> {
308    if axes.rows_outside_time.is_empty() {
309        return Cow::Borrowed(groups);
310    }
311    Cow::Owned(
312        groups
313            .iter()
314            .filter_map(|(key, samples)| {
315                let selected = samples
316                    .iter()
317                    .copied()
318                    .filter(|sample| !axes.rows_outside_time.contains(&sample.row))
319                    .collect::<Vec<_>>();
320                (!selected.is_empty()).then_some((*key, selected))
321            })
322            .collect(),
323    )
324}

334pub(super) fn rflag_group_projections(
335    group: &[Sample],
336    description: Option<i32>,
337    context: RflagContext<'_>,
338) -> Result<Vec<ClipProjection>, FlaggingError> {
339    if let Some((selection, ddid)) = context.correlations.zip(description) {
340        return selection
341            .for_description(ddid)
342            .map(<[ClipProjection]>::to_vec);
343    }
344    Ok(samples_by_corr(group)
345        .into_iter()
346        .filter(|plane| {
347            context
348                .calibration
349                .is_none_or(|projection| projection.includes(plane[0].corr))
350        })
351        .map(|plane| ClipProjection::Stored(plane[0].corr))
352        .collect())
353}

355pub(super) fn validate_rflag_matrix_scales(
356    groups: &[(AutoflagChunk, Vec<Sample>)],
357    request: &FlagDataRequest,
358    context: RflagContext<'_>,
359) -> Result<(), FlaggingError> {
360    for (scale_name, scale, thresholds) in [
361        (
362            "timedevscale",
363            request.timedevscale,
364            &request.timedev_by_field_spw,
365        ),
366        (
367            "freqdevscale",
368            request.freqdevscale,
369            &request.freqdev_by_field_spw,
370        ),
371    ] {
372        if scale > 0.0 || thresholds.is_empty() {
373            continue;
374        }
375        for (chunk, group) in groups {
376            let Some(first) = group.first() else {
377                continue;
378            };
379            if !thresholds
380                .get(&(first.field, first.spw))
381                .is_some_and(|&value| value > 0.0)
382            {
383                continue;
384            }
385            let ddid = context.correlations.map(|_| chunk.0[4]);
386            if !rflag_group_projections(group, ddid, context)?.is_empty() {
387                return Err(FlaggingError::InvalidRFlagScale(scale_name));
388            }
389        }
390    }
391    Ok(())
392}

401pub(super) fn fit_rflag_chunks(
402    groups: &[(AutoflagChunk, Vec<Sample>)],
403    request: &FlagDataRequest,
404    context: RflagContext<'_>,
405    masks: &mut TfcropFitMasks<'_>,
406) -> Result<RflagFitOutcome, FlaggingError> {
407    let mut thresholds = RFlagThresholds::default();
408    let mut buffer_cutoffs = Vec::new();
409    let mut threshold_ms = 0;
410    let mut roundoff_buf = if context.model_roundoff.is_empty() {
411        Vec::new()
412    } else {
413        vec![0.0; masks.modified.len()]
414    };
415    // RFlag applies each buffer's own fitted thresholds. The calculate report
416    // instead accumulates all buffers for each field/SPW.
417    for chunk in groups.chunk_by(|(left, _), (right, _)| left == right) {
418        request.check_cancelled()?;
419        let threshold_started = Instant::now();
420        let chunk_thresholds =
421            rflag_threshold_maps_with_roundoff(chunk, request, context, &mut roundoff_buf)?
422                .scaled(request, context.casa_arithmetic);
423        threshold_ms += threshold_started.elapsed().as_millis();
424        for (chunk_key, group) in chunk {
425            let ddid = context.correlations.map(|_| chunk_key.0[4]);
426            rflag_group_flags(
427                ddid,
428                group,
429                &chunk_thresholds,
430                request,
431                context,
432                &mut roundoff_buf,
433                masks,
434            )?;
435        }
436        if context.casa_arithmetic {
437            let ([observation, array, scan, field, window], _) = chunk[0].0;
438            let cutoff = chunk_thresholds.representative_pair();
439            buffer_cutoffs.push(cutoff);
440            trace_flagdata(|| {
441                json!({
442                    "mode": "rflag",
443                    "calibration_buffer": {
444                        "observation": observation,
445                        "array": array,
446                        "scan": scan,
447                        "field": field,
448                        "spw": window,
449                    },
450                    "timedev": cutoff.0,
451                    "freqdev": cutoff.1,
452                })
453            });
454        } else {
455            thresholds = chunk_thresholds;
456        }
457    }
458    Ok(RflagFitOutcome {
459        thresholds,
460        buffer_cutoffs,
461        threshold_ms,
462    })
463}

465pub(super) fn extend_autoflag_grouped<'a>(
466    groups: impl IntoIterator<Item = &'a [Sample]>,
467    samples: &[Sample],
468    modified: &mut [bool],
469) {
470    let mut grown = modified.to_vec();
471    for group in groups {
472        for by_corr in samples_by_corr(group) {
473            for by_row in sample_row_runs(&by_corr) {
474                if percent_modified(by_row, modified) >= 80.0 {
475                    for sample in by_row {
476                        grown[sample.index] = true;
477                    }
478                }
479            }
480            for by_chan in samples_by_chan_dense(&by_corr) {
481                if percent_modified(&by_chan, modified) >= 50.0 {
482                    for sample in &by_chan {
483                        grown[sample.index] = true;
484                    }
485                }
486            }
487        }
488    }
489    modified.copy_from_slice(&grown);
490    for group in samples_by_row_chan(samples).values() {
491        if group
492            .iter()
493            .any(|sample| sample.flag || modified[sample.index])
494        {
495            for sample in group {
496                modified[sample.index] = true;
497            }
498        }
499    }
500}

502pub(super) fn rflag_threshold_maps_for_groups(
503    groups: &[(AutoflagChunk, Vec<Sample>)],
504    request: &FlagDataRequest,
505    context: RflagContext<'_>,
506) -> Result<RFlagThresholds, FlaggingError> {
507    if request.rflag_window.is_none() {
508        return Ok(RFlagThresholds::default());
509    }
510    let mut roundoff_buf = if context.model_roundoff.is_empty() {
511        Vec::new()
512    } else {
513        vec![0.0; context.model_roundoff.len()]
514    };
515    rflag_threshold_maps_with_roundoff(groups, request, context, &mut roundoff_buf)
516}
517
518pub(super) fn rflag_threshold_maps_with_roundoff(
519    groups: &[(AutoflagChunk, Vec<Sample>)],
520    request: &FlagDataRequest,
521    context: RflagContext<'_>,
522    roundoff_buf: &mut [f32],
523) -> Result<RFlagThresholds, FlaggingError> {
524    let Some(window_size) = request.rflag_window else {
525        return Ok(RFlagThresholds::default());
526    };
527    let mut histograms = RFlagHistograms::default();
528    for (chunk_key, group) in groups {
529        request.check_cancelled()?;
530        let ddid = context.correlations.map(|_| chunk_key.0[4]);
531        rflag_accumulate_group_histograms(
532            ddid,
533            group,
534            &mut histograms,
535            window_size,
536            context,
537            roundoff_buf,
538        )?;
539    }
540    Ok(histograms.into_thresholds(request, context.casa_arithmetic))
541}

543#[derive(Debug, Default)]
544pub(super) struct RFlagHistograms {
545    pub(super) time: BTreeMap<FieldSpwKey, ChannelHistogram>,
546    pub(super) freq: BTreeMap<FieldSpwKey, ChannelHistogram>,
547}
548
549impl RFlagHistograms {
550    /// Unscaled thresholds for the public calculate report and flagging.
551    /// CASA applies the scale factors only when testing for flags.
552    fn into_thresholds(self, request: &FlagDataRequest, casa_arithmetic: bool) -> RFlagThresholds {
553        let cutoffs = |histograms: BTreeMap<FieldSpwKey, ChannelHistogram>,
554                       scalar: Option<f64>,
555                       supplied: &BTreeMap<FieldSpwKey, f64>| {
556            histograms
557                .into_iter()
558                .map(|(key, histogram)| {
559                    let threshold = supplied
560                        .get(&key)
561                        .copied()
562                        .filter(|&value| value > 0.0)
563                        .or(scalar.filter(|&value| !casa_arithmetic || value > 0.0))
564                        .unwrap_or_else(|| histogram.threshold());
565                    (key, threshold)
566                })
567                .collect()
568        };
569        RFlagThresholds {
570            timedev: cutoffs(self.time, request.timedev, &request.timedev_by_field_spw),
571            freqdev: cutoffs(self.freq, request.freqdev, &request.freqdev_by_field_spw),
572        }
573    }
574}

576#[derive(Debug, Default)]
577pub(super) struct RFlagThresholds {
578    pub(super) timedev: BTreeMap<FieldSpwKey, f64>,
579    pub(super) freqdev: BTreeMap<FieldSpwKey, f64>,
580}
581
582impl RFlagThresholds {
583    pub(super) fn scaled(mut self, request: &FlagDataRequest, casa_arithmetic: bool) -> Self {
584        // CASA uses -1 for an unavailable applied cutoff. A negative scalar
585        // skips the fitted cutoff, and a negative scale skips the prepass
586        // histogram. CASA scales its sentinel together with defined cutoffs.
587        let applied = |threshold: f64, scalar: Option<f64>, supplied: Option<f64>, scale: f64| {
588            if let Some(value) = supplied {
589                return if value > 0.0 { value * scale } else { -scale };
590            }
591            if casa_arithmetic
592                && !scalar.is_some_and(|value| value > 0.0)
593                && (threshold == 0.0 || scale < 0.0 || scalar.is_some_and(|value| value < 0.0))
594            {
595                -scale
596            } else {
597                threshold * scale
598            }
599        };
600        for (key, value) in &mut self.timedev {
601            *value = applied(
602                *value,
603                request.timedev,
604                request.timedev_by_field_spw.get(key).copied(),
605                request.timedevscale,
606            );
607        }
608        for (key, value) in &mut self.freqdev {
609            *value = applied(
610                *value,
611                request.freqdev,
612                request.freqdev_by_field_spw.get(key).copied(),
613                request.freqdevscale,
614            );
615        }
616        self
617    }
618
619    fn public_report(&self, request: &FlagDataRequest, active: bool) -> RFlagThresholdReport {
620        let rows = |values: &BTreeMap<FieldSpwKey, f64>,
621                    scalar: Option<f64>,
622                    supplied: &BTreeMap<FieldSpwKey, f64>| {
623            if !active || request.rflag_window.is_none() {
624                return Vec::new();
625            }
626            let mut report = supplied.clone();
627            if !scalar.is_some_and(|value| value > 0.0) {
628                report.extend(values);
629            }
630            report
631                .into_iter()
632                .map(|((field, spw), value)| [f64::from(field), f64::from(spw), value])
633                .collect()
634        };
635        RFlagThresholdReport {
636            timedev: rows(
637                &self.timedev,
638                request.timedev,
639                &request.timedev_by_field_spw,
640            ),
641            freqdev: rows(
642                &self.freqdev,
643                request.freqdev,
644                &request.freqdev_by_field_spw,
645            ),
646        }
647    }
648
649    pub(super) fn representative_pair(&self) -> (f64, f64) {
650        (
651            representative_threshold(&self.timedev),
652            representative_threshold(&self.freqdev),
653        )
654    }
655}

657#[derive(Debug, Clone)]
658pub(super) struct ChannelHistogram {
659    pub(super) sum: Vec<f64>,
660    pub(super) counts: Vec<f64>,
661}
662
663impl ChannelHistogram {
664    pub(super) fn new(nchan: usize) -> Self {
665        Self {
666            sum: vec![0.0; nchan],
667            counts: vec![0.0; nchan],
668        }
669    }
670
671    pub(super) fn add(&mut self, chan: usize, value: f64) {
672        if let (Some(sum), Some(count)) = (self.sum.get_mut(chan), self.counts.get_mut(chan)) {
673            *sum += value;
674            *count += 1.0;
675        }
676    }
677
678    pub(super) fn threshold(&self) -> f64 {
679        let samples = self
680            .sum
681            .iter()
682            .zip(self.counts.iter())
683            .map(|(sum, count)| if *count > 0.0 { *sum / *count } else { 0.0 })
684            .collect::<Vec<_>>();
685        let med = median(samples.clone());
686        let mad = median(samples.iter().map(|value| (*value - med).abs()).collect());
687        med + 1.4826 * mad
688    }
689}

698pub(super) fn rflag_accumulate_group_histograms(
699    description: Option<i32>,
700    group: &[Sample],
701    histograms: &mut RFlagHistograms,
702    window_size: usize,
703    context: RflagContext<'_>,
704    roundoff_buf: &mut [f32],
705) -> Result<(), FlaggingError> {
706    let Some(first) = group.first() else {
707        return Ok(());
708    };
709    let planes = samples_by_corr(group);
710    let projections = rflag_group_projections(group, description, context)?;
711    if projections.is_empty() {
712        return Ok(());
713    }
714    let key = (first.field, first.spw);
715    // CASA histograms have one bin per selected channel.
716    let channels = context.casa_arithmetic.then(|| {
717        let mut channels = unique_sample_chans(group);
718        channels.sort_unstable();
719        channels
720    });
721    let nchan = channels.as_ref().map_or_else(
722        || group.iter().map(|sample| sample.chan).max().unwrap_or(0) + 1,
723        Vec::len,
724    );
725    let index = |channel| {
726        channels.as_ref().map_or(channel, |channels| {
727            channels.binary_search(&channel).expect("loaded channel")
728        })
729    };
730    histograms
731        .time
732        .entry(key)
733        .or_insert_with(|| ChannelHistogram::new(nchan));
734    histograms
735        .freq
736        .entry(key)
737        .or_insert_with(|| ChannelHistogram::new(nchan));
738    for projection in projections {
739        let Some(series) = tfcrop_series(
740            &planes,
741            projection,
742            None,
743            &[],
744            context.model_roundoff,
745            roundoff_buf,
746        ) else {
747            continue;
748        };
749        let time_hist = histograms
750            .time
751            .get_mut(&key)
752            .expect("initialized time histogram");
753        for by_chan in samples_by_chan_dense(&series.samples) {
754            for (start, stop) in rflag_time_windows(by_chan.len(), window_size) {
755                let window = &by_chan[start..=stop];
756                if let Some(value) = rflag_time_std_total(
757                    window,
758                    context.casa_arithmetic,
759                    RflagRoundoff::Slice(roundoff_buf),
760                ) {
761                    time_hist.add(index(by_chan[0].chan), value);
762                }
763            }
764        }
765        let freq_hist = histograms
766            .freq
767            .get_mut(&key)
768            .expect("initialized frequency histogram");
769        for by_row in sample_row_runs(&series.samples) {
770            let stats = rflag_spectral_stats(
771                by_row,
772                context.casa_arithmetic,
773                RflagRoundoff::Slice(roundoff_buf),
774            );
775            let row_roundoff = rflag_row_roundoff(by_row, RflagRoundoff::Slice(roundoff_buf));
776            for sample in by_row {
777                if sample.flag {
778                    continue;
779                }
780                let bound = f64::from(
781                    roundoff_buf.get(sample.index).copied().unwrap_or(0.0) + row_roundoff,
782                );
783                if stats.sum_weight_real > 0.0 {
784                    freq_hist.add(
785                        index(sample.chan),
786                        rflag_clamped_deviation((sample.real - stats.average_real).abs(), bound),
787                    );
788                }
789                if stats.sum_weight_imag > 0.0 {
790                    freq_hist.add(
791                        index(sample.chan),
792                        rflag_clamped_deviation((sample.imag - stats.average_imag).abs(), bound),
793                    );
794                }
795            }
796        }
797    }
798    Ok(())
799}

801#[derive(Clone, Copy)]
802pub(super) enum RflagRoundoff<'a> {
803    Slice(&'a [f32]),
804    Sparse(&'a HashMap<usize, f32>),
805}
806
807impl RflagRoundoff<'_> {
808    fn is_empty(self) -> bool {
809        match self {
810            Self::Slice(slice) => slice.is_empty(),
811            Self::Sparse(map) => map.is_empty(),
812        }
813    }
814
815    fn get(self, index: usize) -> f32 {
816        match self {
817            Self::Slice(slice) => slice.get(index).copied().unwrap_or(0.0),
818            Self::Sparse(map) => map.get(&index).copied().unwrap_or(0.0),
819        }
820    }
821}

830pub(super) fn rflag_projection_series(
831    planes: &[Vec<Sample>],
832    projection: ClipProjection,
833    model_roundoff: &[f32],
834    roundoff_buf: &mut [f32],
835) -> Option<RflagProjectionSeries> {
836    let series = tfcrop_series(planes, projection, None, &[], model_roundoff, roundoff_buf)?;
837    let rows = sample_row_runs(&series.samples)
838        .into_iter()
839        .map(<[Sample]>::to_vec)
840        .collect::<Vec<_>>();
841    let chans = samples_by_chan_dense(&series.samples);
842    let roundoff = if roundoff_buf.is_empty() {
843        HashMap::new()
844    } else {
845        series
846            .samples
847            .iter()
848            .map(|sample| {
849                (
850                    sample.index,
851                    roundoff_buf.get(sample.index).copied().unwrap_or(0.0),
852                )
853            })
854            .collect()
855    };
856    Some(RflagProjectionSeries {
857        rows,
858        chans,
859        partners: series.pairs.into_iter().collect(),
860        roundoff,
861    })
862}

890pub(super) fn rflag_group_flags(
891    description: Option<i32>,
892    group: &[Sample],
893    thresholds: &RFlagThresholds,
894    request: &FlagDataRequest,
895    context: RflagContext<'_>,
896    roundoff_buf: &mut [f32],
897    masks: &mut TfcropFitMasks<'_>,
898) -> Result<(), FlaggingError> {
899    let Some(window_size) = request.rflag_window else {
900        return Ok(());
901    };
902    let Some(first) = group.first() else {
903        return Ok(());
904    };
905    let key = (first.field, first.spw);
906    let timedev = thresholds
907        .timedev
908        .get(&key)
909        .copied()
910        .unwrap_or(f64::INFINITY);
911    let freqdev = thresholds
912        .freqdev
913        .get(&key)
914        .copied()
915        .unwrap_or(f64::INFINITY);
916    let planes = samples_by_corr(group);
917    let mut projections = Vec::new();
918    for projection in rflag_group_projections(group, description, context)? {
919        if let Some(series) =
920            rflag_projection_series(&planes, projection, context.model_roundoff, roundoff_buf)
921        {
922            projections.push(series);
923        }
924    }
925    let row_count = projections.first().map_or(0, |series| series.rows.len());
926    let do_time = timedev >= 0.0 || !context.casa_arithmetic;
927    let do_freq = freqdev >= 0.0 || !context.casa_arithmetic;
928    for timestep in 0..row_count {
929        if do_time
930            && let Some((start, stop)) = rflag_time_window_at(row_count, window_size, timestep)
931        {
932            for series in &projections {
933                let roundoff = RflagRoundoff::Sparse(&series.roundoff);
934                for by_chan in &series.chans {
935                    let Some(window) = by_chan.get(start..=stop) else {
936                        continue;
937                    };
938                    if rflag_time_std_total(window, context.casa_arithmetic, roundoff)
939                        .is_some_and(|deviation| deviation > timedev)
940                    {
941                        for sample in window {
942                            if !is_rflag_flagged(sample, &series.partners, masks.modified) {
943                                mark_rflag_flagged(
944                                    &series.partners,
945                                    sample.index,
946                                    masks.time_flags,
947                                    masks.modified,
948                                );
949                            }
950                        }
951                    }
952                }
953            }
954        }
955        if do_freq {
956            for series in &projections {
957                let Some(by_row) = series.rows.get(timestep) else {
958                    continue;
959                };
960                let roundoff = RflagRoundoff::Sparse(&series.roundoff);
961                let stats = rflag_spectral_stats(by_row, context.casa_arithmetic, roundoff);
962                let row_roundoff = rflag_row_roundoff(by_row, roundoff);
963                let flag_all = stats.std_real > request.spectralmax
964                    || stats.std_imag > request.spectralmax
965                    || stats.std_real < request.spectralmin
966                    || stats.std_imag < request.spectralmin;
967                for sample in by_row {
968                    if is_rflag_flagged(sample, &series.partners, masks.modified) {
969                        continue;
970                    }
971                    let bound = f64::from(roundoff.get(sample.index) + row_roundoff);
972                    let dev_real =
973                        rflag_clamped_deviation((sample.real - stats.average_real).abs(), bound);
974                    let dev_imag =
975                        rflag_clamped_deviation((sample.imag - stats.average_imag).abs(), bound);
976                    if flag_all || dev_real > freqdev || dev_imag > freqdev {
977                        mark_rflag_flagged(
978                            &series.partners,
979                            sample.index,
980                            masks.freq_flags,
981                            masks.modified,
982                        );
983                    }
984                }
985            }
986        }
987    }
988    Ok(())
989}

991pub(super) fn rflag_time_window_at(
992    len: usize,
993    window_size: usize,
994    center: usize,
995) -> Option<(usize, usize)> {
996    let effective = len.min(window_size);
997    let delta = effective.saturating_sub(1) / 2;
998    if delta == 0 {
999        (center < len).then_some((center, center))
1000    } else if center >= delta && center < len - delta {
1001        Some((center - delta, center + delta))
1002    } else {
1003        None
1004    }
1005}

1007pub(super) fn rflag_time_windows(
1008    len: usize,
1009    window_size: usize,
1010) -> impl Iterator<Item = (usize, usize)> {
1011    (0..len).filter_map(move |center| rflag_time_window_at(len, window_size, center))
1012}

1014pub(super) fn rflag_time_std_total(
1015    samples: &[Sample],
1016    casa_arithmetic: bool,
1017    roundoff: RflagRoundoff<'_>,
1018) -> Option<f64> {
1019    // CASA's visibility mapper supplies Float values and sums v and v * v in Double.
1020    let term = |value: f64| {
1021        if casa_arithmetic {
1022            let value = value as f32;
1023            (f64::from(value), f64::from(value * value))
1024        } else {
1025            (value, value * value)
1026        }
1027    };
1028    let mut count = 0usize;
1029    let mut sum_real = 0.0;
1030    let mut sum_imag = 0.0;
1031    let mut sumsq_real = 0.0;
1032    let mut sumsq_imag = 0.0;
1033    for sample in samples {
1034        if sample.flag {
1035            continue;
1036        }
1037        count += 1;
1038        let (real, real_sq) = term(sample.real);
1039        let (imag, imag_sq) = term(sample.imag);
1040        sum_real += real;
1041        sum_imag += imag;
1042        sumsq_real += real_sq;
1043        sumsq_imag += imag_sq;
1044    }
1045    if count == 0 {
1046        return (!casa_arithmetic).then_some(0.0);
1047    }
1048    let sum_weight = count as f64;
1049    let avg_real = sum_real / sum_weight;
1050    let avg_imag = sum_imag / sum_weight;
1051    if !roundoff.is_empty()
1052        && samples.iter().filter(|sample| !sample.flag).all(|sample| {
1053            let bound = f64::from(roundoff.get(sample.index)) * 2.0;
1054            (sample.real - avg_real).abs() <= bound && (sample.imag - avg_imag).abs() <= bound
1055        })
1056    {
1057        return Some(0.0);
1058    }
1059    let var_real = sumsq_real / sum_weight - avg_real * avg_real;
1060    let var_imag = sumsq_imag / sum_weight - avg_imag * avg_imag;
1061    Some((var_real.max(0.0) + var_imag.max(0.0)).sqrt())
1062}
1063
1064pub(super) fn rflag_row_roundoff(samples: &[Sample], roundoff: RflagRoundoff<'_>) -> f32 {
1065    if roundoff.is_empty() {
1066        return 0.0;
1067    }
1068    samples
1069        .iter()
1070        .filter(|sample| !sample.flag)
1071        .map(|sample| roundoff.get(sample.index))
1072        .filter(|bound| bound.is_finite())
1073        .fold(0.0, f32::max)
1074}
1075
1076pub(super) fn rflag_clamped_deviation(deviation: f64, roundoff_bound: f64) -> f64 {
1077    if roundoff_bound > 0.0 && deviation <= roundoff_bound {
1078        0.0
1079    } else {
1080        deviation
1081    }
1082}

1084#[derive(Debug, Clone, Copy, Default)]
1085pub(super) struct RFlagSpectralStats {
1086    pub(super) average_real: f64,
1087    pub(super) average_imag: f64,
1088    pub(super) std_real: f64,
1089    pub(super) std_imag: f64,
1090    pub(super) sum_weight_real: f64,
1091    pub(super) sum_weight_imag: f64,
1092}
1093
1094pub(super) fn rflag_spectral_stats(
1095    samples: &[Sample],
1096    casa_arithmetic: bool,
1097    roundoff: RflagRoundoff<'_>,
1098) -> RFlagSpectralStats {
1099    let mut real_values = Vec::with_capacity(samples.len());
1100    let mut imag_values = Vec::with_capacity(samples.len());
1101    for sample in samples {
1102        if sample.flag {
1103            continue;
1104        }
1105        real_values.push(sample.real);
1106        imag_values.push(sample.imag);
1107    }
1108    if real_values.is_empty() {
1109        return RFlagSpectralStats::default();
1110    }
1111    let average_real = median_in_place(&mut real_values);
1112    let average_imag = median_in_place(&mut imag_values);
1113    let row_bound = f64::from(rflag_row_roundoff(samples, roundoff)) * 2.0;
1114    let residual = |value: f64, reference: f64| {
1115        let delta = value - reference;
1116        if row_bound > 0.0 && delta.abs() <= row_bound {
1117            0.0
1118        } else if casa_arithmetic {
1119            delta
1120        } else {
1121            delta.abs()
1122        }
1123    };
1124    for value in &mut real_values {
1125        *value = residual(*value, average_real);
1126    }
1127    for value in &mut imag_values {
1128        *value = residual(*value, average_imag);
1129    }
1130    let std_real = 1.4826 * median_in_place(&mut real_values);
1131    let std_imag = 1.4826 * median_in_place(&mut imag_values);
1132    let weight = real_values.len() as f64;
1133    RFlagSpectralStats {
1134        average_real,
1135        average_imag,
1136        std_real,
1137        std_imag,
1138        sum_weight_real: weight,
1139        sum_weight_imag: weight,
1140    }
1141}
