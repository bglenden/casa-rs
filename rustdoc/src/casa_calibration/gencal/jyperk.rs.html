27#[derive(Debug, Error)]
28pub enum JyperkError {
29    #[error("There is no Jy/K db-derived factor file: {0}")]
30    MissingFile(PathBuf),
31    #[error("read Jy/K factor file {path}: {source}")]
32    ReadFile {
33        path: PathBuf,
34        source: std::io::Error,
35    },
36    #[error("read Jy/K factor file {path}: {source}")]
37    ParseFile {
38        path: PathBuf,
39        source: std::str::Utf8Error,
40    },
41    #[error("generator raised StopIteration")]
42    EmptyFile,
43    #[error("No valid Jy/K factors found for this MS")]
44    NoFactors,
45    #[error("could not convert string to float: {0}")]
46    InvalidNumber(String),
47    #[error(
48        "Jy/K factor {factor:?} must yield a finite positive gain for antenna {antenna:?} and SPW {window:?}"
49    )]
50    InvalidFactor {
51        factor: String,
52        antenna: String,
53        window: String,
54    },
55    #[error("{0}")]
56    Service(String),
57    #[error("No Jy/K factors available: JYPERKDB_URL, backup_hosts, and infile all failed")]
58    AllSourcesFailed,
59    #[error(transparent)]
60    Generate(#[from] GencalError),
61}

63#[derive(Debug)]
64pub(super) struct Factor {
65    antenna: String,
66    window: String,
67    polarization: &'static str,
68    gain: f64,
69}
70
71impl Factor {
72    fn parse(
73        antenna: &str,
74        window: &str,
75        polarization: &str,
76        value: &str,
77    ) -> Result<Self, JyperkError> {
78        let factor = value
79            .trim()
80            .parse::<f64>()
81            .map_err(|_| JyperkError::InvalidNumber(python_string_repr(value)))?;
82        let gain = 1.0 / factor.sqrt();
83        if !(gain as f32).is_finite() || gain as f32 <= 0.0 {
84            return Err(JyperkError::InvalidFactor {
85                factor: value.to_string(),
86                antenna: antenna.to_string(),
87                window: window.to_string(),
88            });
89        }
90        let polarization = match polarization {
91            "XX" => "X",
92            "YY" => "Y",
93            "RR" => "R",
94            "LL" => "L",
95            _ => "",
96        };
97        Ok(Self {
98            antenna: antenna.to_string(),
99            window: window.to_string(),
100            polarization,
101            gain,
102        })
103    }
104}

106pub fn calibrate(
107    request: &GencalRequest,
108    path: Option<&Path>,
109    options: &ServiceOptions,
110    backup_hosts: Option<&[String]>,
111    mode: CalibrationOutputMode,
112) -> Result<GencalReport, JyperkError> {
113    let ms = open_measurement_set(&request.measurement_set)?;
114    let file_factors = |path| {
115        refuse_prior_input_overlap(request, path)?;
116        let name = request.measurement_set.file_name().unwrap_or_default();
117        read_factors(path, &name.to_string_lossy())
118    };
119    let env_url = std::env::var("JYPERKDB_URL").ok();
120    let factors = if let Some(hosts) = backup_hosts {
121        let mut result = Vec::new();
122        for host in env_url
123            .iter()
124            .filter(|url| !url.trim().is_empty())
125            .chain(hosts)
126        {
127            match service::lookup(&ms, request, options, host) {
128                Ok(factors) if !factors.is_empty() => {
129                    result = factors;
130                    break;
131                }
132                Ok(_) => {}
133                Err(error) => tracing::warn!(%error, "Jy/K service lookup failed"),
134            }
135        }
136        if !result.is_empty() {
137            service_factors(result)?
138        } else {
139            file_factors(path.ok_or(JyperkError::AllSourcesFailed)?)?
140        }
141    } else if let Some(path) = path {
142        file_factors(path)?
143    } else {
144        service_factors(service::lookup(
145            &ms,
146            request,
147            options,
148            env_url.as_deref().unwrap_or(service::DEFAULT_HOST),
149        )?)?
150    };
151    Ok(generate(&ms, request, &factors, mode)?)
152}

154fn service_factors(rows: Vec<service::DatabaseFactor>) -> Result<Vec<Factor>, JyperkError> {
155    if rows.is_empty() {
156        return Err(JyperkError::NoFactors);
157    }
158    rows.iter()
159        .map(|row| Factor::parse(&row.antenna, &row.window, "I", &row.value))
160        .collect()
161}

163fn read_factors(path: &Path, measurement_set: &str) -> Result<Vec<Factor>, JyperkError> {
164    if !path.is_file() {
165        return Err(JyperkError::MissingFile(path.to_path_buf()));
166    }
167    let io_error = |source| JyperkError::ReadFile {
168        path: path.to_path_buf(),
169        source,
170    };
171    let content = fs::read(path).map_err(io_error)?;
172    if content.is_empty() {
173        return Err(JyperkError::EmptyFile);
174    }
175    // Validate before CSV removes separators. Otherwise malformed adjacent
176    // fields could form a UTF-8 character spanning one of the field offsets.
177    let encoding_error = |source| JyperkError::ParseFile {
178        path: path.to_path_buf(),
179        source,
180    };
181    std::str::from_utf8(&content).map_err(encoding_error)?;
182    let mut input = content.as_slice();
183    let mut reader = csv_core::Reader::new();
184    let mut record = vec![0; 256];
185    let mut ends = vec![0; 8];
186    let mut factors = Vec::new();
187    let mut first = !input.starts_with(b"\r") && !input.starts_with(b"\n");
188    let mut len = 0;
189    let mut fields = 0;
190    loop {
191        let (status, consumed, bytes, cells) =
192            reader.read_record(input, &mut record[len..], &mut ends[fields..]);
193        input = &input[consumed..];
194        len += bytes;
195        fields += cells;
196        match status {
197            ReadRecordResult::End => break,
198            ReadRecordResult::InputEmpty => continue,
199            ReadRecordResult::OutputFull => {
200                record.resize(record.len() * 2, 0);
201                continue;
202            }
203            ReadRecordResult::OutputEndsFull => {
204                ends.resize(ends.len() * 2, 0);
205                continue;
206            }
207            ReadRecordResult::Record => {}
208        }
209        let text = std::str::from_utf8(&record[..len]).map_err(encoding_error)?;
210        let name = &text[..ends[0]];
211        if !(name.is_empty()
212            || name.starts_with('#')
213            || (first
214                && (name.trim().eq_ignore_ascii_case("MS") || name.trim_start().starts_with('#'))))
215        {
216            if fields != 5 {
217                tracing::warn!("Jy/K factor file is invalid format");
218            } else if name == measurement_set {
219                let cell = |index: usize| &text[ends[index - 1]..ends[index]];
220                factors.push(Factor::parse(cell(1), cell(2), cell(3), cell(4))?);
221            }
222        }
223        first = false;
224        len = 0;
225        fields = 0;
226    }
227    if factors.is_empty() {
228        return Err(JyperkError::NoFactors);
229    }
230    Ok(factors)
231}

233fn generate(
234    ms: &MeasurementSet,
235    request: &GencalRequest,
236    factors: &[Factor],
237    mode: CalibrationOutputMode,
238) -> Result<GencalReport, GencalError> {
239    let output = CalibrationOutput::prepare(&request.output_table, mode)?;
240    let updating = mode == CalibrationOutputMode::Update && request.output_table.exists();
241    let mut rows = None;
242    let mut update = request.clone();
243    update.parameter = vec![1.0];
244    for factor in factors {
245        update.antenna.clone_from(&factor.antenna);
246        update.spw.clone_from(&factor.window);
247        update.parameter[0] = factor.gain;
248        apply_specified_rows(ms, &update, factor.polarization, updating, &mut rows)?;
249    }
250    let mut rows = rows.expect("validated Jy/K factors");
251    if factors.len() > 1 {
252        // The first specifycal creates the full grid. Each subsequent
253        // specifycal update leaves the observation identifier at -1.
254        for row in &mut rows {
255            row.observation_id = -1;
256        }
257    }
258    write_prior_caltable(ms, request, "G Jones", &rows, output, mode, true)
259}
