10#[derive(Clone, Copy, Default)]
11struct TileRun {
12    offset: usize,
13    count: usize,
14}
15
16impl TiledFileIO {
17    pub(super) fn get_strided_slice<T: TilePixel>(
18        &mut self,
19        start: &[usize],
20        shape: &[usize],
21        stride: &[usize],
22    ) -> Result<ArrayD<T>, StorageError> {
23        let rank = self.cube_shape.len();
24        if rank == 0
25            || start.len() != rank
26            || shape.len() != rank
27            || stride.len() != rank
28            || stride.contains(&0)
29            || shape.contains(&0)
30            || (0..rank).any(|axis| {
31                (shape[axis] - 1)
32                    .checked_mul(stride[axis])
33                    .and_then(|range| start[axis].checked_add(range))
34                    .is_none_or(|last| last >= self.cube_shape[axis])
35            })
36        {
37            return Err(StorageError::FormatMismatch(
38                "invalid tiled slice bounds or destination".into(),
39            ));
40        }
41
42        let mut values = vec![T::default(); checked_storage_product(shape, "slice size")?];
43        let output_strides = fortran_order_strides(shape);
44        let mut samples = vec![0; rank];
45        let mut runs = vec![TileRun::default(); rank];
46        let tile_strides = self.tile_strides.clone();
47
48        loop {
49            let mut tile = 0;
50            let mut output_base = 0;
51            for axis in 0..rank {
52                let position = start[axis] + samples[axis] * stride[axis];
53                let tile_len = self.tile_shape[axis];
54                let offset = position % tile_len;
55                tile += position / tile_len * self.tiles_per_dim_strides[axis];
56                output_base += samples[axis] * output_strides[axis];
57                runs[axis] = TileRun {
58                    offset,
59                    count: (1 + (tile_len - offset - 1) / stride[axis])
60                        .min(shape[axis] - samples[axis]),
61                };
62            }
63            let tile = tile_as_typed::<T>(self.get_cached_tile(tile)?);
64            gather_tile(
65                tile,
66                &mut values[output_base..],
67                &runs,
68                stride,
69                &tile_strides,
70                &output_strides,
71            );
72
73            // Advance along the fastest remaining axis with selected samples.
74            // The traversal makes no lookups for tiles between samples.
75            let mut more_tiles = false;
76            for axis in 0..rank {
77                samples[axis] += runs[axis].count;
78                if samples[axis] < shape[axis] {
79                    more_tiles = true;
80                    break;
81                }
82                samples[axis] = 0;
83            }
84            if !more_tiles {
85                break;
86            }
87        }
88        ArrayD::from_shape_vec(IxDyn(shape).f(), values)
89            .map_err(|error| StorageError::FormatMismatch(format!("array shape: {error}")))
90    }
91}
92
93fn gather_tile<T: TilePixel>(
94    tile: &[T],
95    output: &mut [T],
96    runs: &[TileRun],
97    strides: &[usize],
98    tile_strides: &[usize],
99    output_strides: &[usize],
100) {
101    let rows = runs[1..].iter().map(|run| run.count).product::<usize>();
102    let width = runs[0].count;
103    let inner_stride = strides[0];
104    for mut row in 0..rows {
105        let mut tile_offset = runs[0].offset;
106        let mut output_offset = 0;
107        for axis in 1..runs.len() {
108            let position = row % runs[axis].count;
109            row /= runs[axis].count;
110            tile_offset += (runs[axis].offset + position * strides[axis]) * tile_strides[axis];
111            output_offset += position * output_strides[axis];
112        }
113        let row = &tile[tile_offset..=tile_offset + (width - 1) * inner_stride];
114        let destination = &mut output[output_offset..output_offset + width];
115        if inner_stride == 1 {
116            destination.copy_from_slice(row);
117        } else {
118            for (value, &sample) in destination.iter_mut().zip(row.iter().step_by(inner_stride)) {
119                *value = sample;
120            }
121        }
122    }
123}
