58impl BaselineTable {
59    pub(super) fn open(path: &Path) -> Result<Self, String> {
60        let table = Table::open(TableOptions::new(path)).map_err(error_text)?;
61        validate_layout(&table)?;
62        let mut rows = BTreeMap::<_, Vec<IndexedRow>>::new();
63        for index in 0..table.row_count() {
64            let window = scalar_uint(&table, index, "IFNO")?;
65            let antenna = scalar_uint(&table, index, "ANTNO")?;
66            let beam = scalar_uint(&table, index, "BEAMNO")?;
67            let scan = scalar_uint(&table, index, "SCANNO")?;
68            let cell = table
69                .cell_accessor(index, "TIME")
70                .and_then(|cell| cell.scalar())
71                .map_err(error_text)?;
72            let ScalarValue::Float64(time) = cell else {
73                return Err("Expected a baseline TIME with type Double.".into());
74            };
75            if !time.is_finite() {
76                return Err("Nonfinite baseline TIME.".into());
77            }
78            rows.entry((window, antenna, beam))
79                .or_default()
80                .push(IndexedRow {
81                    time: *time,
82                    scan,
83                    index,
84                });
85        }
86        for group in rows.values_mut() {
87            group.sort_by(|left, right| left.time.total_cmp(&right.time));
88            group.dedup_by(|later, earlier| {
89                if later.time == earlier.time {
90                    *earlier = *later;
91                    true
92                } else {
93                    false
94                }
95            });
96        }
97        Ok(Self { table, rows })
98    }

100    pub(super) fn matching_row(
101        &self,
102        window: u32,
103        antenna: u32,
104        beam: u32,
105        scan: u32,
106        time: f64,
107        interval: f64,
108    ) -> Option<usize> {
109        let rows = self.rows.get(&(window, antenna, beam))?;
110        if time < rows.first()?.time || time > rows.last()?.time {
111            return None;
112        }
113        let matching = matching_time(rows, time, interval)?;
114        (matching.scan == scan).then_some(matching.index)
115    }

117    pub(super) fn model(
118        &self,
119        row: usize,
120        receptor: usize,
121    ) -> Result<Option<AppliedBaseline>, String> {
122        let cell = |column| {
123            self.table
124                .cell_accessor(row, column)
125                .and_then(|cell| cell.array())
126                .map_err(error_text)
127        };
128        let ArrayValue::Bool(apply) = cell("APPLY")? else {
129            return Err("Expected a Boolean baseline APPLY cell.".into());
130        };
131        if !baseline_scalar(apply, receptor).copied().unwrap_or(false) {
132            return Ok(None);
133        }
134        let ArrayValue::UInt32(families) = cell("FUNC_TYPE")? else {
135            return Err("Expected an unsigned FUNC_TYPE cell.".into());
136        };
137        let ArrayValue::Int32(parameters) = cell("FUNC_PARAM")? else {
138            return Err("Expected an integer FUNC_PARAM cell.".into());
139        };
140        let parameter = || {
141            baseline_scalar(parameters, receptor)
142                .and_then(|&value| usize::try_from(value).ok())
143                .ok_or("Invalid baseline order or piece count.")
144        };
145        let channels = scalar_uint(&self.table, row, "NCHAN")? as usize;
146        let basis = match baseline_scalar(families, receptor) {
147            Some(0) => ModelBasis::Polynomial(Polynomial::Monomial, parameter()?),
148            Some(1) => ModelBasis::Polynomial(Polynomial::Chebyshev, parameter()?),
149            Some(2) => {
150                let pieces = parameter()?;
151                let ArrayValue::Float32(params) = cell("FUNC_FPARAM")? else {
152                    return Err("Expected floating-point spline boundaries.".into());
153                };
154                let boundaries = spline_boundaries(params, receptor, pieces, channels)?;
155                ModelBasis::Spline { pieces, boundaries }
156            }
157            Some(3) => {
158                let waves = sinusoid_waves(parameters, receptor)?;
159                sinusoid::validate_context(*waves.last().expect("nonempty waves"), channels)?;
160                ModelBasis::Sinusoid(waves)
161            }
162            _ => return Err("Unsupported baseline type.".into()),
163        };
164        let coefficients_count = match &basis {
165            ModelBasis::Polynomial(_, order) => order + 1,
166            ModelBasis::Spline { pieces, .. } => pieces
167                .checked_mul(4)
168                .ok_or("Invalid number of cubic-spline coefficients.")?,
169            ModelBasis::Sinusoid(waves) => sinusoid::terms(waves),
170        };
171        let ArrayValue::Float32(result) = cell("RESULT")? else {
172            return Err("Expected a floating-point RESULT cell.".into());
173        };
174        if result.shape().get(1).copied().unwrap_or(0) < coefficients_count {
175            return Err("Baseline order or piece count exceeds the stored coefficients.".into());
176        }
177        let mut coefficients = Vec::with_capacity(coefficients_count);
178        for term in 0..coefficients_count {
179            coefficients.push(f64::from(
180                *result
181                    .get([receptor, term])
182                    .ok_or("Baseline order exceeds the stored coefficient count.")?,
183            ));
184        }
185        let ArrayValue::UInt32(mask) = cell("MASKLIST")? else {
186            return Err("Expected an unsigned MASKLIST cell.".into());
187        };
188        if mask.ndim() != 2 || receptor >= mask.shape()[0] {
189            return Err("Invalid polarization axis in baseline MASKLIST.".into());
190        }
191        let ranges = mask
192            .index_axis(Axis(0), receptor)
193            .iter()
194            .copied()
195            .collect::<Vec<_>>();
196        let included = decode_ranges(&ranges, channels)?;
197        Ok(Some(AppliedBaseline {
198            basis,
199            coefficients,
200            included,
201        }))
202    }
203}

205fn sinusoid_waves(values: &ndarray::ArrayD<i32>, receptor: usize) -> Result<Vec<usize>, String> {
206    if values.ndim() != 2 || receptor >= values.shape()[0] {
207        return Err("Invalid polarization axis in sinusoidal FUNC_PARAM.".into());
208    }
209    let mut waves = Vec::new();
210    for &value in values.index_axis(Axis(0), receptor) {
211        // A leading zero represents the DC term. Later zeros are table padding.
212        if value == 0 && !waves.is_empty() {
213            break;
214        }
215        waves.push(usize::try_from(value).map_err(|_| "Negative sinusoidal wavenumber.")?);
216    }
217    if waves.is_empty() || !waves.windows(2).all(|pair| pair[0] < pair[1]) {
218        return Err("Invalid sinusoidal wavenumbers in baseline table.".into());
219    }
220    Ok(waves)
221}

223fn spline_boundaries(
224    params: &ndarray::ArrayD<f32>,
225    receptor: usize,
226    pieces: usize,
227    channels: usize,
228) -> Result<Vec<usize>, String> {
229    if pieces == 0 || params.shape().get(1).copied().unwrap_or(0) <= pieces {
230        return Err("Missing cubic-spline boundaries in baseline table.".into());
231    }
232    (0..=pieces)
233        .map(|index| {
234            let value = *params
235                .get([receptor, index])
236                .ok_or("Invalid polarization axis in FUNC_FPARAM.")?;
237            if value.is_finite() && value >= 0. && value <= channels as f32 && value.fract() == 0. {
238                Ok(value as usize)
239            } else {
240                Err("Cubic-spline boundary is outside the selected spectrum.".into())
241            }
242        })
243        .collect()
244}

246fn matching_time(rows: &[IndexedRow], time: f64, interval: f64) -> Option<&IndexedRow> {
247    let (mut first, mut last) = (0, rows.len() - 1);
248    while last - first > 1 {
249        let middle = first + (last - first) / 2;
250        if let Some(exact) = [first, middle, last]
251            .into_iter()
252            .map(|index| &rows[index])
253            .find(|row| row.time == time)
254        {
255            return Some(exact);
256        }
257        if rows[middle].time < time {
258            first = middle;
259        } else {
260            last = middle;
261        }
262    }
263    // In its final bracket CASA checks both candidates in time order.
264    let half = interval / 2.;
265    rows[first..=last]
266        .iter()
267        .find(|row| time - half < row.time && row.time < time + half)
268}

270fn validate_layout(table: &Table) -> Result<(), String> {
271    let actual = table.schema().ok_or("Missing baseline table schema.")?;
272    for expected in schema().columns() {
273        let name = expected.name();
274        let column = actual
275            .column(name)
276            .ok_or_else(|| format!("Table column {name} is unknown"))?;
277        if column.data_type() != expected.data_type()
278            || std::mem::discriminant(column.column_type())
279                != std::mem::discriminant(expected.column_type())
280        {
281            return Err(format!("Unexpected data type for baseline column {name}."));
282        }
283    }
284    Ok(())
285}

287fn baseline_scalar<T>(values: &ndarray::ArrayD<T>, receptor: usize) -> Option<&T> {
288    match values.ndim() {
289        1 => values.get([receptor]),
290        2 => values.get([receptor, 0]),
291        _ => None,
292    }

295fn decode_ranges(ranges: &[u32], channels: usize) -> Result<Vec<bool>, String> {
296    let (pairs, tail) = ranges.as_chunks::<2>();
297    if !tail.is_empty() {
298        return Err("masklist must have even number of elements.".into());
299    }
300    let mut included = vec![false; channels];
301    let Some(last) = channels.checked_sub(1) else {
302        return Ok(included);
303    };
304    for &[start, end] in pairs {
305        let end = end as usize;
306        if let Some(section) = included.get_mut(start as usize..=end.min(last)) {
307            section.fill(true);
308        }
309        // Casacore pads narrower polarization masks to make a rectangular
310        // cell. The first pair ending at the last channel terminates the mask.
311        if end == last {
312            break;
313        }
314    }
315    Ok(included)
316}

318fn scalar_uint(table: &Table, row: usize, column: &str) -> Result<u32, String> {
319    match table
320        .cell_accessor(row, column)
321        .and_then(|cell| cell.scalar())
322        .map_err(error_text)?
323    {
324        ScalarValue::UInt32(value) => Ok(*value),
325        _ => Err(format!(
326            "Expected unsigned integer baseline column {column}."
327        )),
328    }

331impl BaselineRow {
332    pub(super) fn record(&self, fixed_family: Option<BaselineFamily>) -> RecordValue {
333        let receptors = self.models.len();
334        let widest = |size: fn(&SpectrumModel) -> usize| {
335            self.models
336                .iter()
337                .flatten()
338                .map(size)
339                .max()
340                .unwrap_or_default()
341        };
342        let coefficients = widest(|model| model.coefficients.len());
343        let parameters = widest(|model| match &model.basis {
344            ModelBasis::Sinusoid(waves) => waves.len(),
345            _ => 1,
346        });
347        let boundaries = widest(|model| match &model.basis {
348            ModelBasis::Spline { boundaries, .. } => boundaries.len(),
349            _ => 0,
350        });
351        let ranges = widest(|model| model.mask.len());
352        let mut apply = Array2::from_elem((receptors, 1), false);
353        let storage_family = |family| match family {
354            BaselineFamily::Polynomial(Polynomial::Monomial) => 0,
355            BaselineFamily::Polynomial(Polynomial::Chebyshev) => 1,
356            BaselineFamily::CubicSpline => 2,
357            BaselineFamily::Sinusoid => 3,
358        };
359        let mut family = Array2::from_elem((receptors, 1), fixed_family.map_or(0, storage_family));
360        let mut parameter = Array2::<i32>::zeros((receptors, parameters));
361        let mut boundaries = Array2::<f32>::zeros((receptors, boundaries));
362        let mut result = Array2::<f32>::zeros((receptors, coefficients));
363        let mut mask = Array2::<u32>::zeros((receptors, ranges));
364        let mut rms = Array2::<f32>::zeros((receptors, 1));
365        let mut clip = Array2::<f32>::zeros((receptors, 1));
366        let mut iterations = Array2::<u32>::zeros((receptors, 1));
367        let mut use_line_finder = Array2::from_elem((receptors, 1), false);
368        let mut line_threshold = Array2::<f32>::zeros((receptors, 1));
369        let mut line_average = Array2::<u32>::zeros((receptors, 1));
370        let mut line_edges = Array2::<u32>::zeros((receptors, 2));
371        for (receptor, model) in self.models.iter().enumerate() {
372            let Some(model) = model else { continue };
373            apply[[receptor, 0]] = true;
374            family[[receptor, 0]] = storage_family(model.basis.family());
375            match &model.basis {
376                ModelBasis::Polynomial(_, order) => parameter[[receptor, 0]] = *order as i32,
377                ModelBasis::Spline {
378                    pieces,
379                    boundaries: knots,
380                } => {
381                    parameter[[receptor, 0]] = *pieces as i32;
382                    for (index, &value) in knots.iter().enumerate() {
383                        boundaries[[receptor, index]] = value as f32;
384                    }
385                }
386                ModelBasis::Sinusoid(waves) => {
387                    for (index, &value) in waves.iter().enumerate() {
388                        parameter[[receptor, index]] = value as i32;
389                    }
390                }
391            }
392            for (index, &value) in model.coefficients.iter().enumerate() {
393                result[[receptor, index]] = value as f32;
394            }
395            for (index, slot) in mask.row_mut(receptor).iter_mut().enumerate() {
396                // CASA's variable fits zero-pad the cell. The fixed-family
397                // writer repeats the last range to preserve the model mask.
398                let previous = model.mask.len().saturating_sub(2) + index % 2;
399                *slot = model.mask.get(index).copied().unwrap_or_else(|| {
400                    fixed_family
401                        .and_then(|_| model.mask.get(previous).copied())
402                        .unwrap_or_default()
403                });
404            }
405            rms[[receptor, 0]] = model.rms;
406            clip[[receptor, 0]] = model.clip_threshold;
407            iterations[[receptor, 0]] = model.clip_iterations;
408            use_line_finder[[receptor, 0]] = model.line_finder.enabled;
409            line_threshold[[receptor, 0]] = model.line_finder.threshold;
410            line_average[[receptor, 0]] = model.line_finder.average;
411            [line_edges[[receptor, 0]], line_edges[[receptor, 1]]] = model.line_finder.edges;
412        }
413        let uint = |name, value| RecordField::new(name, Value::Scalar(ScalarValue::UInt32(value)));
414        let array = |name, value| RecordField::new(name, Value::Array(value));
415        RecordValue::new(vec![
416            uint("SCANNO", self.scan),
417            uint("BEAMNO", self.beam),
418            uint("ANTNO", self.antenna),
419            uint("IFNO", self.window),
420            uint("FREQ_ID", 0),
421            RecordField::new("TIME", Value::Scalar(ScalarValue::Float64(self.time))),
422            array("APPLY", ArrayValue::Bool(apply.into_dyn())),
423            array("FUNC_TYPE", ArrayValue::UInt32(family.into_dyn())),
424            array("FUNC_PARAM", ArrayValue::Int32(parameter.into_dyn())),
425            array("FUNC_FPARAM", ArrayValue::Float32(boundaries.into_dyn())),
426            array("MASKLIST", ArrayValue::UInt32(mask.into_dyn())),
427            array("RESULT", ArrayValue::Float32(result.into_dyn())),
428            array("RMS", ArrayValue::Float32(rms.into_dyn())),
429            uint("NCHAN", self.channels as u32),
430            array("CLIP_THRESHOLD", ArrayValue::Float32(clip.into_dyn())),
431            array("CLIP_ITERATION", ArrayValue::UInt32(iterations.into_dyn())),
432            array("USE_LF", ArrayValue::Bool(use_line_finder.into_dyn())),
433            array(
434                "LF_THRESHOLD",
435                ArrayValue::Float32(line_threshold.into_dyn()),
436            ),
437            array("LF_AVERAGE", ArrayValue::UInt32(line_average.into_dyn())),
438            array("LF_EDGE", ArrayValue::UInt32(line_edges.into_dyn())),
439        ])
440    }
441}

443pub(super) fn mask_ranges(mask: &[bool]) -> Vec<u32> {
444    let mut result = Vec::new();
445    let mut first = None;
446    for (channel, included) in mask.iter().copied().chain([false]).enumerate() {
447        if included {
448            first.get_or_insert(channel);
449        } else if let Some(start) = first.take() {
450            result.extend([start as u32, (channel - 1) as u32]);
451        }
452    }
453    result
454}

456pub(super) fn write(
457    output: &Path,
458    input: &MeasurementSet,
459    rows: impl IntoIterator<Item = RecordValue>,
460) -> Result<(), String> {
461    let mut table = Table::with_schema(schema());
462    table.set_info(TableInfo {
463        table_type: "ApplyTable".into(),
464        sub_type: String::new(),
465        readme: Vec::new(),
466    });
467    let source = input
468        .path()
469        .map(|path| path.display().to_string())
470        .unwrap_or_default();
471    for (key, value) in [
472        (
473            "ApplyType",
474            Value::Scalar(ScalarValue::String("BASELINE".into())),
475        ),
476        ("VERSION", Value::Scalar(ScalarValue::Int32(1))),
477        ("MSName", Value::Scalar(ScalarValue::String(source))),
478        (
479            "FREQUENCIES",
480            Value::table_ref(strip_directory(&output.join("SPECTRAL_WINDOW"), output)),
481        ),
482    ] {
483        table.keywords_mut().upsert(key, value);
484    }
485    TableMeasDesc::new_fixed("TIME", MeasureType::Epoch, "UTC")
486        .write(&mut table)
487        .map_err(error_text)?;
488    TableQuantumDesc::with_unit("TIME", "d")
489        .write(&mut table)
490        .map_err(error_text)?;
491    table.set_column_keywords(
492        "FUNC_TYPE",
493        RecordValue::new(
494            ["Polynomial", "Chebyshev", "Cubic Spline", "Sinusoid"]
495                .into_iter()
496                .enumerate()
497                .map(|(id, label)| {
498                    RecordField::new(label, Value::Scalar(ScalarValue::Int32(id as i32)))
499                })
500                .collect(),
501        ),
502    );
503    for row in rows {
504        table.add_row(row).map_err(error_text)?;
505    }
506    table.save(TableOptions::new(output)).map_err(error_text)?;
507    input
508        .subtable(SubtableId::SpectralWindow)
509        .ok_or("No SPECTRAL_WINDOW subtable for the baseline table.")?
510        .deep_copy(TableOptions::new(output.join("SPECTRAL_WINDOW")))
511        .map_err(error_text)?;
512    Ok(())
513}

515fn schema() -> TableSchema {
516    use PrimitiveType::{Bool, Float32, Float64, Int32, UInt32};
517    let array = |name, primitive| ColumnSchema::array_variable(name, primitive, Some(2));
518    TableSchema::new(vec![
519        ColumnSchema::scalar("SCANNO", UInt32),
520        ColumnSchema::scalar("BEAMNO", UInt32),
521        ColumnSchema::scalar("ANTNO", UInt32),
522        ColumnSchema::scalar("IFNO", UInt32),
523        ColumnSchema::scalar("FREQ_ID", UInt32),
524        ColumnSchema::scalar("TIME", Float64),
525        array("APPLY", Bool),
526        array("FUNC_TYPE", UInt32),
527        array("FUNC_PARAM", Int32),
528        array("FUNC_FPARAM", Float32),
529        array("MASKLIST", UInt32),
530        array("RESULT", Float32),
531        array("RMS", Float32),
532        ColumnSchema::scalar("NCHAN", UInt32),
533        array("CLIP_THRESHOLD", Float32),
534        array("CLIP_ITERATION", UInt32),
535        array("USE_LF", Bool),
536        array("LF_THRESHOLD", Float32),
537        array("LF_AVERAGE", UInt32),
538        array("LF_EDGE", UInt32),
539    ])
540    .expect("valid baseline table schema")
541}
