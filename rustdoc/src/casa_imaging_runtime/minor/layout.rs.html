2//! What a minor cycle holds ([`prepare_bytes`], [`run_bytes`],
3//! [`mask_bytes`]), from the layout of the image state it cleans
4//! ([`MinorLayout`]): a completion's normal state, or the layout a run's
5//! first pass will form, from which the run chooses its workers before any
6//! image exists (#737).

16/// The layout of the image state a minor cycle cleans, as its charges read
17/// it: how many planes it loads and the largest, its image domains'
18/// shapes, its residual and PSF term counts, and whether its planes are
19/// read from a paged channel-local state.
20#[derive(Clone, Debug, PartialEq, Eq)]
21pub struct MinorLayout {
22    planes: usize,
23    shape: PlaneShape,
24    domains: Vec<[usize; 2]>,
25    terms: (u64, u64),
26    paged: bool,
27}
28
29impl MinorLayout {
30    /// The layout of `normal`.
31    #[must_use]
32    pub fn of(normal: &FinalNormalState) -> Self {
33        let taylor = is_taylor(normal);
34        let keys = plane_keys(normal, taylor);
35        Self {
36            planes: keys.len(),
37            shape: keys
38                .iter()
39                .map(|key| plane_shape(normal, *key))
40                .max_by_key(|shape| shape.len())
41                .unwrap_or(PlaneShape::new(0, 0)),
42            domains: (0..normal.domain_count())
43                .filter_map(|domain| normal.domain_shape(domain))
44                .collect(),
45            terms: if taylor {
46                (
47                    normal.coefficient_term_count() as u64,
48                    normal.normal_moment_count() as u64,
49                )
50            } else {
51                (1, 1)
52            },
53            paged: normal.basis() == NormalBasis::ChannelLocal,
54        }
55    }
56
57    /// The layout the first major cycle of `problem` forms: one Taylor
58    /// family of `T` residual terms and `2T − 1` normal moments on the first
59    /// image domain, or a plane per polarization, image domain and output
60    /// channel, paged when there is a plane per channel.
61    #[must_use]
62    pub fn planned(problem: &CompiledProblem) -> Self {
63        let target = problem.model_lifecycle().target();
64        let domains = target
65            .domains()
66            .iter()
67            .map(|domain| domain.pixels())
68            .collect::<Vec<_>>();
69        let largest = |domains: &[[usize; 2]]| {
70            domains
71                .iter()
72                .map(|&[nx, ny]| PlaneShape::new(nx, ny))
73                .max_by_key(|shape| shape.len())
74                .unwrap_or(PlaneShape::new(0, 0))
75        };
76        let per_channel = target.polarizations() * domains.len();
77        let (planes, shape, terms, paged) = match problem.reconstruction().basis() {
78            ReconstructionBasis::Taylor { terms } => (
79                1,
80                largest(&domains[..domains.len().min(1)]),
81                (terms as u64, (2 * terms).saturating_sub(1) as u64),
82                false,
83            ),
84            ReconstructionBasis::ChannelLocal { channels } => {
85                (per_channel * channels, largest(&domains), (1, 1), true)
86            }
87            ReconstructionBasis::Constant => (per_channel, largest(&domains), (1, 1), false),
88        };
89        Self {
90            planes,
91            shape,
92            domains,
93            terms,
94            paged,
95        }
96    }

98    /// Heap bytes the masks a minor cycle forms keep while the cycle runs
99    /// and after it: one support per image domain.
100    #[must_use]
101    pub fn mask_bytes(&self) -> u64 {
102        self.domains
103            .iter()
104            .map(|[width, height]| (width * height * size_of::<bool>()) as u64)
105            .sum()
106    }
107
108    /// Heap bytes the record of one minor cycle keeps for the rest of the
109    /// run: a copy of the primary mask's support, the first image domain's,
110    /// and at most `TRACE` traced components (the run's `record_bytes`).
111    #[must_use]
112    pub fn record_bytes(&self) -> u64 {
113        let support = self
114            .domains
115            .first()
116            .map_or(0, |[width, height]| width * height * size_of::<bool>());
117        (support + TRACE * size_of::<super::TracedComponent>()) as u64
118    }
119
120    /// Heap bytes the model terms of a minor cycle under `setup` hold, from
121    /// the solve until its refresh pass takes them: [`super::result_bytes`]
122    /// of every plane's update, cleaning at most `iterations`, at the most
123    /// entries its solver's components can make
124    /// ([`casa_imaging_deconvolution::update_entries`]). A multiscale
125    /// component makes an entry for each cell of its scale, not one (#778
126    /// review).
127    #[must_use]
128    pub fn result_bytes(&self, setup: &MinorCycleSetup, iterations: usize) -> u64 {
129        let (_, entries) = solve_bound(self, setup, iterations);
130        self.planes as u64 * super::result_bytes(entries)
131    }
132
133    /// Heap bytes the Clark refresh a run caches across its minor cycles
134    /// holds ([`PsfCache::refresh_bytes`]): a single-plane Clark run keeps
135    /// one; any other keeps none.
136    #[must_use]
137    pub fn refresh_bytes(&self, setup: &MinorCycleSetup) -> u64 {
138        if matches!(setup.algorithm, ReconstructionAlgorithm::Clark) && self.planes == 1 {
139            casa_imaging_deconvolution::LinearRefresh::bytes(self.shape)
140        } else {
141            0
142        }
143    }
144
145    /// Heap bytes one loaded plane holds at most ([`super::load`]): its
146    /// residual and PSF terms (`f64`), its support and validity, its
147    /// sensitivity under a response and, from a paged channel-local state,
148    /// the plane's fields read as `Complex64`; or, before any of them, its
149    /// validity beside the one model plane read at a time to find the valid
150    /// cells. A constant or Taylor state is resident and its planes are
151    /// borrowed, not read (#738).
152    fn load_bytes(&self, response: bool) -> u64 {
153        let terms = self.terms.0 + self.terms.1;
154        let loaded = terms * size_of::<f64>() as u64
155            + 2 * size_of::<bool>() as u64
156            + if response { size_of::<f64>() as u64 } else { 0 }
157            + if self.paged {
158                terms * size_of::<num_complex::Complex64>() as u64
159            } else {
160                0
161            };
162        let validity = (size_of::<casa_imaging_model::ModelSample>() + size_of::<bool>()) as u64;
163        self.shape.len() as u64 * loaded.max(validity)
164    }
165
166    /// Planes a phase of `workers` loads at once: one when a single plane
167    /// spreads over the team.
168    fn concurrent(&self, workers: usize) -> u64 {
169        workers.min(self.planes).max(1) as u64
170    }
171}

173/// Heap bytes [`super::prepare_minor_cycle`] holds at most: the masks it
174/// materializes, and on each of the planes loaded at once on `workers`, the
175/// load and its measurement (the robust noise's two copies under `nsigma`,
176/// the PSF fit's single-precision copies for a plane not yet in `cache`).
177#[must_use]
178pub fn prepare_bytes(
179    completion: &MajorCycleCompletion,
180    mask_plans: &ImageDomainReconstructionMaskPlans,
181    setup: &MinorCycleSetup,
182    cache: &PsfCache,
183    workers: usize,
184) -> u64 {
185    let normal = completion.normal_state();
186    let fits = plane_keys(normal, is_taylor(normal))
187        .iter()
188        .any(|key| !cache.summaries.contains_key(key));
189    prepare_bytes_for(&MinorLayout::of(normal), mask_plans, setup, fits, workers)
190}
191
192/// [`prepare_bytes`] for a state of `layout`, with `fits` when some plane's
193/// PSF is not yet measured, as before the first minor cycle.
194#[must_use]
195pub fn prepare_bytes_for(
196    layout: &MinorLayout,
197    mask_plans: &ImageDomainReconstructionMaskPlans,
198    setup: &MinorCycleSetup,
199    fits: bool,
200    workers: usize,
201) -> u64 {
202    let measure = layout.shape.len() as u64
203        * if setup.nsigma > 0.0 {
204            2 * size_of::<f64>() as u64
205        } else {
206            0
207        }
208        .max(if fits { 2 * size_of::<f32>() as u64 } else { 0 });
209    mask_plans.materialize_bytes_for(|domain| layout.domains.get(domain).copied())
210        + layout.concurrent(workers) * (layout.load_bytes(setup.response.is_some()) + measure)
211}

213/// Heap bytes the masks a minor cycle forms keep while the cycle runs and
214/// after it: one support per image domain.
215#[must_use]
216pub fn mask_bytes(completion: &MajorCycleCompletion) -> u64 {
217    MinorLayout::of(completion.normal_state()).mask_bytes()
218}

220/// Heap bytes [`super::run_minor_cycle`] holds at most under `controls`,
221/// besides each plane's terms ([`super::result_bytes`]): on each of the
222/// planes solved at once on `workers` (one when a single plane spreads over
223/// the team), the load and the solve
224/// ([`casa_imaging_deconvolution::solve_bytes`]), and a record of every
225/// plane. A Clark refresh already in `cache` is charged with the cache, not
226/// here.
227#[must_use]
228pub fn run_bytes(
229    completion: &MajorCycleCompletion,
230    setup: &MinorCycleSetup,
231    controls: &CycleControls,
232    cache: &PsfCache,
233    workers: usize,
234) -> u64 {
235    run_bytes_for(
236        &MinorLayout::of(completion.normal_state()),
237        setup,
238        controls.iterations,
239        cache.refresh_bytes(completion),
240        workers,
241    )
242}
243
244/// [`run_bytes`] for a state of `layout`, cleaning at most `iterations` per
245/// plane beside a cached Clark refresh of `refresh` bytes.
246#[must_use]
247pub fn run_bytes_for(
248    layout: &MinorLayout,
249    setup: &MinorCycleSetup,
250    iterations: usize,
251    refresh: u64,
252    workers: usize,
253) -> u64 {
254    let (solve, _) = solve_bound(layout, setup, iterations);
255    let solve = if matches!(setup.algorithm, ReconstructionAlgorithm::Clark) && layout.planes == 1 {
256        solve.saturating_sub(refresh)
257    } else {
258        solve
259    };
260    // Every plane's slot, solved record and stop, which the cycle keeps
261    // until it merges them.
262    let records =
263        layout.planes * (size_of::<Slot>() + size_of::<Solved>() + size_of::<PlaneStop>());
264    layout.concurrent(workers) * (layout.load_bytes(setup.response.is_some()) + solve)
265        + records as u64
266}

268/// What `setup`'s solver holds for one plane of `layout`, cleaning at most
269/// `iterations` and so placing at most one more component
270/// ([`casa_imaging_deconvolution::solve_bytes`]), and the entries of the
271/// plane's model update ([`casa_imaging_deconvolution::update_entries`]);
272/// nothing for an algorithm with no solver here.
273fn solve_bound(layout: &MinorLayout, setup: &MinorCycleSetup, iterations: usize) -> (u64, usize) {
274    fn bound<S: Solver>(solver: &S, layout: &MinorLayout, components: usize) -> (u64, usize) {
275        let (shape, terms) = (layout.shape, layout.terms.0 as usize);
276        (
277            casa_imaging_deconvolution::solve_bytes(solver, shape, terms, components, TRACE),
278            casa_imaging_deconvolution::update_entries(solver, shape, terms, components),
279        )
280    }
281    let components = iterations + 1;
282    match &setup.algorithm {
283        ReconstructionAlgorithm::Hogbom => bound(&Hogbom::new(false), layout, components),
284        ReconstructionAlgorithm::Clark => bound(&Clark::new(None), layout, components),
285        ReconstructionAlgorithm::Multiscale {
286            scales_px,
287            small_scale_bias,
288        } => bound(
289            &Multiscale::new(scales_px.clone(), *small_scale_bias),
290            layout,
291            components,
292        ),
293        ReconstructionAlgorithm::Mtmfs {
294            scales_px,
295            small_scale_bias,
296        } => bound(
297            &Taylor::new(
298                layout.terms.0 as usize,
299                scales_px.clone(),
300                *small_scale_bias,
301            ),
302            layout,
303            components,
304        ),
305        _ => (0, 0),
306    }
