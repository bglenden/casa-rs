28/// Build a FITS header for an attached image without reading its pixels.
29///
30/// The shape is supplied separately so that virtual and native attachments
31/// use the same header path.
32///
33/// # Errors
34/// Returns an error for invalid WCS, metadata or an unreadable clock.
35pub fn image_fits_header(
36    image: &dyn ImageInterface<f32>,
37    shape: &[usize],
38    name: &str,
39    history: &[ImageHistoryEntry],
40) -> Result<ImageFitsHeader, ImageError> {
41    let coords = image.coordinates();
42    let plan = ExportAxisPlan::build(coords, shape, false, false, false)?;
43    let standard_wcs = || {
44        let base = try_to_fits_header(coords, shape)?;
45        let mut wcs = build_export_keywords(coords, &base, &plan, true, true)?
46            .into_iter()
47            .collect();
48        normalize_wcs(coords, &mut wcs)?;
49        Ok::<_, ImageError>(wcs)
50    };
51    let (mut wcs, projection) = match standard_wcs() {
52        Ok(wcs) => (wcs, Some(projection_parameters(coords))),
53        Err(_) => (linear_fallback(coords, shape.len())?, None),
54    };
55    let mut header = ImageFitsHeader(Vec::new());
56    header.add("SIMPLE", FitsValue::Logical(true), Some("Standard FITS"));
57    header.add(
58        "BITPIX",
59        FitsValue::Integer(-32),
60        Some("Floating point (32 bit)"),
61    );
62    header.integer("NAXIS", shape.len() as i64);
63    for (index, length) in shape.iter().enumerate() {
64        header.integer(format!("NAXIS{}", index + 1), *length as i64);
65    }
66    header.add("EXTEND", FitsValue::Logical(true), None);
67    header.add(
68        "BSCALE",
69        FitsValue::Float(1.0),
70        Some("PHYSICAL = PIXEL*BSCALE + BZERO"),
71    );
72    header.add("BZERO", FitsValue::Float(0.0), None);
73    let info = image.image_info()?;
74    header.image_metadata(image.units(), &info)?;
75    let standard = projection.is_some();
76    header.axes(shape.len(), &mut wcs, projection);
77    for (name, comment) in [
78        ("RESTFRQ", "Rest Frequency (Hz)"),
79        ("SPECSYS", "Spectral reference frame"),
80        ("ALTRVAL", "Alternate frequency reference value"),
81        ("ALTRPIX", "Alternate frequency reference pixel"),
82        ("VELREF", "1 LSR, 2 HEL, 3 OBS, +256 Radio"),
83    ] {
84        if let Some(value) = wcs.remove(name) {
85            header.add(name, value, Some(comment));
86            if name == "VELREF" {
87                header.text(
88                    "COMMENT",
89                    "casacore non-standard usage: 4 LSD, 5 GEO, 6 SOU, 7 GAL",
90                );
91            }
92        }
93    }
94    if standard {
95        header.observation(image, &mut wcs)?;
96    } else {
97        header.text("TIMESYS", "UTC");
98    }
99    header.extend_misc(image.misc_info())?;
100    header.add(
101        "DATE",
102        FitsValue::String(format_time(
103            ImageHistoryEntry::current_time()? / 86_400.0,
104            FITS,
105            12,
106        )),
107        Some("Date FITS file was written"),
108    );
109    header.text("ORIGIN", "casacore-3.8.0");
110    if info.beam_set.is_multi() {
111        header.add(
112            "CASAMBM",
113            FitsValue::Logical(true),
114            Some("CASA multiple BEAMS table present"),
115        );
116    }
117    if !history.is_empty() {
118        visit_history_cards(history, false, |text| {
119            header.text("HISTORY", text);
120            Ok(())
121        })?;
122    }
123    header.text("END", "");
124    let start = name.len().saturating_sub(68);
125    let suffix = name.get(start..).unwrap_or_else(|| {
126        let boundary = name.char_indices().find(|(i, _)| *i >= start);
127        &name[boundary.map_or(name.len(), |(i, _)| i)..]
128    });
129    header.text("IMAGENME", suffix);
130    Ok(header)
131}

133impl ImageFitsHeader {
134    fn add(&mut self, name: impl Into<String>, value: FitsValue, comment: Option<&str>) {
135        let name = name.into();
136        let group = group_name(&name);
137        self.0.push(HeaderCard {
138            keyword: FitsKeyword {
139                name,
140                value,
141                comment: comment.map(str::to_owned),
142            },
143            group,
144        });
145    }
146
147    fn text(&mut self, name: impl Into<String>, text: impl Into<String>) {
148        self.add(name, FitsValue::String(text.into()), None);
149    }
150
151    fn integer(&mut self, name: impl Into<String>, number: i64) {
152        self.add(name, FitsValue::Integer(number), None);
153    }
154
155    fn image_metadata(&mut self, units: &str, info: &ImageInfo) -> Result<(), ImageError> {
156        if let Some(beam) = info.beam_set.single_beam().filter(|beam| !beam.is_null()) {
157            self.add("BMAJ", FitsValue::Float(beam.major_in("deg")?), None);
158            self.add("BMIN", FitsValue::Float(beam.minor_in("deg")?), None);
159            let mut angle = beam.position_angle_in("deg")? % 180.0;
160            if angle > 90.0 {
161                angle -= 180.0;
162            } else if angle <= -90.0 {
163                angle += 180.0;
164            }
165            self.add("BPA", FitsValue::Float(angle), None);
166        }
167        let image_type = if info.image_type == ImageType::Undefined {
168            ImageType::Intensity
169        } else {
170            info.image_type
171        };
172        self.text("BTYPE", image_type.to_string());
173        self.text("OBJECT", &info.object_name);
174        self.add(
175            "BUNIT",
176            FitsValue::String(units.to_owned()),
177            Some("Brightness (pixel) unit"),
178        );
179        Ok(())
180    }
181
182    fn axes(
183        &mut self,
184        rank: usize,
185        wcs: &mut BTreeMap<String, FitsValue>,
186        projection: Option<Vec<(String, FitsValue)>>,
187    ) {
188        for keyword in ["EQUINOX", "RADESYS", "LONPOLE", "LATPOLE"] {
189            if let Some(value) = wcs.remove(keyword) {
190                self.add(keyword, value, None);
191            }
192        }
193        if projection.is_some() {
194            for col in 1..=rank {
195                for row in 1..=rank {
196                    let name = format!("PC{row}_{col}");
197                    let value = wcs
198                        .remove(&name)
199                        .unwrap_or(FitsValue::Float(f64::from(row == col)));
200                    self.add(name, value, None);
201                }
202            }
203        }
204        let projection = projection.unwrap_or_default();
205        let interleave_projection = projection.len() == rank;
206        let mut projection = projection.into_iter();
207        for axis in 1..=rank {
208            for key in ["CTYPE", "CRVAL", "CDELT", "CRPIX", "CUNIT"] {
209                let name = format!("{key}{axis}");
210                if let Some(value) = wcs.remove(&name) {
211                    self.add(name, value, None);
212                } else if key == "CUNIT" {
213                    self.text(name, "");
214                }
215            }
216            if interleave_projection && let Some((name, value)) = projection.next() {
217                self.add(name, value, None);
218            }
219        }
220        for (name, value) in projection {
221            self.add(name, value, None);
222        }
223    }
224
225    fn observation(
226        &mut self,
227        image: &dyn ImageInterface<f32>,
228        wcs: &mut BTreeMap<String, FitsValue>,
229    ) -> Result<(), ImageError> {
230        for keyword in ["TELESCOP", "OBSERVER", "DATE-OBS", "TIMESYS"] {
231            if let Some(value) = wcs.remove(keyword) {
232                self.add(keyword, value, None);
233            }
234        }
235        let obs = image.coordinates().obs_info();
236        if let Some(position) = &obs.telescope_position {
237            let xyz = position
238                .convert_to(PositionRef::ITRF)
239                .map_err(|error| ImageError::InvalidMetadata(error.to_string()))?
240                .values();
241            for (key, value) in ["OBSGEO-X", "OBSGEO-Y", "OBSGEO-Z"].into_iter().zip(xyz) {
242                self.add(key, FitsValue::Float(value), None);
243            }
244        }
245        if !obs.pointing_center_initial {
246            for (key, angle) in ["OBSRA", "OBSDEC"].into_iter().zip(obs.pointing_center_rad) {
247                self.add(key, FitsValue::Float(angle.to_degrees()), None);
248            }
249        }
250        Ok(())
251    }
252
253    fn extend_misc(&mut self, info: RecordValue) -> Result<(), ImageError> {
254        let mut fields = info.fields().iter().collect::<Vec<_>>();
255        fields.sort_by_key(|field| field.name.to_ascii_uppercase());
256        let mut cards_by_field = Vec::new();
257        for field in fields {
258            let name = field.name.to_ascii_uppercase();
259            if name.is_empty() || !name.is_ascii() {
260                continue;
261            }
262            let values = misc_values(&name, &field.value)?;
263            if values.is_empty() {
264                continue;
265            }
266            let shape = match &field.value {
267                Value::Array(array) => array.shape().to_vec(),
268                _ => Vec::new(),
269            };
270            cards_by_field.push((name, shape, values));
271        }
272        // casacore interleaves arrays of the same rank and element count,
273        // naming matrix cards using the last array's row count in the run.
274        for run in cards_by_field.chunk_by(|(_, first, values), (_, next, other)| {
275            first.len() == next.len() && values.len() == other.len()
276        }) {
277            let shape = &run.last().expect("nonempty run").1;
278            let mut fields = Vec::with_capacity(run.len());
279            for (name, _, values) in run {
280                let names = misc_keyword_names(name, shape, values.len())?;
281                fields.push(names.into_iter().zip(values));
282            }
283            for _ in 0..run[0].2.len() {
284                for cards in &mut fields {
285                    let (mut key, value) = cards.next().expect("one card per value");
286                    if key == "COMMENT"
287                        && matches!(value, FitsValue::String(text) if text.is_empty())
288                    {
289                        key.clear();
290                    }
291                    self.0.push(HeaderCard {
292                        keyword: FitsKeyword {
293                            name: key,
294                            value: value.clone(),
295                            comment: None,
296                        },
297                        group: None,
298                    });
299                }
300            }
301        }
302        Ok(())
303    }

305    /// The FITS header as consecutive 80-byte cards, with no block padding.
306    pub fn card_string(&self) -> String {
307        self.0
308            .iter()
309            .map(|card| render_card(&card.keyword))
310            .collect()
311    }

313    /// The FITS header as a CASA record with the given entry names omitted.
314    pub fn record(&self, exclude: &[String]) -> RecordValue {
315        let mut grouped = BTreeMap::<String, Vec<FitsValue>>::new();
316        for card in &self.0 {
317            let keyword = &card.keyword;
318            let name = card.group.unwrap_or(&keyword.name);
319            let name = if name.is_empty() { "        " } else { name };
320            grouped
321                .entry(name.to_owned())
322                .or_default()
323                .push(keyword.value.clone());
324        }
325        let mut record = RecordValue::default();
326        for (name, values) in grouped {
327            if !exclude.contains(&name) {
328                let axis_vector = matches!(
329                    name.as_str(),
330                    "NAXIS" | "CTYPE" | "CRVAL" | "CRPIX" | "CDELT"
331                );
332                let value = if !axis_vector && values.len() == 1 {
333                    scalar(&name, &values[0]).into()
334                } else {
335                    Value::Array(array(&name, &values))
336                };
337                record.upsert(name, value);
338            }
339        }
340        record
341    }
342}

344fn linear_fallback(
345    coords: &CoordinateSystem,
346    rank: usize,
347) -> Result<BTreeMap<String, FitsValue>, ImageError> {
348    if coords.n_world_axes() != rank {
349        return Err(header_creation_error());
350    }
351    let mut axis = 1;
352    let mut wcs = BTreeMap::new();
353    for index in 0..coords.n_coordinates() {
354        let coordinate = coords.coordinate(index);
355        let (reference, increment, units) = match coordinate {
356            CoordinateModel::Direction(direction) => {
357                let (reference, increment) = direction.reference_and_increment_in_world_units();
358                (reference, increment, direction.visible_world_units())
359            }
360            _ => (
361                coordinate.reference_value(),
362                coordinate.increment(),
363                coordinate.axis_units(),
364            ),
365        };
366        let names = coordinate.axis_names();
367        let pixels = coordinate.reference_pixel();
368        for local in 0..coordinate.n_pixel_axes() {
369            let mut name = names[local].to_ascii_uppercase();
370            let mut unit = units[local].clone();
371            truncate_bytes(&mut name, 8);
372            truncate_bytes(&mut unit, 8);
373            for (key, value) in [
374                ("CTYPE", FitsValue::String(name)),
375                ("CRVAL", FitsValue::Float(reference[local])),
376                ("CDELT", FitsValue::Float(increment[local])),
377                ("CRPIX", FitsValue::Float(pixels[local] + 1.0)),
378                ("CUNIT", FitsValue::String(unit)),
379            ] {
380                wcs.insert(format!("{key}{axis}"), value);
381            }
382            axis += 1;
383        }
384    }
385    Ok(wcs)
386}

388fn normalize_wcs(
389    coords: &CoordinateSystem,
390    wcs: &mut BTreeMap<String, FitsValue>,
391) -> Result<(), ImageError> {
392    if !wcs.contains_key("RESTFRQ") {
393        wcs.remove("SPECSYS");
394    }
395    let mut axis = 1;
396    for index in 0..coords.n_coordinates() {
397        let coordinate = coords.coordinate(index);
398        match coordinate {
399            CoordinateModel::Direction(direction) => {
400                let frame = direction.direction_ref();
401                let (longitude, latitude) = match frame {
402                    DirectionRef::GALACTIC => ("GLON", "GLAT"),
403                    DirectionRef::SUPERGAL => ("SLON", "SLAT"),
404                    DirectionRef::ECLIPTIC | DirectionRef::MECLIPTIC | DirectionRef::TECLIPTIC => {
405                        ("ELON", "ELAT")
406                    }
407                    DirectionRef::J2000
408                    | DirectionRef::JMEAN
409                    | DirectionRef::APP
410                    | DirectionRef::B1950
411                    | DirectionRef::B1950Vla
412                    | DirectionRef::BMEAN
413                    | DirectionRef::BTRUE
414                    | DirectionRef::ICRS
415                    | DirectionRef::HADEC => ("RA--", "DEC-"),
416                    _ => ("??LN", "??LT"),
417                };
418                let projection = direction.projection().name();
419                for (offset, name) in [longitude, latitude].into_iter().enumerate() {
420                    wcs.insert(
421                        format!("CTYPE{}", axis + offset),
422                        FitsValue::String(format!("{name}-{projection}")),
423                    );
424                }
425                wcs.remove("EQUINOX");
426                wcs.remove("RADESYS");
427                let reference = match frame {
428                    DirectionRef::J2000 => Some(("FK5", Some(2000.0))),
429                    DirectionRef::B1950 => Some(("FK4", Some(1950.0))),
430                    DirectionRef::B1950Vla => Some(("FK4", Some(1979.9))),
431                    DirectionRef::ICRS => Some(("ICRS", None)),
432                    _ => None,
433                };
434                if let Some((system, equinox)) = reference {
435                    wcs.insert("RADESYS".into(), FitsValue::String(system.into()));
436                    if let Some(equinox) = equinox {
437                        wcs.insert("EQUINOX".into(), FitsValue::Float(equinox));
438                    }
439                }
440            }
441            CoordinateModel::Stokes(stokes) => {
442                let (reference, increment) =
443                    fits_stokes_axis(stokes.stokes()).ok_or_else(header_creation_error)?;
444                wcs.insert(format!("CRVAL{axis}"), FitsValue::Float(reference));
445                wcs.insert(format!("CDELT{axis}"), FitsValue::Float(increment));
446                wcs.insert(format!("CRPIX{axis}"), FitsValue::Float(1.0));
447            }
448            _ => {}
449        }
450        axis += coordinate.n_pixel_axes();
451    }
452    for (key, value) in wcs.iter_mut() {
453        if matches!(split_axis_keyword(key), Some(("CTYPE" | "CUNIT", _)))
454            && let FitsValue::String(text) = value
455        {
456            truncate_bytes(text, 8);
457        }
458    }
459    Ok(())
460}

462fn projection_parameters(coords: &CoordinateSystem) -> Vec<(String, FitsValue)> {
463    let mut result = Vec::new();
464    let mut offset = 0;
465    for index in 0..coords.n_coordinates() {
466        let axis = coords.coordinate(index);
467        if let CoordinateModel::Direction(direction) = axis {
468            let projection = direction.projection();
469            let params = projection.parameters();
470            let count = if projection.projection_type() == ProjectionType::SIN {
471                2
472            } else {
473                params.len()
474            };
475            for index in 0..count {
476                result.push((
477                    format!("PV{}_{}", offset + 2, index + 1),
478                    FitsValue::Float(params.get(index).copied().unwrap_or(0.0)),
479                ));
480            }
481        }
482        offset += axis.n_pixel_axes();
483    }
484    result
485}

487fn header_creation_error() -> ImageError {
488    ImageError::InvalidMetadata("header creation failed: Error creating initial FITS header".into())
489}

491fn misc_keyword_names(
492    name: &str,
493    shape: &[usize],
494    count: usize,
495) -> Result<Vec<String>, ImageError> {
496    match shape {
497        [] => Ok(vec![if name.starts_with("COMMENT") {
498            "COMMENT".into()
499        } else if name.starts_with("HISTORY") {
500            "HISTORY".into()
501        } else {
502            name[..name.len().min(8)].to_owned()
503        }]),
504        [length] if *length <= 999 && name.len() + length.to_string().len() <= 8 => {
505            Ok((1..=count).map(|index| format!("{name}{index}")).collect())
506        }
507        [rows, cols] if name.len() <= 2 && *rows <= 99 && *cols <= 99 => {
508            let width = if *rows > 9 { 2 } else { 1 };
509            Ok((0..count)
510                .map(|index| {
511                    let row = index % rows + 1;
512                    let col = index / rows + 1;
513                    format!("{name}{row:0width$}_{col:0width$}")
514                })
515                .collect())
516        }
517        _ => Err(header_creation_error()),
518    }
519}

521fn misc_values(name: &str, value: &Value) -> Result<Vec<FitsValue>, ImageError> {
522    let card_integer = |value| FitsValue::Integer(i64::from(value as i32));
523    macro_rules! array {
524        ($values:expr, $variant:ident, $map:expr) => {
525            $values
526                .t()
527                .iter()
528                .map(|value| FitsValue::$variant($map(value)))
529                .collect()
530        };
531    }
532    let values = match value {
533        Value::Scalar(value) => {
534            let scalar = match value {
535                ScalarValue::Bool(value) => FitsValue::Logical(*value),
536                ScalarValue::UInt8(value) => FitsValue::Integer(i64::from(*value)),
537                ScalarValue::UInt16(value) => FitsValue::Integer(i64::from(*value)),
538                ScalarValue::UInt32(value) => card_integer(i64::from(*value)),
539                ScalarValue::Int16(value) => FitsValue::Integer(i64::from(*value)),
540                ScalarValue::Int32(value) => FitsValue::Integer(i64::from(*value)),
541                ScalarValue::Int64(value) => card_integer(*value),
542                ScalarValue::Float32(value) => FitsValue::Float(f64::from(*value)),
543                ScalarValue::Float64(value) => FitsValue::Float(*value),
544                ScalarValue::String(value) => {
545                    let mut text = value.clone();
546                    if name.starts_with("COMMENT") || name.starts_with("HISTORY") {
547                        truncate_bytes(&mut text, 72);
548                        return Ok(vec![FitsValue::String(text)]);
549                    }
550                    FitsValue::String(text)
551                }
552                ScalarValue::Complex32(_) | ScalarValue::Complex64(_) => {
553                    return Err(header_creation_error());
554                }
555            };
556            vec![scalar]
557        }
558        Value::Array(ArrayValue::Complex32(_) | ArrayValue::Complex64(_)) => Vec::new(),
559        Value::Array(value) if value.shape().len() > 2 => return Err(header_creation_error()),
560        Value::Array(value) => match value {
561            ArrayValue::Bool(values) => array!(values, Logical, |value: &bool| *value),
562            ArrayValue::UInt8(values) => array!(values, Integer, |value: &u8| i64::from(*value)),
563            ArrayValue::UInt16(values) => array!(values, Integer, |value: &u16| i64::from(*value)),
564            ArrayValue::UInt32(values) => {
565                array!(values, Integer, |value: &u32| i64::from(*value as i32))
566            }
567            ArrayValue::Int16(values) => array!(values, Integer, |value: &i16| i64::from(*value)),
568            ArrayValue::Int32(values) => array!(values, Integer, |value: &i32| i64::from(*value)),
569            ArrayValue::Int64(values) => {
570                array!(values, Integer, |value: &i64| i64::from(*value as i32))
571            }
572            ArrayValue::Float32(values) => array!(values, Float, |value: &f32| f64::from(*value)),
573            ArrayValue::Float64(values) => array!(values, Float, |value: &f64| *value),
574            ArrayValue::String(values) => array!(values, String, Clone::clone),
575            _ => Vec::new(),
576        },
577        _ => Vec::new(),
578    };
579    if values
580        .iter()
581        .any(|value| matches!(value, FitsValue::String(text) if text.len() > 68))
582    {
583        return Err(header_creation_error());
584    }
585    Ok(values)
586}

588fn group_name(name: &str) -> Option<&'static str> {
589    if name == "NAXIS" {
590        return Some("NAXIS");
591    }
592    if let Some((base, _)) = split_axis_keyword(name) {
593        return (base != "CUNIT").then_some(base);
594    }
595    name.strip_prefix("NAXIS")
596        .filter(|axis| !axis.is_empty() && axis.bytes().all(|ch| ch.is_ascii_digit()))
597        .map(|_| "NAXIS")
598}

600fn padded_text(name: &str, value: &str) -> String {
601    if matches!(name, "        " | "END" | "HISTORY" | "COMMENT") {
602        value.to_owned()
603    } else {
604        let mut text = value.to_owned();
605        text.extend(std::iter::repeat_n(' ', 8_usize.saturating_sub(text.len())));
606        text
607    }
608}
609
610fn scalar(name: &str, value: &FitsValue) -> ScalarValue {
611    match value {
612        FitsValue::String(value) => ScalarValue::String(padded_text(name, value)),
613        FitsValue::Integer(value) => ScalarValue::Int64(*value),
614        FitsValue::Float(value) => ScalarValue::Float64(*value),
615        FitsValue::Logical(value) => ScalarValue::Bool(*value),
616    }
617}

619fn array(name: &str, values: &[FitsValue]) -> ArrayValue {
620    match values.first().expect("at least one FITS card in a group") {
621        FitsValue::String(_) => ArrayValue::from_string_vec(
622            values
623                .iter()
624                .filter_map(|value| match value {
625                    FitsValue::String(text) => Some(padded_text(name, text)),
626                    _ => None,
627                })
628                .collect(),
629        ),
630        FitsValue::Integer(_) => ArrayValue::Int64(
631            ndarray::Array1::from_iter(values.iter().filter_map(|value| match value {
632                FitsValue::Integer(integer) => Some(*integer),
633                _ => None,
634            }))
635            .into_dyn(),
636        ),
637        FitsValue::Float(_) => ArrayValue::from_f64_vec(
638            values
639                .iter()
640                .filter_map(|value| match value {
641                    FitsValue::Float(float) => Some(*float),
642                    _ => None,
643                })
644                .collect(),
645        ),
646        FitsValue::Logical(_) => ArrayValue::Bool(
647            ndarray::Array1::from_iter(values.iter().filter_map(|value| match value {
648                FitsValue::Logical(flag) => Some(*flag),
649                _ => None,
650            }))
651            .into_dyn(),
652        ),
653    }
654}
655
656fn render_card(keyword: &FitsKeyword) -> String {
657    let name = &keyword.name;
658    let mut card = match &keyword.value {
659        FitsValue::String(text) if matches!(name.as_str(), "" | "END" | "HISTORY" | "COMMENT") => {
660            format!("{name:<8}{text}")
661        }
662        value => {
663            let field = match value {
664                FitsValue::String(text) => {
665                    format!("'{}'", padded_text(name, text).replace('\'', "''"))
666                }
667                FitsValue::Integer(integer) => format!("{integer:>20}"),
668                FitsValue::Logical(flag) => format!("{:>20}", if *flag { 'T' } else { 'F' }),
669                FitsValue::Float(number) => {
670                    let token = format!("{number:.12E}");
671                    let field = token.split_once('E').map_or_else(
672                        || token.clone(),
673                        |(mantissa, exponent)| {
674                            format!(
675                                "{mantissa}E{:+03}",
676                                exponent.parse::<i32>().expect("decimal exponent")
677                            )
678                        },
679                    );
680                    format!("{field:>20}")
681                }
682            };
683            format!("{name:<8}= {field}")
684        }
685    };
686    if let Some(comment) = &keyword.comment {
687        card.extend(std::iter::repeat_n(
688            ' ',
689            30_usize.saturating_sub(card.len()),
690        ));
691        card.push_str(" /");
692        card.push_str(comment);
693    }
694    truncate_bytes(&mut card, 80);
695    card.extend(std::iter::repeat_n(' ', 80 - card.len()));
696    card
697}
698
699fn truncate_bytes(text: &mut String, limit: usize) {
700    if text.len() > limit {
701        let boundary = (0..=limit)
702            .rev()
703            .find(|index| text.is_char_boundary(*index))
704            .expect("zero is a boundary");
705        text.truncate(boundary);
706    }
707}
