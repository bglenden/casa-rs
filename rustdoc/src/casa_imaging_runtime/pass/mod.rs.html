2//! The major-cycle pass (plan section 5.4): one traversal of the selected
3//! visibilities that accumulates the dirty or residual image terms, and the
4//! PSF on the initial pass, of every image domain, for every capability.
5//!
6//! A pass reads blocks of native rows, projected on every image domain, from
7//! a [`BoundedSource`] through a two-slot stream, places each row with each
8//! domain's spectral resampler and the imaging weights, routes the
9//! placements to the owners of the domain's [`Partition`] and accumulates
10//! them with the CPU backend: the residual `V − Σ_d A_d·m_d` of every
11//! domain's model when a model is present (CASA `SIMapperCollection::degrid`
12//! sums every mapper's prediction before `grid` forms the residual), `V`
13//! otherwise. Each wave of planes ([`Residency`]) ends with the per-plane
14//! transforms, single-threaded per worker, and hands each domain's
15//! [`NormalImages`] to the caller before the next wave starts.

61/// A failure of a major-cycle or density pass.
62#[derive(Debug, thiserror::Error)]
63pub enum PassError {
64    /// A worker team was asked for no workers.
65    #[error("a worker team needs at least one worker")]
66    Workers,
67    /// A pass was given no image domain.
68    #[error("a pass needs at least one image domain")]
69    NoDomains,
70    /// The image domains of a pass do not share one plane axis.
71    #[error("the image domains of a pass must have the same planes")]
72    PlaneAxes,
73    /// A row's native channels lie further apart than the spacing that
74    /// sized the waves' model halo.
75    #[error("native channels {observed_hz} Hz apart exceed the planned spacing {bound_hz} Hz")]
76    NativeSpacing {
77        /// The widest spacing in the row.
78        observed_hz: f64,
79        /// The pass's `native_spacing_hz`.
80        bound_hz: f64,
81    },
82    /// The measurement operator rejected a row, model or accumulator.
83    #[error("measurement operator: {0}")]
84    Operator(#[from] OperatorError),
85    /// The visibility source failed.
86    #[error("visibility source: {0}")]
87    Source(#[source] SourceError),
88    /// The model grids of a wave could not be prepared.
89    #[error("model preparation: {0}")]
90    Model(#[source] SourceError),
91    /// The consumer of a wave's images failed.
92    #[error("wave images: {0}")]
93    Images(#[source] SourceError),
94    /// The visibility writer failed.
95    #[error("visibility write: {0}")]
96    VisibilityWrite(#[source] SourceError),
97    /// A writing pass must hold every plane at once: a native sample's
98    /// prediction can draw on any output channel, and each row is written
99    /// once.
100    #[error("writing visibilities needs every plane resident")]
101    VisibilityWriteWaves,
102    /// Not even one plane of the pass fits the memory budget.
103    #[error("one plane needs {required} bytes but the pass may use {available}")]
104    Memory {
105        /// Bytes one plane needs.
106        required: u64,
107        /// Bytes the pass may use.
108        available: u64,
109    },
110    /// The pass was cancelled at a block boundary.
111    #[error("the pass was cancelled")]
112    Cancelled,
113    /// The source thread panicked or could not start.
114    #[error("the visibility source thread failed")]
115    ProducerPanicked,
116}

118/// Cooperative cancellation shared between the caller and running passes;
119/// a pass stops at the next block boundary.
120#[derive(Clone, Debug, Default)]
121pub struct Cancel(Arc<AtomicBool>);
122
123impl Cancel {
124    /// A token that is not cancelled.
125    #[must_use]
126    pub fn new() -> Self {
127        Self::default()
128    }
129
130    /// Request cancellation.
131    pub fn cancel(&self) {
132        self.0.store(true, Ordering::Release);
133    }
134
135    /// Whether cancellation was requested.
136    #[must_use]
137    pub fn is_cancelled(&self) -> bool {
138        self.0.load(Ordering::Acquire)
139    }
140}

142/// A re-traversable source of native rows.
143///
144/// Each wave of a pass begins one traversal. Rows are delivered in a fixed
145/// order, so repeated traversals see the same samples.
146pub trait BoundedSource: Send {
147    /// Start a traversal for a wave over `planes`. With `restrict`, the
148    /// source may deliver only the native channels whose samples reach
149    /// `planes`, with both interpolation partners of each; without it, it
150    /// delivers every selected channel of each row.
151    fn begin(&mut self, planes: PlaneRange, restrict: bool) -> Result<(), SourceError>;
152
153    /// Fill `block` with the next rows of the traversal; `Ok(false)` once
154    /// the traversal is exhausted.
155    fn fill(&mut self, block: &mut NativeBlock) -> Result<bool, SourceError>;
156}

170/// One major-cycle pass over the selected visibilities.
171pub struct MajorCyclePass<'a> {
172    /// The image domains; rows arrive projected on each, in this order.
173    pub domains: &'a [PassDomain<'a>],
174    /// The run's imaging weights.
175    pub weighting: &'a WeightingGeneration,
176    /// Modes accumulated: data and PSF on the initial pass, data later.
177    pub modes: ModeSet,
178    /// Model grids of a domain's planes; `None` grids the data themselves
179    /// (an initial pass without a start model).
180    pub model: Option<&'a ModelPreparation<'a>>,
181    /// Planes held at once.
182    pub residency: Residency,
183    /// The widest spacing between adjacent selected native channels; with
184    /// a model it sizes each wave's model halo
185    /// ([`SpectralResampler::model_planes`]).
186    pub native_spacing_hz: f64,
187}
188
189impl MajorCyclePass<'_> {
190    /// Whether the pass forms its residual at native channels: with a model,
191    /// when several domains' predictions must be summed before subtracting,
192    /// or when the resampler interpolates onto the output channels
193    /// ([`SpectralResampler::forms_native_residuals`]).
194    fn native_residuals(&self) -> bool {
195        self.model.is_some()
196            && (self.domains.len() > 1
197                || self
198                    .domains
199                    .iter()
200                    .any(|domain| domain.resampler.forms_native_residuals()))
201    }
202
203    /// Whether a wave must read whole rows: a native-channel prediction under
204    /// linear interpolation depends on the whole row's channel map
205    /// ([`SpectralResampler::predict_row`]), not only the channels that
206    /// reach the wave.
207    fn whole_rows(&self) -> bool {
208        self.model.is_some()
209            && self
210                .domains
211                .iter()
212                .any(|domain| domain.resampler.forms_native_residuals())
213    }
214
215    /// What one wave of this pass holds, for [`Residency::plan`].
216    #[must_use]
217    pub fn demand(&self, workers: usize) -> WaveDemand<'_> {
218        WaveDemand {
219            domains: self.domains,
220            modes: self.modes,
221            with_model: self.model.is_some(),
222            native_spacing_hz: self.native_spacing_hz,
223            workers,
224        }
225    }
226}

228/// What a pass traversed.
229#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
230pub struct PassSummary {
231    /// Samples placed on the first image domain, over every wave: the same
232    /// count for every pass over the same selection and weights.
233    pub samples: u64,
234    /// Source blocks consumed, over every wave.
235    pub blocks: u64,
236}

238/// Run `pass`, handing each wave's normal images of each domain to
239/// `images` (domain, images) in plane order and, on a final pass that
240/// writes visibilities, every block to `visibilities`.
241pub fn run_major_cycle(
242    pass: &MajorCyclePass<'_>,
243    source: &mut dyn BoundedSource,
244    team: &WorkerTeam,
245    cancel: &Cancel,
246    images: &mut dyn FnMut(usize, NormalImages) -> Result<(), PassError>,
247    mut visibilities: Option<&mut VisibilitySink<'_>>,
248) -> Result<PassSummary, PassError> {
249    if visibilities.is_some() && pass.residency != Residency::All {
250        return Err(PassError::VisibilityWriteWaves);
251    }
252    let Some(main) = pass.domains.first() else {
253        return Err(PassError::NoDomains);
254    };
255    let total = main.operator.basis().planes();
256    if pass
257        .domains
258        .iter()
259        .any(|domain| domain.operator.basis().planes() != total)
260    {
261        return Err(PassError::PlaneAxes);
262    }
263    let restrict = !pass.whole_rows();
264    let native_residuals = pass.native_residuals();
265    let mut summary = PassSummary::default();
266    for planes in pass.residency.waves(total) {
267        source.begin(planes, restrict).map_err(PassError::Source)?;
268        let models = pass
269            .model
270            .map(|prepare| {
271                pass.domains
272                    .iter()
273                    .enumerate()
274                    .map(|(index, domain)| {
275                        prepare(
276                            index,
277                            domain
278                                .resampler
279                                .model_planes(planes, pass.native_spacing_hz),
280                        )
281                    })
282                    .collect::<Result<Vec<_>, _>>()
283            })
284            .transpose()?;
285        let mut wave = Wave::new(pass, planes, models.as_deref(), native_residuals)
286            .checking_spacing(!restrict && pass.residency != Residency::All);
287        summary.blocks += stream_blocks(source, cancel, |block| {
288            wave.consume(block, team, visibilities.as_deref_mut())
289        })?;
290        summary.samples += wave.samples();
291        for (domain, domain_images) in wave.finish(team)?.into_iter().enumerate() {
292            images(domain, domain_images)?;
293        }
294    }
295    Ok(summary)
296}

298/// Accumulate the weight-density grid of `shape` over one traversal of the
299/// first image domain's projection: each row is placed with
300/// [`SpectralResampler::place_density`] on row chunks and the chunks are
301/// added in row order.
302pub fn run_density_pass(
303    operator: &MeasurementOperator,
304    resampler: &SpectralResampler,
305    shape: DensityGridShape,
306    source: &mut dyn BoundedSource,
307    team: &WorkerTeam,
308    cancel: &Cancel,
309) -> Result<DensityGrid, PassError> {
310    source
311        .begin(PlaneRange::new(0, operator.basis().planes()), false)
312        .map_err(PassError::Source)?;
313    let mut grid = DensityGrid::new(shape);
314    let mut chunks = (0..team.workers() * 4)
315        .map(|_| (0..0, SampleBuffer::new(1)))
316        .collect::<Vec<_>>();
317    stream_blocks(source, cancel, |block| {
318        let rows = block.len();
319        let count = chunks.len().min(rows.max(1));
320        for (index, (range, _)) in chunks[..count].iter_mut().enumerate() {
321            *range = index * rows / count..(index + 1) * rows / count;
322        }
323        team.for_each_mut(&mut chunks[..count], |_, (range, buffer)| {
324            buffer.clear();
325            for row in range.clone() {
326                resampler.place_density(operator, &block.row(0, row), &shape, buffer)?;
327            }
328            Ok::<_, PassError>(())
329        })?;
330        for (_, buffer) in &chunks[..count] {
331            grid.accumulate(&buffer.block());
332        }
333        Ok(())
334    })?;
335    Ok(grid)
