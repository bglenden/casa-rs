3//! The visibility-weighting contract: CASA's imaging-weight formulas
4//! (`SynthesisImagerVi2::weight`), the scope and cells of the visibility
5//! density the density formulas divide by, and the Gaussian uv taper
6//! (`VisImagingWeight::setFilter`).
7//!
8//! CASA's density formulas are one law, `w / (D·f² + d²)` over the gridded
9//! weight density `D` (`VisImagingWeight::weightUniform`,
10//! `BriggsCubeWeightor::getWeightUniform`): uniform is `f² = 1`, `d² = 0`,
11//! and Briggs's two robustness modes choose `f²` and `d²`. What CASA's
12//! `npixels` changes is the grid `D` lives on ([`DensityCells`]).
13
14use std::num::{NonZeroU32, NonZeroUsize};
15
16/// Visibility-weighting formula.
17#[derive(Debug, Clone, Copy, PartialEq)]
18pub enum WeightingScheme {
19    /// Natural weighting.
20    Natural,
21    /// CASA `radial`: each sample's weight times its uv distance in
22    /// wavelengths (`VisImagingWeight::weightRadial`); no density.
23    Radial,
24    /// Uniform density weighting, `w / D`: CASA `uniform`, and CASA
25    /// `superuniform` with a [`DensityCells::Box`] density.
26    Uniform,
27    /// Briggs robust weighting (CASA `rmode = "norm"`):
28    /// `f² = (5·10^{−robust})² / (ΣD² / ΣD)`, `d² = 1`.
29    Briggs {
30        /// Robustness in the conventional interval `[-2, 2]`.
31        robust: f64,
32    },
33    /// CASA `briggsabs` (`rmode = "abs"`): `f² = robust²` and
34    /// `d² = 2·noise²`, the expected noise of one visibility of unit weight.
35    BriggsAbsolute {
36        /// Robustness in the conventional interval `[-2, 2]`.
37        robust: f64,
38        /// The noise in Jy; it and `robust` are not both zero.
39        noise_jy: f64,
40    },
41    /// Briggs bandwidth-taper weighting.
42    BriggsBandwidthTaper {
43        /// Robustness in the conventional interval `[-2, 2]`.
44        robust: f64,
45    },
46}
47
48impl WeightingScheme {
49    /// Whether the formula divides by a gridded visibility density.
50    #[must_use]
51    pub const fn uses_density(self) -> bool {
52        !matches!(self, Self::Natural | Self::Radial)
53    }
54}
55
56/// Domain over which visibility-density weights are derived.
57#[derive(Debug, Clone, Copy, PartialEq, Eq)]
58pub enum WeightDensityScope {
59    /// The weighting formula does not use a density generation.
60    NotApplicable,
61    /// All selected data contributing to the logical product.
62    GlobalSelection,
63    /// An explicit global density generation per output channel.
64    PerOutputChannel,
65    /// One density per field over all its selected data, each with its own
66    /// Briggs factors: tclean's `mosweight` (`VisImagingWeight`'s
67    /// `multiField`, which keys a density by MeasurementSet and FIELD_ID).
68    /// The densities are those of FIELD_IDs `first_field` to
69    /// `first_field + fields − 1`; a field in the span with no data keeps an
70    /// empty density.
71    PerField {
72        /// The first FIELD_ID with a density.
73        first_field: u32,
74        /// The FIELD_IDs the densities span.
75        fields: NonZeroU32,
76    },
77    /// One density per field and output channel: tclean's `mosweight` on a
78    /// cube's per-channel density (`BriggsCubeWeightor`'s `multiField`).
79    /// Each field's densities are the [`Self::PerOutputChannel`] densities
80    /// of that field's samples alone, each channel with its own Briggs
81    /// factors, over the fields [`Self::PerField`] spans.
82    PerFieldAndOutputChannel {
83        /// The first FIELD_ID with densities.
84        first_field: u32,
85        /// The FIELD_IDs the densities span.
86        fields: NonZeroU32,
87    },
88}
89
90impl WeightDensityScope {
91    /// Whether one density covers all the data of what it weighs, a field or
92    /// the whole selection, rather than one output channel: the continuum
93    /// density of CASA's `VisImagingWeight`.
94    #[must_use]
95    pub const fn is_continuum(self) -> bool {
96        matches!(self, Self::GlobalSelection | Self::PerField { .. })
97    }
98
99    /// Whether each output channel has its own density: CASA's
100    /// `BriggsCubeWeightor`.
101    #[must_use]
102    pub const fn is_per_channel(self) -> bool {
103        matches!(
104            self,
105            Self::PerOutputChannel | Self::PerFieldAndOutputChannel { .. }
106        )
107    }
108
109    /// The first FIELD_ID and the FIELD_IDs spanned by a density per field
110    /// (`mosweight`); `None` when one density covers every field.
111    #[must_use]
112    pub const fn fields(self) -> Option<(u32, NonZeroU32)> {
113        match self {
114            Self::PerField {
115                first_field,
116                fields,
117            }
118            | Self::PerFieldAndOutputChannel {
119                first_field,
120                fields,
121            } => Some((first_field, fields)),
122            Self::NotApplicable | Self::GlobalSelection | Self::PerOutputChannel => None,
123        }
124    }
125}

127/// The cells a visibility density is gridded on: CASA's `npixels`, whose
128/// meaning `SynthesisImagerVi2::weight` takes from the weighting and the
129/// density scope.
130#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
131pub enum DensityCells {
132    /// The image's own uv cells; a sample adds to the cell it falls in.
133    #[default]
134    Image,
135    /// An `n × n` grid spanning the image's uv extent, so each cell is
136    /// `N / n` image uv cells wide along an image axis of `N` pixels: CASA's
137    /// continuum (`VisImagingWeight`) uniform and Briggs density with
138    /// `npixels = n`, its sidelobes suppressed over the inner `n` pixels.
139    Square(NonZeroUsize),
140    /// The image's uv cells, each sample spread over the `(2b + 1)²` cells
141    /// about its own: CASA `superuniform`'s box (`VisImagingWeight`'s
142    /// `uBox = vBox = b`, with `b = npixels`, or 3 when `npixels ≤ 0`), and a
143    /// cube's per-channel density with `npixels = b`
144    /// (`BriggsCubeWeightor`'s `superUniformBox`).
145    Box(NonZeroUsize),
146}

186/// Complete logical visibility-weighting requirements.
187#[derive(Debug, Clone, Copy, PartialEq)]
188pub struct WeightingContract {
189    scheme: WeightingScheme,
190    density_scope: WeightDensityScope,
191    density_cells: DensityCells,
192    uv_taper: Option<UvTaper>,
193    casa_cube_density_padding: Option<usize>,
194}
195
196impl WeightingContract {
197    /// Construct weighting requirements on the image's own density cells.
198    #[must_use]
199    pub const fn new(scheme: WeightingScheme, density_scope: WeightDensityScope) -> Self {
200        Self {
201            scheme,
202            density_scope,
203            density_cells: DensityCells::Image,
204            uv_taper: None,
205            casa_cube_density_padding: None,
206        }
207    }
208
209    /// Grid the density on `cells` (CASA's `npixels`).
210    #[must_use]
211    pub const fn with_density_cells(mut self, cells: DensityCells) -> Self {
212        self.density_cells = cells;
213        self
214    }
215
216    /// Add a Gaussian UV taper to the weighting metric.
217    #[must_use]
218    pub const fn with_uv_taper(mut self, uv_taper: UvTaper) -> Self {
219        self.uv_taper = Some(uv_taper);
220        self
221    }
222
223    /// Bind CASA's cube density and native-weight transfer law, with a
224    /// metadata-derived number of density channels on each side of the image.
225    /// Zero padding still selects the cube law. The published axis is unchanged.
226    #[must_use]
227    pub const fn with_casa_cube_density_padding(mut self, channels_per_side: usize) -> Self {
228        self.casa_cube_density_padding = Some(channels_per_side);
229        self
230    }
231
232    /// Return the bound CASA cube law's symmetric density-axis padding.
233    #[must_use]
234    pub const fn casa_cube_density_padding(self) -> Option<usize> {
235        self.casa_cube_density_padding
236    }
237
238    /// Return the weighting formula.
239    #[must_use]
240    pub const fn scheme(self) -> WeightingScheme {
241        self.scheme
242    }
243
244    /// Return the density-generation scope.
245    #[must_use]
246    pub const fn density_scope(self) -> WeightDensityScope {
247        self.density_scope
248    }
249
250    /// Return the cells the density is gridded on.
251    #[must_use]
252    pub const fn density_cells(self) -> DensityCells {
253        self.density_cells
254    }
255
256    /// Return the optional Gaussian UV taper.
257    #[must_use]
258    pub const fn uv_taper(self) -> Option<UvTaper> {
259        self.uv_taper
260    }
261
262    /// Why the requirements are outside the logical domain, if they are.
263    pub(crate) fn validate(self) -> Result<(), &'static str> {
264        if self.casa_cube_density_padding.is_some() && !self.density_scope.is_per_channel() {
265            return Err("CASA cube density requires per-output-channel weighting");
266        }
267        let density = self.scheme.uses_density();
268        if !density && self.density_scope != WeightDensityScope::NotApplicable {
269            return Err("natural and radial weighting have no density generation");
270        }
271        if density && self.density_scope == WeightDensityScope::NotApplicable {
272            return Err("density weighting requires an explicit global density scope");
273        }
274        match self.density_cells {
275            DensityCells::Image => {}
276            _ if !density => return Err("a weighting without a density has no density cells"),
277            DensityCells::Square(_) if !self.density_scope.is_continuum() => {
278                return Err("a square density grid is CASA's continuum density");
279            }
280            _ if matches!(self.scheme, WeightingScheme::BriggsBandwidthTaper { .. }) => {
281                return Err("CASA's bandwidth-tapered Briggs density has the image's cells");
282            }
283            DensityCells::Square(_) | DensityCells::Box(_) => {}
284        }
285        let robust = match self.scheme {
286            WeightingScheme::Briggs { robust }
287            | WeightingScheme::BriggsAbsolute { robust, .. }
288            | WeightingScheme::BriggsBandwidthTaper { robust } => Some(robust),
289            WeightingScheme::Natural | WeightingScheme::Radial | WeightingScheme::Uniform => None,
290        };
291        if robust.is_some_and(|value| !(value.is_finite() && (-2.0..=2.0).contains(&value))) {
292            return Err("Briggs robustness must be finite and in [-2, 2]");
293        }
294        if let WeightingScheme::BriggsAbsolute { robust, noise_jy } = self.scheme
295            && (!noise_jy.is_finite() || (robust == 0.0 && noise_jy == 0.0))
296        {
297            // CASA's denominator robust²·D + 2·noise² would vanish
298            // (`SynthesisImagerVi2::weight` refuses it for cubes).
299            return Err("absolute Briggs noise must be finite, and not zero with robust zero");
300        }
301        if self.uv_taper.is_some_and(|taper| {
302            !(taper.major_lambda.is_finite()
303                && taper.major_lambda > 0.0
304                && taper.minor_lambda.is_finite()
305                && taper.minor_lambda > 0.0
306                && taper.position_angle_rad.is_finite())
307        }) {
308            return Err("UV taper axes must be finite and positive and its angle must be finite");
309        }
310        Ok(())
311    }
312}

314#[cfg(test)]
315mod tests {
316    use super::*;
317
318    fn cells(count: usize) -> NonZeroUsize {
319        NonZeroUsize::new(count).expect("non-zero")
320    }
321
322    fn contract(scheme: WeightingScheme, scope: WeightDensityScope) -> WeightingContract {
323        WeightingContract::new(scheme, scope)
324    }
325
326    #[test]
327    fn density_cells_follow_casas_npixels_readings() {
328        use WeightDensityScope::{GlobalSelection, NotApplicable, PerOutputChannel};
329        let briggs = WeightingScheme::Briggs { robust: 0.5 };
330        let absolute = WeightingScheme::BriggsAbsolute {
331            robust: 1.0,
332            noise_jy: 0.1,
333        };
334        let per_field = WeightDensityScope::PerField {
335            first_field: 2,
336            fields: NonZeroU32::new(3).expect("non-zero"),
337        };
338        let per_field_channel = WeightDensityScope::PerFieldAndOutputChannel {
339            first_field: 2,
340            fields: NonZeroU32::new(3).expect("non-zero"),
341        };
342        for valid in [
343            contract(WeightingScheme::Radial, NotApplicable),
344            // `mosweight`'s densities are continuum densities.
345            contract(briggs, per_field).with_density_cells(DensityCells::Square(cells(7))),
346            contract(WeightingScheme::Uniform, per_field)
347                .with_density_cells(DensityCells::Box(cells(3))),
348            contract(WeightingScheme::Uniform, GlobalSelection)
349                .with_density_cells(DensityCells::Square(cells(32))),
350            contract(briggs, GlobalSelection).with_density_cells(DensityCells::Square(cells(7))),
351            contract(absolute, GlobalSelection),
352            // Superuniform is uniform over a box.
353            contract(WeightingScheme::Uniform, GlobalSelection)
354                .with_density_cells(DensityCells::Box(cells(3))),
355            contract(absolute, PerOutputChannel)
356                .with_density_cells(DensityCells::Box(cells(2)))
357                .with_casa_cube_density_padding(4),
358            // `mosweight` on a cube: `BriggsCubeWeightor`'s `multiField`.
359            contract(briggs, per_field_channel)
360                .with_density_cells(DensityCells::Box(cells(2)))
361                .with_casa_cube_density_padding(2),
362        ] {
363            assert_eq!(valid.validate(), Ok(()), "{valid:?}");
364        }
365        for invalid in [
366            contract(WeightingScheme::Radial, GlobalSelection),
367            contract(WeightingScheme::Natural, per_field),
368            contract(briggs, per_field).with_casa_cube_density_padding(1),
369            contract(briggs, per_field_channel)
370                .with_density_cells(DensityCells::Square(cells(8)))
371                .with_casa_cube_density_padding(2),
372            contract(WeightingScheme::Natural, NotApplicable)
373                .with_density_cells(DensityCells::Box(cells(3))),
374            contract(briggs, PerOutputChannel).with_density_cells(DensityCells::Square(cells(8))),
375            contract(
376                WeightingScheme::BriggsBandwidthTaper { robust: 0.5 },
377                PerOutputChannel,
378            )
379            .with_density_cells(DensityCells::Box(cells(1))),
380            contract(
381                WeightingScheme::BriggsAbsolute {
382                    robust: 0.0,
383                    noise_jy: 0.0,
384                },
385                GlobalSelection,
386            ),
387            contract(
388                WeightingScheme::BriggsAbsolute {
389                    robust: 2.5,
390                    noise_jy: 1.0,
391                },
392                GlobalSelection,
393            ),
394            contract(
395                WeightingScheme::BriggsAbsolute {
396                    robust: 0.5,
397                    noise_jy: f64::NAN,
398                },
399                GlobalSelection,
400            ),
401        ] {
402            assert!(invalid.validate().is_err(), "{invalid:?}");
403        }
404        assert_eq!(per_field.fields(), per_field_channel.fields());
405        assert_eq!(per_field.fields().map(|(first, _)| first), Some(2));
406        assert!(per_field_channel.is_per_channel() && !per_field_channel.is_continuum());
407        assert!(per_field.is_continuum() && !per_field.is_per_channel());
408        assert_eq!(PerOutputChannel.fields(), None);
409        // Either term alone keeps the absolute denominator positive.
410        for (robust, noise_jy) in [(0.0, 1.0), (-1.0, 0.0)] {
411            let scheme = WeightingScheme::BriggsAbsolute { robust, noise_jy };
412            assert_eq!(contract(scheme, GlobalSelection).validate(), Ok(()));
413        }
414    }
