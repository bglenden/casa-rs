3//! Direct standard-gridder, natural-weight, Stokes-I cube band kernels.
4//!
5//! The input is borrowed channel-major/correlation-minor numeric arrays, with
6//! geometry once per row. Grids are contiguous [channel, x, y] Complex64 arrays;
7//! compensation is a separate, equally shaped allocation. Plane views neither
8//! allocate nor copy. The coarse epoch job loads bounded model planes before
9//! numerical work; payload elements carry no workflow or publication state.

37/// Shape and conservative row-union dependencies for one output band. Runtime

49impl BandPlan {
50    /// Whether the compiled geometry is implemented by the native cube kernel.

52    pub fn supports(specification: &SpectralOperatorSpecification) -> bool {
53        specification.cube_geometry().is_ok()

56    /// Compile the supported standard, Stokes-I, linear cube band shape.
57    pub fn new(
58        specification: &SpectralOperatorSpecification,
59        pass: SpectralOperatorPass,
60    ) -> Result<Self, SpectralOperatorError> {
61        Ok(Self {
62            geometry: specification.cube_geometry()?,
63            core: specification.slab().core_range(),
64            total_channels: specification.slab().total_channels(),
65            phase: match pass {
66                SpectralOperatorPass::InitialMajor
67                    if specification.is_initial_certified_zero(pass) =>
68                {
69                    BandPhase::InitialZero
70                }
71                SpectralOperatorPass::InitialMajor => BandPhase::Full,
72                SpectralOperatorPass::ResidualRefresh => BandPhase::Residual,
73            },
74            support: BandSupport {
75                native: 0..0,
76                model: Vec::new(),
77            },
78        })
79    }

81    /// Observe each full selected native block once during preparation. The
82    /// caller retains source/run coverage; this records only band dependencies.
83    pub fn observe(
84        &mut self,
85        block: &NativeBlock,
86        output_hz: &[f64],
87    ) -> Result<(), SpectralOperatorError> {
88        Self::observe_all(std::slice::from_mut(self), block, output_hz).map(|_| ())
89    }

91    /// Discover every band's row support with one native-pair sweep per distinct
92    /// consecutive spectral row. The returned count measures actual pair visits,
93    /// independent of the band count.
94    /// Only one range per band is temporary; model support is merged directly
95    /// into the retained sorted channel lists, without per-row/band vectors.
96    pub fn observe_all(
97        bands: &mut [Self],
98        block: &NativeBlock,
99        output_hz: &[f64],
100    ) -> Result<u64, SpectralOperatorError> {
101        if bands.is_empty()
102            || bands
103                .iter()
104                .any(|band| output_hz.len() != band.total_channels)
105            || bands
106                .windows(2)
107                .any(|pair| pair[0].core.end > pair[1].core.start)
108        {
109            return Err(SpectralOperatorError::ProblemMismatch);
110        }
111        block
112            .validate_shape(block.metadata.len(), block.channels, block.correlations)
113            .map_err(|_| SpectralOperatorError::InvalidSample)?;
114        if block.channels < 2 {
115            return Err(SpectralOperatorError::InvalidSample);
116        }
117        let mut ranges = vec![block.channels..0; bands.len()];
118        let mut pairs = 0_u64;
119        for (row, metadata) in block.metadata.iter().enumerate() {
120            let native_hz = &block.frequencies_hz[row * block.channels..(row + 1) * block.channels];
121            if row > 0
122                && metadata.original_pair_hz == block.metadata[row - 1].original_pair_hz
123                && native_hz
124                    == &block.frequencies_hz[(row - 1) * block.channels..row * block.channels]
125            {
126                continue;
127            }
128            ranges.fill(block.channels..0);
129            let grid = CasaLinearGrid::compile(
130                output_hz,
131                metadata.original_pair_hz[0],
132                metadata.original_pair_hz[1],
133            )
134            .ok_or(SpectralOperatorError::MissingRowSpectralGeometry)?;
135            let mut next_fine = 0;
136            for (left, pair) in native_hz.windows(2).enumerate() {
137                pairs = pairs
138                    .checked_add(1)
139                    .ok_or(SpectralOperatorError::ResidencyOverflow)?;
140                for fine in grid
141                    .samples_for_pair(&mut next_fine, pair[0], pair[1])
142                    .map_err(|_| SpectralOperatorError::InvalidSample)?
143                {
144                    let channel = fine.output_channel();
145                    let band = bands.partition_point(|band| band.core.end <= channel);
146                    if band < bands.len() && bands[band].core.contains(&channel) {
147                        ranges[band].start = ranges[band].start.min(left);
148                        ranges[band].end = ranges[band].end.max(left + 2);
149                    }
150                }
151            }
152            for (band, native) in bands.iter_mut().zip(&ranges) {
153                if native.is_empty() {
154                    continue;
155                }
156                band.support.native = if band.support.native.is_empty() {
157                    native.clone()
158                } else {
159                    band.support.native.start.min(native.start)
160                        ..band.support.native.end.max(native.end)
161                };
162                // Preserve the complete contiguous native closure, including
163                // interior samples that did not themselves emit a fine channel.
164                for &frequency in &native_hz[native.clone()] {
165                    for term in casa_linear_prediction_terms(
166                        output_hz,
167                        frequency,
168                        metadata.original_pair_hz,
169                    )
170                    .map_err(|_| SpectralOperatorError::InvalidSample)?
171                    {
172                        let channel = term.output_channel() as usize;
173                        if let Err(position) = band.support.model.binary_search(&channel) {
174                            if band.support.model.len() == band.support.model.capacity() {
175                                let capacity = band
176                                    .support
177                                    .model
178                                    .capacity()
179                                    .saturating_mul(2)
180                                    .max(4)
181                                    .min(band.total_channels);
182                                band.support
183                                    .model
184                                    .try_reserve_exact(capacity - band.support.model.len())
185                                    .map_err(|_| SpectralOperatorError::ResidencyOverflow)?;
186                            }
187                            band.support.model.insert(position, channel);
188                        }
189                    }
190                }
191            }
192        }
193        Ok(pairs)
194    }

196    /// Selected native ordinals required across all observed rows.
197    pub fn native_range(&self) -> Range<usize> {
198        self.support.native.clone()
199    }
200
201    /// Exclusive output channels.
202    pub fn core(&self) -> Range<usize> {
203        self.core.clone()
204    }

206    /// Preparation retains at most one entry per output model channel plus
207    /// one row-local range. Capacity growth is capped at this shape bound.
208    /// Runtime charges this before observing row-dependent support.
209    pub fn preparation_metadata_bytes(&self) -> Result<usize, SpectralOperatorError> {
210        self.total_channels
211            .checked_mul(size_of::<usize>())
212            .and_then(|bytes| bytes.checked_add(size_of::<Self>() + size_of::<Range<usize>>()))
213            .ok_or(SpectralOperatorError::ResidencyOverflow)
214    }

216    /// Initial-phase admission may omit forward grids only for this compiled
217    /// certified-empty phase. A seeded model needs its own support-aware plan.
218    pub fn is_certified_empty_initial(&self) -> bool {
219        self.phase == BandPhase::InitialZero
220    }
221
222    /// Reuse fixed dependencies for a later immutable model epoch.
223    pub fn residual_refresh(&self) -> Self {
224        Self {
225            phase: BandPhase::Residual,
226            ..self.clone()
227        }
228    }

230    /// Allocate only after runtime admission. Recycled FFT state retains the
231    /// same spatial geometry; model reads stay within one admitted plane.
232    pub fn prepare<'a>(
233        self,
234        generation: &'a ModelGeneration,
235        fft: Option<PreparedFft>,
236    ) -> Result<EpochBand<'a>, SpectralOperatorError> {
237        self.memory()?;
238        let fft = match fft {
239            Some(fft) if fft.shape() == self.geometry.grid_shape => fft,
240            Some(_) => return Err(SpectralOperatorError::UnsupportedGeometry),
241            None => PreparedFft::new(
242                self.geometry.grid_shape,
243                fft_resident_complex_values_for_shape(self.geometry.grid_shape)?,
244            )?,
245        };
246        EpochBand::prepare(
247            BandWorkspace::new(
248                self.geometry,
249                self.core,
250                self.support.model,
251                fft,
252                self.phase,
253            ),
254            generation,
255            self.support.native,
256        )
257    }

310impl<'a> EpochBand<'a> {
311    fn prepare(
312        mut workspace: BandWorkspace,
313        generation: &'a ModelGeneration,
314        native_range: Range<usize>,
315    ) -> Result<Self, SpectralOperatorError> {
316        let shape = generation.shape();
317        if shape.domains().len() != 1
318            || shape.domains()[0].pixels() != workspace.geometry.image_shape
319            || shape.polarizations() != 1
320            || !matches!(
321                shape.coefficient_space().basis(),
322                ReconstructionBasis::ChannelLocal { .. }
323            )
324            || workspace.core.end > shape.coefficients()
325            || workspace
326                .model_channels
327                .iter()
328                .any(|&channel| channel >= shape.coefficients())
329        {
330            return Err(SpectralOperatorError::ModelShape);
331        }
332        if workspace.phase == BandPhase::InitialZero
333            && generation.origin() != crate::ModelGenerationOrigin::Empty
334        {
335            return Err(SpectralOperatorError::ReusableNormalStateMismatch);
336        }
337        let width = workspace.geometry.image_shape[0];
338        for plane in 0..workspace.forward.len_of(Axis(0)) {
339            // The existing storage owner enforces its admitted window. Only one
340            // canonical y/x plane is live alongside the prepared support grids.
341            let samples = generation.read_plane(0, workspace.model_channels[plane], 0)?;
342            workspace.prepare_plane(plane, |x, y| {
343                let sample = samples[y * width + x];
344                if sample.support() == ModelSupport::Valid {
345                    Complex64::new(sample.value().value(), 0.0)
346                } else {
347                    Complex64::default()
348                }
349            })?;
350        }
351        Ok(Self {
352            generation,
353            workspace,
354            native_range,
355        })
356    }
357
358    /// Evaluate this block using borrowed subwindows, keeping all writable
359    /// grids exclusively within this epoch job.
360    pub fn consume(
361        &mut self,
362        block: &NativeBlock,
363        layout: &NativeLayout,
364        window: Range<usize>,
365        output_hz: &[f64],
366        polarization: &PolarizationOperator,
367    ) -> Result<(), SpectralOperatorError> {
368        if output_hz.len() != self.generation.shape().coefficients() {
369            return Err(SpectralOperatorError::ProblemMismatch);
370        }
371        self.workspace.consume_block(
372            block,
373            layout,
374            window,
375            self.native_range.clone(),
376            output_hz,
377            polarization,
378        )
379    }
380
381    /// Prepared model planes and actual forward FFTs. Exact-zero planes are
382    /// recognized during grid population, not by a separate verification pass.
383    pub fn model_plane_counts(&self) -> (usize, usize) {
384        (
385            self.workspace.forward_nonzero.len(),
386            self.workspace
387                .forward_nonzero
388                .iter()
389                .filter(|&&active| active)
390                .count(),
391        )
392    }
393
394    /// Join into owned normal fields for the same model and return reusable FFT
395    /// state. Complete source coverage is the runtime owner's separate duty.
396    pub fn complete(
397        self,
398        expected: &ModelGeneration,
399    ) -> Result<(BandResult, PreparedFft), SpectralOperatorError> {
400        if !std::ptr::eq(self.generation, expected) {
401            return Err(SpectralOperatorError::ModelMismatch);
402        }
403        let channels = self.generation.shape().coefficients();
404        let model = self.generation.generation_id();
405        let (images, fft) = self.workspace.finish_images()?;
406        let result = if images.phase == BandPhase::Residual {
407            BandResult::Residual(CubeResidual {
408                shape: images.shape,
409                core: images.core,
410                total_channels: channels,
411                model,
412                values: images.residual.into_boxed_slice(),
413            })
414        } else {
415            BandResult::Initial(SpectralOperatorPrimitives::from_cube_band(
416                images, channels, model,
417            )?)
418        };
419        Ok((result, fft))
420    }
421}

445/// Exact native-pair and model support for one row and output band. Native
446/// ordinals address the supplied row, not physical MS channel numbers.
447#[derive(Debug, Clone, PartialEq)]
448struct BandSupport {
449    native: Range<usize>,
450    model: Vec<usize>,
451}
452
453impl BandSupport {
454    #[cfg(test)]
455    fn compile(
456        output_hz: &[f64],
457        core: Range<usize>,
458        native_hz: &[f64],
459        original_pair_hz: [f64; 2],
460    ) -> Result<Self, SpectralOperatorError> {
461        let native = Self::native_window(output_hz, core, native_hz, original_pair_hz)?;
462        let mut model = Vec::new();
463        for frequency in &native_hz[native.clone()] {
464            for term in casa_linear_prediction_terms(output_hz, *frequency, original_pair_hz)
465                .map_err(|_| SpectralOperatorError::InvalidSample)?
466            {
467                let channel = term.output_channel() as usize;
468                if let Err(position) = model.binary_search(&channel) {
469                    model.insert(position, channel);
470                }
471            }
472        }
473        Ok(Self { native, model })
474    }
475
476    /// Recomputed over the shared decoded window for each row. This narrows
477    /// row-varying support without storing a per-row/band directory or predicting
478    /// unrelated native channels from a conservatively wider stored window.
479    fn native_window(
480        output_hz: &[f64],
481        core: Range<usize>,
482        native_hz: &[f64],
483        original_pair_hz: [f64; 2],
484    ) -> Result<Range<usize>, SpectralOperatorError> {
485        if core.is_empty() || core.end > output_hz.len() || native_hz.len() < 2 {
486            return Err(SpectralOperatorError::InvalidSample);
487        }
488        let grid = CasaLinearGrid::compile(output_hz, original_pair_hz[0], original_pair_hz[1])
489            .ok_or(SpectralOperatorError::MissingRowSpectralGeometry)?;
490        let mut next_fine = 0;
491        let mut native = native_hz.len()..0;
492        for (left, pair) in native_hz.windows(2).enumerate() {
493            for fine in grid
494                .samples_for_pair(&mut next_fine, pair[0], pair[1])
495                .map_err(|_| SpectralOperatorError::InvalidSample)?
496            {
497                if core.contains(&fine.output_channel()) {
498                    native.start = native.start.min(left);
499                    native.end = native.end.max(left + 2);
500                }
501            }
502        }
503        if native.is_empty() {
504            native = 0..0;
505        }
506        Ok(native)
507    }
508
509    /// Union during the one preparation traversal. Only band-sized metadata is
510    /// retained; no payload is reread after preparation to discover dependencies.
511    #[cfg(test)]
512    fn include(&mut self, row: Self) {
513        if !row.native.is_empty() {
514            self.native = if self.native.is_empty() {
515                row.native
516            } else {
517                self.native.start.min(row.native.start)..self.native.end.max(row.native.end)
518            };
519        }
520        for channel in row.model {
521            if let Err(position) = self.model.binary_search(&channel) {
522                self.model.insert(position, channel);
523            }
524        }
525    }
526}

528/// Borrowed compact native row, including a selected native-channel window.
529///
530/// Values/weights retain Complex64/f64 precision, flags occupy one byte each.
531/// `flags` have already applied finite-value/input acceptance; `weight_flags`
532/// are the native weight-group/parallel-hand/row flags used by nearest weight
533/// transfer. Keeping the latter distinct preserves CASA's endpoint behavior.
534/// Correlation identity belongs to the block's shared polarization operator.
535struct VisibilityRow<'a> {
536    address: SelectedSampleAddress,
537    uvw_m: [f64; 3],
538    phase_shift_m: f64,
539    original_pair_hz: [f64; 2],
540    channels: &'a [u32],
541    frequencies_hz: &'a [f64],
542    values: ArrayView2<'a, Complex64>,
543    weights: ArrayView2<'a, f64>,
544    flags: ArrayView2<'a, bool>,
545    weight_flags: ArrayView2<'a, bool>,
546}
547
548impl NativeBlock {
549    /// Borrow a decoded native window directly; no intermediate sample objects.
550    /// `channels` identifies the store window within the shared selected layout.
551    fn row<'a>(
552        &'a self,
553        layout: &'a NativeLayout,
554        row: usize,
555        channels: Range<usize>,
556    ) -> Result<VisibilityRow<'a>, SpectralOperatorError> {
557        if row >= self.metadata.len()
558            || channels.len() != self.channels
559            || channels.end > layout.channels.len()
560            || self.correlations != layout.correlations.len()
561        {
562            return Err(SpectralOperatorError::InvalidSample);
563        }
564        self.validate_shape(self.metadata.len(), self.channels, self.correlations)
565            .map_err(|_| SpectralOperatorError::InvalidSample)?;
566        let metadata = self.metadata[row];
567        let cells = row * self.channels..(row + 1) * self.channels;
568        let samples = cells.start * self.correlations..cells.end * self.correlations;
569        let shape = (self.channels, self.correlations);
570        Ok(VisibilityRow {
571            address: SelectedSampleAddress {
572                physical_row: metadata.physical_row,
573                ..layout.address
574            },
575            uvw_m: metadata.uvw_m,
576            phase_shift_m: metadata.phase_shift_m,
577            original_pair_hz: metadata.original_pair_hz,
578            channels: &layout.channels[channels],
579            frequencies_hz: &self.frequencies_hz[cells],
580            values: ArrayView2::from_shape(shape, &self.values[samples.clone()]).unwrap(),
581            weights: ArrayView2::from_shape(shape, &self.weights[samples.clone()]).unwrap(),
582            flags: ArrayView2::from_shape(shape, &self.flags[samples.clone()]).unwrap(),
583            weight_flags: ArrayView2::from_shape(shape, &self.weight_flags[samples]).unwrap(),
584        })
585    }
586}
587
588impl<'a> VisibilityRow<'a> {
589    fn window(self, channels: Range<usize>) -> Result<Self, SpectralOperatorError> {
590        if channels.len() < 2 || channels.end > self.channels.len() {
591            return Err(SpectralOperatorError::InvalidSample);
592        }
593        Ok(Self {
594            channels: &self.channels[channels.clone()],
595            frequencies_hz: &self.frequencies_hz[channels.clone()],
596            values: self.values.slice_move(s![channels.clone(), ..]),
597            weights: self.weights.slice_move(s![channels.clone(), ..]),
598            flags: self.flags.slice_move(s![channels.clone(), ..]),
599            weight_flags: self.weight_flags.slice_move(s![channels, ..]),
600            ..self
601        })
602    }
603
604    fn validate(&self, correlations: usize) -> Result<(), SpectralOperatorError> {
605        let shape = (self.channels.len(), correlations);
606        if self.channels.len() < 2
607            || self.frequencies_hz.len() != shape.0
608            || correlations == 0
609            || correlations > 4
610            || self.values.dim() != shape
611            || self.weights.dim() != shape
612            || self.flags.dim() != shape
613            || self.weight_flags.dim() != shape
614            || !self.values.is_standard_layout()
615            || !self.weights.is_standard_layout()
616            || !self.flags.is_standard_layout()
617            || !self.weight_flags.is_standard_layout()
618        {
619            return Err(SpectralOperatorError::InvalidSample);
620        }
621        Ok(())
622    }
623
624    fn geometry(&self) -> NativeRowSpectralGeometry {
625        NativeRowSpectralGeometry {
626            channels: self.channels.len(),
627            first: (self.channels[0], self.frequencies_hz[0]),
628            second: Some((self.channels[1], self.frequencies_hz[1])),
629            lattice_first_pair_hz: Some(self.original_pair_hz),
630        }
631    }
632}

652    sum_weight_error: Vec<f64>,
653    mapped: Vec<u64>,
654}

656impl BandWorkspace {
657    fn new(
658        geometry: SpectralOperatorGeometry,
659        core: Range<usize>,
660        model_channels: Vec<usize>,
661        fft: PreparedFft,
662        phase: BandPhase,
663    ) -> Self {
664        let shape = (core.len(), geometry.grid_shape[0], geometry.grid_shape[1]);
665        let normal = if phase == BandPhase::Residual {
666            (0, shape.1, shape.2)
667        } else {
668            shape
669        };
670        let residual = if phase == BandPhase::InitialZero {
671            (0, shape.1, shape.2)
672        } else {
673            shape
674        };
675        let normal_planes = normal.0;
676        Self {
677            phase,
678            forward_nonzero: vec![
679                false;
680                if phase == BandPhase::InitialZero {
681                    0
682                } else {
683                    model_channels.len()
684                }
685            ],
686            forward: Array3::zeros((
687                if phase == BandPhase::InitialZero {
688                    0
689                } else {
690                    model_channels.len()
691                },
692                shape.1,
693                shape.2,
694            )),
695            dirty: Array3::zeros(normal),
696            dirty_error: Array3::zeros(normal),
697            residual: Array3::zeros(residual),
698            residual_error: Array3::zeros(residual),
699            psf: Array3::zeros(normal),
700            psf_error: Array3::zeros(normal),
701            sum_weight: vec![0.0; normal_planes],
702            sum_weight_error: vec![0.0; normal_planes],
703            mapped: vec![0; normal_planes],
704            convolution: StandardConvolution::new(&geometry),
705            geometry,
706            core,
707            model_channels,
708            fft,
709        }
710    }
711
712    #[cfg(test)]
713    fn prepare_model(
714        &mut self,
715        model: ArrayView3<'_, Complex64>,
716    ) -> Result<(), SpectralOperatorError> {
717        if model.shape()[1..] != self.geometry.image_shape {
718            return Err(SpectralOperatorError::ModelShape);
719        }
720        for plane in 0..self.model_channels.len() {
721            let channel = self.model_channels[plane];
722            if channel >= model.len_of(Axis(0)) {
723                return Err(SpectralOperatorError::IncompleteSpectralHalo);
724            }
725            self.prepare_plane(plane, |x, y| model[(channel, x, y)])?;
726        }
727        Ok(())
728    }
729
730    fn prepare_plane(
731        &mut self,
732        plane: usize,
733        value: impl Fn(usize, usize) -> Complex64,
734    ) -> Result<bool, SpectralOperatorError> {
735        let mut grid = self.forward.index_axis_mut(Axis(0), plane);
736        grid.fill(Complex64::default());
737        let mut nonzero = false;
738        for y in 0..self.geometry.image_shape[1] {
739            for x in 0..self.geometry.image_shape[0] {
740                let corrected = value(x, y) * self.convolution.image_correction(x, y);
741                nonzero |= corrected != Complex64::default();
742                grid[(
743                    self.geometry.image_blc[0] + x,
744                    self.geometry.image_blc[1] + y,
745                )] = corrected;
746            }
747        }
748        self.forward_nonzero[plane] = nonzero;
749        if !nonzero {
750            return Ok(false);
751        }
752        self.fft.transform(&mut grid, false);
753        if grid
754            .iter()
755            .any(|value| !value.re.is_finite() || !value.im.is_finite())
756        {
757            return Err(SpectralOperatorError::GeneratedNonfinite);
758        }
759        Ok(true)
760    }
761
762    fn finish_images(self) -> Result<(BandImages, PreparedFft), SpectralOperatorError> {
763        let Self {
764            phase,
765            geometry,
766            core,
767            convolution,
768            mut fft,
769            dirty,
770            residual,
771            psf,
772            sum_weight,
773            mapped,
774            forward,
775            forward_nonzero,
776            dirty_error,
777            residual_error,
778            psf_error,
779            model_channels,
780            sum_weight_error,
781        } = self;
782        // Prediction and compensation are dead before image allocations begin.
783        drop((
784            forward,
785            forward_nonzero,
786            dirty_error,
787            residual_error,
788            psf_error,
789            model_channels,
790            sum_weight_error,
791        ));
792        let mut image = |mut grids: Array3<Complex64>| {
793            let mut values = Vec::with_capacity(
794                grids.len_of(Axis(0)) * geometry.image_shape[0] * geometry.image_shape[1],
795            );
796            for mut grid in grids.axis_iter_mut(Axis(0)) {
797                fft.transform(&mut grid, true);
798                append_image_plane(
799                    grid.view(),
800                    &geometry,
801                    &mut values,
802                    |x, y| convolution.image_correction(x, y),
803                    false,
804                )?;
805            }
806            Ok::<_, SpectralOperatorError>(values)
807        };
808        let dirty = image(dirty)?;
809        let psf = image(psf)?;
810        let residual = image(residual)?;
811        Ok((
812            BandImages {
813                phase,
814                shape: geometry.image_shape,
815                core,
816                dirty,
817                residual,
818                psf,
819                sum_weight,
820                mapped,
821            },
822            fft,
823        ))
824    }
825
826    fn predict_native(
827        &self,
828        row: &VisibilityRow<'_>,
829        frequency_hz: f64,
830        output_hz: &[f64],
831        polarization: &PolarizationOperator,
832    ) -> Result<SmallVec<[Complex64; 4]>, SpectralOperatorError> {
833        if self.phase == BandPhase::InitialZero {
834            return Ok(std::iter::repeat_n(
835                Complex64::default(),
836                polarization.correlations().len(),
837            )
838            .collect());
839        }
840        let mut predicted = Complex64::default();
841        for term in casa_linear_prediction_terms(output_hz, frequency_hz, row.original_pair_hz)
842            .map_err(|_| SpectralOperatorError::InvalidSample)?
843        {
844            let plane = self
845                .model_channels
846                .binary_search(&(term.output_channel() as usize))
847                .map_err(|_| SpectralOperatorError::IncompleteSpectralHalo)?;
848            if !self.forward_nonzero[plane] {
849                continue;
850            }
851            let frequency = term.evaluation_frequency_hz();
852            let wavelength_scale = frequency / SPEED_OF_LIGHT_M_PER_S;
853            if let Some(taps) = self.convolution.taps([
854                row.uvw_m[0] * wavelength_scale,
855                row.uvw_m[1] * wavelength_scale,
856            ]) {
857                predicted += self
858                    .convolution
859                    .degrid(&self.forward.index_axis(Axis(0), plane), taps)
860                    * phase(row.phase_shift_m, frequency).conj()
861                    * term.factor();
862            }
863        }
864        polarization
865            .predict(&[predicted])
866            .map_err(|_| SpectralOperatorError::GeneratedNonfinite)
867    }
868
869    fn begin_row<'a>(
870        &'a mut self,
871        row: VisibilityRow<'a>,
872        output_hz: &'a [f64],
873        polarization: &'a PolarizationOperator,
874    ) -> Result<RowAccumulator<'a>, SpectralOperatorError> {
875        row.validate(polarization.correlations().len())?;
876        if polarization.model_coordinates() != [PolarizationCoordinate::StokesI] {
877            return Err(SpectralOperatorError::InvalidSample);
878        }
879        let output = CasaLinearOutputGrid::compile(output_hz)
880            .ok_or(SpectralOperatorError::MissingRowSpectralGeometry)?;
881        Ok(RowAccumulator {
882            band: self,
883            row,
884            output_hz,
885            output,
886            polarization,
887            cursor: CasaLinearRowCursor::new(),
888            next: 0,
889            previous_prediction: SmallVec::new(),
890        })
891    }
892
893    fn grid_sample(
894        &mut self,
895        fine: CasaLinearSample,
896        row: &VisibilityRow<'_>,
897        observed: Complex64,
898        predicted: Complex64,
899        weight: f64,
900    ) -> Result<(), SpectralOperatorError> {
901        let plane = fine.output_channel() - self.core.start;
902        if self.phase != BandPhase::Residual {
903            self.mapped[plane] = self.mapped[plane]
904                .checked_add(1)
905                .ok_or(SpectralOperatorError::CoverageOverflow)?;
906        }
907        if weight == 0.0 {
908            return Ok(());
909        }
910        let wavelength_scale = fine.frequency_hz() / SPEED_OF_LIGHT_M_PER_S;
911        let Some(taps) = self.convolution.taps([
912            row.uvw_m[0] * wavelength_scale,
913            row.uvw_m[1] * wavelength_scale,
914        ]) else {
915            return Ok(());
916        };
917        let rotation = phase(row.phase_shift_m, fine.frequency_hz());
918        for (grid, error, value) in [
919            (
920                &mut self.dirty,
921                &mut self.dirty_error,
922                observed * rotation * weight,
923            ),
924            (
925                &mut self.residual,
926                &mut self.residual_error,
927                (observed - predicted) * rotation * weight,
928            ),
929            (
930                &mut self.psf,
931                &mut self.psf_error,
932                Complex64::new(weight, 0.0),
933            ),
934        ] {
935            if grid.is_empty() {
936                continue;
937            }
938            self.convolution.grid_compensated(
939                &mut grid.index_axis_mut(Axis(0), plane),
940                &mut error.index_axis_mut(Axis(0), plane),
941                taps,
942                value,
943            );
944        }
945        if self.phase != BandPhase::Residual {
946            let increment = weight - self.sum_weight_error[plane];
947            let updated = self.sum_weight[plane] + increment;
948            self.sum_weight_error[plane] = (updated - self.sum_weight[plane]) - increment;
949            self.sum_weight[plane] = updated;
950        }
951        Ok(())
952    }
953
954    fn consume_block(
955        &mut self,
956        block: &NativeBlock,
957        layout: &NativeLayout,
958        window: Range<usize>,
959        native_range: Range<usize>,
960        output_hz: &[f64],
961        polarization: &PolarizationOperator,
962    ) -> Result<(), SpectralOperatorError> {
963        if native_range.is_empty() {
964            return Ok(());
965        }
966        if native_range.start < window.start || native_range.end > window.end {
967            return Err(SpectralOperatorError::IncompleteSpectralHalo);
968        }
969        let local = native_range.start - window.start..native_range.end - window.start;
970        for row in 0..block.metadata.len() {
971            let row = block
972                .row(layout, row, window.clone())?
973                .window(local.clone())?;
974            let native = BandSupport::native_window(
975                output_hz,
976                self.core.clone(),
977                row.frequencies_hz,
978                row.original_pair_hz,
979            )?;
980            if native.is_empty() {
981                continue;
982            }
983            let channels = native.len();
984            let mut accumulator = self.begin_row(row.window(native)?, output_hz, polarization)?;
985            accumulator.push(0..channels)?;
986            accumulator.finish()?;
987        }
988        Ok(())
989    }
990}
991
992fn phase(shift_m: f64, frequency_hz: f64) -> Complex64 {
993    Complex64::from_polar(
994        1.0,
995        std::f64::consts::TAU * shift_m * frequency_hz / SPEED_OF_LIGHT_M_PER_S,
996    )
997}

999/// A row cursor survives chunk boundaries; only the previous prediction is
1000/// retained. All other input accesses are borrowed from the compact row.
1001struct RowAccumulator<'a> {
1002    band: &'a mut BandWorkspace,
1003    row: VisibilityRow<'a>,
1004    output_hz: &'a [f64],
1005    output: CasaLinearOutputGrid,
1006    polarization: &'a PolarizationOperator,
1007    cursor: CasaLinearRowCursor,
1008    next: usize,
1009    previous_prediction: SmallVec<[Complex64; 4]>,
1010}
1011
1012impl RowAccumulator<'_> {
1013    fn push(&mut self, channels: Range<usize>) -> Result<(), SpectralOperatorError> {
1014        if channels.start != self.next || channels.end > self.row.channels.len() {
1015            return Err(SpectralOperatorError::IncompleteCoverage);
1016        }
1017        for channel in channels {
1018            let mut address = self.row.address;
1019            address.channel_index = self.row.channels[channel];
1020            let prediction = self.band.predict_native(
1021                &self.row,
1022                self.row.frequencies_hz[channel],
1023                self.output_hz,
1024                self.polarization,
1025            )?;
1026            let fine_points = self
1027                .cursor
1028                .push(
1029                    address,
1030                    self.row.geometry(),
1031                    self.row.frequencies_hz[channel],
1032                    self.output,
1033                )
1034                .map_err(|_| SpectralOperatorError::IncompleteCoverage)?;
1035            if let Some(fine_points) = fine_points {
1036                for fine in fine_points {
1037                    if !self.band.core.contains(&fine.output_channel()) {
1038                        continue;
1039                    }
1040                    let nearest = if fine.nearest_is_right() {
1041                        channel
1042                    } else {
1043                        channel - 1
1044                    };
1045                    let mut observed = SmallVec::<[Complex64; 4]>::new();
1046                    let mut predicted = SmallVec::<[Complex64; 4]>::new();
1047                    let mut weights = SmallVec::<[f64; 4]>::new();
1048                    let mut flags = SmallVec::<[bool; 4]>::new();
1049                    for correlation in 0..prediction.len() {
1050                        observed.push(interpolate_complex_pair(
1051                            self.row.values[(channel - 1, correlation)],
1052                            self.row.values[(channel, correlation)],
1053                            fine.factors(),
1054                        ));
1055                        predicted.push(interpolate_complex_pair(
1056                            self.previous_prediction[correlation],
1057                            prediction[correlation],
1058                            fine.factors(),
1059                        ));
1060                        weights.push(self.row.weights[(nearest, correlation)]);
1061                        flags.push(
1062                            fine.linear_flag(
1063                                self.row.flags[(channel - 1, correlation)],
1064                                self.row.flags[(channel, correlation)],
1065                            ) || self.row.weight_flags[(nearest, correlation)],
1066                        );
1067                    }
1068                    // Interpolate observed and predicted separately, then apply
1069                    // polarization/weights, then subtract in the existing order.
1070                    let observed_adjoint = self
1071                        .polarization
1072                        .weighted_adjoint(&observed, &weights, &flags)
1073                        .map_err(|_| SpectralOperatorError::InvalidSample)?[0];
1074                    let predicted_adjoint = self
1075                        .polarization
1076                        .weighted_adjoint(&predicted, &weights, &flags)
1077                        .map_err(|_| SpectralOperatorError::InvalidSample)?[0];
1078                    let weight = polarization_diagonal(self.polarization, &weights, &flags)[0];
1079                    let direct = (self.polarization.feed_basis() == crate::FeedBasis::Stokes)
1080                        .then(|| {
1081                            self.polarization
1082                                .coefficients()
1083                                .iter()
1084                                .position(|value| *value == Complex64::new(1.0, 0.0))
1085                        })
1086                        .flatten();
1087                    let normalize = |adjoint, values: &[Complex64]| {
1088                        if weight == 0.0 {
1089                            Complex64::default()
1090                        } else {
1091                            direct.map_or_else(|| adjoint / weight, |index| values[index])
1092                        }
1093                    };
1094                    self.band.grid_sample(
1095                        fine,
1096                        &self.row,
1097                        normalize(observed_adjoint, &observed),
1098                        normalize(predicted_adjoint, &predicted),
1099                        weight,
1100                    )?;
1101                }
1102            }
1103            self.previous_prediction = prediction;
1104            self.next += 1;
1105        }
1106        Ok(())
1107    }
1108
1109    fn finish(mut self) -> Result<(), SpectralOperatorError> {
1110        if self.next != self.row.channels.len() {
1111            return Err(SpectralOperatorError::IncompleteCoverage);
1112        }
1113        self.cursor
1114            .finish()
1115            .map_err(|_| SpectralOperatorError::IncompleteCoverage)
1116    }
1117}
