2//! What CASA clears from a MeasurementSet before it writes a scratch
3//! column, so that the column casa-rs writes is the one CASA reads (CASA
4//! 6.7.5).
5//!
6//! - **Cached sort keywords.** CASA's iterators may cache a pre-sorted
7//!   reference table of MAIN and record it in MAIN's keywords as
8//!   `SORTED_TABLE`, with the columns it is sorted by as `SORT_COLUMNS`
9//!   (`MSIter2`). A sorted table made before a column was added does not
10//!   see it, so adding `MODEL_DATA` or `CORRECTED_DATA` removes both
11//!   keywords (`VisSetUtil::addScrCols`; the table itself is left, as CASA
12//!   leaves it). The MS definition's own `SORTED_TABLES` is untouched.
13//! - **The virtual model.** CASA can hold a field's model as a recipe
14//!   instead of a column: `setjy` (`usescratch=False`) and `tclean`
15//!   (`savemodel='virtual'`) write one, and CASA's iterator prefers it to
16//!   `MODEL_DATA` (`VisibilityIteratorImpl2::visibilityModel`). A task that
17//!   writes `MODEL_DATA` makes the column "the exclusive origin for model
18//!   visibilities", whether or not it adds it: it clears every field's
19//!   virtual model first. `addScrCols` does so when it adds the column
20//!   (`remOTFModel`, `VisModelData::clearModel` of
21//!   `synthesis/TransformMachines/VisModelData.cc`), and the imagers do so
22//!   whenever they write it (`SynthesisImagerVi2::selectData`,
23//!   `Imager::setjy` and `ft`, `refim::VisModelData::clearModel` of
24//!   `TransformMachines2`). The two `clearModel` bodies are the same.
25//!
26//! A field's virtual model is a keyword `definedmodel_field_<id>` naming a
27//! model key, in SOURCE's keywords when SOURCE has rows and MAIN's
28//! otherwise. Under SOURCE the model key is a keyword holding the SOURCE
29//! row whose `SOURCE_MODEL` cell holds the model record; under MAIN it is a
30//! keyword holding the record itself. Clearing a field removes both
31//! keywords wherever they are and empties the `SOURCE_MODEL` cell, which
32//! is what CASA leaves (measured with `setjy` then `delmod` on CASA 6.7.5).
33//! A pointer under SOURCE naming a key held in MAIN is removed with the
34//! key, where CASA keeps the pointer; either way the field has no model.
35//!
36//! `clearModel` also deletes the model images a gridder's record names
37//! (`ft_<k>.container.diskimage`), wherever they lie, and every `FT_MODEL*`
38//! table in the MeasurementSet and its SOURCE, where `tclean` writes them.
39//! casa-rs deletes only the `FT_MODEL*` tables, so that a path in a record
40//! never deletes anything else. A multi-MS is cleared as one table, not
41//! part by part as CASA clears it.

57/// Clear what CASA clears before it writes scratch column `column`
58/// (`MODEL_DATA` or `CORRECTED_DATA`) to `ms`, adding it when `adds`:
59/// for an added column the cached sort keywords, and for `MODEL_DATA`,
60/// added or not, every field's virtual model, as the module states. Every
61/// path that adds either column to an existing MeasurementSet, or writes
62/// `MODEL_DATA`, calls it before it changes the column.
63///
64/// What is cleared is found first, and a table with nothing to clear is
65/// not changed. A MeasurementSet on disk then has SOURCE and MAIN changed
66/// and written in turn, each under casacore's write lock on it: SOURCE in
67/// full as [`MeasurementSet::save`] writes it, MAIN's metadata alone. A
68/// table opened with locking is written under its own lock, held already
69/// or taken and released here; one opened without, under a
70/// [`TableWriteLock`] taken and released here, so a caller holds MAIN's
71/// write lock, if at all, through its handle, never through such a guard.
72/// A lock is waited for as casacore waits, except SOURCE's while the
73/// caller holds MAIN's write lock: one attempt is then made, as a save
74/// never waits for one table while it holds another, where CASA waits
75/// (casa-rs holders do not yet release a lock on request, #694). The
76/// `FT_MODEL*` model images are deleted last and, as CASA deletes them, at
77/// best: one that is held open or cannot be deleted is logged and left.
78///
79/// # Errors
80///
81/// [`MsError::InvalidInput`] for `DATA`, which is no scratch column. A
82/// [`TableError::LockFailed`] when SOURCE's lock is held elsewhere while
83/// the caller holds MAIN's write lock, and any error reading the fields
84/// or locking and writing SOURCE or MAIN. A table whose write fails is
85/// reloaded as it is on disk, holding whatever of the write reached it,
86/// as after an interrupted casacore write; when SOURCE's fails, MAIN is
87/// not changed.
88pub fn prepare_write(
89    ms: &mut MeasurementSet,
90    column: VisibilityDataColumn,
91    adds: bool,
92) -> MsResult<()> {
93    prepare(ms, column, adds, None)
94}

96/// [`prepare_write`] for a caller that holds MAIN's write lock through
97/// `main_lock`, a [`TableWriteLock`] on the MAIN of `ms` (opened without
98/// locking), as a writer of several mutation sessions in turn holds it:
99/// MAIN's metadata is written under it, the write recorded so that the
100/// caller's release publishes it, and SOURCE's lock is tried once, as while
101/// a caller holds MAIN's through its handle.
102///
103/// # Errors
104///
105/// As [`prepare_write`]; a MAIN whose write fails is reloaded as it is on
106/// disk, and the caller's lock publishes what reached it.
107pub fn prepare_write_holding(
108    ms: &mut MeasurementSet,
109    column: VisibilityDataColumn,
110    adds: bool,
111    main_lock: &mut TableWriteLock,
112) -> MsResult<()> {
113    prepare(ms, column, adds, Some(main_lock))
114}

116fn prepare(
117    ms: &mut MeasurementSet,
118    column: VisibilityDataColumn,
119    adds: bool,
120    main_lock: Option<&mut TableWriteLock>,
121) -> MsResult<()> {
122    if column == VisibilityDataColumn::Data {
123        return Err(MsError::InvalidInput(
124            "DATA is not a scratch column".to_string(),
125        ));
126    }
127    let mut clearing = Clearing::default();
128    if column == VisibilityDataColumn::ModelData {
129        clearing.plan_virtual_model(ms)?;
130    }
131    if adds {
132        let keywords = ms.main_table().keywords();
133        clearing.main.extend(
134            SORT_KEYWORDS
135                .iter()
136                .filter(|name| keywords.get(name).is_some())
137                .map(|name| name.to_string()),
138        );
139    }
140    let holds_main = main_lock.is_some() || holds_write_lock(ms.main_table());
141    if let Some(source) = ms.subtable_mut(SubtableId::Source)
142        && !(clearing.source.is_empty() && clearing.source_cells.is_empty())
143    {
144        let attempts = if holds_main { 1 } else { 0 };
145        change_in_place(source, attempts, Write::Full, None, |source| {
146            for row in &clearing.source_cells {
147                source
148                    .cell_accessor_mut(*row, "SOURCE_MODEL")
149                    .and_then(|mut cell| cell.set(Value::Record(RecordValue::default())))?;
150            }
151            remove(source, &clearing.source);
152            Ok(())
153        })?;
154    }
155    if !clearing.main.is_empty() {
156        change_in_place(ms.main_table_mut(), 0, Write::Metadata, main_lock, |main| {
157            remove(main, &clearing.main);
158            Ok(())
159        })?;
160    }
161    if column == VisibilityDataColumn::ModelData {
162        delete_model_images(ms);
163    }
164    Ok(())
165}

167/// The keywords to remove from MAIN and SOURCE and the SOURCE rows whose
168/// `SOURCE_MODEL` is emptied: each present, so that a table with nothing
169/// to clear is not changed.
170#[derive(Default)]
171struct Clearing {
172    main: Vec<String>,
173    source: Vec<String>,
174    source_cells: Vec<usize>,
175}
176
177impl Clearing {
178    /// `VisModelData::clearModel`: for every field MAIN names, its pointer
179    /// and model keys wherever they are, and the SOURCE row of a model
180    /// under SOURCE (`removeRecordByKey`).
181    fn plan_virtual_model(&mut self, ms: &MeasurementSet) -> MsResult<()> {
182        let fields = fields_with_models(ms)?;
183        let main = ms.main_table().keywords();
184        let source = ms.subtable(SubtableId::Source);
185        let source_rows = source.map_or(0, Table::row_count);
186        let source_keywords = source.map(Table::keywords);
187        let source_model = source
188            .and_then(Table::schema)
189            .is_some_and(|schema| schema.contains_column("SOURCE_MODEL"));
190        for field in fields {
191            let pointer = format!("definedmodel_field_{field}");
192            let in_source = source_rows > 0
193                && source_keywords.is_some_and(|keywords| keywords.get(&pointer).is_some());
194            let model = if in_source {
195                source_keywords.and_then(|keywords| string(keywords, &pointer))
196            } else {
197                string(main, &pointer)
198            };
199            if let Some(model) = model {
200                if let Some(Value::Scalar(ScalarValue::Int32(row))) =
201                    source_keywords.and_then(|keywords| keywords.get(&model))
202                    && let Ok(row) = usize::try_from(*row)
203                    && row < source_rows
204                    && source_model
205                {
206                    self.source_cells.push(row);
207                }
208                present(&mut self.source, source_keywords, &model);
209                present(&mut self.main, Some(main), &model);
210            }
211            present(&mut self.source, source_keywords, &pointer);
212            present(&mut self.main, Some(main), &pointer);
213        }
214        self.source_cells.sort_unstable();
215        self.source_cells.dedup();
216        Ok(())
217    }
218}

220/// Add `name` to `names` when `keywords` holds it and `names` does not.
221fn present(names: &mut Vec<String>, keywords: Option<&RecordValue>, name: &str) {
222    if keywords.is_some_and(|keywords| keywords.get(name).is_some())
223        && !names.iter().any(|held| held == name)
224    {
225        names.push(name.to_string());
226    }
227}
228
229fn remove(table: &mut Table, names: &[String]) {
230    let keywords = table.keywords_mut();
231    for name in names {
232        keywords.remove(name);
233    }
234}

236/// MAIN rows read at a time while looking for the fields that have models.
237const FIELD_SCAN_ROWS: usize = 65_536;
238
239/// The fields that have a model pointer, in MAIN's or SOURCE's keywords,
240/// and rows in MAIN: those `clearModel` reaches, since it visits the fields
241/// MAIN names and a field without a pointer has nothing to clear. MAIN's
242/// `FIELD_ID` is read only when a pointer exists, a bounded chunk at a
243/// time, and no further than to find every pointed field.
244fn fields_with_models(ms: &MeasurementSet) -> MsResult<BTreeSet<i32>> {
245    let pointed = |keywords: &RecordValue| -> Vec<i32> {
246        keywords
247            .fields()
248            .iter()
249            .filter_map(|field| field.name.strip_prefix("definedmodel_field_")?.parse().ok())
250            .collect()
251    };
252    let mut wanted: BTreeSet<i32> = pointed(ms.main_table().keywords()).into_iter().collect();
253    if let Some(source) = ms.subtable(SubtableId::Source) {
254        wanted.extend(pointed(source.keywords()));
255    }
256    let mut found = BTreeSet::new();
257    if wanted.is_empty() {
258        return Ok(found);
259    }
260    let main = ms.main_table();
261    let column = main.column_accessor("FIELD_ID")?;
262    let mut start = 0;
263    while start < main.row_count() && found.len() < wanted.len() {
264        let rows: Vec<usize> = (start..main.row_count().min(start + FIELD_SCAN_ROWS)).collect();
265        for cell in column.scalar_cells_owned_for_rows(&rows)? {
266            if let Some(ScalarValue::Int32(field)) = cell
267                && wanted.contains(&field)
268            {
269                found.insert(field);
270            }
271        }
272        start += FIELD_SCAN_ROWS;
273    }
274    Ok(found)
275}

277fn string(keywords: &RecordValue, name: &str) -> Option<String> {
278    match keywords.get(name) {
279        Some(Value::Scalar(ScalarValue::String(value))) if !value.is_empty() => Some(value.clone()),
280        _ => None,
281    }
282}

284/// How a changed table is written.
285#[derive(Clone, Copy)]
286enum Write {
287    /// In full, with the storage managers [`MeasurementSet::save`] uses.
288    Full,
289    /// Its keywords and description alone, its storage left as it is.
290    Metadata,
291}

293/// Whether `table` was opened with locking and holds its write lock.
294fn holds_write_lock(table: &Table) -> bool {
295    #[cfg(unix)]
296    {
297        table.has_lock(casa_tables::LockType::Write)
298    }
299    #[cfg(not(unix))]
300    {
301        let _ = table;
302        false
303    }
304}
305
306/// Apply `change` to `table` and write it to its directory under
307/// casacore's write lock on it, taken in `attempts` (0 waits) unless the
308/// handle holds it or the caller does through `held`, whose write is then
309/// recorded for the caller's release to publish. A table not on disk is
310/// changed in memory alone. When the change or the write fails, a lock
311/// taken here has the table reloaded as it is on disk and is released, and
312/// under `held` the table is reloaded and the lock left to the caller; a
313/// lock the handle holds is left to it.
314fn change_in_place(
315    table: &mut Table,
316    attempts: u32,
317    write: Write,
318    held: Option<&mut TableWriteLock>,
319    change: impl FnOnce(&mut Table) -> MsResult<()>,
320) -> MsResult<()> {
321    let Some(path) = table.path().map(Path::to_path_buf) else {
322        return change(table);
323    };
324    let written = |table: &mut Table| -> MsResult<()> {
325        change(table)?;
326        match write {
327            Write::Full => table.save(measurement_set_table_options(&path))?,
328            Write::Metadata => table.save_metadata_only(TableOptions::new(&path))?,
329        }
330        Ok(())
331    };
332    if let Some(lock) = held {
333        lock.record_write();
334        let result = written(table);
335        if result.is_err() {
336            let _ = table.resync();
337        }
338        return result;
339    }
340    #[cfg(unix)]
341    if table.lock_options().is_some() {
342        if holds_write_lock(table) {
343            // Published when the caller releases its lock.
344            return written(table);
345        }
346        if !table.lock(casa_tables::LockType::Write, attempts)? {
347            return Err(lock_refused(&path));
348        }
349        let result = written(table);
350        if result.is_err() {
351            let _ = table.resync();
352        }
353        let released = table.unlock_metadata_only_or_discard();
354        return result.and(released.map_err(MsError::from));
355    }
356    let mut lock = TableWriteLock::acquire(&path, attempts)?;
357    lock.record_write();
358    let result = written(table);
359    if result.is_err() {
360        let _ = table.resync();
361    }
362    let released = lock.release();
363    result.and(released.map_err(MsError::from))
364}
365
366fn lock_refused(path: &Path) -> MsError {
367    TableError::LockFailed {
368        path: path.display().to_string(),
369        message: "another process holds the table's lock".to_string(),
370    }
371    .into()
372}

374/// Where the table at `path` is held open, as `TableUtil::canDeleteTable`
375/// asks, read from its lock file without opening the table or waiting for
376/// a lock.
377fn held_open(path: &Path) -> std::io::Result<Option<casa_tables::TableHolder>> {
378    #[cfg(unix)]
379    {
380        casa_tables::table_holder(path)
381    }
382    #[cfg(not(unix))]
383    {
384        let _ = path;
385        Ok(None)
386    }
387}

389/// Delete every `FT_MODEL*` table in the MeasurementSet's directory and in
390/// its SOURCE's, as `clearModel` does, at best: a table held open, in this
391/// process or another, is left, as `TableUtil::canDeleteTable` leaves it,
392/// and so is one whose lock file cannot be read or that cannot be deleted,
393/// each logged.
394fn delete_model_images(ms: &MeasurementSet) {
395    let directories = [
396        ms.path(),
397        ms.subtable(SubtableId::Source).and_then(Table::path),
398    ];
399    for directory in directories.into_iter().flatten() {
400        let Ok(entries) = std::fs::read_dir(directory) else {
401            continue;
402        };
403        for entry in entries.flatten() {
404            let image = entry.path();
405            if !entry.file_name().to_string_lossy().starts_with("FT_MODEL")
406                || !image.join("table.dat").is_file()
407            {
408                continue;
409            }
410            let deleted = match held_open(&image) {
411                Ok(None) => std::fs::remove_dir_all(&image).map_err(|error| error.to_string()),
412                Ok(Some(holder)) => Err(format!("it is held open {holder}")),
413                Err(error) => Err(error.to_string()),
414            };
415            if let Err(error) = deleted {
416                tracing::warn!(
417                    image = %image.display(),
418                    %error,
419                    "a cleared virtual model's image is left"
420                );
421            }
422        }
423    }
424}

827    /// Restores `table.dat`'s mode when dropped.
828    struct Writable(PathBuf);
829
830    impl Drop for Writable {
831        fn drop(&mut self) {
832            use std::os::unix::fs::PermissionsExt;
833            let _ = std::fs::set_permissions(&self.0, std::fs::Permissions::from_mode(0o644));
834        }
835    }

837    /// Make MAIN's `table.dat` read-only, or `None` where it stays
838    /// writable (a process running as root).
839    fn read_only_table_dat(path: &Path) -> Option<Writable> {
840        use std::os::unix::fs::PermissionsExt;
841        let table_dat = path.join("table.dat");
842        std::fs::set_permissions(&table_dat, std::fs::Permissions::from_mode(0o444))
843            .expect("read only");
844        let restore = Writable(table_dat);
845        if std::fs::OpenOptions::new()
846            .write(true)
847            .open(&restore.0)
848            .is_ok()
849        {
850            eprintln!("skipped: table.dat stays writable for this process");
851            return None;
852        }
853        Some(restore)
854    }
