24enum MeasureArgument {
25    Numeric(f64),
26    Quantity(Quantity),
27}
28
29impl MeasureArgument {
30    fn value(&self) -> ComponentFitQuantity<'_> {
31        match self {
32            Self::Numeric(value) => ComponentFitQuantity::Numeric(*value),
33            Self::Quantity(value) => ComponentFitQuantity::Quantity(value),
34        }
35    }
36}

38#[pymethods]
39impl PyImageTool {
40    /// Fit Gaussian sky components and write optional model and residual images.
41    #[allow(clippy::too_many_arguments)]
42    #[pyo3(signature = (r#box="", region=None, chans=None, stokes="", mask=None, includepix=DoubleList(vec![-1.]), excludepix=DoubleList(vec![-1.]), residual="", model="", estimates="", logfile="", append=true, newestimates="", complist="", overwrite=false, dooff=false, offset=0., fixoffset=false, stretch=false, rms=None, noisefwhm=None, summary=""), text_signature = "($self, box='', region=[], chans=[], stokes='', mask='', includepix=[-1.0], excludepix=[-1.0], residual='', model='', estimates='', logfile='', append=True, newestimates='', complist='', overwrite=False, dooff=False, offset=0.0, fixoffset=False, stretch=False, rms='', noisefwhm='', summary='')")]
43    fn fitcomponents<'py>(
44        &self,
45        py: Python<'py>,
46        r#box: &str,
47        region: Option<&Bound<'py, PyAny>>,
48        chans: Option<&Bound<'py, PyAny>>,
49        stokes: &str,
50        mask: Option<&Bound<'py, PyAny>>,
51        includepix: DoubleList,
52        excludepix: DoubleList,
53        residual: &str,
54        model: &str,
55        estimates: &str,
56        logfile: &str,
57        #[pyo3(from_py_with = require_bool)] append: bool,
58        newestimates: &str,
59        complist: &str,
60        #[pyo3(from_py_with = require_bool)] overwrite: bool,
61        #[pyo3(from_py_with = require_bool)] dooff: bool,
62        offset: f64,
63        #[pyo3(from_py_with = require_bool)] fixoffset: bool,
64        #[pyo3(from_py_with = require_bool)] stretch: bool,
65        rms: Option<&Bound<'py, PyAny>>,
66        noisefwhm: Option<&Bound<'py, PyAny>>,
67        summary: &str,
68    ) -> PyResult<Py<PyAny>> {
69        if py.detach(|| self.state().image.is_none()) {
70            return record_to_py(py, RecordValue::default());
71        }
72        let region: RegionArgument =
73            region.map_or_else(|| Ok(RegionArgument::default()), |arg| arg.extract())?;
74        let (channels, channels_history) = channel_argument(chans)?;
75        let (rms, rms_history) = measure_argument(rms, "rms")?;
76        let (noise, noise_history) = measure_argument(noisefwhm, "noisefwhm")?;
77        let mask = mask_argument(mask)?;
78        let region_history = if region.selection().is_none() {
79            "[]".to_owned()
80        } else {
81            region.history()
82        };
83        let record = py.detach(|| -> PyResult<_> {
84            let state = self.state();
85            let Some(source) = &state.image else {
86                return Ok(RecordValue::default());
87            };
88            let request = FitComponentsRequest {
89                box_selection: nonempty(r#box),
90                region: region.selection(),
91                chans: nonempty(&channels),
92                stokes: nonempty(stokes),
93                mask: Some(&mask),
94                stretch,
95                includepix: &includepix.0,
96                excludepix: &excludepix.0,
97                residual,
98                model,
99                estimates: nonempty(estimates).map(Path::new),
100                logfile,
101                append,
102                newestimates,
103                complist,
104                summary,
105                overwrite,
106                fit_zero_level: dooff,
107                zero_level: (offset, fixoffset),
108                rms: rms.as_ref().map(MeasureArgument::value),
109                noise_fwhm: noise.as_ref().map(MeasureArgument::value),
110                history: FitComponentsHistory {
111                    region: &region_history,
112                    channels: &channels_history,
113                    rms: &rms_history,
114                    noise_fwhm: &noise_history,
115                },
116                record_history: state.history,
117            };
118            fit_components(source, request, external_open_check()).map_err(raised)
119        })?;
120        record_to_py(py, record)
121    }
122}

124fn nonempty(value: &str) -> Option<&str> {
125    (!value.is_empty()).then_some(value)
126}
127
128fn channel_argument(value: Option<&Bound<'_, PyAny>>) -> PyResult<(String, String)> {
129    let Some(value) = value else {
130        return Ok((String::new(), "[]".into()));
131    };
132    if let Ok(text) = value.extract::<String>() {
133        let history = format!("{text:?}");
134        return Ok((text, history));
135    }
136    if !value.is_instance_of::<PyBool>()
137        && let Ok(channel) = value.extract::<i64>()
138    {
139        let text = channel.to_string();
140        return Ok((text.clone(), text));
141    }
142    if value.extract::<Variant>()?.is_empty_sequence()
143        || value.try_iter().is_ok_and(|mut values| {
144            values.all(|item| item.is_ok_and(|item| item.is_instance_of::<PyBool>()))
145        })
146    {
147        return Ok((String::new(), "[]".into()));
148    }
149    Err(raised(
150        "Unsupported type for chans. chans must be either an integer or a string",
151    ))
152}
153
154fn measure_argument(
155    value: Option<&Bound<'_, PyAny>>,
156    name: &str,
157) -> PyResult<(Option<MeasureArgument>, String)> {
158    let Some(value) = value else {
159        return Ok((None, r#""""#.into()));
160    };
161    let value = value.extract::<Variant>()?;
162    let history = input_history(&value);
163    let parsed = match &value {
164        Variant::Empty => None,
165        Variant::Text(text) if text.is_empty() => None,
166        Variant::Number(value) => Some(MeasureArgument::Numeric(*value)),
167        Variant::Text(_) | Variant::Record(_) => {
168            Some(MeasureArgument::Quantity(quantity(&value).map_err(raised)?))
169        }
170        _ => return Err(raised(format!("Unsupported data type for {name}"))),
171    };
172    Ok((parsed, history))
173}
