116/// Whether a sample's weights (`[channel, correlation]`) are an
117/// acquisition's rather than padding's, whose every weight is non-finite.
118///
119/// # Errors
120///
121/// An acquisition whose weights vary over channels: an MSv2 `WEIGHT` holds
122/// one per correlation, and the conversion writes no `WEIGHT_SPECTRUM`.
123pub(super) fn acquired(weights: ArrayView2<'_, f32>) -> Result<bool, ObservationError> {
124    let Some(first) = weights.outer_iter().next() else {
125        return Ok(false);
126    };
127    if weights.iter().all(|weight| !weight.is_finite()) {
128        return Ok(false);
129    }
130    let varies = weights.outer_iter().skip(1).any(|channel| {
131        channel
132            .iter()
133            .zip(first.iter())
134            .any(|(weight, first)| weight.to_bits() != first.to_bits())
135    });
136    if varies {
137        return Err(ObservationError::Unsupported {
138            what: "converting weights that vary over channels into an MSv2 (WEIGHT_SPECTRUM)",
139        });
140    }
141    Ok(true)
142}

144/// CASA's `SIGMA` for a weight, as `mstransform` derives it
145/// (`weightToSigma`): `1/√w`, and -1 where `w` is not positive.
146pub(super) fn sigma(weight: f32) -> f32 {
147    if weight > f32::MIN_POSITIVE {
148        1.0 / weight.sqrt()
149    } else {
150        -1.0
151    }
152}

210/// Each sample's `STATE_ID` in an MSv2 input's batches, read from its
211/// MAIN where an input partition's rows are written row by row
212/// ([`StateIds::PerRow`]); one input partition's batches in turn.
213#[derive(Default)]
214pub(super) struct RowStates {
215    /// The input partition's times the batches so far hold.
216    times: usize,
217    rows: Vec<usize>,
218    states: Vec<i32>,
219}
220
221impl RowStates {
222    /// Start an input partition's batches.
223    pub(super) fn restart(&mut self) {
224        self.times = 0;
225    }
226
227    /// Each sample's input `STATE_ID` in `batch`, the next of partition
228    /// `id` of `observation`.
229    ///
230    /// # Errors
231    ///
232    /// A batch that is not its grid's next times, or a failed read.
233    pub(super) fn of(
234        &mut self,
235        observation: &Msv2Observation,
236        id: PartitionId,
237        batch: &VisibilityBatch<'_>,
238    ) -> Result<&[i32], ObservationError> {
239        let times = batch.samples() / batch.samples_per_time().max(1);
240        observation.cell_states_into(
241            id,
242            self.times..self.times + times,
243            (&mut self.rows, &mut self.states),
244        )?;
245        if self.states.len() != batch.samples() {
246            return Err(request_error(
247                "a batch's samples are not its grid's cells at its times",
248            ));
249        }
250        self.times += times;
251        Ok(&self.states)
252    }
253}

289impl MainRows {
290    /// Start MAIN's writers at `staging` under `plan` ([`write_plan`]) for
291    /// the `occupancy` of `catalogue`'s (output) partitions, with
292    /// `data_columns` (`DATA`, `CORRECTED_DATA`, `MODEL_DATA`), the
293    /// subtables' `layout` and the `padding` policy the occupancy was
294    /// counted under.
295    pub(super) fn start(
296        (staging, plan): (&Path, MeasurementSetWritePlan),
297        catalogue: &Catalogue,
298        data_columns: Vec<&'static str>,
299        occupancy: &Occupancy,
300        layout: Layout,
301        padding: PaddingRows,
302    ) -> Result<Self, ObservationError> {
303        let planned = occupancy.rows.iter().sum();
304        let session = MeasurementSetWriteSession::start(staging, plan).map_err(storage)?;
305        let id = |name: &str| session.scalar_column(name).map_err(storage);
306        let scalars = Scalars {
307            antenna1: id("ANTENNA1")?,
308            antenna2: id("ANTENNA2")?,
309            field: id("FIELD_ID")?,
310            scan: id("SCAN_NUMBER")?,
311            time: id("TIME")?,
312            centroid: id("TIME_CENTROID")?,
313            interval: id("INTERVAL")?,
314            exposure: id("EXPOSURE")?,
315            constant: [
316                id("ARRAY_ID")?,
317                id("DATA_DESC_ID")?,
318                id("FEED1")?,
319                id("FEED2")?,
320                id("FLAG_ROW")?,
321                id("OBSERVATION_ID")?,
322                id("PROCESSOR_ID")?,
323            ],
324            state: id("STATE_ID")?,
325        };
326        let outputs = catalogue
327            .partitions
328            .iter()
329            .zip(&occupancy.intervals)
330            .map(|(entry, interval)| {
331                let [feed1, feed2] = entry.feeds.unwrap_or([0, 0]);
332                Output {
333                    observation: entry.observation.0 as i32,
334                    feeds: [feed1 as i32, feed2 as i32],
335                    interval: *interval,
336                }
337            })
338            .collect();
339        Ok(Self {
340            session,
341            layout,
342            outputs,
343            padding,
344            data_columns,
345            scalars,
346            planned,
347            rows: 0,
348            keys: Keys::default(),
349            scratch: Scratch::default(),
350        })
351    }
352
353    /// Write a row for every sample of `batches` (every acquisition, with
354    /// padding omitted), one batch per group the partition reads, of the
355    /// same times, the first with every role; the input's identities
356    /// written as `map` gives them, and each sample's input `STATE_ID`
357    /// as `states` give them where they are written row by row.
358    pub(super) fn push(
359        &mut self,
360        map: &RowMap,
361        batches: &[VisibilityBatch<'_>],
362        states: Option<&[i32]>,
363    ) -> Result<(), ObservationError> {
364        let batch = &batches[0];
365        let missing = |role: &str| request_error(format!("a converted batch carries its {role}"));
366        let weights = batch.group.weights.ok_or_else(|| missing("weights"))?;
367        let flags = batch.group.flags.ok_or_else(|| missing("flags"))?;
368        let uvw = batch.group.uvw.ok_or_else(|| missing("UVW"))?;
369        let centroid = batch
370            .timing
371            .centroid
372            .ok_or_else(|| missing("time centroids"))?;
373        if batches.iter().any(|each| {
374            each.partition != batch.partition
375                || each.samples() != batch.samples()
376                || each.time.offsets_s() != batch.time.offsets_s()
377        }) {
378            return Err(request_error(
379                "a further group's batch is not the first's partition and times",
380            ));
381        }
382        let placement = &map.partitions[batch.partition.0 as usize];
383        if matches!(placement.state, StateIds::PerRow(_)) != states.is_some() {
384            return Err(request_error(
385                "a batch's row states are given exactly where they are written row by row",
386            ));
387        }
388        let values = placement
389            .columns
390            .iter()
391            .map(|source| match source {
392                ColumnSource::Group(index) => batches[*index]
393                    .group
394                    .values
395                    .map(Some)
396                    .ok_or_else(|| missing("visibilities")),
397                ColumnSource::InitialModel => Ok(None),
398            })
399            .collect::<Result<Vec<_>, _>>()?;
400        let shape = [batch.correlations.len(), batch.channel_count()];
401        if placement.columns.contains(&ColumnSource::InitialModel) {
402            self.scratch.model.clear();
403            for _ in 0..shape[1] {
404                self.scratch.model.extend(
405                    batch
406                        .correlations
407                        .iter()
408                        .map(|correlation| initial_model_visibility(correlation.casacore_code())),
409                );
410            }
411        }
412        let origin = MjdHighPrec::new(0.0, 0.0);
413        let interval = self.outputs[placement.output].interval;
414        self.keys.clear();
415        for sample in 0..batch.samples() {
416            let sample_weights = weights.index_axis(Axis(0), sample);
417            let acquisition = acquired(sample_weights)?;
418            if !acquisition && self.padding == PaddingRows::Omitted {
419                continue;
420            }
421            self.push_cells(
422                sample,
423                shape,
424                (&values, flags, sample_weights),
425                (acquisition, map.weight_scale),
426            )?;
427            self.session
428                .push_f64_row("UVW", &[3], &uvw.metres_at(sample))
429                .map_err(storage)?;
430            let coordinate = batch.sample(sample);
431            let scan = batch.scan(sample).ok_or_else(|| missing("scans"))?;
432            let time = batch
433                .time
434                .seconds_since(sample / batch.samples_per_time(), origin);
435            let keys = &mut self.keys;
436            keys.antenna1
437                .push(map.antennas[coordinate.baseline.antenna1.0 as usize]);
438            keys.antenna2
439                .push(map.antennas[coordinate.baseline.antenna2.0 as usize]);
440            keys.field.push(map.fields[coordinate.field.0 as usize]);
441            keys.scan
442                .push(map.scans[scan.0 as usize] + placement.scan_offset);
443            if let Some(states) = states {
444                let state = usize::try_from(states[sample])
445                    .ok()
446                    .and_then(|state| map.states.get(state));
447                keys.state.push(state.copied().unwrap_or(-1));
448            }
449            keys.time.push(time);
450            if acquisition {
451                keys.centroid.push(centroid.seconds_since(sample, origin));
452                keys.interval.push(
453                    batch
454                        .interval_s(sample)
455                        .ok_or_else(|| missing("intervals"))?,
456                );
457                keys.exposure.push(
458                    batch
459                        .exposure_s(sample)
460                        .ok_or_else(|| missing("exposures"))?,
461                );
462            } else {
463                // A padded row's keys stay valid: nothing was integrated,
464                // over the grid's interval, centred on its time.
465                keys.centroid.push(time);
466                keys.interval.push(interval);
467                keys.exposure.push(0.0);
468            }
469        }
470        self.push_keys(placement)
471    }
472
473    /// One sample's visibility, flag, weight and sigma cells; padding's
474    /// flagged throughout with NaN weights, padding's one mark. An
475    /// acquisition's weights are multiplied by `scale`, and its sigmas
476    /// derived from the scaled weights, so `SIGMA` is `1/√(W·w)`.
477    fn push_cells(
478        &mut self,
479        sample: usize,
480        shape: [usize; 2],
481        (values, flags, weights): Cells<'_, '_>,
482        (acquisition, scale): (bool, f32),
483    ) -> Result<(), ObservationError> {
484        // A sample's `[channel, correlation]` cells, correlation fastest,
485        // are an MSv2 cell of shape `[correlation, channel]` as casacore
486        // stores it.
487        let scratch = &mut self.scratch;
488        for (column, values) in self.data_columns.iter().zip(values) {
489            let cell = match values {
490                Some(values) => contiguous(values.index_axis(Axis(0), sample), &mut scratch.values),
491                None => &scratch.model,
492            };
493            self.session
494                .push_complex32_row(column, &shape, cell)
495                .map_err(storage)?;
496        }
497        let cell = if acquisition {
498            contiguous(flags.index_axis(Axis(0), sample), &mut scratch.flags)
499        } else {
500            scratch.flags.clear();
501            scratch.flags.resize(shape[0] * shape[1], true);
502            &scratch.flags
503        };
504        self.session
505            .push_bool_row("FLAG", &shape, cell)
506            .map_err(storage)?;
507        self.session
508            .push_undefined_row("FLAG_CATEGORY")
509            .map_err(storage)?;
510        scratch.weights.clear();
511        if acquisition {
512            scratch
513                .weights
514                .extend(weights.row(0).iter().map(|weight| weight * scale));
515        } else {
516            scratch.weights.resize(shape[0], f32::NAN);
517        }
518        scratch.sigmas.clear();
519        scratch
520            .sigmas
521            .extend(scratch.weights.iter().map(|weight| sigma(*weight)));
522        let correlations = [shape[0]];
523        self.session
524            .push_f32_row("WEIGHT", &correlations, &scratch.weights)
525            .map_err(storage)?;
526        self.session
527            .push_f32_row("SIGMA", &correlations, &scratch.sigmas)
528            .map_err(storage)
529    }

531    /// The batch's gathered keys, and its input partition's constant ones.
532    fn push_keys(&mut self, placement: &Placement) -> Result<(), ObservationError> {
533        let keys = &self.keys;
534        let count = keys.antenna1.len();
535        let scalars = &self.scalars;
536        for (column, values) in [
537            (
538                scalars.antenna1,
539                StreamedScalarValues::Int32(&keys.antenna1),
540            ),
541            (
542                scalars.antenna2,
543                StreamedScalarValues::Int32(&keys.antenna2),
544            ),
545            (scalars.field, StreamedScalarValues::Int32(&keys.field)),
546            (scalars.scan, StreamedScalarValues::Int32(&keys.scan)),
547            (scalars.time, StreamedScalarValues::Float64(&keys.time)),
548            (
549                scalars.centroid,
550                StreamedScalarValues::Float64(&keys.centroid),
551            ),
552            (
553                scalars.interval,
554                StreamedScalarValues::Float64(&keys.interval),
555            ),
556            (
557                scalars.exposure,
558                StreamedScalarValues::Float64(&keys.exposure),
559            ),
560        ] {
561            self.session
562                .push_scalar_column(column, values)
563                .map_err(storage)?;
564        }
565        let output = placement.output;
566        let [feed1, feed2] = self.outputs[output].feeds;
567        let constants = [
568            ScalarValue::Int32(0),
569            ScalarValue::Int32(self.layout.data_description[output]),
570            ScalarValue::Int32(feed1),
571            ScalarValue::Int32(feed2),
572            ScalarValue::Bool(false),
573            ScalarValue::Int32(self.outputs[output].observation),
574            ScalarValue::Int32(placement.processor),
575        ];
576        for (column, value) in scalars.constant.iter().zip(constants) {
577            self.session
578                .push_repeated_scalar(*column, Some(value), count)
579                .map_err(storage)?;
580        }
581        let state = match placement.state {
582            StateIds::One(state) => Some(state),
583            StateIds::Intents => Some(self.layout.intent_states[output].unwrap_or(-1)),
584            StateIds::PerRow(_) => None,
585        };
586        match state {
587            Some(state) => self.session.push_repeated_scalar(
588                scalars.state,
589                Some(ScalarValue::Int32(state)),
590                count,
591            ),
592            None => self
593                .session
594                .push_scalar_column(scalars.state, StreamedScalarValues::Int32(&keys.state)),
595        }
596        .map_err(storage)?;
597        self.rows += count;
598        Ok(())
599    }

601    /// Save `ms` with MAIN's columns installed; the rows written.
602    ///
603    /// # Errors
604    ///
605    /// Fewer rows than planned (an input changed between the passes), or
606    /// a write failure.
607    pub(super) fn finish(self, ms: &mut MeasurementSet) -> Result<usize, ObservationError> {
608        if self.rows != self.planned {
609            return Err(request_error(format!(
610                "{} rows written of the {} the occupancy pass counted",
611                self.rows, self.planned
612            )));
613        }
614        let mut overrides = ColumnOverrides::for_row_count(self.rows);
615        for column in self.data_columns.iter().chain(&ARRAY_COLUMNS) {
616            overrides.insert_deferred(*column);
617        }
618        self.session
619            .save_and_finish(ms, &overrides)
620            .map_err(storage)?;
621        Ok(self.rows)
622    }
623}

625/// `view`'s elements in order: the view itself when contiguous, else a
626/// copy in `scratch`.
627fn contiguous<'a, T: Copy>(view: ArrayView2<'a, T>, scratch: &'a mut Vec<T>) -> &'a [T] {
628    match view.to_slice() {
629        Some(slice) => slice,
630        None => {
631            scratch.clear();
632            scratch.extend(view.iter().copied());
633            scratch
634        }
635    }
636}

638/// MAIN's variable-shape creation plan for the `occupancy` of
639/// `catalogue`'s partitions with `data_columns`, each array column's writer
640/// given `column_bytes` ([`array_columns`]).
641pub(super) fn write_plan(
642    columns: Vec<MeasurementSetArrayColumnPlan>,
643    occupancy: &Occupancy,
644    column_bytes: usize,
645) -> Result<MeasurementSetWritePlan, ObservationError> {
646    let scalars = standard_main_scalar_column_plans();
647    let resources = MeasurementSetWritePlan::variable_array_creation_resources(
648        columns.len(),
649        scalars.len(),
650        column_bytes,
651    )
652    .ok_or_else(|| request_error("the MAIN writers' memory overflows"))?;
653    MeasurementSetWritePlan::variable_array_creation(
654        occupancy.rows.iter().sum(),
655        columns,
656        scalars,
657        resources,
658    )
659    .map_err(storage)
660}
661
662/// What one batch's rows hold in [`MainRows`] beside the writers: the keys
663/// of `samples` samples with the reading of their states ([`RowStates`]),
664/// and a cell of `cell` elements of each kind.
665pub(super) fn held_bytes(samples: usize, cell: usize) -> usize {
666    let keys = 5 * size_of::<i32>() + 4 * size_of::<f64>();
667    let states = size_of::<usize>() + size_of::<i32>();
668    let scratch = 2 * size_of::<Complex32>() + size_of::<bool>() + 2 * size_of::<f32>();
669    samples * (keys + states) + cell * scratch
670}

672/// MAIN's array columns for the `occupancy` of `catalogue`'s partitions
673/// with `data_columns`: one hypercube per visibility shape and per
674/// correlation count, sized by the rows each holds.
675pub(super) fn array_columns(
676    catalogue: &Catalogue,
677    data_columns: &[&'static str],
678    occupancy: &Occupancy,
679) -> Result<Vec<MeasurementSetArrayColumnPlan>, ObservationError> {
680    let (rows, total) = (&occupancy.rows, occupancy.rows.iter().sum::<usize>());
681    if total == 0 {
682        return Err(request_error(
683            "the source holds nothing to write as an MSv2 row",
684        ));
685    }
686    let telescope = catalogue
687        .observations
688        .first()
689        .map_or("", |observation| observation.telescope.as_str());
690    let mut by_shape = BTreeMap::<(usize, usize), usize>::new();
691    let mut by_correlations = BTreeMap::<usize, usize>::new();
692    for (entry, rows) in catalogue.partitions.iter().zip(rows) {
693        let correlations = entry.correlations.len();
694        *by_shape
695            .entry((correlations, entry.frequencies_hz.len()))
696            .or_default() += rows;
697        *by_correlations.entry(correlations).or_default() += rows;
698    }
699    let visibility: Vec<_> = by_shape
700        .iter()
701        .map(|(&(correlations, channels), &rows)| {
702            MeasurementSetArrayShapePlan::visibility(correlations, channels, rows, telescope)
703        })
704        .collect();
705    let flag_category = by_shape
706        .iter()
707        .map(|(&(correlations, channels), &rows)| {
708            MeasurementSetArrayShapePlan::flag_category(correlations, channels, 1, rows, telescope)
709        })
710        .collect();
711    let weight: Vec<_> = by_correlations
712        .iter()
713        .map(|(&correlations, &rows)| {
714            MeasurementSetArrayShapePlan::weight(correlations, rows, telescope)
715        })
716        .collect();
717    let uvw = MeasurementSetArrayShapePlan {
718        cell_shape: vec![3],
719        row_count: total,
720        tile_shape: crate::ms::casa_uvw_tile_shape(&visibility[0].tile_shape),
721    };
722    let column = |name: &str,
723                  value_type: StreamedTiledShapeValueType,
724                  shapes: Vec<MeasurementSetArrayShapePlan>,
725                  storage_manager: MeasurementSetColumnStorage| {
726        MeasurementSetArrayColumnPlan {
727            name: name.to_string(),
728            value_type,
729            shapes,
730            storage_manager,
731        }
732    };
733    let shaped = MeasurementSetColumnStorage::TiledShape;
734    let mut columns: Vec<_> = data_columns
735        .iter()
736        .map(|name| {
737            column(
738                name,
739                StreamedTiledShapeValueType::Complex32,
740                visibility.clone(),
741                shaped,
742            )
743        })
744        .collect();
745    columns.extend([
746        column(
747            "FLAG",
748            StreamedTiledShapeValueType::Bool,
749            visibility,
750            shaped,
751        ),
752        column(
753            "FLAG_CATEGORY",
754            StreamedTiledShapeValueType::Bool,
755            flag_category,
756            shaped,
757        ),
758        column(
759            "UVW",
760            StreamedTiledShapeValueType::Float64,
761            vec![uvw],
762            MeasurementSetColumnStorage::TiledColumn,
763        ),
764        column(
765            "WEIGHT",
766            StreamedTiledShapeValueType::Float32,
767            weight.clone(),
768            shaped,
769        ),
770        column(
771            "SIGMA",
772            StreamedTiledShapeValueType::Float32,
773            weight,
774            shaped,
775        ),
776    ]);
777    Ok(columns)
778}
