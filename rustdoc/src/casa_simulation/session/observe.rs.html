2//! `sm.settimes`, `sm.observe` and `sm.observemany`: the schedule of scans.
3//!
4//! `observe` records a scan, and `observemany` a run of pointings, each
5//! pointing a scan of its own; the session writes every scan's rows at
6//! `close`. The bookkeeping follows `NewMSSimulator::observe`, whose
7//! one-source form is `observemany` with one pointing and no direction:
8//! - scan numbers continue from the last scan, and a run of pointings of one
9//!   field shares one;
10//! - an OBSERVATION row is added when asked for or when there is none;
11//! - STATE rows are shared by identical states;
12//! - each pointing's integration count is
13//!   `max(1, Int(0.5 + (stop − start)/Tint))`.
14//!
15//! **Times honour the requested UTC** (plan decision 5). CASA adds the
16//! reference time in TAI to start and stop and stores the result in a TIME
17//! column labelled UTC, so its rows are TAI − UTC (37 s) late. casa-rs adds
18//! the reference time in UTC.
19

44/// The arguments of `sm.observe`.

62    pub observer: String,
63    /// PROJECT (`project`).
64    pub project: String,

67/// One pointing of `sm.observemany`: a field, by name, observed from
68/// `start_s` to `stop_s`, its phase centre moved to `direction` when one is
69/// given.
70#[derive(Debug, Clone)]
71pub struct Pointing {
72    /// The field, by name (an element of `sourcenames`).
73    pub source: String,
74    /// Start, seconds, as [`ObserveSpec::start_s`] (an element of
75    /// `starttimes`).
76    pub start_s: f64,
77    /// Stop, seconds (an element of `stoptimes`).
78    pub stop_s: f64,
79    /// The pointing's phase centre (an element of `directions`). `None`
80    /// keeps the field's, or the previous pointing's when the field is the
81    /// same; casatools passes the pole for an empty direction, which
82    /// `NewMSSimulator` reads as none.
83    pub direction: Option<MDirection>,
84}

167/// One scan's rows, as `observe` planned them.
168#[derive(Debug, Clone)]
169pub(crate) struct Scan {
170    pub(crate) scan_number: i32,
171    pub(crate) field_id: usize,
172    pub(crate) spectral_window_id: usize,
173    /// The scan's start (CASA's `Tstart`), UTC seconds.
174    pub(crate) start_s: f64,
175    pub(crate) integration_s: f64,
176    pub(crate) integration_count: usize,
177    pub(crate) observation_id: i32,
178    pub(crate) state_id: usize,
179    /// Antennas and feeds defined when `observe` was called.
180    pub(crate) antenna_count: usize,
181    /// The J2000 phase centre, radians.
182    pub(crate) phase_center_rad: [f64; 2],
183    /// Autocorrelation rows are written when the weight is positive.
184    pub(crate) autocorrelation_weight: f32,
185    pub(crate) elevation_limit_rad: f64,
186    pub(crate) shadow_fraction_limit: f64,
187}
188
189impl Scan {
190    /// Baselines per integration, autocorrelations included when written.
191    pub(crate) fn baseline_count(&self) -> usize {
192        let n = self.antenna_count;
193        if self.autocorrelation_weight > 0.0 {
194            n * (n + 1) / 2
195        } else {
196            n * n.saturating_sub(1) / 2
197        }
198    }
199
200    /// MAIN rows of the scan.
201    pub(crate) fn row_count(&self) -> usize {
202        self.baseline_count() * self.integration_count
203    }
204
205    /// TIME of integration `index`: CASA accumulates `Time += Tint` from
206    /// the start and stores the integration's centre.
207    pub(crate) fn integration_times_s(&self) -> impl Iterator<Item = f64> + '_ {
208        let mut time = self.start_s;
209        (0..self.integration_count).map(move |_| {
210            let centre = time + self.integration_s / 2.0;
211            time += self.integration_s;
212            centre
213        })
214    }
215}

217impl super::Simulator {
218    /// `NewMSSimulator::observe`'s bookkeeping for `spec`'s pointings, each
219    /// planned as a [`Scan`] of its own:
220    /// - a new scan number at the first pointing and whenever the field
221    ///   changes, so consecutive pointings of one field share one;
222    /// - each pointing's phase centre is its direction when it has one,
223    ///   else its field's at a change of field, else the previous
224    ///   pointing's;
225    /// - the hour-angle offset is the first field's, and the OBSERVATION
226    ///   row spans the first pointing's start to the last one's stop.
227    ///
228    /// CASA adds every pointing's MAIN rows before it looks up the fields
229    /// after the first, so a missing one raises with rows left behind;
230    /// casa-rs looks every field up first and plans nothing.
231    pub(super) fn plan_pointings(&mut self, spec: &ObserveManySpec) -> SimulatorResult<()> {
232        let spectral_window_id = self.observable_window(&spec.spectral_window)?;
233        let Some((first, last)) = spec.pointings.first().zip(spec.pointings.last()) else {
234            return Err(SimulatorError::refused(
235                "observemany: no pointings to observe",
236            ));
237        };
238        if self.fields.is_empty() {
239            return Err(SimulatorError::refused("Field information not yet defined"));
240        }
241        let field_of = |name: &str| self.fields.iter().position(|field| field.name == name);
242        let first_field = field_of(&first.source).ok_or_else(|| {
243            SimulatorError::refused(format!("Field named {} not yet defined", first.source))
244        })?;
245        let mut field_ids = Vec::with_capacity(spec.pointings.len());
246        for (index, pointing) in spec.pointings.iter().enumerate() {
247            field_ids.push(field_of(&pointing.source).ok_or_else(|| {
248                SimulatorError::refused(format!(
249                    "   Field {index}: {} not yet defined",
250                    pointing.source
251                ))
252            })?);
253        }
254        let directions = spec
255            .pointings
256            .iter()
257            .map(|pointing| {
258                pointing
259                    .direction
260                    .as_ref()
261                    .map(|direction| self.j2000_rad(direction, "pointing direction"))
262                    .transpose()
263            })
264            .collect::<SimulatorResult<Vec<_>>>()?;
265        let first_direction_rad = self.fields[first_field].direction_rad;
266        tracing::info!(
267            "First source: {} @ {first_direction_rad:?}",
268            self.fields[first_field].name
269        );
270
271        let integration_s = self.times.integration_s;
272        if integration_s <= 0.0 {
273            return Err(SimulatorError::refused(
274                "observe: the integration time must be positive",
275            ));
276        }
277        if self.times.use_hour_angle && self.times.hour_angle_offset_s.is_none() {
278            self.times.hour_angle_offset_s = Some(hour_angle_offset_s(
279                first_direction_rad,
280                &self.times.reference,
281                self.antennas[0].position_itrf_m,
282                self.options.measures.clone(),
283            )?);
284        }
285        let offset_s = if self.times.use_hour_angle {
286            self.times.hour_angle_offset_s.unwrap_or(0.0)
287        } else {
288            0.0
289        };
290        let reference_s = self.times.reference_utc_s();
291        let utc_s = move |seconds: f64| seconds + reference_s + offset_s;
292        let (start_s, stop_s) = (utc_s(first.start_s), utc_s(last.stop_s));
293        tracing::info!("Full time range: {start_s} -- {stop_s} UTC with int = {integration_s}");
294
295        let observation_id = self.observation_for(spec, [start_s, stop_s]);
296        let state_id = match self.states.iter().rposition(|state| *state == spec.state) {
297            Some(existing) => existing,
298            None => {
299                self.states.push(spec.state.clone());
300                self.states.len() - 1
301            }
302        };
303        let mut scan_number = self.scans.last().map_or(0, |scan| scan.scan_number);
304        let mut phase_center_rad = first_direction_rad;
305        let mut integrations = 0;
306        for (index, pointing) in spec.pointings.iter().enumerate() {
307            let field_id = field_ids[index];
308            if index == 0 || pointing.source != spec.pointings[index - 1].source {
309                scan_number += 1;
310                phase_center_rad = self.fields[field_id].direction_rad;
311            }
312            if let Some(direction) = directions[index] {
313                phase_center_rad = direction;
314            }
315            let start_s = utc_s(pointing.start_s);
316            let integration_count =
317                integration_count(start_s, utc_s(pointing.stop_s), integration_s);
318            integrations += integration_count;
319            self.scans.push(Scan {
320                scan_number,
321                field_id,
322                spectral_window_id,
323                start_s,
324                integration_s,
325                integration_count,
326                observation_id,
327                state_id,
328                antenna_count: self.antennas.len(),
329                phase_center_rad,
330                autocorrelation_weight: self.autocorrelation_weight,
331                elevation_limit_rad: self.elevation_limit_rad,
332                shadow_fraction_limit: self.shadow_fraction_limit,
333            });
334        }
335        tracing::info!("Calculating a total of {integrations} integrations");
336        Ok(())
337    }

339    /// The checks `NewMSSimulator::observe` makes before its fields, and
340    /// the default feeds it sets: antennas, feeds as one per antenna, and
341    /// the spectral window named `name`, whose index it returns.
342    fn observable_window(&mut self, name: &str) -> SimulatorResult<usize> {
343        if self.antennas.is_empty() {
344            return Err(SimulatorError::refused(
345                "Antenna information not yet defined",
346            ));
347        }
348        if !self.feeds_set && self.feeds.is_empty() {
349            tracing::warn!("Feeds have not been set - using default perfect R L");
350            self.feeds = super::feeds::perfect_feeds("perfect R L", self.antennas.len());
351        }
352        if !self.times.set {
353            tracing::warn!("Times have not been set - using defaults");
354            tracing::warn!("     Times will be interpreted as hour angles for first source");
355        }
356        if self.feeds.len() != self.antennas.len() {
357            return Err(SimulatorError::refused(
358                "Feed table format is incompatible with existing code of NewMSSimulator::observe",
359            ));
360        }
361        if self.spectral_windows.is_empty() {
362            return Err(SimulatorError::refused(
363                "Spectral window information not yet defined",
364            ));
365        }
366        self.spectral_windows
367            .iter()
368            .position(|window| window.name == name)
369            .ok_or_else(|| {
370                SimulatorError::refused(format!("Spectral window named {name} not yet defined"))
371            })
372    }
373
374    /// The OBSERVATION_ID of a planned observation spanning `time_range_s`:
375    /// a new OBSERVATION row when `spec` asks for one or there is none,
376    /// numbered one past the largest OBSERVATION_ID in MAIN so far, else 0.
377    fn observation_for(&mut self, spec: &ObserveManySpec, time_range_s: [f64; 2]) -> i32 {
378        if !spec.add_observation && !self.observations.is_empty() {
379            return 0;
380        }
381        self.observations.push(Observation {
382            telescope: self.telescope.clone(),
383            time_range_s,
384            observer: spec.observer.clone(),
385            project: spec.project.clone(),
386        });
387        self.scans
388            .iter()
389            .map(|scan| scan.observation_id)
390            .max()
391            .map_or(0, |largest| largest + 1)
392    }
393}

395/// CASA's integration count for one pointing: `max(1, Int(0.5 + span/Tint))`,
396/// with C++'s truncation toward zero.
397pub(crate) fn integration_count(start_s: f64, stop_s: f64, integration_s: f64) -> usize {
398    let count = (0.5 + (stop_s - start_s) / integration_s).trunc();
399    if count >= 1.0 { count as usize } else { 1 }
400}

402/// `NewMSSimulator`'s hour-angle offset: minus the hour angle of `direction`
403/// at the reference epoch and the first antenna, as seconds, converting
404/// hours of hour angle at 3600 seconds each (no sidereal factor).
405pub(crate) fn hour_angle_offset_s(
406    direction_rad: [f64; 2],
407    reference: &MEpoch,
408    first_antenna_itrf_m: [f64; 3],
409    measures: Arc<dyn MeasuresProvider>,
410) -> SimulatorResult<f64> {
411    let [x, y, z] = first_antenna_itrf_m;
412    let frame = MeasFrame::new()
413        .with_epoch(reference.clone())
414        .with_position(MPosition::new_itrf(x, y, z))
415        .with_measures(measures);
416    let hadec = MDirection::from_angles(direction_rad[0], direction_rad[1], DirectionRef::J2000)
417        .convert_to(DirectionRef::HADEC, &frame)
418        .map_err(|error| {
419            SimulatorError::refused(format!("hour angle of the first source: {error}"))
420        })?;
421    let mut hour_angle = hadec.as_angles().0;
422    // casacore's MVDirection longitude is in (−π, π].
423    if hour_angle > std::f64::consts::PI {
424        hour_angle -= std::f64::consts::TAU;
425    } else if hour_angle <= -std::f64::consts::PI {
426        hour_angle += std::f64::consts::TAU;
427    }
428    Ok(-hour_angle * (3600.0 * 180.0 / std::f64::consts::PI / 15.0))
429}
