2//! `casars.tools.simulator`: CASA's `simulator` tool (`sm`) over
3//! [`casa_simulation::Simulator`].
4//!
5//! The class declares every method of CASA's installed binding with CASA's
6//! names, parameter order and effective defaults ([`surface`]), and each
7//! method delegates to a free function: the session's lifecycle in
8//! `lifecycle`, the setters and `observe` in `setup`, prediction
9//! (`predict`, `setoptions`, `setvp`) in `prediction`, thermal noise
10//! (`setseed`, `setnoise`, `corrupt`, `reset`) in `corruption`, and the
11//! methods casa-rs does not implement yet, or that CASA disables, in `stubs`.
12//!
13//! CASA's outcomes are kept: before `open` every method but `type`, `close`
14//! and `done` returns `False` (or `''`) without converting its arguments, and
15//! `setconfig` also logs "The sm tool must be opened first."; a failure CASA
16//! reports by logging and returning `False` is logged at ERROR on the Python
17//! logger `casars.tools.simulator`; a failure CASA raises is a
18//! `RuntimeError`. What CASA does and casa-rs does not yet (a whole method,
19//! or a mode such as `setnoise(mode='tsys-atm')`) raises
20//! `NotImplementedError` naming the phase that implements it.
21//!
22//! Divergence: the MeasurementSet is written at `close` or `done` (and by an
23//! `open` that replaces the session), never while observing, so `predict`
24//! prepares the model and `corrupt` records the noise, and `close` writes
25//! both. A tool dropped with planned
26//! scans and no `close` writes nothing, where CASA leaves the partly written
27//! MS on disk; the drop emits a `ResourceWarning`.

57/// CASA's `simulator` tool.
58///
59/// `simulator(*, memory_budget=None, backend='auto', workers=None)` takes
60/// only casa-rs's controls, keyword-only (ADR-0019 item 3):
61/// - `memory_budget` caps, in bytes, what a session holds at once while
62///   `close` writes: the MeasurementSet writer gets half of it and the
63///   geometry and noise the rest. `None` takes three quarters of the
64///   machine's free memory.
65/// - `backend` is the device to predict on: `'auto'` (the machine's Metal
66///   device when it has one, else the CPU), `'cpu'` or `'metal'`; `'metal'`
67///   on a machine without a Metal device raises `RuntimeError`. Noise is
68///   computed on the CPU whatever the backend.
69/// - `workers` is the number of threads `close` keeps busy, the writer's
70///   among them; `None` takes three quarters of the machine's threads, at
71///   most its performance cores.
72///
73/// The MeasurementSet is bitwise the same for any budget or workers, and
74/// for any backend when nothing is predicted; a Metal prediction agrees with
75/// the CPU's to `1e-4` of the largest visibility.
76#[pyclass(name = "simulator", module = "casars.tools", frozen)]
77pub(crate) struct PySimulator {
78    state: Mutex<Option<Session>>,
79    resources: SessionResources,
80}

82impl PySimulator {
83    /// The session lock, taken without the GIL (inside `Python::detach`).
84    fn lock(&self) -> MutexGuard<'_, Option<Session>> {
85        self.state.lock().unwrap_or_else(PoisonError::into_inner)
86    }
87
88    /// The session lock, taken with the GIL attached; waiting for it
89    /// releases the GIL.
90    fn lock_attached(&self, py: Python<'_>) -> MutexGuard<'_, Option<Session>> {
91        self.state
92            .lock_py_attached(py)
93            .unwrap_or_else(PoisonError::into_inner)
94    }
95
96    /// Runs `work` on the open session with the GIL released, and logs what
97    /// it reports once the GIL is back. Before `open`, returns `None` without
98    /// running it.
99    ///
100    /// The method has converted its Python arguments to Rust values (the
101    /// signature's `Variant`s and lists) before it calls this; `work` turns
102    /// them into the session's types, which can need the session's measures,
103    /// and does the session's work, all without the GIL (ADR-0019 §4). An
104    /// error it returns is created lazily and raised once the GIL is back.
105    fn on_session<T: Send>(
106        &self,
107        py: Python<'_>,
108        work: impl FnOnce(&mut Session, &mut Vec<LogLine>) -> PyResult<T> + Send,
109    ) -> PyResult<Option<T>> {
110        let (result, lines) = py.detach(|| {
111            let mut lines = Vec::new();
112            let result = match self.lock().as_mut() {
113                Some(session) => work(session, &mut lines).map(Some),
114                None => Ok(None),
115            };
116            (result, lines)
117        });
118        emit(py, LOGGER, &lines)?;
119        result
120    }
121
122    /// Whether a session is open.
123    fn is_open(&self, py: Python<'_>) -> bool {
124        self.lock_attached(py).is_some()
125    }
126}

128/// CASA's outcome for a session result: `True`, `False` with the message
129/// logged at ERROR, or a `RuntimeError`; or `NotImplementedError` for what
130/// casa-rs does not do yet.
131fn outcome(result: SimulatorResult<()>, lines: &mut Vec<LogLine>) -> PyResult<bool> {
132    match result {
133        Ok(()) => Ok(true),
134        Err(error) => match error.outcome() {
135            CasaOutcome::ReturnsFalse => {
136                lines.push(LogLine::error(error.message()));
137                Ok(false)
138            }
139            CasaOutcome::Raises => Err(raised(&error)),
140            CasaOutcome::NotYetImplemented => {
141                Err(PyNotImplementedError::new_err(error.message().to_string()))
142            }
143        },
144    }
145}
146
147/// A session error as the `RuntimeError` CASA raises.
148fn raised(error: &SimulatorError) -> PyErr {
149    PyRuntimeError::new_err(error.message().to_string())
150}

152impl Drop for PySimulator {
153    fn drop(&mut self) {
154        let state = self.state.get_mut().unwrap_or_else(PoisonError::into_inner);
155        let Some(session) = state.take() else {
156            return;
157        };
158        let scans = session.simulator.scan_count();
159        if scans == 0 {
160            return;
161        }
162        let message = format!(
163            "simulator for {} was dropped with {scans} planned scan(s) and no close(); \
164             nothing was written (casa-rs writes the MeasurementSet at close or done)",
165            session.simulator.name().display()
166        );
167        Python::attach(|py| {
168            let category = py.get_type::<PyResourceWarning>();
169            let message = CString::new(message).unwrap_or_default();
170            if let Err(error) = PyErr::warn(py, category.as_any(), &message, 1) {
171                error.write_unraisable(py, None);
172            }
173        });
174    }
175}
