13/// A task in the application catalog and the executable that provides it.
14#[derive(Debug, Clone)]
15pub(crate) struct RegistryApp {
16    pub id: String,
17    binary_name: String,
18    cargo_package: String,
19    override_env: String,
20}
21
22fn registry_app_from_catalog(entry: &ApplicationDefinition) -> Option<RegistryApp> {
23    (entry.kind == ApplicationKind::Task).then(|| RegistryApp {
24        id: entry.id.clone(),
25        binary_name: entry.launch.executable.clone(),
26        cargo_package: entry.launch.cargo_package.clone(),
27        override_env: entry.launch.override_env.clone(),
28    })
29}
30
31fn configured_launch_mode() -> Result<ApplicationLaunchMode, String> {
32    match env::var("CASARS_LAUNCH_MODE") {
33        Ok(value) if value == "installed_suite" => Ok(ApplicationLaunchMode::InstalledSuite),
34        Ok(value) if value == "development_workspace" => {
35            Ok(ApplicationLaunchMode::DevelopmentWorkspace)
36        }
37        Ok(value) => Err(format!(
38            "invalid CASARS_LAUNCH_MODE {value:?}; expected installed_suite or development_workspace"
39        )),
40        Err(env::VarError::NotPresent) if cfg!(debug_assertions) => {
41            Ok(ApplicationLaunchMode::DevelopmentWorkspace)
42        }
43        Err(env::VarError::NotPresent) => Ok(ApplicationLaunchMode::InstalledSuite),
44        Err(error) => Err(format!("read CASARS_LAUNCH_MODE: {error}")),
45    }
46}

48/// A program and the arguments that precede the task's own arguments.
49#[derive(Debug, Clone)]
50pub(crate) struct ResolvedCommand {
51    program: OsString,
52    prefix_args: Vec<OsString>,
53}
54
55impl ResolvedCommand {
56    pub(crate) fn command(&self) -> Command {
57        let mut command = Command::new(&self.program);
58        command.args(&self.prefix_args);
59        command
60    }
61
62    #[cfg(test)]
63    pub(crate) fn direct(program: impl Into<OsString>) -> Self {
64        Self {
65            program: program.into(),
66            prefix_args: Vec::new(),
67        }
68    }
69}

71impl RegistryApp {
72    /// Resolve the provider executable: the `CASARS_<APP>_BIN` override
73    /// first, then the launch mode (an installed suite's sibling binary, or
74    /// `cargo run` in a development workspace).
75    pub(crate) fn resolve_command(&self) -> Result<ResolvedCommand, String> {
76        if let Some(path) = env::var_os(&self.override_env) {
77            return Ok(ResolvedCommand {
78                program: path,
79                prefix_args: Vec::new(),
80            });
81        }
82        match configured_launch_mode()? {
83            ApplicationLaunchMode::InstalledSuite => {
84                let path = sibling_binary(&self.binary_name).ok_or_else(|| {
85                    format!(
86                        "installed-suite executable {:?} is missing beside the casars launcher; set {} to its installed path",
87                        self.binary_name, self.override_env
88                    )
89                })?;
90                Ok(ResolvedCommand {
91                    program: path.into_os_string(),
92                    prefix_args: Vec::new(),
93                })
94            }
95            ApplicationLaunchMode::DevelopmentWorkspace => {
96                let cargo = env::var_os("CARGO").unwrap_or_else(|| OsString::from("cargo"));
97                Ok(ResolvedCommand {
98                    program: cargo,
99                    prefix_args: vec![
100                        OsString::from("run"),
101                        OsString::from("--manifest-path"),
102                        workspace_manifest_path().into_os_string(),
103                        OsString::from("-q"),
104                        OsString::from("-p"),
105                        OsString::from(&self.cargo_package),
106                        OsString::from("--bin"),
107                        OsString::from(&self.binary_name),
108                        OsString::from("--"),
109                    ],
110                })
111            }
112        }
113    }
114}

116/// The catalog task `id`.
117pub(crate) fn resolve_app(id: &str) -> Result<RegistryApp, String> {
118    registered_apps()
119        .into_iter()
120        .find(|app| app.id == id)
121        .ok_or_else(|| {
122            let expected = registered_apps()
123                .into_iter()
124                .map(|app| app.id)
125                .collect::<Vec<_>>()
126                .join(", ");
127            format!("unknown casars task {id:?}; expected one of: {expected}")
128        })
129}
130
131/// Every task in the built-in application catalog.
132pub(crate) fn registered_apps() -> Vec<RegistryApp> {
133    builtin_application_catalog()
134        .expect("built-in application catalog should validate")
135        .applications
136        .iter()
137        .filter_map(registry_app_from_catalog)
138        .collect()
139}

141fn sibling_binary(binary_name: &str) -> Option<PathBuf> {
142    let mut path = env::current_exe().ok()?;
143    path.pop();
144    path.push(binary_name);
145    path.set_extension(env::consts::EXE_EXTENSION);
146    if path.exists() { Some(path) } else { None }
147}
148
149fn workspace_manifest_path() -> PathBuf {
150    let manifest_dir = PathBuf::from(env!("CARGO_MANIFEST_DIR"));
151    manifest_dir
152        .parent()
153        .and_then(|path| path.parent())
154        .unwrap_or_else(|| {
155            panic!("casars manifest dir should live under <workspace>/crates/casars")
156        })
157        .join("Cargo.toml")
158}

160#[cfg(test)]
161mod tests {
162    use std::sync::{Mutex, MutexGuard};

164    use super::*;

166    fn env_lock() -> MutexGuard<'static, ()> {
167        static LOCK: Mutex<()> = Mutex::new(());
168        LOCK.lock().unwrap_or_else(|poisoned| poisoned.into_inner())
169    }
170
171    #[test]
172    fn every_catalog_task_has_a_parameter_surface() {
173        let apps = registered_apps();
174        assert_eq!(apps.len(), 27);
175        for app in apps {
176            let bundle = casa_provider_contracts::builtin_surface_bundle(&app.id)
177                .unwrap_or_else(|error| panic!("{}: {error}", app.id));
178            assert_eq!(bundle.surface.id(), app.id);
179        }
180    }
181
182    #[test]
183    fn surfaces_may_share_one_provider_executable() {
184        assert_eq!(
185            resolve_app("split").unwrap().binary_name,
186            resolve_app("mstransform").unwrap().binary_name
187        );
188        assert_eq!(
189            resolve_app("plotms").unwrap().binary_name,
190            resolve_app("msexplore").unwrap().binary_name
191        );
192        assert_eq!(resolve_app("imstat").unwrap().binary_name, "imexplore");
193    }

195    #[test]
196    fn unknown_tasks_are_rejected_with_the_catalog_list() {
197        let error = resolve_app("tablebrowser").unwrap_err();
198        assert!(error.contains("unknown casars task"), "{error}");
199        assert!(error.contains("imager"), "{error}");
200    }

202    #[test]
203    fn command_override_remains_runtime_only() {
204        let app = resolve_app("imager").unwrap();
205        let variable = "CASARS_IMAGER_BIN";
206        let _guard = env_lock();
207        // SAFETY: the test holds the module's environment lock.
208        unsafe { env::set_var("CASARS_LAUNCH_MODE", "installed_suite") };
209        // SAFETY: the test holds the module's environment lock.
210        unsafe { env::set_var(variable, "/tmp/canonical-imager") };
211        let command = app.resolve_command().unwrap().command();
212        assert_eq!(command.get_program(), "/tmp/canonical-imager");
213        // SAFETY: the test holds the module's environment lock.
214        unsafe { env::remove_var(variable) };
215        // SAFETY: the test holds the module's environment lock.
216        unsafe { env::remove_var("CASARS_LAUNCH_MODE") };
217    }

219    #[test]
220    fn invalid_launch_mode_fails_without_searching_or_switching_modes() {
221        let app = resolve_app("imager").unwrap();
222        let _guard = env_lock();

230        // SAFETY: the test holds the module's environment lock.
