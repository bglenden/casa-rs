2//! `cl.addcomponent`: a component built as CASA's binding builds it, from
3//! its own setters on a new component (`simulate(1)`, then `setlabel`,
4//! `setflux`, the direction, `setshape`, `setspectrum`, `setfreq` and
5//! `setfreqframe`, in that order).
6//!
7//! What a setter raises is raised and the component is not added, as CASA's
8//! `addcomponent` removes the component it began. `setshape` raises what
9//! reading its axes or constructing the shape raises (an axis that does not
10//! read, axes that are not positive or not ordered, a Gaussian's position
11//! angle outside ±2π). It refuses, logging and leaving the component a
12//! point, a shape name it does not know, a limb-darkened disk and an axis
13//! that is not an angle.
14//!
15//! As CASA's does, `setflux` stores the flux's values as Stokes whatever
16//! `polarization` says (a CASA bug casa-rs follows, recorded in
17//! `docs/CASA (C++) bugs.md`), and takes the real parts of complex values.
18//!
19//! Divergence, recorded in `docs/CASA (C++) bugs.md`: **the flux keeps
20//! double precision.** CASA's `setflux` reads the values into a
21//! single-precision `Vector<Complex>` (0.1 is stored as
22//! 0.10000000149011612); casa-rs keeps the values given.
23

39/// The direction `addcomponent` takes for `dir=[]`.
40const DEFAULT_DIRECTION: &str = "J2000 00h00m00.0 90d00m00";

42/// The frequency `addcomponent` takes for `freq=[]`.
43const DEFAULT_FREQUENCY: &str = "LSRK 1.420GHz";

45/// The axes of `GaussianShape`'s default Gaussian, one arcminute, rad.
46const DEFAULT_GAUSSIAN_AXIS: f64 = PI / 10_800.0;

65impl AddComponent {
66    /// The component, as CASA's setters leave it; `lines` gets what
67    /// `setshape` and the frequency reader log.
68    pub(crate) fn component(
69        &self,
70        measures: &Arc<dyn MeasuresProvider>,
71        lines: &mut Vec<LogLine>,
72    ) -> ConvertResult<SkyComponent> {
73        let flux = flux(&self.flux, &self.fluxunit, &self.polarization)?;
74        let dir = match &self.dir {
75            Variant::Empty => Variant::Text(DEFAULT_DIRECTION.to_string()),
76            other => other.clone(),
77        };
78        let direction = measures::direction(&dir, measures)?
79            .ok_or_else(|| ConvertError::runtime("Could not interpret direction parameter"))?;
80        let shape = match self.shape(direction.clone())? {
81            Ok(shape) => shape,
82            Err(message) => {
83                lines.push(LogLine::error(message));
84                ComponentShape::point(direction)
85            }
86        };
87        let kind = spectral_kind(&self.spectrumtype, &self.index)?;
88        let freq = match &self.freq {
89            Variant::Empty => Variant::Text(DEFAULT_FREQUENCY.to_string()),
90            other => other.clone(),
91        };
92        let (frequency, warning) = measures::frequency(&freq)?;
93        lines.extend(warning.map(LogLine::warning));
94        let frequency = frequency
95            .ok_or_else(|| ConvertError::runtime("Could not interpret frequency parameter"))?;
96        let spectrum = SpectralModel::new(frequency, kind);
97        Ok(SkyComponent::new(flux, shape, spectrum).with_label(self.label.clone()))
98    }

100    /// `setshape`'s shape at `direction`: `Ok(Err(message))` when it refuses
101    /// the shape, logging `message` and leaving the shape unchanged, and an
102    /// error for what it raises.
103    fn shape(&self, direction: MDirection) -> ConvertResult<Result<ComponentShape, String>> {
104        let unchanged = "Shape not changed.";
105        let Ok(shape) = ShapeType::parse_casa(&self.shape) else {
106            return Ok(Err(format!(
107                "Could not translate the shape type to a known value. Known types are: Point, \
108                 Gaussian, Disk. {unchanged}"
109            )));
110        };
111        match shape {
112            ShapeType::Point => return Ok(Ok(ComponentShape::point(direction))),
113            ShapeType::LimbDarkenedDisk => {
114                return Ok(Err(format!(
115                    "Limbdarkeneddisk is not implemented yet. {unchanged}"
116                )));
117            }
118            ShapeType::Gaussian | ShapeType::Disk => {}
119        }
120        // `casaQuantity` reads all three, raising for one it cannot read,
121        // before the shape's `fromRecord` refuses one that is not an angle.
122        let quantities = [
123            ("majoraxis", quantity(&self.majoraxis)?),
124            ("minoraxis", quantity(&self.minoraxis)?),
125            ("positionangle", quantity(&self.positionangle)?),
126        ];
127        let mut radians = [0.0; 3];
128        for ((name, quantity), radian) in quantities.iter().zip(&mut radians) {
129            if quantity.unit().val().dim != UnitDim::basis(Dimension::Angle) {
130                return Ok(Err(format!(
131                    "Could not parse the shape parameters. The error was: The {name} field does \
132                     not have angular units {unchanged}"
133                )));
134            }
135            *radian = quantity.get_si_value();
136        }
137        let [major, minor, position_angle] = radians;
138        if let Some(message) = constructor_error(shape, major, minor, position_angle) {
139            return Err(ConvertError::runtime(message));
140        }
141        let raised =
142            |error: casa_components::ComponentError| ConvertError::runtime(error.to_string());
143        let axes = Ellipse::new(major, minor, position_angle).map_err(raised)?;
144        let errors = EllipseErrors::default();
145        let kind = if shape == ShapeType::Gaussian {
146            ShapeKind::Gaussian { axes, errors }
147        } else {
148            ShapeKind::Disk { axes, errors }
149        };
150        ComponentShape::new(direction, kind).map(Ok).map_err(raised)
151    }
152}

154/// What casacore raises making a Gaussian or a disk with these axes and
155/// position angle (rad), or `None` when it makes it.
156///
157/// `GaussianShape` sets the axes through `Gaussian2DParam::setWidth` on its
158/// default one-arcminute circle, which sets the major axis first only when
159/// it is longer than an arcminute, so that order decides which axis is
160/// reported; then `setPA` refuses an angle outside ±2π. `DiskShape` asserts
161/// its axes, taking two within 1e-17 rad as equal, and takes any angle.
162fn constructor_error(
163    shape: ShapeType,
164    major: f64,
165    minor: f64,
166    position_angle: f64,
167) -> Option<&'static str> {
168    if shape == ShapeType::Disk {
169        let minor = if (major - minor).abs() < 1e-17 {
170            major
171        } else {
172            minor
173        };
174        let asserted = major > 0.0 && minor > 0.0 && major >= minor;
175        return (!asserted).then_some(
176            "Failed AlwaysAssert _majorAxis > 0 && _minorAxis > 0 && _majorAxis >=_minorAxis",
177        );
178    }
179    let set_major = |width: f64, minor_axis: f64| {
180        if width <= 0.0 {
181            Some("Gaussian2DParam<T>::setMajorAxis(const T &width) - width must be positive")
182        } else if width < minor_axis {
183            Some(
184                "Gaussian2DParam<T>::setMajorAxis(const T &width) - major axis is smaller than \
185                 minor axis",
186            )
187        } else {
188            None
189        }
190    };
191    let set_minor = |width: f64, major_axis: f64| {
192        if width <= 0.0 {
193            Some("Gaussian2DParam<T>::setMinorAxis(const T &width) - width must be positive")
194        } else if width > major_axis {
195            Some(
196                "Gaussian2DParam<T>::setMinorAxis(const T &width) - minor axis is greater than \
197                 major axis",
198            )
199        } else {
200            None
201        }
202    };
203    let widths = if major.abs() > DEFAULT_GAUSSIAN_AXIS {
204        set_major(major, DEFAULT_GAUSSIAN_AXIS).or_else(|| set_minor(minor, major))
205    } else {
206        set_minor(minor, DEFAULT_GAUSSIAN_AXIS).or_else(|| set_major(major, minor))
207    };
208    widths.or_else(|| {
209        (position_angle.abs() > 2.0 * PI).then_some(
210            "Gaussian2DParam<T>::setPA(const T &pa) - PA must be in radians and between -2pi \
211             and 2pi",
212        )
213    })
214}

216/// `setflux`'s flux: one value (Stokes I) or four, in `unit` (a flux
217/// density), checked in CASA's order (the representation, the unit, then
218/// the values) with its messages.
219///
220/// `polarization` must name a representation, and one value needs Stokes,
221/// but the values are stored as Stokes whatever it names, as CASA's
222/// `FluxRep::setValue` stores them. Complex values (a `complex`, or a
223/// sequence with one) give their real parts, as CASA's setter takes them.
224fn flux(value: &Variant, unit: &str, polarization: &str) -> ConvertResult<Flux> {
225    let polarisation = Polarisation::parse_casa(polarization).map_err(|_| {
226        ConvertError::runtime(
227            "Unknown polarization. Possible values are:\n 'Stokes'  'Linear'  'Circular' ",
228        )
229    })?;
230    let unchanged = "Flux not changed on any components";
231    let mut flux =
232        Flux::new([Complex64::new(0.0, 0.0); 4], unit, Polarisation::Stokes).map_err(|_| {
233            ConvertError::runtime(format!(
234                "The flux units must have the same dimensions as the Jansky\n{unchanged}"
235            ))
236        })?;
237    let values: Vec<f64> = match value {
238        Variant::Empty => vec![0.0; 4],
239        other => complex_values(other)
240            .ok_or_else(|| {
241                ConvertError::runtime("Could not understand the type of flux value variable")
242            })?
243            .iter()
244            .map(|value| value.re)
245            .collect(),
246    };
247    let values = match (values.as_slice(), polarisation) {
248        ([i], Polarisation::Stokes) => [*i, 0.0, 0.0, 0.0],
249        ([a, b, c, d], _) => [*a, *b, *c, *d],
250        _ => {
251            return Err(ConvertError::runtime(format!(
252                "The flux must have one or four elements,\none element can only be used if the \
253                 polarization is 'Stokes'.\n{unchanged}"
254            )));
255        }
256    };
257    flux.set_values(values.map(|value| Complex64::new(value, 0.0)));
258    Ok(flux)
259}

261/// `setspectrum`'s model for `spectrumtype` (its first letter, upper-cased)
262/// and `index`, before `setfreq` gives it its reference frequency.
263fn spectral_kind(spectrumtype: &str, index: &Variant) -> ConvertResult<SpectralKind> {
264    match spectrumtype.to_ascii_uppercase().chars().next() {
265        // `addcomponent` passes `setspectrum`'s one-point default tabulation.
266        Some('T') => Err(ConvertError::runtime(
267            "There need to be at least 2 points in a tabular spectrum",
268        )),
269        Some('S') => {
270            let index = match index {
271                Variant::Number(index) => *index,
272                Variant::Numbers(indices) => indices[0],
273                _ => {
274                    return Err(ConvertError::runtime(
275                        "Unrecognized type for index, must be double or doubleArray",
276                    ));
277                }
278            };
279            Ok(SpectralKind::SpectralIndex(SpectralIndex::new(index)))
280        }
281        Some('P') => match index {
282            Variant::Numbers(coefficients) => Ok(SpectralKind::PowerLogPolynomial(
283                PowerLogPolynomial::new(coefficients.clone()),
284            )),
285            _ => Err(ConvertError::runtime(
286                "for a spectrum='plp', index must be a list of numbers",
287            )),
288        },
289        Some('C') => Ok(SpectralKind::Constant),
290        _ => Err(ConvertError::runtime(format!(
291            "Unkinown spectral type {spectrumtype}"
292        ))),
293    }
