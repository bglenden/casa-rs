2//! CASA's per-record project, epoch and receiver-band tests.
3
4use casa_core::quanta::{Quantity, Unit};
5
6use crate::disk::LogicalRecord;
7use crate::record::CdaId;
8use crate::{BandName, ImportVlaOptions, VlaError};
9
10#[derive(Debug)]
11pub(crate) struct ArchiveSelection {
12    project: Option<String>,
13    start_mjd: f64,
14    stop_mjd: f64,
15    band: Option<(f64, f64)>,
16}
17
18impl ArchiveSelection {
19    pub(crate) fn new(options: &ImportVlaOptions) -> Result<Self, VlaError> {
20        options.require_archivefiles()?;
21        let project = options
22            .project
23            .as_deref()
24            .map(str::trim)
25            .filter(|name| !matches!(*name, "" | "all"));
26        if project.is_some_and(|name| name.len() > 6) {
27            return Err(VlaError::InvalidArgument {
28                argument: "project",
29                message: "The project name must have no more than 6 characters".into(),
30            });
31        }
32        // A malformed date disables that endpoint in CASA. Its empty
33        // arguments instead install the filler defaults.
34        let start_mjd =
35            archive_date(options.starttime.as_deref(), "1970/1/31/00:00:00").unwrap_or(0.0);
36        let stop_mjd = archive_date(options.stoptime.as_deref(), "2199/1/31/23:59:59")
37            .unwrap_or(f64::MAX / 2.0);
38        if start_mjd > stop_mjd {
39            return Err(VlaError::InvalidArgument {
40                argument: "stoptime",
41                message: "The start time must precede the stop time".into(),
42            });
43        }
44        Ok(Self {
45            project: project.map(str::to_ascii_uppercase),
46            start_mjd,
47            stop_mjd,
48            band: options
49                .bandname
50                .map(|band| band_range(band, options.evlabands)),
51        })
52    }
53
54    pub(crate) fn matches_project(&self, project: &str) -> bool {
55        self.project
56            .as_ref()
57            .is_none_or(|name| project.trim().eq_ignore_ascii_case(name))
58    }
59
60    pub(crate) fn matches_time_and_band(&self, record: &LogicalRecord) -> Result<bool, VlaError> {
61        let error = |field, source| VlaError::import(format!("decode {field}: {source}"));
62        let rca = record.rca();
63        let sda = record.sda().map_err(|source| error("SDA", source))?;
64        let day = rca.obs_day().map_err(|source| error("obs day", source))?;
65        let seconds = sda
66            .observation_time_seconds()
67            .map_err(|source| error("observation time", source))?;
68        let time = f64::from(day) + seconds / 86_400.0;
69        if time <= self.start_mjd || time >= self.stop_mjd {
70            return Ok(false);
71        }
72        let Some((low, high)) = self.band else {
73            return Ok(true);
74        };
75        for index in [CdaId::Cda0, CdaId::Cda1, CdaId::Cda2, CdaId::Cda3] {
76            if rca
77                .cda_offset_bytes(index.index())
78                .map_err(|source| error("CDA offset", source))?
79                == 0
80                || !record
81                    .cda(index)
82                    .map_err(|source| error("CDA", source))?
83                    .is_valid()
84            {
85                continue;
86            }
87            let edge = sda
88                .edge_frequency_hz(index)
89                .map_err(|source| error("band edge", source))?;
90            let width = sda
91                .channel_width_hz(index)
92                .map_err(|source| error("channel width", source))?;
93            let channels = sda
94                .n_channels(index)
95                .map_err(|source| error("channel count", source))?;
96            if high > edge && low < edge + f64::from(channels) * width {
97                return Ok(true);
98            }
99        }
100        Ok(false)
101    }
102}

104fn archive_date(value: Option<&str>, default: &str) -> Option<f64> {
105    let value = value.filter(|value| !value.is_empty()).unwrap_or(default);
106    let quantity = Quantity::read(value).ok()?;
107    if quantity.unit().name().is_empty() {
108        return Some(quantity.value());
109    }
110    quantity
111        .get_value_in(&Unit::new("d").ok()?)
112        .ok()
113        .or_else(|| {
114            let radians = quantity.get_value_in(&Unit::new("rad").ok()?).ok()?;
115            Some(radians / std::f64::consts::TAU)
116        })
117}

119fn band_range(band: BandName, evla: bool) -> (f64, f64) {
120    use BandName::*;
121    // Centre and full width in MHz, as selected by CASA's vlafiller.
122    let (centre, width) = match (band, evla) {
123        (Four, _) => (75.0, 48.0),
124        (P, _) => (321.5, 47.0),
125        (L, false) => (1450.0, 600.0),
126        (L, true) => (1500.0, 1000.0),
127        (S, _) => (3000.0, 2000.0),
128        (C, false) => (4650.0, 900.0),
129        (C, true) => (6000.0, 4000.0),
130        (X, false) => (8200.0, 2800.0),
131        (X, true) => (10_000.0, 4000.0),
132        (U, false) => (14_900.0, 2800.0),
133        (U, true) => (15_000.0, 6000.0),
134        (K, false) => (23_300.0, 5000.0),
135        (K, true) => (22_260.0, 8000.0),
136        (Ka, _) => (33_250.0, 13_500.0),
137        (Q, false) => (44_500.0, 13_000.0),
138        (Q, true) => (45_000.0, 10_000.0),
139    };
140    ((centre - width * 0.5) * 1e6, (centre + width * 0.5) * 1e6)
141}
