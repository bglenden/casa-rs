4pub(super) fn nearest_index(coordinates: &[f64], coordinate: f64) -> usize {
5    let upper = coordinates.partition_point(|&value| value < coordinate);
6    let upper = upper.clamp(1, coordinates.len() - 1);
7    if coordinate < (coordinates[upper - 1] + coordinates[upper]) * 0.5 {
8        upper - 1
9    } else {
10        upper
11    }
12}

14pub(super) fn window(count: usize, nearest: usize) -> std::ops::Range<usize> {
15    let first = nearest.saturating_sub(5);
16    let last = (nearest + 5).min(count);
17    if first == 0 {
18        0..count.min(11)
19    } else if last == count {
20        count.saturating_sub(11)..count
21    } else {
22        first..last
23    }
24}

26pub(super) fn interpolate(coordinates: &[f64], values: &[f64], coordinate: f64) -> f64 {
27    if values.len() == 1 {
28        return values[0];
29    }
30    let nearest = nearest_index(coordinates, coordinate);
31    let range = window(values.len(), nearest);
32    interpolate_window(&coordinates[range.clone()], &values[range], coordinate)
33}

35pub(super) fn interpolate_window(x: &[f64], y: &[f64], coordinate: f64) -> f64 {
36    if x.len() == 1 {
37        return y[0];
38    }
39    let nearest = nearest_index(x, coordinate);
40    let upper = x
41        .partition_point(|&value| value < coordinate)
42        .clamp(1, x.len() - 1);
43    let left = upper - 1;
44    let width = x[upper] - x[left];
45    let t = (coordinate - x[left]) / width;
46    let linear = y[left] + t * (y[upper] - y[left]);
47    let min = y.iter().copied().fold(f64::INFINITY, f64::min);
48    let max = y.iter().copied().fold(f64::NEG_INFINITY, f64::max);
49    let half_span = (max - min) * 0.5;
50    let credible = (min - half_span)..=(max + half_span);
51    if x.len() >= 4 {
52        let [a, b] = spline_slopes(x, y, left);
53        let t2 = t * t;
54        let t3 = t2 * t;
55        let cubic = (2. * t3 - 3. * t2 + 1.) * y[left]
56            + (t3 - 2. * t2 + t) * width * a
57            + (-2. * t3 + 3. * t2) * y[upper]
58            + (t3 - t2) * width * b;
59        if credible.contains(&cubic) {
60            return cubic;
61        }
62    }
63    if credible.contains(&linear) {
64        linear
65    } else {
66        y[nearest]
67    }
68}

70fn spline_slopes(x: &[f64], y: &[f64], segment: usize) -> [f64; 2] {
71    // Solve the tridiagonal first-derivative equations for a not-a-knot
72    // cubic spline, using the same end conditions as scipy interp1d.
73    let count = x.len();
74    let mut upper = [0.; 11];
75    let mut rhs = [0.; 11];
76    let step = |row: usize| x[row + 1] - x[row];
77    let slope = |row: usize| (y[row + 1] - y[row]) / step(row);
78    let first = step(0);
79    let second = step(1);
80    let span = first + second;
81    upper[0] = span / second;
82    rhs[0] = ((first + 2. * span) * second * slope(0) + first * first * slope(1)) / (span * second);
83    for i in 1..count {
84        let left = step(i - 1);
85        let (lower, diagonal, above, target) = if i + 1 == count {
86            let prior = step(i - 2);
87            let span = prior + left;
88            (
89                span,
90                prior,
91                0.,
92                (left * left * slope(i - 2) + (2. * span + left) * prior * slope(i - 1)) / span,
93            )
94        } else {
95            let right = step(i);
96            (
97                right,
98                2. * (left + right),
99                left,
100                3. * (right * slope(i - 1) + left * slope(i)),
101            )
102        };
103        let pivot = diagonal - lower * upper[i - 1];
104        upper[i] = above / pivot;
105        rhs[i] = (target - lower * rhs[i - 1]) / pivot;
106    }
107    for i in (segment..count - 1).rev() {
108        rhs[i] -= upper[i] * rhs[i + 1];
109    }
110    [rhs[segment], rhs[segment + 1]]
111}

113pub(super) fn average_over_band(
114    coordinates: &[f64],
115    values: &[f64],
116    band: [f64; 2],
117) -> Option<f64> {
118    let [low, high] = band;
119    if low < *coordinates.first()? || high > *coordinates.last()? || high < low {
120        return None;
121    }
122    let low_value = interpolate(coordinates, values, low);
123    if low == high {
124        return Some(low_value);
125    }
126    let start = coordinates.partition_point(|&value| value < low);
127    let end = coordinates.partition_point(|&value| value <= high);
128    let high_value = interpolate(coordinates, values, high);
129    let mut area = 0.;
130    let (mut previous, mut brightness) = (low, low_value);
131    for (&next, &value) in coordinates[start..end].iter().zip(&values[start..end]) {
132        area += (next - previous) * (brightness + value) * 0.5;
133        (previous, brightness) = (next, value);
134    }
135    area += (high - previous) * (brightness + high_value) * 0.5;
136    Some(area / (high - low))
137}
