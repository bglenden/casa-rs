13/// Why a selected sample did or did not acquire output spectral support.
14#[derive(Debug, Clone, Copy, PartialEq, Eq)]
15pub enum SpectralStencilValidity {
16    /// The valid source sample has one or more output terms.
17    Mapped,
18    /// Source flags or effective weight rejected the sample.
19    Flagged,
20    /// The valid source sample is outside the law's edge support.
21    Unmapped,
22}
23
24/// The sole compiled sparse stencil consumed by prediction and adjoint paths.
25#[derive(Debug, Clone, PartialEq)]
26pub struct SpectralStencilReceipt {
27    contributions: SelectedSpectralContributions,
28    evaluation: SelectedSpectralEvaluation,
29    validity: SpectralStencilValidity,
30    covariance: SpectralCovariance,
31}
32
33impl SpectralStencilReceipt {
34    /// Return sparse coefficients in output-channel order.
35    #[must_use]
36    pub const fn contributions(&self) -> &SelectedSpectralContributions {
37        &self.contributions
38    }
39
40    /// Return the source-backed trace used to compile the stencil.
41    #[must_use]
42    pub const fn evaluation(&self) -> SelectedSpectralEvaluation {
43        self.evaluation
44    }
45
46    /// Return mapped, flagged, or edge-unmapped validity.
47    #[must_use]
48    pub const fn validity(&self) -> SpectralStencilValidity {
49        self.validity
50    }
51
52    /// Return the covariance propagation declaration.
53    #[must_use]
54    pub const fn covariance(&self) -> SpectralCovariance {
55        self.covariance
56    }
57}

59/// A spectral law or geometry could not produce a coherent sparse stencil.
60#[derive(Debug, Clone, Copy, PartialEq, Eq, Error)]
61pub enum SpectralStencilError {
62    /// Output WCS centres or boundaries were invalid.
63    #[error("compiled output spectral geometry is invalid")]
64    InvalidOutputGeometry,
65    /// Identity sampling could not locate the selected source channel.
66    #[error("identity sampling could not locate the selected source channel")]
67    IdentitySourceMismatch,
68    /// Integration exceeded its planner-declared sparse term bound.
69    #[error("spectral integration exceeded its planner term bound")]
70    PlannerTermBound,
71    /// Generated terms were not a valid sparse contribution set.
72    #[error("compiled spectral coefficients are invalid")]
73    InvalidCoefficients,
74}

76/// Compile one paired sparse stencil from a source trace and logical sampling law.
77pub fn compile_spectral_stencil(
78    problem: &CompiledProblem,
79    sample: &SelectedObservationSample,
80    evaluation: SelectedSpectralEvaluation,
81) -> Result<SpectralStencilReceipt, SpectralStencilError> {
82    let law = problem.science().spectral().sampling();
83    let frequency_hz = evaluation.output_frame().centre_hz();
84    // Preserve paired forward support even when flags force the adjoint weight
85    // to zero: MODEL_DATA refreshes every selected cell, including flagged
86    // cells. A constant-basis model evaluates the same coefficient at every
87    // selected frequency rather than treating source-channel ordinal as a
88    // model coefficient.
89    let terms = match problem.reconstruction().basis() {
90        ReconstructionBasis::Constant => one_term(0, 1.0, frequency_hz)?,
91        ReconstructionBasis::ChannelLocal { .. } | ReconstructionBasis::Taylor { .. } => {
92            channel_local_terms(problem, sample, evaluation, frequency_hz)?
93        }
94    };
95    let contributions = SelectedSpectralContributions::new(terms)
96        .ok_or(SpectralStencilError::InvalidCoefficients)?;
97    let validity = if !evaluation.is_valid() {
98        SpectralStencilValidity::Flagged
99    } else if contributions.is_empty() {
100        SpectralStencilValidity::Unmapped
101    } else {
102        SpectralStencilValidity::Mapped
103    };
104    Ok(receipt(
105        contributions,
106        evaluation,
107        validity,
108        law.covariance(),
109    ))
110}

112fn channel_local_terms(
113    problem: &CompiledProblem,
114    sample: &SelectedObservationSample,
115    evaluation: SelectedSpectralEvaluation,
116    frequency_hz: f64,
117) -> Result<SmallVec<[SelectedSpectralContribution; 4]>, SpectralStencilError> {
118    let law = problem.science().spectral().sampling();
119    let spectral = problem.geometry().spectral();
120    let centres = (0..spectral.output_channels())
121        .map(|channel| spectral.channel_centre_hz(channel))
122        .collect::<Option<Vec<_>>>()
123        .ok_or(SpectralStencilError::InvalidOutputGeometry)?;
124    let boundaries = (0..=spectral.output_channels())
125        .map(|boundary| spectral.channel_boundary_hz(boundary))
126        .collect::<Option<Vec<_>>>()
127        .ok_or(SpectralStencilError::InvalidOutputGeometry)?;
128    validate_axis(&centres, &boundaries)?;
129    match law.kernel() {
130        SpectralKernel::Identity => identity_terms(problem, sample, frequency_hz),
131        SpectralKernel::Nearest => Ok(nearest_terms(&centres, &boundaries, frequency_hz)),
132        SpectralKernel::Linear => Ok(linear_terms(&centres, &boundaries, frequency_hz)),
133        SpectralKernel::Cubic => Ok(cubic_terms(&centres, frequency_hz)),
134        SpectralKernel::ChannelIntegration { maximum_terms } => integration_terms(
135            &boundaries,
136            evaluation.output_frame().boundaries_hz(),
137            law.edge_policy(),
138            maximum_terms,
139            frequency_hz,
140        ),
141    }
142}

158fn validate_axis(centres: &[f64], boundaries: &[f64]) -> Result<(), SpectralStencilError> {
159    if centres.is_empty()
160        || boundaries.len() != centres.len() + 1
161        || centres
162            .iter()
163            .chain(boundaries)
164            .any(|value| !value.is_finite() || *value <= 0.0)
165    {
166        return Err(SpectralStencilError::InvalidOutputGeometry);
167    }
168    let direction = (centres[centres.len() - 1] - centres[0]).signum();
169    if centres.len() > 1
170        && (direction == 0.0
171            || centres
172                .windows(2)
173                .any(|pair| (pair[1] - pair[0]).signum() != direction))
174    {
175        return Err(SpectralStencilError::InvalidOutputGeometry);
176    }
177    Ok(())
178}

180fn identity_terms(
181    problem: &CompiledProblem,
182    sample: &SelectedObservationSample,
183    frequency_hz: f64,
184) -> Result<SmallVec<[SelectedSpectralContribution; 4]>, SpectralStencilError> {
185    if let Some(transform) = problem.visibility_transform()
186        && let Some(rule) =
187            transform.rule(sample.metadata.field_id, sample.address.spectral_window_id)
188    {
189        let ordinal = rule
190            .channels()
191            .iter()
192            .filter(|channel| channel.use_role().contributes_to_output())
193            .position(|channel| channel.channel_index() == sample.address.channel_index)
194            .ok_or(SpectralStencilError::IdentitySourceMismatch)?;
195        if ordinal >= problem.geometry().spectral().output_channels() {
196            return Ok(SmallVec::new());
197        }
198        return one_term(ordinal, 1.0, frequency_hz);
199    }
200    let source = problem
201        .inputs()
202        .observation_snapshot()
203        .sources()
204        .iter()
205        .find(|source| source.identity() == sample.address.measurement_set)
206        .ok_or(SpectralStencilError::IdentitySourceMismatch)?;
207    let selection = source
208        .selection()
209        .spectral_windows()
210        .iter()
211        .find(|selection| selection.spectral_window_id() == sample.address.spectral_window_id)
212        .ok_or(SpectralStencilError::IdentitySourceMismatch)?;
213    let ordinal = selection
214        .channel_indices()
215        .iter()
216        .position(|channel| *channel == sample.address.channel_index)
217        .ok_or(SpectralStencilError::IdentitySourceMismatch)?;
218    if ordinal >= problem.geometry().spectral().output_channels() {
219        return Ok(SmallVec::new());
220    }
221    one_term(ordinal, 1.0, frequency_hz)
222}

224fn nearest_terms(
225    centres: &[f64],
226    boundaries: &[f64],
227    frequency_hz: f64,
228) -> SmallVec<[SelectedSpectralContribution; 4]> {
229    let Some((index, _)) = centres.iter().enumerate().min_by(|(_, left), (_, right)| {
230        (*left - frequency_hz)
231            .abs()
232            .total_cmp(&(*right - frequency_hz).abs())
233    }) else {
234        return SmallVec::new();
235    };
236    let low = boundaries[index].min(boundaries[index + 1]);
237    let high = boundaries[index].max(boundaries[index + 1]);
238    if !(low..=high).contains(&frequency_hz) {
239        return SmallVec::new();
240    }
241    one_term(index, 1.0, frequency_hz).unwrap_or_default()
242}

244fn linear_terms(
245    centres: &[f64],
246    boundaries: &[f64],
247    frequency_hz: f64,
248) -> SmallVec<[SelectedSpectralContribution; 4]> {
249    if centres.len() == 1 {
250        return nearest_terms(centres, boundaries, frequency_hz);
251    }
252    interpolation_interval(centres, frequency_hz)
253        .and_then(|index| {
254            let first = centres[index];
255            let second = centres[index + 1];
256            let upper = (frequency_hz - first) / (second - first);
257            sparse_terms([(index, 1.0 - upper), (index + 1, upper)], frequency_hz)
258        })
259        .unwrap_or_default()
260}
261
262fn cubic_terms(centres: &[f64], frequency_hz: f64) -> SmallVec<[SelectedSpectralContribution; 4]> {
263    if centres.len() < 4 {
264        return SmallVec::new();
265    }
266    let Some(interval) = interpolation_interval(centres, frequency_hz) else {
267        return SmallVec::new();
268    };
269    let bracket = interval + 1;
270    let start = if bracket > 1 && bracket < centres.len() - 1 {
271        bracket - 2
272    } else if bracket <= 1 {
273        0
274    } else {
275        centres.len() - 4
276    };
277    let mut terms = SmallVec::<[(usize, f64); 4]>::new();
278    for index in start..start + 4 {
279        let mut coefficient = 1.0;
280        for other in start..start + 4 {
281            if other != index {
282                coefficient *= (frequency_hz - centres[other]) / (centres[index] - centres[other]);
283            }
284        }
285        terms.push((index, coefficient));
286    }
287    sparse_terms(terms, frequency_hz).unwrap_or_default()
288}

290fn integration_terms(
291    output_boundaries_hz: &[f64],
292    source_boundaries_hz: [f64; 2],
293    edge_policy: SpectralEdgePolicy,
294    maximum_terms: usize,
295    frequency_hz: f64,
296) -> Result<SmallVec<[SelectedSpectralContribution; 4]>, SpectralStencilError> {
297    let source_low = source_boundaries_hz[0].min(source_boundaries_hz[1]);
298    let source_high = source_boundaries_hz[0].max(source_boundaries_hz[1]);
299    let axis_end = output_boundaries_hz[output_boundaries_hz.len() - 1];
300    let axis_low = output_boundaries_hz[0].min(axis_end);
301    let axis_high = output_boundaries_hz[0].max(axis_end);
302    if edge_policy == SpectralEdgePolicy::CompleteSupport
303        && (source_low < axis_low || source_high > axis_high)
304    {
305        return Ok(SmallVec::new());
306    }
307    let mut terms = SmallVec::<[SelectedSpectralContribution; 4]>::new();
308    for (output_channel, pair) in output_boundaries_hz.windows(2).enumerate() {
309        let output_low = pair[0].min(pair[1]);
310        let output_high = pair[0].max(pair[1]);
311        let overlap = source_high.min(output_high) - source_low.max(output_low);
312        if overlap > 0.0 {
313            if terms.len() == maximum_terms {
314                return Err(SpectralStencilError::PlannerTermBound);
315            }
316            terms.push(
317                SelectedSpectralContribution::new(
318                    u32::try_from(output_channel)
319                        .map_err(|_| SpectralStencilError::InvalidCoefficients)?,
320                    overlap / (output_high - output_low),
321                    frequency_hz,
322                )
323                .ok_or(SpectralStencilError::InvalidCoefficients)?,
324            );
325        }
326    }
327    Ok(terms)
328}
329
330fn interpolation_interval(centres: &[f64], frequency_hz: f64) -> Option<usize> {
331    centres.windows(2).position(|pair| {
332        pair[0].min(pair[1]) <= frequency_hz && frequency_hz <= pair[0].max(pair[1])
333    })
334}
335
336fn one_term(
337    output_channel: usize,
338    factor: f64,
339    frequency_hz: f64,
340) -> Result<SmallVec<[SelectedSpectralContribution; 4]>, SpectralStencilError> {
341    sparse_terms([(output_channel, factor)], frequency_hz)
342        .ok_or(SpectralStencilError::InvalidCoefficients)
343}
344
345fn sparse_terms(
346    terms: impl IntoIterator<Item = (usize, f64)>,
347    frequency_hz: f64,
348) -> Option<SmallVec<[SelectedSpectralContribution; 4]>> {
349    terms
350        .into_iter()
351        .filter(|(_, factor)| *factor != 0.0)
352        .map(|(output_channel, factor)| {
353            SelectedSpectralContribution::new(
354                u32::try_from(output_channel).ok()?,
355                factor,
356                frequency_hz,
357            )
358        })
359        .collect()
360}

366    #[test]
367    fn t35_one_channel_identity_matches_constant_basis_and_is_exactly_paired() {
368        let stencil = one_term(0, 1.0, 1.4e9).expect("identity stencil");
369        assert_eq!(
370            stencil
371                .iter()
372                .map(|term| (term.output_channel(), term.factor()))
373                .collect::<Vec<_>>(),
374            vec![(0, 1.0)]
375        );
376
377        let model = 2.5_f64;
378        let visibility = -3.0_f64;
379        let weight = 4.0_f64;
380        let channel_prediction = stencil
381            .iter()
382            .map(|term| term.factor() * model)
383            .sum::<f64>();
384        let channel_dirty = stencil
385            .iter()
386            .map(|term| term.factor() * weight * visibility)
387            .sum::<f64>();
388        let channel_psf = stencil
389            .iter()
390            .map(|term| term.factor() * weight * term.factor())
391            .sum::<f64>();
392        let channel_sum_weight = weight;
393        assert_eq!(channel_prediction, model);
394        assert_eq!(channel_dirty, weight * visibility);
395        assert_eq!(channel_psf, weight);
396        assert_eq!(channel_sum_weight, weight);
397
398        let lhs = channel_prediction * visibility;
399        let adjoint_visibility = stencil
400            .iter()
401            .map(|term| term.factor() * visibility)
402            .sum::<f64>();
403        let rhs = model * adjoint_visibility;
404        assert_eq!(lhs, rhs, "prediction and adjoint must use the same stencil");
405    }

409        let centres = [10.0, 20.0, 30.0, 40.0];

493    fn t36_nearest_linear_cubic_edge_and_covariance_match_casacore_oracles() {
