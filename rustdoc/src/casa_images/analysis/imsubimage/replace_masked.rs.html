12/// The pixels and validity selection for an in-place mask edit.
13pub struct MaskReplacementOptions<'a> {
14    pub expression: &'a str,
15    pub region: Option<SubimageRegion<'a>>,
16    pub mask: &'a str,
17    pub update: bool,
18    pub stretch: bool,
19}
20
21/// The result after writing replacement pixels.
22pub enum MaskReplacementOutcome {
23    Applied,
24    /// An external or regional mask supplied the invalid pixels, but the
25    /// image has no native pixel mask for the requested update. CASA reports
26    /// this after writing the replacement values.
27    MissingNativeMask,
28}
29
30/// Check an edit's region before acquiring the image write lock.
31///
32/// # Errors
33/// Returns an error if the region cannot be resolved or is compound.
34pub fn validate_mask_replacement_region<T: ImagePixel>(
35    image: &dyn ImageInterface<T>,
36    region: SubimageRegion<'_>,
37) -> Result<(), ImageError> {
38    replacement_selection(image, Some(region)).map(|_| ())
39}
40
41fn replacement_selection<T: ImagePixel>(
42    image: &dyn ImageInterface<T>,
43    region: Option<SubimageRegion<'_>>,
44) -> Result<SubimageSelection, ImageError> {
45    let selection = SubimageSelection::resolve(image, None, region, None, None)?;
46    if selection.has_multiple_regions {
47        return Err(ImageError::CasaSelection(
48            "Exception: Only a single n-dimensional rectangular region is supported.".into(),
49        ));
50    }
51    Ok(selection)
52}

54/// Replace invalid single-precision real pixels with a numerical expression.
55pub fn replace_masked_real_pixels(
56    image: &mut dyn MutableImageInterface<f32>,
57    options: MaskReplacementOptions<'_>,
58) -> Result<MaskReplacementOutcome, ImageError> {
59    replace_masked(
60        image,
61        options,
62        |expression, shape| match evaluate_replacement_pixels(expression, shape, false)? {
63            ReplacementPixels::Real(values) => Ok(values.mapv(|value| value as f32)),
64            ReplacementPixels::Complex(_) => unreachable!("real replacement expression"),
65        },
66    )
67}
68
69/// Replace invalid single-precision complex pixels with a numerical expression.
70pub fn replace_masked_complex_pixels(
71    image: &mut dyn MutableImageInterface<Complex32>,
72    options: MaskReplacementOptions<'_>,
73) -> Result<MaskReplacementOutcome, ImageError> {
74    replace_masked(image, options, |expression, shape| {
75        Ok(
76            match evaluate_replacement_pixels(expression, shape, true)? {
77                ReplacementPixels::Real(values) => {
78                    values.mapv(|value| Complex32::new(value as f32, 0.0))
79                }
80                ReplacementPixels::Complex(values) => {
81                    values.mapv(|value| Complex32::new(value.re as f32, value.im as f32))
82                }
83            },
84        )
85    })
86}

88fn replace_masked<T: ImagePixel>(
89    image: &mut dyn MutableImageInterface<T>,
90    options: MaskReplacementOptions<'_>,
91    values: impl FnOnce(&str, &[usize]) -> Result<ArrayD<T>, ImageError>,
92) -> Result<MaskReplacementOutcome, ImageError> {
93    if options.expression.is_empty() {
94        return Err(ImageError::CasaSelection(
95            "You must specify an expression".into(),
96        ));
97    }
98    let selection = replacement_selection(image, options.region)?;
99    let stride = vec![1; image.ndim()];
100    let mut valid = image.get_mask_slice(&selection.start, &selection.shape, &stride)?;
101    selection.apply_selection_masks(&mut valid);
102    if let Some(external) = evaluate_mask(
103        Some(options.mask),
104        image.shape(),
105        image.coordinates(),
106        options.stretch,
107        (&selection.start, &selection.shape),
108    )? {
109        and_opt_mask(&mut valid, external);
110    }
111    let valid = valid.ok_or_else(|| {
112        ImageError::CasaSelection("Exception: Selected region of image has no mask.".into())
113    })?;
114    if valid.iter().all(|&pixel| pixel) {
115        return Err(ImageError::CasaSelection(
116            "Exception: Mask for selected region has no bad pixels.".into(),
117        ));
118    }
119    let replacements = values(options.expression, &selection.shape)?;
120    let mut pixels = image.get_slice(&selection.start, &selection.shape, &stride)?;
121    if replacements.ndim() == 0 {
122        let replacement = replacements[IxDyn(&[])];
123        Zip::from(&mut pixels)
124            .and(&valid)
125            .for_each(|pixel, &valid| {
126                if !valid {
127                    *pixel = replacement;
128                }
129            });
130    } else {
131        Zip::from(&mut pixels)
132            .and(&valid)
133            .and(&replacements)
134            .for_each(|pixel, &valid, &replacement| {
135                if !valid {
136                    *pixel = replacement;
137                }
138            });
139    }
140    image.put_slice(&pixels, &selection.start)?;
141    if options.update {
142        let Some(name) = image.active_pixel_mask_name() else {
143            return Ok(MaskReplacementOutcome::MissingNativeMask);
144        };
145        let all_valid = ArrayD::from_elem(IxDyn(&selection.shape), true);
146        image.put_mask_slice(&name, &all_valid, &selection.start)?;
147    }
148    Ok(MaskReplacementOutcome::Applied)
149}
