2//! An existing binary CLEAN mask, read once on its checked grid. CASA's
3//! SynthesisDeconvolver::setupMask reuses it when no new user mask is given.
4//!
5//! A mask has a plane per Stokes parameter and channel, and is reused
6//! plane by plane: as one support when its planes agree, as a static mask's
7//! do, else as a support for each plane, as an automatic mask's may.
8
9use std::path::{Path, PathBuf};
10
11use casa_images::PagedImage;
12use casa_imaging_deconvolution::MaskPlanes;
13
14use crate::PrepareError;
15
16const CACHE_BYTES: usize = 64 * 1024;
17
18pub(crate) struct MaskInput {
19    path: PathBuf,
20    shape: Vec<usize>,
21    tile_bytes: u64,
22}
23
24impl MaskInput {
25    pub(crate) fn open(path: &Path) -> Result<Self, PrepareError> {
26        let image = PagedImage::<f32>::open_with_cache(path, CACHE_BYTES).map_err(|error| {
27            PrepareError::Restart {
28                path: path.into(),
29                reason: error.to_string(),
30            }
31        })?;
32        let tile_bytes = image
33            .tile_shape()
34            .iter()
35            .fold(4_u64, |n, &side| n.saturating_mul(side as u64));
36        Ok(Self {
37            path: path.into(),
38            shape: image.shape().to_vec(),
39            tile_bytes,
40        })
41    }
42
43    fn pixels(&self) -> u64 {
44        self.shape.iter().map(|&side| side as u64).product()
45    }
46
47    /// The read of every plane and its conversion, two support copies of
48    /// every plane (compile and run clone their plans), the bounded tile
49    /// cache and two decoded/I/O tile buffers.
50    pub(crate) fn reading_bytes(&self) -> u64 {
51        self.pixels()
52            .saturating_mul(2 * 4)
53            .saturating_add(self.pixels().saturating_mul(2))
54            .saturating_add(self.tile_bytes.saturating_mul(2))
55            .saturating_add(CACHE_BYTES as u64)
56    }
57
58    /// Both the original and the cloned static reconstruction-mask plan of
59    /// the `planes` of `cells` pixels each that [`Self::read`] kept.
60    pub(crate) fn retained_bytes(planes: MaskPlanes, cells: usize) -> u64 {
61        (cells as u64)
62            .saturating_mul(planes.count() as u64)
63            .saturating_mul(2)
64    }
65
66    /// The mask's support, its planes as CASA's `.mask` holds them: one
67    /// plane when every Stokes plane and channel agrees, as a static mask
68    /// casa-rs writes does, else a plane for each.
69    pub(crate) fn read(self) -> Result<(MaskPlanes, Box<[bool]>), PrepareError> {
70        let refuse = |reason| PrepareError::Restart {
71            path: self.path.clone(),
72            reason,
73        };
74        // Reopen after admission: even the bounded tiled reader can allocate one
75        // physical tile larger than its requested cache budget.
76        let image = PagedImage::<f32>::open_with_cache(&self.path, CACHE_BYTES)
77            .map_err(|error| refuse(error.to_string()))?;
78        let pixels = image.get().map_err(|error| refuse(error.to_string()))?;
79        let binary = |value: f32| match value {
80            0.0 => Ok(false),
81            1.0 => Ok(true),
82            _ => Err(refuse(
83                "continuation with a non-binary CLEAN mask is not implemented yet".into(),
84            )),
85        };
86        // The grid check holds the mask to the image's four axes,
87        // [x, y, stokes, channel]; reconstruction support is x-major (y
88        // fastest), unlike model samples, plane after plane, channel-major.
89        let &[nx, ny, stokes, channels] = self.shape.as_slice() else {
90            return Err(refuse(format!(
91                "a CLEAN mask has four axes, not {:?}",
92                self.shape
93            )));
94        };
95        let mut support = Vec::with_capacity(nx * ny * stokes * channels);
96        for channel in 0..channels {
97            for polarization in 0..stokes {
98                for x in 0..nx {
99                    for y in 0..ny {
100                        support.push(binary(pixels[&[x, y, polarization, channel][..]])?);
101                    }
102                }
103            }
104        }
105        let cells = nx * ny;
106        if support
107            .chunks(cells)
108            .all(|plane| plane == &support[..cells])
109        {
110            support.truncate(cells);
111            return Ok((MaskPlanes::SHARED, support.into_boxed_slice()));
112        }
113        let planes =
114            MaskPlanes::new(channels, stokes).map_err(|error| refuse(error.to_string()))?;
115        Ok((planes, support.into_boxed_slice()))
116    }
117}
