2//! Translate developer sky inputs to native images/component lists.
3use super::*;
4use casa_components::{
5    ComponentList, ComponentShape, Ellipse, EllipseErrors, Flux, ShapeKind, SkyComponent,
6    SpectralKind, SpectralModel, TabularSpectrum,
7};
8use casa_core::measures::frequency::MFrequency;
9use casa_imaging_runtime::{Demand, admit};
10use std::path::Path;
11
12pub(super) fn predict(
13    request: &ObservationRequest,
14    sm: &mut Simulator,
15    root: &Path,
16    resources: SessionResources,
17) -> Result<()> {
18    sm.set_vp(&crate::VoltagePatternSpec {
19        do_squint: false,
20        use_default_vp: request.vp_table.is_none(),
21        vp_table: request
22            .vp_table
23            .as_ref()
24            .map_or_else(String::new, |path| path.to_string_lossy().into_owned()),
25        ..crate::VoltagePatternSpec::default()
26    })?;
27    let mut prediction = crate::PredictSpec::default();
28    let model = request.model.clone().unwrap_or(SkyModel::FitsImage {
29        path: request.model_image.clone(),
30        model_peak_jy_per_pixel: request.model_peak_jy_per_pixel,
31        direction_reference_rad: None,
32        cell_size_rad: None,
33    });
34    match model {
35        SkyModel::FitsImage {
36            path,
37            model_peak_jy_per_pixel,
38            direction_reference_rad,
39            cell_size_rad,
40        } => {
41            let shape = if path.is_dir() {
42                casa_images::AnyPagedImage::open(&path)
43                    .map_err(error)?
44                    .shape()
45                    .to_vec()
46            } else {
47                casa_images::fits_image_shape(&path).map_err(error)?
48            };
49            let bytes = shape
50                .iter()
51                .try_fold(64u64, |n, &axis| n.checked_mul(axis as u64))
52                .and_then(|n| n.checked_add(32 << 20))
53                .ok_or_else(|| error("model dimensions overflow"))?;
54            let _memory = admit(
55                resources.host(),
56                resources.policy(),
57                &Demand {
58                    phase: "synthetic model preparation",
59                    memory: bytes,
60                },
61            )
62            .map_err(error)?;
63            let output = root.join("model.image");
64            crate::simutil::skymodel::modify_model(
65                &path,
66                &output,
67                &crate::simutil::skymodel::ModelOverrides {
68                    inbright: model_peak_jy_per_pixel
69                        .map_or_else(String::new, |n| format!("{n}Jy/pixel")),
70                    indirection: direction_reference_rad
71                        .map_or_else(Vec::new, |d| vec![format!("J2000 {}rad {}rad", d[0], d[1])]),
72                    incell: cell_size_rad
73                        .map_or_else(Vec::new, |c| c.map(|n| format!("{n}rad")).to_vec()),
74                    incenter: String::new(),
75                    inwidth: String::new(),
76                },
77            )
78            .map_err(error)?;
79            prediction.images.push(output);
80        }
81        SkyModel::AnalyticComponents {
82            path, components, ..
83        } => {
84            let components = if components.is_empty() {
85                let path = path.ok_or_else(|| error("analytic model has no components or path"))?;
86                serde_json::from_reader::<_, AnalyticComponentModel>(
87                    std::fs::File::open(path).map_err(error)?,
88                )
89                .map_err(error)?
90                .components
91            } else {
92                components
93            };
94            let mut list = ComponentList::new();
95            for component in &components {
96                list.push(component_model(request, component)?);
97            }
98            let path = root.join("model.cl");
99            list.write_table(&path).map_err(error)?;
100            prediction.component_lists.push(path);
101        }
102    }
103    sm.predict(&prediction)
104}

106fn component_model(
107    request: &ObservationRequest,
108    component: &AnalyticComponent,
109) -> Result<SkyComponent> {
110    let (name, l, m, spectrum, ellipse) = match component {
111        AnalyticComponent::Point {
112            name,
113            l_rad,
114            m_rad,
115            spectrum,
116        } => (name, *l_rad, *m_rad, spectrum, None),
117        AnalyticComponent::Gaussian {
118            name,
119            l_rad,
120            m_rad,
121            spectrum,
122            major_fwhm_rad,
123            minor_fwhm_rad,
124            position_angle_rad,
125        } => (
126            name,
127            *l_rad,
128            *m_rad,
129            spectrum,
130            Some((
131                *major_fwhm_rad,
132                if *minor_fwhm_rad > 0. {
133                    *minor_fwhm_rad
134                } else {
135                    *major_fwhm_rad
136                },
137                *position_angle_rad,
138            )),
139        ),
140    };
141    let [ra, dec] = request.phase_center_rad;
142    let n = (1. - l * l - m * m).sqrt();
143    let direction = MDirection::from_angles(
144        ra + l.atan2(n * dec.cos() - m * dec.sin()),
145        (m * dec.cos() + n * dec.sin()).asin(),
146        DirectionRef::J2000,
147    );
148    let shape = if let Some((major, minor, angle)) = ellipse {
149        ComponentShape::new(
150            direction,
151            ShapeKind::Gaussian {
152                axes: Ellipse::new(major, minor, angle).map_err(error)?,
153                errors: EllipseErrors::default(),
154            },
155        )
156        .map_err(error)?
157    } else {
158        ComponentShape::point(direction)
159    };
160    let reference = request.spectral_windows[0].reference_frequency_hz();
161    let mut samples = Vec::new();
162    let [q, u, v] = request.polarization_setup.source_stokes_fractional_quv;
163    for window in &request.spectral_windows {
164        for channel in 0..window.channel_count {
165            let flux = spectrum.flux_for_channel(window, channel, reference);
166            samples.push((
167                window.start_frequency_hz + channel as f64 * window.channel_width_hz,
168                [flux, q * flux, u * flux, v * flux],
169            ));
170        }
171    }
172    samples.sort_by(|a, b| a.0.total_cmp(&b.0));
173    samples.dedup_by(|a, b| a.0 == b.0);
174    if samples.len() == 1 {
175        let (frequency, [i, q, u, v]) = samples[0];
176        return Ok(SkyComponent::new(
177            Flux::stokes_jy(i, q, u, v),
178            shape,
179            SpectralModel::constant(MFrequency::new(frequency, FrequencyRef::LSRK)),
180        )
181        .with_label(name.clone().unwrap_or_default()));
182    }
183    let (frequencies, fluxes) = samples.into_iter().unzip();
184    let table = TabularSpectrum::new(FrequencyRef::LSRK, frequencies, fluxes).map_err(error)?;
185    Ok(SkyComponent::new(
186        Flux::stokes_jy(
187            spectrum.flux_jy,
188            q * spectrum.flux_jy,
189            u * spectrum.flux_jy,
190            v * spectrum.flux_jy,
191        ),
192        shape,
193        SpectralModel::new(
194            MFrequency::new(reference, FrequencyRef::LSRK),
195            SpectralKind::Tabular(table),
196        ),
197    )
198    .with_label(name.clone().unwrap_or_default()))
199}
