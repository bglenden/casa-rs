12fn quantities(argument: Option<Variant>, label: &str) -> PyResult<Vec<Quantity>> {
13    let strings = match argument {
14        Some(Variant::Text(value)) => {
15            let separator = if value.contains(',') { ',' } else { ' ' };
16            if value.contains(separator) {
17                value
18                    .split(separator)
19                    .filter(|part| !part.is_empty() && *part != " ")
20                    .map(str::to_owned)
21                    .collect()
22            } else {
23                vec![value]
24            }
25        }
26        Some(Variant::Texts(values)) => values,
27        _ => return Err(raised(format!("Cannot understand {label} values"))),
28    };
29    strings
30        .iter()
31        .map(|value| {
32            if value.contains("pix") {
33                // CASA accepts this unit in the list reader. All beam
34                // components that use it fail the angular-unit check.
35                return Quantity::new(0.0, "pix").map_err(raised);
36            }
37            read_quantity(value, &format!(" In converting quantity {value}")).map_err(Into::into)
38        })
39        .collect()
40}

42fn stored(values: &[Quantity]) -> PyResult<StoredGaussianBeam> {
43    let major = &values[0];
44    let minor = values.get(1).unwrap_or(major);
45    let zero = Quantity::new(0.0, "deg").expect("built-in unit");
46    let pa = values.get(2).unwrap_or(&zero);
47    let rad = Unit::new("rad").expect("built-in unit");
48    for (label, quantity) in [
49        ("Major axis", major),
50        ("Minor axis", minor),
51        ("Position angle", pa),
52    ] {
53        if !quantity.unit().conformant(&rad) {
54            return Err(raised(format!(
55                "Exception: {label} must have angular units ({} is not)..",
56                quantity.unit().name()
57            )));
58        }
59    }
60    for (label, quantity) in [("Major axis", major), ("Minor axis", minor)] {
61        if quantity.value() < 0.0 {
62            return Err(raised(format!(
63                "Exception: {label} cannot be less than zero.."
64            )));
65        }
66    }
67    if major.value() < casacore::get_conformant_value(minor, major.unit()).map_err(raised)? {
68        return Err(raised(
69            "Exception: Major axis must be greater or equal to minor axis.",
70        ));
71    }
72    StoredGaussianBeam::new(major.clone(), minor.clone(), pa.clone()).map_err(raised)
73}

75fn radians(beam: &StoredGaussianBeam) -> PyResult<GaussianBeam> {
76    let unit = Unit::new("rad").expect("built-in unit");
77    let angular =
78        |quantity: &Quantity| casacore::get_conformant_value(quantity, &unit).map_err(raised);
79    Ok(GaussianBeam::new(
80        angular(beam.component(BeamComponent::Major))?,
81        angular(beam.component(BeamComponent::Minor))?,
82        angular(&beam.wrapped_position_angle().map_err(raised)?)?,
83    ))
84}

86/// CASA's tool distinguishes point sources, failures, and resolved fits.
87/// The point-source limit depends on the width of the smaller beam.
88fn fit_size(total: GaussianBeam, applied: GaussianBeam) -> Option<(GaussianBeam, bool)> {
89    let (total_sin, total_cos) = (total.position_angle.sin(), total.position_angle.cos());
90    let (beam_sin, beam_cos) = (applied.position_angle.sin(), applied.position_angle.cos());
91    let alpha = (total.major * total_cos).powi(2) + (total.minor * total_sin).powi(2)
92        - (applied.major * beam_cos).powi(2)
93        - (applied.minor * beam_sin).powi(2);
94    let beta = (total.major * total_sin).powi(2) + (total.minor * total_cos).powi(2)
95        - (applied.major * beam_sin).powi(2)
96        - (applied.minor * beam_cos).powi(2);
97    let gamma = 2.0
98        * ((total.minor.powi(2) - total.major.powi(2)) * total_sin * total_cos
99            - (applied.minor.powi(2) - applied.major.powi(2)) * beam_sin * beam_cos);
100    let sum = alpha + beta;
101    let separation = ((alpha - beta).powi(2) + gamma.powi(2)).sqrt();
102    let smallest = total.minor.min(applied.minor);
103    let limit = 0.1 * smallest * smallest;
104    if alpha < 0.0 || beta < 0.0 || sum < separation {
105        return (0.5 * (sum - separation) < limit && alpha > -limit && beta > -limit)
106            .then_some((applied, true));
107    }
108    let pa = if gamma.abs() + (alpha - beta).abs() == 0.0 {
109        0.0
110    } else {
111        0.5 * (-gamma).atan2(alpha - beta)
112    };
113    Some((
114        GaussianBeam::new(
115            (0.5 * (sum + separation)).sqrt(),
116            (0.5 * (sum - separation)).sqrt(),
117            pa,
118        ),
119        false,
120    ))
121}

123fn fit_record(beam: GaussianBeam) -> PyResult<Vec<(&'static str, CasaValue)>> {
124    let angle = Quantity::new(beam.position_angle, "rad").map_err(raised)?;
125    let angle = casa_images::beam::wrap_position_angle(&angle).map_err(raised)?;
126    let unit = Unit::new("deg").expect("built-in unit");
127    Ok(vec![
128        (
129            "major",
130            CasaValue::quantity(beam.major_in("arcsec").map_err(raised)?, "arcsec"),
131        ),
132        (
133            "minor",
134            CasaValue::quantity(beam.minor_in("arcsec").map_err(raised)?, "arcsec"),
135        ),
136        (
137            "pa",
138            CasaValue::quantity(
139                casacore::get_conformant_value(&angle, &unit).map_err(raised)?,
140                "deg",
141            ),
142        ),
143    ])
144}

146fn describe(beam: &StoredGaussianBeam) -> String {
147    let quantity = |component| {
148        let quantity = beam.component(component);
149        format!(
150            "{} {}",
151            casacore::stream_double(quantity.value()),
152            quantity.unit().name()
153        )
154    };
155    format!(
156        "major: {}, minor: {}, pa: {}",
157        quantity(BeamComponent::Major),
158        quantity(BeamComponent::Minor),
159        quantity(BeamComponent::PositionAngle)
160    )
161}

163#[pymethods]
164impl PyImageTool {
165    #[allow(clippy::unused_self)]
166    #[pyo3(signature = (source=None, convolved=None), text_signature = "($self, source=[], convolved=[])")]
167    fn beamforconvolvedsize(
168        &self,
169        py: Python<'_>,
170        source: Option<Variant>,
171        convolved: Option<Variant>,
172    ) -> PyResult<Py<PyAny>> {
173        py.detach(|| {
174            let source = quantities(source, "source")?;
175            if source.len() != 3 {
176                return Err(raised("Cannot understand source values"));
177            }
178            let convolved = quantities(convolved, "target")?;
179            if convolved.len() < 3 {
180                return Err(raised("Cannot understand target values"));
181            }
182            let source = stored(&source)?;
183            let convolved = stored(&convolved)?;
184            let Some((fit, false)) = fit_size(radians(&convolved)?, radians(&source)?) else {
185                return Err(raised(format!(
186                    "Unable to reach target resolution of {} Input source {} is probably too large.",
187                    describe(&convolved), describe(&source),
188                )));
189            };
190            Ok(CasaValue::record(fit_record(fit)?))
191        })?
192        .into_python(py)
193    }
194
195    #[allow(clippy::unused_self)]
196    #[pyo3(signature = (source=None, beam=None), text_signature = "($self, source=[], beam=[])")]
197    fn deconvolvefrombeam(
198        &self,
199        py: Python<'_>,
200        source: Option<Variant>,
201        beam: Option<Variant>,
202    ) -> PyResult<Py<PyAny>> {
203        py.detach(|| {
204            let source = quantities(source, "source")?;
205            if !(1..=3).contains(&source.len()) {
206                return Err(raised("Cannot understand source values"));
207            }
208            let source = stored(&source)?;
209            let beam = quantities(beam, "beam")?;
210            if beam.is_empty() {
211                return Err(raised("Cannot understand beam values"));
212            }
213            let beam = stored(&beam)?;
214            let result = fit_size(radians(&source)?, radians(&beam)?);
215            let (fit, unresolved) = result.unwrap_or((GaussianBeam::new(0.0, 0.0, 0.0), false));
216            let mut fit = fit_record(fit)?;
217            fit.push(("success", CasaValue::Bool(result.is_some())));
218            Ok(CasaValue::record([
219                ("fit", CasaValue::record(fit)),
220                ("return", CasaValue::Bool(unresolved)),
221            ]))
222        })?
223        .into_python(py)
224    }
