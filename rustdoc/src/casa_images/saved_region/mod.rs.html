2//! World-coordinate polygon regions saved with an image.
3//!
4//! casacore keeps named regions in an image's `regions` table keyword
5//! (C++ `ImageInterface::defineRegion` and `getRegion`, through
6//! `RegionHandler`). This module reads and writes the world-coordinate
7//! polygon classes stored there, a single `WCPolygon` or a `WCUnion` whose
8//! members are all `WCPolygon`s, and turns a region into a pixel mask or into
9//! statistics over the pixels it selects.
10//!
11//! | Rust                                   | C++ casacore                                   |
12//! |----------------------------------------|------------------------------------------------|
13//! | [`PolygonRegion`]                      | `WCPolygon`, `WCUnion` of `WCPolygon`          |
14//! | [`PagedImage::get_polygon_region`]     | `ImageInterface::getRegion`                    |
15//! | [`PagedImage::put_polygon_region`]     | `ImageInterface::defineRegion` (`Regions`)     |
16//! | [`PolygonRegion::mask`]                | `WCRegion::toLCRegion` then `LCRegion::getMask` |
17//! | [`PagedImage::region_statistics`]      | statistics of `SubImage(image, region)`        |
18//!
19//! As in casacore's `WCRegion::toLCRegion`, a polygon drawn on two pixel axes
20//! covers the full extent of every other image axis. Its vertices are
21//! converted to pixel coordinates plane by plane, at the centre of the
22//! polygon axes and the plane's position on the others. A pixel belongs to a
23//! polygon when its centre lies inside it or on its boundary.
24//!
25//! Other casacore region classes (`WCBox`, `WCEllipsoid`, the `LC*` pixel
26//! regions and compounds other than a union of polygons) are reported by
27//! class name and not converted. Removing a saved region is
28//! [`PagedImage::remove_region`].
29//!
30//! # Example
31//!
32//! ```rust
33//! use casa_coordinates::{CoordinateSystem, DirectionCoordinate, Projection, ProjectionType};
34//! use casa_images::{PagedImage, PolygonRegion};
35//! use casa_core::measures::direction::DirectionRef;
36//!
37//! let mut coords = CoordinateSystem::new();
38//! coords.add_coordinate(DirectionCoordinate::new(
39//!     DirectionRef::J2000,
40//!     Projection::new(ProjectionType::SIN),
41//!     [0.0, 0.5],
42//!     [-1e-4, 1e-4],
43//!     [2.0, 2.0],
44//! ));
45//! let dir = tempfile::tempdir().unwrap();
46//! let mut image = PagedImage::<f32>::create(vec![5, 5], coords.clone(), dir.path().join("a.image")).unwrap();
47//!
48//! let mut region = PolygonRegion::new(&coords, [0, 1]).unwrap();
49//! let corners = [[1.0, 1.0], [3.0, 1.0], [3.0, 3.0], [1.0, 3.0]];
50//! region
51//!     .add_polygon(corners.iter().map(|&[x, y]| {
52//!         let world = coords.to_world(&[x, y]).unwrap();
53//!         [world[0], world[1]]
54//!     }).collect())
55//!     .unwrap();
56//! image.put_polygon_region("box", &region).unwrap();
57//! image.save().unwrap();
58//!
59//! let mask = region.mask(&coords, image.shape()).unwrap();
60//! assert_eq!(mask.iter().filter(|&&inside| inside).count(), 9);
61//! assert_eq!(image.get_polygon_region("box").unwrap().polygons.len(), 1);
62//! ```

80/// A region made of world-coordinate polygons on two pixel axes.
81///
82/// One polygon is saved as a casacore `WCPolygon`; several are saved as a
83/// `WCUnion` of `WCPolygon`s, and the region is their union. Each vertex is
84/// `[x, y]` in world coordinates on [`pixel_axes`](Self::pixel_axes),
85/// expressed in [`units`](Self::units). Polygons are closed implicitly: the
86/// last vertex joins the first.
87#[derive(Debug, Clone, PartialEq)]
88pub struct PolygonRegion {
89    /// The two zero-based image pixel axes the polygons are drawn on.
90    pub pixel_axes: [usize; 2],
91    /// World units of the vertex coordinates on each axis, such as `"rad"`.
92    pub units: [String; 2],
93    /// The polygons, each a list of `[x, y]` world-coordinate vertices.
94    pub polygons: Vec<Vec<[f64; 2]>>,
95}
96
97/// Statistics of the pixels a region selects.
98///
99/// A pixel counts when it lies inside the region, the image's default mask
100/// (if any) selects it, and its value is finite. This is the data C++
101/// casacore returns from a `SubImage` of the image over the region, with
102/// non-finite values dropped.
103#[derive(Debug, Clone, PartialEq)]
104pub struct RegionStatistics {
105    /// Number of pixels used.
106    pub pixel_count: usize,
107    /// Sum of the values.
108    pub sum: f64,
109    /// Mean value.
110    pub mean: f64,
111    /// Median value; the mean of the two middle values for an even count.
112    pub median: f64,
113    /// Root mean square of the values.
114    pub rms: f64,
115    /// Population standard deviation about the mean (divides by the count).
116    pub sigma: f64,
117    /// Smallest value.
118    pub min: f64,
119    /// Largest value.
120    pub max: f64,
121    /// The image brightness unit, such as `"Jy/beam"`.
122    pub unit: String,
123}

144impl PolygonRegion {
145    /// Creates an empty region on `pixel_axes`, taking the vertex units from
146    /// the world axes of `coords`.
147    pub fn new(coords: &CoordinateSystem, pixel_axes: [usize; 2]) -> Result<Self, ImageError> {
148        check_pixel_axes(pixel_axes, coords.n_pixel_axes())?;
149        let [x, y] = pixel_axes;
150        Ok(Self {
151            pixel_axes,
152            units: [pixel_axis_unit(coords, x)?, pixel_axis_unit(coords, y)?],
153            polygons: Vec::new(),
154        })
155    }
156
157    /// Adds a polygon given by its `[x, y]` world-coordinate vertices.
158    ///
159    /// The polygon needs at least three vertices and must not cross itself.
160    pub fn add_polygon(&mut self, vertices: Vec<[f64; 2]>) -> Result<(), ImageError> {
161        validate_polygon(&vertices)?;
162        self.polygons.push(vertices);
163        Ok(())
164    }

166    /// Encodes the region as a casacore region record.
167    ///
168    /// One polygon becomes a `WCPolygon` record and several a `WCUnion`. The
169    /// record embeds `coords`, the coordinate system of the image the region
170    /// is saved with, as casacore does.
171    pub fn to_record(&self, coords: &CoordinateSystem) -> Result<RecordValue, ImageError> {
172        record::encode(self, coords)
173    }
174
175    /// Decodes the casacore region record saved as `name` with an image whose
176    /// coordinate system is `coords`.
177    ///
178    /// Accepts a `WCPolygon` with absolute coordinates, or a `WCUnion` whose
179    /// members are such polygons on the same pixel axes. Vertex values are
180    /// converted to the units of the image's world axes. Any other class is
181    /// an error that names the region and the class.
182    pub fn from_record(
183        name: &str,
184        record: &RecordValue,
185        coords: &CoordinateSystem,
186    ) -> Result<Self, ImageError> {
187        record::decode(name, record, coords)
188    }
189
190    /// Returns a mask of an image of `shape` with coordinate system `coords`
191    /// that is `true` for every pixel inside the region.
192    ///
193    /// The polygons cover the full extent of every axis other than the
194    /// region's pixel axes.
195    pub fn mask(
196        &self,
197        coords: &CoordinateSystem,
198        shape: &[usize],
199    ) -> Result<ArrayD<bool>, ImageError> {
200        let mut mask = ArrayD::from_elem(IxDyn(shape), false);
201        self.visit_planes(coords, shape, |plane| {
202            let [x_axis, y_axis] = self.pixel_axes;
203            let mut view = mask.slice_each_axis_mut(|axis| {
204                let start = plane.start[axis.axis.index()];
205                Slice::from(start..start + plane.extent[axis.axis.index()])
206            });
207            for (index, value) in view.indexed_iter_mut() {
208                *value = plane.inside[[index[x_axis], index[y_axis]]];
209            }
210            Ok(())
211        })?;
212        Ok(mask)
213    }
214
215    /// The bounds of selected pixels, computed one spatial plane at a time.
216    pub fn bounds(
217        &self,
218        coords: &CoordinateSystem,
219        shape: &[usize],
220    ) -> Result<Option<PolygonBounds>, ImageError> {
221        let [x_axis, y_axis] = self.pixel_axes;
222        let mut lower = shape.to_vec();
223        let mut upper = vec![0; shape.len()];
224        self.visit_planes(coords, shape, |plane| {
225            for ((x, y), &selected) in plane.inside.indexed_iter() {
226                if selected {
227                    for (axis, offset) in [(x_axis, x), (y_axis, y)] {
228                        let position = plane.start[axis] + offset;
229                        lower[axis] = lower[axis].min(position);
230                        upper[axis] = upper[axis].max(position + 1);
231                    }
232                }
233            }
234            for axis in (0..shape.len()).filter(|&axis| axis != x_axis && axis != y_axis) {
235                lower[axis] = lower[axis].min(plane.start[axis]);
236                upper[axis] = upper[axis].max(plane.start[axis] + 1);
237            }
238            Ok(())
239        })?;
240        if upper[x_axis] == 0 {
241            return Ok(None);
242        }
243        let count = upper
244            .iter()
245            .zip(&lower)
246            .map(|(end, start)| end - start)
247            .collect();
248        Ok(Some(PolygonBounds {
249            start: lower,
250            shape: count,
251        }))
252    }

254    fn check_axes(&self, coords: &CoordinateSystem) -> Result<(), ImageError> {
255        check_pixel_axes(self.pixel_axes, coords.n_pixel_axes())
256    }
257
258    /// Calls `visit` for every plane in which the region selects at least
259    /// one pixel.
260    fn visit_planes(
261        &self,
262        coords: &CoordinateSystem,
263        shape: &[usize],
264        mut visit: impl FnMut(&PlaneSelection) -> Result<(), ImageError>,
265    ) -> Result<(), ImageError> {
266        if coords.n_pixel_axes() != shape.len() {
267            return Err(ImageError::InvalidMetadata(format!(
268                "polygon regions need a coordinate system for every image axis; the image has {} axes and its coordinate system {}",
269                shape.len(),
270                coords.n_pixel_axes()
271            )));
272        }
273        self.check_axes(coords)?;
274        for polygon in &self.polygons {
275            validate_polygon(polygon)?;
276        }
277        let [x_axis, y_axis] = self.pixel_axes;
278        let other_axes = (0..shape.len())
279            .filter(|&axis| axis != x_axis && axis != y_axis)
280            .collect::<Vec<_>>();
281        let lengths = other_axes
282            .iter()
283            .map(|&axis| shape[axis])
284            .collect::<Vec<_>>();
285        let mut base_pixel = shape
286            .iter()
287            .map(|&len| (len.saturating_sub(1) / 2) as f64)
288            .collect::<Vec<_>>();
289        let mut previous: Option<(PixelPolygons, Option<PlaneSelection>)> = None;
290        for context in enumerate_axis_indices(&lengths) {
291            for (&axis, &index) in other_axes.iter().zip(&context) {
292                base_pixel[axis] = index as f64;
293            }
294            let polygons = self.plane_pixel_polygons(coords, &base_pixel)?;
295            let reuse = previous
296                .as_ref()
297                .is_some_and(|(previous_polygons, _)| *previous_polygons == polygons);
298            if !reuse {
299                let selection = select_plane(&polygons, shape, self.pixel_axes);
300                previous = Some((polygons, selection));
301            }
302            let Some((_, Some(selection))) = previous.as_mut() else {
303                continue;
304            };
305            for (&axis, &index) in other_axes.iter().zip(&context) {
306                selection.start[axis] = index;
307            }
308            visit(selection)?;
309        }
310        Ok(())
311    }

313    /// Converts the polygons to pixel coordinates on the region axes in the
314    /// plane through `base_pixel`.
315    fn plane_pixel_polygons(
316        &self,
317        coords: &CoordinateSystem,
318        base_pixel: &[f64],
319    ) -> Result<PixelPolygons, ImageError> {
320        let base_world = coords.to_world(base_pixel)?;
321        self.pixel_polygons_at_world(coords, &base_world)
322    }
323
324    fn pixel_polygons_at_world(
325        &self,
326        coords: &CoordinateSystem,
327        base_world: &[f64],
328    ) -> Result<PixelPolygons, ImageError> {
329        let [x_axis, y_axis] = self.pixel_axes;
330        let x_world = world_axis_for_pixel(coords, x_axis)?;
331        let y_world = world_axis_for_pixel(coords, y_axis)?;
332        self.polygons
333            .iter()
334            .map(|polygon| {
335                polygon
336                    .iter()
337                    .map(|vertex| {
338                        let mut world = base_world.to_vec();
339                        world[x_world] = vertex[0];
340                        world[y_world] = vertex[1];
341                        let pixel = coords.to_pixel(&world)?;
342                        let pixel = [pixel[x_axis], pixel[y_axis]];
343                        if pixel.iter().all(|value| value.is_finite()) {
344                            Ok(pixel)
345                        } else {
346                            Err(ImageError::InvalidMetadata(format!(
347                                "polygon vertex {vertex:?} has no pixel position in this image"
348                            )))
349                        }
350                    })
351                    .collect()
352            })
353            .collect()
354    }

356    /// Projects the region at the coordinate system's reference world position. Returns None if no
357    /// pixel centers are selected, or an error if region axes or the transform are invalid.
358    pub fn lattice_plane(
359        &self,
360        coords: &CoordinateSystem,
361        shape: &[usize],
362    ) -> Result<Option<PolygonPixelPlane>, ImageError> {
363        self.check_axes(coords)?;
364        // WCPolygon creates one LCPolygon at the reference world position.
365        // Unspecified axes extend that plane without multiplying its storage.
366        let vertices = self.pixel_polygons_at_world(coords, &coords.reference_value())?;
367        Ok(polygon_pixel_plane(&vertices, shape, self.pixel_axes))
368    }
369}

371impl<T: ImagePixel> PagedImage<T> {
372    /// Reads the saved region `name` as a polygon region.
373    ///
374    /// Mirrors C++ `ImageInterface::getRegion` for the `WCPolygon` class and
375    /// a `WCUnion` of `WCPolygon`s; any other class is an error that names
376    /// it. See [`PolygonRegion::from_record`].
377    pub fn get_polygon_region(&self, name: &str) -> Result<PolygonRegion, ImageError> {
378        let record = self.get_region_record(name)?;
379        PolygonRegion::from_record(name, &record, self.coordinates())
380    }
381
382    /// Saves `region` as the image region `name`, replacing any region of
383    /// that name.
384    ///
385    /// Mirrors C++ `ImageInterface::defineRegion` with
386    /// `RegionHandler::Regions`. Like other metadata changes, it reaches disk
387    /// when the image is [saved](Self::save).
388    pub fn put_polygon_region(
389        &mut self,
390        name: &str,
391        region: &PolygonRegion,
392    ) -> Result<(), ImageError> {
393        let record = region.to_record(self.coordinates())?;
394        self.put_region_record(name, &record)
395    }
396
397    /// Writes `region` as the mask `mask_name`, `true` inside the region,
398    /// and optionally makes it the default mask.
399    ///
400    /// The mask pixels are written at once, as [`put_mask`](Self::put_mask)
401    /// does; the default-mask choice reaches disk when the image is
402    /// [saved](Self::save).
403    pub fn put_region_mask(
404        &mut self,
405        mask_name: &str,
406        region: &PolygonRegion,
407        make_default: bool,
408    ) -> Result<(), ImageError> {
409        let mask = region.mask(self.coordinates(), self.shape())?;
410        self.put_mask(mask_name, &mask)?;
411        if make_default {
412            self.set_default_mask(mask_name)?;
413        }
414        Ok(())
415    }
416}

418impl<T: ImagePixel + StatsElement> PagedImage<T> {
419    /// Computes statistics of the pixels `region` selects.
420    ///
421    /// Returns `None` when the region selects no finite pixel that the
422    /// default mask allows. See [`RegionStatistics`] for which pixels count.
423    /// The image is read one plane's bounding box at a time.
424    pub fn region_statistics(
425        &self,
426        region: &PolygonRegion,
427    ) -> Result<Option<RegionStatistics>, ImageError> {
428        let [x_axis, y_axis] = region.pixel_axes;
429        let mut values = Vec::new();
430        region.visit_planes(self.coordinates(), self.shape(), |plane| {
431            let data = self.get_slice(&plane.start, &plane.extent)?;
432            let stride = vec![1; plane.extent.len()];
433            let mask = self.get_mask_slice(&plane.start, &plane.extent, &stride)?;
434            for (index, pixel) in data.indexed_iter() {
435                if !plane.inside[[index[x_axis], index[y_axis]]] {
436                    continue;
437                }
438                if mask.as_ref().is_some_and(|mask| !mask[index.slice()]) {
439                    continue;
440                }
441                let value = pixel.to_f64_stats();
442                if value.is_finite() {
443                    values.push(value);
444                }
445            }
446            Ok(())
447        })?;
448        Ok(statistics_of(values, self.units()))
449    }
450}
451
452fn statistics_of(mut values: Vec<f64>, unit: &str) -> Option<RegionStatistics> {
453    if values.is_empty() {
454        return None;
455    }
456    let pixel_count = values.len();
457    let count = pixel_count as f64;
458    let sum = values.iter().sum::<f64>();
459    let sum_sq = values.iter().map(|value| value * value).sum::<f64>();
460    let mean = sum / count;
461    let variance = values
462        .iter()
463        .map(|value| {
464            let delta = value - mean;
465            delta * delta
466        })
467        .sum::<f64>()
468        / count;
469    let min = values.iter().copied().fold(f64::INFINITY, f64::min);
470    let max = values.iter().copied().fold(f64::NEG_INFINITY, f64::max);
471    values.sort_by(f64::total_cmp);
472    let median = (values[(pixel_count - 1) / 2] + values[pixel_count / 2]) / 2.0;
473    Some(RegionStatistics {
474        pixel_count,
475        sum,
476        mean,
477        median,
478        rms: (sum_sq / count).sqrt(),
479        sigma: variance.sqrt(),
480        min,
481        max,
482        unit: unit.to_string(),
483    })
484}

486/// Finds the pixels inside `polygons` (pixel coordinates on the region axes)
487/// within their bounding box, or `None` when the box misses the image.
488pub struct PolygonPixelPlane {
489    pub(crate) start: [usize; 2],
490    extent: [usize; 2],
491    polygons: Vec<Vec<[f64; 2]>>,
492}
493
494impl PolygonPixelPlane {
495    /// Expands the polygon's tight pixel bounds into a full-rank image section. Axes outside the
496    /// polygon plane span their full extents.
497    pub fn section(&self, shape: &[usize], [x, y]: [usize; 2]) -> (Vec<usize>, Vec<usize>) {
498        let mut start = vec![0; shape.len()];
499        start[x] = self.start[0];
500        start[y] = self.start[1];
501        let mut shape = shape.to_vec();
502        shape[x] = self.extent[0];
503        shape[y] = self.extent[1];
504        (start, shape)
505    }
506
507    /// Tests whether the center of a zero-based pixel in the polygon plane lies in any member
508    /// polygon.
509    pub fn contains(&self, x: usize, y: usize) -> bool {
510        let point = [x as f64, y as f64];
511        self.polygons
512            .iter()
513            .any(|polygon| polygon_contains_pixel(polygon, point))
514    }
515}

517/// Builds a plane for the union of pixel-coordinate polygons, clipped to the two specified image
518/// axes. Returns None when no pixel center is selected.
519pub fn polygon_pixel_plane(
520    polygons: &[Vec<[f64; 2]>],
521    shape: &[usize],
522    [x, y]: [usize; 2],
523) -> Option<PolygonPixelPlane> {
524    let (x0, x1) = pixel_range(polygons, 0, shape[x])?;
525    let (y0, y1) = pixel_range(polygons, 1, shape[y])?;
526    let mut low = [x1, y1];
527    let mut high = [x0, y0];
528    let mut any = false;
529    for x in x0..=x1 {
530        for y in y0..=y1 {
531            let point = [x as f64, y as f64];
532            if polygons
533                .iter()
534                .any(|polygon| polygon_contains_pixel(polygon, point))
535            {
536                low[0] = low[0].min(x);
537                low[1] = low[1].min(y);
538                high[0] = high[0].max(x);
539                high[1] = high[1].max(y);
540                any = true;
541            }
542        }
543    }
544    any.then(|| PolygonPixelPlane {
545        start: low,
546        extent: [high[0] - low[0] + 1, high[1] - low[1] + 1],
547        polygons: polygons.to_vec(),
548    })
549}

551fn select_plane(
552    polygons: &[Vec<[f64; 2]>],
553    shape: &[usize],
554    [x_axis, y_axis]: [usize; 2],
555) -> Option<PlaneSelection> {
556    let (x_range, y_range) = (
557        pixel_range(polygons, 0, shape[x_axis])?,
558        pixel_range(polygons, 1, shape[y_axis])?,
559    );
560    let inside = Array2::from_shape_fn(
561        (x_range.1 - x_range.0 + 1, y_range.1 - y_range.0 + 1),
562        |(x, y)| {
563            let centre = [(x_range.0 + x) as f64, (y_range.0 + y) as f64];
564            polygons
565                .iter()
566                .any(|polygon| polygon_contains_pixel(polygon, centre))
567        },
568    );
569    if !inside.iter().any(|&selected| selected) {
570        return None;
571    }
572    let mut start = vec![0; shape.len()];
573    let mut extent = vec![1; shape.len()];
574    start[x_axis] = x_range.0;
575    start[y_axis] = y_range.0;
576    extent[x_axis] = inside.nrows();
577    extent[y_axis] = inside.ncols();
578    Some(PlaneSelection {
579        start,
580        extent,
581        inside,
582    })
583}

585/// The inclusive range of pixel indices on one region axis that the polygons
586/// can touch, clipped to an axis of `len` pixels.
587fn pixel_range(
588    polygons: &[Vec<[f64; 2]>],
589    coordinate: usize,
590    len: usize,
591) -> Option<(usize, usize)> {
592    let values = polygons.iter().flatten().map(|vertex| vertex[coordinate]);
593    let low = values.clone().fold(f64::INFINITY, f64::min).floor();
594    let high = values.fold(f64::NEG_INFINITY, f64::max).ceil();
595    if len == 0 || high < 0.0 || low > (len - 1) as f64 {
596        return None;
597    }
598    Some((low.max(0.0) as usize, high.min((len - 1) as f64) as usize))
599}
600
601fn check_pixel_axes(pixel_axes: [usize; 2], n_axes: usize) -> Result<(), ImageError> {
602    if pixel_axes[0] == pixel_axes[1] || pixel_axes.iter().any(|&axis| axis >= n_axes) {
603        return Err(ImageError::InvalidMetadata(format!(
604            "polygon region axes {pixel_axes:?} must be two different axes of a {n_axes}-axis coordinate system"
605        )));
606    }
607    Ok(())
608}
609
610fn world_axis_for_pixel(coords: &CoordinateSystem, pixel: usize) -> Result<usize, ImageError> {
611    coords.pixel_axis_to_world_axis(pixel).ok_or_else(|| {
612        ImageError::InvalidMetadata(format!(
613            "polygon region pixel axis {pixel} has no world axis"
614        ))
615    })
616}
617
618fn pixel_axis_unit(coords: &CoordinateSystem, pixel: usize) -> Result<String, ImageError> {
619    let world = world_axis_for_pixel(coords, pixel)?;
620    let (index, local) = coords.world_axis_owner(world).expect("public world axis");
621    let units = coords.coordinate(index).axis_units();
622    Ok(units.get(local).cloned().unwrap_or_default())
623}
