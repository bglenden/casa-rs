26/// Reservations are attached at the scheduler's existing export boundaries.
27/// Every physical backing retains this owner after its arrays and directory.
28#[derive(Debug, Default)]
29struct CubeStateRetention {
30    capacity: OnceLock<RetainedArtifactPermit>,
31    heap: OnceLock<RetainedArtifactPermit>,
32}
33
34/// One independently releasable memory allocation; shared capacity is small for
35/// resident normals and remains conservative for paged file/storage resources.
36#[derive(Debug)]
37struct CubeBackingRetention {
38    heap: OnceLock<std::sync::Mutex<RetainedArtifactPermit>>,
39    _shared: Arc<CubeStateRetention>,
40}
41
42/// One retained lease and one physical cache across every cube major cycle.
43pub(crate) struct ManagedCubeRun {
44    pub(crate) residency: Arc<CubeResidency>,
45    retention: Arc<CubeBackingRetention>,
46}
47
48impl ManagedCubeRun {
49    /// Physical eviction must precede returning any part of the retained lease.
50    pub(crate) fn shrink_cache_to(&self, target: usize) -> io::Result<()> {
51        self.residency.shrink_to(target)?;
52        self.retention
53            .heap
54            .get()
55            .ok_or_else(|| io::Error::other("managed cache permit is not retained"))?
56            .lock()
57            .map_err(|_| io::Error::other("managed cache permit lock poisoned"))?
58            .narrow_memory_to(target as u64)
59            .map_err(io::Error::other)
60    }
61}

63impl std::fmt::Debug for ManagedCubeRun {
64    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
65        f.debug_struct("ManagedCubeRun")
66            .field("used_bytes", &self.residency.used_bytes())
67            .finish_non_exhaustive()
68    }

72/// Prior generations keep their own reservations while this generation is built.

76    normal: Arc<dyn NormalStorageFactory>,

80    metrics: Arc<CubeBackingMetrics>,

95    /// Admission floor and full-residency ceiling for two overlapping model
96    /// generations, one invariant PSF and two residual epochs. All owner,
97    /// registry, staging and active-operation terms use the same typed array
98    /// formulas as physical creation.
99    pub(crate) fn managed_cache_limits(
100        problem: &CompiledProblem,
101        storage: &ManagedSpillStorage,
102        workers: usize,
103    ) -> io::Result<(usize, usize)> {
104        let shape = problem.model_lifecycle().target();
105        let [domain] = shape.domains() else {
106            return Err(io::Error::other("managed cube requires one image domain"));
107        };
108        let [width, height] = domain.pixels();
109        let cells = width.checked_mul(height).ok_or_else(overflow)?;
110        if cells == 0 || workers == 0 || !shape.sample_count().is_multiple_of(cells) {
111            return Err(overflow());
112        }
113        let planes = shape.sample_count() / cells;
114        Self::managed_cache_limits_for_shape(storage.directory(), height, width, planes, workers)
115    }
116
117    fn managed_cache_limits_for_shape(
118        directory: &std::path::Path,
119        height: usize,
120        width: usize,
121        planes: usize,
122        workers: usize,
123    ) -> io::Result<(usize, usize)> {
124        let value = ManagedPlaneArray::<f32>::footprint(directory, height, width, planes)?;
125        let support = ManagedPlaneArray::<bool>::footprint(directory, height, width, planes)?;
126        let owner = add(
127            value.owner_bytes.checked_mul(5).ok_or_else(overflow)?,
128            support.owner_bytes.checked_mul(2).ok_or_else(overflow)?,
129        )?;
130        let registry = add(
131            value.registry_bytes.checked_mul(5).ok_or_else(overflow)?,
132            support.registry_bytes.checked_mul(2).ok_or_else(overflow)?,
133        )?;
134        let staging = add(
135            value.staging_bytes.checked_mul(5).ok_or_else(overflow)?,
136            support.staging_bytes.checked_mul(2).ok_or_else(overflow)?,
137        )?;
138        let worker_blocks = add(
139            value.block_bytes.checked_mul(3).ok_or_else(overflow)?,
140            support.block_bytes.checked_mul(2).ok_or_else(overflow)?,
141        )?;
142        let active = worker_blocks.checked_mul(workers).ok_or_else(overflow)?;
143        let fixed = CubeResidency::fixed_owner_bytes();
144        let operation =
145            CubeResidency::operation_overhead(workers.checked_mul(5).ok_or_else(overflow)?)?;
146        let normal_floor = [fixed, owner, registry, staging, active, operation]
147            .into_iter()
148            .try_fold(0usize, add)?;
149        let creation_overlap = add(
150            add(fixed, add(owner, registry)?)?,
151            add(staging, value.creation_bytes.max(support.creation_bytes))?,
152        )?;
153        let minimum = normal_floor.max(creation_overlap);
154        let payload = add(
155            value
156                .block_bytes
157                .checked_mul(5)
158                .and_then(|n| n.checked_mul(planes))
159                .ok_or_else(overflow)?,
160            support
161                .block_bytes
162                .checked_mul(2)
163                .and_then(|n| n.checked_mul(planes))
164                .ok_or_else(overflow)?,
165        )?;
166        let full = add(
167            add(add(fixed, owner)?, registry)?,
168            add(add(staging, payload)?, operation)?,
169        )?
170        .max(minimum);
171        Ok((minimum, full))
172    }

174    /// Bind channel-local Float owners to the one lease-backed run cache.
175    /// The first major reserves its storage and memory capacity; later majors
176    /// reuse that retained capacity while admitting only their own workspaces.
177    #[allow(clippy::too_many_arguments)]
178    pub(crate) fn managed_streaming_cube(
179        problem: &CompiledProblem,
180        storage: &ManagedSpillStorage,
181        window_channels: usize,
182        acquire: WorkNodeId,
183        terminal: WorkNodeId,
184        run: Option<Arc<ManagedCubeRun>>,
185        cache_bytes: usize,
186    ) -> io::Result<Self> {
187        let shape = problem.model_lifecycle().target();
188        let [domain] = shape.domains() else {
189            return Err(io::Error::other("managed cube requires one image domain"));
190        };
191        let [width, height] = domain.pixels();
192        let cells = width.checked_mul(height).ok_or_else(overflow)?;
193        let planes = shape.sample_count() / cells;
194        if cells == 0 || planes == 0 || cells.checked_mul(planes) != Some(shape.sample_count()) {
195            return Err(overflow());
196        }
197        let initial = run.is_none();
198        let retention = run.as_ref().map_or_else(
199            || Arc::new(CubeStateRetention::default()),
200            |run| run.retention._shared.clone(),
201        );
202        let run = match run {
203            Some(run) => run,
204            None => Arc::new(ManagedCubeRun {
205                residency: CubeResidency::new(cache_bytes)?,
206                retention: Arc::new(CubeBackingRetention {
207                    heap: OnceLock::new(),
208                    _shared: retention.clone(),
209                }),
210            }),
211        };
212        let permit_owner: Arc<dyn std::fmt::Debug + Send + Sync> = run.clone();
213        let model: Arc<dyn ModelStorageFactory> = Arc::new(ManagedModelFactory::new(
214            run.residency.clone(),
215            permit_owner.clone(),
216            storage.directory(),
217            [height, width],
218            planes,
219        )?);
220        let normal: Arc<dyn NormalStorageFactory> = Arc::new(ManagedNormalFactory::new(
221            run.residency.clone(),
222            permit_owner,
223            storage.directory(),
224            [height, width],
225        )?);
226        let storage_id = format!("cube-state-storage-{}", acquire.as_str());
227        let value_disk =
228            ManagedPlaneArray::<f32>::footprint(storage.directory(), height, width, planes)?
229                .storage_bytes;
230        let support_disk =
231            ManagedPlaneArray::<bool>::footprint(storage.directory(), height, width, planes)?
232                .storage_bytes;
233        let storage_bytes = if initial {
234            let model_pair = add(value_disk, support_disk)?;
235            as_u64(add(
236                model_pair.checked_mul(2).ok_or_else(overflow)?,
237                value_disk.checked_mul(3).ok_or_else(overflow)?,
238            )?)?
239        } else {
240            0
241        };
242        let file_handles = if initial { 7 } else { 0 };
243        let heap_id = format!("cube-state-retained-{}", acquire.as_str());
244        let heap_bytes = size_of::<Self>()
245            .checked_add(size_of::<ManagedModelFactory>())
246            .and_then(|n| n.checked_add(size_of::<ManagedNormalFactory>()))
247            .and_then(|n| n.checked_add(size_of::<ManagedCubeRun>()))
248            .and_then(|n| n.checked_add(size_of::<CubeStateRetention>()))
249            .and_then(|n| n.checked_add(size_of::<CubeBackingRetention>()))
250            .and_then(|n| n.checked_add(storage.directory().as_os_str().len() * 2))
251            .and_then(|n| n.checked_add(12 * size_of::<usize>()))
252            .ok_or_else(overflow)?;
253        let permit_bytes = if initial {
254            RetainedArtifactPermit::heap_bytes_for_resources(
255                &[
256                    LeaseResource::Storage {
257                        demand_id: storage_id.clone(),
258                        use_kind: StorageUseKind::Temporary,
259                    },
260                    LeaseResource::FileDescriptors,
261                ],
262                0,
263                "host-memory",
264                storage.resources().domain().as_str(),
265            )
266            .and_then(|n| {
267                RetainedArtifactPermit::heap_bytes_for_resources(
268                    &[LeaseResource::Memory {
269                        allocation_id: format!("cube-state-manager-{}", acquire.as_str()),
270                    }],
271                    1,
272                    "host-memory",
273                    storage.resources().domain().as_str(),
274                )
275                .and_then(|extra| n.checked_add(extra))
276            })
277            .ok_or_else(overflow)? as usize
278        } else {
279            0
280        };
281        let heap = allocation(
282            heap_id,
283            as_u64(add(heap_bytes, permit_bytes)?)?,
284            &acquire,
285            &terminal,
286            initial,
287        );
288        let backings = if initial {
289            vec![(
290                allocation(
291                    format!("cube-state-manager-{}", acquire.as_str()),
292                    as_u64(cache_bytes)?,
293                    &acquire,
294                    &terminal,
295                    true,
296                ),
297                run.retention.clone(),
298            )]
299            .into_boxed_slice()
300        } else {
301            Box::new([])
302        };
303        let plane_complex = cells
304            .checked_mul(size_of::<num_complex::Complex64>())
305            .ok_or_else(overflow)?;
306        let scratch_bytes = add(
307            plane_complex.checked_mul(2).ok_or_else(overflow)?,
308            cells
309                .checked_mul(size_of::<ModelSample>())
310                .ok_or_else(overflow)?,
311        )?;
312        let scratch = allocation(
313            format!("cube-state-access-{}", acquire.as_str()),
314            as_u64(scratch_bytes)?,
315            &acquire,
316            &terminal,
317            false,
318        );
319        let retained_bytes = if initial {
320            heap.bytes
321                .checked_add(as_u64(cache_bytes)?)
322                .ok_or_else(overflow)?
323        } else {
324            0
325        };
326        Ok(Self {
327            model,
328            normal,
329            managed: Some(run),
330            managed_initial: initial,
331            retention,
332            metrics: Arc::new(CubeBackingMetrics::default()),
333            model_window_samples: cells,
334            normal_window_channels: window_channels,
335            acquire,
336            terminal,
337            heap,
338            backings,
339            retained_bytes,
340            scratch,
341            storage_id,
342            storage_bytes,
343            file_handles,
344        })
345    }

351    pub(crate) fn new(
352        problem: &CompiledProblem,
353        storage: &ManagedSpillStorage,
354        window_channels: usize,
355        acquire: WorkNodeId,
356        terminal: WorkNodeId,
357    ) -> io::Result<Self> {
358        let specification =
359            SpectralOperatorSpecification::new(problem).map_err(io::Error::other)?;
360        let requirements =
361            ChannelNormalStorageRequirement::for_specification(&specification, window_channels)
362                .map_err(io::Error::other)?;
363        Self::with_normal(
364            problem,
365            storage,
366            window_channels,
367            acquire,
368            terminal,
369            requirements,
370        )
371    }

373    #[allow(clippy::too_many_arguments)]
374    fn with_normal(
375        problem: &CompiledProblem,
376        storage: &ManagedSpillStorage,
377        window_channels: usize,
378        acquire: WorkNodeId,
379        terminal: WorkNodeId,
380        requirements: Box<[ChannelNormalStorageRequirement]>,
381    ) -> io::Result<Self> {
382        let shape = problem.model_lifecycle().target();
383        let model_window_samples = shape.domains().iter().try_fold(0usize, |largest, domain| {
384            let [width, height] = domain.pixels();
385            width
386                .checked_mul(height)
387                .and_then(|n| n.checked_mul(shape.polarizations()))
388                .map(|n| largest.max(n))
389                .ok_or_else(overflow)
390        })?;
391        let retention = Arc::new(CubeStateRetention::default());
392        let model_retention = Arc::new(CubeBackingRetention {
393            heap: OnceLock::new(),
394            _shared: retention.clone(),
395        });
396        let metrics = Arc::new(CubeBackingMetrics::default());
397        let model_layout = if let [domain] = shape.domains() {
398            let [width, height] = domain.pixels();
399            CubeArrayLayout::new_spatial(
400                shape.sample_count(),
401                height,
402                width
403                    .checked_mul(shape.polarizations())
404                    .ok_or_else(overflow)?,
405                model_window_samples,
406                1,
407            )
408        } else {
409            CubeArrayLayout::new(
410                shape.sample_count(),
411                model_window_samples,
412                model_window_samples,
413                1,
414            )
415        }
416        .map_err(io::Error::other)?;
417        let model = Arc::new(
418            PagedModelStorageFactory::new(
419                storage.directory(),
420                model_layout,
421                model_retention.clone(),
422                metrics.clone(),
423            )
424            .map_err(io::Error::other)?,
425        );
426        let model_ledger = model.ledger().map_err(io::Error::other)?;
427        let layouts = requirements
428            .iter()
429            .map(|requirement| {
430                CubeArrayLayout::new_spatial(
431                    requirement.scalar_capacity(),
432                    requirement.image_axes()[0],
433                    requirement.image_axes()[1],
434                    requirement.maximum_window_scalars(),
435                    1,
436                )
437                .map(|layout| (requirement.allocation_ordinal(), layout))
438                .map_err(io::Error::other)
439            })
440            .collect::<io::Result<Box<[_]>>>()?;
441        let mut retained_bytes = 0usize;
442        let mut backings = vec![(
443            allocation(
444                format!("cube-state-model-{}", acquire.as_str()),
445                as_u64(model_ledger.retained_bytes)?,
446                &acquire,
447                &terminal,
448                true,
449            ),
450            model_retention,
451        )];
452        let mut normal_retentions: Vec<(usize, Arc<dyn std::fmt::Debug + Send + Sync>)> =
453            Vec::new();
454        let mut storage_bytes = model_ledger.storage_bytes;
455        let mut file_handles = model_ledger.file_handles;
456        let mut scratch_bytes = model_ledger
457            .read_scratch_bytes
458            .max(model_ledger.write_scratch_bytes)
459            .max(model_ledger.flush_scratch_bytes)
460            .checked_add(
461                model_window_samples
462                    .checked_mul(size_of::<ModelSample>())
463                    .ok_or_else(overflow)?,
464            )
465            .ok_or_else(overflow)?;
466        for ((_, layout), requirement) in layouts.iter().zip(&requirements) {
467            let ledger = layout
468                .normal_ledger(storage.directory())
469                .map_err(io::Error::other)?;
470            let normal_retention = Arc::new(CubeBackingRetention {
471                heap: OnceLock::new(),
472                _shared: retention.clone(),
473            });
474            normal_retentions.push((requirement.allocation_ordinal(), normal_retention.clone()));
475            backings.push((
476                allocation(
477                    format!(
478                        "cube-state-normal-{}-{}",
479                        acquire.as_str(),
480                        requirement.allocation_ordinal()
481                    ),
482                    as_u64(add(
483                        ledger.retained_bytes,
484                        requirement.retained_metadata_bytes(),
485                    )?)?,
486                    &acquire,
487                    &terminal,
488                    true,
489                ),
490                normal_retention,
491            ));
492            storage_bytes = add(storage_bytes, ledger.storage_bytes)?;
493            file_handles = add(file_handles, ledger.file_handles)?;
494            // The normal owner converts a complex source to f64 before the
495            // backing's ndarray copy. Both coexist with the source primitive.
496            let conversion = requirement
497                .maximum_window_scalars()
498                .checked_mul(size_of::<f64>())
499                .ok_or_else(overflow)?;
500            scratch_bytes = scratch_bytes.max(
501                add(conversion, ledger.write_scratch_bytes)?
502                    .max(ledger.read_scratch_bytes)
503                    .max(ledger.flush_scratch_bytes),
504            );
505        }
506        let factory = PagedNormalStorageFactory::new(
507            storage.directory(),
508            layouts,
509            normal_retentions.into_boxed_slice(),
510            metrics.clone(),
511            false,
512        );
513        let normal_metadata = factory.owned_metadata_bytes().map_err(io::Error::other)?;
514        let normal: Arc<dyn NormalStorageFactory> = Arc::new(factory);
515        retained_bytes = add(
516            retained_bytes,
517            model.owned_metadata_bytes().map_err(io::Error::other)?,
518        )?;
519        retained_bytes = add(retained_bytes, normal_metadata)?;
520        retained_bytes = add(
521            retained_bytes,
522            size_of::<Self>()
523                + size_of::<CubeStateRetention>()
524                + size_of::<CubeBackingMetrics>()
525                + backings.len()
526                    * (size_of::<(LogicalAllocation, Arc<CubeBackingRetention>)>()
527                        + size_of::<CubeBackingRetention>()
528                        + 2 * size_of::<usize>())
529                + 10 * size_of::<usize>(),
530        )?;
531        let storage_id = format!("cube-state-storage-{}", acquire.as_str());
532        let heap_id = format!("cube-state-retained-{}", acquire.as_str());
533        let capacity_resources = [
534            LeaseResource::Storage {
535                demand_id: storage_id.clone(),
536                use_kind: StorageUseKind::Temporary,
537            },
538            LeaseResource::FileDescriptors,
539        ];
540        let heap_resources = [LeaseResource::Memory {
541            allocation_id: heap_id.clone(),
542        }];
543        let permit_bytes = RetainedArtifactPermit::heap_bytes_for_resources(
544            &capacity_resources,
545            0,
546            "host-memory",
547            storage.resources().domain().as_str(),
548        )
549        .and_then(|bytes| {
550            RetainedArtifactPermit::heap_bytes_for_resources(
551                &heap_resources,
552                1,
553                "host-memory",
554                storage.resources().domain().as_str(),
555            )
556            .and_then(|other| bytes.checked_add(other))
557        })
558        .ok_or_else(overflow)?;
559        let permit_bytes = backings
560            .iter()
561            .try_fold(permit_bytes, |bytes, (allocation, _)| {
562                RetainedArtifactPermit::heap_bytes_for_resources(
563                    &[LeaseResource::Memory {
564                        allocation_id: allocation.id.as_str().to_owned(),
565                    }],
566                    1,
567                    "host-memory",
568                    storage.resources().domain().as_str(),
569                )
570                .and_then(|other| bytes.checked_add(other))
571                .ok_or_else(overflow)
572            })?;
573        let retained_bytes = as_u64(retained_bytes)?
574            .checked_add(permit_bytes)
575            .ok_or_else(overflow)?;
576        let heap = allocation(heap_id, retained_bytes, &acquire, &terminal, true);
577        let retained_bytes = backings
578            .iter()
579            .try_fold(heap.bytes, |bytes, (allocation, _)| {
580                bytes.checked_add(allocation.bytes).ok_or_else(overflow)
581            })?;
582        let scratch = allocation(
583            format!("cube-state-access-{}", acquire.as_str()),
584            as_u64(scratch_bytes)?,
585            &acquire,
586            &terminal,
587            false,
588        );
589        Ok(Self {
590            model,
591            normal,
592            managed: None,
593            managed_initial: false,
594            retention,
595            metrics,
596            model_window_samples,
597            normal_window_channels: window_channels,
598            acquire,
599            terminal,
600            heap,
601            backings: backings.into_boxed_slice(),
602            retained_bytes,
603            scratch,
604            storage_id,
605            storage_bytes: as_u64(storage_bytes)?,
606            file_handles: as_u64(file_handles)?,
607        })
608    }

614    /// Allocation capabilities must be live before any mutable backing is made.
615    pub(crate) fn model_storage(
616        &self,
617        context: WorkExecutionContext<'_>,
618    ) -> io::Result<ModelStoragePlan> {
619        let retained_capacity = self.managed.is_some()
620            && !self.managed_initial
621            && self.retention.capacity.get().is_some()
622            && self
623                .managed
624                .as_ref()
625                .is_some_and(|run| run.retention.heap.get().is_some());
626        if context.node().id != self.acquire
627            || !context
628                .node()
629                .allocations
630                .iter()
631                .any(|use_| use_.allocation == self.heap.id)
632            || self.backings.iter().any(|(allocation, _)| {
633                !context
634                    .node()
635                    .allocations
636                    .iter()
637                    .any(|use_| use_.allocation == allocation.id)
638            })
639            || !(retained_capacity
640                || context.node().claims.iter().any(|claim| {
641                    claim.resource == self.storage_resource()
642                        && claim.amount == self.storage_bytes
643                        && claim.lifetime == ClaimLifetime::Artifact
644                }))
645            || !(retained_capacity
646                || context.node().claims.iter().any(|claim| {
647                    claim.resource == LeaseResource::FileDescriptors
648                        && claim.amount == self.file_handles
649                        && claim.lifetime == ClaimLifetime::Artifact
650                }))
651        {
652            return Err(io::Error::other(
653                "cube storage lacks its plan-issued allocation and capacity claims",
654            ));
655        }
656        ModelStoragePlan::new(self.model.clone(), self.model_window_samples)
657            .map_err(io::Error::other)
658    }

704    pub(crate) fn retain(
705        &self,
706        node: &WorkNodeId,
707        permit: RetainedArtifactPermit,
708    ) -> io::Result<()> {
709        if node == &self.acquire {
710            if !permit.covers_exact_resources(&[
711                (self.storage_resource(), self.storage_bytes),
712                (LeaseResource::FileDescriptors, self.file_handles),
713            ]) {
714                return Err(io::Error::other(
715                    "cube backing reservation differs from its admitted capacity",
716                ));
717            }
718            self.retention
719                .capacity
720                .set(permit)
721                .map_err(|_| io::Error::other("cube backing capacity was retained twice"))
722        } else if node == &self.terminal {
723            let allocations = std::iter::once(&self.heap)
724                .chain(self.backings.iter().map(|(allocation, _)| allocation))
725                .collect::<Vec<_>>();
726            let mut partitions = permit
727                .partition_immutable_allocations(node, &allocations)
728                .map_err(io::Error::other)?
729                .into_iter();
730            self.retention
731                .heap
732                .set(partitions.next().expect("shared allocation is present"))
733                .map_err(|_| io::Error::other("cube backing cache was retained twice"))?;
734            for ((_, retention), permit) in self.backings.iter().zip(partitions) {
735                retention
736                    .heap
737                    .set(std::sync::Mutex::new(permit))
738                    .map_err(|_| io::Error::other("cube backing allocation was retained twice"))?;
739            }
740            Ok(())
741        } else {
742            Err(io::Error::other(
743                "cube backing reservation was exported by an unrelated node",
744            ))
745        }
746    }

751            use_kind: StorageUseKind::Temporary,

755    pub(crate) fn compose<R: ImplementationRegistry>(
756        &self,
757        registry: &R,
758        implementation: WorkImplementationId,
759        storage: &ManagedSpillStorage,
760        base: PhysicalWorkBinding,
761        replay: &WorkNodeId,
762        reconcile: &WorkNodeId,
763    ) -> Result<PhysicalWorkBinding, SpectralCyclePlanError> {
764        // Cube I/O runs synchronously on the coordinator, using the existing
765        // transaction queue. It does not add a concurrent reader or writer.
766        let existing_queue = base
767            .execution_dag()
768            .resource_alternative()
769            .demand
770            .queues
771            .iter()
772            .find(|queue| &queue.resource == storage.resources().queue())
773            .map(|queue| queue.demand_id.clone());
774        let new_queue = existing_queue.is_none().then(|| QueueDemand {
775            demand_id: format!("{}-queue", self.storage_id),
776            resource: storage.resources().queue().clone(),
777            slots: CountDemand::new(1, 1),
778        });
779        let queue = LeaseResource::Queue {
780            demand_id: existing_queue.unwrap_or_else(|| {
781                new_queue
782                    .as_ref()
783                    .expect("new queue needed")
784                    .demand_id
785                    .clone()
786            }),
787        };
788        let mut nodes = base
789            .execution_dag()
790            .nodes()
791            .values()
792            .cloned()
793            .collect::<Vec<_>>();
794        let users = BTreeSet::from([
795            self.acquire.clone(),
796            replay.clone(),
797            reconcile.clone(),
798            self.terminal.clone(),
799        ]);
800        for node in &mut nodes {
801            if !users.contains(&node.id) {
802                continue;
803            }
804            let lifetime = if node.fences.contains(&FenceKind::Io) {
805                ClaimLifetime::through_fence(FenceKind::Io)
806            } else {
807                ClaimLifetime::Work
808            };
809            node.allocations.extend([
810                AllocationUse {
811                    allocation: self.heap.id.clone(),
812                    lifetime: lifetime.clone(),
813                },
814                AllocationUse {
815                    allocation: self.scratch.id.clone(),
816                    lifetime: lifetime.clone(),
817                },
818            ]);
819            node.allocations
820                .extend(self.backings.iter().map(|(allocation, _)| AllocationUse {
821                    allocation: allocation.id.clone(),
822                    lifetime: lifetime.clone(),
823                }));
824            node.claims.extend([
825                ResourceClaim {
826                    resource: LeaseResource::StorageReadRate {
827                        demand_id: self.storage_id.clone(),
828                    },
829                    amount: 1,
830                    lifetime: lifetime.clone(),
831                },
832                ResourceClaim {
833                    resource: LeaseResource::StorageWriteRate {
834                        demand_id: self.storage_id.clone(),
835                    },
836                    amount: 1,
837                    lifetime: lifetime.clone(),
838                },
839            ]);
840            if !node.claims.iter().any(|claim| claim.resource == queue) {
841                node.claims.push(ResourceClaim {
842                    resource: queue.clone(),
843                    amount: 1,
844                    lifetime,
845                });
846            }
847            if node.id == self.acquire && self.storage_bytes != 0 {
848                node.claims.extend([
849                    ResourceClaim {
850                        resource: self.storage_resource(),
851                        amount: self.storage_bytes,
852                        lifetime: ClaimLifetime::Artifact,
853                    },
854                    ResourceClaim {
855                        resource: LeaseResource::FileDescriptors,
856                        amount: self.file_handles,
857                        lifetime: ClaimLifetime::Artifact,
858                    },
859                ]);
860            }
861        }
862        let mut alternative = base.execution_dag().resource_alternative().clone();
863        alternative.demand.queues.extend(new_queue);
864        for allocation in [&self.heap, &self.scratch]
865            .into_iter()
866            .chain(self.backings.iter().map(|(allocation, _)| allocation))
867        {
868            alternative.demand.memory.push(MemoryDemand {
869                allocation_id: allocation.id.as_str().to_owned(),
870                hard_bytes: allocation.bytes,
871                preferred_bytes: allocation.bytes,
872                views: vec![CapacityViewId::new("host-memory")],
873            });
874        }
875        alternative.demand.file_descriptors = CountDemand::new(
876            alternative
877                .demand
878                .file_descriptors
879                .hard()
880                .checked_add(self.file_handles)
881                .ok_or(SpectralCyclePlanError::Overflow)?,
882            alternative
883                .demand
884                .file_descriptors
885                .preferred()
886                .checked_add(self.file_handles)
887                .ok_or(SpectralCyclePlanError::Overflow)?,
888        );
889        alternative.demand.storage.push(StorageDemand {
890            demand_id: self.storage_id.clone(),
891            domain: storage.resources().domain().clone(),
892            temporary_bytes: self.storage_bytes,
893            staged_output_bytes: 0,
894            final_output_bytes: 0,
895            persistent_cache_bytes: 0,
896            read_rate: CountDemand::new(1, 1),
897            write_rate: CountDemand::new(1, 1),
898            operations_rate: CountDemand::zero(),
899            queue_slots: CountDemand::zero(),
900        });
901        let dag = ExecutionDag::new(ExecutionDagSpecification {
902            required_resource_capabilities: base
903                .execution_dag()
904                .required_resource_capabilities()
905                .clone(),
906            resource_alternative: alternative,
907            nodes,
908            logical_allocations: base
909                .execution_dag()
910                .logical_allocations()
911                .values()
912                .cloned()
913                .chain([self.heap.clone(), self.scratch.clone()])
914                .chain(
915                    self.backings
916                        .iter()
917                        .map(|(allocation, _)| allocation.clone()),
918                )
919                .collect(),
920            physical_slots: base
921                .execution_dag()
922                .physical_slots()
923                .values()
924                .cloned()
925                .chain(
926                    [&self.heap, &self.scratch]
927                        .into_iter()
928                        .chain(self.backings.iter().map(|(allocation, _)| allocation))
929                        .map(|allocation| PhysicalSlot {
930                            id: allocation.physical_slot.clone(),
931                            lease_resource: LeaseResource::Memory {
932                                allocation_id: allocation.id.as_str().to_owned(),
933                            },
934                            capacity_bytes: allocation.bytes,
935                            compatibility: allocation.compatibility.clone(),
936                        }),
937                )
938                .collect(),
939            initial_knobs: base.execution_dag().initial_knobs().clone(),
940            adaptations: base
941                .execution_dag()
942                .adaptations()
943                .values()
944                .cloned()
945                .collect(),
946        })?;
947        let catalog = ImplementationContractCatalog::from_registry(registry, [implementation])?;
948        Ok(PhysicalWorkBinding::new_reconstruction(
949            catalog,
950            dag,
951            base.prediction().clone(),
952            base.artifacts().to_vec(),
953            base.observation_transaction().clone(),
954            base.publication_layouts().clone(),
955        )?)
956    }
957}

959fn allocation(
960    id: String,
961    bytes: u64,
962    acquire: &WorkNodeId,
963    terminal: &WorkNodeId,
964    retained: bool,
965) -> LogicalAllocation {
966    LogicalAllocation {
967        physical_slot: PhysicalSlotId::new(format!("{id}-slot")),
968        id: AllocationId::new(id),
969        bytes,
970        purpose: AllocationPurpose::Data,
971        compatibility: SlotCompatibility {
972            memory_domain: CapacityDomainId::new("host-memory"),
973            views: BTreeSet::from([CapacityViewId::new("host-memory")]),
974            alignment_bytes: 64,
975            storage_mode: StorageMode::Host,
976            layout: AllocationLayout::new(if retained {
977                "cube-state-cache-and-metadata"
978            } else {
979                "cube-state-access"
980            }),
981            initialization: InitializationPolicy::OverwriteBeforeRead,
982            access: if retained {
983                AllocationAccess::ReadOnly
984            } else {
985                AllocationAccess::ReadWrite
986            },
987        },
988        lifetime: AllocationLifetime {
989            acquire_at: acquire.clone(),
990            release_after: BTreeSet::from([WorkDependency::Work(terminal.clone())]),
991            disposition: if retained {
992                AllocationDisposition::ExportImmutableArtifact {
993                    owner_node: terminal.clone(),
994                }
995            } else {
996                AllocationDisposition::Release
997            },
998        },
999    }

1002fn overflow() -> io::Error {
1003    io::Error::other("cube state storage layout overflow")
1004}
1005fn add(left: usize, right: usize) -> io::Result<usize> {
1006    left.checked_add(right).ok_or_else(overflow)
1007}
1008fn as_u64(value: usize) -> io::Result<u64> {
1009    u64::try_from(value).map_err(|_| overflow())
