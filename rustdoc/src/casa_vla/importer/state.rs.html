115    fn initialize_epoch_measure_references(&mut self) -> Result<(), VlaError> {
116        for column in ["TIME", "TIME_CENTROID"] {
117            set_epoch_measure_reference(self.ms.main_table_mut(), column, EpochRef::TAI)?;
118        }
119        {
120            let mut field = self.ms.field_mut().map_err(|error| {
121                VlaError::import(format!("open FIELD for epoch metadata update: {error}"))
122            })?;
123            set_epoch_measure_reference(field.table_mut(), "TIME", EpochRef::TAI)?;
124        }
125        {
126            let mut feed = self.ms.feed_mut().map_err(|error| {
127                VlaError::import(format!("open FEED for epoch metadata update: {error}"))
128            })?;
129            set_epoch_measure_reference(feed.table_mut(), "TIME", EpochRef::TAI)?;
130        }
131        {
132            let mut source = self.ms.source_mut().map_err(|error| {
133                VlaError::import(format!("open SOURCE for epoch metadata update: {error}"))
134            })?;
135            set_epoch_measure_reference(source.table_mut(), "TIME", EpochRef::TAI)?;
136        }
137        {
138            let mut observation = self.ms.observation_mut().map_err(|error| {
139                VlaError::import(format!(
140                    "open OBSERVATION for epoch metadata update: {error}"
141                ))
142            })?;
143            for column in ["TIME_RANGE", "RELEASE_DATE"] {
144                set_epoch_measure_reference(observation.table_mut(), column, EpochRef::TAI)?;
145            }
146        }
147        Ok(())
148    }

169                })?;
170        }
171        Ok(())

174    pub(super) fn push_record(&mut self, record: NormalizedRecord) -> Result<(), VlaError> {
175        let mut new_scan = self.new_scan_pending;
176        self.new_scan_pending = false;
177        self.pending_record_freq_group = None;
178        let ms_direction_epoch = self.ensure_direction_epoch(record.direction_epoch)?;
179        let source_direction = align_direction_for_epoch(
180            record.source_direction,
181            record.direction_epoch,
182            ms_direction_epoch,
183        )?;
184        let observation_id = self.observe_record(&record)?;
185        let field_key = FieldKey {
186            source_name: record.source_name.clone(),
187            calibration_code: record.calibration_code.clone(),
188            ra_bits: source_direction[0].to_bits(),
189            dec_bits: source_direction[1].to_bits(),
190        };
191        let source_id = self.ensure_source(&record, &field_key, source_direction)?;
192        let field_id = self.ensure_field(&record, &field_key, source_direction, source_id)?;
193        if self.last_field_by_array.get(&record.array_id).copied() != Some(field_id) {
194            new_scan = true;
195            self.last_field_by_array.insert(record.array_id, field_id);
196        }
197
198        let array_id = record.array_id;
199        let time_seconds = record.time_seconds;
200        let integration_seconds = record.integration_seconds;
201        let antennas = record.antennas;
202        let mut sorted_antennas = antennas.iter().enumerate().collect::<Vec<_>>();
203        sorted_antennas.sort_by_key(|(_, antenna)| antenna.antenna_id);
204        if self.last_antenna_rows.len() != antennas.len() {
205            self.last_antenna_rows.clear();
206        }
207        let mut antenna_ids = vec![0; antennas.len()];
208        let mut antenna_added = false;
209        for (archive_index, antenna) in sorted_antennas {
210            let preferred = self.last_antenna_rows.get(archive_index).copied();
211            let (id, inserted) = self.ensure_antenna(antenna, preferred)?;
212            antenna_ids[archive_index] = id;
213            antenna_added |= inserted;
214        }
215        new_scan |= antenna_added;
216
217        let mut group_data_desc_ids = Vec::with_capacity(record.groups.len());
218        let mut group_rows = Vec::with_capacity(record.groups.len());
219        let mut current_spectral_windows = [None; 4];
220
221        for group in record.groups {
222            let polarization_id = self.ensure_polarization(&group.baselines[0].corr_types)?;
223            let preferred = group
224                .cda_ids
225                .first()
226                .and_then(|id| self.last_spectral_windows[id.index()]);
227            let spectral_window_id = self.ensure_spectral_window(
228                &group.descriptor,
229                source_id,
230                time_seconds,
231                source_direction,
232                ms_direction_epoch,
233                preferred,
234            )?;
235            for id in group.cda_ids {
236                current_spectral_windows[id.index()] = Some(spectral_window_id as usize);
237            }
238            let data_desc_id = self.ensure_data_description(spectral_window_id, polarization_id)?;
239            group_data_desc_ids.push(data_desc_id);
240            group_rows.push((group.baselines, data_desc_id));
241        }
242        self.last_spectral_windows = current_spectral_windows;
243
244        if self
245            .last_data_desc_ids_by_array
246            .get(&array_id)
247            .is_none_or(|existing| existing != &group_data_desc_ids)
248        {
249            new_scan = true;
250            self.last_data_desc_ids_by_array
251                .insert(array_id, group_data_desc_ids);
252        }
253
254        let scan_number = if new_scan {
255            let scan_number = self.next_scan_number;
256            self.next_scan_number += 1;
257            self.current_scan_by_array.insert(array_id, scan_number);
258            scan_number
259        } else {
260            *self.current_scan_by_array.get(&array_id).ok_or_else(|| {
261                VlaError::import(format!(
262                    "missing current scan number for subarray {}",
263                    array_id
264                ))
265            })?
266        };
267
268        for (baselines, data_desc_id) in group_rows {
269            for baseline in baselines {
270                let antenna1_id = antenna_ids[baseline.antenna1_archive];
271                let antenna2_id = antenna_ids[baseline.antenna2_archive];
272                self.add_main_row(
273                    antenna1_id,
274                    antenna2_id,
275                    field_id,
276                    observation_id,
277                    data_desc_id,
278                    scan_number,
279                    array_id,
280                    time_seconds,
281                    integration_seconds,
282                    baseline,
283                )?;
284            }
285        }
286        self.last_antenna_rows = antenna_ids;
287
288        Ok(())
289    }

291    pub(super) fn finish(&mut self) -> Result<(), VlaError> {
292        let finish_started = Instant::now();
293        self.finish_archive()?;
294        let main = self.ms.main_table_mut();
295        let mut scan_keywords = main
296            .column_keywords("SCAN_NUMBER")
297            .cloned()
298            .unwrap_or_default();
299        scan_keywords.upsert(
300            "LAST_SCAN",
301            Value::Array(ArrayValue::from_i32_vec(
302                (0..VLA_SUBARRAY_COUNT)
303                    .map(|array| self.current_scan_by_array.get(&array).copied().unwrap_or(0))
304                    .collect(),
305            )),
306        );
307        main.set_column_keywords("SCAN_NUMBER", scan_keywords);
308        self.perf_summary.finish_metadata_ns = finish_started.elapsed().as_nanos() as u64;
309        let save_started = Instant::now();
310        let main_write_session = self
311            .main_write_session
312            .take()
313            .ok_or_else(|| VlaError::import("MAIN write session was already finalized"))?;
314        let mut overrides = casa_tables::ColumnOverrides::for_row_count(self.main_rows_written);
315        for column in [
316            "DATA",
317            "CORRECTED_DATA",
318            "MODEL_DATA",
319            "FLAG",
320            "FLAG_CATEGORY",
321            "WEIGHT",
322            "SIGMA",
323        ] {
324            overrides.insert_deferred(column);
325        }
326        overrides.insert_deferred("UVW");
327        let write_telemetry = main_write_session
328            .save_and_finish(&mut self.ms, &overrides)
329            .map_err(|error| VlaError::import(error.to_string()))?;
330        self.perf_summary.write_producer_ns = seconds_to_nanos(write_telemetry.producer_seconds);
331        self.perf_summary.write_queue_wait_ns =
332            seconds_to_nanos(write_telemetry.queue_wait_seconds);
333        self.perf_summary.write_assemble_ns = seconds_to_nanos(write_telemetry.assemble_seconds);
334        self.perf_summary.write_physical_ns = seconds_to_nanos(write_telemetry.write_seconds);
335        self.perf_summary.write_finalize_ns = seconds_to_nanos(write_telemetry.finalize_seconds);
336        self.perf_summary.write_rows = write_telemetry.rows_written;
337        self.perf_summary.write_bytes = write_telemetry.bytes_written;
338        self.perf_summary.write_peak_planned_resident_bytes =
339            write_telemetry.maximum_resident_bytes;
340        let result = Ok(());
341        self.perf_summary.save_ns = save_started.elapsed().as_nanos() as u64;
342        result
343    }

345    fn ensure_direction_epoch(
346        &mut self,
347        epoch: DirectionEpoch,
348    ) -> Result<DirectionEpoch, VlaError> {
349        if let Some(existing) = self.ms_direction_epoch {
350            return Ok(existing);
351        }
352        set_direction_measure_reference(self.ms.main_table_mut(), "UVW", epoch)?;
353        {
354            let mut field = self.ms.field_mut().map_err(|error| {
355                VlaError::import(format!("open FIELD for metadata update: {error}"))
356            })?;
357            for column in ["DELAY_DIR", "PHASE_DIR", "REFERENCE_DIR"] {
358                set_direction_measure_reference(field.table_mut(), column, epoch)?;
359            }
360        }
361        {
362            let mut source = self.ms.source_mut().map_err(|error| {
363                VlaError::import(format!("open SOURCE for metadata update: {error}"))
364            })?;
365            set_direction_measure_reference(source.table_mut(), "DIRECTION", epoch)?;
366        }
367        self.ms_direction_epoch = Some(epoch);
368        Ok(epoch)
369    }

371    fn ensure_antenna(
372        &mut self,
373        antenna: &NormalizedAntenna,
374        preferred: Option<i32>,
375    ) -> Result<(i32, bool), VlaError> {
376        if let Some(entries) = self.antenna_rows.get(&antenna.name) {
377            let matches_position = |position: &[f64; 3]| {
378                position
379                    .iter()
380                    .zip(antenna.position_itrf_m)
381                    .all(|(stored, incoming)| (stored - incoming).abs() <= 1.0)
382            };
383            let previous = preferred.and_then(|id| entries.iter().find(|(row, _)| *row == id));
384            if let Some(&(row, _)) = previous
385                .into_iter()
386                .chain(entries.iter().rev())
387                .find(|(_, position)| matches_position(position))
388            {
389                return Ok((row, false));
390            }
391        }
392        let row = {
393            let mut ant = self
394                .ms
395                .antenna_mut()
396                .map_err(|error| VlaError::import(format!("open ANTENNA for write: {error}")))?;
397            i32::try_from(
398                ant.add_antenna(
399                    &antenna.name,
400                    &antenna.station,
401                    "GROUND-BASED",
402                    "ALT-AZ",
403                    antenna.position_itrf_m,
404                    [0.0, 0.0, 0.0],
405                    25.0,
406                )
407                .map_err(|error| VlaError::import(format!("add ANTENNA row: {error}")))?,
408            )
409            .map_err(|_| VlaError::import("ANTENNA row index does not fit in i32"))?
410        };
411        self.add_feed_row(row)?;
412        self.antenna_rows
413            .entry(antenna.name.clone())
414            .or_default()
415            .push((row, antenna.position_itrf_m));
416        Ok((row, true))
417    }

419    fn add_feed_row(&mut self, antenna_id: i32) -> Result<(), VlaError> {
420        let feed_table = self
421            .ms
422            .subtable_mut(SubtableId::Feed)
423            .ok_or_else(|| VlaError::import("missing FEED subtable"))?;
424        let pol_response = ArrayValue::Complex32(
425            ArrayD::from_shape_vec(
426                IxDyn(&[2, 2]).f(),
427                vec![
428                    Complex32::new(1.0, 0.0),
429                    Complex32::new(0.0, 0.0),
430                    Complex32::new(0.0, 0.0),
431                    Complex32::new(1.0, 0.0),
432                ],
433            )
434            .map_err(|error| VlaError::import(format!("shape FEED.POL_RESPONSE: {error}")))?,
435        );
436        let polarization_type = ArrayValue::String(
437            ArrayD::from_shape_vec(IxDyn(&[2]).f(), vec!["R".to_string(), "L".to_string()])
438                .map_err(|error| {
439                    VlaError::import(format!("shape FEED.POLARIZATION_TYPE: {error}"))
440                })?,
441        );
442        let receptor_angle = ArrayValue::Float64(
443            ArrayD::from_shape_vec(IxDyn(&[2]).f(), vec![0.0, 0.0])
444                .map_err(|error| VlaError::import(format!("shape FEED.RECEPTOR_ANGLE: {error}")))?,
445        );
446        let beam_offset = ArrayValue::Float64(
447            ArrayD::from_shape_vec(IxDyn(&[2, 2]).f(), vec![0.0; 4])
448                .map_err(|error| VlaError::import(format!("shape FEED.BEAM_OFFSET: {error}")))?,
449        );
450        let position = ArrayValue::Float64(
451            ArrayD::from_shape_vec(IxDyn(&[3]).f(), vec![0.0, 0.0, 0.0])
452                .map_err(|error| VlaError::import(format!("shape FEED.POSITION: {error}")))?,
453        );
454        let row = make_row_from_columns(
455            casa_ms::schema::feed::REQUIRED_COLUMNS,
456            &[
457                ("ANTENNA_ID", Value::Scalar(ScalarValue::Int32(antenna_id))),
458                ("BEAM_ID", Value::Scalar(ScalarValue::Int32(-1))),
459                ("BEAM_OFFSET", Value::Array(beam_offset)),
460                ("FEED_ID", Value::Scalar(ScalarValue::Int32(0))),
461                ("INTERVAL", Value::Scalar(ScalarValue::Float64(0.0))),
462                ("NUM_RECEPTORS", Value::Scalar(ScalarValue::Int32(2))),
463                ("POL_RESPONSE", Value::Array(pol_response)),
464                ("POLARIZATION_TYPE", Value::Array(polarization_type)),
465                ("POSITION", Value::Array(position)),
466                ("RECEPTOR_ANGLE", Value::Array(receptor_angle)),
467                ("SPECTRAL_WINDOW_ID", Value::Scalar(ScalarValue::Int32(-1))),
468                ("TIME", Value::Scalar(ScalarValue::Float64(0.0))),
469            ],
470        )?;
471        feed_table
472            .prepare_row_appender()
473            .and_then(|mut appender| appender.append(row))
474            .map_err(|error| VlaError::import(format!("add FEED row: {error}")))
475    }

477    fn observe_record(&mut self, record: &NormalizedRecord) -> Result<i32, VlaError> {
478        if self.archive_observation.is_none() {
479            let table = self
480                .ms
481                .subtable(SubtableId::Observation)
482                .ok_or_else(|| VlaError::import("missing OBSERVATION subtable"))?;
483            let id = i32::try_from(table.row_count())
484                .map_err(|_| VlaError::import("OBSERVATION row count too large"))?;
485            self.archive_observation = Some(ArchiveObservation {
486                id,
487                projects: String::new(),
488                start: record.time_seconds,
489                end: record.time_seconds,
490            });
491        }
492        let observation = self
493            .archive_observation
494            .as_mut()
495            .expect("initialized above");
496        if !observation.projects.contains(&record.project) {
497            if !observation.projects.is_empty() {
498                observation.projects.push(' ');
499            }
500            observation.projects.push_str(&record.project);
501        }
502        observation.end = record.time_seconds;
503        Ok(observation.id)
504    }

506    pub(super) fn finish_archive(&mut self) -> Result<(), VlaError> {
507        self.new_scan_pending = true;
508        self.last_field_by_array.clear();
509        self.last_data_desc_ids_by_array.clear();
510        self.last_spectral_windows = [None; 4];
511        self.last_antenna_rows.clear();
512        let Some(observation) = self.archive_observation.take() else {
513            return Ok(());
514        };
515        let log = ArrayValue::String(
516            ArrayD::from_shape_vec(IxDyn(&[1]).f(), vec!["unavailable".to_string()])
517                .map_err(|error| VlaError::import(format!("shape OBSERVATION.LOG: {error}")))?,
518        );
519        let schedule = ArrayValue::String(
520            ArrayD::from_shape_vec(IxDyn(&[1]).f(), vec!["unavailable".to_string()]).map_err(
521                |error| VlaError::import(format!("shape OBSERVATION.SCHEDULE: {error}")),
522            )?,
523        );
524        let time_range = ArrayValue::Float64(
525            ArrayD::from_shape_vec(IxDyn(&[2]).f(), vec![observation.start, observation.end])
526                .map_err(|error| {
527                    VlaError::import(format!("shape OBSERVATION.TIME_RANGE: {error}"))
528                })?,
529        );
530        let table = self
531            .ms
532            .subtable_mut(SubtableId::Observation)
533            .ok_or_else(|| VlaError::import("missing OBSERVATION subtable"))?;
534        let row = make_row_from_columns(
535            casa_ms::schema::observation::REQUIRED_COLUMNS,
536            &[
537                ("FLAG_ROW", Value::Scalar(ScalarValue::Bool(false))),
538                ("LOG", Value::Array(log)),
539                (
540                    "OBSERVER",
541                    Value::Scalar(ScalarValue::String("unavailable".to_string())),
542                ),
543                (
544                    "PROJECT",
545                    Value::Scalar(ScalarValue::String(observation.projects)),
546                ),
547                (
548                    "RELEASE_DATE",
549                    Value::Scalar(ScalarValue::Float64(
550                        observation.end + 1.5 * SECONDS_PER_JULIAN_YEAR,
551                    )),
552                ),
553                ("SCHEDULE", Value::Array(schedule)),
554                (
555                    "SCHEDULE_TYPE",
556                    Value::Scalar(ScalarValue::String("unknown".to_string())),
557                ),
558                (
559                    "TELESCOPE_NAME",
560                    Value::Scalar(ScalarValue::String(DEFAULT_TELESCOPE_NAME.to_string())),
561                ),
562                ("TIME_RANGE", Value::Array(time_range)),
563            ],
564        )?;
565        table
566            .prepare_row_appender()
567            .and_then(|mut appender| appender.append(row))
568            .map_err(|error| VlaError::import(format!("add OBSERVATION row: {error}")))?;
569        Ok(())
570    }

572    fn ensure_source(

574        record: &NormalizedRecord,
575        field_key: &FieldKey,
576        source_direction: [f64; 2],
577    ) -> Result<i32, VlaError> {
578        if let Some(&row) = self.source_rows.get(field_key) {
579            return Ok(row);
580        }
581        let direction = ArrayValue::Float64(
582            ArrayD::from_shape_vec(
583                IxDyn(&[2]).f(),
584                vec![source_direction[0], source_direction[1]],
585            )
586            .map_err(|error| VlaError::import(format!("shape SOURCE.DIRECTION: {error}")))?,
587        );
588        let proper_motion = ArrayValue::Float64(
589            ArrayD::from_shape_vec(IxDyn(&[2]).f(), vec![0.0, 0.0]).map_err(|error| {
590                VlaError::import(format!("shape SOURCE.PROPER_MOTION: {error}"))
591            })?,
592        );
593        let row = {
594            let table = self
595                .ms
596                .subtable_mut(SubtableId::Source)
597                .ok_or_else(|| VlaError::import("missing SOURCE subtable"))?;
598            let row_index = i32::try_from(table.row_count())
599                .map_err(|_| VlaError::import("SOURCE row count too large"))?;
600            let row = make_row_from_columns(
601                casa_ms::schema::source::REQUIRED_COLUMNS,
602                &[
603                    ("CALIBRATION_GROUP", Value::Scalar(ScalarValue::Int32(0))),
604                    (
605                        "CODE",
606                        Value::Scalar(ScalarValue::String(record.calibration_code.clone())),
607                    ),
608                    ("DIRECTION", Value::Array(direction)),
609                    ("INTERVAL", Value::Scalar(ScalarValue::Float64(0.0))),
610                    (
611                        "NAME",
612                        Value::Scalar(ScalarValue::String(record.source_name.clone())),
613                    ),
614                    (
615                        "NUM_LINES",
616                        Value::Scalar(ScalarValue::Int32(record.source_num_lines)),
617                    ),
618                    ("PROPER_MOTION", Value::Array(proper_motion)),
619                    ("SOURCE_ID", Value::Scalar(ScalarValue::Int32(row_index))),
620                    ("SPECTRAL_WINDOW_ID", Value::Scalar(ScalarValue::Int32(-1))),
621                    (
622                        "TIME",
623                        Value::Scalar(ScalarValue::Float64(record.time_seconds)),
624                    ),
625                ],
626            )?;
627            table
628                .prepare_row_appender()
629                .and_then(|mut appender| appender.append(row))
630                .map_err(|error| VlaError::import(format!("add SOURCE row: {error}")))?;
631            row_index
632        };
633        self.source_rows.insert(field_key.clone(), row);
634        self.last_source_row = Some(row);
635        Ok(row)
636    }

740    fn ensure_doppler(
741        &mut self,
742        source_id: i32,
743        descriptor: &SpectralDescriptor,
744    ) -> Result<i32, VlaError> {
745        let source_id = self.last_source_row.unwrap_or(source_id);
746        if let Some((_, row)) = self
747            .doppler_rows
748            .iter()
749            .find(|((existing_source, existing), _)| {
750                *existing_source == source_id
751                    && same_spectral_descriptor(existing, descriptor, self.options.frequencytol_hz)
752            })
753        {
754            return Ok(*row);
755        }
756        let row = {
757            let table = self
758                .ms
759                .subtable_mut(SubtableId::Doppler)
760                .ok_or_else(|| VlaError::import("missing DOPPLER subtable"))?;
761            let row_index = i32::try_from(table.row_count())
762                .map_err(|_| VlaError::import("DOPPLER row count too large"))?;
763            let row = make_row_from_columns(
764                casa_ms::schema::doppler::REQUIRED_COLUMNS,
765                &[
766                    ("DOPPLER_ID", Value::Scalar(ScalarValue::Int32(row_index))),
767                    ("SOURCE_ID", Value::Scalar(ScalarValue::Int32(source_id))),
768                    ("TRANSITION_ID", Value::Scalar(ScalarValue::Int32(0))),
769                    (
770                        "VELDEF",
771                        Value::Scalar(ScalarValue::Float64(if descriptor.doppler_tracking {
772                            descriptor.doppler_velocity_mps
773                        } else {
774                            0.0
775                        })),
776                    ),
777                ],
778            )?;
779            table
780                .prepare_row_appender()
781                .and_then(|mut appender| appender.append(row))
782                .map_err(|error| VlaError::import(format!("add DOPPLER row: {error}")))?;
783            row_index
784        };
785        self.doppler_rows
786            .push(((source_id, descriptor.clone()), row));
787        Ok(row)
788    }

790    fn ensure_spectral_window(
791        &mut self,
792        descriptor: &SpectralDescriptor,
793        source_id: i32,
794        time_seconds: f64,
795        source_direction: [f64; 2],
796        direction_epoch: DirectionEpoch,
797        preferred: Option<usize>,
798    ) -> Result<i32, VlaError> {
799        let measures: Arc<dyn MeasuresProvider> = self.measures.clone();
800        let previous = preferred.and_then(|index| self.spectral_windows.get(index));
801        let newest = self
802            .spectral_windows
803            .iter()
804            .rev()
805            .filter(|entry| Some(entry.row as usize) != preferred);
806        for entry in previous.into_iter().chain(newest) {
807            if spectral_window_matches(
808                entry,
809                descriptor,
810                time_seconds,
811                source_direction,
812                direction_epoch,
813                self.options.frequencytol_hz,
814                measures.clone(),
815            )? {
816                return Ok(entry.row);
817            }
818        }
819        let freq_group = if let Some(group) = self.pending_record_freq_group {
820            group
821        } else {
822            self.current_freq_group += 1;
823            self.pending_record_freq_group = Some(self.current_freq_group);
824            self.current_freq_group
825        };
826
827        let frame =
828            spectral_conversion_frame(time_seconds, source_direction, direction_epoch, measures)?;
829        let ref_freq = spectral_reference_frequency(descriptor);
830        let meas_freq_ref = frequency_reference_for_descriptor(descriptor);
831        let chan_freq: Vec<f64> = (0..descriptor.num_chan)
832            .map(|index| {
833                descriptor.edge_frequency_hz + (index as f64 + 0.5) * descriptor.channel_width_hz
834            })
835            .map(|hz| convert_topocentric_frequency(hz, meas_freq_ref, &frame))
836            .collect::<Result<Vec<_>, _>>()?;
837        let converted_width = if meas_freq_ref == FrequencyRef::TOPO {
838            descriptor.channel_width_hz
839        } else {
840            convert_topocentric_frequency(descriptor.channel_width_hz, meas_freq_ref, &frame)?
841        };
842        let widths = vec![converted_width; descriptor.num_chan];
843        let stored_bandwidth_hz = widths.iter().sum();
844        let resolution = widths.clone();
845        let chan_freq_array = ArrayValue::Float64(
846            ArrayD::from_shape_vec(IxDyn(&[descriptor.num_chan]).f(), chan_freq).map_err(
847                |error| VlaError::import(format!("shape SPECTRAL_WINDOW.CHAN_FREQ: {error}")),
848            )?,
849        );
850        let width_array = |values: Vec<f64>, column: &str| -> Result<ArrayValue, VlaError> {
851            Ok(ArrayValue::Float64(
852                ArrayD::from_shape_vec(IxDyn(&[descriptor.num_chan]).f(), values).map_err(
853                    |error| VlaError::import(format!("shape SPECTRAL_WINDOW.{column}: {error}")),
854                )?,
855            ))
856        };
857        let doppler_id = self.ensure_doppler(source_id, descriptor)?;
858        let row = {
859            let table = self
860                .ms
861                .subtable_mut(SubtableId::SpectralWindow)
862                .ok_or_else(|| VlaError::import("missing SPECTRAL_WINDOW subtable"))?;
863            let row_index = i32::try_from(table.row_count())
864                .map_err(|_| VlaError::import("SPECTRAL_WINDOW row count too large"))?;
865            let mut row = make_row_from_columns(
866                casa_ms::schema::spectral_window::REQUIRED_COLUMNS,
867                &[
868                    (
869                        "NUM_CHAN",
870                        Value::Scalar(ScalarValue::Int32(
871                            i32::try_from(descriptor.num_chan)
872                                .map_err(|_| VlaError::import("NUM_CHAN does not fit in i32"))?,
873                        )),
874                    ),
875                    (
876                        "NAME",
877                        Value::Scalar(ScalarValue::String(spectral_window_name(
878                            descriptor,
879                            ref_freq.hz(),
880                            meas_freq_ref,
881                        ))),
882                    ),
883                    (
884                        "REF_FREQUENCY",
885                        Value::Scalar(ScalarValue::Float64(ref_freq.hz())),
886                    ),
887                    (
888                        "TOTAL_BANDWIDTH",
889                        Value::Scalar(ScalarValue::Float64(stored_bandwidth_hz)),
890                    ),
891                    ("CHAN_FREQ", Value::Array(chan_freq_array)),
892                    (
893                        "CHAN_WIDTH",
894                        Value::Array(width_array(widths.clone(), "CHAN_WIDTH")?),
895                    ),
896                    (
897                        "EFFECTIVE_BW",
898                        Value::Array(width_array(widths, "EFFECTIVE_BW")?),
899                    ),
900                    (
901                        "RESOLUTION",
902                        Value::Array(width_array(resolution, "RESOLUTION")?),
903                    ),
904                    (
905                        "MEAS_FREQ_REF",
906                        Value::Scalar(ScalarValue::Int32(meas_freq_ref.casacore_code())),
907                    ),
908                    ("NET_SIDEBAND", Value::Scalar(ScalarValue::Int32(1))),
909                    ("FREQ_GROUP", Value::Scalar(ScalarValue::Int32(freq_group))),
910                    (
911                        "FREQ_GROUP_NAME",
912                        Value::Scalar(ScalarValue::String(format!("Group {freq_group}"))),
913                    ),
914                    (
915                        "IF_CONV_CHAIN",
916                        Value::Scalar(ScalarValue::Int32(descriptor.if_conv_chain)),
917                    ),
918                    ("FLAG_ROW", Value::Scalar(ScalarValue::Bool(false))),
919                ],
920            )?;
921            row.push(RecordField::new(
922                "DOPPLER_ID",
923                Value::Scalar(ScalarValue::Int32(doppler_id)),
924            ));
925            table
926                .prepare_row_appender()
927                .and_then(|mut appender| appender.append(row))
928                .map_err(|error| VlaError::import(format!("add SPECTRAL_WINDOW row: {error}")))?;
929            row_index
930        };
931        self.spectral_windows.push(SpectralWindowEntry {
932            descriptor: descriptor.clone(),
933            stored_bandwidth_hz,
934            row,
935            time_seconds,
936            source_direction,
937            direction_epoch,
938        });
939        Ok(row)
940    }

985    #[allow(clippy::too_many_arguments)]
986    fn add_main_row(
987        &mut self,
988        antenna1_id: i32,
989        antenna2_id: i32,
990        field_id: i32,
991        observation_id: i32,
992        data_desc_id: i32,
993        scan_number: i32,
994        array_id: i32,
995        time_seconds: f64,
996        integration_seconds: f64,
997        baseline: NormalizedBaselineRow,
998    ) -> Result<(), VlaError> {
999        let NormalizedBaselineRow {
1000            corr_types,
1001            uvw_m,
1002            data,
1003            corrected_data,
1004            model_data,
1005            flag,
1006            flag_category,
1007            weight,
1008            sigma,
1009            ..
1010        } = baseline;
1011        let (
1012            row_ant1,
1013            row_ant2,
1014            row_data,
1015            row_corrected,
1016            row_model,
1017            row_flag,
1018            row_flag_category,
1019            row_weight,
1020            row_sigma,
1021            _row_corr_types,
1022        ) = {
1023            let reorder_started = Instant::now();
1024            let result = reorder_baseline_for_ms(
1025                antenna1_id,
1026                antenna2_id,
1027                data,
1028                corrected_data,
1029                model_data,
1030                flag,
1031                flag_category,
1032                weight,
1033                sigma,
1034                &corr_types,
1035            )?;
1036            self.perf_summary.reorder_baseline_ns += reorder_started.elapsed().as_nanos() as u64;
1037            result
1038        };
1039        let row_uvw = if row_ant1 == antenna1_id && row_ant2 == antenna2_id {
1040            uvw_m
1041        } else {
1042            [-uvw_m[0], -uvw_m[1], -uvw_m[2]]
1043        };
1044        let row_flag_row = {
1045            let flag_started = Instant::now();
1046            let value = flag_row_value(&row_flag)?;
1047            self.perf_summary.flag_row_ns += flag_started.elapsed().as_nanos() as u64;
1048            value
1049        };
1050        {
1051            let writer = self
1052                .main_write_session
1053                .as_mut()
1054                .ok_or_else(|| VlaError::import("MAIN write session is unavailable"))?;
1055            let append_started = Instant::now();
1056            push_import_complex32_row(writer, "DATA", &row_data)?;
1057            push_import_complex32_row(writer, "CORRECTED_DATA", &row_corrected)?;
1058            push_import_complex32_row(writer, "MODEL_DATA", &row_model)?;
1059            push_import_bool_row(writer, "FLAG", &row_flag)?;
1060            push_import_bool_row(writer, "FLAG_CATEGORY", &row_flag_category)?;
1061            push_import_f32_row(writer, "WEIGHT", &row_weight)?;
1062            push_import_f32_row(writer, "SIGMA", &row_sigma)?;
1063            writer
1064                .push_f64_row("UVW", &[3], &row_uvw)
1065                .map_err(|error| VlaError::import(error.to_string()))?;
1066            push_import_scalar_row(
1067                writer,
1068                row_ant1,
1069                row_ant2,
1070                array_id,
1071                data_desc_id,
1072                integration_seconds,
1073                field_id,
1074                row_flag_row,
1075                observation_id,
1076                scan_number,
1077                time_seconds,
1078            )?;
1079            self.perf_summary.append_main_row_ns += append_started.elapsed().as_nanos() as u64;
1080        }
1081        self.main_rows_written += 1;
1082        Ok(())
1083    }
