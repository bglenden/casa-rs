218impl MsTransformTaskOptions {
219    fn check_cancelled(&self) -> Result<(), MsTransformError> {
220        if self.cancel.as_ref().is_some_and(Cancellation::is_cancelled) {
221            return Err(MsTransformError::Interrupted);
222        }
223        Ok(())
224    }

558
559fn materialize_transform(
560    request: &MsTransformRequest,
561    grammar: RowGrammar,
562    options: &MsTransformTaskOptions,
563) -> Result<MsTransformReport, MsTransformError> {
564    if request.input_ms == request.output_ms {
565        return Err(MsTransformError::SameInputOutput {
566            path: request.input_ms.display().to_string(),
567        });
568    }
569    if request.output_ms.symlink_metadata().is_ok() {
570        return Err(MsTransformError::OutputExists {
571            path: request.output_ms.display().to_string(),
572        });
573    }
574    let flagversions = casa_ms::flagversions_dir(&request.output_ms);
575    if flagversions.exists() {
576        return Err(MsTransformError::OutputFlagVersionsExist {
577            path: flagversions.display().to_string(),
578        });
579    }
580    options.check_cancelled()?;
581    let output = &request.output_ms;
582    if let Some(parent) = output.parent().filter(|path| !path.as_os_str().is_empty()) {
583        fs::create_dir_all(parent).map_err(|error| output_error(output, error))?;
584    }
585    let stage = Stage::reserve(output, "casa-rs-incomplete")
586        .map_err(|error| output_error(output, error))?;
587    let result = stage_then_publish_transform(request, grammar, options, &stage);
588    stage.remove();
589    result
590}
591
592fn output_error(output: &Path, error: impl ToString) -> MsTransformError {
593    MsTransformError::PrepareOutput {
594        path: output.display().to_string(),
595        reason: error.to_string(),
596    }
597}

599fn stage_then_publish_transform(
600    request: &MsTransformRequest,
601    grammar: RowGrammar,
602    options: &MsTransformTaskOptions,
603    stage: &Stage,
604) -> Result<MsTransformReport, MsTransformError> {
605    let mut physical_request = request.clone();
606    physical_request.output_ms = stage.path().to_path_buf();
607    if let Some(configuration) = options.multi_ms
608        && let Some(mut report) = multi_ms::transform(
609            &physical_request,
610            &request.output_ms,
611            grammar,
612            options,
613            configuration,
614        )?
615    {
616        if report.row_count > 0 {
617            publish_transform_stage(request, options, stage)?;
618        } else {
619            options.check_cancelled()?;
620        }
621        report.output_ms.clone_from(&request.output_ms);
622        return Ok(report);
623    }
624    let members = if options.keep_mms.is_some() {
625        multi_ms::members(&request.input_ms)?
626    } else {
627        None
628    };
629    let mut report = match (options.keep_mms, members) {
630        (Some(true), Some(members)) => {
631            let Some(report) = multi_ms::retain(
632                &physical_request,
633                &request.output_ms,
634                grammar,
635                options,
636                &members,
637            )?
638            else {
639                options.check_cancelled()?;
640                return Ok(empty_transform_report(request));
641            };
642            report
643        }
644        (Some(false), Some(members)) => transform_product(
645            &physical_request,
646            grammar,
647            options,
648            members.first().map(PathBuf::as_path),
649            None,
650        )?,
651        _ => transform_product(&physical_request, grammar, options, None, None)?,
652    };
653    publish_transform_stage(request, options, stage)?;
654    report.output_ms = request.output_ms.clone();
655    Ok(report)
656}
657
658fn publish_transform_stage(
659    request: &MsTransformRequest,
660    options: &MsTransformTaskOptions,
661    stage: &Stage,
662) -> Result<(), MsTransformError> {
663    #[cfg(feature = "transform-test-hooks")]
664    test_hooks::notify(test_hooks::Checkpoint::BeforePublication);
665    options.check_cancelled()?;
666    stage
667        .publish_new(&request.output_ms)
668        .map_err(|error| output_error(&request.output_ms, error))
669}

705fn transform_selected_product(
706    request: &MsTransformRequest,
707    grammar: RowGrammar,
708    options: &MsTransformTaskOptions,
709    metadata_template: Option<&Path>,
710    operation: Option<SpectralOperation<'_>>,
711    partition: Option<TransformSelection>,
712) -> Result<MsTransformReport, MsTransformError> {
713    options.check_cancelled()?;
714    let report = mstransform_staged(
715        request,
716        grammar,
717        options,
718        metadata_template,
719        operation,
720        partition,
721    )?;
722    options.check_cancelled()?;

725    }
726    Ok(report)

729struct TransformSelection {
730    selection: MsSelection,
731    correlations: Option<CorrelationSelection>,
732    time: Option<TimeSelection>,
733    rows: Vec<ResolvedMsSelectionRow>,
734    antenna_ids: Option<Vec<i32>>,
735    tile_window: Option<i32>,
736}
737
738fn resolve_transform_selection(
739    input: &MeasurementSet,
740    request: &MsTransformRequest,
741    grammar: RowGrammar,
742    budget: MsSelectionIoBudget,
743    auto_correlations: bool,
744) -> Result<TransformSelection, MsTransformError> {
745    let error = |reason: MsSelectionError| MsTransformError::SelectRows {
746        path: request.input_ms.display().to_string(),
747        source: match reason {
748            MsSelectionError::Domain(source) => source,
749            other => MsError::InvalidInput(other.to_string()),
750        },
751    };
752    let mut selection = request.selection.clone();
753    if auto_correlations {
754        selection.antenna = Some(convolution::auto_correlations(
755            input,
756            selection.antenna.as_deref(),
757        )?);
758    }
759    if selection.spw.is_none() {
760        let spw = request.spw.trim();
761        selection.spw = Some(if spw.is_empty() { "*" } else { spw }.into());
762    }
763    let correlations = if grammar == RowGrammar::Transform {
764        selection
765            .correlation
766            .take()
767            .filter(|text| !text.is_empty())
768            .map(|text| CorrelationSelection::resolve(input, &text))
769            .transpose()
770            .map_err(|source| error(MsSelectionError::Domain(source)))?
771    } else {
772        None
773    };
774    if let Some(correlations) = correlations.as_ref()
775        && !correlations
776            .constrain(&mut selection)
777            .map_err(|source| error(MsSelectionError::Domain(source)))?
778    {
779        return Err(MsTransformError::EmptySelection {
780            path: request.input_ms.display().to_string(),
781        });
782    }
783    let (resolved, time) = input
784        .resolve_selection_with_time(&selection, budget, grammar)
785        .map_err(error)?;
786    let rows = resolved.selected_rows;
787    let antenna_ids =
788        selection_metadata::selected_antenna_ids(input, selection.antenna.as_deref(), &rows)?;
789    Ok(TransformSelection {
790        selection,
791        correlations,
792        time,
793        rows,
794        antenna_ids,
795        tile_window: None,
796    })
797}

816    #[test]
817    fn select_channels_covers_contiguous_noncontiguous_and_full_width_paths() {
818        let complex = ArrayValue::Complex32(
819            ArrayD::from_shape_vec(
820                IxDyn(&[2, 4]).f(),
821                (0..8)
822                    .map(|value| Complex32::new(value as f32, value as f32 + 0.5))
823                    .collect(),
824            )
825            .unwrap(),
826        );
827        let selected = select_channels(complex, &[1, 2]).expect("contiguous selection");
828        let ArrayValue::Complex32(selected) = selected else {
829            panic!("expected Complex32");
830        };
831        assert_eq!(selected.shape(), &[2, 2]);
832        assert_eq!(selected[[0, 0]], Complex32::new(2.0, 2.5));
833        assert_eq!(selected[[1, 1]], Complex32::new(5.0, 5.5));
834
835        let flags = ArrayValue::Bool(
836            ArrayD::from_shape_vec(
837                IxDyn(&[2, 4]).f(),
838                vec![false, true, false, true, true, false, true, false],
839            )
840            .unwrap(),
841        );
842        let selected = select_channels(flags, &[0, 3]).expect("non-contiguous selection");
843        let ArrayValue::Bool(selected) = selected else {
844            panic!("expected Bool");
845        };
846        assert_eq!(selected.shape(), &[2, 2]);
847        assert!(!selected[[0, 0]]);
848        assert!(!selected[[1, 1]]);
849
850        let weights = ArrayValue::Float32(
851            ArrayD::from_shape_vec(IxDyn(&[1, 3]).f(), vec![1.0, 2.0, 3.0]).unwrap(),
852        );
853        let selected = select_channels(weights, &[0, 1, 2]).expect("full-width selection");
854        let ArrayValue::Float32(selected) = selected else {
855            panic!("expected Float32");
856        };

858        assert_eq!(selected[[0, 2]], 3.0);
859    }
