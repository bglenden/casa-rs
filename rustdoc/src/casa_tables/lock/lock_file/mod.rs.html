2//! Low-level lock file protocol handler.
3//!
4//! Handles `fcntl`-based byte-range locking, request-list I/O, and sync-data
5//! I/O on a casacore `table.lock` file.
6
7// libc::F_RDLCK et al. are i16 on macOS but i32 on Linux; the `as i32` casts
8// are necessary on macOS but flagged as unnecessary on Linux.
9#![allow(clippy::unnecessary_cast)]
10//!
11//! # One descriptor per lock file per process
12//!
13//! `fcntl` record locks belong to the process, not to a descriptor: two
14//! descriptors on the same `table.lock` in one process never exclude each
15//! other, and closing *any* descriptor on the file releases every lock the
16//! process holds on it. casacore avoids both through its process-wide table
17//! cache, which gives each table one `LockFile` per process. This module
18//! keeps the same invariant with a process registry: every [`LockFile`]
19//! handle on one lock file (identified by device and inode) shares one
20//! descriptor, which is closed only when the last handle is dropped, and the
21//! registry arbitrates the locks the handles hold through it:
22//!
23//! - a write lock is held by at most one handle in the process, and a second
24//!   handle is refused it at once, without waiting, because the holder may
25//!   be the thread that would wait;
26//! - read locks never conflict within the process, as with casacore's shared
27//!   table;
28//! - the process's `fcntl` lock is the strongest lock any handle holds, so
29//!   releasing one handle never drops a lock another handle still holds.
30//!
31//! Nothing here is persisted: the registry is process memory, and the file
32//! protocol is casacore's own, so casacore and casa-rs processes exclude each
33//! other through the same byte ranges.
34//!
35//! # Waiting for another process
36//!
37//! A lock another process holds is waited for as casacore's
38//! `LockFile::acquire` waits: after one failed attempt the process's id is
39//! added to the request list at the start of the lock file, which a casacore
40//! holder using `AutoLocking` inspects and answers by releasing its lock, and
41//! it is removed again once the wait ends. The request list is casacore's
42//! layout: a big-endian `Int` count followed by 32 big-endian
43//! `(pid, hostid)` pairs, with host id 0 as casacore writes it. casacore's
44//! holder reads only the count. A wait is reported once when it starts, every
45//! ten seconds while it lasts, and once more when the lock is acquired.
46//!
47//! An indefinite wait blocks in the kernel (`F_SETLKW`), as casacore's
48//! `FileLocker` does, so that the kernel detects deadlocks between waiting
49//! processes, such as two processes upgrading their read locks: the request
50//! that would close the cycle fails with `EDEADLK`. The process mutex is not
51//! held while a thread is blocked; the granted lock is claimed afterwards
52//! under the mutex, and no handle claims the write lock while a thread of the
53//! process is blocked for a read lock, whose grant would replace it.
54//!
55//! # File systems without locking
56//!
57//! casacore counts a lock refused with `ENOLCK` ("locking over a network
58//! file system is not working", NFS without `lockd`) as acquired, so tables
59//! there are used without cross-process exclusion. macOS `smbfs` refuses
60//! `fcntl` locks with `ENOTSUP`/`EOPNOTSUPP` for the same condition, and
61//! casa-rs treats all three alike: the lock counts as acquired, read or
62//! write, and one warning per lock file names the path and errno. Handles in
63//! one process still exclude each other through the registry.
64//!
65//! # C++ reference
66//!
67//! `LockFile.cc`, `FileLocker.cc`, `PlainTable::tableCache`
68

84/// Size of a canonical `Int` in the lock file (4 bytes, big-endian).

107/// The process's descriptor on one `table.lock` and the locks held through it.
108struct SharedLockFd {
109    fd: RawFd,
110    writable: bool,
111    path: PathBuf,
112    state: Mutex<ProcessLockState>,
113}
114
115impl SharedLockFd {
116    fn state(&self) -> MutexGuard<'_, ProcessLockState> {
117        self.state.lock().unwrap_or_else(PoisonError::into_inner)
118    }
119
120    /// Set (or clear) an `fcntl` lock on `start..start + len` without
121    /// waiting; `false` when another process holds a conflicting lock.
122    fn set_lock(&self, lock_type: i32, start: i64, len: i64) -> io::Result<bool> {
123        self.granted(fcntl_lock(self.fd, lock_type, start, len))
124    }
125
126    /// Whether an `fcntl` outcome grants the lock. A file system without
127    /// lock support grants every lock, as casacore does for `ENOLCK`.
128    fn granted(&self, outcome: io::Result<FcntlOutcome>) -> io::Result<bool> {
129        Ok(match outcome? {
130            FcntlOutcome::Granted => true,
131            FcntlOutcome::Held | FcntlOutcome::Deadlock => false,
132            FcntlOutcome::Unsupported(errno) => {
133                report_unsupported_locking(&self.path, errno);
134                true
135            }
136        })
137    }
138}

140/// Warn, once per lock file in this process, that its file system does not
141/// support `fcntl` locking. Returns whether this call warned.
142fn report_unsupported_locking(path: &Path, errno: i32) -> bool {
143    static REPORTED: OnceLock<Mutex<HashSet<PathBuf>>> = OnceLock::new();
144    let first = REPORTED
145        .get_or_init(Default::default)
146        .lock()
147        .unwrap_or_else(PoisonError::into_inner)
148        .insert(path.to_path_buf());
149    if first {
150        tracing::warn!(
151            path = %path.display(),
152            errno,
153            error = %io::Error::from_raw_os_error(errno),
154            "the file system does not support table locking; the table is used without \
155             cross-process locks, as casacore does when locking over a network file system \
156             is not working (ENOLCK)"
157        );
158    }
159    first
160}

162/// Locks the handles of one process hold on one lock file.
163#[derive(Default)]
164struct ProcessLockState {
165    /// Handles holding a read lock.
166    readers: usize,
167    /// The handle holding the write lock.
168    writer: Option<u64>,
169    /// Threads of this process blocked in the kernel for a read lock. While
170    /// any is, no handle claims the write lock: the read lock the kernel
171    /// grants such a thread replaces the process's write lock.
172    blocked_readers: usize,
173    /// Length of the in-use read lock at byte 1 (2 under permanent locking).
174    in_use_len: i64,
175    /// Acquisitions that may still obtain the data lock, including kernel waits.
176    acquisitions: usize,
177    /// The reservation handle that delegates access to one task.
178    delegated_write: Option<u64>,
179}

189/// One registered lock file: its shared descriptor and the handles using it.
190struct RegistryEntry {
191    shared: Arc<SharedLockFd>,
192    handles: usize,
193    /// Descriptors opened while another process replaced the file between
194    /// `stat` and `open`; closing them early would drop the process's locks.
195    extra_fds: Vec<RawFd>,
196}

198/// Open lock files of this process. Descriptors are opened, registered and
199/// closed only while this mutex is held, so no descriptor on a registered
200/// file is ever closed while a lock is held through another one.
201fn registry() -> MutexGuard<'static, HashMap<LockFileKey, RegistryEntry>> {
202    static REGISTRY: OnceLock<Mutex<HashMap<LockFileKey, RegistryEntry>>> = OnceLock::new();
203    REGISTRY
204        .get_or_init(Default::default)
205        .lock()
206        .unwrap_or_else(PoisonError::into_inner)
207}

209/// Device and inode of `path`, or `None` when it does not exist.
210pub(super) fn path_key(path: &Path) -> io::Result<Option<(u64, u64)>> {
211    use std::os::unix::fs::MetadataExt;
212    match std::fs::metadata(path) {
213        Ok(metadata) => Ok(Some((metadata.dev(), metadata.ino()))),
214        Err(error) if error.kind() == io::ErrorKind::NotFound => Ok(None),
215        Err(error) => Err(error),
216    }
217}
218
219/// Device and inode of an open descriptor.
220fn fd_key(fd: RawFd) -> io::Result<LockFileKey> {
221    let mut stat = std::mem::MaybeUninit::<libc::stat>::uninit();
222    if unsafe { libc::fstat(fd, stat.as_mut_ptr()) } != 0 {
223        return Err(io::Error::last_os_error());
224    }
225    let stat = unsafe { stat.assume_init() };
226    Ok((stat.st_dev as u64, stat.st_ino as u64))
227}

229/// Outcome of a lock request.
230#[derive(Debug, Clone, Copy, PartialEq, Eq)]
231pub(crate) enum LockOutcome {
232    /// This handle holds the lock.
233    Acquired,
234    /// Another handle in this process holds the write lock. It is never
235    /// waited for, because it may belong to the waiting thread.
236    HeldInProcess,
237    /// Another process holds a conflicting lock, still after the attempts
238    /// the request allowed.
239    HeldByAnotherProcess,
240    /// Waiting would deadlock: the kernel refused the blocking request
241    /// (`EDEADLK`) because the holder waits, directly or through others, for
242    /// a lock this process holds. casacore's blocking wait gives up the same
243    /// way.
244    Deadlock,
245}
246
247impl LockOutcome {
248    /// Whether the lock was acquired.
249    pub(crate) fn is_acquired(self) -> bool {
250        self == Self::Acquired
251    }
252
253    /// Why a request for `lock_type` with `nattempts` attempts was refused,
254    /// worded for [`TableError::LockFailed`](crate::TableError::LockFailed).
255    pub(crate) fn refusal(self, lock_type: LockType, nattempts: u32) -> String {
256        match self {
257            Self::Acquired => format!("the {} lock was acquired", lock_name(lock_type)),
258            Self::HeldInProcess => "the write lock is held by another handle on this table in \
259                                    this process; a lock held in this process is never waited for"
260                .to_string(),
261            Self::HeldByAnotherProcess => {
262                let held = match lock_type {
263                    LockType::Read => "another process holds the write lock",
264                    LockType::Write => "another process holds a read or write lock",
265                };
266                match nattempts {
267                    0 | 1 => format!("{held} (one attempt, without waiting)"),
268                    attempts => format!("{held}, still after {attempts} attempts one second apart"),
269                }
270            }
271            Self::Deadlock => format!(
272                "waiting for the {} lock would deadlock with another process waiting for this \
273                 table: it holds a lock this request needs and waits for a lock this process holds",
274                lock_name(lock_type)
275            ),
276        }
277    }
278}

280/// The lock type as casacore's messages name it.
281fn lock_name(lock_type: LockType) -> &'static str {
282    match lock_type {
283        LockType::Read => "read",
284        LockType::Write => "write",
285    }
286}

288/// Low-level lock file protocol handler.
289///
290/// One handle on a `table.lock` file. Handles on the same file in one
291/// process share its descriptor through the process registry (see the
292/// module documentation), and provide methods for acquiring/releasing
293/// `fcntl` advisory locks, reading/writing the request list, and
294/// reading/writing sync data.
295///
296/// C++ equivalent: `LockFile`.
297#[allow(dead_code)] // fields used for AutoLocking (wave 3+)
298pub(crate) struct LockFile {
299    /// The process's descriptor on this lock file.
300    shared: Arc<SharedLockFd>,
301    /// Registry key of `shared`.
302    key: LockFileKey,
303    /// This handle's identity within the process.
304    id: u64,
305    /// Whether to add our PID to the request list when waiting.
306    add_to_list: bool,
307    /// Our process ID.
308    pid: i32,
309    /// Host ID (always 0, matching C++ which comments out `gethostid()`).
310    host_id: i32,
311    /// Inspection interval in seconds for auto-locking.
312    interval: f64,
313    /// Time of last inspection.
314    last_inspect: Instant,
315    /// Counter for inspection throttling (check every 25 calls).
316    inspect_count: u32,
317    /// Whether permanent locking is in use (affects in-use byte range).
318    perm_locking: bool,
319    /// Path to the lock file (for diagnostics).
320    path: PathBuf,
321    /// The lock this handle holds on the main byte range.
322    held: Option<LockType>,
323}

326    pub(crate) fn matches_table_path(&self, dir: &Path) -> bool {
327        path_key(&dir.join(LOCK_FILE_NAME)).is_ok_and(|key| key == Some(self.key))

342    /// Create or open a `table.lock` file at the given path.
343    ///
344    /// If `create` is true, the file is created (or truncated) with mode 0666
345    /// and the request list is initialized to all zeros. A lock file this
346    /// process already has open is shared instead, and only its request list
347    /// is reset.
348    ///
349    /// An in-use read lock is acquired on the file to signal that the table
350    /// is open.
351    pub fn create_or_open(
352        table_dir: &Path,
353        create: bool,
354        interval: f64,
355        perm_locking: bool,
356    ) -> io::Result<Self> {
357        static NEXT_HANDLE: AtomicU64 = AtomicU64::new(1);
358        let path = table_dir.join(LOCK_FILE_NAME);
359        let pid = unsafe { libc::getpid() };
360
361        let mut registry = registry();
362        let existing = path_key(&path)?;
363        let (shared, key) = match existing.filter(|key| registry.contains_key(key)) {
364            Some(key) => {
365                let entry = registry.get_mut(&key).expect("registered lock file");
366                entry.handles += 1;
367                if create && entry.shared.writable {
368                    write_at(entry.shared.fd, &[0u8; SIZEREQID], 0)?;
369                }
370                (Arc::clone(&entry.shared), key)
371            }
372            None => {
373                let (fd, writable) = open_lock_file(&path, create || existing.is_none())?;
374                let key = match fd_key(fd) {
375                    Ok(key) => key,
376                    Err(error) => {
377                        unsafe { libc::close(fd) };
378                        return Err(error);
379                    }
380                };
381                if let Some(entry) = registry.get_mut(&key) {
382                    entry.handles += 1;
383                    entry.extra_fds.push(fd);
384                    (Arc::clone(&entry.shared), key)
385                } else {
386                    let shared = Arc::new(SharedLockFd {
387                        fd,
388                        writable,
389                        path: path.clone(),
390                        state: Mutex::new(ProcessLockState::default()),
391                    });
392                    registry.insert(
393                        key,
394                        RegistryEntry {
395                            shared: Arc::clone(&shared),
396                            handles: 1,
397                            extra_fds: Vec::new(),
398                        },
399                    );
400                    (shared, key)
401                }
402            }
403        };
404
405        // Acquire the in-use read lock (byte 1, length 1 or 2 for permanent).
406        {
407            let mut state = shared.state();
408            let use_len = if perm_locking { 2 } else { 1 };
409            if state.in_use_len < use_len {
410                let _ = shared.set_lock(libc::F_RDLCK as i32, 1, use_len);
411                state.in_use_len = use_len;
412            }
413        }
414        drop(registry);
415
416        let lf = Self {
417            add_to_list: shared.writable,
418            shared,
419            key,
420            id: NEXT_HANDLE.fetch_add(1, Ordering::Relaxed),
421            pid,
422            host_id: 0,
423            interval,
424            last_inspect: Instant::now(),
425            inspect_count: 0,
426            perm_locking,
427            path,
428            held: None,
429        };
430
431        // Read any existing request list to clear stale state.
432        lf.read_request_count().ok();
433
434        Ok(lf)
435    }

437    /// Acquire a lock of the given type, as casacore's `LockFile::acquire`
438    /// does.
439    ///
440    /// One attempt is made without waiting. When another process holds a
441    /// conflicting lock and `nattempts` is not 1, this process's id is added
442    /// to the lock file's request list (when the file is writable), so that a
443    /// casacore holder using `AutoLocking` releases its lock at its next
444    /// inspection, and the request waits: with `nattempts == 0` blocked in
445    /// the kernel (`F_SETLKW`) until the lock is granted, as casacore's
446    /// `FileLocker` waits, and otherwise retrying once a second up to
447    /// `nattempts` attempts in all. The id is removed from the request list
448    /// when the wait ends. A wait is logged when it starts, every ten seconds
449    /// while it lasts and when it succeeds.
450    ///
451    /// A blocking wait lets the kernel detect a deadlock, such as two
452    /// processes that each hold a read lock and wait to upgrade it to a write
453    /// lock: the request that would close the cycle is refused with
454    /// [`LockOutcome::Deadlock`], and the other is granted once the refused
455    /// process releases its lock.
456    ///
457    /// A write lock another handle in this process holds is never waited for,
458    /// whatever `nattempts` is: that handle may belong to the waiting thread.
459    /// The request is answered [`LockOutcome::HeldInProcess`] at once.
460    ///
461    /// # Errors
462    ///
463    /// Returns an I/O error when `fcntl` fails for a reason other than a held
464    /// lock, a deadlock or missing lock support.
465    pub fn acquire(&mut self, lock_type: LockType, nattempts: u32) -> io::Result<LockOutcome> {
466        self.acquire_with_reservation(lock_type, nattempts, None)
467    }
468
469    pub(super) fn acquire_with_reservation(
470        &mut self,
471        lock_type: LockType,
472        nattempts: u32,
473        reservations: Option<&[DelegatedTableWrite]>,
474    ) -> io::Result<LockOutcome> {
475        let shared = Arc::clone(&self.shared);
476        let _acquisition = {
477            let mut state = shared.state();
478            if !delegated_write::permits(state.delegated_write, self, reservations) {
479                return Err(io::Error::new(
480                    io::ErrorKind::WouldBlock,
481                    "a task provider is using this table",
482                ));
483            }
484            state.acquisitions += 1;
485            Acquisition(&shared)
486        };
487        let first = self.try_acquire(lock_type)?;
488        let result = if first == LockOutcome::HeldByAnotherProcess && nattempts != 1 {
489            let added = self.add_to_list && self.shared.writable && self.add_request_id().is_ok();
490            let result = self.wait_for_other_process(lock_type, nattempts);
491            if added {
492                self.remove_request_id().ok();
493            }
494            result
495        } else {
496            Ok(first)
497        };
498        self.last_inspect = Instant::now();
499        self.inspect_count = 0;
500        result
501    }

503    /// Wait while another process holds a conflicting lock: blocked in the
504    /// kernel for `nattempts == 0`, otherwise up to `nattempts` attempts in
505    /// all, one second apart. The wait is logged.
506    fn wait_for_other_process(
507        &mut self,
508        lock_type: LockType,
509        nattempts: u32,
510    ) -> io::Result<LockOutcome> {
511        let table = self.path.parent().unwrap_or(&self.path).to_path_buf();
512        let lock = lock_name(lock_type);
513        let started = Instant::now();
514        // The fields are repeated in the message, which is all a plain-text
515        // log shows.
516        let outcome = if nattempts == 0 {
517            tracing::warn!(
518                table = %table.display(),
519                lock,
520                "table {} is locked by another process; waiting for the {lock} lock until it \
521                 is released, as casacore does",
522                table.display()
523            );
524            let dispatch = tracing::dispatcher::get_default(Clone::clone);
525            let (stop, stopped) = std::sync::mpsc::channel::<()>();
526            std::thread::scope(|scope| {
527                let table = table.as_path();
528                scope.spawn(move || {
529                    tracing::dispatcher::with_default(&dispatch, move || {
530                        while let Err(std::sync::mpsc::RecvTimeoutError::Timeout) =
531                            stopped.recv_timeout(WAIT_REPORT_INTERVAL)
532                        {
533                            let waited_s = started.elapsed().as_secs_f64();
534                            tracing::info!(
535                                table = %table.display(),
536                                lock,
537                                waited_s,
538                                "still waiting for the {lock} lock on table {} after {waited_s:.0} s",
539                                table.display()
540                            );
541                        }
542                    });
543                });
544                let outcome = self.wait_blocking(lock_type);
545                drop(stop);
546                outcome
547            })?
548        } else {
549            tracing::warn!(
550                table = %table.display(),
551                lock,
552                attempts = nattempts,
553                "table {} is locked by another process; retrying the {lock} lock once a \
554                 second, {nattempts} attempts in all",
555                table.display()
556            );
557            self.retry_once_a_second(lock_type, nattempts - 1)?
558        };
559        let waited_s = started.elapsed().as_secs_f64();
560        match outcome {
561            LockOutcome::Acquired => tracing::info!(
562                table = %table.display(),
563                lock,
564                waited_s,
565                "acquired the {lock} lock on table {} after waiting {waited_s:.1} s",
566                table.display()
567            ),
568            LockOutcome::Deadlock => tracing::warn!(
569                table = %table.display(),
570                lock,
571                waited_s,
572                "gave up waiting for the {lock} lock on table {}: waiting would deadlock with \
573                 another process that waits for a lock this process holds (EDEADLK)",
574                table.display()
575            ),
576            LockOutcome::HeldInProcess | LockOutcome::HeldByAnotherProcess => {}
577        }
578        Ok(outcome)
579    }

581    /// Up to `retries` more attempts, one second apart.
582    fn retry_once_a_second(
583        &mut self,
584        lock_type: LockType,
585        retries: u32,
586    ) -> io::Result<LockOutcome> {
587        for _ in 0..retries {
588            std::thread::sleep(Duration::from_secs(1));
589            let outcome = self.try_acquire(lock_type)?;
590            if outcome != LockOutcome::HeldByAnotherProcess {
591                return Ok(outcome);
592            }
593        }
594        Ok(LockOutcome::HeldByAnotherProcess)
595    }
596
597    /// Wait in the kernel (`F_SETLKW`) until the lock is granted, then claim
598    /// it for this handle.
599    ///
600    /// The process mutex is not held while the thread is blocked, so other
601    /// handles of this process keep working. The kernel grants the process's
602    /// lock; the claim is then re-validated under the mutex with a
603    /// non-blocking attempt, because meanwhile another handle may have taken
604    /// the write lock (the request is then [`LockOutcome::HeldInProcess`]) or
605    /// changed the process's lock, in which case the request waits again. A
606    /// thread blocked for a read lock is counted, and no handle claims the
607    /// write lock until it returns: the read lock the kernel grants it would
608    /// replace the process's write lock.
609    fn wait_blocking(&mut self, lock_type: LockType) -> io::Result<LockOutcome> {
610        loop {
611            let fcntl_type = match lock_type {
612                LockType::Write => libc::F_WRLCK,
613                LockType::Read => {
614                    let mut state = self.shared.state();
615                    if state.writer.is_some() || state.readers > 0 {
616                        // Another handle's lock covers reading now.
617                        drop(state);
618                        match self.try_acquire(lock_type)? {
619                            LockOutcome::HeldByAnotherProcess => continue,
620                            outcome => return Ok(outcome),
621                        }
622                    }
623                    state.blocked_readers += 1;
624                    libc::F_RDLCK
625                }
626            };
627            let blocked = fcntl_lock_wait(self.shared.fd, fcntl_type as i32, 0, 1);
628            if lock_type == LockType::Read {
629                self.shared.state().blocked_readers -= 1;
630            }
631            match blocked {
632                Ok(FcntlOutcome::Deadlock) => return Ok(LockOutcome::Deadlock),
633                Ok(_) => {}
634                Err(error) => {
635                    self.settle_unclaimed_grant();
636                    return Err(error);
637                }
638            }
639            match self.try_acquire(lock_type) {
640                Ok(LockOutcome::HeldByAnotherProcess) => std::thread::sleep(GRANT_RECLAIM_INTERVAL),
641                Ok(outcome) => return Ok(outcome),
642                Err(error) => {
643                    self.settle_unclaimed_grant();
644                    return Err(error);
645                }
646            }
647        }
648    }
649
650    /// Return the process's lock to the strongest lock its handles hold,
651    /// after the kernel granted a lock that this handle did not claim.
652    fn settle_unclaimed_grant(&self) {
653        let state = self.shared.state();
654        if state.writer.is_none() {
655            let held = if state.readers > 0 {
656                libc::F_RDLCK
657            } else {
658                libc::F_UNLCK
659            };
660            let _ = self.shared.set_lock(held as i32, 0, 1);
661        }
662    }

664    /// One non-blocking attempt, arbitrated against the other handles of this
665    /// process before the process's `fcntl` lock is changed.
666    fn try_acquire(&mut self, lock_type: LockType) -> io::Result<LockOutcome> {
667        let shared = &self.shared;
668        let mut state = shared.state();
669        match lock_type {
670            LockType::Write => {
671                match state.writer {
672                    Some(writer) if writer == self.id => return Ok(LockOutcome::Acquired),
673                    Some(_) => return Ok(LockOutcome::HeldInProcess),
674                    None => {}
675                }
676                // A thread of this process blocked for a read lock waits for
677                // another process's write lock, and its grant would replace
678                // this one.
679                if state.blocked_readers > 0 || !shared.set_lock(libc::F_WRLCK as i32, 0, 1)? {
680                    return Ok(LockOutcome::HeldByAnotherProcess);
681                }
682                if self.held == Some(LockType::Read) {
683                    state.readers -= 1;
684                }
685                state.writer = Some(self.id);
686                self.held = Some(LockType::Write);
687                Ok(LockOutcome::Acquired)
688            }
689            LockType::Read => match self.held {
690                // A held write lock covers reading and is kept, as casacore's
691                // `FileLocker::acquire(Read)` keeps it: dropping it would let
692                // other processes in before this handle's changes are flushed
693                // when it is released.
694                Some(LockType::Read | LockType::Write) => Ok(LockOutcome::Acquired),
695                None => {
696                    // The process's lock already covers reading when another
697                    // handle holds a read or write lock.
698                    if state.readers == 0
699                        && state.writer.is_none()
700                        && !shared.set_lock(libc::F_RDLCK as i32, 0, 1)?
701                    {
702                        return Ok(LockOutcome::HeldByAnotherProcess);
703                    }
704                    state.readers += 1;
705                    self.held = Some(LockType::Read);
706                    Ok(LockOutcome::Acquired)
707                }
708            },
709        }
710    }

712    /// Release the currently held lock.
713    ///
714    /// The process's `fcntl` lock drops to the strongest lock another handle
715    /// in this process still holds.
716    ///
717    /// Returns `true` if a lock was released, `false` if no lock was held.
718    pub fn release(&mut self) -> io::Result<bool> {
719        let shared = &self.shared;
720        let mut state = shared.state();
721        let Some(held) = self.held.take() else {
722            return Ok(false);
723        };
724        match held {
725            LockType::Read => {
726                state.readers -= 1;
727                if state.readers == 0 && state.writer.is_none() {
728                    shared.set_lock(libc::F_UNLCK as i32, 0, 1)?;
729                }
730            }
731            LockType::Write => {
732                state.writer = None;
733                let remaining = if state.readers > 0 {
734                    libc::F_RDLCK
735                } else {
736                    libc::F_UNLCK
737                };
738                shared.set_lock(remaining as i32, 0, 1)?;
739            }
740        }
741        Ok(true)
742    }

761    fn replace_sync_data(&self, payload: &[u8]) -> io::Result<()> {
762        if !self.shared.writable {
763            return Err(io::Error::new(
764                io::ErrorKind::PermissionDenied,
765                "table.lock is read-only",
766            ));
767        }

773                    "writing table.lock sync data failed: {error}; restoring the previous record also failed: {recovery}"
774                )));
775            }
776            return Err(error);
777        }
778        Ok(())
779    }

784    /// process is waiting. Throttled to check at most every 25 calls and
785    /// only after the inspection interval has elapsed.

787        if !always {
788            if self.interval > 0.0 && self.inspect_count < 25 {
789                self.inspect_count += 1;
790                return Ok(false);
791            }
792            self.inspect_count = 0;
793            if self.interval > 0.0 && self.last_inspect.elapsed().as_secs_f64() < self.interval {
794                return Ok(false);
795            }
796        }
797
798        let nr = self.read_request_count()?;
799        self.last_inspect = Instant::now();
800        Ok(nr > 0)
801    }

826    fn read_request_count(&self) -> io::Result<u32> {
827        let mut buf = [0u8; SIZEINT];
828        let n = read_at(self.shared.fd, &mut buf, 0)?;

835    /// Add our PID to the request list.
836    fn add_request_id(&self) -> io::Result<()> {
837        let fd = self.shared.fd;
838        let mut header = [0u8; SIZEREQID];
839        let n = read_at(fd, &mut header, 0)?;
840        if n < SIZEREQID {
841            // Pad with zeros if short.
842            header[n..].fill(0);
843        }
844
845        let count = i32::from_be_bytes(header[0..4].try_into().unwrap());
846        let inx = count.min(NRREQID as i32 - 1) as usize;
847
848        // Write our PID and host ID at the next slot.
849        let pid_offset = (1 + 2 * inx) * SIZEINT;
850        let host_offset = pid_offset + SIZEINT;
851        header[pid_offset..pid_offset + 4].copy_from_slice(&self.pid.to_be_bytes());
852        header[host_offset..host_offset + 4].copy_from_slice(&self.host_id.to_be_bytes());
853
854        // Increment count.
855        let new_count = (count + 1).min(NRREQID as i32);
856        header[0..4].copy_from_slice(&new_count.to_be_bytes());
857
858        write_at(fd, &header, 0)?;
859        unsafe { libc::fsync(fd) };
860        Ok(())
861    }

865        wait::remove_request_id_from_fd(self.shared.fd, self.pid)

869/// Open `path` read-write (creating and initializing it when `create`),
870/// falling back to read-only for an existing file.
871fn open_lock_file(path: &Path, create: bool) -> io::Result<(RawFd, bool)> {
872    let c_path = path_to_cstring(path)?;
873    if create {
874        // Create with world read/write access, matching C++.
875        let fd = unsafe {
876            libc::open(
877                c_path.as_ptr(),
878                libc::O_RDWR | libc::O_CREAT | libc::O_TRUNC,
879                0o666,
880            )
881        };
882        if fd < 0 {
883            return Err(io::Error::last_os_error());
884        }
885        // Initialize the request list header to zeros.
886        if let Err(error) = write_at(fd, &[0u8; SIZEREQID], 0) {
887            unsafe { libc::close(fd) };
888            return Err(error);
889        }
890        return Ok((fd, true));
891    }
892    // Open existing, try read-write first, fall back to read-only.
893    let fd = unsafe { libc::open(c_path.as_ptr(), libc::O_RDWR) };
894    if fd >= 0 {
895        return Ok((fd, true));
896    }
897    let fd = unsafe { libc::open(c_path.as_ptr(), libc::O_RDONLY) };
898    if fd < 0 {
899        return Err(io::Error::last_os_error());
900    }
901    // Read-only: can't add to request list.
902    Ok((fd, false))
903}

905/// Read `TableSyncData` from an existing `table.lock` file without acquiring
906/// or holding any locks.
907///
908/// This mirrors the information that C++ `PlainTable` consults during open
909/// before deciding which row count to trust for the subsequent table-data
910/// load. A lock file this process already has open is read through its
911/// shared descriptor; closing a second descriptor would release the
912/// process's locks on it.
913pub(crate) fn read_sync_data_from_table_dir(table_dir: &Path) -> io::Result<Option<SyncData>> {
914    let path = table_dir.join(LOCK_FILE_NAME);
915    let registry = registry();
916    let Some(key) = path_key(&path)? else {
917        return Ok(None);
918    };
919    if let Some(entry) = registry.get(&key) {
920        return read_sync_data_from_fd(entry.shared.fd);
921    }
922
923    // The process holds no lock on this file, so closing this descriptor
924    // releases nothing; the registry stays locked until it is closed.
925    let c_path = path_to_cstring(&path)?;
926    let fd = unsafe { libc::open(c_path.as_ptr(), libc::O_RDONLY) };
927    if fd < 0 {
928        return Err(io::Error::last_os_error());
929    }

931    struct FdGuard(i32);
932    impl Drop for FdGuard {
933        fn drop(&mut self) {
934            unsafe { libc::close(self.0) };
935        }
936    }
937    let _guard = FdGuard(fd);
938    read_sync_data_from_fd(fd)
939}

941/// Where the table at `table_dir` is held open, tested without opening the
942/// table or taking a lock: by a handle in this process that holds its
943/// `table.lock`, or by another process, if a write lock on the lock file's
944/// in-use byte would be refused. A handle in this process is reported
945/// first, as casacore's `TableUtil::canDeleteTable` refuses a table still
946/// open in this process before it asks `isMultiUsed`. Another process is
947/// tested for only when this process holds no handle on the lock file, so
948/// the descriptor opened for the test holds no lock that closing it would
949/// release.
950///
951/// `None` means that no holder was detected, not that none exists. A
952/// handle opened without locking (casacore's `NoLocking`, and casa-rs's
953/// [`Table::open`](crate::Table::open)) leaves no mark, nor does a table
954/// casacore opened without a lock file, so neither is seen; casacore's own
955/// check does not see them either. And the answer is a snapshot: a process
956/// can open the table as soon as it is given.
957///
958/// C++ equivalent: `TableUtil::canDeleteTable`'s "still open in this
959/// process" test and `Table::isMultiUsed` (`LockFile::isMultiUsed`), which
960/// `Table::deleteTable` and `Table::rename` consult before they act.
961///
962/// # Errors
963///
964/// The lock file exists but cannot be opened or tested.
965pub fn table_holder(table_dir: &Path) -> io::Result<Option<TableHolder>> {
966    let path = table_dir.join(LOCK_FILE_NAME);
967    let registry = registry();
968    let Some(key) = path_key(&path)? else {
969        return Ok(None);
970    };
971    if registry.contains_key(&key) {
972        return Ok(Some(TableHolder::ThisProcess));
973    }
974
975    // As above: the process holds no lock on this file, so closing this
976    // descriptor releases nothing.
977    let c_path = path_to_cstring(&path)?;
978    let fd = unsafe { libc::open(c_path.as_ptr(), libc::O_RDONLY) };
979    if fd < 0 {
980        return Err(io::Error::last_os_error());
981    }
982
983    struct FdGuard(i32);
984    impl Drop for FdGuard {
985        fn drop(&mut self) {
986            unsafe { libc::close(self.0) };
987        }
988    }
989    let _guard = FdGuard(fd);
990    let free = fcntl_can_lock(fd, libc::F_WRLCK as i32, 1, 1)?;
991    Ok((!free).then_some(TableHolder::AnotherProcess))
992}

994/// Read the sync data that follows the request list in an open lock file.
995fn read_sync_data_from_fd(fd: RawFd) -> io::Result<Option<SyncData>> {
996    let payload = read_sync_payload(fd)?;
997    if payload.is_empty() {
998        return Ok(None);
999    }

1003const SYNC_PAYLOAD_OFFSET: i64 = (SIZEREQID + SIZEINT) as i64;
1004
1005/// Keep the original bytes, including records written by casacore.
1006fn read_sync_payload(fd: RawFd) -> io::Result<Vec<u8>> {
1007    let mut len_buf = [0u8; SIZEINT];
1008    if read_up_to_at(fd, &mut len_buf, SIZEREQID as i64)? < SIZEINT {
1009        return Ok(Vec::new());
1010    }
1011    let info_len = u32::from_be_bytes(len_buf) as usize;
1012    let mut payload = vec![0u8; info_len];
1013    let n = read_up_to_at(fd, &mut payload, SYNC_PAYLOAD_OFFSET)?;
1014    if n < info_len {
1015        return Err(io::Error::new(
1016            io::ErrorKind::UnexpectedEof,
1017            format!("sync data truncated: expected {info_len}, got {n}"),
1018        ));
1019    }
1020    Ok(payload)
1021}

1023fn read_up_to_at(fd: RawFd, buf: &mut [u8], offset: i64) -> io::Result<usize> {
1024    let mut read = 0;
1025    while read < buf.len() {
1026        match read_at(fd, &mut buf[read..], offset + read as i64) {
1027            Ok(0) => break,
1028            Ok(count) => read += count,
1029            Err(error) if error.kind() == io::ErrorKind::Interrupted => {}
1030            Err(error) => return Err(error),
1031        }
1032    }
1033    Ok(read)
1034}
1035
1036fn write_sync_payload(fd: RawFd, payload: &[u8]) -> io::Result<()> {
1037    let info_len = u32::try_from(payload.len()).map_err(io::Error::other)?;
1038    write_all_at(fd, payload, SYNC_PAYLOAD_OFFSET)?;
1039    write_all_at(fd, &info_len.to_be_bytes(), SIZEREQID as i64)?;
1040    loop {
1041        // SAFETY: the caller retains the registered descriptor.
1042        if unsafe { libc::fsync(fd) } == 0 {
1043            return Ok(());
1044        }
1045        let error = io::Error::last_os_error();
1046        if error.kind() != io::ErrorKind::Interrupted {
1047            return Err(error);
1048        }
1049    }
1050}
1051
1052fn write_all_at(fd: RawFd, mut buf: &[u8], mut offset: i64) -> io::Result<()> {
1053    while !buf.is_empty() {
1054        match write_at(fd, buf, offset) {
1055            Ok(0) => return Err(io::Error::from(io::ErrorKind::WriteZero)),
1056            Ok(count) => {
1057                buf = &buf[count..];
1058                offset += count as i64;
1059            }
1060            Err(error) if error.kind() == io::ErrorKind::Interrupted => {}
1061            Err(error) => return Err(error),
1062        }
1063    }
1064    Ok(())
1065}

1067impl Drop for LockFile {
1068    fn drop(&mut self) {
1069        // Release any held lock (ignore errors in Drop).
1070        let _ = self.release();
1071        let mut registry = registry();
1072        let Some(entry) = registry.get_mut(&self.key) else {
1073            return;
1074        };
1075        entry.handles -= 1;
1076        if entry.handles == 0 {
1077            let entry = registry.remove(&self.key).expect("registered lock file");
1078            // Closing the last descriptor also drops the in-use lock.
1079            unsafe { libc::close(entry.shared.fd) };
1080            for fd in entry.extra_fds {
1081                unsafe { libc::close(fd) };
1082            }
1083        }
1084    }
1085}

1089/// Outcome of one `fcntl` lock request.
1090#[derive(Debug, Clone, Copy, PartialEq, Eq)]
1091enum FcntlOutcome {
1092    /// The lock was set (or cleared).
1093    Granted,
1094    /// Another process holds a conflicting lock.
1095    Held,
1096    /// A blocking request would deadlock (`EDEADLK`).
1097    Deadlock,
1098    /// The file system does not support `fcntl` locking; carries the errno.
1099    Unsupported(i32),
1100}
1101
1102/// Classify a failed `fcntl` lock request by its errno.
1103///
1104/// `EAGAIN`/`EACCES` mean another process holds the lock, and `EDEADLK` that
1105/// a blocking request would deadlock. `ENOLCK` is what NFS returns when
1106/// locking over the network is not working (casacore's `FileLocker` counts it
1107/// as acquired), and macOS `smbfs` returns `ENOTSUP`/`EOPNOTSUPP` for the
1108/// same condition. Any other errno is an error.
1109fn outcome_of_failed_lock(error: io::Error) -> io::Result<FcntlOutcome> {
1110    match error.raw_os_error() {
1111        Some(libc::EAGAIN | libc::EACCES) => Ok(FcntlOutcome::Held),
1112        Some(libc::EDEADLK) => Ok(FcntlOutcome::Deadlock),
1113        Some(errno)
1114            if errno == libc::ENOLCK || errno == libc::ENOTSUP || errno == libc::EOPNOTSUPP =>
1115        {
1116            Ok(FcntlOutcome::Unsupported(errno))
1117        }
1118        _ => Err(error),
1119    }
1120}

1122/// Perform a non-blocking `fcntl` lock operation (`F_SETLK`).
1123fn fcntl_lock(fd: RawFd, lock_type: i32, start: i64, len: i64) -> io::Result<FcntlOutcome> {
1124    let mut flock = libc::flock {
1125        l_type: lock_type as i16,
1126        l_whence: libc::SEEK_SET as i16,
1127        l_start: start,
1128        l_len: len,
1129        l_pid: 0,
1130    };
1131    if unsafe { libc::fcntl(fd, libc::F_SETLK, &mut flock) } == -1 {
1132        outcome_of_failed_lock(io::Error::last_os_error())
1133    } else {
1134        Ok(FcntlOutcome::Granted)
1135    }
1136}

1138/// Perform a blocking `fcntl` lock operation (`F_SETLKW`): wait until the
1139/// lock is granted, or until the kernel finds that waiting would deadlock.
1140/// An interrupted wait is resumed.
1141///
1142/// C++ equivalent: `FileLocker::acquire` with `nattempts == 0`.
1143fn fcntl_lock_wait(fd: RawFd, lock_type: i32, start: i64, len: i64) -> io::Result<FcntlOutcome> {
1144    loop {
1145        let mut flock = libc::flock {
1146            l_type: lock_type as i16,
1147            l_whence: libc::SEEK_SET as i16,
1148            l_start: start,
1149            l_len: len,
1150            l_pid: 0,
1151        };
1152        if unsafe { libc::fcntl(fd, libc::F_SETLKW, &mut flock) } != -1 {
1153            return Ok(FcntlOutcome::Granted);
1154        }
1155        let error = io::Error::last_os_error();
1156        if error.kind() != io::ErrorKind::Interrupted {
1157            return outcome_of_failed_lock(error);
1158        }
1159    }
1160}

1190/// Write to a file descriptor at a given offset using `pwrite`.
1191fn write_at(fd: RawFd, buf: &[u8], offset: i64) -> io::Result<usize> {
1192    #[cfg(test)]
1193    let buf = if offset >= SYNC_PAYLOAD_OFFSET {
1194        match SYNC_WRITE_FAULT.take() {
1195            Some(0) => return Err(io::Error::from_raw_os_error(libc::ENOSPC)),
1196            Some(count) => {
1197                SYNC_WRITE_FAULT.set(Some(0));
1198                &buf[..count.min(buf.len())]
1199            }
1200            None => buf,
1201        }
1202    } else {
1203        buf
1204    };
1205    let n = unsafe { libc::pwrite(fd, buf.as_ptr().cast(), buf.len(), offset) };
1206    if n < 0 {
1207        Err(io::Error::last_os_error())
1208    } else {
1209        Ok(n as usize)
1210    }
1211}
