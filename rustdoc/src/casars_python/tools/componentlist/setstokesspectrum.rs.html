14pub(super) struct Request {
15    name: String,
16    index: setspectrum::VectorArg,
17    vectors: [setspectrum::VectorArg; 5],
18    reference: Variant,
19    frame: String,
20}

22pub(super) struct Plan {

24    pub(super) reference: Option<(f64, FrequencyRef)>,
25    pub(super) refused: bool,

28fn frame(name: &str) -> ConvertResult<FrequencyRef> {
29    frequency_frame(name).ok_or_else(|| {
30        ConvertError::runtime(format!(
31            "{name} is not a frequency frame that is understood"
32        ))
33    })
34}

36/// Construct all spectral values before mutating the selected component.
37pub(super) fn prepare(request: &Request) -> ConvertResult<Option<Plan>> {
38    let name = request.name.to_ascii_uppercase();
39    if name.contains("TABU") {
40        return tabular(request).map(Some);
41    }
42    let spectral = name.contains("SPECTRAL");
43    if !spectral && name != "CONSTANT" && name != "POWER LOGARITHMIC POLYNOMIAL" {
44        return Ok(None);
45    }
46    let frame = frame(&request.frame)?;
47    let hz = if request.reference.is_empty_sequence()
48        || matches!(&request.reference, Variant::Numbers(values) if values.is_empty())
49    {
50        1.4e9
51    } else {
52        let value = quantity(&request.reference)?;
53        let hz = casacore_unit("Hz").map_err(ConvertError::runtime)?;
54        if !value.unit().conformant(&hz) || !value.get_si_value().is_finite() {
55            return Err(ConvertError::runtime("invalid frequency refused (#1053)"));
56        }
57        value.get_si_value()
58    };
59    if request.index.invalid || request.index.values.is_empty() {
60        return Err(ConvertError::runtime(
61            "invalid or empty spectrum index refused (#1053)",
62        ));
63    }
64    if name == "POWER LOGARITHMIC POLYNOMIAL" {
65        // CASA returns false after changing the reference frequency.
66        return Ok(Some(Plan {
67            kind: SpectralKind::Constant,
68            reference: None,
69            refused: true,
70        }));
71    }
72    let kind = if spectral {
73        let indices = &request.index.values;
74        if indices.iter().any(|value| !value.is_finite()) {
75            return Err(ConvertError::runtime(
76                "non-finite spectrum index refused (#1053)",
77            ));
78        }
79        let index = SpectralIndex::new(indices[0]);
80        let index = if let [i, q, u, v] = indices.as_slice() {
81            index.with_stokes_indices([*i, *q, *u, *v])
82        } else {
83            index
84        };
85        SpectralKind::SpectralIndex(index)
86    } else {
87        SpectralKind::Constant
88    };
89    Ok(Some(Plan {
90        kind,
91        reference: Some((hz, frame)),
92        refused: false,
93    }))
94}

96fn tabular(request: &Request) -> ConvertResult<Plan> {
97    let [freq, i, q, u, v] = &request.vectors;
98    if freq.values.len() < 2 {
99        return Err(ConvertError::runtime(
100            "There need to be at least 2 points in a tabular spectrum to interpolate in between",
101        ));
102    }
103    for (name, values) in [("I", i), ("Q", q), ("U", u), ("V", v)] {
104        if freq.values.len() != values.values.len() {
105            return Err(ConvertError::runtime(format!(
106                "lengths of tabular frequencies and {name} values have to be the same"
107            )));
108        }
109    }
110    let frame = frame(&request.frame)?;
111    if request.vectors.iter().any(|v| v.invalid) {
112        return Err(ConvertError::runtime(
113            "unrepresentable tabular value refused (#1053)",
114        ));
115    }
116    if request
117        .vectors
118        .iter()
119        .flat_map(|v| &v.values)
120        .any(|value| !value.is_finite())
121    {
122        return Err(ConvertError::runtime(
123            "non-finite tabular spectrum refused (#1053)",
124        ));
125    }
126    let values = (0..freq.values.len())
127        .map(|n| [i.values[n], q.values[n], u.values[n], v.values[n]])
128        .collect();
129    let table = TabularSpectrum::new(frame, freq.values.clone(), values)
130        .map_err(|e| ConvertError::runtime(e.to_string()))?;
131    Ok(Plan {
132        kind: SpectralKind::Tabular(table),
133        reference: None,
134        refused: false,
135    })
136}

138#[pymethods]
139impl PyComponentList {
140    /// Set one component's Stokes-dependent spectrum.
141    #[pyo3(signature = (which, r#type = None, index = None, tabularfreqs = None, tabulari = None, tabularq = None, tabularu = None, tabularv = None, reffreq = None, frame = None), text_signature = "($self, which, type='spectral index', index=[0.0], tabularfreqs=[100000000000.0], tabulari=[1.0], tabularq=[0.0], tabularu=[0.0], tabularv=[0.0], reffreq=[], frame='LSRK')")]
142    #[allow(clippy::too_many_arguments)]
143    fn setstokesspectrum(
144        &self,
145        py: Python<'_>,
146        which: &Bound<'_, PyAny>,
147        #[pyo3(from_py_with = supplied)] r#type: Option<Bound<'_, PyAny>>,
148        #[pyo3(from_py_with = supplied)] index: Option<Bound<'_, PyAny>>,
149        #[pyo3(from_py_with = supplied)] tabularfreqs: Option<Bound<'_, PyAny>>,
150        #[pyo3(from_py_with = supplied)] tabulari: Option<Bound<'_, PyAny>>,
151        #[pyo3(from_py_with = supplied)] tabularq: Option<Bound<'_, PyAny>>,
152        #[pyo3(from_py_with = supplied)] tabularu: Option<Bound<'_, PyAny>>,
153        #[pyo3(from_py_with = supplied)] tabularv: Option<Bound<'_, PyAny>>,
154        #[pyo3(from_py_with = supplied)] reffreq: Option<Bound<'_, PyAny>>,
155        #[pyo3(from_py_with = supplied)] frame: Option<Bound<'_, PyAny>>,
156    ) -> PyResult<Py<PyAny>> {
157        let (which, index_pending) = if !which.is_instance_of::<PyInt>()
158            && which.is_instance(&py.import("numpy")?.getattr("bool_")?)?
159        {
160            (-1, true)
161        } else {
162            match which.extract::<i64>() {
163                Ok(index) => (i64::from(index as i32), false),
164                Err(error) if which.is_none() || error.is_instance_of::<PyOverflowError>(py) => {
165                    (-1, true)
166                }
167                Err(_) => return Err(PyTypeError::new_err("argument _which must be an integer")),
168            }
169        };
170        let name = setspectrum::text(r#type, "spectral index", "type")?;
171        let (index, mut pending) = setspectrum::vector(index, 0.)?;
172        let mut vectors = Vec::new();
173        for (value, default) in [
174            (tabularfreqs, 1e11),
175            (tabulari, 1.),
176            (tabularq, 0.),
177            (tabularu, 0.),
178            (tabularv, 0.),
179        ] {
180            let (value, item_pending) = setspectrum::vector(value, default)?;
181            pending |= item_pending;
182            vectors.push(value);
183        }
184        let (reference, reference_pending) = setflux::argument(reffreq)?;
185        pending |= reference_pending || index_pending;
186        let frame = setspectrum::text(frame, "LSRK", "frame")?;
187        let request = Request {
188            name,
189            index,
190            vectors: vectors
191                .try_into()
192                .map_err(|_| PyTypeError::new_err("invalid tabular vectors"))?,
193            reference,
194            frame,
195        };
196        let result = self.call(py, move |state, lines| {
197            state
198                .setstokesspectrum(which, &request, pending, lines)
199                .map(CasaValue::Bool)
200        })?;
201        if pending {
202            return Err(PySystemError::new_err(
203                "<built-in function componentlist_setstokesspectrum> returned a result with an exception set",
204            ));
205        }
206        Ok(result)
207    }
