2//! The spline used by the legacy CASA Fourier machine to interpolate the
3//! degridded model from its image channels to native visibility channels.

5use num_complex::Complex32;

7use super::{LegacyPredictionMap, SpectralAxis, SpectralKernel, SpectralResampler};

9impl SpectralResampler {
10    /// Legacy spline prediction uses the cubic channel map and interpolates
11    /// the original image frequencies with casacore's second derivatives.
12    /// Use the row prediction methods for this resampler. The bounded
13    /// backend stencil cannot express a spline across the whole image axis.
14    #[must_use]
15    pub fn legacy_spline(output: SpectralAxis, matching: SpectralAxis) -> Self {
16        let mut resampler = Self::channel_local(output, SpectralKernel::Cubic);
17        resampler.legacy_prediction_map = Some(LegacyPredictionMap {
18            axis: matching,
19            selection: None,
20            spline: true,
21        });
22        resampler
23    }
24
25    pub(super) fn uses_legacy_spline(&self) -> bool {
26        self.legacy_prediction_map
27            .as_ref()
28            .is_some_and(|mapping| mapping.spline)
29    }
30}

32#[derive(Debug, Default)]
33pub(super) struct SplineScratch {
34    upper: Vec<f64>,
35    derivatives: Vec<Complex32>,
36}
37
38impl SplineScratch {
39    pub(super) fn with_capacity(planes: usize, npol: usize) -> Self {
40        Self {
41            upper: Vec::with_capacity(planes),
42            derivatives: Vec::with_capacity(planes * npol),
43        }
44    }
45
46    pub(super) const fn bytes(planes: usize, npol: usize) -> u64 {
47        (planes * (size_of::<f64>() + npol * size_of::<Complex32>())) as u64
48    }
49
50    pub(super) fn capacity_bytes(&self) -> u64 {
51        (self.upper.capacity() * size_of::<f64>()
52            + self.derivatives.capacity() * size_of::<Complex32>()) as u64
53    }
54
55    pub(super) fn prepare(
56        &mut self,
57        axis: &SpectralAxis,
58        npol: usize,
59        value: impl Fn(usize, usize) -> Complex32,
60    ) {
61        let planes = axis.channels as usize;
62        self.upper.clear();
63        self.upper.resize(planes, 0.0);
64        self.derivatives.clear();
65        self.derivatives.resize(planes * npol, Complex32::default());
66
67        let mut width = axis.centre_hz(1) - axis.centre_hz(0);
68        for index in 1..planes - 1 {
69            let previous_width = width;
70            width = axis.centre_hz(index as u32 + 1) - axis.centre_hz(index as u32);
71            let diagonal = 2.0 * (previous_width + width) - previous_width * self.upper[index - 1];
72            self.upper[index] = width / diagonal;
73            for pol in 0..npol {
74                let current = value(index, pol);
75                let right_slope = (value(index + 1, pol) - current) * (1.0 / width) as f32;
76                let left_slope = (current - value(index - 1, pol)) * (1.0 / previous_width) as f32;
77                let previous = self.derivatives[(index - 1) * npol + pol];
78                self.derivatives[index * npol + pol] = ((right_slope - left_slope) * 6.0
79                    - previous * previous_width as f32)
80                    * (1.0 / diagonal) as f32;
81            }
82        }
83        // InterpolateArray1D's back substitution stops before plane 1.
84        for index in (2..planes - 1).rev() {
85            for pol in 0..npol {
86                let right = self.derivatives[(index + 1) * npol + pol];
87                self.derivatives[index * npol + pol] -= right * self.upper[index] as f32;
88            }
89        }
90    }
91
92    pub(super) fn evaluate(
93        &self,
94        axis: &SpectralAxis,
95        frequency: f64,
96        value: impl Fn(usize, usize) -> Complex32,
97        flags: &[bool],
98        output: &mut [Complex32],
99    ) {
100        let npol = output.len();
101        let position = axis
102            .tabular_pixel(frequency)
103            .unwrap_or_else(|| (frequency - axis.first_hz) / axis.increment_hz);
104        let right = (position.ceil() as usize).clamp(1, axis.channels as usize - 1);
105        let left = right - 1;
106        let left_hz = axis.centre_hz(left as u32);
107        let right_hz = axis.centre_hz(right as u32);
108        let width = right_hz - left_hz;
109        let a = (right_hz - frequency) / width;
110        let b = 1.0 - a;
111        let scale = width * width / 6.0;
112        let second_left = (scale * (a * a * a - a)) as f32;
113        let second_right = (scale * (b * b * b - b)) as f32;
114        for (pol, cell) in output.iter_mut().enumerate() {
115            if !flags[pol] {
116                *cell = value(left, pol) * a as f32
117                    + value(right, pol) * b as f32
118                    + self.derivatives[left * npol + pol] * second_left
119                    + self.derivatives[right * npol + pol] * second_right;
120            }
121        }
122    }
123}
