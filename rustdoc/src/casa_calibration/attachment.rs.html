13/// An open MeasurementSet and the owned write lock held across a calibration mutation.
14pub struct OpenedMs {
15    path: PathBuf,
16    /// MeasurementSet read while holding the MAIN-table lock.
17    pub measurement_set: MeasurementSet,
18    /// Exclusive MAIN-table lock passed to storage-changing operations that reopen the table.
19    pub lock: TableWriteLock,
20}
21
22impl OpenedMs {
23    /// Canonicalizes the table path, acquires its MAIN write lock, then reads the MeasurementSet.
24    /// Lock and open failures propagate.
25    pub fn open(path: &Path) -> MsResult<Self> {
26        let path = table_cache_name(path)?;
27        let lock = TableWriteLock::acquire_to_reread(&path, 0)?;
28        Self::read_holding(path, lock)
29    }
30
31    fn read_holding(path: PathBuf, lock: TableWriteLock) -> MsResult<Self> {
32        let measurement_set = MeasurementSet::open(&path)?;
33        Ok(Self {
34            path,
35            measurement_set,
36            lock,
37        })
38    }
39
40    /// Initializes the requested scratch visibility column and retains the resulting lock.
41    pub fn initialize(mut self, column: VisibilityDataColumn) -> MsResult<Self> {
42        self.lock = initialize_scratch_column(&mut self.measurement_set, column, self.lock)?;
43        Ok(self)
44    }
45
46    /// Initializes the requested corrected and model columns under the retained write lock,
47    /// optionally using compressed storage.
48    pub fn initialize_calibrater(
49        mut self,
50        compress: bool,
51        add_corrected: bool,
52        add_model: bool,
53    ) -> MsResult<Self> {
54        self.lock = initialize_calibrater_columns(
55            &mut self.measurement_set,
56            compress,
57            add_corrected,
58            add_model,
59            self.lock,
60        )?;
61        Ok(self)
62    }
63
64    /// Resolves CASA calibration selectors against the locked MeasurementSet's current metadata.
65    pub fn resolve(
66        &self,
67        selection: &MsSelection,
68    ) -> Result<ResolvedMsSelection, MsSelectionError> {
69        resolve_casa_calibration_selection(&self.measurement_set, selection)
70    }
71
72    /// Drops and rereads the MeasurementSet while retaining the exclusive lock.
73    pub fn reload(self) -> MsResult<Self> {
74        drop(self.measurement_set);
75        Self::read_holding(self.path, self.lock)
76    }
77
78    /// Closes the MeasurementSet and explicitly releases the lock, returning any release error.
79    pub fn release(self) -> MsResult<()> {
80        drop(self.measurement_set);
81        self.lock.release()?;
82        Ok(())
83    }
84}

113/// An attachment or selection failed before prediction could write data.
114#[derive(Debug, thiserror::Error)]
115pub enum AttachmentError {
116    /// No MeasurementSet is attached to this tool.
117    #[error("no MeasurementSet is attached")]
118    Unattached,
119    /// The requested selection matched no MAIN rows.
120    #[error("Selection is empty: you may want to review this MSs selection")]
121    EmptySelection,
122    /// An explicit selection was rejected and needs a successful replacement.
123    #[error("the current visibility selection was rejected")]
124    RejectedSelection,
125    /// A direct open has no selected spectral windows for a UV cut.
126    #[error("uvrange requires a configured visibility selection")]
127    UvRangeRequiresSelection,
128    /// The named attachment was opened read-only.
129    #[error("Please make sure MS is writable when using Imager::ft")]
130    ReadOnly,
131    /// The active Multi-MS iterator includes an earlier read-only dataset.
132    #[error("an active MeasurementSet is read-only")]
133    ReadOnlyInput,
134    /// The supplied UV interval is outside CASA's accepted domain.
135    #[error("Invalid uvmin and uvmax: {0}, {1}")]
136    InvalidUvRange(f64, f64),
137    /// A MeasurementSet or its write lock could not be opened.
138    #[error(transparent)]
139    MeasurementSet(#[from] casa_ms::MsError),
140    /// The stored selection could not be resolved against the attached dataset.
141    #[error(transparent)]
142    Selection(#[from] MsSelectionError),
143    /// The supplied image or component model could not be prepared.
144    #[error(transparent)]
145    PredictionPreparation(casa_ms::MsError),
146}
147
148impl From<TableError> for AttachmentError {
149    fn from(error: TableError) -> Self {
150        Self::MeasurementSet(error.into())
151    }
152}

161/// Canonical path and selector inputs retained between calibration calls.
162///
163/// Physical row ids and channel mappings are resolved on the reopened
164/// MeasurementSet under its MAIN lock whenever a caller mutates the dataset.
165#[derive(Clone, Debug, PartialEq)]
166pub struct CalibrationAttachment {
167    path: PathBuf,
168    mode: AttachmentMode,
169    options: AttachmentOptions,
170    selection: MsSelection,
171    imager_channels: Option<ImagerChannelSelection>,
172    state: SelectionState,
173    uv_ranges: Vec<[f64; 2]>,
174}
175
176impl CalibrationAttachment {
177    /// Attach the specified MeasurementSet, optionally initializing its scratch model.
178    ///
179    /// # Errors
180    ///
181    /// Returns an error if the table, its lock, or the selection cannot be read.
182    pub fn attach(
183        path: &Path,
184        mode: AttachmentMode,
185        options: AttachmentOptions,
186        selection: MsSelection,
187    ) -> Result<Self, AttachmentError> {
188        Self::attach_with_channels(path, mode, options, selection, None)
189    }
190
191    fn attach_with_channels(
192        path: &Path,
193        mode: AttachmentMode,
194        options: AttachmentOptions,
195        selection: MsSelection,
196        imager_channels: Option<ImagerChannelSelection>,
197    ) -> Result<Self, AttachmentError> {
198        let path = table_cache_name(path)?;
199        let attachment = Self {
200            path,
201            mode,
202            options,
203            selection,
204            imager_channels,
205            state: match mode {
206                AttachmentMode::Open => SelectionState::Initial,
207                AttachmentMode::Selection => SelectionState::Selected,
208            },
209            uv_ranges: Vec::new(),
210        };
211        if options.write_access && options.use_scratch {
212            let opened = OpenedMs::open(&attachment.path)?;
213            let opened = opened.initialize(VisibilityDataColumn::ModelData)?;
214            if mode == AttachmentMode::Selection {
215                attachment.require_selected_rows(&opened.measurement_set)?;
216            }
217            opened.release()?;
218        } else {
219            let ms = MeasurementSet::open(&attachment.path)?;
220            if mode == AttachmentMode::Selection {
221                if attachment.imager_channels.is_some()
222                    && options.use_scratch
223                    && !has_column(&ms, VisibilityDataColumn::ModelData)
224                {
225                    return Err(casa_ms::MsError::InvalidInput(format!(
226                        "Invalid Table operation: Table::addColumn; table {} is not writable",
227                        attachment.path.display(),
228                    ))
229                    .into());
230                }
231                attachment.require_selected_rows(&ms)?;
232            }
233        }
234        Ok(attachment)
235    }
236
237    fn require_selected_rows(&self, ms: &MeasurementSet) -> Result<(), AttachmentError> {
238        if self.resolve_holding(ms, &[])?.selected_rows.is_empty() {
239            return Err(AttachmentError::EmptySelection);
240        }
241        Ok(())
242    }
243
244    /// Canonical path used for both table access and locking.
245    pub fn path(&self) -> &Path {
246        &self.path
247    }
248
249    /// Mode that established the association between this tool and the MS.
250    pub fn mode(&self) -> AttachmentMode {
251        self.mode
252    }
253
254    /// Retained write and scratch options.
255    pub fn options(&self) -> AttachmentOptions {
256        self.options
257    }
258
259    /// Resolve the current selection for a read.
260    ///
261    /// # Errors
262    ///
263    /// Returns an error if the attached dataset or selector cannot be read.
264    pub fn resolve(&self) -> Result<ResolvedMsSelection, AttachmentError> {
265        let ms = MeasurementSet::open(&self.path)?;
266        self.resolve_holding(&ms, &self.uv_ranges)
267    }
268
269    /// Replace the selection on this attachment and reset prior UV narrowing.
270    ///
271    /// An empty or invalid selector disables mutations until the next
272    /// successful selection, matching CASA's direct-open state.
273    ///
274    /// # Errors
275    ///
276    /// Returns an error if the table or selection cannot be read.
277    pub fn replace_selection(&mut self, selection: MsSelection) -> Result<bool, AttachmentError> {
278        self.replace_selection_with_channels(selection, None)
279    }
280
281    fn replace_selection_with_channels(
282        &mut self,
283        selection: MsSelection,
284        channels: Option<ImagerChannelSelection>,
285    ) -> Result<bool, AttachmentError> {
286        match self.select_with_channels(selection, channels) {
287            Ok(()) => Ok(true),
288            Err(
289                AttachmentError::EmptySelection
290                | AttachmentError::Selection(MsSelectionError::Domain(
291                    casa_ms::MsError::InvalidInput(_) | casa_ms::MsError::VersionError(_),
292                )),
293            ) => Ok(false),
294            Err(error) => Err(error),
295        }
296    }
297
298    /// Validates and replaces the attached input's selection. An empty or invalid selection is
299    /// reported and leaves the attachment in a rejected selection state.
300    pub fn select(&mut self, selection: MsSelection) -> Result<(), AttachmentError> {
301        self.select_with_channels(selection, None)
302    }
303
304    fn select_with_channels(
305        &mut self,
306        selection: MsSelection,
307        channels: Option<ImagerChannelSelection>,
308    ) -> Result<(), AttachmentError> {
309        let ms = MeasurementSet::open(&self.path)?;
310        self.state = SelectionState::Rejected;
311        self.uv_ranges.clear();
312        let resolved = imager_selection(&ms, &selection, channels.as_ref())?;
313        if resolve_casa_calibration_selection(&ms, &resolved)?
314            .selected_rows
315            .is_empty()
316        {
317            return Err(AttachmentError::EmptySelection);
318        }
319        self.selection = selection;
320        self.imager_channels = channels;
321        self.state = SelectionState::Selected;
322        Ok(())
323    }
324
325    /// Borrows the currently stored selector. Selection state separately tracks whether it was
326    /// rejected.
327    pub fn selection(&self) -> &MsSelection {
328        &self.selection
329    }
330
331    /// Clears explicit row and channel selections and wavelength bounds.
332    pub fn reset_selection(&mut self) {
333        self.selection = MsSelection::default();
334        self.imager_channels = None;
335        self.state = SelectionState::Initial;
336        self.uv_ranges.clear();
337    }
338
339    /// Narrow the current selection using strict wavelength bounds.
340    ///
341    /// CASA leaves the current selection in place when a valid interval
342    /// matches no rows. A nonpositive upper bound uses its 1e10 default.
343    ///
344    /// # Errors
345    ///
346    /// Returns an error for invalid bounds or a failed selection.
347    pub fn narrow_uvrange(&mut self, minimum: f64, maximum: f64) -> Result<(), AttachmentError> {
348        let maximum = if maximum <= 0.0 { 1e10 } else { maximum };
349        if !(minimum >= 0.0 && maximum > minimum && maximum.is_finite()) {
350            return Err(AttachmentError::InvalidUvRange(minimum, maximum));
351        }
352        match self.state {
353            SelectionState::Initial => return Err(AttachmentError::UvRangeRequiresSelection),
354            SelectionState::Rejected => return Err(AttachmentError::RejectedSelection),
355            SelectionState::Selected => {}
356        }
357        let ms = MeasurementSet::open(&self.path)?;
358        let mut ranges = self.uv_ranges.clone();
359        ranges.push([minimum, maximum]);
360        if !self.resolve_holding(&ms, &ranges)?.selected_rows.is_empty() {
361            self.uv_ranges = ranges;
362        }
363        Ok(())
364    }
365
366    /// Reopen under MAIN and supply the selection and its matching write lock.
367    ///
368    /// Publish through operations that accept the held lock.
369    ///
370    /// # Errors
371    ///
372    /// Returns an error for read-only inputs, failed locking or selection, a
373    /// rejected operation, or a failed lock release.
374    pub fn with_write<T>(
375        &self,
376        operation: impl FnOnce(
377            &mut MeasurementSet,
378            &ResolvedMsSelection,
379            &mut TableWriteLock,
380        ) -> Result<T, AttachmentError>,
381    ) -> Result<T, AttachmentError> {
382        if !self.options.write_access {
383            return Err(AttachmentError::ReadOnly);
384        }
385        if self.state == SelectionState::Rejected {
386            return Err(AttachmentError::RejectedSelection);
387        }
388        let mut opened = OpenedMs::open(&self.path)?;
389        let selected = self.resolve_holding(&opened.measurement_set, &self.uv_ranges);
390        let result = selected.and_then(|selected| {
391            operation(&mut opened.measurement_set, &selected, &mut opened.lock)
392        });
393        let release = opened.release();
394        result.and_then(|value| {
395            release?;
396            Ok(value)
397        })
398    }
399
400    fn resolve_holding(
401        &self,
402        ms: &MeasurementSet,
403        uv_ranges: &[[f64; 2]],
404    ) -> Result<ResolvedMsSelection, AttachmentError> {
405        if self.state == SelectionState::Rejected {
406            return Err(AttachmentError::RejectedSelection);
407        }
408        let selected = imager_selection(ms, &self.selection, self.imager_channels.as_ref())?;
409        if uv_ranges.is_empty() {
410            return Ok(resolve_casa_calibration_selection(ms, &selected)?);
411        }
412        let mut selection = if selected.selectdata {
413            selected.into_owned()
414        } else {
415            MsSelection::default()
416        };
417        let factor = mean_inverse_wavelength(ms, selection.spw.as_deref())?.powi(2);
418        let distance = format!("(SQUARE(UVW[1]) + SQUARE(UVW[2]))*{factor:.5e}");
419        let mut predicates = selection
420            .msselect
421            .take()
422            .into_iter()
423            .filter(|text| !text.trim().is_empty())
424            .map(|text| format!("({text})"))
425            .collect::<Vec<_>>();
426        for &[minimum, maximum] in uv_ranges {
427            predicates.push(format!(
428                "({distance}>{:.5e} AND {distance}<{:.5e})",
429                minimum * minimum,
430                maximum * maximum,
431            ));
432        }
433        selection.msselect = Some(predicates.join(" AND "));
434        Ok(resolve_casa_calibration_selection(ms, &selection)?)
435    }
436}

438/// Channel counts, offsets and strides for the legacy imager selector.
439#[derive(Clone, Debug, PartialEq)]
440pub struct ImagerChannelSelection {
441    /// Number of channels for each selected spectral window, or one for all windows.
442    pub count: Vec<i64>,
443    /// First channel for each selected window, or one for all windows.
444    pub start: Vec<i64>,
445    /// Stride for each selected window, or one for all windows.
446    pub step: Vec<i64>,
447}
448
449fn imager_selection<'a>(
450    ms: &MeasurementSet,
451    selection: &'a MsSelection,
452    channels: Option<&ImagerChannelSelection>,
453) -> Result<std::borrow::Cow<'a, MsSelection>, AttachmentError> {
454    let Some(channels) =
455        channels.filter(|channels| !matches!(channels.count.as_slice(), [n] if *n < 0))
456    else {
457        return Ok(std::borrow::Cow::Borrowed(selection));
458    };
459    let windows = ms.resolve_spw_selector(selection.spw.as_deref().unwrap_or("*"))?;
460    let mut ids = windows.spw_ids().to_vec();
461    ids.sort_unstable();
462    ids.dedup();
463    // The legacy selector keeps its SPW and row selection when the iterator
464    // cannot apply a malformed explicit channel vector.
465    if [&channels.count, &channels.start, &channels.step]
466        .iter()
467        .any(|values| values.len() != 1 && values.len() != ids.len())
468    {
469        return Ok(std::borrow::Cow::Borrowed(selection));
470    }
471    let argument = |values: &[i64], index| {
472        if values.len() == 1 {
473            values[0]
474        } else {
475            values[index]
476        }
477    };
478    let spectral = ms.spectral_window()?;
479    let mut ranges = Vec::with_capacity(ids.len());
480    for (index, &id) in ids.iter().enumerate() {
481        let count = argument(&channels.count, index);
482        let start = argument(&channels.start, index);
483        let step = argument(&channels.step, index).max(1);
484        let total = i64::from(spectral.num_chan(id)?);
485        if !(0..total).contains(&start) {
486            return Ok(std::borrow::Cow::Borrowed(selection));
487        }
488        // Keep a valid first channel when the stride runs past the end of
489        // the window. This boundary can abort the legacy iterator.
490        let available = ((total - start) / step).max(1);
491        let count = if count <= 0 {
492            available
493        } else {
494            count.min(available)
495        };
496        let end = start + (count - 1) * step;
497        ranges.push(format!("{id}:{start}~{end}^{step}"));
498    }
499    let mut selected = selection.clone();
500    selected.spw = Some(ranges.join(","));
501    Ok(std::borrow::Cow::Owned(selected))
502}

504/// MeasurementSet associations for one imager object.
505#[derive(Debug, Default)]
506pub struct ImagerAttachments {
507    inputs: Vec<CalibrationAttachment>,
508    current_only: bool,
509}
510
511impl ImagerAttachments {
512    /// Replace the existing associations with an explicit MeasurementSet.
513    ///
514    /// # Errors
515    ///
516    /// Returns an error if the new table or required scratch preparation fails.
517    pub fn open(&mut self, path: &Path, options: AttachmentOptions) -> Result<(), AttachmentError> {
518        let input = CalibrationAttachment::attach(
519            path,
520            AttachmentMode::Open,
521            options,
522            MsSelection::default(),
523        )?;
524        self.inputs = vec![input];
525        self.current_only = false;
526        Ok(())
527    }
528
529    /// Replace the selection on an explicitly opened MS, or add another input.
530    ///
531    /// A direct open keeps the original dataset and its access options.
532    /// Selection-only mode retains its path, options and selector per input.
533    ///
534    /// # Errors
535    ///
536    /// Returns an error if an attachment or selector cannot be read.
537    pub fn select(
538        &mut self,
539        path: Option<&Path>,
540        options: AttachmentOptions,
541        selection: MsSelection,
542    ) -> Result<bool, AttachmentError> {
543        self.select_channels(path, options, selection, None)
544    }
545
546    /// Select rows and explicit imager channel vectors, retaining the request for each input.
547    ///
548    /// # Errors
549    /// The input, channel vectors or selector cannot be applied.
550    pub fn select_channels(
551        &mut self,
552        path: Option<&Path>,
553        options: AttachmentOptions,
554        selection: MsSelection,
555        channels: Option<ImagerChannelSelection>,
556    ) -> Result<bool, AttachmentError> {
557        if let Some(input) = self.inputs.first_mut()
558            && input.mode == AttachmentMode::Open
559        {
560            return input.replace_selection_with_channels(selection, channels);
561        }
562        let path = path.ok_or(AttachmentError::Unattached)?;
563        let input = CalibrationAttachment::attach_with_channels(
564            path,
565            AttachmentMode::Selection,
566            options,
567            selection,
568            channels,
569        )?;
570        if self.current_only
571            && let Some(previous) = self.inputs.last_mut()
572        {
573            previous.uv_ranges.clear();
574        }
575        self.inputs.push(input);
576        self.current_only = false;
577        Ok(true)
578    }
579
580    /// The attached inputs in the order supplied by the caller.
581    pub fn inputs(&self) -> &[CalibrationAttachment] {
582        &self.inputs
583    }
584
585    /// Inputs currently used by the visibility iterator.
586    pub fn active_inputs(&self) -> &[CalibrationAttachment] {
587        if self.current_only {
588            &self.inputs[self.inputs.len().saturating_sub(1)..]
589        } else {
590            &self.inputs
591        }
592    }
593
594    /// Validate the active group before prediction writes to any input.
595    ///
596    /// # Errors
597    ///
598    /// Returns an error when inputs are absent, read-only, or rejected.
599    pub fn writable_inputs(&self) -> Result<&[CalibrationAttachment], AttachmentError> {
600        let active = self.active_inputs();
601        let current = active.last().ok_or(AttachmentError::Unattached)?;
602        if !current.options.write_access {
603            return Err(AttachmentError::ReadOnly);
604        }
605        if active.iter().any(|input| !input.options.write_access) {
606            return Err(AttachmentError::ReadOnlyInput);
607        }
608        if active
609            .iter()
610            .any(|input| input.state == SelectionState::Rejected)
611        {
612            return Err(AttachmentError::RejectedSelection);
613        }
614        Ok(active)
615    }
616
617    /// Replaces the latest input's setjy selection and updates which attachments are active. Fails
618    /// if the imager has no attached input or validation fails.
619    pub fn retain_setjy_selection(
620        &mut self,
621        selection: MsSelection,
622    ) -> Result<(), AttachmentError> {
623        let input = self.inputs.last_mut().ok_or(AttachmentError::Unattached)?;
624        input.replace_selection(selection)?;
625        self.current_only = input.mode == AttachmentMode::Selection;
626        Ok(())
627    }
628
629    /// Narrow the currently selected input.
630    ///
631    /// # Errors
632    ///
633    /// Returns an error for an unattached imager or a failed UV-range selection.
634    pub fn narrow_uvrange(&mut self, minimum: f64, maximum: f64) -> Result<(), AttachmentError> {
635        let current = self.inputs.last_mut().ok_or(AttachmentError::Unattached)?;
636        current.narrow_uvrange(minimum, maximum)?;
637        // The legacy iterator is rebuilt over the current input, including
638        // when the UV predicate matched no rows and retained its selection.
639        self.current_only = current.mode == AttachmentMode::Selection;
640        Ok(())
641    }
642
643    /// Release the stored table associations and per-input selectors.
644    pub fn close(&mut self) {
645        self.inputs.clear();
646        self.current_only = false;
647    }
648}

650fn mean_inverse_wavelength(ms: &MeasurementSet, selector: Option<&str>) -> MsResult<f64> {
651    const LIGHT_SPEED_M_S: f64 = 299_792_458.0;
652    let spectral = ms.spectral_window()?;
653    let mut windows = match selector
654        .map(|selector| ms.resolve_spw_selector(selector))
655        .transpose()?
656    {
657        Some(selection) if !selection.selects_all() => selection.spw_ids().to_vec(),
658        _ => (0..spectral.row_count()).collect(),
659    };
660    // CASA's UV cut uses the whole selected spectral windows in table order.
661    windows.sort_unstable();
662    windows.dedup();
663    let (mut sum, mut count) = (0.0, 0);
664    for window in windows {
665        let frequencies = spectral.chan_freq(window)?;
666        sum += frequencies.iter().sum::<f64>();
667        count += frequencies.len();
668    }
669    if count == 0 {
670        return Err(casa_ms::MsError::InvalidInput(
671            "selected spectral windows have no channels".into(),
672        ));
673    }
674    Ok(sum / (count as f64 * LIGHT_SPEED_M_S))
675}
