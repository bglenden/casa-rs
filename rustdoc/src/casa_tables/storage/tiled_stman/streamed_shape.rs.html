5/// Element type for a bounded, heterogeneous `TiledShapeStMan` column.
6///
7/// This is intentionally a storage-side construction type. Table users keep
8/// using row, column, and cell accessors; the type exists so higher-level
9/// writers can build a canonical casacore data-manager payload without first
10/// materializing every array cell.
11#[doc(hidden)]
12#[derive(Debug, Clone, Copy, Eq, PartialEq)]
13pub enum StreamedTiledShapeValueType {
14    Bool,
15    Int32,
16    Float32,
17    Float64,
18    Complex32,
19}
20
21impl StreamedTiledShapeValueType {
22    pub(super) fn casacore_type(self) -> CasacoreDataType {
23        match self {
24            Self::Bool => CasacoreDataType::TpBool,
25            Self::Int32 => CasacoreDataType::TpInt,
26            Self::Float32 => CasacoreDataType::TpFloat,
27            Self::Float64 => CasacoreDataType::TpDouble,
28            Self::Complex32 => CasacoreDataType::TpComplex,
29        }
30    }
31
32    pub(super) fn elem_size(self) -> usize {
33        tile_element_size(self.casacore_type())
34    }
35}

128/// Direct, bounded writer for one heterogeneous `TiledShapeStMan` column.
129///
130/// The shape histogram is immutable. Every distinct shape maps to exactly one
131/// extensible hypercube, matching casacore's `TiledShapeStMan` convention.
132/// Each row is written once to its final tile offsets; no full-payload spool,
133/// snapshot, or second assembly pass is used.
134#[doc(hidden)]
135#[derive(Debug)]
136pub struct StreamingTiledShapeWriter {
137    pub(super) value_type: StreamedTiledShapeValueType,
138    pub(super) big_endian: bool,
139    pub(super) cell_rank: Option<usize>,
140    pub(super) default_tile_shape: Vec<usize>,
141    pub(super) row_count: usize,
142    pub(super) rows_written: usize,
143    pub(super) shape_to_cube: HashMap<Vec<usize>, usize>,
144    pub(super) cubes: Vec<DirectShapeCubeWriter>,
145    /// The tile-row scratch every cube writes its rows through, sized for
146    /// the largest cube's tile row.
147    pub(super) tile_row: Vec<u8>,
148    pub(super) row_map: Vec<u32>,
149    pub(super) cube_map: Vec<u32>,
150    pub(super) pos_map: Vec<u32>,
151    pub(super) memory_budget_bytes: usize,
152    pub(super) working_bytes: usize,
153    pub(super) max_resident_bytes: usize,
154    pub(super) assemble_seconds: f64,
155    pub(super) write_seconds: f64,
156}

158impl StreamingTiledShapeWriter {
159    /// Set the manager's default tile for future hypercubes. The explicit
160    /// layout of each existing cube remains the one passed at creation.
161    pub fn set_default_tile_shape(&mut self, default: Vec<usize>) -> Result<(), StorageError> {
162        if default.len() != self.default_tile_shape.len()
163            || default.iter().any(|&extent| extent > i32::MAX as usize)
164        {
165            return Err(StorageError::FormatMismatch(
166                "streamed TiledShape default tile has an invalid rank or extent".into(),
167            ));
168        }
169        self.default_tile_shape = default;
170        Ok(())
171    }

173    #[allow(clippy::too_many_arguments)]
174    pub fn create(
175        temp_data_path: impl Into<PathBuf>,
176        value_type: StreamedTiledShapeValueType,
177        layouts: Vec<StreamedTiledShapeCubeLayout>,
178        memory_budget_bytes: usize,
179        big_endian: bool,
180    ) -> Result<Self, StorageError> {
181        if memory_budget_bytes == 0 {
182            return Err(StorageError::FormatMismatch(
183                "streamed TiledShape memory budget must be positive".to_string(),
184            ));
185        }
186
187        let temp_data_path = temp_data_path.into();
188        if let Some(parent) = temp_data_path.parent() {
189            std::fs::create_dir_all(parent)?;
190        }
191
192        let mut shape_to_cube = HashMap::with_capacity(layouts.len());
193        let mut cubes = Vec::with_capacity(layouts.len());
194        let mut row_count = 0usize;
195        let mut cell_rank = None;
196        let mut default_tile_shape = Vec::new();
197        let mut working_set = ShapeCubeWorkingSet::default();
198
199        // Each layout's extents are checked as its cube is created.
200        for (cube_index, layout) in layouts.into_iter().enumerate() {
201            match cell_rank {
202                Some(rank) if rank != layout.cell_shape.len() => {
203                    return Err(StorageError::FormatMismatch(
204                        "streamed TiledShape layouts must have one cell rank".to_string(),
205                    ));
206                }
207                None => cell_rank = Some(layout.cell_shape.len()),
208                _ => {}
209            }
210            if default_tile_shape.is_empty() {
211                default_tile_shape = layout.tile_shape.clone();
212            }
213            if shape_to_cube
214                .insert(layout.cell_shape.clone(), cube_index)
215                .is_some()
216            {
217                return Err(StorageError::FormatMismatch(format!(
218                    "streamed TiledShape has duplicate layout for shape {:?}",
219                    layout.cell_shape
220                )));
221            }
222
223            row_count = row_count.checked_add(layout.row_count).ok_or_else(|| {
224                StorageError::FormatMismatch(
225                    "streamed TiledShape row count overflows usize".to_string(),
226                )
227            })?;
228            if row_count > u32::MAX as usize || layout.row_count > u32::MAX as usize {
229                return Err(StorageError::FormatMismatch(
230                    "streamed TiledShape row maps support at most u32::MAX rows".to_string(),
231                ));
232            }
233            let (cube, cube_working_set) = DirectShapeCubeWriter::create(
234                streamed_shape_cube_temp_path(&temp_data_path, cube_index),
235                layout,
236                value_type,
237            )?;
238            working_set = working_set.max(cube_working_set);
239            cubes.push(cube);
240        }
241
242        // The largest encoded row, tile row and packed Bool tile row are
243        // held at once, though they may belong to different cubes: the
244        // tile-row scratch is shared and sized for the largest.
245        let mut working_bytes = working_set.bytes().ok_or_else(|| {
246            StorageError::FormatMismatch(
247                "streamed TiledShape working set overflows usize".to_string(),
248            )
249        })?;
250        if working_bytes > memory_budget_bytes {
251            return Err(StorageError::FormatMismatch(format!(
252                "streamed TiledShape working set {working_bytes} exceeds memory budget {memory_budget_bytes}"
253            )));
254        }
255        // Each cube assembles its bands in memory when they fit the budget
256        // beside the working set and the largest row map the rows can need
257        // ([`Self::resident_bytes`]).
258        let band_bytes = cubes
259            .iter()
260            .map(DirectShapeCubeWriter::band_bytes)
261            .sum::<usize>();
262        if shape_cube::buffered_bytes(working_bytes, band_bytes, row_count) <= memory_budget_bytes {
263            cubes
264                .iter_mut()
265                .for_each(DirectShapeCubeWriter::buffer_bands);
266            working_bytes += band_bytes;
267        }
268
269        Ok(Self {
270            value_type,
271            big_endian,
272            cell_rank,
273            default_tile_shape,
274            row_count,
275            rows_written: 0,
276            shape_to_cube,
277            cubes,
278            tile_row: vec![0; working_set.tile_row_bytes],
279            row_map: Vec::new(),
280            cube_map: Vec::new(),
281            pos_map: Vec::new(),
282            memory_budget_bytes,
283            working_bytes,
284            max_resident_bytes: working_bytes,
285            assemble_seconds: 0.0,
286            write_seconds: 0.0,
287        })
288    }

290    pub fn push_bool_row(
291        &mut self,
292        cell_shape: &[usize],
293        values: &[bool],
294    ) -> Result<(), StorageError> {
295        self.require_value_type(StreamedTiledShapeValueType::Bool)?;
296        let assemble_started = Instant::now();
297        let encoded: Vec<u8> = values.iter().map(|&value| u8::from(value)).collect();
298        self.assemble_seconds += assemble_started.elapsed().as_secs_f64();
299        self.push_encoded_row(cell_shape, &encoded)
300    }

302    pub fn push_f32_row(
303        &mut self,
304        cell_shape: &[usize],
305        values: &[f32],
306    ) -> Result<(), StorageError> {
307        self.require_value_type(StreamedTiledShapeValueType::Float32)?;
308        let assemble_started = Instant::now();
309        let mut encoded = vec![0u8; values.len() * 4];
310        for (idx, &value) in values.iter().enumerate() {
311            if self.big_endian {
312                write_f32_be(&mut encoded[idx * 4..], value);
313            } else {
314                write_f32_le(&mut encoded[idx * 4..], value);
315            }
316        }
317        self.assemble_seconds += assemble_started.elapsed().as_secs_f64();
318        self.push_encoded_row(cell_shape, &encoded)
319    }

321    /// Append one Int32 cell in casacore axis order.
322    pub fn push_i32_row(
323        &mut self,
324        cell_shape: &[usize],
325        values: &[i32],
326    ) -> Result<(), StorageError> {
327        self.require_value_type(StreamedTiledShapeValueType::Int32)?;
328        let assemble_started = Instant::now();
329        let mut encoded = Vec::with_capacity(size_of_val(values));
330        for &value in values {
331            encoded.extend_from_slice(&if self.big_endian {
332                value.to_be_bytes()
333            } else {
334                value.to_le_bytes()
335            });
336        }
337        self.assemble_seconds += assemble_started.elapsed().as_secs_f64();
338        self.push_encoded_row(cell_shape, &encoded)
339    }

360    pub fn push_complex32_row(
361        &mut self,
362        cell_shape: &[usize],
363        values: &[Complex32],
364    ) -> Result<(), StorageError> {
365        self.require_value_type(StreamedTiledShapeValueType::Complex32)?;
366        let assemble_started = Instant::now();
367        let mut encoded = vec![0u8; values.len() * 8];
368        if self.big_endian {
369            for (idx, &value) in values.iter().enumerate() {
370                write_complex32_component_bytes(&mut encoded[idx * 8..], value, true);
371            }
372        } else {
373            copy_complex32_native_bytes(&mut encoded, values);
374        }
375        self.assemble_seconds += assemble_started.elapsed().as_secs_f64();
376        self.push_encoded_row(cell_shape, &encoded)
377    }

379    /// Append an undefined variable-shape cell using casacore's dummy cube 0.
380    pub fn push_undefined_row(&mut self) -> Result<(), StorageError> {
381        if self.rows_written >= self.row_count {
382            return Err(StorageError::FormatMismatch(format!(
383                "streamed TiledShape writer received too many rows: expected {}",
384                self.row_count
385            )));
386        }
387        let global_row = self.rows_written as u32;
388        let position = if self.cube_map.last() == Some(&0) {
389            self.pos_map.last().copied().unwrap_or(0).saturating_add(1)
390        } else {
391            0
392        };
393        self.append_row_mapping(global_row, 0, position)?;
394        self.rows_written += 1;
395        Ok(())
396    }

398    pub fn finish(mut self) -> Result<StreamedTiledShapeColumn, StorageError> {
399        if self.rows_written != self.row_count {
400            return Err(StorageError::FormatMismatch(format!(
401                "streamed TiledShape writer finished with {} rows, expected {}",
402                self.rows_written, self.row_count
403            )));
404        }
405        let mut cubes = Vec::with_capacity(self.cubes.len());
406        let mut remapped_cube_indices = vec![0u32; self.cubes.len() + 1];
407        for (old_cube_index, mut cube) in self.cubes.into_iter().enumerate() {
408            if cube.rows_written > cube.row_count {
409                return Err(StorageError::FormatMismatch(format!(
410                    "streamed TiledShape shape {:?} finished with {} rows, capacity {}",
411                    cube.cell_shape, cube.rows_written, cube.row_count
412                )));
413            }
414            if cube.rows_written == 0 {
415                drop(cube.file);
416                std::fs::remove_file(&cube.temp_data_path)?;
417                continue;
418            }
419            let row_tile_count = cube.rows_written.div_ceil(cube.rows_per_tile);
420            let actual_bytes_written = row_tile_count
421                .checked_mul(cube.non_row_tile_count)
422                .and_then(|tiles| tiles.checked_mul(cube.bucket_size))
423                .ok_or_else(|| {
424                    StorageError::FormatMismatch(
425                        "streamed TiledShape final file size overflows usize".to_string(),
426                    )
427                })?;
428            cube.finish_band()?;
429            cube.file.flush()?;
430            cube.file.set_len(actual_bytes_written as u64)?;
431            remapped_cube_indices[old_cube_index + 1] = (cubes.len() + 1) as u32;
432            cubes.push(DirectShapeCubePayload {
433                temp_data_path: cube.temp_data_path,
434                cell_shape: cube.cell_shape,
435                row_count: cube.rows_written,
436                tile_shape: cube.tile_shape,
437                bytes_written: actual_bytes_written,
438            });
439        }
440        for cube_index in &mut self.cube_map {
441            if *cube_index != 0 {
442                *cube_index = remapped_cube_indices[*cube_index as usize];
443                debug_assert_ne!(*cube_index, 0);
444            }
445        }
446        if cubes.is_empty() {
447            self.row_map.clear();
448            self.cube_map.clear();
449            self.pos_map.clear();
450        }
451        Ok(StreamedTiledShapeColumn {
452            row_count: self.row_count,
453            value_type: self.value_type,
454            big_endian: self.big_endian,
455            cell_rank: self.cell_rank.unwrap_or(0),
456            default_tile_shape: self.default_tile_shape,
457            cubes,
458            row_map: self.row_map,
459            cube_map: self.cube_map,
460            pos_map: self.pos_map,
461            max_resident_bytes: self.max_resident_bytes,
462            assemble_seconds: self.assemble_seconds,
463            write_seconds: self.write_seconds,
464        })
465    }

481    fn push_encoded_row(
482        &mut self,
483        cell_shape: &[usize],
484        encoded: &[u8],
485    ) -> Result<(), StorageError> {
486        if self.rows_written >= self.row_count {
487            return Err(StorageError::FormatMismatch(format!(
488                "streamed TiledShape writer received too many rows: expected {}",
489                self.row_count
490            )));
491        }
492        let cube_index = self.shape_to_cube.get(cell_shape).copied().ok_or_else(|| {
493            StorageError::FormatMismatch(format!(
494                "streamed TiledShape row has undeclared shape {cell_shape:?}"
495            ))
496        })?;
497        let cube = &mut self.cubes[cube_index];
498        if cube.rows_written >= cube.row_count {
499            return Err(StorageError::FormatMismatch(format!(
500                "streamed TiledShape received too many rows for shape {cell_shape:?}: expected {}",
501                cube.row_count
502            )));
503        }
504        let expected_bytes =
505            cube.cell_shape.iter().product::<usize>() * self.value_type.elem_size();
506        if encoded.len() != expected_bytes {
507            return Err(StorageError::FormatMismatch(format!(
508                "streamed TiledShape row has {} bytes, expected {expected_bytes} for shape {cell_shape:?}",
509                encoded.len()
510            )));
511        }
512
513        let global_row = self.rows_written as u32;
514        let pos_in_cube = cube.rows_written as u32;
515        let write_started = Instant::now();
516        cube.write_row(self.value_type, encoded, &mut self.tile_row)?;
517        self.write_seconds += write_started.elapsed().as_secs_f64();
518        self.rows_written += 1;
519        self.append_row_mapping(global_row, (cube_index + 1) as u32, pos_in_cube)?;
520        Ok(())
521    }
522
523    fn append_row_mapping(
524        &mut self,
525        global_row: u32,
526        cube: u32,
527        position: u32,
528    ) -> Result<(), StorageError> {
529        let extends_last = self.row_map.last().is_some_and(|&last_row| {
530            self.cube_map.last() == Some(&cube)
531                && last_row + 1 == global_row
532                && self
533                    .pos_map
534                    .last()
535                    .is_some_and(|&last_pos| last_pos + 1 == position)
536        });
537        let map_entries_after = self.row_map.len() + usize::from(!extends_last);
538        let map_bytes_after = map_entries_after
539            .checked_mul(3 * std::mem::size_of::<u32>())
540            .ok_or_else(|| {
541                StorageError::FormatMismatch(
542                    "streamed TiledShape row-map size overflows usize".to_string(),
543                )
544            })?;
545        if self.working_bytes + map_bytes_after > self.memory_budget_bytes {
546            return Err(StorageError::FormatMismatch(format!(
547                "streamed TiledShape row-map working set exceeds memory budget {}",
548                self.memory_budget_bytes
549            )));
550        }
551        if extends_last {
552            *self.row_map.last_mut().expect("checked row-map entry") = global_row;
553            *self.pos_map.last_mut().expect("checked position entry") = position;
554        } else {
555            self.row_map.push(global_row);
556            self.cube_map.push(cube);
557            self.pos_map.push(position);
558        }
559        self.max_resident_bytes = self
560            .max_resident_bytes
561            .max(self.working_bytes + self.row_map.len() * 3 * std::mem::size_of::<u32>());
562        Ok(())
563    }
564}

572/// Install a completed heterogeneous payload using casacore's canonical
573/// `TiledShapeStMan` header and row-map conventions.
574#[doc(hidden)]
575pub fn install_streamed_tiled_shape_column(
576    table_path: &Path,
577    dm_seq_nr: u32,
578    dm_name: &str,
579    streamed: StreamedTiledShapeColumn,
580) -> Result<(), StorageError> {
581    std::fs::create_dir_all(table_path)?;
582    let mut files = Vec::with_capacity(streamed.cubes.len());
583    let mut cubes = vec![TsmCubeInfo {
584        values: RecordValue::default(),
585        extensible: false,
586        cube_shape: vec![],
587        tile_shape: vec![],
588        file_seq_nr: -1,
589        file_offset: 0,
590    }];
591    let cell_rank = streamed.cell_rank;
592    let default_tile_shape = streamed.default_tile_shape.clone();
593    for (file_index, cube) in streamed.cubes.into_iter().enumerate() {
594        let target = tsm_data_path(table_path, dm_seq_nr, file_index as u32);
595        match std::fs::rename(&cube.temp_data_path, &target) {
596            Ok(()) => {}
597            Err(_) => {
598                std::fs::copy(&cube.temp_data_path, &target)?;
599                std::fs::remove_file(&cube.temp_data_path)?;
600            }
601        }
602        files.push(Some(TsmFileInfo {
603            seq_nr: file_index as u32,
604            length: cube.bytes_written as i64,
605        }));
606        let mut cube_shape = cube.cell_shape;
607        cube_shape.push(cube.row_count);
608        cubes.push(TsmCubeInfo {
609            values: RecordValue::default(),
610            extensible: true,
611            cube_shape,
612            tile_shape: cube.tile_shape,
613            file_seq_nr: file_index as i32,
614            file_offset: 0,
615        });
616    }
617
618    let header = TiledStManHeader {
619        big_endian: streamed.big_endian,
620        seq_nr: dm_seq_nr,
621        nrrow: streamed.row_count as u64,
622        col_data_types: vec![streamed.value_type.casacore_type()],
623        hypercolumn_name: dm_name.to_string(),
624        max_cache_size: 0,
625        nrdim: if cell_rank == 0 {
626            0
627        } else {
628            (cell_rank + 1) as u32
629        },
630        files,
631        cubes,
632    };
633    let variant = TiledVariant::Shape {
634        default_tile_shape: default_tile_shape
635            .iter()
636            .map(|&value| value as i32)
637            .collect(),
638        nr_used_row_map: streamed.row_map.len() as u32,
639        row_map: streamed.row_map,
640        cube_map: streamed.cube_map,
641        pos_map: streamed.pos_map,
642    };
643    write_tiled_header(
644        &table_path.join(format!("table.f{dm_seq_nr}")),
645        &variant,
646        &header,
647    )?;
648    invalidate_shared_tile_cache_for_table(table_path);
649    Ok(())
650}

652/// Install a completed single-shape payload using casacore's canonical
653/// `TiledColumnStMan` header conventions.
654///
655/// The direct writer is shared with heterogeneous `TiledShapeStMan` output,
656/// but a column manager has exactly one defined shape and no row map.
657#[doc(hidden)]
658pub fn install_streamed_tiled_column(
659    table_path: &Path,
660    dm_seq_nr: u32,
661    dm_name: &str,
662    mut streamed: StreamedTiledShapeColumn,
663) -> Result<(), StorageError> {
664    if streamed.row_count == 0
665        || streamed.cubes.len() != 1
666        || streamed.row_map != vec![(streamed.row_count - 1) as u32]
667        || streamed.cube_map != vec![1]
668        || streamed.pos_map != vec![(streamed.row_count - 1) as u32]
669    {
670        return Err(StorageError::FormatMismatch(
671            "streamed TiledColumnStMan payload must contain one defined shape for every row"
672                .to_string(),
673        ));
674    }
675
676    std::fs::create_dir_all(table_path)?;
677    let cube = streamed.cubes.pop().expect("validated one cube");
678    let target = tsm_data_path(table_path, dm_seq_nr, 0);
679    match std::fs::rename(&cube.temp_data_path, &target) {
680        Ok(()) => {}
681        Err(_) => {
682            std::fs::copy(&cube.temp_data_path, &target)?;
683            std::fs::remove_file(&cube.temp_data_path)?;
684        }
685    }
686    let mut cube_shape = cube.cell_shape;
687    cube_shape.push(cube.row_count);
688    let header = TiledStManHeader {
689        big_endian: streamed.big_endian,
690        seq_nr: dm_seq_nr,
691        nrrow: streamed.row_count as u64,
692        col_data_types: vec![streamed.value_type.casacore_type()],
693        hypercolumn_name: dm_name.to_string(),
694        max_cache_size: 0,
695        nrdim: (streamed.cell_rank + 1) as u32,
696        files: vec![Some(TsmFileInfo {
697            seq_nr: 0,
698            length: cube.bytes_written as i64,
699        })],
700        cubes: vec![TsmCubeInfo {
701            values: RecordValue::default(),
702            extensible: true,
703            cube_shape,
704            tile_shape: cube.tile_shape,
705            file_seq_nr: 0,
706            file_offset: 0,
707        }],
708    };
709    let variant = TiledVariant::Column {
710        default_tile_shape: streamed
711            .default_tile_shape
712            .iter()
713            .map(|&value| value as i32)
714            .collect(),
715    };
716    write_tiled_header(
717        &table_path.join(format!("table.f{dm_seq_nr}")),
718        &variant,
719        &header,
720    )?;
721    invalidate_shared_tile_cache_for_table(table_path);
722    Ok(())
723}

725#[allow(clippy::too_many_arguments)]
726pub(super) fn write_single_column_shape_cube_direct_to_file(
727    tsm_path: &Path,
728    values: &[Option<&Value>],
729    cell_shape: &[usize],
730    tile_shape: &[usize],
731    cell_nelem: usize,
732    elem_size: usize,
733    nr_tiles: usize,
734    bucket_size: usize,
735    col_offset: usize,
736    col_data_type: CasacoreDataType,
737    big_endian: bool,
738) -> Result<Option<StreamedShapeCubeWrite>, StorageError> {
739    if tile_shape.len() != cell_shape.len() + 1 {
740        return Ok(None);
741    }
742    if tile_shape[..cell_shape.len()] != *cell_shape {
743        return Ok(None);
744    }
745    let rows_per_tile = tile_shape[cell_shape.len()];
746    if rows_per_tile == 0 {
747        return Ok(None);
748    }
749    if col_data_type != CasacoreDataType::TpBool && col_data_type != CasacoreDataType::TpComplex {
750        return Ok(None);
751    }
752
753    for value in values.iter().flatten() {
754        match (col_data_type, *value) {
755            (CasacoreDataType::TpBool, Value::Array(ArrayValue::Bool(array))) => {
756                if array.len() != cell_nelem {
757                    return Ok(None);
758                }
759            }
760            (CasacoreDataType::TpComplex, Value::Array(ArrayValue::Complex32(array))) => {
761                if array.len() != cell_nelem || array.shape() != cell_shape {
762                    return Ok(None);
763                }
764            }
765            _ => return Ok(None),
766        }
767    }
768
769    let row_bytes = cell_nelem * elem_size;
770    let tile_nelem: usize = tile_shape.iter().product();
771    let tile_storage_len = tile_storage_bytes(col_data_type, tile_nelem);
772    let tile_end = col_offset + tile_storage_len;
773    if tile_end > bucket_size {
774        return Ok(None);
775    }
776
777    let mut file = std::fs::File::create(tsm_path)?;
778    let mut tile_data = vec![0u8; bucket_size];
779    let mut assemble_seconds = 0.0;
780    let mut write_seconds = 0.0;
781    for tile_idx in 0..nr_tiles {
782        tile_data.fill(0);
783        let assemble_started = Instant::now();
784        for row_in_tile in 0..rows_per_tile {
785            let pos_in_cube = tile_idx * rows_per_tile + row_in_tile;
786            let Some(value) = values.get(pos_in_cube).and_then(|value| *value) else {
787                continue;
788            };
789            if col_data_type == CasacoreDataType::TpBool {
790                let Value::Array(ArrayValue::Bool(array)) = value else {
791                    unreachable!("preflight checked bool tiled shape values");
792                };
793                if array.iter().all(|value| !*value) {
794                    continue;
795                }
796                let bools = array_memory_order_values(array);
797                write_bool_bits(
798                    &mut tile_data[col_offset..tile_end],
799                    row_in_tile * cell_nelem,
800                    &bools,
801                );
802            } else {
803                let Value::Array(ArrayValue::Complex32(array)) = value else {
804                    unreachable!("preflight checked complex tiled shape values");
805                };
806                let dst_start = col_offset + row_in_tile * row_bytes;
807                let dst_end = dst_start + row_bytes;
808                if dst_end > tile_data.len() {
809                    return Ok(None);
810                }
811                encode_complex32_array_into_tile_row(
812                    &mut tile_data[dst_start..dst_end],
813                    array,
814                    cell_shape,
815                    big_endian,
816                )?;
817            }
818        }
819        assemble_seconds += assemble_started.elapsed().as_secs_f64();
820        let write_started = Instant::now();
821        file.write_all(&tile_data)?;
822        write_seconds += write_started.elapsed().as_secs_f64();
823    }
824
825    Ok(Some(StreamedShapeCubeWrite {
826        bytes: nr_tiles * bucket_size,
827        assemble_seconds,
828        write_seconds,
829    }))
830}

832pub(super) fn encode_complex32_array_into_tile_row(
833    dst: &mut [u8],
834    array: &ArrayD<Complex32>,
835    cell_shape: &[usize],
836    big_endian: bool,
837) -> Result<(), StorageError> {
838    if dst.len() != array.len() * 8 {
839        return Err(StorageError::FormatMismatch(
840            "complex tile row has unexpected byte length".to_string(),
841        ));
842    }
843
844    if !big_endian
845        && is_fortran_contiguous(array.shape(), array.strides())
846        && let Some(slice) = array.as_slice_memory_order()
847    {
848        copy_complex32_native_bytes(dst, slice);
849        return Ok(());
850    }
851
852    if let Some(slice) = array.as_slice_memory_order()
853        && cell_shape.len() == 2
854        && array.strides() == [cell_shape[1] as isize, 1]
855    {
856        let n0 = cell_shape[0];
857        let n1 = cell_shape[1];
858        for axis1 in 0..n1 {
859            for axis0 in 0..n0 {
860                let value = slice[axis0 * n1 + axis1];
861                let dst_offset = (axis1 * n0 + axis0) * 8;
862                write_complex32_component_bytes(
863                    &mut dst[dst_offset..dst_offset + 8],
864                    value,
865                    big_endian,
866                );
867            }
868        }
869        return Ok(());
870    }
871
872    let slice = array_memory_order_values(array);
873    for (idx, &value) in slice.iter().enumerate() {
874        let dst_offset = idx * 8;
875        write_complex32_component_bytes(&mut dst[dst_offset..dst_offset + 8], value, big_endian);
876    }
877    Ok(())
878}

880pub(super) fn copy_complex32_native_bytes(dst: &mut [u8], values: &[Complex32]) {
881    debug_assert_eq!(dst.len(), std::mem::size_of_val(values));
882    // `num_complex::Complex32` is the same in-memory pair of f32 values that
883    // casacore writes for little-endian complex cells. This fast path is only
884    // used after the caller has already established the array's MS storage
885    // order, so no logical transposition happens here.
886    let src = unsafe {
887        std::slice::from_raw_parts(values.as_ptr().cast::<u8>(), std::mem::size_of_val(values))
888    };
889    dst.copy_from_slice(src);
890}

907pub(super) fn write_complex32_component_bytes(dst: &mut [u8], value: Complex32, big_endian: bool) {
908    if big_endian {
909        write_f32_be(dst, value.re);
910        write_f32_be(&mut dst[4..], value.im);
911    } else {
912        write_f32_le(dst, value.re);
913        write_f32_le(&mut dst[4..], value.im);
914    }
