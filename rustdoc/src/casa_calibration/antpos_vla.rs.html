4use casa_ms::{MeasurementSet, MsResult, derived::engine::MsCalEngine};
5
6pub(crate) const TROPO_KEYWORD: &str = "VLATrDelCorr";

8pub(crate) fn only_vla_observations(ms: &MeasurementSet) -> MsResult<bool> {
9    let observations = ms.observation()?;
10    for row in 0..observations.row_count() {
11        if !observations.string(row, "TELESCOPE_NAME")?.contains("VLA") {
12            return Ok(false);
13        }
14    }
15    Ok(observations.row_count() != 0)
16}
17
18pub(crate) struct TropoDelay {
19    arms: Vec<Option<Arm>>,
20}
21
22struct Arm {
23    azimuth_rad: f64,
24    distance_m: f64,
25}
26
27impl TropoDelay {
28    pub(crate) fn new(ms: &MeasurementSet, engine: &MsCalEngine, scale: f64) -> MsResult<Self> {
29        let antennas = ms.antenna()?;
30        let center = engine.observatory_position().as_itrf();
31        let arms = (0..antennas.row_count())
32            .map(|row| {
33                let station = antennas.station(row)?;
34                let stripped = station
35                    .strip_prefix("EVLA:")
36                    .or_else(|| station.strip_prefix("VLA:"));
37                let azimuth = arm_azimuth(stripped.unwrap_or(&station), stripped.is_some());
38                let Some(azimuth_rad) = azimuth else {
39                    return Ok(None);
40                };
41                let position = antennas.position(row)?;
42                let radius = position
43                    .into_iter()
44                    .zip(center)
45                    .map(|(position, center)| (position - center).powi(2))
46                    .sum::<f64>()
47                    .sqrt();
48                let distance_m =
49                    radius * (-1.0e-15 * scale) * crate::execute::SPEED_OF_LIGHT_M_PER_S;
50                Ok(Some(Arm {
51                    azimuth_rad,
52                    distance_m,
53                }))
54            })
55            .collect::<MsResult<_>>()?;
56        Ok(Self { arms })
57    }
58
59    pub(crate) fn metres(
60        &self,
61        engine: &MsCalEngine,
62        time_seconds: f64,
63        field: usize,
64        antenna: usize,
65    ) -> MsResult<f64> {
66        let Some(Some(arm)) = self.arms.get(antenna) else {
67            return Ok(0.0);
68        };
69        let (azimuth, elevation) = engine.azel(time_seconds, field, antenna)?;
70        Ok((azimuth - arm.azimuth_rad).cos() / elevation.tan() / elevation.sin() * arm.distance_m)
71    }
72}
73
74fn arm_azimuth(station: &str, has_prefix: bool) -> Option<f64> {
75    [('N', 355.0_f64), ('E', 115.0), ('W', 236.0)]
76        .into_iter()
77        .find(|(arm, _)| {
78            if has_prefix {
79                station.starts_with(*arm)
80            } else {
81                station.contains(*arm)
82            }
83        })
84        .map(|(_, degrees)| degrees.to_radians())
85}
