14/// Relative allowance for the Doppler factor between the frame of a
15/// spectral window's stored `CHAN_FREQ` and the output frame rows reach the
16/// passes in: 3,000 km/s.
17pub const FRAME_MARGIN: f64 = 0.01;
18
19/// One row of the blocks `problem`'s source delivers: the most channels any
20/// selected spectral window gives a row and their frequencies' extent, the
21/// most correlations any selection routes, and the image domains each row
22/// is projected on; no rows.
23#[must_use]
24pub fn row_layout(problem: &CompiledProblem) -> BlockShape {
25    let sources = problem.observation_transaction().read_set().sources();
26    BlockShape {
27        rows: 0,
28        spectrum: row_spectrum(
29            sources
30                .iter()
31                .flat_map(|source| source.selection().spectral_windows()),
32        ),
33        correlations: sources
34            .iter()
35            .flat_map(|source| source.selection().correlations())
36            .map(|selection| selection.products().len())
37            .max()
38            .unwrap_or(0),
39        domains: problem.geometry().domains().len(),
40    }
41}

43/// What bounds the samples a row of `windows` places: its most selected
44/// channels, and from the storage owner's `CHAN_FREQ` the smallest first
45/// separation and the widest span of any window's selected channels,
46/// widened by [`FRAME_MARGIN`]. A window without a catalog leaves its rows
47/// evenly spaced ([`RowSpectrum::evenly_spaced`]); the storage owner binds
48/// one to every MeasurementSet selection.
49fn row_spectrum<'a>(windows: impl Iterator<Item = &'a SpectralWindowSelection>) -> RowSpectrum {
50    let mut spectrum = RowSpectrum::evenly_spaced(0);
51    let mut known = true;
52    for window in windows {
53        let channels = window.channel_indices();
54        spectrum.channels = spectrum.channels.max(channels.len());
55        let frequency = |index: usize| {
56            let catalog = window.coordinate_catalog()?;
57            catalog.channel_frequency_hz(*channels.get(index)? as usize)
58        };
59        if channels.len() < 2 {
60            continue;
61        }
62        let (Some(first), Some(second), Some(last)) =
63            (frequency(0), frequency(1), frequency(channels.len() - 1))
64        else {
65            known = false;
66            continue;
67        };
68        let spacing = (second - first).abs() * (1.0 - FRAME_MARGIN);
69        if spacing > 0.0
70            && (spectrum.first_spacing_hz == 0.0 || spacing < spectrum.first_spacing_hz)
71        {
72            spectrum.first_spacing_hz = spacing;
73        }
74        spectrum.span_hz = spectrum
75            .span_hz
76            .max((last - first).abs() * (1.0 + FRAME_MARGIN));
77    }
78    if known {
79        spectrum
80    } else {
81        RowSpectrum::evenly_spaced(spectrum.channels)
82    }
83}

85/// The budget that bounds source inspection before the compiled problem
86/// can quote its complete initialization and traversal requirements.
87#[must_use]
88pub const fn bootstrap_source_budget() -> SelectedObservationContentBudget {
89    SelectedObservationContentBudget::new(64 << 20, 2, 4)
90}

92/// Why a source's execution envelope could not be finalized.
93#[derive(Debug, thiserror::Error)]
94pub enum SourceAccessError {
95    /// The storage owner could not quote or bind the source's requirements.
96    #[error("selected-observation access: {0}")]
97    Access(#[from] BoundSelectedObservationError),
98    /// The requirement curve could not be evaluated or planned.
99    #[error("selected-observation content plan: {0}")]
100    ContentPlan(#[from] SelectedObservationContentPlanError),
101    /// The policy's free memory cannot hold the source's envelope.
102    #[error(transparent)]
103    Admission(#[from] Admission),
104    /// The preferred envelope does not fit the address space.
105    #[error("the selected-observation envelope overflows")]
106    Overflow,
107}

109/// Finalize an unopened source's bounded execution envelope and admit it,
110/// with the stream that converts its blocks for the passes.
111///
112/// The storage owner supplies the requirement curve; the source takes its
113/// mandatory minimum and grows beyond it by at most the bootstrap budget.
114/// Each row a block holds is also converted into the native blocks a pass's
115/// stream double-buffers, and predicted by a pass with a model
116/// ([`source_block_bytes`]); the source and that storage together grow by
117/// at most a quarter of what `policy` leaves free on `host` past the
118/// minimum, shared in proportion to their bytes per row, so the paged cube
119/// cache and the passes, admitted after them, keep the rest. A row of
120/// [`row_layout`] produces at most `row` samples, which the passes' row
121/// chunks hold until they reach their cap. The reservation holds the
122/// envelope the source plans and what the passes hold for its blocks, and
123/// the run keeps it while the source is open.
124pub fn finalize_source_access(
125    problem: &CompiledProblem,
126    access: ResolvedSelectedObservationAccess,
127    row: RowSamples,
128    host: &HostResources,
129    policy: &ResourcePolicy,
130) -> Result<(ResolvedSelectedObservationAccess, Reservation), SourceAccessError> {
131    let requirements = access.content_requirements(problem)?;
132    let maximum_live_blocks = access
133        .source_binding()
134        .content_budget()
135        .maximum_live_blocks();
136    let minimum = requirements.minimum_bytes()?;
137    let free = usize::try_from(free_memory(host, policy)).unwrap_or(usize::MAX);
138    let bootstrap = bootstrap_source_budget().available_bytes();
139    let quarter = free.saturating_sub(minimum) / 4;
140    let budget = |growth: usize| {
141        Ok::<_, SourceAccessError>(SelectedObservationContentBudget::new(
142            minimum
143                .checked_add(growth)
144                .ok_or(SourceAccessError::Overflow)?,
145            maximum_live_blocks,
146            requirements.maximum_pointing_polynomial_terms(),
147        ))
148    };
149    let layout = row_layout(problem);
150    let passes = |rows: usize| source_block_bytes(BlockShape { rows, ..layout });
151    // The passes' row chunks also grow with the block, under the passes'
152    // own charge, until they reach their cap; the passes stream on the
153    // allocation's streaming workers.
154    let workers = policy.cores(host).streaming_workers;
155    let passes_per_row = passes(1) - passes(0) + chunk_bytes_per_row(layout, row, workers);
156    let unshared = requirements.plan(budget(bootstrap.min(quarter))?)?;
157    let source_per_row =
158        (unshared.maximum_resident_bytes() / unshared.rows_per_block().max(1)) as u64;
159    let share = u128::from(source_per_row) * quarter as u128
160        / u128::from((source_per_row + passes_per_row).max(1));
161    let planned = requirements.plan(budget(
162        bootstrap.min(usize::try_from(share).map_err(|_| SourceAccessError::Overflow)?),
163    )?)?;
164    let passes = passes(planned.rows_per_block());
165    tracing::info!(
166        rows_per_block = planned.rows_per_block(),
167        envelope_bytes = planned.maximum_resident_bytes(),
168        pass_bytes = passes,
169        minimum_bytes = minimum,
170        live_blocks = maximum_live_blocks,
171        "selected-observation source plan"
172    );
173    let reservation = admit(
174        host,
175        policy,
176        &Demand {
177            phase: "selected-observation source",
178            memory: u64::try_from(planned.maximum_resident_bytes())
179                .map_err(|_| SourceAccessError::Overflow)?
180                .checked_add(passes)
181                .ok_or(SourceAccessError::Overflow)?,
182        },
183    )?;
184    let budget = SelectedObservationContentBudget::new(
185        planned.maximum_resident_bytes(),
186        maximum_live_blocks,
187        requirements.maximum_pointing_polynomial_terms(),
188    );
189    Ok((access.with_content_budget(budget), reservation))
190}
