2//! Shifted subtraction of a PSF or scale patch from a residual plane.
3
4use crate::Error;
5use crate::plane::PlaneShape;
6
7/// Subtract `scale × source`, shifted so that `centre` of the source lands
8/// on `at` of the target, over CASA's update box: offsets `−h ..= h − 1`
9/// from `at` on each axis, clipped to both planes.
10///
11/// This is the box of `hclean` (`x1 = px − nx/2`, `x2 = px + nx/2 − 1`), of
12/// `MatrixCleaner::clean` (a full-image box) and of the multi-term cleaner's
13/// PSF patch (`buildImagePatches`). Their image and patch boxes stay
14/// aligned because the source centre sits `h` from its low edge.
15#[allow(clippy::too_many_arguments)]
16pub(crate) fn subtract_window(
17    target: &mut [f64],
18    target_shape: PlaneShape,
19    source: &[f64],
20    source_shape: PlaneShape,
21    centre: [usize; 2],
22    at: [usize; 2],
23    half: [usize; 2],
24    scale: f64,
25) -> Result<(), Error> {
26    if !scale.is_finite() {
27        return Err(Error::NonFinite);
28    }
29    let axis = |at: usize, centre: usize, half: usize, target: usize, source: usize| {
30        // Offsets d with 0 <= at + d < target and 0 <= centre + d < source.
31        let low = (at.min(half)).min(centre) as isize;
32        let high = ((target - at).min(half)).min(source - centre) as isize;
33        (-low)..high
34    };
35    let xs = axis(at[0], centre[0], half[0], target_shape.nx, source_shape.nx);
36    let ys = axis(at[1], centre[1], half[1], target_shape.ny, source_shape.ny);
37    if ys.is_empty() {
38        return Ok(());
39    }
40    let length = (ys.end - ys.start) as usize;
41    for dx in xs {
42        let tx = (at[0] as isize + dx) as usize;
43        let sx = (centre[0] as isize + dx) as usize;
44        let ty = (at[1] as isize + ys.start) as usize;
45        let sy = (centre[1] as isize + ys.start) as usize;
46        let row = &mut target[tx * target_shape.ny + ty..][..length];
47        let kernel = &source[sx * source_shape.ny + sy..][..length];
48        for (value, coefficient) in row.iter_mut().zip(kernel) {
49            *value -= scale * coefficient;
50        }
51    }
52    Ok(())
53}
54
55/// Subtract `flux ×` the PSF shifted from its peak to `at`, over `hclean`'s
56/// box of half the plane on each side.
57pub(crate) fn subtract_shifted(
58    residual: &mut [f64],
59    psf: &[f64],
60    shape: PlaneShape,
61    psf_peak: usize,
62    at: usize,
63    flux: f64,
64) -> Result<(), Error> {
65    subtract_window(
66        residual,
67        shape,
68        psf,
69        shape,
70        shape.pixel(psf_peak),
71        shape.pixel(at),
72        shape.centre(),
73        flux,
74    )
75}

77#[cfg(test)]
78mod tests {
79    use super::*;
80
81    /// The window subtraction equals a direct sum over the CASA box for
82    /// every target position, on an odd non-square plane with an
83    /// off-centre source centre.
84    #[test]
85    fn the_window_is_the_clipped_casa_box() {
86        let target_shape = PlaneShape::new(7, 5);
87        let source_shape = PlaneShape::new(6, 9);
88        let source = (0..source_shape.len())
89            .map(|i| (i as f64 * 0.37).sin())
90            .collect::<Vec<_>>();
91        let centre = [2, 5];
92        let half = [3, 2];
93        for at in 0..target_shape.len() {
94            let mut actual = vec![1.0; target_shape.len()];
95            let at_pixel = target_shape.pixel(at);
96            subtract_window(
97                &mut actual,
98                target_shape,
99                &source,
100                source_shape,
101                centre,
102                at_pixel,
103                half,
104                0.5,
105            )
106            .unwrap();
107            let mut expected = vec![1.0; target_shape.len()];
108            for dx in -(half[0] as isize)..half[0] as isize {
109                for dy in -(half[1] as isize)..half[1] as isize {
110                    let t = [at_pixel[0] as isize + dx, at_pixel[1] as isize + dy];
111                    let s = [centre[0] as isize + dx, centre[1] as isize + dy];
112                    if (0..7).contains(&t[0])
113                        && (0..5).contains(&t[1])
114                        && (0..6).contains(&s[0])
115                        && (0..9).contains(&s[1])
116                    {
117                        expected[target_shape.index(t[0] as usize, t[1] as usize)] -=
118                            0.5 * source[source_shape.index(s[0] as usize, s[1] as usize)];
119                    }
120                }
121            }
122            assert_eq!(actual, expected, "at {at_pixel:?}");
123        }
124    }
125
126    #[test]
127    fn a_nonfinite_scale_is_refused() {
128        let shape = PlaneShape::new(4, 4);
129        let mut target = vec![0.0; 16];
130        assert_eq!(
131            subtract_shifted(&mut target, &[1.0; 16], shape, 10, 5, f64::NAN),
132            Err(Error::NonFinite)
133        );
134    }
