2//! Prepare the circular Fourier model consumed by the legacy AW imager.
3
4use ndarray::ArrayD;
5use num_complex::Complex32;
6
7use super::AwCatalog;
8use crate::error::OperatorError;
9use crate::fft::PlaneFft;
10
11impl AwCatalog {
12    /// Prepare the four single-channel circular Fourier planes from a legacy
13    /// `AWProjectFT` IQUV image and its optional cached sensitivity image.
14    ///
15    /// # Errors
16    /// The image, sensitivity image or PB cutoff is incompatible.
17    pub fn prepare_legacy_model(
18        &self,
19        model: &ArrayD<f32>,
20        average_beam: Option<&ArrayD<f32>>,
21        cutoff: Option<f32>,
22    ) -> Result<[Vec<Complex32>; 4], OperatorError> {
23        let [nx, ny, 4, 1] = *model.shape() else {
24            return Err(invalid(
25                "legacy AW prediction needs a single-channel IQUV image",
26            ));
27        };
28        if nx != ny || !nx.is_multiple_of(2) || nx == 0 {
29            return Err(invalid(
30                "legacy AW needs a square image with even pixel lengths",
31            ));
32        }
33        if cutoff.is_some_and(|limit| !limit.is_finite() || limit < 0.)
34            || (cutoff.is_some() && average_beam.is_none())
35            || average_beam.is_some_and(|beam| {
36                beam.shape() != model.shape()
37                    || beam.iter().any(|pixel| !pixel.is_finite() || *pixel < 0.)
38            })
39        {
40            return Err(invalid("invalid legacy AW average beam or cutoff"));
41        }
42        let sinc = (0..nx)
43            .map(|pixel| {
44                let x = std::f32::consts::PI * (pixel as isize - (nx / 2) as isize) as f32
45                    / (nx as f32 * f32::from(self.oversampling));
46                if x == 0. { 1. } else { x.sin() / x }
47            })
48            .collect::<Vec<_>>();
49        let mut grids = std::array::from_fn(|_| Vec::with_capacity(nx * ny));
50        for (y, &sy) in sinc.iter().enumerate() {
51            for (x, &sx) in sinc.iter().enumerate() {
52                let mut stokes = std::array::from_fn(|pol| model[[x, y, pol, 0]]);
53                if let (Some(beam), Some(limit)) = (average_beam, cutoff) {
54                    for (pol, value) in stokes.iter_mut().enumerate() {
55                        let power = beam[[x, y, pol, 0]];
56                        if power > limit {
57                            *value /= power.sqrt();
58                        }
59                    }
60                }
61                let [i, q, u, v] = stokes;
62                let circular = [
63                    Complex32::from(i + v),
64                    Complex32::new(q, u),
65                    Complex32::new(q, -u),
66                    Complex32::from(i - v),
67                ];
68                for (grid, value) in grids.iter_mut().zip(circular) {
69                    let taper = sx * sy;
70                    let value = match (average_beam, cutoff) {
71                        (Some(_), Some(_)) => value / taper,
72                        (Some(_), None) => value * taper,
73                        (None, _) => value,
74                    };
75                    grid.push(value);
76                }
77            }
78        }
79        let mut fft = PlaneFft::new([nx, ny], false)?;
80        for grid in &mut grids {
81            fft.transform(grid, false)?;
82        }
83        Ok(grids)
84    }
85}
86
87fn invalid(reason: &'static str) -> OperatorError {
88    OperatorError::ConvolutionFunction { reason }
89}
