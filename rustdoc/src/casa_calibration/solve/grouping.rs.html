24pub(crate) fn validate_solve_interval(
25    solve_interval: GainSolveInterval,
26) -> Result<(), GainSolveError> {
27    match solve_interval {
28        GainSolveInterval::Infinite | GainSolveInterval::Integration => Ok(()),
29        GainSolveInterval::Seconds(seconds) if seconds > 0.0 => Ok(()),
30        _ => Err(GainSolveError::UnsupportedSolveInterval { solve_interval }),
31    }
32}

34pub(crate) fn validate_smodel(smodel: [f32; 4]) -> Result<(), GainSolveError> {
35    if smodel[0] <= 0.0 || smodel[1] != 0.0 || smodel[2] != 0.0 || smodel[3] != 0.0 {
36        return Err(GainSolveError::UnsupportedSkyModel { smodel });
37    }
38    Ok(())
39}

41pub(crate) fn load_preapplied_rows(
42    ms: &MeasurementSet,
43    selection: &MsSelection,
44    prior_calibration_tables: &[ApplyCalibrationTableSpec],
45    parang: bool,
46) -> Result<Option<HashMap<usize, EvaluatedApplyRow>>, GainSolveError> {
47    if prior_calibration_tables.is_empty() && !parang {
48        return Ok(None);
49    }
50    let plan = plan_apply(
51        ms,
52        &ApplyPlanRequest {
53            selection: selection.clone(),
54            apply_mode: crate::ApplyMode::CalFlag,
55            parang,
56            calibration_tables: prior_calibration_tables.to_vec(),
57        },
58    )
59    .map_err(|source| GainSolveError::PriorCalibrationPlan {
60        source: Box::new(source),
61    })?;
62    evaluate_apply_rows(ms, &plan).map(Some).map_err(|source| {
63        GainSolveError::PriorCalibrationApply {
64            source: Box::new(source),
65        }
66    })
67}

69pub(crate) fn build_solve_groups(
70    ms: &MeasurementSet,
71    rows: &[SelectedSolveRow],
72    preapplied_rows: Option<&HashMap<usize, EvaluatedApplyRow>>,
73    options: SolveGroupOptions,
74) -> Result<BTreeMap<(SolveBaseKey, SolveBucketKey), SolveAccumulator>, GainSolveError> {
75    let mut sorted_rows = rows.to_vec();
76    sorted_rows.sort_by_key(|row| {
77        (
78            row.field_id,
79            row.data_spw_id,
80            row.observation_id,
81            row.scan_number,
82            row.time_seconds.to_bits(),
83            row.interval_seconds.to_bits(),
84            row.row_index,
85        )
86    });
87
88    let mut groups = BTreeMap::<(SolveBaseKey, SolveBucketKey), SolveAccumulator>::new();
89    let mut bucket_starts = BTreeMap::<SolveBaseKey, f64>::new();
90
91    for row in sorted_rows {
92        if row.antenna1 == row.antenna2 {
93            continue;
94        }
95        let base_key = SolveBaseKey {
96            field_id: if options.combine.fields {
97                0
98            } else {
99                row.field_id
100            },
101            spw_id: row.data_spw_id,
102            observation_id: row.observation_id,
103            scan_number: if options.combine.scans {
104                0
105            } else {
106                row.scan_number
107            },
108        };
109        let bucket_key = match options.solve_interval {
110            GainSolveInterval::Infinite => SolveBucketKey::Infinite,
111            GainSolveInterval::Integration => SolveBucketKey::Integration {
112                time_bits: row.time_seconds.to_bits(),
113                interval_bits: row.interval_seconds.to_bits(),
114            },
115            GainSolveInterval::Seconds(seconds) => {
116                let bucket_start = bucket_starts.entry(base_key).or_insert(row.time_seconds);
117                let bucket_index = ((row.time_seconds - *bucket_start) / seconds)
118                    .floor()
119                    .max(0.0);
120                SolveBucketKey::Seconds(bucket_index as u64)
121            }
122        };
123
124        let group = groups.entry((base_key, bucket_key)).or_insert_with(|| {
125            SolveAccumulator::new(row.field_id, row.data_spw_id, row.observation_id)
126        });
127        let preapplied = preapplied_rows.and_then(|rows| rows.get(&row.row_index));
128        group.observe(ms, &row, preapplied, &options)?;
129    }
130
131    Ok(groups)
132}

134pub(crate) fn all_antenna_ids(ms: &MeasurementSet) -> Result<BTreeSet<i32>, GainSolveError> {
135    let antenna = ms
136        .antenna()
137        .map_err(|source| GainSolveError::OpenMeasurementSet {
138            path: ms
139                .path()
140                .map(|path| path.display().to_string())
141                .unwrap_or_else(|| "<in-memory>".to_string()),
142            source,
143        })?;
144    Ok((0..antenna.row_count())
145        .map(|row| i32::try_from(row).expect("antenna row count should fit in i32"))
146        .collect())

149pub(crate) fn resolve_refant(
150    ms: &MeasurementSet,
151    selector: &RefAntSelector,
152) -> Result<i32, GainSolveError> {
153    match selector {
154        RefAntSelector::AntennaId(id) => Ok(*id),
155        RefAntSelector::AntennaName(name) => {
156            let antenna = ms
157                .antenna()
158                .map_err(|error| GainSolveError::ResolveRefAnt {
159                    selector: name.clone(),
160                    reason: error.to_string(),
161                })?;
162            for row in 0..antenna.row_count() {
163                if antenna
164                    .name(row)
165                    .map_err(|error| GainSolveError::ResolveRefAnt {
166                        selector: name.clone(),
167                        reason: error.to_string(),
168                    })?
169                    == *name
170                {
171                    return Ok(row as i32);
172                }
173            }
174            Err(GainSolveError::ResolveRefAnt {
175                selector: name.clone(),
176                reason: "no ANTENNA.NAME match".to_string(),
177            })
178        }
179    }
180}

182pub(crate) fn collect_selected_rows(
183    ms: &MeasurementSet,
184    selection: &MsSelection,
185) -> Result<Vec<SelectedSolveRow>, GainSolveError> {
186    let selected_rows = resolve_calibration_selection(ms, selection).map_err(|error| {
187        GainSolveError::OpenMeasurementSet {
188            path: ms
189                .path()
190                .map(|path| path.display().to_string())
191                .unwrap_or_else(|| "<in-memory>".to_string()),
192            source: match error {
193                casa_ms::MsSelectionError::Domain(source) => source,
194                other => MsError::InvalidInput(other.to_string()),
195            },
196        }
197    })?;
198    if selected_rows.selected_rows.is_empty() {
199        return Err(GainSolveError::EmptySelection);
200    }
201
202    selected_rows
203        .selected_rows
204        .into_iter()
205        .map(|row| {
206            Ok(SelectedSolveRow {
207                row_index: row.row_index,
208                field_id: row.field_id,
209                observation_id: get_i32(ms.main_table(), row.row_index, "OBSERVATION_ID")?,
210                data_desc_id: row.data_desc_id,
211                data_spw_id: row.spectral_window_id,
212                channel_selection: selection.channel_selection_for_spw(row.spectral_window_id),
213                antenna1: row.antenna1_id,
214                antenna2: row.antenna2_id,
215                time_seconds: row.time_mjd_seconds,
216                time_centroid_seconds: get_f64(ms.main_table(), row.row_index, "TIME_CENTROID")?,
217                interval_seconds: get_f64(ms.main_table(), row.row_index, "INTERVAL")?,
218                scan_number: get_i32(ms.main_table(), row.row_index, "SCAN_NUMBER")?,
219                flag_row: row.flag_row,
220            })
221        })
222        .collect()
223}

225pub(crate) fn correlation_types_for_ddid(
226    ms: &MeasurementSet,
227    data_desc_id: i32,
228) -> Result<Vec<i32>, GainSolveError> {
229    let dd = ms
230        .data_description()
231        .map_err(|source| GainSolveError::OpenMeasurementSet {
232            path: ms
233                .path()
234                .map(|path| path.display().to_string())
235                .unwrap_or_else(|| "<in-memory>".to_string()),
236            source,
237        })?;
238    let pol = ms
239        .polarization()
240        .map_err(|source| GainSolveError::OpenMeasurementSet {
241            path: ms
242                .path()
243                .map(|path| path.display().to_string())
244                .unwrap_or_else(|| "<in-memory>".to_string()),
245            source,
246        })?;
247    let row = usize::try_from(data_desc_id).map_err(|_| {
248        GainSolveError::UnsupportedCorrelationLayout {
249            data_desc_id,
250            correlation_types: Vec::new(),
251        }
252    })?;
253    let pol_id = dd
254        .polarization_id(row)
255        .map_err(|source| GainSolveError::OpenMeasurementSet {
256            path: ms
257                .path()
258                .map(|path| path.display().to_string())
259                .unwrap_or_else(|| "<in-memory>".to_string()),
260            source,
261        })?;
262    pol.corr_type(usize::try_from(pol_id).unwrap_or(usize::MAX))
263        .map_err(|source| GainSolveError::OpenMeasurementSet {
264            path: ms
265                .path()
266                .map(|path| path.display().to_string())
267                .unwrap_or_else(|| "<in-memory>".to_string()),
268            source,
269        })
270}

457impl SolveAccumulator {
458    pub(crate) fn new(field_id: i32, spw_id: i32, observation_id: i32) -> Self {
459        Self {
460            field_id,
461            spw_id,
462            observation_id,
463            min_time: f64::INFINITY,
464            max_time: f64::NEG_INFINITY,
465            unique_time_sum: 0.0,
466            unique_time_count: 0,
467            total_interval: 0.0,
468            sample_rows: 0,
469            field_ids: BTreeSet::new(),
470            scan_numbers: BTreeSet::new(),
471            antenna_ids: BTreeSet::new(),
472            unique_time_bits: BTreeSet::new(),
473            solution_channels: 1,
474            channel_graphs: Vec::new(),
475            solution_time: 0.0,
476            attempted: false,
477            baseline_centroids: HashMap::new(),
478            receptor_accumulations: Vec::new(),
479        }
480    }
481
482    /// Add one MAIN row: its unflagged, positively weighted samples divided by
483    /// the model, each weighted by its solve weight times |model|² as
484    /// `VisEquation::divideCorrByModel` does, and its TIME_CENTROID.
485    pub(crate) fn observe(
486        &mut self,
487        ms: &MeasurementSet,
488        row: &SelectedSolveRow,
489        preapplied_row: Option<&EvaluatedApplyRow>,
490        options: &SolveGroupOptions,
491    ) -> Result<(), GainSolveError> {
492        let inputs = read_row_inputs(ms, row, preapplied_row, options.model_source)?;
493        let correlation_types = correlation_types_for_ddid(ms, row.data_desc_id)?;
494        let channel_count = inputs.data.shape()[1];
495        let channels = if options.channelized {
496            (0..channel_count).collect()
497        } else {
498            selected_channels(row.channel_selection.as_ref(), channel_count)?
499        };
500        self.observe_samples(
501            &SolveSampleRow::of(row),
502            &inputs.samples(),
503            &correlation_types,
504            &channels,
505            options,
506        )
507    }
508
509    /// Accumulate one sample's `channels` (every channel for a channelized
510    /// solve): the arithmetic of [`Self::observe`] on arrays from any source.
511    /// The MSv2/MSv4 trial (`docs/measurement-set/trial/README.md`) feeds it
512    /// from `casa_observation::VisibilityBatch` views without copying them.
513    pub(crate) fn observe_samples(
514        &mut self,
515        row: &SolveSampleRow,
516        samples: &SolveSamples<'_>,
517        correlation_types: &[i32],
518        channels: &[usize],
519        options: &SolveGroupOptions,
520    ) -> Result<(), GainSolveError> {
521        if options.channelized {
522            let channel_count = samples.data.shape()[1];
523            if self.sample_rows > 0 && self.solution_channels != channel_count {
524                return Err(GainSolveError::UnsupportedParameterShape {
525                    path: "<measurement-set DATA>".to_string(),
526                    shape: samples.data.shape().to_vec(),
527                });
528            }
529            self.solution_channels = channel_count;
530        }
531
532        let graph_count = match options.gain_type {
533            GainType::G => 2,
534            GainType::T => 1,
535        };
536        if self.receptor_accumulations.len() < graph_count {
537            self.receptor_accumulations
538                .resize_with(graph_count, HashMap::new);
539        }
540        self.observe_metadata(row);
541
542        let mut row_weight = 0.0_f64;
543        for (corr_index, &correlation_type) in correlation_types.iter().enumerate() {
544            let Some(receptors) = correlation_receptors(correlation_type) else {
545                return Err(GainSolveError::UnsupportedCorrelationLayout {
546                    data_desc_id: row.data_desc_id,
547                    correlation_types: correlation_types
548                        .iter()
549                        .map(|code| stokes_name(*code).to_string())
550                        .collect(),
551                });
552            };
553            if receptors.0 != receptors.1 {
554                continue;
555            }
556            let graph_index = match options.gain_type {
557                GainType::G => receptors.0,
558                GainType::T => 0,
559            };
560            let mut channel_weights = Vec::with_capacity(channels.len());
561            for &chan_index in channels {
562                let weight = self.accumulate_sample(
563                    graph_index,
564                    row,
565                    samples,
566                    options,
567                    corr_index,
568                    chan_index,
569                );
570                channel_weights.push(weight);
571            }
572            row_weight += if options.channelized {
573                f64::from(median_with_even_mean(&mut channel_weights))
574            } else {
575                channel_weights
576                    .iter()
577                    .map(|weight| f64::from(*weight))
578                    .sum()
579            };
580        }
581        if row_weight > 0.0 {
582            let centroid = self
583                .baseline_centroids
584                .entry((row.antenna1, row.antenna2))
585                .or_default();
586            centroid.weighted_sum += row_weight * row.time_centroid_seconds;
587            centroid.weight += row_weight;
588        }
589        Ok(())
590    }
591
592    fn observe_metadata(&mut self, row: &SolveSampleRow) {
593        self.min_time = self.min_time.min(row.time_seconds);
594        self.max_time = self.max_time.max(row.time_seconds);
595        if self.unique_time_bits.insert(row.time_seconds.to_bits()) {
596            self.unique_time_sum += row.time_seconds;
597            self.unique_time_count += 1;
598        }
599        self.total_interval += row.interval_seconds;
600        self.sample_rows += 1;
601        self.field_ids.insert(row.field_id);
602        self.scan_numbers.insert(row.scan_number);
603        self.antenna_ids.insert(row.antenna1);
604        self.antenna_ids.insert(row.antenna2);
605    }
606
607    /// Accumulate one sample if it can enter the solve; return its effective
608    /// weight, zero when it is flagged or unusable.
609    fn accumulate_sample(
610        &mut self,
611        graph_index: usize,
612        row: &SolveSampleRow,
613        samples: &SolveSamples<'_>,
614        options: &SolveGroupOptions,
615        corr_index: usize,
616        chan_index: usize,
617    ) -> f32 {
618        let weight = samples.weights[[corr_index, chan_index]];
619        if weight <= 0.0 || solve_sample_flagged(&samples.flags, options.gain_type, chan_index) {
620            return 0.0;
621        }
622        let model = match samples.model {
623            Some(model_data) => model_data[[corr_index, chan_index]],
624            None => Complex32::new(options.stokes_i, 0.0),
625        };
626        let model_norm_sqr = model.norm_sqr();
627        if model_norm_sqr <= f32::EPSILON {
628            return 0.0;
629        }
630        self.attempted = true;
631        let sample = samples.data[[corr_index, chan_index]] / model;
632        if sample.norm() <= f32::EPSILON {
633            return 0.0;
634        }
635        let effective_weight = weight * model_norm_sqr;
636        let accumulation = self.receptor_accumulations[graph_index]
637            .entry(SolveAccumulationKey {
638                baseline: (row.antenna1, row.antenna2),
639                correlation_index: corr_index,
640                channel_index: if options.channelized { chan_index } else { 0 },
641            })
642            .or_default();
643        accumulation.weighted_sum += sample * Complex32::new(effective_weight, 0.0);
644        accumulation.weight += effective_weight;
645        accumulation.weighted_sample_power +=
646            f64::from(effective_weight) * f64::from(sample.norm_sqr());
647        accumulation.sample_count += 1;
648        effective_weight
649    }
650
651    /// Collapse each baseline's samples per correlation and solution channel
652    /// into the solve graphs, and set the solution TIME.
653    ///
654    /// A phase-only solve sees each collapsed sample at unit amplitude with
655    /// its weight times its squared amplitude
656    /// (`SolveDataBuffer::enforceAPonData`). TIME is
657    /// `SDBList::aggregateTimeCentroid`: each baseline's TIME_CENTROID,
658    /// averaged over its rows by row weight as time averaging does, weighted
659    /// by the baseline's solve weight after that phase-only scaling.
660    pub(crate) fn finalize_for_solve(&mut self, solve_mode: GainSolveMode) {
661        self.channel_graphs = vec![
662            vec![SolveGraph::default(); self.receptor_accumulations.len()];
663            self.solution_channels
664        ];
665        let mut baseline_weights = HashMap::<(i32, i32), f64>::new();
666        for (receptor, accumulations) in self.receptor_accumulations.iter().enumerate() {
667            for (key, accumulation) in accumulations {
668                if accumulation.weight <= f32::EPSILON {
669                    continue;
670                }
671                let averaged = accumulation.weighted_sum / Complex32::new(accumulation.weight, 0.0);
672                let (sample, weight) = match solve_mode {
673                    GainSolveMode::Phase => {
674                        let amplitude = averaged.norm();
675                        if amplitude <= f32::EPSILON {
676                            continue;
677                        }
678                        (
679                            averaged / Complex32::new(amplitude, 0.0),
680                            accumulation.weight * amplitude * amplitude,
681                        )
682                    }
683                    GainSolveMode::AmplitudePhase => (averaged, accumulation.weight),
684                };
685                self.channel_graphs[key.channel_index][receptor].accumulate(
686                    key.baseline,
687                    weight,
688                    sample * Complex32::new(weight, 0.0),
689                    SolveEdgeStats {
690                        weighted_sample_power: f64::from(weight) * f64::from(sample.norm_sqr()),
691                        raw_weighted_sample_power: accumulation.weighted_sample_power,
692                        sample_count: accumulation.sample_count,
693                        collapsed_count: 1,
694                    },
695                );
696                *baseline_weights.entry(key.baseline).or_default() += f64::from(weight);
697            }
698        }
699
700        let mut solution_time = WeightedTime::default();
701        for (baseline, weight) in baseline_weights {
702            let Some(centroid) = self.baseline_centroids.get(&baseline) else {
703                continue;
704            };
705            if centroid.weight > 0.0 {
706                solution_time.weighted_sum += weight * centroid.weighted_sum / centroid.weight;
707                solution_time.weight += weight;
708            }
709        }
710        self.solution_time = if solution_time.weight > 0.0 {
711            solution_time.weighted_sum / solution_time.weight
712        } else {
713            self.aggregate_time()
714        };
715    }
716
717    /// Add `other`'s accumulation, of the same group, to this one before
718    /// [`Self::finalize_for_solve`], as if its samples had been observed
719    /// here: the fragment work units of the MSv2/MSv4 trial accumulate apart
720    /// and merge before the group closes. Sums change order, so results
721    /// agree to rounding, not bitwise.
722    #[cfg(test)]
723    pub(crate) fn merge(&mut self, other: Self) {
724        self.min_time = self.min_time.min(other.min_time);
725        self.max_time = self.max_time.max(other.max_time);
726        for bits in other.unique_time_bits {
727            if self.unique_time_bits.insert(bits) {
728                self.unique_time_sum += f64::from_bits(bits);
729                self.unique_time_count += 1;
730            }
731        }
732        self.total_interval += other.total_interval;
733        self.sample_rows += other.sample_rows;
734        self.field_ids.extend(other.field_ids);
735        self.scan_numbers.extend(other.scan_numbers);
736        self.antenna_ids.extend(other.antenna_ids);
737        self.solution_channels = self.solution_channels.max(other.solution_channels);
738        self.attempted |= other.attempted;
739        for (baseline, centroid) in other.baseline_centroids {
740            let mine = self.baseline_centroids.entry(baseline).or_default();
741            mine.weighted_sum += centroid.weighted_sum;
742            mine.weight += centroid.weight;
743        }
744        if self.receptor_accumulations.len() < other.receptor_accumulations.len() {
745            self.receptor_accumulations
746                .resize_with(other.receptor_accumulations.len(), HashMap::new);
747        }
748        for (mine, theirs) in self
749            .receptor_accumulations
750            .iter_mut()
751            .zip(other.receptor_accumulations)
752        {
753            for (key, accumulation) in theirs {
754                let entry = mine.entry(key).or_default();
755                entry.weighted_sum += accumulation.weighted_sum;
756                entry.weight += accumulation.weight;
757                entry.weighted_sample_power += accumulation.weighted_sample_power;
758                entry.sample_count += accumulation.sample_count;
759            }
760        }
761    }
762
763    /// Entries the accumulation's sets and maps hold: its state grows with
764    /// the group's distinct times, antennas, baselines and baseline
765    /// correlations, not with its samples.
766    #[cfg(test)]
767    pub(crate) fn state_entries(&self) -> usize {
768        self.unique_time_bits.len()
769            + self.scan_numbers.len()
770            + self.antenna_ids.len()
771            + self.baseline_centroids.len()
772            + self
773                .receptor_accumulations
774                .iter()
775                .map(HashMap::len)
776                .sum::<usize>()
777    }
778
779    /// The mean of the interval's distinct TIMEs (`SDBList::aggregateTime`,
780    /// CASA's TIME when no sample carries weight).
781    pub(crate) fn aggregate_time(&self) -> f64 {
782        if self.unique_time_count > 0 {
783            self.unique_time_sum / self.unique_time_count as f64
784        } else {
785            (self.min_time + self.max_time) / 2.0
786        }
787    }
788}

790/// Read one MAIN row's data, flags, model and solve weights, from the row or
791/// from its pre-applied calibration.
792///
793/// The weights are CASA's (`CalSolvingVi2::calibrateCurrentVB`, which calls
794/// `VisBufferImpl2::resetWeightsUsingSigma`): 1/σ² from SIGMA_SPECTRUM when
795/// the MeasurementSet has WEIGHT_SPECTRUM and SIGMA_SPECTRUM, otherwise from
796/// SIGMA broadcast over channels; σ ≤ 0 gives no weight. WEIGHT and
797/// WEIGHT_SPECTRUM are not read. Pre-applied `calwt` tables then scale them.
798fn read_row_inputs<'a>(
799    ms: &'a MeasurementSet,
800    row: &SelectedSolveRow,
801    preapplied_row: Option<&'a EvaluatedApplyRow>,
802    model_source: GainSolveModelSource,
803) -> Result<SolveRowInputs<'a>, GainSolveError> {
804    let (data, flags) = match preapplied_row {
805        Some(preapplied) => (&preapplied.corrected_data, &preapplied.flags),
806        None => (
807            main_cell(ms, row.row_index, "DATA")?,
808            main_cell(ms, row.row_index, "FLAG")?,
809        ),
810    };
811    let ArrayValue::Complex32(data) = data else {
812        return Err(unsupported_shape("DATA", data));
813    };
814    let ArrayValue::Bool(flags) = flags else {
815        return Err(unsupported_shape("FLAG", flags));
816    };
817    if data.ndim() != 2 || data.shape() != flags.shape() {
818        return Err(GainSolveError::UnsupportedParameterShape {
819            path: "<measurement-set row>".to_string(),
820            shape: data.shape().to_vec(),
821        });
822    }
823    let model = match model_source {
824        GainSolveModelSource::ModelColumn => {
825            let model = main_cell(ms, row.row_index, "MODEL_DATA")?;
826            match model {
827                ArrayValue::Complex32(model) if model.shape() == data.shape() => Some(model),
828                other => return Err(unsupported_shape("MODEL_DATA", other)),
829            }
830        }
831        GainSolveModelSource::PointSource => None,
832    };
833    let mut weights = solve_weights(ms, row.row_index, data.shape())?;
834    if let Some(factors) = preapplied_row.and_then(|row| row.weight_factors.as_ref()) {
835        if factors.shape() != weights.shape() {
836            return Err(GainSolveError::UnsupportedParameterShape {
837                path: "<pre-applied weight factors>".to_string(),
838                shape: factors.shape().to_vec(),
839            });
840        }
841        weights *= factors;
842    }
843    Ok(SolveRowInputs {
844        data,
845        flags: solve_flags(row, flags),
846        weights,
847        model,
848    })
849}
850
851fn solve_weights(
852    ms: &MeasurementSet,
853    row_index: usize,
854    shape: &[usize],
855) -> Result<ArrayD<f32>, GainSolveError> {
856    let has_column = |name| {
857        ms.main_table()
858            .schema()
859            .is_some_and(|schema| schema.contains_column(name))
860    };
861    let inverse_square = |sigma: f32| {
862        if sigma > 0.0 {
863            1.0 / (sigma * sigma)
864        } else {
865            0.0
866        }
867    };
868    if has_column("WEIGHT_SPECTRUM") && has_column("SIGMA_SPECTRUM") {
869        let sigma = main_cell(ms, row_index, "SIGMA_SPECTRUM")?;
870        return match sigma {
871            ArrayValue::Float32(sigma) if sigma.shape() == shape => Ok(sigma.mapv(inverse_square)),
872            other => Err(unsupported_shape("SIGMA_SPECTRUM", other)),
873        };
874    }
875    let sigma = main_cell(ms, row_index, "SIGMA")?;
876    let ArrayValue::Float32(sigma) = sigma else {
877        return Err(unsupported_shape("SIGMA", sigma));
878    };
879    if sigma.ndim() != 1 || sigma.shape()[0] != shape[0] {
880        return Err(GainSolveError::UnsupportedParameterShape {
881            path: "<measurement-set SIGMA>".to_string(),
882            shape: sigma.shape().to_vec(),
883        });
884    }
885    Ok(ArrayD::from_shape_fn(shape, |index| {
886        inverse_square(sigma[[index[0]]])
887    }))
888}
889
890fn main_cell<'a>(
891    ms: &'a MeasurementSet,
892    row_index: usize,
893    column: &str,
894) -> Result<&'a ArrayValue, GainSolveError> {
895    ms.main_table()
896        .cell_accessor(row_index, column)
897        .and_then(|cell| cell.array())
898        .map_err(|source| GainSolveError::OpenMeasurementSet {
899            path: ms
900                .path()
901                .map(|path| path.display().to_string())
902                .unwrap_or_else(|| "<in-memory>".to_string()),
903            source: MsError::from(source),
904        })
905}
906
907fn unsupported_shape(column: &str, value: &ArrayValue) -> GainSolveError {
908    GainSolveError::UnsupportedParameterShape {
909        path: format!("<measurement-set {column}>"),
910        shape: value.shape().to_vec(),
911    }
912}

914/// The median, the mean of the middle two for an even count, as casacore's
915/// `partialMedians(..., takeEvenMean=True)` that CASA uses for a row's
916/// channel-independent weight.
917fn median_with_even_mean(values: &mut [f32]) -> f32 {
918    if values.is_empty() {
919        return 0.0;
920    }
921    values.sort_by(f32::total_cmp);
922    let middle = values.len() / 2;
923    if values.len().is_multiple_of(2) {
924        (values[middle - 1] + values[middle]) / 2.0
925    } else {
926        values[middle]
927    }
928}

930/// The channels of a window of `channel_count` channels that a solve
931/// uses: those `selection` names, else all. The gain table's
932/// `SPECTRAL_WINDOW` row collapses the same channels.
933pub(crate) fn selected_channels(
934    selection: Option<&ChannelSelection>,
935    channel_count: usize,
936) -> Result<Vec<usize>, GainSolveError> {
937    match selection {
938        Some(selection) => {
939            selection
940                .indices(channel_count)
941                .map_err(|source| GainSolveError::OpenMeasurementSet {
942                    path: "<measurement-set channel selection>".to_string(),
943                    source,
944                })
945        }
946        None => Ok((0..channel_count).collect()),
947    }
948}

950/// The sample flags a solve uses for `row`: its `FLAG` cell, or every sample
951/// flagged when the row's `FLAG_ROW` is set, whatever `FLAG` holds. CASA's
952/// solve does the same: `CalSolvingVi2::calibrateCurrentVB` calls
953/// `VisEquation::collapse2`, whose `divideCorrByModel` flags the whole cube
954/// of such a row and zeroes its weights before G, T or B accumulates it.
955pub(crate) fn solve_flags<'a>(
956    row: &SelectedSolveRow,
957    flags: &'a ArrayD<bool>,
958) -> Cow<'a, ArrayD<bool>> {
959    if row.flag_row {
960        Cow::Owned(ArrayD::from_elem(flags.raw_dim(), true))
961    } else {
962        Cow::Borrowed(flags)
963    }
964}

966/// Correlation-independent flags, as `CalSolvingVi2::corrIndepFlags` makes
967/// them: a channel any correlation flags is flagged for all.
968fn solve_sample_flagged(
969    flags: &ArrayView2<'_, bool>,
970    gain_type: GainType,
971    chan_index: usize,
972) -> bool {
973    match gain_type {
974        GainType::G | GainType::T => {
975            (0..flags.shape()[0]).any(|corr_index| flags[[corr_index, chan_index]])
976        }
977    }
