2//! CASA-style display formatting of world coordinates.
3//!
4//! casacore's `CoordinateSystem::format` chain renders each world axis
5//! for display, and the statistics surfaces join one field per world
6//! axis with `", "`. This module reproduces that rendering for the
7//! forms those surfaces reach:
8//!
9//! - equatorial directions as sexagesimal coordinates and Galactic,
10//!   ecliptic, and horizontal directions as fixed decimal degrees,
11//! - Stokes and Quality planes by name,
12//! - every other axis as C++'s default `ostream <<` rendering of the
13//!   world value with its unit appended directly.
14
15use casa_core::measures::direction::DirectionRef;

18use crate::coordinate_system::CoordinateSystem;
19use crate::error::CoordinateError;

21/// Format the world coordinates of `pixel` the way CASA's statistics
22/// surfaces render them: one field per world axis, comma-separated.
23///
24/// The pixel vector has one value per pixel axis, in axis order. The
25/// returned string is casacore's `CoordinateSystem::format` rendering
26/// of every world axis, joined with `", "` as `ia.statistics` builds
27/// its `blcf`, `trcf`, `minposf` and `maxposf` strings.
28///
29/// # Errors
30///
31/// A pixel the projection cannot place or whose length is not the
32/// coordinate system's pixel-axis count.
33///
34/// # Examples
35///
36/// ```
37/// use casa_coordinates::format_world_display;
38/// use casa_coordinates::{
39///     DirectionCoordinate, Projection, ProjectionType, StokesCoordinate, StokesType,
40/// };
41/// use casa_core::measures::direction::DirectionRef;
42///
43/// let arcmin = std::f64::consts::PI / 180.0 / 60.0;
44/// let mut coords = casa_coordinates::CoordinateSystem::new();
45/// coords.add_coordinate(
46///     DirectionCoordinate::new(
47///         DirectionRef::J2000,
48///         Projection::new(ProjectionType::SIN),
49///         [0.0, 0.0],
50///         [-arcmin, arcmin],
51///         [4.0, 4.0],
52///     )
53///     .with_world_units(["'", "'"])
54///     .expect("arcminutes"),
55/// );
56/// coords.add_coordinate(StokesCoordinate::new(vec![StokesType::I]));
57/// let text = format_world_display(&coords, &[0.0, 0.0, 0.0]).expect("inside the projection");
58/// assert_eq!(text, "00:00:16.000, -00.04.00.000, I");
59/// ```
60pub fn format_world_display(
61    coords: &CoordinateSystem,
62    pixel: &[f64],
63) -> Result<String, CoordinateError> {
64    let world = coords.to_world(pixel)?;
65    let fields: Vec<_> = world
66        .into_iter()
67        .enumerate()
68        .map(|(axis, value)| {
69            let (index, local) = coords.world_axis_owner(axis).expect("public world axis");
70            let coord = coords.coordinate(index);
71            match coord {
72                CoordinateModel::Direction(direction) => direction_field(
73                    direction.direction_ref(),
74                    direction.longitude_axis() == Some(local),
75                    value,
76                ),
77                CoordinateModel::Stokes(_) => {
78                    let code = value as i32;
79                    crate::stokes::StokesType::from_code(code)
80                        .map_or_else(|| code.to_string(), |kind| kind.name().into())
81                }
82                CoordinateModel::Quality(_) => crate::QualityType::from_world(value).name().into(),
83                _ => format!(
84                    "{}{}",
85                    format_stream_value(value),
86                    coord.axis_units()[local]
87                ),
88            }
89        })
90        .collect();
91    Ok(fields.join(", "))
92}

94fn direction_field(frame: DirectionRef, longitude: bool, radians: f64) -> String {
95    let signed = matches!(
96        frame,
97        DirectionRef::ECLIPTIC | DirectionRef::MECLIPTIC | DirectionRef::TECLIPTIC
98    );
99    let positive = matches!(
100        frame,
101        DirectionRef::GALACTIC
102            | DirectionRef::SUPERGAL
103            | DirectionRef::AZEL
104            | DirectionRef::AZELSW
105            | DirectionRef::AZELGEO
106            | DirectionRef::AZELSWGEO
107    );
108    if !(signed || positive) {
109        return if longitude {
110            format_longitude_hours(radians)
111        } else {
112            format_latitude_dms(radians)
113        };
114    }
115    let degrees = radians.to_degrees();
116    let wrapped = if !longitude {
117        degrees
118    } else if signed {
119        (degrees + 180.0).rem_euclid(360.0) - 180.0
120    } else {
121        degrees.rem_euclid(360.0)
122    };
123    format!("{wrapped:.6}deg")
124}

126/// Format a longitude in radians as hours `HH:MM:SS.sss`, wrapped into
127/// one revolution, with the seconds rounded to three decimals.
128pub fn format_longitude_hours(radians: f64) -> String {
129    format_longitude_hours_with_precision(radians, 3)
130}
131
132/// Format a longitude in hours, with the requested number of second decimals.
133pub fn format_longitude_hours_with_precision(radians: f64, decimals: usize) -> String {
134    let hours = (radians * 12.0 / std::f64::consts::PI).rem_euclid(24.0);
135    let mut hour = hours.floor() as u32;
136    let mut minute = ((hours - hour as f64) * 60.0).floor() as u32;
137    let seconds = (hours - hour as f64) * 3600.0 - minute as f64 * 60.0;
138    let scale = 10.0_f64.powi(decimals as i32);
139    let seconds = (seconds * scale).round() / scale;
140    if seconds >= 60.0 {
141        minute += 1;
142        if minute >= 60 {
143            minute = 0;
144            hour = (hour + 1) % 24;
145        }
146        return format!(
147            "{hour:02}:{minute:02}:{:0width$.decimals$}",
148            0.0,
149            width = decimals + 3
150        );
151    }
152    format!(
153        "{hour:02}:{minute:02}:{seconds:0width$.decimals$}",
154        width = decimals + 3
155    )
156}

158/// Format a latitude in radians as signed `DDD.MM.SS.sss` with the
159/// seconds rounded to three decimals and the sign carried at zero
160/// degrees, as casacore's ANGLE rendering gives it.
161pub fn format_latitude_dms(radians: f64) -> String {
162    format_latitude_dms_with_precision(radians, 3)
163}
164
165/// Format a signed angle in degrees, with the requested number of second decimals.
166pub fn format_latitude_dms_with_precision(radians: f64, decimals: usize) -> String {
167    let degrees = radians * 180.0 / std::f64::consts::PI;
168    let sign = if degrees.is_sign_negative() { "-" } else { "+" };
169    let absolute = degrees.abs();
170    let mut degree = absolute.floor() as u32;
171    let mut minute = ((absolute - degree as f64) * 60.0).floor() as u32;
172    let seconds = (absolute - degree as f64) * 3600.0 - minute as f64 * 60.0;
173    let scale = 10.0_f64.powi(decimals as i32);
174    let seconds = (seconds * scale).round() / scale;
175    if seconds >= 60.0 {
176        minute += 1;
177        if minute >= 60 {
178            minute = 0;
179            degree += 1;
180        }
181        return format!(
182            "{sign}{degree:02}.{minute:02}.{:0width$.decimals$}",
183            0.0,
184            width = decimals + 3
185        );
186    }
187    format!(
188        "{sign}{degree:02}.{minute:02}.{seconds:0width$.decimals$}",
189        width = decimals + 3
190    )
191}

193/// Format a world value as C++'s default `ostream <<` rendering: `%g`
194/// with six significant digits, trailing zeros stripped.
195pub fn format_stream_value(value: f64) -> String {
196    format_stream_value_with_precision(value, 6)
197}
198
199/// Format a world value with the given number of significant digits.
200pub fn format_stream_value_with_precision(value: f64, precision: usize) -> String {
201    if value == 0.0 {
202        return "0".to_string();
203    }
204    if !value.is_finite() {
205        return if value.is_nan() {
206            "nan".to_string()
207        } else if value > 0.0 {
208            "inf".to_string()
209        } else {
210            "-inf".to_string()
211        };
212    }
213
214    // In scientific form the mantissa carries
215    // the digits, and the exponent decides %g's fixed-versus-scientific
216    // choice.
217    let precision = precision.max(1);
218    let fractional = precision - 1;
219    let scientific = format!("{value:.fractional$e}");
220    let (mantissa, exponent) = scientific
221        .split_once('e')
222        .expect("scientific formatting carries an exponent");
223    let exponent: i32 = exponent.parse().expect("a decimal exponent");
224    let digits: String = mantissa.chars().filter(|c| c.is_ascii_digit()).collect();
225    let sign = if mantissa.starts_with('-') { "-" } else { "" };
226
227    if !(-4..precision as i32).contains(&exponent) {
228        let mut text = format!("{}.{}", &digits[..1], &digits[1..]);
229        while text.ends_with('0') {
230            text.pop();
231        }
232        if text.ends_with('.') {
233            text.pop();
234        }
235        let sign_char = if exponent < 0 { '-' } else { '+' };
236        let magnitude = exponent.unsigned_abs();
237        format!("{sign}{text}e{sign_char}{magnitude:02}")
238    } else {
239        let decimals = (fractional as i32 - exponent).max(0) as usize;
240        let mut text = format!("{:.decimals$}", value.abs());
241        if text.contains('.') {
242            while text.ends_with('0') {
243                text.pop();
244            }
245            if text.ends_with('.') {
246                text.pop();
247            }
248        }
249        format!("{sign}{text}")
250    }
251}

253#[cfg(test)]
254mod tests {
255    use super::*;
256
257    #[test]
258    fn hours_wrap_and_round() {
259        let to_hours = |deg: f64| deg * std::f64::consts::PI / 180.0;
260        assert_eq!(format_longitude_hours(to_hours(0.0)), "00:00:00.000");
261        assert_eq!(
262            format_longitude_hours(to_hours(-1.0 / 60.0)),
263            "23:59:56.000"
264        );
265        assert_eq!(format_longitude_hours(to_hours(180.0)), "12:00:00.000");
266        // Four arcminutes of sky is sixteen seconds of time.
267        let arcmin = to_hours(1.0 / 60.0);
268        assert_eq!(format_longitude_hours(4.0 * arcmin), "00:00:16.000");
269        // Rounding past a whole second carries into the minute.
270        let nearly = to_hours(10.0) + 59.9996 * std::f64::consts::PI / 12.0 / 3600.0;
271        assert_eq!(format_longitude_hours(nearly), "00:41:00.000");
272    }
273
274    #[test]
275    fn latitude_carries_the_sign_at_every_magnitude() {
276        let to_deg = |deg: f64| deg * std::f64::consts::PI / 180.0;
277        assert_eq!(format_latitude_dms(to_deg(0.0)), "+00.00.00.000");
278        assert_eq!(format_latitude_dms(to_deg(-4.0 / 60.0)), "-00.04.00.000");
279        assert_eq!(format_latitude_dms(to_deg(-90.0)), "-90.00.00.000");
280        assert_eq!(format_latitude_dms(to_deg(45.5)), "+45.30.00.000");
281        assert_eq!(format_latitude_dms(to_deg(3.0 / 60.0)), "+00.03.00.000");
282        // Rounding past a whole second carries into the minute.
283        let nearly = to_deg(45.0) + 59.9996 * std::f64::consts::PI / 180.0 / 3600.0;
284        assert_eq!(format_latitude_dms(nearly), "+45.01.00.000");
285    }
