3//! Thin PyO3 `ia.continuumsub` wrapper delegating to [`casa_imageanalysis::tool::ops_transform::continuum_sub`].
4
5use casa_imageanalysis::tool::ops_transform::continuum_sub;
6use pyo3::prelude::*;
7
8use crate::convert::IntList;
9
10use super::PyImageTool;
11use super::handle::{external_open_check, raised};
12use super::subimage::{RegionArgument, require_bool};
13
14#[pymethods]
15impl PyImageTool {
16    /// Fit a polynomial to the chosen channels and return the residual image.
17    #[allow(clippy::too_many_arguments)]
18    #[pyo3(signature = (outline="", outcont="continuumsub.im", region=RegionArgument::default(), channels=IntList::of(&[-1]), pol="", fitorder=0, overwrite=false), text_signature = "($self, outline='', outcont='continuumsub.im', region={}, channels=[-1], pol='', fitorder=0, overwrite=False)")]
19    fn continuumsub(
20        &self,
21        py: Python<'_>,
22        outline: &str,
23        outcont: &str,
24        region: RegionArgument,
25        channels: IntList,
26        pol: &str,
27        fitorder: i64,
28        #[pyo3(from_py_with = require_bool)] overwrite: bool,
29    ) -> PyResult<Py<Self>> {
30        let product = py.detach(|| -> PyResult<_> {
31            let state = self.state();
32            let Some(attached) = state.image.as_ref() else {
33                return Ok(None);
34            };
35            let history_command = state.history.then(|| {
36                let chan_str = channels
37                    .0
38                    .iter()
39                    .map(i64::to_string)
40                    .collect::<Vec<_>>()
41                    .join(",");
42                let region_hist = region.history();
43                format!(
44                    "ia.continuumsub(outline={outline:?}, outcont={outcont:?}, region={region_hist}, channels=[{chan_str}], pol={pol:?}, fitorder={fitorder}, overwrite={overwrite})"
45                )
46            });
47            continuum_sub(
48                attached,
49                outline,
50                outcont,
51                region.selection(),
52                &channels.0,
53                fitorder,
54                overwrite,
55                history_command,
56                external_open_check(),
57            )
58            .map(Some)
59            .map_err(raised)
60        })?;
61        Py::new(py, product.map_or_else(Self::new, Self::attached))
62    }
63}
