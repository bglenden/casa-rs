17pub(super) fn run(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
18    let path = task.input_path("vis")?;
19    let root = OpenedMs::open(&path).map_err(execution)?;
20    let parts = partition_paths(&path)?
21        .unwrap_or_default()
22        .iter()
23        .map(|path| OpenedMs::open(path)?.initialize_calibrater(false, true, true))
24        .collect::<Result<Vec<_>, _>>()
25        .map_err(execution)?;
26    let root = if parts.is_empty() {
27        root.initialize_calibrater(false, true, true)
28            .map_err(execution)?
29    } else {
30        root.reload().map_err(execution)?
31    };
32    let ms = &root.measurement_set;
33    let selected = root.resolve(&selection(task)?).map_err(selection_error)?;
34    if selected.selected_rows.is_empty() {
35        return Err(selection_error(
36            "MSSelectionNullSelection : The selected table has zero rows.",
37        ));
38    }
39    let shape = VisibilityFitShape::parse(&task.text("comptype")?.unwrap_or_default())
40        .map_err(failed_fit)?;
41    let parameters = task.numbers("sourcepar")?;
42    let vary = task.flags("varypar")?;
43    let fitted = fit_visibility_component(
44        ms,
45        &selected,
46        VisibilityFitRequest {
47            shape,
48            initial: &parameters,
49            vary: &vary,
50            iterations: task.integer("niter")?,
51        },
52    )
53    .map_err(failed_fit)?;
54    if let Some(path) = task.text("outfile")?.filter(|path| !path.is_empty()) {
55        let list = ComponentList::from(vec![fitted.component]);
56        list.write_table(task.workspace().join(path))
57            .map_err(failed_fit)?;
58    }
59    for part in parts {
60        part.release().map_err(execution)?;
61    }
62    root.release().map_err(execution)?;
63    let mut report = TaskSuccess::new(CasaValue::None, &fitted.parameters)?
64        .with_output_line("Formal errors assume the visibility weights are correctly scaled.");
65    let labels = [
66        ("I", " Jy"),
67        ("x", " arcsec"),
68        ("y", " arcsec"),
69        ("a", " arcsec"),
70        ("r", ""),
71        ("p", " deg"),
72    ];
73    for ((value, error), (label, unit)) in fitted
74        .parameters
75        .into_iter()
76        .zip(fitted.formal_errors)
77        .zip(labels)
78    {
79        report = report.with_output_line(format!("{label} = {value:.9e} +/- {error:.9e}{unit}"));
80    }
81    Ok(report)
82}

84fn selection(task: &TaskInvocation<'_>) -> Result<MsSelection, TaskFailure> {
85    let select = task.flag("selectdata")?;
86    let text = |name| -> Result<Option<String>, TaskFailure> {
87        Ok(task.text(name)?.filter(|value| !value.is_empty()))
88    };
89    let optional = |name| {
90        if select { text(name) } else { Ok(None) }
91    };
92    Ok(MsSelection {
93        field: text("field")?,
94        spw: text("spw")?,
95        timerange: optional("timerange")?,
96        uvrange: select
97            .then(|| super::casa_read::uv_selector_text(task.value("uvrange")))
98            .flatten(),
99        antenna: optional("antenna")?,
100        scan: optional("scan")?,
101        msselect: optional("msselect")?,
102        ..MsSelection::default()
103    })
104}

106fn failed_fit(error: impl std::fmt::Display) -> TaskFailure {
107    tracing::error!("visibility model fit: {error}");
108    TaskFailure::execution("Error in Calibrater::modelfit.")
109}
110
111fn selection_error(error: impl std::fmt::Display) -> TaskFailure {
112    execution(format!("Error in data selection specification: {error}"))
113}
114
115fn execution(error: impl std::fmt::Display) -> TaskFailure {
116    TaskFailure::execution(error.to_string())
117}
