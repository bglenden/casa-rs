29pub(super) fn remap_associations(
30    mut row: RecordValue,
31    spw_map: &BTreeMap<i32, i32>,
32) -> RecordValue {
33    let Some(Value::Array(ArrayValue::Int32(ids))) = row.get("ASSOC_SPW_ID") else {
34        return row;
35    };
36    let selected = ids
37        .iter()
38        .enumerate()
39        .filter_map(|(index, id)| spw_map.get(id).map(|&id| (index, id)))
40        .collect::<Vec<_>>();
41    if let Some(Value::Array(ArrayValue::String(nature))) = row.get("ASSOC_NATURE") {
42        let nature = nature.iter().collect::<Vec<_>>();
43        let values = selected
44            .iter()
45            .map(|&(index, _)| {
46                nature
47                    .get(index)
48                    .map_or_else(String::new, |&value| value.clone())
49            })
50            .collect();
51        row.upsert(
52            "ASSOC_NATURE",
53            Value::Array(ArrayValue::from_string_vec(values)),
54        );
55    }
56    row.upsert(
57        "ASSOC_SPW_ID",
58        Value::Array(ArrayValue::from_i32_vec(
59            selected.into_iter().map(|(_, id)| id).collect(),
60        )),
61    );
62    row
63}

65pub(super) fn write_channel_selection(
66    output_ms: &Path,
67    compact_ids: &BTreeMap<i32, i32>,
68    selections: &BTreeMap<i32, TransformChannelSelection>,
69    averaged: Option<&BTreeMap<i32, Vec<ChannelWindow>>>,
70    split_widths: Option<&BTreeMap<i32, usize>>,
71) -> Result<(), MsTransformError> {
72    let path = output_ms.join(SubtableId::SpectralWindow.name());
73    let error = |source: casa_tables::TableError| MsTransformError::MutateMeasurementSet {
74        path: path.display().to_string(),
75        source: Box::new(source),
76    };
77    let mut table = Table::open(TableOptions::new(&path)).map_err(error)?;
78    let mut changed = false;
79    for (old_id, &new_id) in compact_ids {
80        let row = new_id as usize;
81        let selection = &selections[old_id];
82        let freq = f64_vector_cell(&table, row, "CHAN_FREQ", &path)?;
83        let averaged = averaged.map(|windows| windows[old_id].as_slice());
84        let split_width = split_widths.map(|widths| widths[old_id]);
85        if all_channels_selected(&selection.channels, freq.len())
86            && averaged.is_none_or(|bins| bins.len() == freq.len())
87            && split_width.is_none_or(|width| width == 1 || freq.len() == 1)
88        {
89            continue;
90        }
91        let widths = f64_vector_cell(&table, row, "CHAN_WIDTH", &path)?;
92        let resolution = f64_vector_cell(&table, row, "RESOLUTION", &path)?;
93        let planned;
94        let windows = if let Some(width) = split_width {
95            planned = split_windows(selection, width);
96            &planned
97        } else if let Some(windows) = averaged {
98            windows
99        } else {
100            planned = selection_windows(selection);
101            &planned
102        };
103        let bad_axis = || MsTransformError::SpectralMetadata {
104            path: path.display().to_string(),
105            reason: format!("SPW {old_id} channel metadata does not cover the selection"),
106        };
107        let mut grid =
108            selected_windows(windows, &freq, &widths, &resolution).ok_or_else(bad_axis)?;
109        let effective_bandwidth = if let Some(width) = split_width {
110            let input = f64_vector_cell(&table, row, "EFFECTIVE_BW", &path)?;
111            let stride = selection.ranges[0].step.max(1) as usize;
112            let bandwidths =
113                split_effective_bandwidths(windows, &input, stride).ok_or_else(bad_axis)?;
114            grid.bandwidth = bandwidths.iter().sum();
115            if width == 1 && stride == 1 {
116                for (width, window) in grid.widths.iter_mut().zip(windows) {
117                    *width = widths[window.first];
118                }
119            }
120            bandwidths
121        } else {
122            grid.resolution.clone()
123        };
124        let channel_count = grid.centres.len();
125        for (name, values) in [
126            ("CHAN_FREQ", grid.centres),
127            ("CHAN_WIDTH", grid.widths),
128            ("EFFECTIVE_BW", effective_bandwidth),
129            ("RESOLUTION", grid.resolution),
130        ] {
131            let value = Value::Array(ArrayValue::Float64(Array1::from_vec(values).into_dyn()));
132            table
133                .column_accessor_mut(name)
134                .and_then(|mut column| column.set(row, value))
135                .map_err(error)?;
136        }
137        for (name, value) in [
138            ("NUM_CHAN", ScalarValue::Int32(channel_count as i32)),
139            ("REF_FREQUENCY", ScalarValue::Float64(grid.reference)),
140            ("TOTAL_BANDWIDTH", ScalarValue::Float64(grid.bandwidth)),
141        ] {
142            table
143                .column_accessor_mut(name)
144                .and_then(|mut column| column.set(row, Value::Scalar(value)))
145                .map_err(error)?;
146        }
147        changed = true;
148    }
149    if changed {
150        table
151            .prepare_write()
152            .save(TableOptions::new(&path))
153            .map_err(error)?;
154    }
155    Ok(())
156}

158fn split_windows(selection: &TransformChannelSelection, width: usize) -> Vec<ChannelWindow> {
159    let mut windows = selection.metadata_for_width(width);
160    for window in &mut windows {
161        window.dropped = false;
162    }
163    windows
164}
165
166fn split_effective_bandwidths(
167    windows: &[ChannelWindow],
168    input: &[f64],
169    stride: usize,
170) -> Option<Vec<f64>> {
171    windows
172        .iter()
173        .map(|window| {
174            (window.first..=window.last)
175                .step_by(stride)
176                .try_fold(0.0, |sum, channel| Some(sum + input.get(channel)?))
177        })
178        .collect()
179}
180
181fn selection_windows(selection: &TransformChannelSelection) -> Vec<ChannelWindow> {
182    selection
183        .channels
184        .iter()
185        .zip(&selection.metadata_ends)
186        .map(|(&first, &last)| ChannelWindow {
187            first,
188            last,
189            dropped: false,
190        })
191        .collect()
192}

194/// Use the existing copy-task channel geometry before sampling a new grid.
195pub(super) fn selected_footprints(
196    selection: &TransformChannelSelection,
197    frequencies: &[f64],
198    widths: &[f64],
199) -> Option<(Vec<f64>, Vec<f64>)> {
200    let grid = selected_windows(&selection_windows(selection), frequencies, widths, widths)?;
201    Some((grid.centres, grid.widths))
202}

204fn selected_windows(
205    windows: &[ChannelWindow],
206    freq: &[f64],
207    widths: &[f64],
208    resolution: &[f64],
209) -> Option<SpectralGrid> {
210    let sign = if freq.last()? < freq.first()? {
211        -1.0
212    } else {
213        1.0
214    };
215    let mut grid = SpectralGrid {
216        centres: Vec::with_capacity(windows.len()),
217        widths: Vec::with_capacity(windows.len()),
218        resolution: Vec::with_capacity(windows.len()),
219        reference: 0.0,
220        bandwidth: 0.0,
221    };
222    // MSTransformDataHandler::fillSPWTable uses the stride's span for the
223    // metadata even though the visibility iterator reads only stride samples.
224    let centre =
225        |window: &ChannelWindow| Some((freq.get(window.first)? + freq.get(window.last)?) * 0.5);
226    grid.reference = centre(windows.first()?)?.min(centre(windows.last()?)?);
227    let mut dropped_width = 0.0;
228    for &ChannelWindow {
229        first,
230        last,
231        dropped,
232    } in windows
233    {
234        let separation = (freq.get(last)? - freq.get(first)?) * sign;
235        let width = (separation + (widths.get(first)? + widths.get(last)?).abs() * 0.5) * sign;
236        grid.bandwidth += width.abs();
237        if dropped {
238            dropped_width += width;
239        } else {
240            grid.centres.push((freq[first] + freq[last]) * 0.5);
241            grid.widths.push(width);
242            grid.resolution
243                .push(separation + (resolution.get(first)? + resolution.get(last)?) * 0.5);
244        }
245    }
246    grid.bandwidth -= dropped_width;
247    Some(grid)
248}
