31pub(crate) fn load_preapplied_rows(
32    ms: &MeasurementSet,
33    selection: &MsSelection,
34    prior_calibration_tables: &[ApplyCalibrationTableSpec],
35    parang: bool,
36) -> Result<Option<HashMap<usize, EvaluatedApplyRow>>, GainSolveError> {
37    if prior_calibration_tables.is_empty() && !parang {
38        return Ok(None);
39    }
40    let plan = plan_apply(
41        ms,
42        &ApplyPlanRequest {
43            selection: selection.clone(),
44            apply_mode: crate::ApplyMode::CalFlag,
45            parang,
46            calibration_tables: prior_calibration_tables.to_vec(),
47        },
48    )
49    .map_err(|source| GainSolveError::PriorCalibrationPlan {
50        source: Box::new(source),
51    })?;
52    evaluate_apply_rows(ms, &plan).map(Some).map_err(|source| {
53        GainSolveError::PriorCalibrationApply {
54            source: Box::new(source),
55        }
56    })
57}

59pub(crate) fn build_solve_groups(
60    ms: &MeasurementSet,
61    rows: &[SelectedSolveRow],
62    preapplied_rows: Option<&HashMap<usize, EvaluatedApplyRow>>,
63    options: SolveGroupOptions,
64) -> Result<BTreeMap<(SolveBaseKey, SolveBucketKey), SolveAccumulator>, GainSolveError> {
65    let mut groups = BTreeMap::<(SolveBaseKey, SolveBucketKey), SolveAccumulator>::new();
66    let mut layouts = SolveRowLayouts::new(ms, options.rotate_point_model)?;
67    let needs_first_buffer = options.gain_type == GainType::T || options.robust_seed;
68    let mut ordered = bucketed_rows(
69        rows,
70        options.solve_interval,
71        options.combine,
72        needs_first_buffer,
73    );
74    let seed_buffers =
75        if needs_first_buffer || options.combine.observations || options.combine.spectral_windows {
76            let buffers = first_t_buffers(&ordered, options.preaverage_max_span);
77            ordered.retain(|(row, _, _)| row.antenna1 != row.antenna2);
78            buffers
79        } else {
80            BTreeMap::new()
81        };
82    let spectral_weights = SpectralWeights::new(ms, &ordered, options.combine)?;
83    let pipelined = overlap_inputs(ordered.len());
84    if std::env::var_os("CASA_RS_CALIBRATION_TIMING").is_some() {
85        eprintln!("calibration_workers {}", if pipelined { 2 } else { 1 });
86    }
87    let read = |batch: &[BucketedRow<'_>]| {
88        let _phase = crate::timing::Phase::start("solve_read_decode");
89        let indices = batch
90            .iter()
91            .map(|(row, _, _)| row.row_index)
92            .collect::<Vec<_>>();
93        read_input_batch(
94            ms,
95            &indices,
96            preapplied_rows.is_none(),
97            options.model_source,
98            batch
99                .iter()
100                .map(|row| spectral_weights.contains(row))
101                .collect(),
102        )
103    };
104    let mut consume = |batch: &[BucketedRow<'_>], inputs: SolveInputBatch| {
105        let _accumulate_phase = crate::timing::Phase::start("solve_accumulate");
106        for (index, &(row, base_key, bucket_key)) in batch.iter().enumerate() {
107            let group = groups.entry((base_key, bucket_key)).or_insert_with(|| {
108                let buffer = seed_buffers.get(&(base_key, bucket_key));
109                let mut group = SolveAccumulator::new(
110                    row.field_id,
111                    buffer.map_or(row.data_spw_id, |first| first.spw_id),
112                    buffer.map_or(row.observation_id, |first| first.observation_id),
113                );
114                group.first_t_buffer = buffer.copied();
115                group
116            });
117            if options.combine.fields {
118                // CalSolVi2Organizer orders the buffers of a grouped time
119                // interval by FIELD_ID. SDBList::aggregateFld uses the first.
120                group.field_id = group.field_id.min(row.field_id);
121            }
122            let preapplied = preapplied_rows.and_then(|rows| rows.get(&row.row_index));
123            group.observe(
124                ms,
125                row,
126                preapplied,
127                &options,
128                &mut layouts,
129                (&inputs, index),
130            )?;
131        }
132        Ok::<_, GainSolveError>(())
133    };
134    if pipelined {
135        std::thread::scope(|scope| {
136            let (sender, receiver) = std::sync::mpsc::sync_channel(1);
137            let producer_rows = &ordered;
138            scope.spawn(move || {
139                for batch in producer_rows.chunks(256) {
140                    let input = read(batch);
141                    let failed = input.is_err();
142                    if sender.send(input).is_err() || failed {
143                        break;
144                    }
145                }
146            });
147            // Drop the receiver before scope joins the reader on an error,
148            // unblocking a producer that is waiting to send its next batch.
149            (|| {
150                for (batch, input) in ordered.chunks(256).zip(receiver) {
151                    consume(batch, input?)?;
152                }
153                Ok::<_, GainSolveError>(())
154            })()
155        })?;
156    } else {
157        for batch in ordered.chunks(256) {
158            consume(batch, read(batch)?)?;
159        }
160    }
161
162    Ok(groups)
163}

167/// One reader and the calling accumulation thread, within the CPU budget.
168fn overlap_inputs(rows: usize) -> bool {
169    let available = std::thread::available_parallelism().map_or(1, usize::from);
170    let requested = std::env::var("RAYON_NUM_THREADS")
171        .ok()
172        .and_then(|value| value.parse::<usize>().ok())
173        .filter(|n| *n > 0)
174        .unwrap_or(available);
175    rows > 256 && available.min(requested) >= 2
176}

184/// Read selected arrays together so headers and tiles are shared by a batch;
185/// release the payloads after accumulation instead of retaining the whole MS.
186fn read_input_batch(
187    ms: &MeasurementSet,
188    rows: &[usize],
189    read_visibilities: bool,
190    model: GainSolveModelSource,
191    spectral_weights: Vec<bool>,
192) -> Result<SolveInputBatch, GainSolveError> {
193    let schema = ms.main_table().schema();
194    let mut columns = Vec::new();
195    if spectral_weights.contains(&true) {
196        columns.push("SIGMA_SPECTRUM");
197    }
198    if spectral_weights.contains(&false) {
199        columns.push("SIGMA");
200    }
201    if read_visibilities {
202        columns.extend(["DATA", "FLAG"]);
203    }
204    if model == GainSolveModelSource::ModelColumn {
205        columns.push("MODEL_DATA");
206    }
207    let mut inputs = SolveInputBatch {
208        columns: Vec::with_capacity(columns.len()),
209        spectral_weights,
210    };
211    for column in columns {
212        let values = ms
213            .main_table()
214            .column_accessor(column)
215            .and_then(|accessor| {
216                read_column_inputs(
217                    &accessor,
218                    column,
219                    schema
220                        .and_then(|s| s.column(column))
221                        .and_then(|c| c.data_type()),
222                    rows,
223                )
224            })
225            .map_err(|source| ms_read_error(ms, source))?;
226        inputs.columns.push((column, values));
227    }
228    Ok(inputs)
229}

231/// Homogeneous cells use the typed selected reader, avoiding reconstruction
232/// of complete tiled cubes. Mixed shapes and pending edits use the ordinary
233/// owned-cell reader, preserving their existing validation and semantics.
234fn read_column_inputs(
235    column: &casa_tables::TableColumn<'_>,
236    name: &str,
237    kind: Option<casa_types::PrimitiveType>,
238    rows: &[usize],
239) -> Result<Vec<Option<ArrayValue>>, casa_tables::TableError> {
240    if name != "SIGMA"
241        && let Ok(shapes) = column.array_shapes_uncached(rows)
242        && let Some(Some(shape)) = shapes.first()
243        && shape.len() == 2
244        && shape.iter().all(|n| *n > 0)
245        && shapes.iter().all(|other| other.as_ref() == Some(shape))
246    {
247        use casa_tables::SelectedArray2DCellsMut;
248        use casa_types::PrimitiveType;
249        let values = match kind {
250            Some(PrimitiveType::Complex32) => read_row_packed(
251                column,
252                rows,
253                shape[1],
254                |values| SelectedArray2DCellsMut::RowChannelComplex32(values),
255                ArrayValue::Complex32,
256            )?,
257            Some(PrimitiveType::Bool) => read_row_packed(
258                column,
259                rows,
260                shape[1],
261                |values| SelectedArray2DCellsMut::RowChannelBool(values),
262                ArrayValue::Bool,
263            )?,
264            Some(PrimitiveType::Float32) => read_row_packed(
265                column,
266                rows,
267                shape[1],
268                |values| SelectedArray2DCellsMut::RowChannelFloat32(values),
269                ArrayValue::Float32,
270            )?,
271            _ => None,
272        };
273        if let Some(values) = values {
274            return Ok(values);
275        }
276    }
277    column.array_cells_owned_uncached(rows)
278}

280/// Request contiguous row cells directly, avoiding a channel/row transpose.
281fn read_row_packed<T: Clone>(
282    column: &casa_tables::TableColumn<'_>,
283    rows: &[usize],
284    channels: usize,
285    destination: for<'a> fn(&'a mut Vec<T>) -> casa_tables::SelectedArray2DCellsMut<'a>,
286    wrap: fn(ArrayD<T>) -> ArrayValue,
287) -> Result<Option<Vec<Option<ArrayValue>>>, casa_tables::TableError> {
288    let mut values = Vec::new();
289    let Some(shape) = column.fill_array_cells_2d_channel_range_typed_uncached(
290        rows,
291        0,
292        channels,
293        destination(&mut values),
294    )?
295    else {
296        return Ok(None);
297    };
298    let shape = [shape.axis0_count, shape.channel_count];
299    let capacity = shape[0].checked_mul(shape[1]).ok_or_else(|| {
300        casa_tables::TableError::Storage("solve input cell size overflows".to_string())
301    })?;
302    values
303        .chunks_exact(capacity)
304        .map(|row| {
305            ArrayD::from_shape_vec(IxDyn(&shape).f(), row.to_vec())
306                .map(|array| Some(wrap(array)))
307                .map_err(|e| casa_tables::TableError::Storage(e.to_string()))
308        })
309        .collect::<Result<Vec<_>, _>>()
310        .map(Some)
311}

313fn input_cell<'a>(
314    ms: &'a MeasurementSet,
315    row: usize,
316    inputs: (&'a SolveInputBatch, usize),
317    column: &str,
318) -> Result<&'a ArrayValue, GainSolveError> {
319    if let Some((_, values)) = inputs.0.columns.iter().find(|(name, _)| *name == column) {
320        return values[inputs.1].as_ref().ok_or_else(|| {
321            ms_read_error(
322                ms,
323                casa_tables::TableError::ColumnNotFound {
324                    row_index: row,
325                    column: column.to_string(),
326                },
327            )
328        });
329    }
330    main_cell(ms, row, column)
331}

343/// Refuses `rows` when two of their data descriptions share a spectral
344/// window, whatever their polarizations: CASA solves each description
345/// apart, where casa-rs's groups and tables are keyed by window.
346pub(crate) fn refuse_shared_spectral_windows(
347    rows: &[SelectedSolveRow],
348) -> Result<(), GainSolveError> {
349    let mut descriptions = BTreeMap::<i32, BTreeSet<i32>>::new();
350    for row in rows {
351        descriptions
352            .entry(row.data_spw_id)
353            .or_default()
354            .insert(row.data_desc_id);
355    }
356    match descriptions.into_iter().find(|(_, ids)| ids.len() > 1) {
357        Some((spw_id, ids)) => Err(GainSolveError::SharedSpectralWindow {
358            spw_id,
359            data_desc_ids: ids.into_iter().collect(),
360        }),
361        None => Ok(()),
362    }
363}

380pub(crate) fn resolve_refants(
381    ms: &MeasurementSet,
382    selector: &RefAntSelector,
383) -> Result<Vec<i32>, GainSolveError> {
384    match selector {
385        RefAntSelector::Unspecified => Ok(Vec::new()),
386        RefAntSelector::AntennaId(id) => Ok(vec![*id]),
387        RefAntSelector::Expression { selection, .. } => ms
388            .resolve_reference_antennas(selection)
389            .map(|ids| ids.into_iter().map(|id| id as i32).collect())
390            .map_err(|error| GainSolveError::ResolveRefAnt {
391                selector: selection.clone(),
392                reason: error.to_string(),
393            }),
394        RefAntSelector::AntennaName(name) => {
395            let antenna = ms
396                .antenna()
397                .map_err(|error| GainSolveError::ResolveRefAnt {
398                    selector: name.clone(),
399                    reason: error.to_string(),
400                })?;
401            for row in 0..antenna.row_count() {
402                if antenna
403                    .name(row)
404                    .map_err(|error| GainSolveError::ResolveRefAnt {
405                        selector: name.clone(),
406                        reason: error.to_string(),
407                    })?
408                    == *name
409                {
410                    return Ok(vec![row as i32]);
411                }
412            }
413            Err(GainSolveError::ResolveRefAnt {
414                selector: name.clone(),
415                reason: "no ANTENNA.NAME match".to_string(),
416            })
417        }
418    }
419}

421pub(crate) fn collect_selected_rows(
422    ms: &MeasurementSet,
423    selection: &MsSelection,
424    full_selection: bool,
425) -> Result<Vec<SelectedSolveRow>, GainSolveError> {
426    let selected_rows = if full_selection {
427        resolve_casa_calibration_selection(ms, selection)
428    } else {
429        resolve_calibration_selection(ms, selection)
430    }
431    .map_err(|error| GainSolveError::OpenMeasurementSet {
432        path: ms
433            .path()
434            .map(|path| path.display().to_string())
435            .unwrap_or_else(|| "<in-memory>".to_string()),
436        source: match error {
437            casa_ms::MsSelectionError::Domain(source) => source,
438            other => MsError::InvalidInput(other.to_string()),
439        },
440    })?;
441    if selected_rows.selected_rows.is_empty() {
442        return Err(GainSolveError::EmptySelection);
443    }
444
445    let channels = &selected_rows.channel_selections;
446    selected_rows
447        .selected_rows
448        .into_iter()
449        .map(|row| {
450            Ok(SelectedSolveRow {
451                row_index: row.row_index,
452                array_id: get_i32(ms.main_table(), row.row_index, "ARRAY_ID")?,
453                field_id: row.field_id,
454                observation_id: get_i32(ms.main_table(), row.row_index, "OBSERVATION_ID")?,
455                data_desc_id: row.data_desc_id,
456                data_spw_id: row.spectral_window_id,
457                channel_selection: if full_selection {
458                    channels.get(&row.spectral_window_id).cloned()
459                } else {
460                    selection.channel_selection_for_spw(row.spectral_window_id)
461                },
462                antenna1: row.antenna1_id,
463                antenna2: row.antenna2_id,
464                time_seconds: row.time_mjd_seconds,
465                time_centroid_seconds: get_f64(ms.main_table(), row.row_index, "TIME_CENTROID")?,
466                interval_seconds: get_f64(ms.main_table(), row.row_index, "INTERVAL")?,
467                scan_number: get_i32(ms.main_table(), row.row_index, "SCAN_NUMBER")?,
468                flag_row: row.flag_row,
469            })
470        })
471        .collect()
472}

474pub(crate) fn correlation_types_for_ddid(
475    ms: &MeasurementSet,
476    data_desc_id: i32,
477) -> Result<Vec<i32>, GainSolveError> {
478    let dd = ms
479        .data_description()
480        .map_err(|source| GainSolveError::OpenMeasurementSet {
481            path: ms
482                .path()
483                .map(|path| path.display().to_string())
484                .unwrap_or_else(|| "<in-memory>".to_string()),
485            source,
486        })?;
487    let pol = ms
488        .polarization()
489        .map_err(|source| GainSolveError::OpenMeasurementSet {
490            path: ms
491                .path()
492                .map(|path| path.display().to_string())
493                .unwrap_or_else(|| "<in-memory>".to_string()),
494            source,
495        })?;
496    let row = usize::try_from(data_desc_id).map_err(|_| {
497        GainSolveError::UnsupportedCorrelationLayout {
498            data_desc_id,
499            correlation_types: Vec::new(),
500        }
501    })?;
502    let pol_id = dd
503        .polarization_id(row)
504        .map_err(|source| GainSolveError::OpenMeasurementSet {
505            path: ms
506                .path()
507                .map(|path| path.display().to_string())
508                .unwrap_or_else(|| "<in-memory>".to_string()),
509            source,
510        })?;
511    pol.corr_type(usize::try_from(pol_id).unwrap_or(usize::MAX))
512        .map_err(|source| GainSolveError::OpenMeasurementSet {
513            path: ms
514                .path()
515                .map(|path| path.display().to_string())
516                .unwrap_or_else(|| "<in-memory>".to_string()),
517            source,
518        })
519}

728    /// Add one MAIN row: its unflagged, positively weighted samples divided by
729    /// the model, each weighted by its solve weight times |model|² as
730    /// `VisEquation::divideCorrByModel` does, and its TIME_CENTROID.
731    fn observe(
732        &mut self,
733        ms: &MeasurementSet,
734        row: &SelectedSolveRow,
735        preapplied_row: Option<&EvaluatedApplyRow>,
736        options: &SolveGroupOptions,
737        layouts: &mut SolveRowLayouts,
738        inputs: (&SolveInputBatch, usize),
739    ) -> Result<(), GainSolveError> {
740        let inputs = read_row_inputs(ms, row, preapplied_row, options.model_source, inputs)?;
741        let channel_count = inputs.data.shape()[1];
742        let layout = match layouts.by_ddid.entry(row.data_desc_id) {
743            std::collections::hash_map::Entry::Occupied(entry) => entry.into_mut(),
744            std::collections::hash_map::Entry::Vacant(entry) => entry.insert(SolveRowLayout {
745                correlations: correlation_types_for_ddid(ms, row.data_desc_id)?,
746                channel_count: 0,
747                selection: None,
748                channels: vec![],
749                all_channels: false,

751        };

753            layout.channels = selected_channels(row.channel_selection.as_ref(), channel_count)?;
754            layout.all_channels = layout.channels.iter().copied().eq(0..channel_count);
755            layout.channel_count = channel_count;
756            layout.selection.clone_from(&row.channel_selection);
757        }
758        let rotated = layouts
759            .rotation
760            .as_ref()
761            .map(|state| {
762                samples::rotate_point_model(
763                    ms,
764                    row,
765                    state,
766                    &layout.correlations,
767                    options.point_stokes,
768                )
769            })
770            .transpose()?;
771        let mut samples = inputs.samples();
772        samples.point_model = rotated.as_deref();
773        self.observe_selected_samples(
774            &SolveSampleRow::of(row),
775            &samples,
776            &layout.correlations,
777            &layout.channels,
778            options,
779            layout.all_channels,
780        )

789        &mut self,
790        row: &SolveSampleRow,
791        samples: &SolveSamples<'_>,
792        correlation_types: &[i32],
793        channels: &[usize],
794        options: &SolveGroupOptions,
795    ) -> Result<(), GainSolveError> {

806    fn observe_selected_samples(
807        &mut self,
808        row: &SolveSampleRow,
809        samples: &SolveSamples<'_>,
810        correlation_types: &[i32],
811        channels: &[usize],
812        options: &SolveGroupOptions,
813        all_channels: bool,
814    ) -> Result<(), GainSolveError> {
815        if options.channelized {
816            let channel_count = channels.len();
817            if self.sample_rows > 0 && self.solution_channels != channel_count {
818                return Err(GainSolveError::UnsupportedParameterShape {
819                    path: "<measurement-set DATA>".to_string(),
820                    shape: samples.data.shape().to_vec(),
821                });
822            }
823            self.solution_channels = channel_count;
824        }
825
826        let graph_count = match options.gain_type {
827            GainType::G => 2,
828            GainType::T => {
829                self.parallel_hands = if matches!(correlation_types.len(), 2 | 4) {
830                    2
831                } else {
832                    1
833                };
834                1
835            }
836        };
837        if self.receptor_accumulations.len() < graph_count {
838            self.receptor_accumulations
839                .resize_with(graph_count, HashMap::new);
840        }
841        self.observe_metadata(row);
842        let shared_flags =
843            (!options.correlation_dependent_flags).then(|| samples::channel_flags(samples));
844        let no_weighted_data = shared_flags
845            .as_ref()
846            .is_some_and(|flags| flags.iter().all(|flag| *flag))
847            || samples.weights.iter().all(|weight| *weight <= 0.0);
848        // CASA gathers each field and data description in a separate buffer.
849        // Keep their averages separate until enforcing the solve mode.
850        let mut baseline = AveragedBaseline {
851            antennas: (row.antenna1, row.antenna2),
852            field_id: if options.channelized { 0 } else { row.field_id },
853            data_desc_id: row.data_desc_id,
854            start_bits: 0,
855        };
856        let start_bits = options.preaverage_max_span.map_or(0, |max_span| {
857            let start = self
858                .preaverage_starts
859                .entry(baseline)
860                .or_insert(row.time_seconds);
861            if row.time_seconds - *start > max_span {
862                *start = row.time_seconds;
863            }
864            start.to_bits()
865        });
866        baseline.start_bits = start_bits;
867        let solution_channels = self.solution_channels;
868        let mut row_weight = 0.0_f64;
869        for (corr_index, &correlation_type) in correlation_types.iter().enumerate() {
870            let Some(receptors) = correlation_receptors(correlation_type) else {
871                return Err(GainSolveError::UnsupportedCorrelationLayout {
872                    data_desc_id: row.data_desc_id,
873                    correlation_types: correlation_types
874                        .iter()
875                        .map(|code| stokes_name(*code).to_string())
876                        .collect(),
877                });
878            };
879            let parallel_hand = receptors.0 == receptors.1;
880            if no_weighted_data {
881                continue;
882            }
883            let graph_index = match options.gain_type {
884                GainType::G => receptors.0,
885                GainType::T => 0,
886            };
887            let point = if options.channelized {
888                Complex32::new(options.point_stokes[0], 0.0)
889            } else if let Some(rotated) = samples.point_model {
890                rotated[corr_index]
891            } else {
892                samples::point_visibility(
893                    options.point_stokes,
894                    correlation_type,
895                    correlation_types[0],
896                )
897            };
898            if !parallel_hand && options.gain_type == GainType::G {
899                if samples.point_model.is_some() {
900                    row_weight += samples::centroid_weight(
901                        samples,
902                        corr_index,
903                        channels,
904                        shared_flags.as_deref(),
905                        point,
906                    );
907                }
908                continue;
909            }
910            // Look the baseline/correlation up once per row; channels are
911            // dense indices, not independent hash keys for every sample.
912            let accumulations = self.receptor_accumulations[graph_index]
913                .entry(SolveAccumulationKey {
914                    baseline,
915                    correlation_index: corr_index,
916                    parallel_hand,
917                })
918                .or_insert_with(|| vec![SolveEdgeAccumulation::default(); solution_channels]);
919            row_weight += samples::SampleAccumulation {
920                values: accumulations,
921                attempted: &mut self.attempted,
922                channelized: options.channelized,
923                retain_zero_model: options.gain_type == GainType::T,
924            }
925            .observe(
926                samples,
927                corr_index,
928                shared_flags.as_deref(),
929                channels,
930                point,
931                all_channels,
932            );
933        }
934        if row_weight > 0.0 {
935            let centroid = self.baseline_centroids.entry(baseline).or_default();
936            centroid.weighted_sum += row_weight * row.time_centroid_seconds;
937            centroid.weight += row_weight;
938        }
939        Ok(())
940    }

942    fn observe_metadata(&mut self, row: &SolveSampleRow) {
943        self.min_time = self.min_time.min(row.time_seconds);
944        self.max_time = self.max_time.max(row.time_seconds);
945        if self.unique_time_bits.insert(row.time_seconds.to_bits()) {
946            self.unique_time_sum += row.time_seconds;
947            self.unique_time_count += 1;
948        }
949        self.total_interval += row.interval_seconds;
950        self.sample_rows += 1;
951        self.field_ids.insert(row.field_id);
952        self.scan_numbers.insert(row.scan_number);
953        self.antenna_ids.insert(row.antenna1);
954        self.antenna_ids.insert(row.antenna2);
955    }

957    /// Collapse each baseline's samples per correlation and solution channel
958    /// into the solve graphs, and set the solution TIME.
959    ///
960    /// A phase-only solve sees each collapsed sample at unit amplitude with
961    /// its weight times its squared amplitude. An amplitude-only solve
962    /// keeps the collapsed magnitude and the original weight
963    /// (`SolveDataBuffer::enforceAPonData`). TIME is
964    /// `SDBList::aggregateTimeCentroid`: each baseline's TIME_CENTROID,
965    /// averaged over its rows by row weight as time averaging does, weighted
966    /// by the baseline's solve weight after that phase-only scaling.
967    pub(crate) fn finalize_for_solve(&mut self, solve_mode: GainSolveMode, preserve_samples: bool) {
968        self.channel_graphs = vec![
969            vec![SolveGraph::default(); self.receptor_accumulations.len()];
970            self.solution_channels
971        ];
972        let mut baseline_weights = HashMap::<AveragedBaseline, f64>::new();
973        for (receptor, accumulations) in self.receptor_accumulations.iter().enumerate() {
974            for (key, channels) in accumulations {
975                for (channel_index, accumulation) in channels.iter().enumerate() {
976                    if accumulation.weight <= 0.0 {
977                        continue;
978                    }
979                    let averaged =
980                        accumulation.weighted_sum / Complex32::new(accumulation.weight, 0.0);
981                    let (sample, weight) = match solve_mode {
982                        GainSolveMode::Phase => {
983                            let amplitude = averaged.norm();
984                            if amplitude <= f32::EPSILON {
985                                continue;
986                            }
987                            (
988                                averaged / Complex32::new(amplitude, 0.0),
989                                accumulation.weight * amplitude * amplitude,
990                            )
991                        }
992                        GainSolveMode::Amplitude => {
993                            (Complex32::new(averaged.norm(), 0.0), accumulation.weight)
994                        }
995                        GainSolveMode::AmplitudePhase => (averaged, accumulation.weight),
996                    };
997                    let model = accumulation.model_weight / accumulation.weight;
998                    let model_weight = weight * model;
999                    self.channel_graphs[channel_index][receptor].accumulate(
1000                        key.baseline.antennas,
1001                        model_weight * model,
1002                        sample * model_weight,
1003                        SolveEdgeStats {
1004                            weighted_sample_power: f64::from(weight) * f64::from(sample.norm_sqr()),
1005                            raw_weighted_sample_power: accumulation.weighted_sample_power,
1006                            sample_count: accumulation.sample_count,
1007                            collapsed_count: 1,
1008                        },
1009                    );
1010                    if preserve_samples {
1011                        self.channel_graphs[channel_index][receptor]
1012                            .robust_samples
1013                            .push(RobustSample {
1014                                baseline: key.baseline.antennas,
1015                                correlation: key.correlation_index,
1016                                visibility: sample,
1017                                weight,
1018                                model,
1019                                in_first_buffer: self
1020                                    .first_t_buffer
1021                                    .is_some_and(|buffer| buffer.contains(key.baseline)),
1022                            });
1023                    }
1024                    *baseline_weights.entry(key.baseline).or_default() += f64::from(weight);
1025                }
1026            }
1027        }
1028
1029        let mut solution_time = WeightedTime::default();
1030        for (baseline, weight) in baseline_weights {
1031            let Some(centroid) = self.baseline_centroids.get(&baseline) else {
1032                continue;
1033            };
1034            if centroid.weight > 0.0 {
1035                solution_time.weighted_sum += weight * centroid.weighted_sum / centroid.weight;
1036                solution_time.weight += weight;
1037            }
1038        }
1039        self.solution_time = if solution_time.weight > 0.0 {
1040            solution_time.weighted_sum / solution_time.weight
1041        } else {
1042            self.aggregate_time()
1043        };
1044    }
1045
1046    /// Add `other`'s accumulation, of the same group, to this one before
1047    /// [`Self::finalize_for_solve`], as if its samples had been observed
1048    /// here: the fragment work units of the MSv2/MSv4 trial accumulate apart
1049    /// and merge before the group closes. Sums change order, so results
1050    /// agree to rounding, not bitwise.
1051    #[cfg(test)]
1052    pub(crate) fn merge(&mut self, other: Self) {
1053        self.min_time = self.min_time.min(other.min_time);
1054        self.max_time = self.max_time.max(other.max_time);
1055        for bits in other.unique_time_bits {
1056            if self.unique_time_bits.insert(bits) {
1057                self.unique_time_sum += f64::from_bits(bits);
1058                self.unique_time_count += 1;
1059            }
1060        }
1061        self.total_interval += other.total_interval;
1062        self.sample_rows += other.sample_rows;
1063        self.field_ids.extend(other.field_ids);
1064        self.scan_numbers.extend(other.scan_numbers);
1065        self.antenna_ids.extend(other.antenna_ids);
1066        self.solution_channels = self.solution_channels.max(other.solution_channels);
1067        self.attempted |= other.attempted;
1068        for (baseline, centroid) in other.baseline_centroids {
1069            let mine = self.baseline_centroids.entry(baseline).or_default();
1070            mine.weighted_sum += centroid.weighted_sum;
1071            mine.weight += centroid.weight;
1072        }
1073        if self.receptor_accumulations.len() < other.receptor_accumulations.len() {
1074            self.receptor_accumulations
1075                .resize_with(other.receptor_accumulations.len(), HashMap::new);
1076        }
1077        for (mine, theirs) in self
1078            .receptor_accumulations
1079            .iter_mut()
1080            .zip(other.receptor_accumulations)
1081        {
1082            for (key, channels) in theirs {
1083                let entries = mine
1084                    .entry(key)
1085                    .or_insert_with(|| vec![SolveEdgeAccumulation::default(); channels.len()]);
1086                for (entry, accumulation) in entries.iter_mut().zip(channels) {
1087                    entry.merge(accumulation);
1088                }
1089            }
1090        }
1091    }

1093    /// Entries the accumulation's sets and maps hold: its state grows with
1094    /// the group's distinct times, antennas, baselines and baseline
1095    /// correlations, not with its samples.
1096    #[cfg(test)]
1097    pub(crate) fn state_entries(&self) -> usize {
1098        self.unique_time_bits.len()
1099            + self.scan_numbers.len()
1100            + self.antenna_ids.len()
1101            + self.baseline_centroids.len()
1102            + self
1103                .receptor_accumulations
1104                .iter()
1105                .flat_map(HashMap::values)
1106                .map(Vec::len)
1107                .sum::<usize>()
1108    }

1110    /// The mean of the interval's distinct TIMEs (`SDBList::aggregateTime`,
1111    /// CASA's TIME when no sample carries weight).
1112    pub(crate) fn aggregate_time(&self) -> f64 {
1113        if self.unique_time_count > 0 {
1114            self.unique_time_sum / self.unique_time_count as f64
1115        } else {
1116            (self.min_time + self.max_time) / 2.0
1117        }
1118    }

1121/// Read one MAIN row's data, flags, model and solve weights, from the row or
1122/// from its pre-applied calibration.
1123///
1124/// The weights are CASA's (`CalSolvingVi2::calibrateCurrentVB`, which calls
1125/// `VisBufferImpl2::resetWeightsUsingSigma`): 1/σ² from SIGMA_SPECTRUM when
1126/// the MeasurementSet has WEIGHT_SPECTRUM and SIGMA_SPECTRUM, otherwise from
1127/// SIGMA broadcast over channels; σ ≤ 0 gives no weight. WEIGHT and
1128/// WEIGHT_SPECTRUM are not read. Pre-applied `calwt` tables then scale them.
1129fn read_row_inputs<'a>(
1130    ms: &'a MeasurementSet,
1131    row: &SelectedSolveRow,
1132    preapplied_row: Option<&'a EvaluatedApplyRow>,
1133    model_source: GainSolveModelSource,
1134    inputs: (&'a SolveInputBatch, usize),
1135) -> Result<SolveRowInputs<'a>, GainSolveError> {
1136    let (data, flags) = match preapplied_row {
1137        Some(preapplied) => (&preapplied.corrected_data, &preapplied.flags),
1138        None => (
1139            input_cell(ms, row.row_index, inputs, "DATA")?,
1140            input_cell(ms, row.row_index, inputs, "FLAG")?,
1141        ),
1142    };
1143    let ArrayValue::Complex32(data) = data else {
1144        return Err(unsupported_shape("DATA", data));
1145    };
1146    let ArrayValue::Bool(flags) = flags else {
1147        return Err(unsupported_shape("FLAG", flags));
1148    };
1149    if data.ndim() != 2 || data.shape() != flags.shape() {
1150        return Err(GainSolveError::UnsupportedParameterShape {
1151            path: "<measurement-set row>".to_string(),
1152            shape: data.shape().to_vec(),
1153        });
1154    }
1155    let model = match model_source {
1156        GainSolveModelSource::ModelColumn => {
1157            let model = input_cell(ms, row.row_index, inputs, "MODEL_DATA")?;
1158            match model {
1159                ArrayValue::Complex32(model) if model.shape() == data.shape() => Some(model),
1160                other => return Err(unsupported_shape("MODEL_DATA", other)),
1161            }
1162        }
1163        GainSolveModelSource::PointSource => None,
1164    };
1165    let mut weights = solve_weights(ms, row.row_index, data.shape(), inputs)?;
1166    if let Some(factors) = preapplied_row.and_then(|row| row.weight_factors.as_ref()) {
1167        if factors.shape() != data.shape() {
1168            return Err(GainSolveError::UnsupportedParameterShape {
1169                path: "<pre-applied weight factors>".to_string(),
1170                shape: factors.shape().to_vec(),
1171            });
1172        }
1173        weights = &weights
1174            .broadcast(data.raw_dim())
1175            .expect("checked weight shape")
1176            * factors;
1177    }
1178    Ok(SolveRowInputs {
1179        data,
1180        flags: solve_flags(row, flags),
1181        weights,
1182        model,
1183    })
1184}

1204/// The median, the mean of the middle two for an even count, as casacore's
1205/// `partialMedians(..., takeEvenMean=True)` that CASA uses for a row's
1206/// channel-independent weight.
1207pub(super) fn median_with_even_mean(values: &mut [f32]) -> f32 {
1208    if values.is_empty() {
1209        return 0.0;
1210    }
1211    let middle = values.len() / 2;
1212    let even = values.len().is_multiple_of(2);
1213    let first = values[0];
1214    if values
1215        .iter()
1216        .all(|value| value.to_bits() == first.to_bits())
1217    {
1218        return if even { (first + first) / 2.0 } else { first };
1219    }
1220    let (lower, upper, _) = values.select_nth_unstable_by(middle, f32::total_cmp);
1221    let upper = *upper;
1222    if even {
1223        let lower = lower
1224            .iter()
1225            .copied()
1226            .max_by(f32::total_cmp)
1227            .expect("nonempty even lower half");
1228        (lower + upper) / 2.0
1229    } else {
1230        upper
1231    }
1232}

1234/// The channels of a window of `channel_count` channels that a solve
1235/// uses: those `selection` names, else all. The gain table's
1236/// `SPECTRAL_WINDOW` row collapses the same channels.
1237pub(crate) fn selected_channels(
1238    selection: Option<&ChannelSelection>,
1239    channel_count: usize,
1240) -> Result<Vec<usize>, GainSolveError> {
1241    match selection {
1242        Some(selection) => {
1243            selection
1244                .indices(channel_count)
1245                .map_err(|source| GainSolveError::OpenMeasurementSet {
1246                    path: "<measurement-set channel selection>".to_string(),
1247                    source,
1248                })
1249        }
1250        None => Ok((0..channel_count).collect()),
1251    }
1252}

1254/// The sample flags a solve uses for `row`: its `FLAG` cell, or every sample
1255/// flagged when the row's `FLAG_ROW` is set, whatever `FLAG` holds. CASA's
1256/// solve does the same: `CalSolvingVi2::calibrateCurrentVB` calls
1257/// `VisEquation::collapse2`, whose `divideCorrByModel` flags the whole cube
1258/// of such a row and zeroes its weights before G, T or B accumulates it.
1259pub(crate) fn solve_flags<'a>(
1260    row: &SelectedSolveRow,
1261    flags: &'a ArrayD<bool>,
1262) -> Cow<'a, ArrayD<bool>> {
1263    if row.flag_row {
1264        Cow::Owned(ArrayD::from_elem(flags.raw_dim(), true))
1265    } else {
1266        Cow::Borrowed(flags)
1267    }
1268}
