2//! An MSv2 destination preserves an MSv2 source's POINTING rows and required
3//! metadata, remapping only antenna IDs. For MSv4, it writes the samples
4//! each input's plan's reads
5//! gather through the boundary ([`Observation::pointing`],
6//! [`PointingPlan::reads`]), one row per sample, input after input, each
7//! input's in (time, antenna) order as casacore's
8//! `MSPointingColumns::pointingIndex` expects it and its antennas the
9//! output's. An MSv4 source states no
10//! interval, so its rows have `INTERVAL` 0 (which casacore reads as covering
11//! half the gap to the antenna's next row, and casa-rs's reader as covering
12//! nothing); it holds no target, so `TARGET` repeats the direction, as MSv2
13//! writers without a separate target write it.

29/// One gathered `POINTING` row, its times in MJD seconds; its coefficients,
30/// `[axis][term]`, are `terms` per axis from `start` in the gathered
31/// coefficients.
32#[derive(Clone, Copy, Debug)]
33struct Row {
34    /// The time's bit pattern, so that equal times compare equal.
35    time_bits: u64,
36    antenna: AntennaId,
37    /// The row's place among its antenna's samples at its time.
38    repeat: u32,
39    interval_s: f64,
40    origin_s: f64,
41    start: usize,
42    terms: usize,
43}
44
45/// About the bytes a `POINTING` row holds in the created MeasurementSet's
46/// table until it is saved (#950), as the counting laws measure it.
47pub(super) const TABLE_ROW_BYTES: usize = 2_560;
48
49/// The bytes the writer holds for `samples` gathered samples of
50/// `coefficients` coefficients in all: it reserves exactly that and
51/// removes repeats in place.
52pub(super) fn gathered_bytes(samples: usize, coefficients: usize) -> usize {
53    samples
54        .saturating_mul(size_of::<Row>())
55        .saturating_add(coefficients.saturating_mul(size_of::<f64>()))
56}

58/// One input's pointing: its opening, the reads its plan carries, and the
59/// output's `ANTENNA` row of each of its antennas.
60pub(super) struct Pointed<'a> {
61    pub(super) observation: &'a dyn Observation,
62    pub(super) msv2: Option<&'a crate::observation_io::Msv2Observation>,
63    pub(super) plan: &'a PointingPlan,
64    pub(super) antennas: &'a [i32],
65}

67/// One input's gathered rows and their coefficients.
68struct Gathered {
69    rows: Vec<Row>,
70    coefficients: Vec<f64>,
71}

73/// Write `inputs`' pointing, as their plans carry it, into `ms`'s
74/// `POINTING`, input after input, each input's antennas as the output
75/// numbers them. Reads of one input that hold the same sample (an MSv4's
76/// datasets each hold their antennas' pointing) write it once: a sample is
77/// the k-th at its antenna and time, and different samples there are
78/// refused. Samples of different inputs are all written. Every input's
79/// directions must be in one frame.
80pub(super) fn write(
81    ms: &mut MeasurementSet,
82    inputs: &[Pointed<'_>],
83) -> Result<(), ObservationError> {
84    // Preserving MSv2 storage decorations belongs here, inside its adapter's
85    // writer; no task reads or reconstructs pointing payload itself.
86    for input in inputs {
87        if input.plan.preserved() {
88            copy_msv2(ms, input)?;
89        } else {
90            write_series(ms, std::slice::from_ref(input))?;
91        }
92    }
93    Ok(())
94}

96fn copy_msv2(ms: &mut MeasurementSet, input: &Pointed<'_>) -> Result<(), ObservationError> {
97    let source = input
98        .msv2
99        .expect("preserved MSv2 pointing")
100        .measurement_set();
101    let Some(source) = source.subtable(SubtableId::Pointing) else {
102        return Ok(());
103    };
104    let table = subtable_mut(ms, SubtableId::Pointing).map_err(storage)?;
105    if source.row_count() == 0 {
106        return Ok(());
107    }
108    for column in schema::pointing::REQUIRED_COLUMNS {
109        if let Some(keywords) = source.column_keywords(column.name) {
110            if table.row_count() > 0 && table.column_keywords(column.name) != Some(keywords) {
111                return Err(crate::observation_io::update::request_error(
112                    "POINTING column metadata differs between inputs (#939)",
113                ));
114            }
115            table.set_column_keywords(column.name, keywords.clone());
116        }
117    }
118    for row in source.rows().map_err(storage)? {
119        let mut row = row.clone();
120        let antenna = match row.get("ANTENNA_ID") {
121            Some(casa_core::Value::Scalar(casa_core::ScalarValue::Int32(id))) => *id,
122            _ => {
123                return Err(crate::observation_io::update::request_error(
124                    "POINTING ANTENNA_ID is not int32",
125                ));
126            }
127        };
128        // -1 is CASA's all-antennas row, which remains -1.
129        if antenna >= 0 {
130            let mapped = *input.antennas.get(antenna as usize).ok_or_else(|| {
131                crate::observation_io::update::request_error("POINTING names an unknown antenna")
132            })?;
133            *row.get_mut("ANTENNA_ID").expect("checked antenna") = i(mapped);
134        }
135        table.add_row(row).map_err(storage)?;
136    }
137    Ok(())
138}

140fn write_series(ms: &mut MeasurementSet, inputs: &[Pointed<'_>]) -> Result<(), ObservationError> {
141    let mut gathered = Vec::with_capacity(inputs.len());
142    let mut frame = None;
143    for input in inputs {
144        let (own, own_frame) = gather_input(input)?;
145        if let Some(own_frame) = own_frame {
146            if frame.is_some_and(|known| known != own_frame) {
147                return Err(ObservationError::Unsupported {
148                    what: "writing MSv2 pointing of inputs in different frames",
149                });
150            }
151            frame = Some(own_frame);
152        }
153        gathered.push(own);
154    }
155    let Some(frame) = frame else {
156        return Ok(());
157    };
158    let terms = gathered
159        .iter()
160        .flat_map(|own| own.rows.iter().map(|row| row.terms))
161        .max()
162        .unwrap_or(1);
163    let table = subtable_mut(ms, SubtableId::Pointing).map_err(storage)?;
164    if frame != DirectionRef::J2000 {
165        for column in ["DIRECTION", "TARGET"] {
166            TableMeasDesc::new_fixed(column, MeasureType::Direction, frame.as_str())
167                .write(table)
168                .map_err(storage)?;
169        }
170    }
171    for own in &gathered {
172        for row in &own.rows {
173            // `[2, terms]`, each axis's terms padded to the most any row has.
174            let mut direction = Vec::with_capacity(2 * terms);
175            for axis in 0..2 {
176                let start = row.start + axis * row.terms;
177                direction.extend_from_slice(&own.coefficients[start..start + row.terms]);
178                direction.extend(std::iter::repeat_n(0.0, terms - row.terms));
179            }
180            let direction = f64_array(&direction, vec![2, terms]);
181            let values = row_from_defs(
182                schema::pointing::REQUIRED_COLUMNS,
183                &[
184                    ("ANTENNA_ID", i(row.antenna.0 as i32)),
185                    ("TIME", f(f64::from_bits(row.time_bits))),
186                    ("INTERVAL", f(row.interval_s)),
187                    ("NAME", s("")),
188                    ("NUM_POLY", i(terms as i32 - 1)),
189                    ("TIME_ORIGIN", f(row.origin_s)),
190                    ("DIRECTION", direction.clone()),
191                    ("TARGET", direction),
192                    ("TRACKING", b(true)),
193                ],
194            );
195            table.add_row(values).map_err(storage)?;
196        }
197    }
198    Ok(())
199}

201/// One input's rows, in (time, antenna) order with its repeats removed and
202/// its antennas the output's, and their frame (`None` for no pointing).
203fn gather_input(input: &Pointed<'_>) -> Result<(Gathered, Option<DirectionRef>), ObservationError> {
204    let (observation, plan) = (input.observation, input.plan);
205    let catalogue = observation.catalogue();
206    let (samples, held) = plan.gathered();
207    let mut rows = Vec::with_capacity(samples);
208    let mut coefficients = Vec::with_capacity(held);
209    let mut frame = None;
210    for read in plan.reads() {
211        let series = observation.pointing(read)?;
212        if series.antennas.is_empty() {
213            continue;
214        }
215        if frame.is_some_and(|known| known != series.frame) {
216            return Err(ObservationError::Unsupported {
217                what: "pointing in different frames in different partitions",
218            });
219        }
220        frame = Some(series.frame);
221        let entry = catalogue.partition(read.partition);
222        gather(&series, entry, &mut rows, &mut coefficients);
223    }
224    rows.sort_by(|a, b| {
225        f64::from_bits(a.time_bits)
226            .total_cmp(&f64::from_bits(b.time_bits))
227            .then(a.antenna.cmp(&b.antenna))
228            .then(a.repeat.cmp(&b.repeat))
229    });
230    // Remove repeats in place: a sample two reads gathered is written once.
231    let key = |row: &Row| (row.time_bits, row.antenna, row.repeat);
232    let mut kept = 0;
233    for index in 0..rows.len() {
234        if kept > 0 && key(&rows[kept - 1]) == key(&rows[index]) {
235            if !same_sample(&rows[kept - 1], &rows[index], &coefficients) {
236                return Err(ObservationError::Unsupported {
237                    what: "partitions with different pointing at one antenna and time",
238                });
239            }
240            continue;
241        }
242        rows[kept] = rows[index];
243        kept += 1;
244    }
245    rows.truncate(kept);
246    for row in &mut rows {
247        row.antenna = AntennaId(input.antennas[row.antenna.0 as usize] as u32);
248    }
249    Ok((Gathered { rows, coefficients }, frame))
250}

252/// Add `series`' samples (in `entry`'s time system) to `rows`, their
253/// coefficients to `coefficients`.
254fn gather(
255    series: &PointingSeries,
256    entry: &casa_observation::PartitionEntry,
257    rows: &mut Vec<Row>,
258    coefficients: &mut Vec<f64>,
259) {
260    // Times in the MSv2's own system: MJD seconds.
261    let mjd = |times: &[f64]| -> Vec<f64> {
262        let view = EpochView::new(entry.time_scale, entry.time_origin, ArrayView1::from(times));
263        (0..times.len())
264            .map(|time| view.seconds_since(time, MjdHighPrec::new(0.0, 0.0)))
265            .collect()
266    };
267    for (antenna, samples) in &series.antennas {
268        let times = mjd(samples.times_s());
269        let origins = mjd(samples.origins_s());
270        let terms = samples.terms();
271        let mut repeat = 0;
272        for (sample, time) in times.iter().enumerate() {
273            repeat = if sample > 0 && times[sample - 1].to_bits() == time.to_bits() {
274                repeat + 1
275            } else {
276                0
277            };
278            rows.push(Row {
279                time_bits: time.to_bits(),
280                antenna: *antenna,
281                repeat,
282                interval_s: samples
283                    .intervals_s()
284                    .map_or(0.0, |intervals| intervals[sample]),
285                origin_s: origins[sample],
286                start: coefficients.len(),
287                terms,
288            });
289            coefficients.extend_from_slice(
290                &samples.coefficients()[sample * 2 * terms..(sample + 1) * 2 * terms],
291            );
292        }
293    }
294}

296/// Whether two gathered rows are one sample: the same interval, origin and
297/// polynomial, terms beyond either's own being zero (a read pads each
298/// antenna's terms to the most any of its antennas has).
299fn same_sample(a: &Row, b: &Row, coefficients: &[f64]) -> bool {
300    let term = |row: &Row, axis: usize, term: usize| {
301        if term < row.terms {
302            coefficients[row.start + axis * row.terms + term]
303        } else {
304            0.0
305        }
306    };
307    let terms = a.terms.max(b.terms);
308    a.interval_s.to_bits() == b.interval_s.to_bits()
309        && a.origin_s.to_bits() == b.origin_s.to_bits()
310        && (0..2).all(|axis| {
311            (0..terms).all(|index| term(a, axis, index).to_bits() == term(b, axis, index).to_bits())
312        })
313}

315#[cfg(test)]
316mod tests {
317    use casa_observation::AntennaId;
318
319    use super::{Row, same_sample};
320
321    /// One sample read with its antenna's own terms and with terms padded
322    /// to another antenna's is the same sample; a different term is not.
323    #[test]
324    fn padding_does_not_make_a_different_sample() {
325        let row = |start, terms| Row {
326            time_bits: 1.0_f64.to_bits(),
327            antenna: AntennaId(1),
328            repeat: 0,
329            interval_s: 1.0,
330            origin_s: 0.5,
331            start,
332            terms,
333        };
334        // [ra, dec]; [ra, 0, dec, 0]; [ra, 1e-6, dec, 0].
335        let coefficients = [1.0, -0.5, 1.0, 0.0, -0.5, 0.0, 1.0, 1e-6, -0.5, 0.0];
336        assert!(same_sample(&row(0, 1), &row(2, 2), &coefficients));
337        assert!(!same_sample(&row(0, 1), &row(6, 2), &coefficients));
338    }
