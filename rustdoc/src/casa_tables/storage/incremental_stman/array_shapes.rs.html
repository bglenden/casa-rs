2//! Shapes follow the ISM's existing row/bucket/indirect-array registration.
3use super::*;
4
5pub(crate) fn read_ism_array_column_shapes(
6    path: &Path,
7    columns: &[&ColumnDescContents],
8    column: usize,
9    rows: &[usize],
10) -> Result<Option<Vec<Option<Vec<usize>>>>, StorageError> {
11    if rows.is_empty() {
12        return Ok(Some(Vec::new()));
13    }
14    let descriptor = columns.get(column).ok_or_else(|| {
15        StorageError::FormatMismatch("ISM shape column index is out of range".to_string())
16    })?;
17    if !descriptor.is_array {
18        return Ok(None);
19    }
20    if !is_ism_indirect_array(descriptor) {
21        if descriptor.shape.is_empty() {
22            return Ok(None);
23        }
24        let shape = descriptor
25            .shape
26            .iter()
27            .map(|&size| size as usize)
28            .collect::<Vec<_>>();
29        return Ok(Some(vec![Some(shape); rows.len()]));
30    }
31    let mut file = File::open(path)?;
32    let header = parse_ism_header(&mut file)?;
33    let index = parse_ism_index(&mut file, &header)?;
34    let array_path = PathBuf::from(format!("{}i", path.display()));
35    let mut reader = array_path
36        .is_file()
37        .then(|| StManArrayFileReader::open(&array_path, header.big_endian))
38        .transpose()?;
39    let mut cached = None;
40    let mut cached_number = None;
41    let mut result = Vec::with_capacity(rows.len());
42    for &row in rows {
43        let interval = index
44            .rows
45            .partition_point(|&start| start <= row as u64)
46            .checked_sub(1)
47            .filter(|&slot| slot < index.bucket_nrs.len())
48            .ok_or_else(|| {
49                StorageError::FormatMismatch(format!("ISM shape row {row} is outside the index"))
50            })?;
51        let number = index.bucket_nrs[interval];
52        if cached_number != Some(number) {
53            let bytes = read_ism_bucket(&mut file, &header, number)?;
54            cached = Some(parse_ism_bucket(&bytes, columns.len(), header.big_endian)?);
55            cached_number = Some(number);
56        }
57        let bucket = cached.as_ref().expect("bucket initialized");
58        let Some(indices) = bucket.col_indices.get(column) else {
59            result.push(None);
60            continue;
61        };
62        let entry = get_interval(indices, (row as u64 - index.rows[interval]) as u32);
63        let value = read_scalar_at(
64            &bucket.data,
65            indices.offsets[entry] as usize,
66            CasacoreDataType::TpInt64,
67            header.big_endian,
68        )?;
69        let casa_types::Value::Scalar(ScalarValue::Int64(offset)) = value else {
70            return Err(StorageError::FormatMismatch(
71                "ISM shape offset is not Int64".to_string(),
72            ));
73        };
74        result.push(match reader.as_mut() {
75            Some(reader) => reader.read_shape_at(offset)?,
76            None if offset == 0 => None,
77            None => return Err(StorageError::MissingDataFile(array_path)),
78        });
79    }
80    Ok(Some(result))
81}
