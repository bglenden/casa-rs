19/// Bounded metadata acquisition; pixel generation remains in the admitted phase.
20pub(super) fn resolve(
21    request: &ImagingRequest,
22    aw: &AwProjection,
23    surveyed: &Surveyed<'_>,
24    first_row: SelectedObservationRow,
25) -> Result<NativeAwRequestInput, PrepareError> {
26    let AwCfSource::NativeEvla {
27        evla_surface,
28        native_cf_working_size,
29        native_cf_oversampling,
30        native_cf_maximum_cells,
31        ..
32    } = &aw.cf_source
33    else {
34        unreachable!("native AW resolves only a native cache");
35    };
36    let surface = evla_dish(surveyed.ms, first_row, evla_surface)?;
37    let frequencies = frequency_groups(surveyed.survey, surveyed.spectral);
38    let sky_cell = request.cell.to_radians() / 3600.0;
39    let (w_values, w_increment) = w_grid(
40        aw.wprojplanes
41            .expect("a validated AW request names its W planes")
42            .get(),
43        sky_cell,
44    );
45    let working_cell = sky_cell * *native_cf_oversampling as f64 * request.imsize as f64
46        / *native_cf_working_size as f64;
47    let pa = surveyed.engine.parallactic_angle(
48        first_row.time_mjd_seconds(),
49        first_row.field_id() as usize,
50        0,
51    )? as f32;
52    let feed_angle = receptor_zero_angle(
53        surveyed.ms,
54        first_row.time_mjd_seconds(),
55        frequencies[0].spectral_window,
56    )? as f32;
57    Ok(NativeAwRequestInput {
58        surface,
59        antenna_diameter_m: 25.0,
60        frequencies,
61        w_values,
62        w_increment,
63        pa_values: vec![f64::from(pa + feed_angle)],
64        mueller_elements: mueller_elements(&request.stokes)?,
65        reference_frequency_hz: surveyed.spectral.reference_frequency_hz,
66        grid: NativeAwGrid {
67            size: *native_cf_working_size,
68            sky_increment_rad: [-working_cell, working_cell],
69            oversampling: *native_cf_oversampling,
70        },
71        // The installed A-projection (see `specification::aw_projection`).
72        terms: NativeAwTerms {
73            aperture: true,
74            w_term: true,
75            prolate_spheroidal: false,
76            wideband: true,
77            conjugate_beams: true,
78        },
79        maximum_cells: *native_cf_maximum_cells,
80    })
81}

83/// The EVLA dish surface, after checking that the observation is EVLA with
84/// the homogeneous 25 m dishes the native model describes.
85fn evla_dish(
86    ms: &MeasurementSet,
87    first_row: SelectedObservationRow,
88    evla_surface: &std::path::Path,
89) -> Result<EvlaDishSurface, PrepareError> {
90    let observation = ms.observation()?;
91    if observation
92        .string(first_row.observation_id() as usize, "TELESCOPE_NAME")?
93        .trim()
94        != "EVLA"
95    {
96        return Err(PrepareError::NativeAwTelescope);
97    }
98    let antenna = ms.antenna()?;
99    for row in 0..antenna.row_count() {
100        if antenna.dish_diameter(row)? != 25.0 {
101            return Err(PrepareError::NativeAwDishes);
102        }
103    }
104    // The reference data is an explicit input with bounded acquisition, never
105    // discovered through a CASA installation or downloaded during execution.
106    let mut bytes = Vec::new();
107    std::fs::File::open(evla_surface)
108        .and_then(|file| file.take(1_048_577).read_to_end(&mut bytes))
109        .map_err(|source| PrepareError::EvlaSurface {
110            path: evla_surface.to_path_buf(),
111            source,
112        })?;
113    if bytes.len() > 1_048_576 {
114        return Err(PrepareError::NativeAwSurfaceSize);
115    }
116    let text = std::str::from_utf8(&bytes).map_err(|_| PrepareError::EvlaSurfaceText {
117        path: evla_surface.to_path_buf(),
118    })?;
119    Ok(EvlaDishSurface::from_surface_text(text)?)
120}

122/// One frequency group per selected SPW, in increasing CF frequency.
123fn frequency_groups(
124    survey: &Survey,
125    spectral: &PreparedSpectralAxis,
126) -> Vec<NativeAwFrequencyGroup> {
127    let mut frequencies = survey
128        .spectral_windows
129        .iter()
130        .map(|window| {
131            let channel_frequencies_hz = spectral
132                .window_channels(window.spw_id)
133                .iter()
134                .map(|index| {
135                    *window
136                        .frequencies_hz
137                        .get(*index)
138                        .expect("selected channels index their window's frequency axis")
139                })
140                .collect::<Vec<_>>();
141            // TransformMachines2::makeFreqValList selects each SPW's high endpoint.
142            let cf_frequency_hz = channel_frequencies_hz
143                .iter()
144                .copied()
145                .reduce(f64::max)
146                .expect("the spectral axis selects at least one channel of every window");
147            NativeAwFrequencyGroup {
148                spectral_window: u32::try_from(window.spw_id)
149                    .expect("SPW ids are nonnegative stored i32 values"),
150                channel_frequencies_hz,
151                cf_frequency_hz,
152            }
153        })
154        .collect::<Vec<_>>();
155    frequencies.sort_by(|left, right| left.cf_frequency_hz.total_cmp(&right.cf_frequency_hz));
156    frequencies
157}

159/// The W values of `planes` planes and their increment. CASA AWConvFunc
160/// derives its W grid from the requested field of view, not the observed W
161/// envelope: maxUVW=1/(4*sky_increment), w=i²/wScale.
162fn w_grid(planes: usize, sky_cell: f64) -> (Vec<f64>, f64) {
163    let max_w = 1.0 / (sky_cell * 4.0);
164    let w_increment = ((planes - 1) * (planes - 1)) as f32 as f64 / max_w;
165    let w_values = (0..planes)
166        .map(|index| (index * index) as f64 / w_increment)
167        .collect();
168    (w_values, w_increment)
169}

171/// The Mueller elements of the imaged polarization. The catalog routes
172/// each hand through its own element and, for the conjugate baseline, the
173/// opposite hand (`makeConjPolMap`), so a single-hand image still needs
174/// both diagonal elements.
175fn mueller_elements(stokes: &[PolarizationCoordinate]) -> Result<Vec<usize>, PrepareError> {
176    match stokes {
177        [PolarizationCoordinate::CircularRr]
178        | [PolarizationCoordinate::CircularLl]
179        | [PolarizationCoordinate::StokesI] => Ok(vec![0, 15]),
180        _ => Err(PrepareError::NativeAwPolarization),
181    }
182}

184fn receptor_zero_angle(ms: &MeasurementSet, time: f64, spw: u32) -> Result<f64, PrepareError> {
185    let feed = ms.feed()?;
186    let mut angle = None;
187    for row in 0..feed.row_count() {
188        if feed.i32(row, "ANTENNA_ID")? != 0 || feed.i32(row, "FEED_ID")? != 0 {
189            continue;
190        }
191        let spectral_window = feed.i32(row, "SPECTRAL_WINDOW_ID")?;
192        if spectral_window != -1 && spectral_window != spw as i32 {
193            continue;
194        }
195        let center = feed.f64(row, "TIME")?;
196        let interval = feed.f64(row, "INTERVAL")?;
197        if interval != 0.0 && (time - center).abs() > interval / 2.0 {
198            continue;
199        }
200        let ArrayValue::Float64(values) = feed.array(row, "RECEPTOR_ANGLE")? else {
201            return Err(PrepareError::FeedAngleType);
202        };
203        let value = values
204            .iter()
205            .next()
206            .copied()
207            .filter(|value| value.is_finite())
208            .ok_or(PrepareError::FeedAngleNotFinite)?;
209        if angle.is_some_and(|previous: f64| previous.to_bits() != value.to_bits()) {
210            return Err(PrepareError::FeedAngleAmbiguous);
211        }
212        angle = Some(value);
213    }
214    angle.ok_or(PrepareError::NoApplicableFeed)
215}
