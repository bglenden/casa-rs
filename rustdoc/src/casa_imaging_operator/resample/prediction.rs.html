22/// Reusable buffers for predicting a batch of rows; one per row chunk.
23///
24/// A batch is [`Self::begin`], then [`SpectralResampler::place_prediction`]
25/// for each row, and either one [`Self::apply_native`] that degrids every
26/// placed sample and interpolates the predictions to the rows' native cells
27/// ([`SpectralResampler::native_terms`]) in a single backend call, or one
28/// [`Self::apply`] and [`SpectralResampler::finish_prediction`] for each
29/// row. [`SpectralResampler::predict_row`] is the batch of one row.
30/// [`SpectralResampler::prediction_scratch`] allocates the buffers once for
31/// a batch of its widest rows; a default scratch grows as batches need.
32#[derive(Debug, Default)]
33pub struct PredictionScratch {
34    /// The batch's placements: a prediction reads no values or weights.
35    placements: Vec<Placement>,
36    npol: usize,
37    predicted: Vec<Complex32>,
38    /// The output or native channels the placements predict, row by row:
39    /// several share a placement that reads the same taps.
40    sources: Vec<Source>,
41    /// While one row's terms are formed, the batch index of the placement
42    /// at each of its output or native channels.
43    slots: Vec<u32>,
44    /// The native terms of the batch's rows.
45    terms: Vec<NativeTerm>,
46    /// The tridiagonal system for the legacy spline's image planes.
47    spline: SplineScratch,
48}
49
50impl PredictionScratch {
51    /// Scratch for `npol` correlations with room for `samples` degridded
52    /// samples, `slots` output or native channels of one row and `terms`
53    /// native cells, so that predicting rows within them never grows it
54    /// beyond [`Self::bytes`].
55    #[must_use]
56    pub fn with_capacity(npol: usize, samples: usize, slots: usize, terms: usize) -> Self {
57        Self {
58            placements: Vec::with_capacity(samples),
59            npol,
60            predicted: Vec::with_capacity(samples * npol),
61            sources: Vec::with_capacity(samples),
62            slots: Vec::with_capacity(slots),
63            terms: Vec::with_capacity(terms),
64            spline: SplineScratch::default(),
65        }
66    }
67
68    /// Bytes [`Self::with_capacity`] allocates.
69    #[must_use]
70    pub const fn bytes(npol: usize, samples: usize, slots: usize, terms: usize) -> u64 {
71        (samples * (size_of::<Placement>() + npol * size_of::<Complex32>() + size_of::<Source>())
72            + slots * size_of::<u32>()
73            + terms * size_of::<NativeTerm>()) as u64
74    }
75
76    /// Samples placed since [`Self::begin`]: one per run of a row's
77    /// channels that read the same taps ([`SpectralResampler::place_prediction`]).
78    #[must_use]
79    pub fn samples(&self) -> usize {
80        self.placements.len()
81    }
82
83    /// Bytes the scratch's buffers hold now, from their capacities.
84    #[must_use]
85    pub fn capacity_bytes(&self) -> u64 {
86        (self.placements.capacity() * size_of::<Placement>()
87            + self.predicted.capacity() * size_of::<Complex32>()
88            + self.sources.capacity() * size_of::<Source>()
89            + self.slots.capacity() * size_of::<u32>()
90            + self.terms.capacity() * size_of::<NativeTerm>()) as u64
91            + self.spline.capacity_bytes()
92    }
93
94    /// Start a batch of rows of `npol` correlations, clearing every buffer.
95    pub fn begin(&mut self, npol: usize) {
96        self.npol = npol;
97        self.placements.clear();
98        self.sources.clear();
99        self.terms.clear();
100    }
101
102    /// The native terms [`SpectralResampler::native_terms`] added since
103    /// [`Self::begin`].
104    #[must_use]
105    pub fn terms(&self) -> &[NativeTerm] {
106        &self.terms
107    }
108
109    /// The batch's placements, as the block a backend predicts.
110    fn placed(&self) -> SampleBlock<'_> {
111        SampleBlock {
112            placements: &self.placements,
113            values: &[],
114            weights: &[],
115            sources: &[],
116            npol: self.npol,
117        }
118    }
119
120    /// Degrid every sample placed since [`Self::begin`] from `model` in one
121    /// [`Work::Predict`] on `backend`, for
122    /// [`SpectralResampler::finish_prediction`].
123    ///
124    /// # Errors
125    ///
126    /// Whatever `backend` returns.
127    pub fn apply(
128        &mut self,
129        backend: &mut dyn GridBackend,
130        cf: &dyn ConvolutionFunctionSet,
131        model: &PreparedModelGrids,
132    ) -> Result<(), OperatorError> {
133        self.predicted.clear();
134        self.predicted
135            .resize(self.placements.len() * self.npol, Complex32::default());
136        if self.placements.is_empty() {
137            return Ok(());
138        }
139        let block = SampleBlock {
140            placements: &self.placements,
141            values: &[],
142            weights: &[],
143            sources: &[],
144            npol: self.npol,
145        };
146        backend.apply(
147            &block,
148            cf,
149            Work::Predict {
150                model,
151                out: &mut self.predicted,
152            },
153        )
154    }
155
156    /// Degrid every sample placed since [`Self::begin`] from `grids` and
157    /// interpolate the predictions to the batch's native cells
158    /// ([`Self::terms`]) in one [`Work::PredictNative`] on `backend`,
159    /// writing `model`, or with `add` adding to it.
160    ///
161    /// # Errors
162    ///
163    /// Whatever `backend` returns.
164    pub fn apply_native(
165        &self,
166        backend: &mut dyn GridBackend,
167        cf: &dyn ConvolutionFunctionSet,
168        grids: &PreparedModelGrids,
169        model: NativeTarget<'_>,
170        add: bool,
171    ) -> Result<(), OperatorError> {
172        if self.terms.is_empty() {
173            return Ok(());
174        }
175        backend.apply(
176            &self.placed(),
177            cf,
178            Work::PredictNative {
179                grids,
180                terms: &self.terms,
181                model,
182                add,
183            },
184        )
185    }
186}

197/// Consecutive output or native channels of a row that one placement
198/// predicts: the first channel's slot in the row, how many, and the
199/// placement's index in the batch. A row's sources run in slot order.
200#[derive(Clone, Copy, Debug)]
201struct Source {
202    slot: u32,
203    count: u32,
204    placement: u32,
205}

207impl Source {
208    /// The row's slots this source covers.
209    fn slots(self) -> std::ops::Range<usize> {
210        self.slot as usize..(self.slot + self.count) as usize
211    }
212}

214fn index_sources(slots: &mut Vec<u32>, sources: &[Source], count: usize) {
215    slots.clear();
216    slots.resize(count, NO_CELL);
217    for source in sources {
218        slots[source.slots()].fill(source.placement);
219    }
220}

222/// The largest difference of phase-centre phase, in radians, at which two
223/// samples that read the same taps of the same model share one placement:
224/// 2⁻³⁰, 64 times below the resolution of an `f32` unit phasor (2⁻²⁴), so a
225/// shared prediction differs from the sample's own by less than its
226/// rounding. A row whose phase centre is shifted has phases that change
227/// with every channel by far more, and shares nothing.
228const SHARED_PHASE_RAD: f64 = 1.0 / (1u64 << 30) as f64;

230/// What a backend's degrid of a placement reads beside its phasor: its
231/// kernel cell and location on the grid, its plane, its signs of w, its
232/// pointing gradient and, for a model of several terms, its Taylor
233/// variable.
234#[derive(Clone, Copy, PartialEq)]
235struct Gathered {
236    cf: CfKey,
237    location: CellLocation,
238    plane: u32,
239    w_positive: bool,
240    prediction_w_positive: bool,
241    gradient: [u32; 2],
242    spectral: u32,
243}
244
245impl Gathered {
246    /// What `placement`, at `location`, reads for a model of `terms` terms.
247    fn of(placement: &Placement, location: CellLocation, terms: usize) -> Self {
248        Self {
249            cf: placement.cf,
250            location,
251            plane: placement.plane,
252            w_positive: placement.w > 0.0,
253            prediction_w_positive: placement.prediction_w_positive,
254            gradient: placement.gradient.map(f32::to_bits),
255            spectral: if terms > 1 {
256                placement.spectral.to_bits()
257            } else {
258                0
259            },
260        }
261    }

264/// The output channels CASA degrids for one row under interpolation
265/// (`FTMachine::getInterpolateArrays`, after `GridFT::get` returns early
266/// when no native maps): each native's `chanMap` is its channel, −2 in the
267/// interpolation's halo or −1; −1 is replaced by the maximum, and the channels from
268/// the minimum to the maximum are degridded, from channel 0 when the
269/// minimum is the halo's −2 or equals the maximum. When every native lies
270/// in the halo the maximum is −2 and CASA's unsigned loop flags nothing, so
271/// every channel is degridded. `None` when nothing maps.
272fn degridded_channels(
273    planes: u32,
274    native: impl Iterator<Item = f64> + Clone,
275    map: impl Fn(f64) -> i64,
276) -> Option<std::ops::RangeInclusive<u32>> {
277    let maximum = native.clone().map(&map).max()?;
278    if maximum == -1 {
279        return None;
280    }
281    if maximum == -2 {
282        return Some(0..=planes - 1);
283    }
284    let minimum = native
285        .map(|frequency_hz| match map(frequency_hz) {
286            -1 => maximum,
287            channel => channel,
288        })
289        .min()?;
290    let first = if minimum == maximum {
291        0
292    } else {
293        minimum.max(0)
294    };
295    Some(first as u32..=maximum as u32)
296}

298/// Whether `frequencies_hz` strictly increase or strictly decrease.
299fn monotonic(frequencies_hz: &[f64]) -> bool {
300    frequencies_hz.windows(2).all(|pair| pair[0] < pair[1])
301        || frequencies_hz.windows(2).all(|pair| pair[0] > pair[1])
302}

304/// Bit `p` set for each flagged correlation `p` of `channel`.
305fn flag_bits(row: &NativeRow<'_>, npol: usize, channel: usize) -> u32 {
306    row.flags[channel * npol..(channel + 1) * npol]
307        .iter()
308        .enumerate()
309        .fold(0, |bits, (pol, flag)| bits | (u32::from(*flag) << pol))
310}

313    /// The most samples [`Self::predict_row`] degrids for one native row of
314    /// `channels` channels: one per channel, or under linear interpolation
315    /// one per output channel.
316    #[must_use]
317    pub fn prediction_samples_per_row(&self, channels: usize) -> usize {
318        match &self.sampling {
319            Sampling::Direct | Sampling::Nearest(_) => channels,
320            Sampling::Interpolated(axis, _) => channels.max(axis.channels as usize),
321        }
322    }

324    /// Reusable buffers for predicting batches of `rows` rows of at most
325    /// `channels` channels of `npol` correlations, allocated once at
326    /// [`Self::prediction_scratch_bytes`] for every resampler in `resamplers`.
327    #[must_use]
328    pub fn prediction_scratch<'r>(
329        resamplers: impl IntoIterator<Item = &'r Self>,
330        npol: usize,
331        channels: usize,
332        rows: usize,
333    ) -> PredictionScratch {
334        let (samples, slots, spline) = Self::prediction_extent(resamplers, channels);
335        let mut scratch =
336            PredictionScratch::with_capacity(npol, rows * samples, slots, rows * channels);
337        scratch.spline = SplineScratch::with_capacity(spline, npol);
338        scratch
339    }
340
341    /// Bytes [`Self::prediction_scratch`] holds for the same arguments.
342    #[must_use]
343    pub fn prediction_scratch_bytes<'r>(
344        resamplers: impl IntoIterator<Item = &'r Self>,
345        npol: usize,
346        channels: usize,
347        rows: usize,
348    ) -> u64 {
349        let (samples, slots, spline) = Self::prediction_extent(resamplers, channels);
350        PredictionScratch::bytes(npol, rows * samples, slots, rows * channels)
351            + SplineScratch::bytes(spline, npol)
352    }

354    /// The most samples any of `resamplers` predicts with for rows of at
355    /// most `channels` channels, and the most output or native channels of
356    /// one row and the largest legacy spline.
357    fn prediction_extent<'r>(
358        resamplers: impl IntoIterator<Item = &'r Self>,
359        channels: usize,
360    ) -> (usize, usize, usize) {
361        resamplers
362            .into_iter()
363            .fold((0, channels, 0), |(samples, slots, spline), resampler| {
364                let planes = match &resampler.sampling {
365                    Sampling::Interpolated(axis, _) => axis.channels as usize,
366                    Sampling::Direct | Sampling::Nearest(_) => 0,
367                };
368                (
369                    samples.max(resampler.prediction_samples_per_row(channels)),
370                    slots.max(planes),
371                    if resampler.uses_legacy_spline() {
372                        spline.max(planes)
373                    } else {
374                        spline
375                    },
376                )
377            })
378    }

380    /// The model visibility of every selected sample of `row`, written to
381    /// `out` as `[channel][correlation]`, the way CASA predicts `MODEL_DATA`
382    /// (`GridFT::get`, then `FTMachine::interpolateFrequencyFromgrid`).
383    ///
384    /// A flagged row, a flagged correlation, a native channel off the output
385    /// axis and a sample whose kernel leaves the grid predict zero, and so
386    /// does an autocorrelation unless the resampler is
387    /// [`Self::predicting_autocorrelations`].
388    /// A constant or Taylor basis, `nearest` mapping and the one-channel
389    /// bypass degrid each native channel at its own frequency on its mapped
390    /// plane. `linear` mapping degrids the output channels CASA degrids for
391    /// the row (see `degridded_channels`) at the channel centre, repeats
392    /// each value `floor(width ratio)` times on CASA's fine grid when output
393    /// channels are wider than native ones, and interpolates linearly to the
394    /// native frequencies on the axis and in its halo (extrapolating from the
395    /// end pair); unmapped output channels contribute zeros to that grid.
396    /// Only the planes `model` holds are degridded; a caller predicting a
397    /// window of planes prepares the window's halo (see
398    /// [`Self::model_planes`]).
399    ///
400    /// This is the batch of one row; a caller predicting many rows places
401    /// them all into one [`PredictionScratch`] batch, so that one backend
402    /// call degrids them.
403    ///
404    /// # Errors
405    ///
406    /// A row inconsistent with `operator`, an `out` of another length than
407    /// the row's values, or whatever `backend` returns.
408    pub fn predict_row(
409        &self,
410        operator: &MeasurementOperator,
411        backend: &mut dyn GridBackend,
412        model: &PreparedModelGrids,
413        row: &NativeRow<'_>,
414        scratch: &mut PredictionScratch,
415        out: &mut [Complex32],
416    ) -> Result<(), OperatorError> {
417        self.predict_row_with_coverage(operator, backend, model, row, scratch, (out, None))
418    }

420    /// Predict a row and optionally mark the samples the gridder assigned.
421    ///
422    /// This distinguishes a sampled zero from an untouched cell in CASA's
423    /// legacy imager, which degrids into the existing model visibility.
424    /// # Errors
425    /// The same errors as [`Self::predict_row`] and a mismatched coverage slice.
426    pub fn predict_row_with_coverage(
427        &self,
428        operator: &MeasurementOperator,
429        backend: &mut dyn GridBackend,
430        model: &PreparedModelGrids,
431        row: &NativeRow<'_>,
432        scratch: &mut PredictionScratch,
433        output: (&mut [Complex32], Option<&mut [bool]>),
434    ) -> Result<(), OperatorError> {
435        scratch.begin(operator.polarization().correlations().len());
436        let placed = self.place_prediction(operator, model, row, scratch)?;
437        scratch.apply(backend, operator.cf(), model)?;
438        self.finish_prediction_with_coverage(operator, row, placed, scratch, output)
439    }

441    /// Interpolate already sampled image channels for a legacy imager that
442    /// samples its sky in the image plane, such as SDGrid. `sample` fills a
443    /// plane's correlations in row order. Returns false for a one-channel
444    /// or nearest-neighbour prediction, which the caller samples directly.
445    ///
446    /// # Errors
447    /// The native flags or output do not have one cell per channel and
448    /// correlation, or the live cubic grid is too short for interpolation.
449    pub fn predict_sampled_spectrum(
450        &self,
451        row: &NativeRow<'_>,
452        correlations: usize,
453        scratch: &mut PredictionScratch,
454        output: &mut [Complex32],
455        mut sample: impl FnMut(u32, &mut [Complex32]),
456    ) -> Result<bool, OperatorError> {
457        let Some((axis, order)) = self.interpolated_prediction(row) else {
458            return Ok(false);
459        };
460        if correlations == 0
461            || output.len() != row.frequencies_hz.len() * correlations
462            || row.flags.len() != output.len()
463        {
464            return Err(OperatorError::NativeRow {
465                reason: "sampled spectrum does not match native channel and correlation dimensions",
466            });
467        }
468        if row.row_flag {
469            output.fill(Complex32::default());
470            return Ok(true);
471        }
472        self.validate_legacy_prediction(row.frequencies_hz)?;
473        scratch.begin(correlations);
474        scratch.predicted.clear();
475        if let Some(channels) = self.interpolated_planes(axis, order, row.frequencies_hz) {
476            for channel in channels {
477                let start = scratch.predicted.len();
478                scratch
479                    .predicted
480                    .resize(start + correlations, Complex32::default());
481                sample(channel, &mut scratch.predicted[start..]);
482                scratch.sources.push(Source {
483                    slot: channel,
484                    count: 1,
485                    placement: (start / correlations) as u32,
486                });
487            }
488        }
489        let placed = PlacedRow {
490            start: 0,
491            end: scratch.sources.len(),
492        };
493        if self.uses_legacy_spline() {
494            self.finish_spline(axis, row, placed, scratch, (output, None));
495        } else {
496            self.native_terms(row, correlations, placed, scratch, 0);
497            interpolate_terms(
498                &scratch.predicted,
499                correlations,
500                &scratch.terms,
501                output,
502                0,
503                false,
504            );
505        }
506        Ok(true)
507    }

509    fn prediction_channels<'a>(&self, native: &'a [f64]) -> impl Iterator<Item = f64> + Clone + 'a {
510        let [start, count, step] = self
511            .legacy_prediction_map
512            .as_ref()
513            .and_then(|mapping| mapping.selection)
514            .unwrap_or([0, native.len(), 1]);
515        native
516            .iter()
517            .copied()
518            .skip(start)
519            .step_by(step.max(1))
520            .take(count)
521    }

523    fn prediction_width(&self, native: &[f64]) -> f64 {
524        let mut selected = self.prediction_channels(native);
525        selected
526            .next()
527            .zip(selected.next())
528            .map_or(0., |(first, next)| next - first)
529    }

531    fn prediction_channel(
532        &self,
533        axis: &SpectralAxis,
534        order: Order,
535        frequency: f64,
536        width: f64,
537    ) -> i64 {
538        if let Some(mapping) = self.legacy_prediction_map.as_ref() {
539            mapping
540                .axis
541                .legacy_prediction_channel(frequency, width, order)
542        } else if let Some(channel) = axis.nearest_channel(frequency) {
543            i64::from(channel)
544        } else if axis.in_halo(frequency, width, order) {
545            -2
546        } else {
547            -1
548        }
549    }

552        &self,

562    /// Validate CASA's four-image-plane minimum for live cubic prediction
563    /// before a caller opens the scratch writer. Single-channel and entirely
564    /// unmapped buffers bypass interpolation.
565    /// # Errors
566    /// This is an active legacy cubic buffer with fewer than four image planes.
567    pub fn validate_legacy_prediction(&self, native: &[f64]) -> Result<(), OperatorError> {
568        if let Sampling::Interpolated(axis, Order::Cubic) = &self.sampling
569            && self.legacy_prediction_map.is_some()
570            && !self.uses_legacy_spline()
571            && (2..4).contains(&axis.channels)
572            && self.prediction_channels(native).nth(1).is_some()
573            && self
574                .interpolated_planes(axis, Order::Cubic, native)
575                .is_some()
576        {
577            return Err(OperatorError::SpectralAxis {
578                reason: "legacy cubic scratch prediction requires at least four image channels",
579            });
580        }
581        Ok(())
582    }

590    /// The axis and order by which `row`'s prediction interpolates
591    /// degridded output channels to its native channels, when it does (see
592    /// [`Self::predict_row`]).
593    fn interpolated_prediction(&self, row: &NativeRow<'_>) -> Option<(&SpectralAxis, Order)> {
594        match &self.sampling {
595            Sampling::Interpolated(axis, order)
596                if axis.channels > 1
597                    && self
598                        .prediction_channels(row.frequencies_hz)
599                        .nth(1)
600                        .is_some() =>
601            {
602                Some((axis, *order))
603            }
604            Sampling::Direct | Sampling::Nearest(_) | Sampling::Interpolated(..) => None,
605        }
606    }

608    /// Add `row`'s degridded samples to the batch `scratch` began, as
609    /// [`Self::predict_row`] degrids them; a flagged row adds none, nor an
610    /// autocorrelation unless the resampler is
611    /// [`Self::predicting_autocorrelations`].
612    ///
613    /// A sample that reads the same taps of the same model as the row's
614    /// sample before it (the same kernel cell, location, plane, signs of w
615    /// and pointing gradient, and Taylor variable for a model of several
616    /// terms), at a phase-centre phase within 2⁻³⁰ rad of the first of
617    /// their run, is not placed again:
618    /// its channel reads that sample's prediction. A row's neighbouring
619    /// channels sit a small fraction of a cell apart in uv, so they mostly
620    /// share their kernel's fine offset, and a backend's degrid of a fine
621    /// offset is the same for every sample at it, as CASA's gather reads
622    /// the nearest one: an ALMA row of 4096 channels across 2 GHz placed
623    /// one sample in 45 (#855).
624    ///
625    /// # Errors
626    ///
627    /// A row inconsistent with `operator`.
628    pub fn place_prediction(
629        &self,
630        operator: &MeasurementOperator,
631        model: &PreparedModelGrids,
632        row: &NativeRow<'_>,
633        scratch: &mut PredictionScratch,
634    ) -> Result<PlacedRow, OperatorError> {
635        let npol = operator.polarization().correlations().len();
636        self.validate_row(operator, row, npol)?;
637        let PredictionScratch {
638            placements,
639            sources,
640            ..
641        } = scratch;
642        let start = sources.len();
643        if row.row_flag || (!self.autocorrelations && is_autocorrelation(row)) {
644            return Ok(PlacedRow { start, end: start });
645        }
646        let native = row.frequencies_hz;
647        let prepared = model.layout().planes();
648        let terms = self.basis.data_terms();
649        let Some(mut placing) = RowPlacing::new(operator.cf(), operator.geometry(), row)? else {
650            return Ok(PlacedRow { start, end: start });
651        };
652        if matches!(self.sampling, Sampling::Direct | Sampling::Nearest(_))
653            && terms == 1
654            && placing.uniform.is_some()
655            && monotonic(native)
656        {
657            self.place_runs(operator, model, row, &mut placing, placements, sources);
658            return Ok(PlacedRow {
659                start,
660                end: sources.len(),
661            });
662        }
663        // The run of samples the last placement serves: what they read, the
664        // first one's phase, and its batch index.
665        let mut run: Option<(Gathered, f64, u32)> = None;
666        let mut place = |plane: u32, frequency: (f64, Option<usize>), slot: usize| {
667            if !prepared.contains(plane) {
668                return;
669            }
670            let Some((placement, location)) = locate_sample(
671                operator.cf(),
672                operator.geometry(),
673                self.basis,
674                row,
675                plane,
676                frequency,
677                &mut placing,
678            ) else {
679                return;
680            };
681            let gathered = Gathered::of(&placement, location, terms);
682            let placement_index = match run {
683                Some((first, phase, index))
684                    if first == gathered && (placement.phase - phase).abs() <= SHARED_PHASE_RAD =>
685                {
686                    index
687                }
688                _ => {
689                    let index = u32::try_from(placements.len()).expect("a batch fits u32 indices");
690                    placements.push(placement);
691                    run = Some((gathered, placement.phase, index));
692                    index
693                }
694            };
695            let slot = u32::try_from(slot).expect("a row's channels fit u32");
696            match sources[start..].last_mut() {
697                Some(last)
698                    if last.placement == placement_index && last.slot + last.count == slot =>
699                {
700                    last.count += 1;
701                }
702                _ => sources.push(Source {
703                    slot,
704                    count: 1,
705                    placement: placement_index,
706                }),
707            }
708        };
709        match self.interpolated_prediction(row) {
710            None => {
711                for (channel, frequency_hz) in native.iter().enumerate() {
712                    let plane = match &self.sampling {
713                        Sampling::Direct => Some(0),
714                        Sampling::Nearest(axis) | Sampling::Interpolated(axis, _) => {
715                            self.matching_axis(axis).nearest_channel(*frequency_hz)
716                        }
717                    };
718                    if let Some(plane) = plane {
719                        place(plane, (*frequency_hz, Some(channel)), channel);
720                    }
721                }
722            }
723            Some((axis, order)) => {
724                if let Some(channels) = self.interpolated_planes(axis, order, native) {
725                    for channel in channels {
726                        place(channel, (axis.centre_hz(channel), None), channel as usize);
727                    }
728                }
729            }
730        }
731        Ok(PlacedRow {
732            start,
733            end: sources.len(),
734        })
735    }
736
737    /// [`Self::place_prediction`] of `row`'s channels under direct or
738    /// nearest sampling with one kernel cell and one term, in the runs the
739    /// channel-by-channel placement finds, placing each run's first
740    /// channel. A channel's plane, its location on the grid and its
741    /// phase-centre phase move monotonically with its frequency, which
742    /// `row`'s channels order, so once a channel reads other taps than its
743    /// run's first, or its phase has moved past [`SHARED_PHASE_RAD`], every
744    /// later one does too. A run's end is then found by locating a few
745    /// channels rather than each: from the last run's length, then doubling
746    /// and halving.
747    fn place_runs(
748        &self,
749        operator: &MeasurementOperator,
750        model: &PreparedModelGrids,
751        row: &NativeRow<'_>,
752        placing: &mut RowPlacing,
753        placements: &mut Vec<Placement>,
754        sources: &mut Vec<Source>,
755    ) {
756        let native = row.frequencies_hz;
757        let channels = native.len();
758        let prepared = model.layout().planes();
759        let mut locate = |channel: usize| {
760            let plane = match &self.sampling {
761                Sampling::Direct => Some(0),
762                Sampling::Nearest(axis) | Sampling::Interpolated(axis, _) => {
763                    self.matching_axis(axis).nearest_channel(native[channel])
764                }
765            }
766            .filter(|plane| prepared.contains(*plane))?;
767            locate_sample(
768                operator.cf(),
769                operator.geometry(),
770                self.basis,
771                row,
772                plane,
773                (native[channel], Some(channel)),
774                placing,
775            )
776        };
777        let (mut channel, mut length) = (0, 1);
778        while channel < channels {
779            let Some((placement, location)) = locate(channel) else {
780                channel += 1;
781                continue;
782            };
783            let (gathered, phase) = (Gathered::of(&placement, location, 1), placement.phase);
784            let index = u32::try_from(placements.len()).expect("a batch fits u32 indices");
785            placements.push(placement);
786            let mut shares = |probe: usize| {
787                locate(probe).is_some_and(|(placement, location)| {
788                    Gathered::of(&placement, location, 1) == gathered
789                        && (placement.phase - phase).abs() <= SHARED_PHASE_RAD
790                })
791            };
792            // The last channel known to share, and the first known not to.
793            let (mut last, mut past) = (channel, channels);
794            let expected = channel + length - 1;
795            if expected > channel && expected < channels {
796                if shares(expected) {
797                    last = expected;
798                } else {
799                    past = expected;
800                }
801            }
802            let mut step = 1;
803            while past == channels && last + step < channels {
804                if shares(last + step) {
805                    last += step;
806                    step *= 2;
807                } else {
808                    past = last + step;
809                }
810            }
811            while past - last > 1 {
812                let middle = last + (past - last) / 2;
813                if shares(middle) {
814                    last = middle;
815                } else {
816                    past = middle;
817                }
818            }
819            sources.push(Source {
820                slot: u32::try_from(channel).expect("a row's channels fit u32"),
821                count: u32::try_from(last + 1 - channel).expect("a row's channels fit u32"),
822                placement: index,
823            });
824            length = last + 1 - channel;
825            channel = last + 1;
826        }
827    }
828
829    /// Add `row`'s native terms to the batch `scratch` began: one per native
830    /// channel, the row's cells numbered from `first_cell`, interpolating
831    /// its samples `placed` in the batch as [`Self::predict_row`] does. A
832    /// cell whose channel is off the output axis and its halo, or whose row
833    /// placed nothing, gets the zero model; a flagged correlation keeps it
834    /// zero.
835    ///
836    /// # Panics
837    /// Active legacy spline interpolation needs [`Self::finish_prediction`].
838    pub fn native_terms(
839        &self,
840        row: &NativeRow<'_>,
841        npol: usize,
842        placed: PlacedRow,
843        scratch: &mut PredictionScratch,
844        first_cell: u32,
845    ) {
846        let PredictionScratch {
847            sources,
848            slots,
849            terms,
850            ..
851        } = scratch;
852        let native = row.frequencies_hz;
853        let sources = &sources[placed.start..placed.end];
854        // A row without a flag packs no flag bits.
855        let flagged = row.flags.contains(&true);
856        let mut term = |channel: usize, points: &[(u32, f64)]| {
857            let mut term = NativeTerm {
858                cell: first_cell + channel as u32,
859                points: [NO_CELL; STENCIL],
860                factors: [0.0; STENCIL],
861                flags: if flagged {
862                    flag_bits(row, npol, channel)
863                } else {
864                    0
865                },
866            };
867            for (slot, (point, factor)) in points.iter().enumerate() {
868                term.points[slot] = *point;
869                term.factors[slot] = *factor as f32;
870            }
871            terms.push(term);
872        };
873        let Some((axis, order)) = self.interpolated_prediction(row) else {
874            // Each native channel reads its source's placement, in slot
875            // order; a channel no source covers reads nothing.
876            let mut channel = 0;
877            for source in sources {
878                debug_assert!(source.slot as usize >= channel, "sources run in slot order");
879                for unplaced in channel..source.slot as usize {
880                    term(unplaced, &[]);
881                }
882                for placed in source.slots() {
883                    term(placed, &[(source.placement, 1.0)]);
884                }
885                channel = source.slots().end;
886            }
887            for unplaced in channel..native.len() {
888                term(unplaced, &[]);
889            }
890            return;
891        };
892        assert!(!self.uses_legacy_spline(), "a spline needs host prediction");
893        // `FTMachine::interpolateFrequencyFromgrid`: the output-channel
894        // predictions on CASA's image-frequency grid, refined to
895        // `floor(width ratio)` repeated points per channel when output
896        // channels are wider than native ones, interpolated to every native
897        // channel that maps to the axis or lies in its halo (`chanMap` 0… or
898        // −2; −1 stays zero); the end points extrapolate (casacore
899        // `InterpolateArray1D`, linear or cubic).
900        index_sources(slots, sources, axis.channels as usize);
901        let native_width_hz = self.prediction_width(native);
902        let width = (axis.increment_hz / native_width_hz).abs();
903        let legacy = self.legacy_prediction_map.is_some();
904        let per_channel = if legacy {
905            1
906        } else {
907            axis.fine_samples_per_plane(width)
908        };
909        let fine_increment = axis.increment_hz / per_channel as f64;
910        let fine_start = if per_channel > 1 {
911            axis.first_hz - axis.increment_hz / 2.0 + fine_increment / 2.0
912        } else {
913            axis.first_hz
914        };
915        let points = axis.channels as usize * per_channel;
916        let point = |index: usize| {
917            if legacy || (width <= 1.0 && (axis.reporting_axis.is_some() || axis.tabular.is_some()))
918            {
919                axis.centre_hz(index as u32)
920            } else {
921                fine_start + index as f64 * fine_increment
922            }
923        };
924        for (channel, frequency_hz) in native.iter().enumerate() {
925            if self.prediction_channel(axis, order, *frequency_hz, native_width_hz) == -1 {
926                term(channel, &[]);
927                continue;
928            }
929            let position = if legacy || width <= 1.0 {
930                axis.tabular_pixel(*frequency_hz)
931            } else {
932                None
933            }
934            .unwrap_or_else(|| (frequency_hz - fine_start) / fine_increment);
935            let slot = |point: usize| slots[point / per_channel];
936            match order {
937                Order::Linear => {
938                    let left = (position.floor().max(0.0) as usize).min(points - 2);
939                    let fraction = (frequency_hz - point(left)) / (point(left + 1) - point(left));
940                    // casacore's `y1 + frac·(y2 − y1)`, as two factors.
941                    term(
942                        channel,
943                        &[(slot(left), 1.0 - fraction), (slot(left + 1), fraction)],
944                    );
945                }
946                Order::Cubic => {
947                    // `binarySearchBrackets`: the point at the frequency,
948                    // else the first beyond it.
949                    let bracket = if position <= 0.0 {
950                        0
951                    } else {
952                        (position.ceil() as usize).min(points)
953                    };
954                    let stencil = Stencil::cubic(points, bracket, *frequency_hz, point);
955                    let mut factors = [(NO_CELL, 0.0); STENCIL];
956                    for (entry, (index, factor)) in factors.iter_mut().zip(stencil.terms()) {
957                        *entry = (slot(index), factor);
958                    }
959                    term(channel, &factors[..stencil.len]);
960                }
961            }
962        }
963    }

965    /// Write `row`'s model visibilities to `out` as `[channel][correlation]`
966    /// from its samples `placed` in the batch `scratch` applied, as
967    /// [`Self::predict_row`] does.
968    ///
969    /// # Errors
970    ///
971    /// An `out` of another length than the row's values.
972    pub fn finish_prediction(
973        &self,
974        operator: &MeasurementOperator,
975        row: &NativeRow<'_>,
976        placed: PlacedRow,
977        scratch: &mut PredictionScratch,
978        out: &mut [Complex32],
979    ) -> Result<(), OperatorError> {
980        self.finish_prediction_with_coverage(operator, row, placed, scratch, (out, None))
981    }
982
983    fn finish_prediction_with_coverage(
984        &self,
985        operator: &MeasurementOperator,
986        row: &NativeRow<'_>,
987        placed: PlacedRow,
988        scratch: &mut PredictionScratch,
989        (out, coverage): (&mut [Complex32], Option<&mut [bool]>),
990    ) -> Result<(), OperatorError> {
991        if out.len() != row.values.len() {
992            return Err(OperatorError::NativeRow {
993                reason: "the prediction holds one value per selected sample",
994            });
995        }
996        if coverage
997            .as_ref()
998            .is_some_and(|bits| bits.len() != out.len())
999        {
1000            return Err(OperatorError::NativeRow {
1001                reason: "coverage holds one bit per selected sample",
1002            });
1003        }
1004        if self.uses_legacy_spline()
1005            && let Some((axis, _)) = self.interpolated_prediction(row)
1006        {
1007            self.finish_spline(axis, row, placed, scratch, (out, coverage));
1008            return Ok(());
1009        }
1010        let npol = operator.polarization().correlations().len();
1011        let first = scratch.terms.len();
1012        self.native_terms(row, npol, placed, scratch, 0);
1013        if let Some(coverage) = coverage {
1014            let polarization = operator.polarization().pol_map();
1015            let legacy = self
1016                .legacy_prediction_map
1017                .as_ref()
1018                .and(self.interpolated_prediction(row))
1019                .map(|(axis, order)| {
1020                    (
1021                        axis,
1022                        order,
1023                        self.prediction_width(row.frequencies_hz),
1024                        self.interpolated_planes(axis, order, row.frequencies_hz)
1025                            .is_some(),
1026                    )
1027                });
1028            for (channel, (term, covered)) in scratch.terms[first..]
1029                .iter()
1030                .zip(coverage.chunks_exact_mut(npol))
1031                .enumerate()
1032            {
1033                let has_sample = term.points.iter().any(|&point| point != NO_CELL);
1034                // Legacy swapyz copies its values even if the degrid skipped
1035                // a row or correlation. In that case the updated model is zero.
1036                let legacy_assignment = legacy.map(|(axis, order, width, reaches_grid)| {
1037                    reaches_grid
1038                        && self.prediction_channel(axis, order, row.frequencies_hz[channel], width)
1039                            != -1
1040                });
1041                for (index, value) in covered.iter_mut().enumerate() {
1042                    let assigned =
1043                        legacy_assignment.unwrap_or(has_sample && polarization[index].is_some());
1044                    *value = assigned && term.flags & (1 << index) == 0;
1045                }
1046            }
1047        }
1048        interpolate_terms(
1049            &scratch.predicted,
1050            npol,
1051            &scratch.terms[first..],
1052            out,
1053            0,
1054            false,
1055        );
1056        scratch.terms.truncate(first);
1057        Ok(())
1058    }
1059
1060    fn finish_spline(
1061        &self,
1062        axis: &SpectralAxis,
1063        row: &NativeRow<'_>,
1064        placed: PlacedRow,
1065        scratch: &mut PredictionScratch,
1066        (out, mut coverage): (&mut [Complex32], Option<&mut [bool]>),
1067    ) {
1068        out.fill(Complex32::default());
1069        if let Some(bits) = coverage.as_deref_mut() {
1070            bits.fill(false);
1071        }
1072        let native = row.frequencies_hz;
1073        if self
1074            .interpolated_planes(axis, Order::Cubic, native)
1075            .is_none()
1076        {
1077            return;
1078        }
1079        let npol = scratch.npol;
1080        let sources = &scratch.sources[placed.start..placed.end];
1081        let PredictionScratch {
1082            slots,
1083            spline,
1084            predicted,
1085            ..
1086        } = scratch;
1087        index_sources(slots, sources, axis.channels as usize);
1088        let value = |plane: usize, pol: usize| match slots[plane] {
1089            NO_CELL => Complex32::default(),
1090            sample => predicted[sample as usize * npol + pol],
1091        };
1092        if !sources.is_empty() {
1093            spline.prepare(axis, npol, value);
1094        }
1095        let width = self.prediction_width(native);
1096        for (channel, &frequency) in native.iter().enumerate() {
1097            if self.prediction_channel(axis, Order::Cubic, frequency, width) == -1 {
1098                continue;
1099            }
1100            let start = channel * npol;
1101            let flags = &row.flags[start..start + npol];
1102            if let Some(bits) = coverage.as_deref_mut() {
1103                for (covered, &flag) in bits[start..start + npol].iter_mut().zip(flags) {
1104                    *covered = !flag;
1105                }
1106            }
1107            if !sources.is_empty() {
1108                spline.evaluate(axis, frequency, value, flags, &mut out[start..start + npol]);
1109            }
1110        }
1111    }
