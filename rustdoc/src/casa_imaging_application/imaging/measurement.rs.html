17/// The operator of one image domain and the resampler that places rows on
18/// its planes.
19pub(crate) struct DomainOperator {
20    pub(crate) operator: MeasurementOperator,
21    pub(crate) resampler: SpectralResampler,
22}

24/// Correlation types every block delivers, in block order: the first
25/// source's selection, which every source must share.
26pub(crate) fn selected_correlations(
27    problem: &CompiledProblem,
28) -> Result<Vec<CorrelationType>, ImagingError> {
29    let mut layouts = problem
30        .selected_observation()
31        .read_set()
32        .sources()
33        .iter()
34        .flat_map(|source| source.selection().correlations())
35        .map(|selection| {
36            selection
37                .products()
38                .iter()
39                .map(|product| product.correlation_type())
40                .collect::<Vec<_>>()
41        });
42    let first = layouts.next().ok_or(ImagingError::Unsupported {
43        reason: "the selection names no correlations",
44    })?;
45    if layouts.any(|layout| layout != first) {
46        return Err(ImagingError::Unsupported {
47            reason: "selected polarization setups differ in their correlations",
48        });
49    }
50    Ok(first)
51}

53/// The grid basis of the compiled reconstruction basis. A Taylor basis
54/// expands about the image's reference frequency. Taylor-via-channel-major
55/// (CASA `mvc`) always carries a primary-beam response and waits for the
56/// primary-beam operators (IF-3).
57pub(crate) fn basis(problem: &CompiledProblem) -> Result<Basis, ImagingError> {
58    Ok(match problem.reconstruction().basis() {
59        ReconstructionBasis::Constant => Basis::Constant,
60        ReconstructionBasis::Taylor { terms } => Basis::Taylor {
61            terms: terms as u32,
62            reference_hz: reference_frequency_hz(problem)?,
63        },
64        ReconstructionBasis::ChannelLocal { channels } => Basis::ChannelLocal {
65            planes: channels as u32,
66        },
67        ReconstructionBasis::TaylorViaChannelMajor { .. } => {
68            return Err(ImagingError::Unsupported {
69                reason: "Taylor terms via channel cubes need the primary-beam operators",
70            });
71        }
72    })
73}

75/// The image's spectral reference frequency (CASA `reffreq`).
76pub(crate) fn reference_frequency_hz(problem: &CompiledProblem) -> Result<f64, ImagingError> {
77    match problem.geometry().spectral().wcs() {
78        SpectralWcs::Linear {
79            reference_frequency_hz,
80            ..
81        } => Ok(*reference_frequency_hz),
82        SpectralWcs::Tabular { .. } => Err(ImagingError::Unsupported {
83            reason: "a tabular spectral axis has no uniform channel width",
84        }),
85    }
86}

88/// Grid precision of plan decision D2 (`gridprecision = auto`): f64 for the
89/// constant and Taylor bases, f32 for channel-local cubes.
90pub(crate) const fn precision(basis: Basis) -> GridPrecision {
91    match basis {
92        Basis::Constant | Basis::Taylor { .. } => GridPrecision::F64,
93        Basis::ChannelLocal { .. } => GridPrecision::F32,
94    }
95}

97/// The operator of `domain` for the standard kernel set.
98pub(crate) fn domain_operator(
99    problem: &CompiledProblem,
100    domain: &CompiledImageDomain,
101    correlations: &[CorrelationType],
102) -> Result<DomainOperator, ImagingError> {
103    let geometry = GridGeometry::new(image_extent(domain)?, GridPadding::CasaComposite)?;
104    let polarization = PolarizationRouting::compile(
105        correlations,
106        problem.reconstruction().polarization().coordinates(),
107    )?;
108    let basis = basis(problem)?;
109    let resampler = resampler(problem, basis)?;
110    let cf = Spheroidal::new(&geometry, &polarization);
111    Ok(DomainOperator {
112        operator: MeasurementOperator::new(
113            geometry,
114            basis,
115            polarization,
116            Box::new(cf),
117            precision(basis),
118        ),
119        resampler,
120    })
121}

123fn image_extent(domain: &CompiledImageDomain) -> Result<ImageExtent, ImagingError> {
124    let direction = domain.direction();
125    let reference = direction.reference_pixel();
126    if reference
127        .iter()
128        .any(|pixel| pixel.fract() != 0.0 || *pixel < 0.0)
129    {
130        return Err(ImagingError::Unsupported {
131            reason: "the image reference pixel must be a whole pixel",
132        });
133    }
134    Ok(ImageExtent {
135        shape: domain.shape().pixels(),
136        increment_rad: direction.increment_rad(),
137        reference_pixel: [reference[0] as usize, reference[1] as usize],
138    })
139}

141/// CASA `FTMachine` channel mapping for a channel-local basis; every native
142/// channel on plane 0 otherwise.
143fn resampler(problem: &CompiledProblem, basis: Basis) -> Result<SpectralResampler, ImagingError> {
144    let Basis::ChannelLocal { planes } = basis else {
145        return Ok(SpectralResampler::direct(basis)?);
146    };
147    let SpectralWcs::Linear {
148        reference_pixel,
149        reference_frequency_hz,
150        increment_hz,
151        ..
152    } = problem.geometry().spectral().wcs()
153    else {
154        return Err(ImagingError::Unsupported {
155            reason: "a tabular spectral axis has no uniform channel width",
156        });
157    };
158    let kernel = match problem.science().spectral().sampling().kernel() {
159        SpectralKernel::Nearest | SpectralKernel::Identity => {
160            casa_imaging_operator::SpectralKernel::Nearest
161        }
162        SpectralKernel::Linear => casa_imaging_operator::SpectralKernel::Linear,
163        SpectralKernel::Cubic | SpectralKernel::ChannelIntegration { .. } => {
164            return Err(ImagingError::Unsupported {
165                reason: "cube interpolation is nearest or linear",
166            });
167        }
168    };
169    let first_hz = reference_frequency_hz - reference_pixel * increment_hz;
170    Ok(SpectralResampler::channel_local(
171        SpectralAxis::new(first_hz, *increment_hz, planes)?,
172        kernel,
173    ))
174}

176/// Relative allowance for the Doppler factor between the stored frame of the
177/// spectral-window catalog and the output frame rows are delivered in
178/// (3000 km/s); the pass checks every row against the widened bound.
179const FRAME_MARGIN: f64 = 0.01;
180
181/// The widest spacing between adjacent selected native channels of any
182/// selected spectral window, from the storage owner's `CHAN_FREQ` catalog,
183/// widened by [`FRAME_MARGIN`]. A window without a catalog makes it
184/// infinite, so every wave holds the whole model.
185pub(crate) fn native_spacing_hz(problem: &CompiledProblem) -> f64 {
186    let mut widest = 0.0_f64;
187    for window in problem
188        .selected_observation()
189        .read_set()
190        .sources()
191        .iter()
192        .flat_map(|source| source.selection().spectral_windows())
193    {
194        let Some(catalog) = window.coordinate_catalog() else {
195            return f64::INFINITY;
196        };
197        for pair in window.channel_indices().windows(2) {
198            let (Some(first), Some(second)) = (
199                catalog.channel_frequency_hz(pair[0] as usize),
200                catalog.channel_frequency_hz(pair[1] as usize),
201            ) else {
202                return f64::INFINITY;
203            };
204            widest = widest.max((second - first).abs());
205        }
206    }
207    widest * (1.0 + FRAME_MARGIN)
208}

210/// Shape of the weight-density grid of `domain`: one global plane on CASA's
211/// `VisImagingWeight` cells, or one plane per output channel, plus the
212/// compiled padding planes on each side, on `BriggsCubeWeightor` cells.
213pub(crate) fn density_shape(
214    problem: &CompiledProblem,
215    domain: &CompiledImageDomain,
216) -> Result<DensityGridShape, ImagingError> {
217    let [width, height] = domain.shape().pixels();
218    let (planes, padding, rule) = match problem.weighting().density_scope() {
219        WeightDensityScope::PerOutputChannel => {
220            // CASA always pads the cube density axis
221            // (`BriggsCubeWeightor::estimateSwingChanPad`); a contract without
222            // the padding cannot reproduce it.
223            let padding = problem.weighting().casa_cube_density_padding().ok_or(
224                ImagingError::Unsupported {
225                    reason: "a per-channel weight density needs its compiled padding",
226                },
227            )?;
228            let planes = padding
229                .checked_mul(2)
230                .and_then(|padding| {
231                    padding.checked_add(problem.geometry().spectral().output_channels())
232                })
233                .ok_or(ImagingError::Unsupported {
234                    reason: "the padded density axis overflows",
235                })?;
236            let padding = u32::try_from(padding).map_err(|_| ImagingError::Unsupported {
237                reason: "the padded density axis overflows",
238            })?;
239            (planes, padding, DensityCellRule::Cube)
240        }
241        WeightDensityScope::NotApplicable | WeightDensityScope::GlobalSelection => {
242            (1, 0, DensityCellRule::Standard)
243        }
244    };
245    Ok(DensityGridShape {
246        width,
247        height,
248        planes,
249        padding,
250        increment_rad: domain.direction().increment_rad(),
251        rule,
252    })
