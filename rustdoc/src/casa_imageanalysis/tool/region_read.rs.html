15/// Resolves region bounding box and optional per-pixel membership mask.
16pub fn resolve_region_section(
17    image: &ImageHandle,
18    region: Option<SubimageRegion<'_>>,
19    include_membership: bool,
20) -> ImageToolResult<ImageRegionSection> {
21    if let ImageHandle::Float(storage) = image {
22        Ok(storage.with_float_view(|image| {
23            ImageRegionSection::resolve(image, region, include_membership)
24        })?)
25    } else {
26        Ok(
27            with_pixels!(image, storage => with_storage!(storage, image =>
28            ImageRegionSection::resolve(image.as_ref(), region, include_membership)))?,
29        )
30    }
31}
32
33/// Resolves a region section into a strided [`Chunk`] and optional membership mask.
34pub fn read_region_section(
35    image: &ImageHandle,
36    region: Option<SubimageRegion<'_>>,
37    include_membership: bool,
38) -> ImageToolResult<(Chunk, Option<ArrayD<bool>>)> {
39    let section = resolve_region_section(image, region, include_membership)?;
40    Ok((
41        Chunk {
42            start: section.start,
43            size: section.count,
44            stride: section.stride,
45        },
46        section.membership,
47    ))
48}

50/// Reads region pixel/mask arrays for CASA `ia.getregion`, including axis collapse and LEL mask.
51#[allow(clippy::too_many_arguments)]
52pub fn read_region_casa(
53    attached: &AttachedImage,
54    region: Option<SubimageRegion<'_>>,
55    axes: &[i64],
56    mask: &str,
57    list: bool,
58    dropdeg: bool,
59    getmask: bool,
60    stretch: bool,
61) -> ImageToolResult<(ChunkData, Vec<ToolLogLine>)> {
62    read_locked(attached, |image| {
63        let shape = image.shape();
64        let axes = collapse_axes(axes, shape.len())?;
65        let (chunk, membership) = read_region_section(image, region, getmask || !axes.is_empty())?;
66        let expression = if mask.is_empty() {
67            None
68        } else {
69            Some(evaluate_image_tool_mask_section(
70                mask,
71                shape,
72                image.coordinates(),
73                (&chunk.start, &chunk.size, &chunk.stride),
74                stretch,
75            )?)
76        };
77        let data = chunk.read_from(
78            image,
79            &ChunkRead {
80                axes: &axes,
81                dropdeg,
82                getmask,
83            },
84            and_mask(membership, expression),
85        )?;
86        let logs = list
87            .then(|| {
88                let end = chunk
89                    .start
90                    .iter()
91                    .zip(&chunk.size)
92                    .zip(&chunk.stride)
93                    .map(|((&start, &count), &step)| start + (count - 1) * step)
94                    .collect::<Vec<_>>();
95                ToolLogLine::info(format!(
96                    "Selected bounding box {:?} to {end:?}",
97                    chunk.start
98                ))
99            })
100            .into_iter()
101            .collect();
102        Ok((data, logs))
103    })
104}
