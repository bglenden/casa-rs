2//! A traversal's times split into slices planned together, so that several
3//! workers can each take a slice of a partition and still finalize every
4//! scientific group once: a group the cuts divide is joined across the
5//! slices by one shared completion.
6//!
7//! A consumer plans a [`TimeSplit`] of its request, binds each slice's
8//! request on its own worker, and gives each worker its slice's view of the
9//! split's [`SplitCompletion`]. It writes the same code as for one
10//! traversal: whatever the cuts, every group becomes ready once, after the
11//! input and the work of each of its parts. A slice that ends without
12//! completing its parts abandons their groups, which the others then
13//! report instead of waiting.
14

24/// A request's times cut into consecutive slices that tile them, planned
25/// together: each slice is the request with its own times, and the split
26/// knows which scientific groups the cuts divide and which slices hold
27/// their parts.
28///
29/// Group keys do not depend on the time selection (they are those of the
30/// grid's field-selected times), so a divided group arrives in
31/// each slice holding part of it under the same key, marked
32/// [`clipped`](crate::GroupFragment::clipped) there; the split's
33/// [`SplitCompletion`] joins the parts. A slice may hold no time of a
34/// partition: its traversal then delivers nothing of it.
35#[derive(Clone, Debug)]
36pub struct TimeSplit {
37    slices: Vec<AccessRequest>,
38    divided: BTreeMap<GroupKey, BTreeSet<usize>>,
39}
40
41impl TimeSplit {
42    /// Split `request` at the consecutive ranges `cuts`: slice `i` holds the
43    /// request's times within `cuts[i]`. Each range ends where the next
44    /// starts, so a planner cuts at the epochs it chose (aligned to the
45    /// store's natural time blocks through [`TimeRange::of_grid`], say),
46    /// and the first must start no later and the last end no earlier than
47    /// the request's times in every partition, so the slices tile them.
48    /// The store's group index ([`Observation::group_keys`]) gives the
49    /// groups the cuts divide; nothing is read.
50    ///
51    /// # Errors
52    ///
53    /// [`ObservationError::InvalidRequest`] when there are no cuts, they are
54    /// not consecutive ranges of one origin and scale (the request's own, if
55    /// it selects a range), or they leave a time of the request outside
56    /// every slice; and as [`Observation::estimate`] for the request and
57    /// each slice.
58    pub fn plan(
59        observation: &(impl Observation + ?Sized),
60        request: &AccessRequest,
61        cuts: &[TimeRange],
62    ) -> Result<Self, ObservationError> {
63        let invalid = |reason: String| ObservationError::InvalidRequest { reason };
64        let (first, rest) = cuts
65            .split_first()
66            .ok_or_else(|| invalid("a time split needs at least one slice".to_string()))?;
67        for (earlier, later) in cuts.iter().zip(rest) {
68            if !later.same_frame(first) || earlier.end_s != later.start_s {
69                return Err(invalid(format!(
70                    "time split slices [{}, {}) s and [{}, {}) s are not consecutive ranges of \
71                     one origin and scale",
72                    earlier.start_s, earlier.end_s, later.start_s, later.end_s
73                )));
74            }
75        }
76        let slices = cuts
77            .iter()
78            .map(|cut| {
79                Ok(AccessRequest {
80                    times: TimeSelection::Within(within(request.times, *cut)?),
81                    ..request.clone()
82                })
83            })
84            .collect::<Result<Vec<_>, ObservationError>>()?;
85        let whole = observation.estimate(request)?;
86        let parts = slices
87            .iter()
88            .map(|slice| observation.estimate(slice))
89            .collect::<Result<Vec<_>, _>>()?;
90        check_tiling(&whole, &parts)?;
91        let divided = if request.grouping == Grouping::None {
92            BTreeMap::new()
93        } else {
94            divided(observation, request, &parts)?
95        };
96        Ok(Self { slices, divided })
97    }
98
99    /// The slices' requests, in time order.
100    #[must_use]
101    pub fn slices(&self) -> &[AccessRequest] {
102        &self.slices
103    }
104
105    /// The groups the cuts divide, each with the slices holding part of
106    /// it: two or more.
107    #[must_use]
108    pub fn divided(&self) -> &BTreeMap<GroupKey, BTreeSet<usize>> {
109        &self.divided
110    }
111
112    /// A completion the slices' traversals share, joining the parts of
113    /// every divided group.
114    #[must_use]
115    pub fn completion(&self) -> SplitCompletion {
116        SplitCompletion {
117            joins: Arc::new(Mutex::new(Joins::divided(self.divided.clone()))),
118            taken: Arc::new(Mutex::new(vec![false; self.slices.len()])),
119        }
120    }
121}

123/// `cut` within the request's times: the cut itself when the request reads
124/// every time, else its intersection with the request's range.
125fn within(times: TimeSelection, cut: TimeRange) -> Result<TimeRange, ObservationError> {
126    let TimeSelection::Within(range) = times else {
127        return Ok(cut);
128    };
129    if !range.same_frame(&cut) {
130        return Err(ObservationError::InvalidRequest {
131            reason: "a time split's slices are stated in another origin or scale than the \
132                     request's time range"
133                .to_string(),
134        });
135    }
136    let start_s = range.start_s.max(cut.start_s);
137    Ok(TimeRange {
138        start_s,
139        end_s: range.end_s.min(cut.end_s).max(start_s),
140        ..cut
141    })
142}

144/// Refuse slices whose selected times, taken in order, are not the
145/// request's in every partition: each held once, none missing.
146fn check_tiling(whole: &Estimate, parts: &[Estimate]) -> Result<(), ObservationError> {
147    for (index, partition) in whole.partitions.iter().enumerate() {
148        // The slices' runs in order, a run a cut divides joined again.
149        let mut joined: Vec<Range<usize>> = Vec::new();
150        let mut ordered = true;
151        for run in parts
152            .iter()
153            .flat_map(|part| part.partitions[index].time_runs.runs())
154        {
155            match joined.last_mut() {
156                Some(last) if last.end == run.start => last.end = run.end,
157                Some(last) if last.end > run.start => ordered = false,
158                _ => joined.push(run.clone()),
159            }
160        }
161        if !ordered || joined != partition.time_runs.runs() {
162            return Err(ObservationError::InvalidRequest {
163                reason: format!(
164                    "the time split's slices do not tile {:?}'s selected times {:?}",
165                    partition.partition,
166                    partition.time_runs.span()
167                ),
168            });
169        }
170    }
171    Ok(())
172}

174/// The groups more than one slice holds part of, with the slices holding
175/// part of each.
176fn divided(
177    observation: &(impl Observation + ?Sized),
178    request: &AccessRequest,
179    parts: &[Estimate],
180) -> Result<BTreeMap<GroupKey, BTreeSet<usize>>, ObservationError> {
181    let mut divided = BTreeMap::new();
182    for (index, id) in request.partitions.iter().enumerate() {
183        let Some(keys) =
184            observation.group_keys(*id, &request.data_group, request.grouping, &request.fields)?
185        else {
186            continue;
187        };
188        let mut holders = BTreeMap::<GroupKey, BTreeSet<usize>>::new();
189        for (slice, part) in parts.iter().enumerate() {
190            for run in part.partitions[index].time_runs.runs() {
191                let held = keys
192                    .get(run.clone())
193                    .ok_or_else(|| ObservationError::InvalidStore {
194                        reason: format!("{id:?}'s group index does not cover its selected times"),
195                    })?;
196                for key in held.iter().flatten() {
197                    holders.entry(*key).or_default().insert(slice);
198                }
199            }
200        }
201        divided.extend(holders.into_iter().filter(|(_, slices)| slices.len() > 1));
202    }
203    Ok(divided)
204}

206/// The completion a [`TimeSplit`]'s traversals share: the joins of
207/// [`GroupCompletion`](crate::GroupCompletion) over every slice, so that a
208/// group the cuts divide is ready once, after each part's input has
209/// completed in its slice and all work begun on it has finished, however
210/// the slices interleave.
211///
212/// Each slice's consumer works through its own [`SliceCompletion`]
213/// ([`Self::slice`]), which is `Send`; any one of them may finish a token
214/// another began. Whichever call makes a group ready returns `true`, once
215/// across the slices. A token dropped unfinished abandons its group, as
216/// with one traversal, and so does a slice's view that ends
217/// ([`SliceCompletion::end`]) or is dropped while a part of the group it
218/// holds is open: a slice whose traversal failed is reported, not waited
219/// for. Such a slice is not run again within the split: its groups are
220/// abandoned, so a retry plans a fresh completion.
221#[derive(Clone, Debug)]
222pub struct SplitCompletion {
223    joins: Arc<Mutex<Joins>>,
224    /// Which slices' views have been given.
225    taken: Arc<Mutex<Vec<bool>>>,
226}
227
228impl SplitCompletion {
229    /// Slice `index`'s view, in the split's slice order. Each slice has one
230    /// view, given once.
231    ///
232    /// # Panics
233    ///
234    /// If the split has no slice `index`, or its view was given before.
235    #[must_use]
236    pub fn slice(&self, index: usize) -> SliceCompletion {
237        let mut taken = self.taken.lock().unwrap_or_else(PoisonError::into_inner);
238        let slices = taken.len();
239        let given = taken
240            .get_mut(index)
241            .unwrap_or_else(|| panic!("slice {index} of a split of {slices}"));
242        assert!(!*given, "slice {index}'s view was given before");
243        *given = true;
244        SliceCompletion {
245            joins: Arc::clone(&self.joins),
246            slice: index,
247            ended: false,
248        }
249    }
250
251    /// Whether no work is pending, no closed group awaits its work and every
252    /// divided group's parts have all completed: true once every slice's
253    /// traversal is consumed and its work finished.
254    ///
255    /// # Errors
256    ///
257    /// [`ObservationError::WorkAbandoned`] if a unit of work was abandoned,
258    /// or a slice ended with a part open.
259    pub fn is_idle(&self) -> Result<bool, ObservationError> {
260        self.lock().is_idle()
261    }
262
263    fn lock(&self) -> std::sync::MutexGuard<'_, Joins> {
264        self.joins.lock().unwrap_or_else(PoisonError::into_inner)
265    }
266}

268/// One slice's view of a [`SplitCompletion`]: the calls of
269/// [`GroupCompletion`](crate::GroupCompletion), for the consumer of that
270/// slice's traversal, and [`Self::end`] when it is over. Dropped before
271/// its traversal has completed every part it holds (the traversal failed,
272/// or its worker did), it abandons their groups.
273#[derive(Debug)]
274pub struct SliceCompletion {
275    joins: Arc<Mutex<Joins>>,
276    slice: usize,
277    ended: bool,
278}
279
280impl SliceCompletion {
281    /// A unit of work on `key` has started in this slice.
282    ///
283    /// # Errors
284    ///
285    /// If this slice has completed its part of the group's input, or work
286    /// on the group was abandoned.
287    pub fn begin(&self, key: GroupKey) -> Result<WorkToken, ObservationError> {
288        self.lock().begin(self.slice, key)
289    }
290
291    /// The unit of work `token` names has finished; `true` if that makes
292    /// its group ready.
293    ///
294    /// # Errors
295    ///
296    /// [`ObservationError::WorkAbandoned`] if a unit of work was abandoned.
297    ///
298    /// # Panics
299    ///
300    /// If `token` was issued by another completion.
301    pub fn finish(&self, token: WorkToken) -> Result<bool, ObservationError> {
302        self.lock().finish(token)
303    }
304
305    /// This slice's part of the group's input is complete; `true` if every
306    /// part's is and no work on the group is pending, so it is ready now.
307    ///
308    /// # Errors
309    ///
310    /// If this slice closed the group, or a later group of its series,
311    /// already; [`ObservationError::WorkAbandoned`] if a unit of work was
312    /// abandoned.
313    pub fn close(&self, key: GroupKey) -> Result<bool, ObservationError> {
314        self.lock().close(self.slice, key)
315    }
316
317    /// Units of work pending on `key`, in every slice.
318    #[must_use]
319    pub fn pending(&self, key: GroupKey) -> usize {
320        self.lock().pending(key)
321    }
322
323    /// This slice's traversal is over. A divided group whose part it has
324    /// not completed is abandoned: its input will not complete, so the
325    /// other slices' calls and [`SplitCompletion::is_idle`] report it.
326    ///
327    /// # Errors
328    ///
329    /// [`ObservationError::WorkAbandoned`] when the slice left a part open,
330    /// or work was abandoned before.
331    pub fn end(mut self) -> Result<(), ObservationError> {
332        self.ended = true;
333        self.lock().end(self.slice)
334    }
335
336    fn lock(&self) -> std::sync::MutexGuard<'_, Joins> {
337        self.joins.lock().unwrap_or_else(PoisonError::into_inner)
338    }
339}
340
341impl Drop for SliceCompletion {
342    fn drop(&mut self) {
343        if !self.ended {
344            // Reported to the other slices and the split.
345            let _ = self.lock().end(self.slice);
346        }
347    }
348}

355    fn key(bin: u64) -> GroupKey {
356        GroupKey {
357            partition: PartitionId(0),
358            field: None,
359            scan: None,
360            bin,
361        }
362    }

364    /// The completion of three slices dividing the groups of `parts`, each
365    /// among the slices named.
366    fn split(parts: &[(u64, &[usize])]) -> SplitCompletion {
367        let divided = parts
368            .iter()
369            .map(|(bin, slices)| (key(*bin), slices.iter().copied().collect()))
370            .collect();
371        SplitCompletion {
372            joins: Arc::new(Mutex::new(Joins::divided(divided))),
373            taken: Arc::new(Mutex::new(vec![false; 3])),
374        }
375    }

377    /// A divided group is ready once, after every part has closed and the
378    /// work begun on any part has finished: a later slice closing its part
379    /// first does not release it while an earlier slice has yet to begin.
380    #[test]
381    fn a_divided_group_waits_for_every_part() {
382        let completion = split(&[(1, &[0, 1])]);
383        let (early, late) = (completion.slice(0), completion.slice(1));
384        assert!(!late.close(key(1)).expect("late part"), "one part of two");
385        let work = early.begin(key(1)).expect("early work");
386        assert!(!early.close(key(1)).expect("early part"), "work pending");
387        assert!(late.finish(work).expect("finish"), "ready once");
388        assert!(completion.is_idle().expect("idle"));
389    }
390
391    /// Each slice closes its own part and refuses work after it; another
392    /// slice's part stays open, and a whole group is ready on its one close.
393    #[test]
394    fn each_slice_closes_its_own_part() {
395        let completion = split(&[(1, &[0, 1, 2])]);
396        let slices: Vec<_> = (0..3).map(|index| completion.slice(index)).collect();
397        assert!(slices[0].close(key(0)).expect("whole group"), "ready now");
398        assert!(!slices[0].close(key(1)).expect("first part"));
399        assert!(slices[0].begin(key(1)).is_err(), "its part is closed");
400        assert!(slices[0].close(key(1)).is_err(), "closed twice");
401        let work = slices[2].begin(key(1)).expect("another part's work");
402        assert!(!slices[2].close(key(1)).expect("third part"));
403        assert!(!completion.is_idle().expect("not idle"), "a part to come");
404        assert!(
405            !slices[1].close(key(1)).expect("second part"),
406            "work pending"
407        );
408        assert!(slices[2].finish(work).expect("finish"));
409        assert!(completion.is_idle().expect("idle"));
410    }

412    /// Slices on threads: whatever the interleaving, each divided group is
413    /// ready exactly once.
414    #[test]
415    fn concurrent_slices_make_each_group_ready_once() {
416        let groups = 64;
417        let completion = split(
418            &(0..groups)
419                .map(|bin| (bin, &[0, 1, 2][..]))
420                .collect::<Vec<_>>(),
421        );
422        let ready = std::sync::atomic::AtomicUsize::new(0);
423        std::thread::scope(|scope| {
424            for index in 0..3 {
425                let slice = completion.slice(index);
426                let ready = &ready;
427                scope.spawn(move || {
428                    for bin in 0..groups {
429                        let work = slice.begin(key(bin)).expect("begin");
430                        let mut done = slice.close(key(bin)).expect("close");
431                        done |= slice.finish(work).expect("finish");
432                        if done {
433                            ready.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
434                        }
435                    }
436                });
437            }
438        });
439        assert_eq!(ready.into_inner(), groups as usize);
440        assert!(completion.is_idle().expect("idle"));
441    }

443    /// A slice whose traversal fails with a part open, or whose view is
444    /// dropped, abandons the group: the other slices and the split report
445    /// it rather than wait. A slice that completed its parts ends cleanly.
446    #[test]
447    fn a_slice_that_ends_with_a_part_open_abandons_its_group() {
448        for dropped in [false, true] {
449            let completion = split(&[(1, &[0, 1]), (2, &[1, 2])]);
450            let (first, second, third) = (
451                completion.slice(0),
452                completion.slice(1),
453                completion.slice(2),
454            );
455            assert!(!first.close(key(1)).expect("first part"));
456            first.end().expect("its parts are complete");
457            if dropped {
458                drop(second);
459            } else {
460                assert!(
461                    matches!(
462                        second.end(),
463                        Err(ObservationError::WorkAbandoned { group }) if group == key(1)
464                    ),
465                    "its parts of 1 and 2 are open"
466                );
467            }
468            assert!(matches!(
469                completion.is_idle(),
470                Err(ObservationError::WorkAbandoned { .. })
471            ));
472            assert!(third.close(key(2)).is_err(), "reported to the others");
473        }
474    }
