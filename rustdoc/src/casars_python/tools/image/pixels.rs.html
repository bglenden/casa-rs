16pub(super) enum Pixels {
17    Real(ArrayD<f64>),
18    Complex(ArrayD<Complex64>),
19}
20
21impl Pixels {
22    pub(super) fn capture(
23        py: Python<'_>,
24        value: Option<&Bound<'_, PyAny>>,
25        creating: bool,
26    ) -> PyResult<Self> {
27        let format_error = if creating {
28            "pixels is not understood, try using an array"
29        } else {
30            "Unsupported type for pixels parameter. It must be either a vector of doubles or ints"
31        };
32        if let Some(value) = value
33            && let Ok(array) = value.cast::<PyUntypedArray>()
34        {
35            let dtype = array.dtype();
36            let pixels = match (dtype.kind(), dtype.itemsize()) {
37                (b'i' | b'u', _) | (b'f', 4 | 8) => Self::Real(snapshot_array(py, array)?),
38                (b'c', 8 | 16) => Self::Complex(snapshot_array(py, array)?),
39                _ => return Err(raised(format_error)),
40            };
41            return pixels.check_shape("The pixels array cannot be empty");
42        }
43        let variant = value.map(|v| v.extract::<Variant>()).transpose()?;
44        Self::from_variant(variant, format_error)
45    }
46
47    fn from_variant(value: Option<Variant>, format_error: &str) -> PyResult<Self> {
48        let pixels = match value.unwrap_or(Variant::Empty) {
49            Variant::Numbers(values) => Self::Real(ndarray::Array1::from_vec(values).into_dyn()),
50            Variant::Complexes(values) => {
51                Self::Complex(ndarray::Array1::from_vec(values).into_dyn())
52            }
53            Variant::Array { shape, values } => {
54                Self::Real(ArrayD::from_shape_vec(IxDyn(&shape).f(), values).map_err(raised)?)
55            }
56            Variant::ComplexArray { shape, values } => {
57                Self::Complex(ArrayD::from_shape_vec(IxDyn(&shape).f(), values).map_err(raised)?)
58            }
59            Variant::Empty => return Err(raised("Exception: The pixels array cannot be empty")),
60            _ => return Err(raised(format_error)),
61        };
62        pixels.check_shape("The pixels array cannot be empty")
63    }
64
65    fn check_shape(self, empty: &str) -> PyResult<Self> {
66        if self.shape().is_empty() || self.shape().contains(&0) {
67            return Err(raised(format!("Exception: {empty}")));
68        }
69        Ok(self)
70    }
71
72    pub(super) fn shape(&self) -> &[usize] {
73        match self {
74            Self::Real(data) => data.shape(),
75            Self::Complex(data) => data.shape(),
76        }
77    }
78
79    pub(super) fn is_complex(&self) -> bool {
80        matches!(self, Self::Complex(_))
81    }
82
83    pub(super) fn f32(&self) -> ArrayD<f32> {
84        match self {
85            Self::Real(data) => data.mapv(|v| v as f32),
86            Self::Complex(_) => unreachable!(),
87        }
88    }
89
90    pub(super) fn f64(&self) -> ArrayD<f64> {
91        match self {
92            Self::Real(data) => data.clone(),
93            Self::Complex(_) => unreachable!(),
94        }
95    }
96
97    pub(super) fn c32(&self) -> ArrayD<Complex32> {
98        match self {
99            Self::Real(data) => data.mapv(|v| Complex32::new(v as f32, 0.0)),
100            Self::Complex(data) => data.mapv(|v| Complex32::new(v.re as f32, v.im as f32)),
101        }
102    }
103
104    pub(super) fn c64(&self) -> ArrayD<Complex64> {
105        match self {
106            Self::Real(data) => data.mapv(|v| Complex64::new(v, 0.0)),
107            Self::Complex(data) => data.clone(),
108        }
109    }
110
111    pub(super) fn history(&self) -> String {
112        if self.shape().iter().product::<usize>() > 100 {
113            return "\"[...]\"".into();
114        }
115        let values: Vec<String> = match self {
116            Self::Real(data) => data
117                .view()
118                .reversed_axes()
119                .iter()
120                .map(|v| format!("{v:.6}"))
121                .collect(),
122            Self::Complex(data) => data
123                .view()
124                .reversed_axes()
125                .iter()
126                .map(|v| format!("({:.6},{:.6})", v.re, v.im))
127                .collect(),
128        };
129        format!("[{}]", values.join(","))
130    }
131}

133// NumPy copies into a private contiguous buffer. No caller can mutate that
134// buffer while Rust reads it with the interpreter detached.
135fn snapshot_array<T: Element + Copy + Send + Sync>(
136    py: Python<'_>,
137    source: &Bound<'_, PyUntypedArray>,
138) -> PyResult<ArrayD<T>> {
139    let shape = source.shape().to_vec();
140    let snapshot = PyArrayDyn::<T>::zeros(py, IxDyn(&shape), true);
141    // SAFETY: both pointers name live NumPy arrays with identical shapes and
142    // numeric dtypes. The destination is freshly allocated and cannot alias
143    // either the input or an object another Python thread can access.
144    let status = unsafe {
145        PY_ARRAY_API.PyArray_CopyInto(py, snapshot.as_array_ptr(), source.as_array_ptr())
146    };
147    if status != 0 {
148        return Err(PyErr::fetch(py));
149    }
150    let snapshot = snapshot.readonly();
151    let values = snapshot.as_slice().map_err(raised)?;
152    py.detach(|| ArrayD::from_shape_vec(IxDyn(&shape).f(), values.to_vec()).map_err(raised))
153}

155struct Chunk {
156    start: Vec<usize>,
157    size: Vec<usize>,
158    stride: Vec<usize>,
159}
160
161impl Chunk {
162    fn read(shape: &[usize], blc: &[i64], trc: &[i64], inc: &[i64]) -> PyResult<Self> {
163        let mut chunk = Self {
164            start: Vec::new(),
165            size: Vec::new(),
166            stride: Vec::new(),
167        };
168        for (axis, &length) in shape.iter().enumerate() {
169            let valid = |v| usize::try_from(v).ok().filter(|&i| i < length);
170            let first = blc.get(axis).and_then(|&v| valid(v)).unwrap_or(0);
171            let last = trc.get(axis).and_then(|&v| valid(v)).unwrap_or(length - 1);
172            let (first, last) = if first <= last {
173                (first, last)
174            } else {
175                (0, length - 1)
176            };
177            let step = inc.get(axis).copied().unwrap_or(1);
178            if step < 1 {
179                return Err(raised("Exception: stride must be positive"));
180            }
181            let step = if step as usize > length {
182                1
183            } else {
184                step as usize
185            };
186            chunk.start.push(first);
187            chunk.size.push((last - first) / step + 1);
188            chunk.stride.push(step);
189        }
190        Ok(chunk)
191    }
192
193    fn write(
194        shape: &[usize],
195        data: &[usize],
196        blc: &[i64],
197        inc: &[i64],
198        replicate: bool,
199    ) -> PyResult<Self> {
200        if data.len() > shape.len() {
201            return Err(raised(
202                "Exception: Pixel array cannot have more dimensions than the image!",
203            ));
204        }
205        let mut chunk = Self {
206            start: Vec::new(),
207            size: Vec::new(),
208            stride: Vec::new(),
209        };
210        for (axis, &length) in shape.iter().enumerate() {
211            let start = blc
212                .get(axis)
213                .and_then(|&v| usize::try_from(v).ok())
214                .filter(|&v| v < length)
215                .unwrap_or(0);
216            let stride = inc.get(axis).copied().unwrap_or(1).max(1) as usize;
217            let count = data.get(axis).copied().unwrap_or(1);
218            if start >= length || (length - 1 - start) / stride < count - 1 {
219                return Err(raised(
220                    "Exception: Pixels array, including inc, extends beyond edge of image.",
221                ));
222            }
223            let extent = if replicate {
224                (length - 1 - start) / stride + 1
225            } else {
226                count
227            };
228            if replicate && extent % count != 0 {
229                return Err(raised(
230                    "Exception: region shape is not an integer multiple of the input array shape",
231                ));
232            }
233            chunk.start.push(start);
234            chunk.size.push(extent);
235            chunk.stride.push(stride);
236        }
237        Ok(chunk)
238    }

240    fn mask<T: ImagePixel>(
241        &self,
242        image: &impl ImageInterface<T>,
243    ) -> PyResult<Option<ArrayD<bool>>> {
244        image
245            .get_mask_slice(&self.start, &self.size, &self.stride)
246            .map_err(raised)
247    }
248
249    fn put<T: ImagePixel>(&self, storage: &mut Storage<T>, mut data: ArrayD<T>) -> PyResult<()> {
250        while data.ndim() < self.start.len() {
251            let axis = ndarray::Axis(data.ndim());
252            data = data.insert_axis(axis);
253        }
254        if data.shape() != self.size {
255            data = ArrayD::from_shape_fn(IxDyn(&self.size).f(), |index| {
256                let source: Vec<_> = index
257                    .slice()
258                    .iter()
259                    .zip(data.shape())
260                    .map(|(&i, &n)| i % n)
261                    .collect();
262                data[IxDyn(&source)]
263            });
264        }
265        if self.stride.iter().all(|&step| step == 1) {
266            with_storage!(storage, image => image.put_slice(&data, &self.start)).map_err(raised)?;
267        } else {
268            let mut pos = self.start.clone();
269            for (index, &value) in data.indexed_iter() {
270                for (axis, &i) in index.slice().iter().enumerate() {
271                    pos[axis] = self.start[axis] + i * self.stride[axis];
272                }
273                with_storage!(storage, image => image.put_at(value, &pos)).map_err(raised)?;
274            }
275        }
276        Ok(())
277    }

279    fn get<T: ImagePixel, U: Average>(
280        &self,
281        image: &impl ImageInterface<T>,
282        axes: &[usize],
283        dropdeg: bool,
284        getmask: bool,
285        wide: (impl Fn(T) -> U, impl Fn(U) -> U),
286    ) -> PyResult<CasaValue> {
287        let mask = if getmask || !axes.is_empty() {
288            self.mask(image)?
289        } else {
290            None
291        };
292        self.format(axes, dropdeg, getmask, wide, mask, || {
293            image
294                .get_slice(&self.start, &self.size, &self.stride)
295                .map_err(raised)
296        })
297    }
298
299    fn format<T: ImagePixel, U: Average>(
300        &self,
301        axes: &[usize],
302        dropdeg: bool,
303        getmask: bool,
304        wide: (impl Fn(T) -> U, impl Fn(U) -> U),
305        mask: Option<ArrayD<bool>>,
306        read: impl FnOnce() -> PyResult<ArrayD<T>>,
307    ) -> PyResult<CasaValue> {
308        let (widen, round) = wide;
309        let mut shape = self.size.clone();
310        for &axis in axes {
311            shape[axis] = 1;
312        }
313        let output_shape = if dropdeg {
314            let axes: Vec<_> = shape.iter().copied().filter(|&n| n != 1).collect();
315            if axes.is_empty() { vec![1] } else { axes }
316        } else {
317            shape.clone()
318        };
319        if axes.is_empty() && getmask {
320            let values = mask.map_or_else(|| vec![true; shape.iter().product()], fortran_values);
321            return Ok(CasaValue::Bools {
322                shape: output_shape,
323                values,
324            });
325        }
326        let pixels = read()?;
327        if axes.is_empty() {
328            return Ok(U::output(
329                output_shape,
330                pixels.reversed_axes().iter().copied().map(&widen).collect(),
331            ));
332        }
333
334        let mut sums = ArrayD::from_elem(IxDyn(&shape).f(), U::default());
335        let mut counts = ArrayD::from_elem(IxDyn(&shape).f(), 0_usize);
336        let mut out_index = vec![0; shape.len()];
337        for (index, &pixel) in pixels.indexed_iter() {
338            if mask.as_ref().is_some_and(|mask| !mask[&index]) {
339                continue;
340            }
341            out_index.copy_from_slice(index.slice());
342            for &axis in axes {
343                out_index[axis] = 0;
344            }
345            counts[IxDyn(&out_index)] += 1;
346            if !getmask {
347                sums[IxDyn(&out_index)] += widen(pixel);
348            }
349        }
350        let counts = fortran_values(counts);
351        if getmask {
352            let values = counts.into_iter().map(|n| n > 0).collect();
353            Ok(CasaValue::Bools {
354                shape: output_shape,
355                values,
356            })
357        } else {
358            let values = fortran_values(sums)
359                .into_iter()
360                .zip(counts)
361                .map(|(sum, n)| {
362                    if n == 0 {
363                        U::default()
364                    } else {
365                        round(sum / (n as f64))
366                    }
367                })
368                .collect();
369            Ok(U::output(output_shape, values))
370        }
371    }
372}

374fn widen_complex(value: Complex32) -> Complex64 {
375    Complex64::new(f64::from(value.re), f64::from(value.im))
376}
377
378fn fortran_values<T: Copy>(data: ArrayD<T>) -> Vec<T> {
379    data.reversed_axes().iter().copied().collect()
380}
381
382trait Average: Copy + Default + std::ops::AddAssign + std::ops::Div<f64, Output = Self> {
383    fn output(shape: Vec<usize>, values: Vec<Self>) -> CasaValue;
384}
385impl Average for f64 {
386    fn output(shape: Vec<usize>, values: Vec<Self>) -> CasaValue {
387        CasaValue::Floats { shape, values }
388    }
389}
390impl Average for Complex64 {
391    fn output(shape: Vec<usize>, values: Vec<Self>) -> CasaValue {
392        CasaValue::Complexes { shape, values }
393    }

396#[pymethods]
397#[allow(clippy::too_many_arguments)]
398impl PyImageTool {
399    #[pyo3(signature = (blc=None, trc=None, inc=None, axes=None, list=false, dropdeg=false, getmask=false), text_signature = "($self, blc=[-1], trc=[-1], inc=[1], axes=[-1], list=False, dropdeg=False, getmask=False)")]
400    fn getchunk(
401        &self,
402        py: Python<'_>,
403        blc: Option<IntList>,
404        trc: Option<IntList>,
405        inc: Option<IntList>,
406        axes: Option<IntList>,
407        list: bool,
408        dropdeg: bool,
409        getmask: bool,
410    ) -> PyResult<Py<PyAny>> {
411        let _ = list;
412        let value = py.detach(|| {
413            let state = self.state();
414            let Some(image) = state.image.as_ref() else {
415                return Ok(CasaValue::Bools { shape: vec![0], values: Vec::new() });
416            };
417            let image = image.lock();
418            let shape = image.shape();
419            let chunk = Chunk::read(shape, &blc.map_or_else(Vec::new, |arg| arg.0), &trc.map_or_else(Vec::new, |arg| arg.0), &inc.map_or_else(Vec::new, |arg| arg.0))?;
420            let axes = axes.map_or_else(Vec::new, |arg| arg.0);
421            let axes = if matches!(axes.as_slice(), [only] if *only < 0) { &[][..] } else { axes.as_slice() };
422            let mut collapse = Vec::with_capacity(axes.len());
423            for &axis in axes {
424                if axis < 0 || axis as usize >= shape.len() || collapse.contains(&(axis as usize)) {
425                    return Err(raised("IPosition::makeAxisPath: invalid defined axes"));
426                }
427                collapse.push(axis as usize);
428            }
429            if let Some(fits) = image.fits() {
430                let pixels = fits.read(&chunk.start, &chunk.size, &chunk.stride)?;
431                let mask = (getmask || !collapse.is_empty()).then(|| pixels.mapv(|value| !value.is_nan()));
432                return chunk.format(&collapse, dropdeg, getmask, (f64::from, |v| f64::from(v as f32)), mask, || Ok(pixels));
433            }
434            match &*image {
435                ImageHandle::Float(storage) => with_storage!(storage, image => chunk.get(image.as_ref(), &collapse, dropdeg, getmask, (f64::from, |v| f64::from(v as f32)))),
436                ImageHandle::Double(storage) => with_storage!(storage, image => chunk.get(image.as_ref(), &collapse, dropdeg, getmask, (f64::from, std::convert::identity))),
437                ImageHandle::Complex(storage) => with_storage!(storage, image => chunk.get(image.as_ref(), &collapse, dropdeg, getmask, (widen_complex, |v: Complex64| widen_complex(Complex32::new(v.re as f32, v.im as f32))))),
438                ImageHandle::DComplex(storage) => with_storage!(storage, image => chunk.get(image.as_ref(), &collapse, dropdeg, getmask, (std::convert::identity, std::convert::identity))),
439            }
440        })?;
441        value.into_python(py)
442    }
443
444    #[pyo3(signature = (pixels=None, blc=None, inc=None, list=false, locking=true, replicate=false), text_signature = "($self, pixels=[], blc=[-1], inc=[1], list=False, locking=True, replicate=False)")]
445    fn putchunk(
446        &self,
447        py: Python<'_>,
448        pixels: Option<&Bound<'_, PyAny>>,
449        blc: Option<IntList>,
450        inc: Option<IntList>,
451        list: bool,
452        locking: bool,
453        replicate: bool,
454    ) -> PyResult<bool> {
455        let pixels = Pixels::capture(py, pixels, false);
456        py.detach(|| {
457            let state = self.state();
458            let history = state.history;
459            let Some(image) = state.image.as_ref() else { return Ok(false) };
460            let mut image = image.lock();
461            let pixels = pixels?;
462            let blc = blc.unwrap_or_else(|| IntList::of(&[-1])).0;
463            let inc = inc.unwrap_or_else(|| IntList::of(&[1])).0;
464            let chunk = Chunk::write(image.shape(), pixels.shape(), &blc, &inc, replicate)?;
465            if pixels.is_complex() && matches!(&*image, ImageHandle::Float(_) | ImageHandle::Double(_)) {
466                return Err(raised("Exception: Unsupported type for pixels parameter. It must be either a vector of doubles or ints"));
467            }
468            if image.fits().is_some() {
469                return Err(raised("FITSImage::putSlice - is not possible as FITSImage is not writable"));
470            }
471            match &mut *image {
472                ImageHandle::Float(storage) => chunk.put(storage, pixels.f32())?,
473                ImageHandle::Double(storage) => chunk.put(storage, pixels.f64())?,
474                ImageHandle::Complex(storage) => chunk.put(storage, pixels.c32())?,
475                ImageHandle::DComplex(storage) => chunk.put(storage, pixels.c64())?,
476            }
477            if history {
478                let command = format!(
479                    "ia.putchunk(pixels={}, blc={}, inc={}, list={list}, locking={locking}, replicate={replicate})",
480                    pixels.history(), format!("{blc:?}").replace(' ', ""), format!("{inc:?}").replace(' ', "")
481                );
482                image.record_history("putchunk", command)?;
483            }
484            image.save().map_err(raised)?;
485            Ok(true)
486        })
487    }
