10pub(super) fn check(p: &Parameters<'_>, plan: &Plan) -> Result<(), SimobserveError> {
11    let mut outputs = Vec::new();
12    let synthesis = if plan.inputs.synthesis.is_empty() {
13        Vec::new()
14    } else {
15        [
16            ".image",
17            ".model",
18            ".psf",
19            ".residual",
20            ".pb",
21            ".image.pbcor",
22            ".sumwt",
23            ".weight",
24            ".mask",
25            ".image.flat",
26            ".residual.flat",
27            ".diff",
28            ".fidelity",
29        ]
30        .map(|suffix| plan.file(suffix))
31        .to_vec()
32    };
33    let mut synthesis = synthesis;
34    if !plan.inputs.synthesis.is_empty()
35        && (!p.text("featherimage").is_empty()
36            || (plan.inputs.total_power.is_some() && p.text("modelimage").is_empty()))
37    {
38        synthesis.extend(
39            [
40                ".feather.image",
41                ".feather.residual.flat",
42                ".feather.image.flat",
43                ".feather.diff",
44                ".feather.fidelity",
45            ]
46            .map(|suffix| plan.file(suffix)),
47        );
48    }
49    outputs.extend(synthesis.iter().cloned());
50    if plan.inputs.synthesis.len() > 1 {
51        outputs.push(plan.ms.clone());
52    }
53    let mut tp = Vec::new();
54    if plan.inputs.total_power.is_some() {
55        let project = plan.directory.file_name().unwrap().to_string_lossy();
56        let image = plan.directory.join(format!("{project}.sd.image"));
57        tp = ["", "0", "0.scaled", "0.image", "0.weight"]
58            .map(|suffix| suffixed(&image, suffix))
59            .to_vec();
60        outputs.extend(tp.iter().cloned());
61    }
62    let mut inputs = plan.inputs.synthesis.clone();
63    inputs.extend(plan.inputs.total_power.iter().cloned());
64    inputs.push(plan.model.clone());
65    refuse_collisions(&inputs, &outputs)?;
66    for name in ["modelimage", "featherimage"] {
67        let supplied = p.text(name);
68        if supplied.is_empty() {
69            continue;
70        }
71        let input = p.locate(&plan.directory, supplied);
72        refuse_collisions(std::slice::from_ref(&input), &synthesis)?;
73        // The final TP image is deliberately available as this invocation's
74        // seed or feather input. Its intermediates are never caller inputs.
75        refuse_collisions(
76            std::slice::from_ref(&input),
77            tp.get(1..).unwrap_or_default(),
78        )?;
79    }
80    if !p.flag("overwrite") {
81        for output in outputs {
82            if std::fs::symlink_metadata(&output).is_ok() {
83                return Err(failure(format!("{} already exists", output.display())));
84            }
85        }
86    }
87    Ok(())
88}

90fn existing(path: &Path) -> Result<Option<PathBuf>, SimobserveError> {
91    match std::fs::canonicalize(path) {
92        Ok(path) => Ok(Some(path)),
93        Err(error) if error.kind() == std::io::ErrorKind::NotFound => Ok(None),
94        Err(error) => Err(failure(error)),
95    }
96}
97
98fn refuse_collisions(inputs: &[PathBuf], outputs: &[PathBuf]) -> Result<(), SimobserveError> {
99    for input in inputs {
100        let Some(input) = existing(input)? else {
101            continue;
102        };
103        for output in outputs {
104            if existing(output)?.is_some_and(|output| output == input) {
105                return Err(failure(format!(
106                    "input {} would be overwritten by simulation imaging",
107                    input.display()
108                )));
109            }
110        }
111    }
112    Ok(())
113}

116mod tests {

119    #[test]
120    fn collisions_include_canonical_aliases() {
121        let scratch = tempfile::tempdir().unwrap();
122        let input = scratch.path().join("input");
123        std::fs::create_dir(&input).unwrap();
124        let output = scratch.path().join("./input");
