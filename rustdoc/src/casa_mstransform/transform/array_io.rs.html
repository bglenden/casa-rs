13pub(super) fn transform_array_dimensions(
14    input: &MeasurementSet,
15    selected_ddids: &[i32],
16    ddid_to_spw: &BTreeMap<i32, i32>,
17    channel_bins: &BTreeMap<i32, Vec<Vec<usize>>>,
18    correlations: Option<&CorrelationSelection>,
19    polarization_projection: Option<&PolarizationProjection>,
20) -> Result<BTreeMap<i32, TransformArrayDimensions>, MsTransformError> {
21    let input_path = input.path().unwrap_or_else(|| Path::new("<memory>"));
22    let invalid = |reason| MsTransformError::SpectralMetadata {
23        path: input_path.display().to_string(),
24        reason,
25    };
26    let data_description = input
27        .data_description()
28        .map_err(|error| invalid(error.to_string()))?;
29    let polarization = input
30        .polarization()
31        .map_err(|error| invalid(error.to_string()))?;
32    let spectral_window = input
33        .spectral_window()
34        .map_err(|error| invalid(error.to_string()))?;
35    let mut dimensions = BTreeMap::new();
36    for &ddid in selected_ddids {
37        let std::collections::btree_map::Entry::Vacant(entry) = dimensions.entry(ddid) else {
38            continue;
39        };
40        let row =
41            usize::try_from(ddid).map_err(|_| invalid(format!("negative DATA_DESC_ID {ddid}")))?;
42        let pol_id = data_description
43            .polarization_id(row)
44            .map_err(|error| invalid(format!("read DATA_DESCRIPTION row {ddid}: {error}")))?;
45        let pol_row = usize::try_from(pol_id).map_err(|_| {
46            invalid(format!(
47                "DATA_DESCRIPTION row {ddid} has negative POLARIZATION_ID {pol_id}"
48            ))
49        })?;
50        let input_correlations = usize::try_from(
51            polarization
52                .num_corr(pol_row)
53                .map_err(|error| invalid(format!("read POLARIZATION row {pol_id}: {error}")))?,
54        )
55        .map_err(|_| invalid(format!("POLARIZATION row {pol_id} has negative NUM_CORR")))?;
56        let spw = ddid_to_spw.get(&ddid).copied().ok_or_else(|| {
57            invalid(format!(
58                "DATA_DESC_ID {ddid} has no spectral-window mapping"
59            ))
60        })?;
61        let output_channels = channel_bins
62            .get(&spw)
63            .map(Vec::len)
64            .ok_or_else(|| invalid(format!("spectral window {spw} has no output channel bins")))?;
65        let input_channels = usize::try_from(
66            spectral_window
67                .num_chan(spw as usize)
68                .map_err(|error| invalid(format!("read SPECTRAL_WINDOW row {spw}: {error}")))?,
69        )
70        .map_err(|_| invalid(format!("SPECTRAL_WINDOW row {spw} has negative NUM_CHAN")))?;
71        entry.insert(TransformArrayDimensions {
72            input_correlations,
73            input_channels,
74            output_correlations: if polarization_projection.is_some_and(|plan| plan.contains(ddid))
75            {
76                1
77            } else {
78                correlations
79                    .and_then(|plan| plan.axes(ddid))
80                    .map_or(input_correlations, <[usize]>::len)
81            },
82            output_channels,
83            output_parts: vec![output_channels],
84        });
85    }
86    Ok(dimensions)
87}

109pub(super) fn transform_array_write_plan(
110    input: &MeasurementSet,
111    TransformArrayRows {
112        selected_rows,
113        selected_ddids,
114        groups,
115        output_rows,
116        output_schema,
117        combined,
118        regrid,
119        tile,
120    }: TransformArrayRows<'_>,
121    ddid_to_spw: &BTreeMap<i32, i32>,
122    channel_bins: &BTreeMap<i32, Vec<Vec<usize>>>,
123    (correlations, polarization): (
124        Option<&CorrelationSelection>,
125        Option<&PolarizationProjection>,
126    ),
127    (visibility, spectra, view_bytes): (&VisibilityColumns, &SpectrumProjection, usize),
128    resources: MeasurementSetWriteResources,
129) -> Result<(MeasurementSetWritePlan, usize), MsTransformError> {
130    let input_path = input.path().unwrap_or_else(|| Path::new("<memory>"));
131    let mut dimensions = transform_array_dimensions(
132        input,
133        combined.map_or(selected_ddids, |rows| rows.source_ddids),
134        ddid_to_spw,
135        channel_bins,
136        correlations,
137        polarization,
138    )?;
139    if let Some(CombinedArrayRows { grid, .. }) = combined {
140        for dimension in dimensions.values_mut() {
141            dimension.output_channels = grid.frequencies.len();
142            dimension.output_parts = vec![dimension.output_channels];
143        }
144    }
145    if let Some(regrid) = regrid {
146        for (&ddid, dimension) in &mut dimensions {
147            let spw = ddid_to_spw[&ddid];
148            dimension.output_channels = regrid.channel_count(spw);
149            dimension.output_parts = regrid
150                .parts(spw)
151                .ranges()
152                .iter()
153                .map(|part| part.len())
154                .collect();
155        }
156    }
157    let mut shape_counts = BTreeMap::<Vec<usize>, usize>::new();
158    let mut weight_shape_counts = BTreeMap::<Vec<usize>, usize>::new();
159    for (&row, &ddid) in selected_rows.iter().zip(selected_ddids) {
160        let shape = &dimensions[&ddid];
161        let field_channels = regrid.and_then(|plan| plan.field_channel_count(row));
162        let parts = field_channels
163            .as_ref()
164            .map_or(&shape.output_parts[..], std::slice::from_ref);
165        for &channels in parts {
166            *shape_counts
167                .entry(vec![shape.output_correlations, channels])
168                .or_default() += 1;
169            *weight_shape_counts
170                .entry(vec![shape.output_correlations])
171                .or_default() += 1;
172        }
173    }
174    let shapes = shape_counts
175        .into_iter()
176        .map(|(cell_shape, row_count)| MeasurementSetArrayShapePlan {
177            tile_shape: casa_ms::casa_visibility_tile_shape(cell_shape[0], cell_shape[1], ""),
178            cell_shape,
179            row_count,
180        })
181        .collect::<Vec<_>>();
182    let weight_shapes = weight_shape_counts
183        .into_iter()
184        .map(|(cell_shape, row_count)| {
185            let visibility_tile = casa_ms::casa_visibility_tile_shape(cell_shape[0], 1, "");
186            MeasurementSetArrayShapePlan {
187                tile_shape: casa_ms::casa_weight_tile_shape(&visibility_tile),
188                cell_shape,
189                row_count,
190            }
191        })
192        .collect::<Vec<_>>();
193    let metadata_budget = MsSelectionIoBudget {
194        available_bytes: resources.available_bytes,
195        maximum_live_blocks: resources.maximum_live_batches,
196        requested_bytes_per_row: std::mem::size_of::<Option<Vec<usize>>>()
197            + 5 * std::mem::size_of::<usize>(),
198        storage_alignment_rows: None,
199    };
200    let (flag_category_shapes, largest_input_category) = plan_flag_category_shapes(
201        input,
202        selected_rows,
203        selected_ddids,
204        &dimensions,
205        correlations.is_some() || polarization.is_some(),
206        (groups, metadata_budget),
207        (combined, regrid),
208    )?;
209    let io_error = |error: MsSelectionError| MsTransformError::WriteMeasurementSet {
210        path: input_path.display().to_string(),
211        reason: error.to_string(),
212    };
213    let (lag_shapes, maximum_lag_bytes) = if visibility.contains_output("LAG_DATA") {
214        plan_lag_shapes(
215            input,
216            selected_rows,
217            selected_ddids,
218            &dimensions,
219            (groups, metadata_budget),
220        )?
221    } else {
222        (Vec::new(), 0)
223    };
224    let (source_count, source_time_groups) = combined
225        .map_or((selected_rows.len(), groups), |rows| {
226            (rows.source_ddids.len(), rows.source_time_groups)
227        });
228    let raw_row_count = source_time_groups.map_or(source_count, |groups| {
229        groups.iter().map(|group| group.rows.len()).sum()
230    });
231    let input_batch_rows = transform_input_batch_rows(
232        raw_row_count,
233        resources,
234        dimensions.values(),
235        spectra.source.len(),
236        visibility
237            .regular_bytes_per_sample()
238            .saturating_add(view_bytes),
239        largest_input_category.saturating_add(maximum_lag_bytes),
240    )
241    .map_err(io_error)?;
242    let visibility_tile = shapes
243        .first()
244        .map(|shape| shape.tile_shape.clone())
245        .unwrap_or_else(|| casa_ms::casa_visibility_tile_shape(1, 1, ""));
246    let mut columns = vec![
247        MeasurementSetArrayColumnPlan {
248            name: "FLAG".to_string(),
249            value_type: StreamedTiledShapeValueType::Bool,
250            shapes: shapes.clone(),
251            storage_manager: MeasurementSetColumnStorage::TiledShape,
252        },
253        MeasurementSetArrayColumnPlan {
254            name: "FLAG_CATEGORY".to_string(),
255            value_type: StreamedTiledShapeValueType::Bool,
256            shapes: flag_category_shapes,
257            storage_manager: MeasurementSetColumnStorage::TiledShape,
258        },
259        MeasurementSetArrayColumnPlan {
260            name: "UVW".to_string(),
261            value_type: StreamedTiledShapeValueType::Float64,
262            shapes: vec![MeasurementSetArrayShapePlan {
263                cell_shape: vec![3],
264                row_count: shapes.iter().map(|shape| shape.row_count).sum(),
265                tile_shape: casa_ms::casa_uvw_tile_shape(&visibility_tile),
266            }],
267            storage_manager: MeasurementSetColumnStorage::TiledColumn,
268        },
269        MeasurementSetArrayColumnPlan {
270            name: "WEIGHT".to_string(),
271            value_type: StreamedTiledShapeValueType::Float32,
272            shapes: weight_shapes.clone(),
273            storage_manager: MeasurementSetColumnStorage::TiledShape,
274        },
275        MeasurementSetArrayColumnPlan {
276            name: "SIGMA".to_string(),
277            value_type: StreamedTiledShapeValueType::Float32,
278            shapes: weight_shapes,
279            storage_manager: MeasurementSetColumnStorage::TiledShape,
280        },
281    ];
282    for &name in &spectra.output {
283        columns.push(MeasurementSetArrayColumnPlan {
284            name: name.to_string(),
285            value_type: StreamedTiledShapeValueType::Float32,
286            shapes: shapes.clone(),
287            storage_manager: MeasurementSetColumnStorage::TiledShape,
288        });
289    }
290    let independent_lag = !regrid.is_some_and(regrid::Plan::is_legacy);
291    for column in visibility.copies() {
292        let shapes = if independent_lag && column.source == OptionalMainColumn::LagData {
293            lag_shapes.clone()
294        } else {
295            shapes.clone()
296        };
297        columns.push(MeasurementSetArrayColumnPlan {
298            name: column.output.name().to_string(),
299            value_type: column.value_type(),
300            shapes,
301            storage_manager: MeasurementSetColumnStorage::TiledShape,
302        });
303    }
304    if output_rows == 0 {
305        columns.clear();
306    }
307    if let Some(tile) = tile {
308        tile.apply(&mut columns);
309    }
310    let writer = MeasurementSetWritePlan::variable_array_creation(
311        output_rows,
312        columns,
313        transform_scalar_column_plans(output_schema),
314        resources,
315    )
316    .map_err(|error| MsTransformError::WriteMeasurementSet {
317        path: input_path.display().to_string(),
318        reason: error.to_string(),
319    })?;
320    let writer = if let Some(tile) = tile {
321        writer.with_array_default_tiles(|name| tile.default_for(name))
322    } else {
323        writer
324    };
325    Ok((writer, input_batch_rows))
326}

335pub(super) struct SpectrumProjection {
336    pub(super) source: Vec<&'static str>,
337    pub(super) output: Vec<&'static str>,
338    pub(super) propagate_weights: bool,
339    pub(super) fill_standard: bool,
340}
341
342impl SpectrumProjection {
343    pub(super) fn existing(table: &Table) -> Self {
344        let source = MAIN_SPECTRA
345            .into_iter()
346            .filter(|name| {
347                table
348                    .schema()
349                    .is_some_and(|schema| schema.contains_column(name))
350            })
351            .collect::<Vec<_>>();
352        Self {
353            output: source.clone(),
354            source,
355            propagate_weights: false,
356            fill_standard: false,
357        }
358    }
359
360    pub(super) fn partition(
361        table: &Table,
362        selected_row: usize,
363        averaging_time: bool,
364    ) -> Result<Self, TableError> {
365        let name = "WEIGHT_SPECTRUM";
366        let present = table
367            .schema()
368            .is_some_and(|schema| schema.contains_column(name));
369        let defined = present && table.array_shape_uncached(selected_row, name)?.is_some();
370        Ok(Self {
371            source: defined.then_some(name).into_iter().collect(),
372            output: vec![name],
373            propagate_weights: defined,
374            fill_standard: defined && averaging_time,
375        })
376    }
377
378    pub(super) fn resolve(
379        table: &Table,
380        selected_row: usize,
381        grammar: RowGrammar,
382        force_spectra: bool,
383        (channel_average, time_average, spectral_convolution, hanning): (bool, bool, bool, bool),
384        visibility: &VisibilityColumns,
385    ) -> Result<Self, TableError> {
386        let exists = |name| {
387            table
388                .schema()
389                .is_some_and(|schema| schema.contains_column(name))
390        };
391        if grammar != RowGrammar::Transform {
392            return Ok(Self::existing(table));
393        }
394        let defined = |row, name| -> Result<bool, TableError> {
395            Ok(exists(name) && table.array_shape_uncached(row, name)?.is_some())
396        };
397        let propagate_weights = defined(selected_row, "WEIGHT_SPECTRUM")?;
398        let fill_standard = force_spectra || propagate_weights;
399        // The selected MS can supply spectra even if row zero of the physical
400        // input has no spectrum. Allocate columns for the selected values.
401        let output = if fill_standard || defined(0, "WEIGHT_SPECTRUM")? {
402            STANDARD_SPECTRA.to_vec()
403        } else {
404            Vec::new()
405        };
406        let mut source = Vec::new();
407        let needs_weight = if time_average {
408            let copies = visibility.copies();
409            copies
410                .iter()
411                .any(|column| column.source == OptionalMainColumn::CorrectedData)
412                || (copies.len() == 1 && copies[0].source == OptionalMainColumn::LagData)
413        } else {
414            channel_average || fill_standard || spectral_convolution || hanning
415        };
416        if needs_weight && exists("WEIGHT_SPECTRUM") {
417            source.push("WEIGHT_SPECTRUM");
418        }
419        let needs_sigma = if channel_average || time_average || hanning {
420            visibility.copies().iter().any(|column| {
421                column.source == OptionalMainColumn::Data
422                    || (time_average && column.source == OptionalMainColumn::FloatData)
423            })
424        } else {
425            (fill_standard || spectral_convolution) && !visibility.corrected_becomes_data()
426        };
427        if needs_sigma && exists("SIGMA_SPECTRUM") {
428            source.push("SIGMA_SPECTRUM");
429        }
430        Ok(Self {
431            source,
432            output,
433            propagate_weights,
434            fill_standard,
435        })
436    }
437}

439pub(super) fn flag_filter_read_plan<'a>(
440    rows: usize,
441    mut budget: MsSelectionIoBudget,
442    dimensions: impl Iterator<Item = &'a TransformArrayDimensions>,
443) -> Result<usize, MsSelectionError> {
444    // FLAG is read at its original shape, even when only a few selected
445    // channels or correlations will appear in the product.
446    let flag_bytes = dimensions
447        .map(|shape| {
448            shape
449                .input_correlations
450                .saturating_mul(shape.input_channels)
451        })
452        .max()
453        .unwrap_or(0);
454    budget.requested_bytes_per_row = flag_bytes.saturating_add(
455        std::mem::size_of::<Option<ArrayValue>>() + std::mem::size_of::<Option<ScalarValue>>() + 32,
456    );
457    Ok(MsReadPlan::new(rows, budget)?.rows_per_block)
458}

460pub(super) struct SelectedFlagAxes<'a> {
461    pub(super) channels: &'a BTreeMap<i32, TransformChannelSelection>,
462    pub(super) ddid_to_spw: &'a BTreeMap<i32, i32>,
463    pub(super) correlations: Option<&'a CorrelationSelection>,
464}
465
466impl SelectedFlagAxes<'_> {
467    fn all_flagged(&self, flags: &ArrayValue, ddid: i32) -> Result<bool, TableError> {
468        let ArrayValue::Bool(flags) = flags else {
469            return Err(TableError::Schema("FLAG must hold Bool".into()));
470        };
471        let flags = flags
472            .view()
473            .into_dimensionality::<ndarray::Ix2>()
474            .map_err(|_| {
475                TableError::Schema("FLAG must have correlation and channel axes".into())
476            })?;
477        let channels = &self.channels[&self.ddid_to_spw[&ddid]].channels;
478        let axes = self.correlations.and_then(|selection| selection.axes(ddid));
479        if channels.iter().any(|&chan| chan >= flags.ncols())
480            || axes.is_some_and(|axes| axes.iter().any(|&corr| corr >= flags.nrows()))
481        {
482            return Err(TableError::Schema(
483                "selected axes exceed the FLAG shape".into(),
484            ));
485        }
486        Ok((0..flags.nrows())
487            .filter(|corr| axes.is_none_or(|axes| axes.contains(corr)))
488            .all(|corr| channels.iter().all(|&chan| flags[[corr, chan]])))
489    }
490}

492pub(super) fn keep_unflagged_rows(
493    input: &MeasurementSet,
494    rows: &[usize],
495    descriptions: &[i32],
496    rows_per_block: usize,
497    selection: Option<&SelectedFlagAxes<'_>>,
498) -> Result<(Vec<usize>, Vec<i32>), MsTransformError> {
499    let error = |source| MsTransformError::MutateMeasurementSet {
500        path: input
501            .path()
502            .unwrap_or_else(|| Path::new("<memory>"))
503            .display()
504            .to_string(),
505        source: Box::new(source),
506    };
507    let table = input.main_table();
508    let flag_row = table.column_accessor("FLAG_ROW").map_err(error)?;
509    let flag = table.column_accessor("FLAG").map_err(error)?;
510    let mut kept_rows = Vec::with_capacity(rows.len());
511    let mut kept_descriptions = Vec::with_capacity(rows.len());
512    for (rows, descriptions) in rows
513        .chunks(rows_per_block)
514        .zip(descriptions.chunks(rows_per_block))
515    {
516        let flag_rows = flag_row.scalar_cells_owned_for_rows(rows).map_err(error)?;
517        let flags = flag.array_cells_owned_uncached(rows).map_err(error)?;
518        for (((&row, &ddid), flag_row), flags) in
519            rows.iter().zip(descriptions).zip(flag_rows).zip(flags)
520        {
521            let flagged_row = scalar_bool(flag_row.as_ref(), "FLAG_ROW", row)?;
522            let flagged_samples = match (selection, flags.as_ref()) {
523                (Some(selection), Some(flags)) => {
524                    selection.all_flagged(flags, ddid).map_err(error)?
525                }
526                _ => bool_array_all_true(flags.as_ref(), "FLAG", row)?,
527            };
528            if !flagged_row && !flagged_samples {
529                kept_rows.push(row);
530                kept_descriptions.push(ddid);
531            }
532        }
533    }
534    Ok((kept_rows, kept_descriptions))
535}

537pub(super) fn transform_input_batch_rows<'a>(
538    rows: usize,
539    resources: MeasurementSetWriteResources,
540    dimensions: impl Iterator<Item = &'a TransformArrayDimensions>,
541    spectrum_columns: usize,
542    visibility_bytes_per_sample: usize,
543    extra_array_bytes: usize,
544) -> Result<usize, MsSelectionError> {
545    // The input arrays are loaded before slicing, so a small output cell
546    // does not imply a small source block. Allow space for array headers too.
547    let elements = dimensions
548        .map(|shape| {
549            shape.input_correlations.saturating_mul(
550                shape
551                    .input_channels
552                    .saturating_mul(1 + visibility_bytes_per_sample + 4 * spectrum_columns)
553                    .saturating_add(8),
554            )
555        })
556        .max()
557        .unwrap_or(0);
558    let overhead = (10 + spectrum_columns) * (std::mem::size_of::<Option<ArrayValue>>() + 32);
559    let bytes_per_row = elements
560        .saturating_add(24 + overhead)
561        .saturating_add(extra_array_bytes);
562    let plan = MsReadPlan::new(
563        rows,
564        MsSelectionIoBudget {
565            available_bytes: resources.available_bytes,
566            maximum_live_blocks: resources.maximum_live_batches,
567            requested_bytes_per_row: bytes_per_row,
568            storage_alignment_rows: None,
569        },
570    )?;
571    Ok(plan.rows_per_block)
572}

574pub(super) fn plan_flag_category_shapes(
575    input: &MeasurementSet,
576    selected_rows: &[usize],
577    selected_ddids: &[i32],
578    dimensions: &BTreeMap<i32, TransformArrayDimensions>,
579    correlation_subset: bool,
580    (groups, budget): (Option<&[time_average::TimeGroup]>, MsSelectionIoBudget),
581    (combined, regrid): (Option<CombinedArrayRows<'_>>, Option<&regrid::Plan>),
582) -> Result<(Vec<MeasurementSetArrayShapePlan>, usize), MsTransformError> {
583    let main = input.main_table();
584    let default_categories = match main
585        .column_keywords("FLAG_CATEGORY")
586        .and_then(|keywords| keywords.get("CATEGORY"))
587    {
588        Some(Value::Array(categories)) => categories.len().max(1),
589        _ => 1,
590    };
591    let mut counts = BTreeMap::new();
592    let mut largest_input = 0;
593    let shapes = if let Some(combined) = combined {
594        let source_shapes = optional_array_shapes(
595            input,
596            combined.source_rows,
597            combined.source_time_groups,
598            "FLAG_CATEGORY",
599            budget,
600        )?;
601        largest_input = source_shapes
602            .iter()
603            .flatten()
604            .map(|shape| shape.iter().copied().fold(1usize, usize::saturating_mul))
605            .max()
606            .unwrap_or(0);
607        combined
608            .groups
609            .iter()
610            .map(|group| {
611                group
612                    .members
613                    .iter()
614                    .find_map(|&(_, pos)| source_shapes[pos].clone())
615            })
616            .collect()
617    } else {
618        optional_array_shapes(input, selected_rows, groups, "FLAG_CATEGORY", budget)?
619    };
620    for ((original, ddid), &row) in shapes.into_iter().zip(selected_ddids).zip(selected_rows) {
621        let shape = &dimensions[ddid];
622        let cell = match original.as_deref() {
623            Some(&[corr, chan, categories]) if corr > 0 && chan > 0 && categories > 0 => {
624                largest_input =
625                    largest_input.max(corr.saturating_mul(chan).saturating_mul(categories));
626                [
627                    if correlation_subset {
628                        shape.output_correlations
629                    } else {
630                        corr
631                    },
632                    if chan == 1 && combined.is_none() {
633                        1
634                    } else {
635                        shape.output_channels
636                    },
637                    categories,
638                ]
639            }
640            _ => [shape.output_correlations, 1, default_categories],
641        };
642        let field_channels = regrid.and_then(|plan| plan.field_channel_count(row));
643        let parts = field_channels
644            .as_ref()
645            .map_or(&shape.output_parts[..], std::slice::from_ref);
646        for &channels in parts {
647            let cell = [cell[0], if cell[1] == 1 { 1 } else { channels }, cell[2]];
648            *counts.entry(cell).or_default() += 1;
649        }
650    }
651    let plans = counts
652        .into_iter()
653        .map(|([corr, chan, categories], row_count)| {
654            let visibility_tile = casa_ms::casa_visibility_tile_shape(corr, chan, "");
655            MeasurementSetArrayShapePlan {
656                cell_shape: vec![corr, chan, categories],
657                row_count,
658                tile_shape: vec![
659                    visibility_tile[0],
660                    visibility_tile[1],
661                    1,
662                    visibility_tile[2],
663                ],
664            }
665        })
666        .collect();
667    Ok((plans, largest_input))
668}

670pub(super) fn plan_lag_shapes(
671    input: &MeasurementSet,
672    rows: &[usize],
673    descriptions: &[i32],
674    dimensions: &BTreeMap<i32, TransformArrayDimensions>,
675    (groups, budget): (Option<&[time_average::TimeGroup]>, MsSelectionIoBudget),
676) -> Result<(Vec<MeasurementSetArrayShapePlan>, usize), MsTransformError> {
677    let path = input.path().unwrap_or_else(|| Path::new("<memory>"));
678    let mut counts = BTreeMap::new();
679    let mut max_bytes = 0;
680    let shapes = optional_array_shapes(input, rows, groups, "LAG_DATA", budget)?;
681    for ((&row, &description), shape) in rows.iter().zip(descriptions).zip(shapes) {
682        let output_correlations = dimensions[&description].output_correlations;
683        let lags = match shape.as_deref() {
684            Some(&[stored_correlations, lags]) if lags > 0 => {
685                max_bytes =
686                    max_bytes.max(stored_correlations.saturating_mul(lags).saturating_mul(8));
687                lags
688            }
689            None => 1, // An undefined cell needs no physical lag extent.
690            _ => {
691                return Err(MsTransformError::SpectralMetadata {
692                    path: path.display().to_string(),
693                    reason: format!("LAG_DATA row {row} has an invalid shape {shape:?}"),
694                });
695            }
696        };
697        *counts.entry([output_correlations, lags]).or_default() +=
698            dimensions[&description].output_parts.len();
699    }
700    let plans = counts
701        .into_iter()
702        .map(
703            |([correlations, lags], row_count)| MeasurementSetArrayShapePlan {
704                cell_shape: vec![correlations, lags],
705                row_count,
706                tile_shape: casa_ms::casa_visibility_tile_shape(correlations, lags, ""),
707            },
708        )
709        .collect();
710    Ok((plans, max_bytes))
711}

713pub(super) fn optional_array_shapes(
714    input: &MeasurementSet,
715    rows: &[usize],
716    groups: Option<&[time_average::TimeGroup]>,
717    column: &str,
718    budget: MsSelectionIoBudget,
719) -> Result<Vec<Option<Vec<usize>>>, MsTransformError> {
720    let path = input.path().unwrap_or_else(|| Path::new("<memory>"));
721    let table_error = |source| MsTransformError::MutateMeasurementSet {
722        path: path.display().to_string(),
723        source: Box::new(source),
724    };
725    let mut members: Box<dyn Iterator<Item = (usize, usize)>> = match groups {
726        Some(groups) => Box::new(groups.iter().enumerate().flat_map(|(index, group)| {
727            group
728                .rows
729                .iter()
730                .map(move |sample| (index, sample.physical))
731        })),
732        None => Box::new(rows.iter().copied().enumerate()),
733    };
734    let raw_rows = groups.map_or(rows.len(), |groups| {
735        groups.iter().map(|group| group.rows.len()).sum()
736    });
737    let plan = MsReadPlan::new(raw_rows, budget).map_err(|error| {
738        MsTransformError::WriteMeasurementSet {
739            path: path.display().to_string(),
740            reason: error.to_string(),
741        }
742    })?;
743    let accessor = input
744        .main_table()
745        .column_accessor(column)
746        .map_err(table_error)?;
747    let mut shapes: Vec<Option<Vec<usize>>> = vec![None; rows.len()];
748    loop {
749        let block = members
750            .by_ref()
751            .take(plan.rows_per_block)
752            .collect::<Vec<_>>();
753        if block.is_empty() {
754            return Ok(shapes);
755        }
756        let indices = block.iter().map(|&(_, row)| row).collect::<Vec<_>>();
757        let projected = accessor
758            .array_shapes_uncached(&indices)
759            .or_else(|_| {
760                indices
761                    .iter()
762                    .map(|&row| input.main_table().array_shape_uncached(row, column))
763                    .collect()
764            })
765            .map_err(table_error)?;
766        for ((index, row), found) in block.into_iter().zip(projected) {
767            let Some(found) = found.filter(|shape| groups.is_none() || !shape.contains(&0)) else {
768                continue;
769            };
770            if let Some(previous) = &shapes[index]
771                && previous != &found
772            {
773                return Err(MsTransformError::SpectralMetadata {
774                    path: path.display().to_string(),
775                    reason: format!(
776                        "{column} row {row} shape {found:?} differs from its time bin's shape {previous:?}"
777                    ),
778                });
779            }
780            shapes[index] = Some(found);
781        }
782    }
783}

785pub(super) fn push_complex32_array_row(
786    session: &mut MeasurementSetWriteSession,
787    column: &str,
788    value: &ArrayValue,
789    output_path: &Path,
790    row: usize,
791) -> Result<(), MsTransformError> {
792    let ArrayValue::Complex32(array) = value else {
793        return Err(MsTransformError::WriteMeasurementSet {
794            path: output_path.display().to_string(),
795            reason: format!("{column} row {row} is not Complex32"),
796        });
797    };
798    if array.ndim() != 2 {
799        return Err(MsTransformError::WriteMeasurementSet {
800            path: output_path.display().to_string(),
801            reason: format!("{column} row {row} is not rank-2"),
802        });
803    }
804    let values = array_in_storage_order(array);
805    session
806        .push_complex32_row(column, array.shape(), &values)
807        .map_err(|error| MsTransformError::WriteMeasurementSet {
808            path: output_path.display().to_string(),
809            reason: error.to_string(),
810        })
811}

886pub(super) fn array_in_storage_order<T: Copy>(array: &ArrayD<T>) -> Cow<'_, [T]> {
887    let storage_order = array.t();
888    match storage_order.to_slice() {
889        Some(values) => Cow::Borrowed(values),
890        None => Cow::Owned(storage_order.iter().copied().collect()),
891    }
