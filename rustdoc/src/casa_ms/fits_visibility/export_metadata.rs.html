25pub(super) struct ExportHeader {
26    pub(super) telescope: String,
27    pub(super) observer: String,
28    pub(super) object: String,
29    pub(super) epoch: f64,
30    pub(super) epoch_reference: EpochRef,
31    pub(super) direction_deg: [f64; 2],
32    pub(super) antenna_numbers: Vec<i32>,
33    pub(super) time: f64,
34    pub(super) rest_frequency: f64,
35    pub(super) measures: Option<Arc<dyn MeasuresProvider>>,
36}

38impl ExportHeader {
39    pub(super) fn read(ms: &MeasurementSet, layout: &OutputLayout) -> MsResult<Self> {
40        let obs = subtable(ms, SubtableId::Observation)?;
41        let telescope = string_cell(obs, 0, "TELESCOPE_NAME")?.to_owned();
42        let antenna = subtable(ms, SubtableId::Antenna)?;
43        let numbers = (0..antenna.row_count())
44            .map(|row| {
45                let name = string_cell(antenna, row, "NAME")?;
46                let parse = if telescope.contains("VLA") {
47                    name.trim_start_matches(|c: char| !c.is_ascii_digit())
48                } else {
49                    name
50                };
51                Ok(parse.parse::<i32>().ok())
52            })
53            .collect::<MsResult<Option<Vec<_>>>>();
54        let mut antenna_numbers = match numbers {
55            Ok(Some(numbers))
56                if numbers.iter().all(|&number| number >= 1)
57                    && numbers
58                        .iter()
59                        .collect::<std::collections::BTreeSet<_>>()
60                        .len()
61                        == numbers.len() =>
62            {
63                numbers
64            }
65            Ok(_) => (1..=antenna.row_count() as i32).collect(),
66            Err(error) => return Err(error),
67        };
68        antenna_numbers.shrink_to_fit();
69        let field = subtable(ms, SubtableId::Field)?;
70        let direction_reference = measure_ref(field, "PHASE_DIR")
71            .parse::<DirectionRef>()
72            .unwrap_or(DirectionRef::J2000);
73        let epoch = if direction_reference.as_str().contains("1950") {
74            1950.0
75        } else {
76            2000.0
77        };
78        let (object, direction_deg) = if layout.multi {
79            ("MULTI".to_owned(), [0.0; 2])
80        } else {
81            let row = layout.field as usize;
82            let mut angles = direction_degrees(field, row)?;
83            angles[0] = angles[0].rem_euclid(360.0);
84            (string_cell(field, row, "NAME")?.to_owned(), angles)
85        };
86        let rest_frequency = ms
87            .subtable(SubtableId::Source)
88            .filter(|source| source.row_count() > 0)
89            .map_or(0.0, |source| {
90                optional_array_first(source, 0, "REST_FREQUENCY")
91            });
92        let first_row = layout
93            .rows
94            .iter()
95            .map(|row| row.index)
96            .min()
97            .expect("export has rows");
98        let epoch_reference = measure_ref(ms.main_table(), "TIME")
99            .parse::<EpochRef>()
100            .unwrap_or(EpochRef::UTC);
101        Ok(Self {
102            telescope,
103            observer: string_cell(obs, 0, "OBSERVER")?.to_owned(),
104            object,
105            epoch,
106            epoch_reference,
107            direction_deg,
108            antenna_numbers,
109            time: f64_cell(ms.main_table(), first_row, "TIME")?,
110            rest_frequency,
111            measures: fits_measures(),
112        })
113    }
114
115    pub(super) fn frame(&self, time: f64, reference: EpochRef) -> MsResult<MeasFrame> {
116        let mjd = time / 86_400.0;
117        let mut frame =
118            MeasFrame::new().with_epoch(MEpoch::new(MjdHighPrec::from_mjd(mjd), reference));
119        if let Some(measures) = &self.measures {
120            frame = frame.with_measures(Arc::clone(measures));
121        }
122        if frame.dut1_for_mjd(mjd)?.is_none() {
123            frame = frame.with_dut1(0.0);
124        }
125        Ok(frame)
126    }
127
128    pub(super) fn separate_baselines(&self) -> bool {
129        self.antenna_numbers.iter().any(|&number| number >= 256)
130    }
131}

133#[cfg(not(test))]
134pub(super) fn fits_measures() -> Option<Arc<dyn MeasuresProvider>> {
135    use casa_measures::MeasuresRuntime;
136    use std::sync::Mutex;
137
138    // Catalogs are immutable for the lifetime of a runtime. Keep one provider
139    // across exports, and follow a newly selected data root on the next call.
140    static RUNTIME: Mutex<Option<Arc<MeasuresRuntime>>> = Mutex::new(None);
141    let root = MeasuresRuntime::discover_candidates()
142        .ok()?
143        .into_iter()
144        .find(|candidate| candidate.complete)?
145        .path
146        .canonicalize()
147        .ok()?;
148    let mut current = RUNTIME.lock().ok()?;
149    if let Some(runtime) = current.as_ref()
150        && runtime.root() == root
151    {
152        return Some(runtime.clone());
153    }
154    let runtime = Arc::new(MeasuresRuntime::open(root, Default::default()).ok()?);
155    *current = Some(runtime.clone());
156    Some(runtime)
157}

164pub(super) fn write_extensions(
165    fits: &mut FitsFile,
166    ms: &MeasurementSet,
167    layout: &OutputLayout,
168    metadata: &ExportHeader,
169    options: &UvFitsExportOptions<'_>,
170) -> MsResult<()> {
171    write_frequency(fits, layout, &metadata.telescope)?;
172    write_antenna(fits, ms, layout, metadata, options.writestation)?;
173    if layout.multi || layout.astron_bin_width.is_some() {
174        write_source(fits, ms, layout, metadata)?;
175    }
176    if options.writesyscal {
177        write_system_calibration(fits, ms, layout, metadata)?;
178    }
179    write_weather(fits, ms, metadata)
180}

182fn write_frequency(fits: &mut FitsFile, layout: &OutputLayout, telescope: &str) -> MsResult<()> {
183    let count = if layout.combine {
184        layout.windows.len()
185    } else {
186        1
187    };
188    let hdu = create_table(
189        fits,
190        "AIPS FQ",
191        &[
192            ("FRQSEL", Type::Int, 1, ""),
193            ("IF FREQ", Type::Double, count, "HZ"),
194            ("CH WIDTH", Type::Float, count, "HZ"),
195            ("TOTAL BANDWIDTH", Type::Float, count, "HZ"),
196            ("SIDEBAND", Type::Int, count, ""),
197        ],
198    )?;
199    hdu.write_key(fits, "EXTVER", 1).map_err(invalid)?;
200    hdu.write_key(fits, "NO_IF", count as i32)
201        .map_err(invalid)?;
202    let ids = if layout.combine {
203        vec![1]
204    } else {
205        (1..=layout.windows.len() as i32).collect()
206    };
207    let offsets = layout
208        .windows
209        .iter()
210        .map(|window| {
211            if matches!(telescope, "IRAM PDB" | "IRAM_PDB") {
212                0.0
213            } else {
214                window.frequency - layout.reference
215            }
216        })
217        .collect::<Vec<_>>();
218    let increments = layout
219        .windows
220        .iter()
221        .map(|window| {
222            (if telescope == "ALMA" {
223                layout.step.abs().copysign(window.increment)
224            } else {
225                layout.step
226            }) as f32
227        })
228        .collect::<Vec<_>>();
229    let bandwidths = layout
230        .windows
231        .iter()
232        .map(|window| window.bandwidth as f32)
233        .collect::<Vec<_>>();
234    let sidebands = layout
235        .windows
236        .iter()
237        .map(|window| {
238            if telescope == "ALMA" {
239                if window.increment < 0.0 { -1 } else { 1 }
240            } else {
241                window.sideband
242            }
243        })
244        .collect::<Vec<_>>();
245    put(&hdu, fits, "FRQSEL", &ids)?;
246    put(&hdu, fits, "IF FREQ", &offsets)?;
247    put(&hdu, fits, "CH WIDTH", &increments)?;
248    put(&hdu, fits, "TOTAL BANDWIDTH", &bandwidths)?;
249    put(&hdu, fits, "SIDEBAND", &sidebands)
250}

252fn write_antenna(
253    fits: &mut FitsFile,
254    ms: &MeasurementSet,
255    layout: &OutputLayout,
256    meta: &ExportHeader,
257    station: bool,
258) -> MsResult<()> {
259    let antenna = subtable(ms, SubtableId::Antenna)?;
260    let obs = subtable(ms, SubtableId::Observation)?;
261    let names = fingerprints(antenna, "NAME")?;
262    let stations = fingerprints(antenna, "STATION")?;
263    let names = if (station || all_equal(&names)) && !all_equal(&stations) {
264        stations
265    } else {
266        names
267    };
268    let frame = measure_ref(antenna, "POSITION")
269        .parse::<PositionRef>()
270        .unwrap_or(PositionRef::ITRF);
271    let feeds = relevant_feeds(ms, layout)?;
272    let cells = antenna_cells(antenna, frame, &feeds)?;
273    for array in 0..obs.row_count() {
274        let telescope = string_cell(obs, array, "TELESCOPE_NAME")?;
275        let hdu = create_table(
276            fits,
277            "AIPS AN",
278            &[
279                ("ANNAME", Type::String, 8, ""),
280                ("STABXYZ", Type::Double, 3, "METERS"),
281                ("ORBPARM", Type::Double, 0, ""),
282                ("NOSTA", Type::Int, 1, ""),
283                ("MNTSTA", Type::Int, 1, ""),
284                ("STAXOF", Type::Float, 1, "METERS"),
285                ("POLTYA", Type::String, 1, ""),
286                ("POLAA", Type::Float, 1, "DEGREES"),
287                ("POLCALA", Type::Float, 1, ""),
288                ("POLTYB", Type::String, 1, ""),
289                ("POLAB", Type::Float, 1, "DEGREES"),
290                ("POLCALB", Type::Float, 1, ""),
291                ("DIAMETER", Type::Float, 1, ""),
292            ],
293        )?;
294        hdu.write_key(fits, "EXTVER", (array + 1) as i32)
295            .map_err(invalid)?;
296        write_antenna_keywords(fits, &hdu, telescope, layout.reference, meta)?;
297        put(&hdu, fits, "ANNAME", &names)?;
298        put(&hdu, fits, "STABXYZ", &cells.positions)?;
299        put(&hdu, fits, "NOSTA", &meta.antenna_numbers)?;
300        put(&hdu, fits, "MNTSTA", &cells.mounts)?;
301        put(&hdu, fits, "STAXOF", &cells.offsets)?;
302        put(&hdu, fits, "DIAMETER", &cells.diameters)?;
303        for (index, suffix) in ["A", "B"].into_iter().enumerate() {
304            put(
305                &hdu,
306                fits,
307                &format!("POLTY{suffix}"),
308                &cells.polarizations[index],
309            )?;
310            put(&hdu, fits, &format!("POLA{suffix}"), &cells.angles[index])?;
311            put(
312                &hdu,
313                fits,
314                &format!("POLCAL{suffix}"),
315                &vec![0.0f32; names.len()],
316            )?;
317        }
318    }
319    Ok(())
320}

322fn write_antenna_keywords(
323    fits: &mut FitsFile,
324    hdu: &FitsHdu,
325    telescope: &str,
326    reference: f64,
327    meta: &ExportHeader,
328) -> MsResult<()> {
329    let time = MjdHighPrec::from_mjd(meta.time / 86_400.0);
330    let epoch = MEpoch::new(time, meta.epoch_reference);
331    let frame = meta.frame(meta.time, meta.epoch_reference)?;
332    let utc = epoch.convert_to(EpochRef::UTC, &frame).map_err(invalid)?;
333    let iat_utc = epoch
334        .convert_to(EpochRef::TAI, &frame)
335        .map(|tai| (tai.value().as_mjd() - utc.value().as_mjd()) * 86_400.0)
336        .unwrap_or(0.0);
337    let utc_mjd = utc.value().as_mjd();
338    let start = (utc_mjd + iat_utc / 86_400.0).floor();
339    let (x, y) = frame
340        .polar_motion_for_mjd(utc_mjd.floor())?
341        .unwrap_or((0.0, 0.0));
342    let polar_m = |arcseconds: f64| 6_356_752.31 * (arcseconds / 3600.0).to_radians();
343    let dut1 = frame.dut1_for_mjd(utc_mjd)?.unwrap_or(0.0);
344    let gst = [start, start + 1.0].map(|day| {
345        MEpoch::new(MjdHighPrec::from_mjd(day), EpochRef::UTC)
346            .convert_to(EpochRef::GMST1, &frame)
347            .map(|value| value.value().as_mjd())
348    });
349    let [gst, next] = gst;
350    let gst = gst.map_err(invalid)?;
351    let degrees = 360.0 * (next.map_err(invalid)? - gst);
352    for (key, value) in [
353        ("ARRAYX", 0.0),
354        ("ARRAYY", 0.0),
355        ("ARRAYZ", 0.0),
356        ("GSTIA0", 360.0 * gst.fract()),
357        ("DEGPDY", degrees),
358        ("FREQ", reference),
359        ("POLARX", polar_m(x)),
360        ("POLARY", polar_m(y)),
361        ("UT1UTC", dut1),
362        ("IATUTC", iat_utc),
363    ] {
364        hdu.write_key(fits, key, value).map_err(invalid)?;
365    }
366    for (key, value) in [("NUMORB", 0), ("NOPCAL", 0)] {
367        hdu.write_key(fits, key, value).map_err(invalid)?;
368    }
369    let date = fits_date(meta.time, false);
370    let reference = meta.epoch_reference.to_string();
371    for (key, value) in [
372        ("RDATE", date.as_str()),
373        ("TIMSYS", reference.as_str()),
374        ("ARRNAM", telescope),
375        ("POLTYPE", " "),
376        ("XYZHAND", "RIGHT"),
377        ("FRAME", "ITRF"),
378    ] {
379        hdu.write_key(fits, key, value).map_err(invalid)?;
380    }
381    Ok(())
382}

384type Feed = [(String, f32); 2];
385
386fn relevant_feeds(ms: &MeasurementSet, layout: &OutputLayout) -> MsResult<BTreeMap<i32, Feed>> {
387    let table = subtable(ms, SubtableId::Feed)?;
388    let mut feeds = BTreeMap::new();
389    for row in 0..table.row_count() {
390        let id = i32_cell(table, row, "ANTENNA_ID")?;
391        let spw = i32_cell(table, row, "SPECTRAL_WINDOW_ID")?;
392        if feeds.contains_key(&id)
393            || (spw >= 0 && !layout.windows.iter().any(|window| window.id == spw))
394        {
395            continue;
396        }
397        let ArrayValue::String(types) = table.cell_accessor(row, "POLARIZATION_TYPE")?.array()?
398        else {
399            return Err(invalid("Invalid feed polarization type"));
400        };
401        let angles = numeric_array(table, row, "RECEPTOR_ANGLE")?;
402        let mut hands = std::array::from_fn(|_| (" ".to_owned(), 0.0));
403        for ((hand, pol), angle) in hands.iter_mut().zip(types.iter()).zip(angles) {
404            *hand = (pol.as_str().to_owned(), angle.to_degrees() as f32);
405        }
406        feeds.insert(id, hands);
407    }
408    Ok(feeds)
409}

411#[derive(Default)]
412struct AntennaCells {
413    positions: Vec<f64>,
414    mounts: Vec<i32>,
415    offsets: Vec<f32>,
416    diameters: Vec<f32>,
417    polarizations: [Vec<String>; 2],
418    angles: [Vec<f32>; 2],
419}
420
421fn antenna_cells(
422    table: &Table,
423    frame: PositionRef,
424    feeds: &BTreeMap<i32, Feed>,
425) -> MsResult<AntennaCells> {
426    let mut cells = AntennaCells::default();
427    for row in 0..table.row_count() {
428        let position = numeric_array(table, row, "POSITION")?;
429        let [x, y, z]: [f64; 3] = position
430            .try_into()
431            .map_err(|_| invalid("Invalid antenna position shape"))?;
432        let position = match frame {
433            PositionRef::ITRF => MPosition::new_itrf(x, y, z),
434            PositionRef::WGS84 => MPosition::new_wgs84(x, y, z),
435        };
436        cells.positions.extend(
437            position
438                .convert_to(PositionRef::ITRF)
439                .map_err(invalid)?
440                .values(),
441        );
442        let mount = string_cell(table, row, "MOUNT")?.to_uppercase();
443        let names = [
444            "ALT-AZ+NASMYTH-R",
445            "ALT-AZ+NASMYTH-L",
446            "ALT-AZ+BWG-R",
447            "ALT-AZ+BWG-L",
448            "ALT-AZ",
449            "EQUATORIAL",
450            "ORBIT",
451            "X-Y",
452            "BIZARRE",
453        ];
454        cells.mounts.push(
455            names
456                .iter()
457                .position(|name| mount.contains(name))
458                .map_or(9, |index| [4, 5, 6, 7, 0, 1, 2, 3, 8][index]),
459        );
460        cells.offsets.push(
461            numeric_array(table, row, "OFFSET")?
462                .first()
463                .copied()
464                .unwrap_or_default() as f32,
465        );
466        cells
467            .diameters
468            .push(f64_cell(table, row, "DISH_DIAMETER")? as f32);
469        for index in 0..2 {
470            let feed = feeds.get(&(row as i32)).map(|hands| &hands[index]);
471            cells.polarizations[index]
472                .push(feed.map_or_else(|| " ".to_owned(), |value| value.0.clone()));
473            cells.angles[index].push(feed.map_or(0.0, |value| value.1));
474        }
475    }
476    Ok(cells)
477}

479fn write_source(
480    fits: &mut FitsFile,
481    ms: &MeasurementSet,
482    layout: &OutputLayout,
483    meta: &ExportHeader,
484) -> MsResult<()> {
485    let field = subtable(ms, SubtableId::Field)?;
486    let fields = fingerprints(field, "NAME")?;
487    let width = layout.windows.len();
488    let hdu = create_table(
489        fits,
490        "AIPS SU",
491        &[
492            ("ID. NO.", Type::Int, 1, ""),
493            ("SOURCE", Type::String, 20, ""),
494            ("QUAL", Type::Int, 1, ""),
495            ("CALCODE", Type::String, 4, ""),
496            ("IFLUX", Type::Float, width, "JY"),
497            ("QFLUX", Type::Float, width, "JY"),
498            ("UFLUX", Type::Float, width, "JY"),
499            ("VFLUX", Type::Float, width, "JY"),
500            ("FREQOFF", Type::Double, width, "HZ"),
501            ("BANDWIDTH", Type::Double, 1, "HZ"),
502            ("RAEPO", Type::Double, 1, "DEGREES"),
503            ("DECEPO", Type::Double, 1, "DEGREES"),
504            ("EPOCH", Type::Double, 1, "YEARS"),
505            ("RAAPP", Type::Double, 1, "DEGREES"),
506            ("DECAPP", Type::Double, 1, "DEGREES"),
507            ("LSRVEL", Type::Double, width, "M/SEC"),
508            ("RESTFREQ", Type::Double, width, "HZ"),
509            ("PMRA", Type::Double, 1, "DEG/DAY"),
510            ("PMDEC", Type::Double, 1, "DEG/DAY"),
511        ],
512    )?;
513    hdu.write_key(fits, "EXTVER", 1).map_err(invalid)?;
514    hdu.write_key(fits, "NO_IF", width as i32)
515        .map_err(invalid)?;
516    hdu.write_key(fits, "FREQID", 1).map_err(invalid)?;
517    if layout.astron_bin_width.is_some() {
518        let spectral = subtable(ms, SubtableId::SpectralWindow)?;
519        for (keyword, column) in [
520            ("VELTYP", "NFRA_VELOCDEFINITION"),
521            ("VELDEF", "NFRA_CONVERSIONTYPE"),
522        ] {
523            if has_column(spectral, column) && spectral.is_cell_defined(0, column)? {
524                hdu.write_key(
525                    fits,
526                    keyword,
527                    string_cell(spectral, 0, column)?.to_ascii_uppercase(),
528                )
529                .map_err(invalid)?;
530            }
531        }
532    }
533    let sources = ms.subtable(SubtableId::Source);
534    let mut cells = SourceCells::default();
535    for (&id, &number) in &layout.fields {
536        let row = id as usize;
537        let name = fields.get(row).ok_or_else(|| invalid("Invalid field ID"))?;
538        cells.ids.push(number);
539        cells.names.push(name.clone());
540        cells.qualifiers.push(
541            if fields.iter().filter(|other| other == &name).count() > 1 {
542                id
543            } else {
544                0
545            },
546        );
547        cells.directions.push(direction_degrees(field, row)?);
548        cells.apparent.push(apparent_directions(field, row, meta)?);
549        let source_id = i32_cell(field, row, "SOURCE_ID")?;
550        let source = sources.and_then(|table| {
551            (0..table.row_count())
552                .find(|&index| i32_cell(table, index, "SOURCE_ID").ok() == Some(source_id))
553                .map(|index| (table, index))
554        });
555        cells.record_source(source, width)?;
556    }
557    cells.write(fits, &hdu, width, layout.windows[0].bandwidth, meta.epoch)
558}

560#[derive(Default)]
561struct SourceCells {
562    ids: Vec<i32>,
563    names: Vec<String>,
564    qualifiers: Vec<i32>,
565    codes: Vec<String>,
566    directions: Vec<[f64; 2]>,
567    apparent: Vec<[f64; 2]>,
568    frequencies: Vec<f64>,
569    velocities: Vec<f64>,
570    motion: Vec<[f64; 2]>,
571}
572
573impl SourceCells {
574    fn record_source(&mut self, source: Option<(&Table, usize)>, width: usize) -> MsResult<()> {
575        if let Some((table, row)) = source {
576            self.codes.push(string_cell(table, row, "CODE")?.to_owned());
577            self.motion.push(proper_motion(table, row)?);
578            self.frequencies.extend(std::iter::repeat_n(
579                optional_array_first(table, row, "REST_FREQUENCY"),
580                width,
581            ));
582            self.velocities.extend(std::iter::repeat_n(
583                optional_array_first(table, row, "SYSVEL"),
584                width,
585            ));
586        } else {
587            self.codes.push(String::new());
588            self.motion.push([0.0; 2]);
589            self.frequencies.extend(std::iter::repeat_n(0.0, width));
590            self.velocities.extend(std::iter::repeat_n(0.0, width));
591        }
592        Ok(())
593    }
594
595    fn write(
596        self,
597        fits: &mut FitsFile,
598        hdu: &FitsHdu,
599        width: usize,
600        bandwidth: f64,
601        epoch: f64,
602    ) -> MsResult<()> {
603        let count = self.ids.len();
604        put(hdu, fits, "ID. NO.", &self.ids)?;
605        put(hdu, fits, "SOURCE", &self.names)?;
606        put(hdu, fits, "QUAL", &self.qualifiers)?;
607        put(hdu, fits, "CALCODE", &self.codes)?;
608        let flux = vec![0.0f32; width * count];
609        for name in ["IFLUX", "QFLUX", "UFLUX", "VFLUX"] {
610            put(hdu, fits, name, &flux)?;
611        }
612        put(hdu, fits, "FREQOFF", &vec![0.0f64; width * count])?;
613        put(hdu, fits, "BANDWIDTH", &vec![bandwidth; count])?;
614        put(hdu, fits, "EPOCH", &vec![epoch; count])?;
615        put(hdu, fits, "RESTFREQ", &self.frequencies)?;
616        put(hdu, fits, "LSRVEL", &self.velocities)?;
617        for (columns, pairs) in [
618            (["RAEPO", "DECEPO"], self.directions),
619            (["RAAPP", "DECAPP"], self.apparent),
620            (["PMRA", "PMDEC"], self.motion),
621        ] {
622            for (index, name) in columns.into_iter().enumerate() {
623                put(
624                    hdu,
625                    fits,
626                    name,
627                    &pairs.iter().map(|pair| pair[index]).collect::<Vec<_>>(),
628                )?;
629            }
630        }
631        Ok(())
632    }
633}

635fn proper_motion(table: &Table, row: usize) -> MsResult<[f64; 2]> {
636    let column = "PROPER_MOTION";
637    if !table.is_cell_defined(row, column)? {
638        return Ok([0.0; 2]);
639    }
640    let motion = numeric_array(table, row, column)?;
641    let mut values = [0.0; 2];
642    let units = match table
643        .column_keywords(column)
644        .and_then(|record| record.get("QuantumUnits"))
645    {
646        Some(Value::Array(ArrayValue::String(units))) => units.as_slice(),
647        _ => None,
648    };
649    let degrees_per_day = Unit::new("deg/d").map_err(invalid)?;
650    for (axis, (output, input)) in values.iter_mut().zip(motion).enumerate() {
651        let unit = units
652            .and_then(|names| names.get(axis).or_else(|| names.first()))
653            .map_or("rad/s", String::as_str);
654        // CASA treats a value with unrecognized or incompatible units as deg/d.
655        *output = Quantity::new(input, unit)
656            .and_then(|value| value.get_value_in(&degrees_per_day))
657            .unwrap_or(input);
658    }
659    Ok(values)
660}

662fn direction_degrees(table: &Table, row: usize) -> MsResult<[f64; 2]> {
663    let ArrayValue::Float64(value) = table.cell_accessor(row, "PHASE_DIR")?.array()? else {
664        return Err(invalid("Invalid phase direction type"));
665    };
666    if value.ndim() != 2 || value.shape()[0] != 2 || value.shape()[1] == 0 {
667        return Err(invalid("Invalid phase direction shape"));
668    }
669    Ok([value[[0, 0]].to_degrees(), value[[1, 0]].to_degrees()])
670}
671
672fn apparent_directions(table: &Table, row: usize, meta: &ExportHeader) -> MsResult<[f64; 2]> {
673    let angles = direction_degrees(table, row)?.map(f64::to_radians);
674    let reference = measure_ref(table, "PHASE_DIR")
675        .parse::<DirectionRef>()
676        .unwrap_or(DirectionRef::J2000);
677    let direction = MDirection::from_angles(angles[0], angles[1], reference);
678    let time = f64_cell(table, row, "TIME")?;
679    let reference = measure_ref(table, "TIME")
680        .parse::<EpochRef>()
681        .unwrap_or(meta.epoch_reference);
682    let frame = meta.frame(time, reference)?;
683    let direction = direction
684        .convert_to(DirectionRef::APP, &frame)
685        .map_err(invalid)?;
686    let (ra, dec) = direction.as_angles();
687    Ok([ra.sin().atan2(ra.cos()).to_degrees(), dec.to_degrees()])
688}

693            .ok()

705fn fingerprints(table: &Table, column: &str) -> MsResult<Vec<String>> {
706    (0..table.row_count())

711fn numeric_array(table: &Table, row: usize, column: &str) -> MsResult<Vec<f64>> {
712    match table.cell_accessor(row, column)?.array()? {
713        ArrayValue::Float64(value) => Ok(value.iter().copied().collect()),
714        _ => Err(invalid(format!(
715            "Invalid floating point array column {column}"
716        ))),
717    }
718}

720fn string_cell<'a>(table: &'a Table, row: usize, column: &str) -> MsResult<&'a str> {
721    match table.cell_accessor(row, column)?.scalar()? {
722        ScalarValue::String(value) => Ok(value),
723        _ => Err(invalid(format!("Invalid string column {column}"))),
724    }
725}

727pub(super) fn measure_ref<'a>(table: &'a Table, column: &str) -> &'a str {
728    match table
729        .column_keywords(column)
730        .and_then(|keywords| keywords.get("MEASINFO"))
731    {
732        Some(Value::Record(info)) => match info.get("Ref") {
733            Some(Value::Scalar(ScalarValue::String(reference))) => reference,
734            _ => "",
735        },
736        _ => "",
737    }
738}

740pub(super) fn create_table(
741    fits: &mut FitsFile,
742    name: &str,
743    columns: &[(&str, Type, usize, &str)],
744) -> MsResult<FitsHdu> {
745    let descriptors = columns
746        .iter()
747        .map(|&(name, kind, count, _)| {
748            ColumnDescription::new(name)
749                .with_type(kind)
750                .that_repeats(count)
751                .create()
752                .map_err(invalid)
753        })
754        .collect::<MsResult<Vec<_>>>()?;
755    let hdu = fits.create_table(name, &descriptors).map_err(invalid)?;
756    for (index, &(_, _, _, units)) in columns.iter().enumerate() {
757        if !units.is_empty() {
758            hdu.write_key(fits, &format!("TUNIT{}", index + 1), units)
759                .map_err(invalid)?;
760        }
761    }
762    Ok(hdu)
763}

766    hdu: &FitsHdu,
