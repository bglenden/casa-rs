2//! `simutil`'s analysis helpers, which CASA's `simanalyze` and `simalma` use
3//! on a simulated image: `ismstp`, `cellsize`, `statim` and `convimage`,
4//! over [`casa_simulation::simutil::analysis`], which records where its
5//! results depart from CASA's.
6//!
7//! Each returns what CASA 6.7.6.14's returns:
8//! - `ismstp`: a `bool`;
9//! - `cellsize`: a list of two quantity records, in the image's direction
10//!   units (radians);
11//! - `statim`: `(min, max, rms, unit)`, the three values NumPy 0-d `float64`
12//!   arrays, as CASA's `np.squeeze` leaves them;
13//! - `convimage`: `None`, having written `<modelflat>.regrid` and
14//!   `<modelflat>.regrid.conv`.
15//!
16//! `statim` draws its native-prepared plane through the shared Python
17//! renderer. `convimage` with a `complist` is refused (#1053).
18

35/// A refusal CASA raises through `simutil.msg(..., priority='error')` is a
36/// `RuntimeError` with its message; a file that cannot be read an `OSError`.
37fn analysis_raised(error: AnalysisError) -> PyErr {
38    match error {
39        AnalysisError::Io { .. } => PyOSError::new_err(error.to_string()),
40        other => PyRuntimeError::new_err(other.to_string()),
41    }
42}

44#[pymethods]
45impl PySimutil {
46    /// Whether `s` is a total-power MeasurementSet: its first antenna's name
47    /// holds `TP`, it has one antenna, or every UVW value is zero. With
48    /// `halt`, a MeasurementSet that is not, or cannot be read, raises with
49    /// CASA's message; otherwise either is `False`.
50    #[pyo3(signature = (s, halt = false), text_signature = "($self, s, halt=False)")]
51    fn ismstp(&self, py: Python<'_>, s: PathBuf, halt: bool) -> PyResult<bool> {
52        let shown = s.display().to_string();
53        match py.detach(|| is_total_power(&s)) {
54            Ok(true) => Ok(true),
55            Ok(false) if halt => Err(PyRuntimeError::new_err(format!(
56                "input file '{shown}' is not a totalpower ms"
57            ))),
58            Err(_) if halt => Err(PyRuntimeError::new_err(format!(
59                "can't understand the file '{shown}'"
60            ))),
61            _ => Ok(false),
62        }
63    }

65    /// The image's cell on the sky: `[cellx, celly]`, quantity records in
66    /// radians, positive. An image rotated with respect to latitude and
67    /// longitude raises CASA's message.
68    #[pyo3(signature = (image), text_signature = "($self, image)")]
69    fn cellsize(&self, py: Python<'_>, image: PathBuf) -> PyResult<Py<PyAny>> {
70        let [x, y] = py
71            .detach(|| {
72                let opened = PagedImage::<f32>::open(&image)?;
73                cell_size(opened.coordinates())
74            })
75            .map_err(analysis_raised)?;
76        CasaValue::List(vec![
77            CasaValue::quantity(x, "rad"),
78            CasaValue::quantity(y, "rad"),
79        ])
80        .into_python(py)
81    }

83    /// The image's minimum, maximum, the rms of its lower-left quarter and
84    /// its brightness unit, as CASA's `statim` gives them. By default draw
85    /// the complete plane, with CASA's display range, colour scale and
86    /// statistics. An empty `disprange` list receives its computed limits.
87    /// `incell` is unused in CASA and here.
88    #[pyo3(
89        signature = (image, plot = true, incell = None, disprange = None, bar = true, showstats = true),
90        text_signature = "($self, image, plot=True, incell=None, disprange=None, bar=True, showstats=True)"
91    )]
92    // CASA's signature: the plot's controls are parameters of their own.
93    #[allow(unused_variables, clippy::too_many_arguments)]
94    fn statim<'py>(
95        &self,
96        py: Python<'py>,
97        image: PathBuf,
98        plot: bool,
99        incell: Option<&Bound<'py, PyAny>>,
100        disprange: Option<&Bound<'py, PyAny>>,
101        bar: bool,
102        showstats: bool,
103    ) -> PyResult<Bound<'py, PyTuple>> {
104        let range_list = disprange.and_then(|value| value.cast::<PyList>().ok());
105        let display_range = disprange
106            .map(|value| {
107                if let Some(list) = range_list {
108                    list.extract::<Vec<f64>>()
109                } else {
110                    value.extract::<f64>().map(|value| vec![value])
111                }
112            })
113            .transpose()?;
114        if display_range
115            .as_ref()
116            .is_some_and(|values| values.len() > 2)
117        {
118            return Err(PyRuntimeError::new_err(
119                "statim disprange has too many elements",
120            ));
121        }
122        let return_range = range_list.is_some_and(|list| list.is_empty());
123        let stats = if plot || return_range {
124            let file = tempfile::NamedTempFile::new()?;
125            let options = ImagePlotOptions {
126                display_range,
127                bar,
128                show_statistics: showstats,
129            };
130            let (document, limits, stats) = py.detach(|| {
131                let resources = SessionResources::from_controls(&ExecutionControls::default())
132                    .map_err(|error| PyRuntimeError::new_err(error.to_string()))?;
133                image_plot(&image, file.path(), &options, resources)
134                    .map_err(|error| PyRuntimeError::new_err(error.to_string()))
135            })?;
136            if return_range {
137                for limit in limits {
138                    range_list.expect("empty caller list").append(limit)?;
139                }
140            }
141            if plot {
142                graphics::draw(py, &document, file.path(), false)?;
143            }
144            stats
145        } else {
146            py.detach(|| statim(&image)).map_err(analysis_raised)?
147        };
148        let zero_d = |value: f64| ndarray::arr0(value).into_pyarray(py).into_any();
149        PyTuple::new(
150            py,
151            [
152                zero_d(stats.min),
153                zero_d(stats.max),
154                zero_d(stats.rms),
155                stats.unit.into_pyobject(py)?.into_any(),
156            ],
157        )
158    }

160    /// Regrid the flat sky model `modelflat` onto the flat image `outflat`
161    /// and convolve it with the image's restoring beam: `<modelflat>.regrid`
162    /// and `<modelflat>.regrid.conv`, as CASA writes them. A `complist` is
163    /// refused (#1053).
164    #[pyo3(signature = (modelflat, outflat, complist = ""), text_signature = "($self, modelflat, outflat, complist='')")]
165    fn convimage(
166        &self,
167        py: Python<'_>,
168        modelflat: PathBuf,
169        outflat: PathBuf,
170        complist: &str,
171    ) -> PyResult<()> {
172        if !complist.is_empty() {
173            return Err(PyNotImplementedError::new_err(
174                "simutil.convimage with a complist is not yet implemented in casa-rs (#1053)",
175            ));
176        }
177        py.detach(|| convimage(&modelflat, &outflat))
178            .map(|_| ())
179            .map_err(analysis_raised)
180    }
