24pub(super) struct AntennaPaths {
25    pub time_seconds: f64,
26    pub field_id: i32,
27    pub meters: Vec<f64>,
28}

30pub(super) struct PhaseWindow {
31    pub id: i32,
32    pub frequency_hz: f64,
33    pub reverse: bool,
34    pub dispersion: f64,
35}

37pub(super) fn publish(
38    ms: &MeasurementSet,
39    request: &WvrRequest,
40    integrations: &[AntennaPaths],
41    windows: &[PhaseWindow],
42    mode: CalibrationOutputMode,
43) -> Result<(), WvrError> {
44    let subtables = [
45        (SubtableId::Antenna, "ANTENNA"),
46        (SubtableId::Field, "FIELD"),
47        (SubtableId::Observation, "OBSERVATION"),
48        (SubtableId::SpectralWindow, "SPECTRAL_WINDOW"),
49    ];
50    for &(id, name) in &subtables {
51        if ms.subtable(id).is_none() {
52            return Err(MsError::MissingSubtable(name.into()).into());
53        }
54    }
55    let output = &request.output_table;
56    let ms_name = ms
57        .path()
58        .and_then(Path::file_name)
59        .map(|part| part.to_string_lossy())
60        .unwrap_or_else(|| "<in-memory>".into());
61    let mut writer = CalibrationTableWriter::with_mode(
62        ms,
63        CalibrationTableDescriptor {
64            output,
65            schema: new_calibration_table_schema(true),
66            subtype: "T Jones",
67            parameter_type: Some("Complex"),
68            measurement_set_name: ms_name.into_owned(),
69            include_polarization_basis: true,
70            time_extra_precision_column: None,
71        },
72        mode,
73    )?;
74    let interval = integrations.get(1).map_or(0.0, |second| {
75        second.time_seconds - integrations[0].time_seconds
76    });
77    for integration in integrations {
78        for window in windows {
79            for (antenna, &path) in integration.meters.iter().enumerate() {
80                writer.append(solution_row(
81                    integration,
82                    window,
83                    interval,
84                    antenna as i32,
85                    path,
86                ))?;
87            }
88        }
89    }
90    writer.copy_subtables(&subtables)?;
91    make_windows_single_channel(
92        &writer.output_path().join("SPECTRAL_WINDOW"),
93        SingleChannelFrequency::MiddleChannel,
94    )?;
95    write_history(&writer.output_path().join("HISTORY"), request)?;
96    Ok(writer.save_main()?)
97}

99fn write_history(path: &Path, request: &WvrRequest) -> Result<(), CalibrationTableWriteError> {
100    let options = TableOptions::new(path);
101    let failure = |source| CalibrationTableWriteError::Save {
102        path: path.display().to_string(),
103        source: Box::new(source),
104    };
105    let schema = build_table_schema(history::REQUIRED_COLUMNS).expect("valid WVR history schema");
106    let mut table = Table::with_schema(schema);
107    crate::writer::set_fixed_unit_keyword(&mut table, COL_TIME, &["s"]);
108    crate::writer::set_measinfo_keyword(&mut table, COL_TIME, "epoch", Some("UTC"));
109    let mut row = RecordValue::new(Vec::new());
110    let parameters = serde_json::to_string(request).expect("WVR requests are serializable");
111    for (column, text) in [
112        ("APPLICATION", "Produced with casa-rs WVR calibration"),
113        ("MESSAGE", &parameters),
114        ("ORIGIN", "wvrgcal"),
115        ("PRIORITY", "INFO"),
116    ] {
117        row.upsert(column, Value::Scalar(ScalarValue::String(text.into())));
118    }
119    for name in ["OBSERVATION_ID", "OBJECT_ID"] {
120        row.upsert(name, Value::Scalar(ScalarValue::Int32(-1)));
121    }
122    let now = std::time::SystemTime::now()
123        .duration_since(std::time::UNIX_EPOCH)
124        .map_or(0.0, |duration| duration.as_secs_f64());
125    row.upsert(
126        "TIME",
127        Value::Scalar(ScalarValue::Float64(now + 40_587.0 * 86_400.0)),
128    );
129    table.add_row(row).map_err(failure)?;
130    table.save(options).map_err(failure)
131}

133fn solution_row(
134    integration: &AntennaPaths,
135    window: &PhaseWindow,
136    interval: f64,
137    antenna: i32,
138    path: f64,
139) -> RecordValue {
140    let (correction, flagged) = gain(path, window.frequency_hz, window.reverse, window.dispersion);
141    let scalar = |name, value| RecordField::new(name, Value::Scalar(value));
142    let array = |name, value| RecordField::new(name, Value::Array(value));
143    let shape = || IxDyn(&[1, 1]);
144    RecordValue::new(vec![
145        scalar(COL_TIME, ScalarValue::Float64(integration.time_seconds)),
146        scalar(COL_FIELD_ID, ScalarValue::Int32(integration.field_id)),
147        scalar(COL_SPECTRAL_WINDOW_ID, ScalarValue::Int32(window.id)),
148        scalar(COL_ANTENNA1, ScalarValue::Int32(antenna)),
149        scalar(COL_ANTENNA2, ScalarValue::Int32(-1)),
150        scalar(COL_INTERVAL, ScalarValue::Float64(interval)),
151        scalar(COL_SCAN_NUMBER, ScalarValue::Int32(-1)),
152        scalar(COL_OBSERVATION_ID, ScalarValue::Int32(0)),
153        array(
154            COL_CPARAM,
155            ArrayValue::Complex32(ArrayD::from_elem(shape(), correction)),
156        ),
157        array(
158            COL_FLAG,
159            ArrayValue::Bool(ArrayD::from_elem(shape(), flagged)),
160        ),
161        array(
162            COL_PARAMERR,
163            ArrayValue::Float32(ArrayD::from_elem(shape(), 0.0)),
164        ),
165        array(
166            COL_SNR,
167            ArrayValue::Float32(ArrayD::from_elem(shape(), 3.0)),
168        ),
169    ])
170}
