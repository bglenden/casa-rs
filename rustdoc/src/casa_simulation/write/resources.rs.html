2//! What `close` holds while it writes MAIN, admitted against the session's
3//! resource policy before anything is written (plan §4: the runtime's
4//! budget, not a second one).
5//!
6//! In order:
7//! 1. **The writer.** The MeasurementSet write plan is sized from half of
8//!    what the policy leaves free, and its resident bytes are admitted with
9//!    the subtable columns its save generates (POINTING's runs, a few
10//!    hundred bytes a scan). Its batches are tile bands of about
11//!    [`casa_ms::TARGET_BATCH_BYTES`], so the budget is a ceiling that only
12//!    a small budget reaches.
13//! 2. **The pipeline.** The batch being filled, the geometry of the
14//!    integrations it spans, each worker's scratch and, when the session
15//!    predicts, the predictor, its block of native rows and (on Metal) the
16//!    device copies of the model grids. A block is the whole batch when
17//!    that fits with the policy's streaming workers; otherwise blocks are
18//!    small, and as many workers as fit what is then free are admitted
19//!    (#737's rule: workers shrink first). With one worker that does not
20//!    fit, `close` is refused.
21//!
22//! The model grids themselves were admitted by `predict`, and are held.
23//! A refusal is a typed [`Admission`] carried by the session error.
24

37/// The writer's phase, as an admission refusal names it.
38pub(super) const WRITER_PHASE: &str = "the simulator's MeasurementSet writer";
39
40/// The pipeline's phase, as an admission refusal names it.
41pub(super) const PIPELINE_PHASE: &str = "the simulator's geometry, prediction and noise";
42
43/// Batches the writer holds at once: the one it writes and one queued.
44const WRITER_LIVE_BATCHES: usize = 2;
45
46/// A writer budget larger than any plan's needs, to tell a budget too small
47/// from a plan that cannot be made.
48const UNBOUNDED_WRITER_BYTES: usize = 1 << 50;
49
50/// Scratch of one worker's integration, per antenna: the antennas' uvw,
51/// their shadow flags and elevations, and for `tsys-manual` noise their
52/// airmasses (and elevations at a moved pointing's phase centre), with room
53/// for the measures frames.
54const ANTENNA_SCRATCH_BYTES: u64 = 64;

56/// The MAIN table `close` writes.
57#[derive(Debug, Clone, Copy)]
58pub(super) struct MainShape<'a> {
59    pub(super) rows: usize,
60    pub(super) correlations: usize,
61    pub(super) channels: usize,
62    pub(super) telescope: &'a str,
63    /// Rows of the session's largest integration.
64    pub(super) largest_integration: usize,
65    /// Antennas of the session's largest scan.
66    pub(super) antennas: usize,
67    /// Heap of the subtable columns the save generates (POINTING's), which
68    /// grows with the scans.
69    pub(super) generated_bytes: u64,
70}

72/// What `close` was admitted: the write plan, the workers that fit, and the
73/// memory, held until the write ends.
74#[derive(Debug)]
75pub(super) struct MainAdmission {
76    pub(super) plan: MeasurementSetWritePlan,
77    pub(super) cores: CoreAllocation,
78    /// The policy's allocation, before admission shrank it.
79    pub(super) policy_cores: CoreAllocation,
80    /// Rows of a prediction block: the write batch's, or fewer when the
81    /// budget is small.
82    pub(super) block_rows: usize,
83    pub(super) reservation: Reservation,
84}

86/// Admit `close`'s writer and pipeline for `shape` within `resources`, the
87/// pipeline predicting with `prediction` when the session predicts.
88///
89/// # Errors
90///
91/// [`SimulatorError::admission`] when the writer, or the pipeline with one
92/// worker, does not fit what the policy leaves free; a raised error when the
93/// MAIN write cannot be planned at all.
94pub(super) fn admit_main_write(
95    resources: &SessionResources,
96    shape: &MainShape<'_>,
97    prediction: Option<&PredictionStage<'_>>,
98) -> SimulatorResult<MainAdmission> {
99    let (host, policy) = (resources.host(), resources.policy());
100    let free = free_memory(host, policy);
101    let plan = writer_plan(shape, to_usize(free / 2)).map_err(|error| {
102        let required =
103            minimum_writer_bytes(shape).map_or(u64::MAX, |bytes| (bytes as u64).saturating_mul(2));
104        refusal_or(error, WRITER_PHASE, required, free)
105    })?;
106    let mut reservation = admit(
107        host,
108        policy,
109        &Demand {
110            phase: WRITER_PHASE,
111            memory: plan.maximum_resident_bytes() as u64 + shape.generated_bytes,
112        },
113    )
114    .map_err(SimulatorError::admission)?;
115    let policy_cores = resources
116        .close_cores(prediction.is_some_and(|stage| stage.backend() == BackendChoice::Metal));
117    let free = free_memory(host, policy);
118    let batch_rows = plan.batch_rows();
119    let demand = |cores: CoreAllocation, block_rows: usize| {
120        pipeline_bytes(shape, batch_rows, cores)
121            + prediction.map_or(0, |stage| stage.bytes(block_rows, cores.streaming_workers))
122    };
123    // A batch predicted as one block when that fits with the policy's
124    // workers; else small blocks, and the workers shrink first (#737).
125    let (cores, block_rows) = if demand(policy_cores, batch_rows) <= free {
126        (policy_cores, batch_rows)
127    } else {
128        let small = prediction.map_or(batch_rows, |stage| stage.small_block_rows(batch_rows));
129        let cores = (1..=policy_cores.streaming_workers)
130            .rev()
131            .map(|workers| policy_cores.at_most(workers))
132            .find(|cores| demand(*cores, small) <= free)
133            .unwrap_or_else(|| policy_cores.at_most(1));
134        (cores, small)
135    };
136    let pipeline = admit(
137        host,
138        policy,
139        &Demand {
140            phase: PIPELINE_PHASE,
141            memory: demand(cores, block_rows),
142        },
143    )
144    .map_err(SimulatorError::admission)?;
145    reservation.join(pipeline);
146    Ok(MainAdmission {
147        plan,
148        cores,
149        policy_cores,
150        block_rows,
151        reservation,
152    })
153}

155/// The MAIN write plan with `available_bytes` for the writer.
156pub(super) fn writer_plan(
157    shape: &MainShape<'_>,
158    available_bytes: usize,
159) -> Result<MeasurementSetWritePlan, MeasurementSetWriteError> {
160    MeasurementSetWritePlan::visibility_creation(
161        shape.rows,
162        shape.correlations,
163        shape.channels,
164        shape.telescope,
165        MeasurementSetWriteResources {
166            available_bytes,
167            maximum_live_batches: WRITER_LIVE_BATCHES,
168            tiled_column_buffer_bytes: STREAMING_TILED_COLUMN_BUFFER_BYTES,
169        },
170    )
171}

173/// The fewest writer bytes that plan `shape`'s MAIN write; `None` when no
174/// budget does.
175fn minimum_writer_bytes(shape: &MainShape<'_>) -> Option<usize> {
176    writer_plan(shape, UNBOUNDED_WRITER_BYTES).ok()?;
177    // Whether a budget plans is monotone in the budget.
178    let (mut refused, mut planned) = (0, UNBOUNDED_WRITER_BYTES);
179    while planned - refused > 1 {
180        let middle = refused + (planned - refused) / 2;
181        if writer_plan(shape, middle).is_ok() {
182            planned = middle;
183        } else {
184            refused = middle;
185        }
186    }
187    Some(planned)
188}

190/// A plan refused for its budget as an [`Admission`] of `required` bytes
191/// against `free`, and any other planning failure as raised.
192fn refusal_or(
193    error: MeasurementSetWriteError,
194    phase: &'static str,
195    required: u64,
196    free: u64,
197) -> SimulatorError {
198    if required == u64::MAX {
199        return SimulatorError::raised(format!("plan the MAIN write: {error}"));
200    }
201    SimulatorError::admission(Admission {
202        phase,
203        required,
204        available: free,
205    })
206}

208/// Bytes the pipeline holds beside the writer, with `cores`' streaming
209/// workers: the batch being filled, the geometry of the integrations it
210/// spans (with the unwritten rows of the one before), the list of those
211/// integrations, one integration's scalar runs, and each worker's scratch.
212pub(super) fn pipeline_bytes(
213    shape: &MainShape<'_>,
214    batch_rows: usize,
215    cores: CoreAllocation,
216) -> u64 {
217    let samples = (shape.correlations * shape.channels) as u64;
218    let rows = batch_rows as u64;
219    let largest = shape.largest_integration as u64;
220    let size = |bytes: usize| bytes as u64;
221    // DATA, FLAG_ROW, UVW, WEIGHT and SIGMA.
222    let row_bytes = samples * size(size_of::<Complex32>()) + 1 + 24 + 4 + 4;
223    let batch = rows * row_bytes;
224    let geometry = (rows + 2 * largest) * size(size_of::<RowGeometry>());
225    let integrations = (rows + 2) * size(size_of::<Integration>());
226    let scalars = largest * (4 + 4 + 1);
227    let workers = cores.streaming_workers as u64 * shape.antennas as u64 * ANTENNA_SCRATCH_BYTES;
228    batch + geometry + integrations + scalars + workers
229}
230
231fn to_usize(bytes: u64) -> usize {
232    usize::try_from(bytes).unwrap_or(usize::MAX)
233}

235#[cfg(test)]
236mod tests {
237    use casa_imaging_runtime::{HostResources, ResourcePolicy};
238
239    use super::*;
240    use crate::session::BackendRequest;
241
242    fn shape() -> MainShape<'static> {
243        MainShape {
244            rows: 100_000,
245            correlations: 2,
246            channels: 64,
247            telescope: "ALMA",
248            largest_integration: 1_000,
249            antennas: 44,
250            generated_bytes: 0,
251        }
252    }
253
254    fn resources(memory: u64, workers: usize) -> SessionResources {
255        SessionResources::pinned(
256            HostResources {
257                threads: 8,
258                performance_cores: 8,
259                available_memory: 8 << 30,
260                metal: false,
261            },
262            ResourcePolicy::Explicit { workers, memory },
263            BackendRequest::Cpu,
264        )
265        .expect("resources")
266    }
267
268    #[test]
269    fn the_writer_takes_half_the_budget_and_the_pipeline_fits_the_rest() {
270        let admitted = admit_main_write(&resources(1 << 30, 5), &shape(), None).expect("fits");
271        assert_eq!(admitted.cores.streaming_workers, 4);
272        let writer = admitted.plan.maximum_resident_bytes() as u64;
273        assert!(writer <= 1 << 29);
274        assert_eq!(
275            admitted.reservation.memory(),
276            writer + pipeline_bytes(&shape(), admitted.plan.batch_rows(), admitted.cores)
277        );
278    }
279
280    #[test]
281    fn a_budget_too_small_for_the_writer_is_a_typed_refusal() {
282        let error =
283            admit_main_write(&resources(1 << 20, 4), &shape(), None).expect_err("too small");
284        let refusal = error.admission_refused().expect("an admission refusal");
285        assert_eq!(refusal.phase, WRITER_PHASE);
286        assert_eq!(refusal.available, 1 << 20);
287        let minimum = minimum_writer_bytes(&shape()).expect("plannable") as u64;
288        assert_eq!(refusal.required, 2 * minimum);
289        assert!(writer_plan(&shape(), to_usize(minimum)).is_ok());
290        assert!(writer_plan(&shape(), to_usize(minimum - 1)).is_err());
291    }
292
293    /// Admission state is the process's, so these laws assume no other
294    /// test holds memory at once (nextest runs each test in its own
295    /// process).
296    #[test]
297    fn workers_shrink_before_the_pipeline_is_refused() {
298        // A budget of twice the writer's minimum, and arrays wide enough
299        // that each worker's scratch is a third of what the writer leaves:
300        // two workers fit, four do not.
301        let writer = minimum_writer_bytes(&shape()).expect("plannable");
302        let plan = writer_plan(&shape(), writer).expect("plan");
303        let budget = 2 * writer as u64;
304        let left = budget - plan.maximum_resident_bytes() as u64;
305        let shape = MainShape {
306            antennas: to_usize(left / 3 / ANTENNA_SCRATCH_BYTES),
307            ..shape()
308        };
309        let four = CoreAllocation {
310            workers: 5,
311            streaming_workers: 4,
312        };
313        let two = pipeline_bytes(&shape, plan.batch_rows(), four.at_most(2));
314        assert!(two <= left, "two workers fit: {two} of {left}");
315        assert!(pipeline_bytes(&shape, plan.batch_rows(), four.at_most(3)) > left);
316        let admitted = admit_main_write(&resources(budget, 5), &shape, None).expect("two fit");
317        assert_eq!(admitted.policy_cores, four);
318        assert_eq!(admitted.cores, four.at_most(2));
319        drop(admitted);
320        let error = admit_main_write(
321            &resources(budget, 5),
322            &MainShape {
323                antennas: shape.antennas * 4,
324                ..shape
325            },
326            None,
327        )
328        .expect_err("one worker does not fit");
329        let refusal = error.admission_refused().expect("an admission refusal");
330        assert_eq!(refusal.phase, PIPELINE_PHASE);
331    }
