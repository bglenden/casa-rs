2//! The `table` tools of this process, as casacore's table cache lists the
3//! tables a process has open.
4//!
5//! casacore's tools share one table object per table, so a call on one tool
6//! can see another's lock, refuse to replace a table another tool holds
7//! (`prepareCopyRename`) and release every automatic lock of the process
8//! (`Table::relinquishAutoLocks`). Here each tool has a handle of its own;
9//! this registry lets a call visit the other tools.
10//!
11//! A call holds its own tool's mutex while it visits the others, so the
12//! others are visited with `try_lock`: one that is busy in another thread is
13//! reported as such, never waited for (two tools visiting each other would
14//! deadlock).
15
16use std::collections::HashMap;
17use std::path::{Path, PathBuf};
18use std::sync::atomic::{AtomicU64, Ordering};
19use std::sync::{Arc, Mutex, OnceLock, PoisonError, TryLockError, Weak};
20
21use super::state::{Fail, State};

23/// Yield a named table's automatic locks before the Python caller waits for
24/// its task provider subprocess to write the table.
25#[cfg(unix)]
26pub(crate) fn prepare_provider_write(
27    targets: &[PathBuf],
28) -> Result<Vec<casa_tables::DelegatedTableWrite>, Fail> {
29    let mut result = Ok(());
30    for_each_other(u64::MAX, |state| {
31        if result.is_ok()
32            && let Some(state) = state
33        {
34            result = state.relinquish_auto_lock(Some(targets));
35        }
36    });
37    result?;
38    targets
39        .iter()
40        .map(|path| {
41            casa_tables::DelegatedTableWrite::acquire(path)
42                .map_err(|error| Fail::Runtime(format!("{}: {error}", path.display())))
43        })
44        .collect()
45}

47/// Every tool made in this process that is still alive, with its
48/// `State::id`.
49static TOOLS: Mutex<Vec<(u64, Weak<Mutex<State>>)>> = Mutex::new(Vec::new());
50/// The plain tables reported by tools with a live MeasurementSet attachment.
51static ATTACHMENTS: Mutex<Vec<Weak<Vec<PathBuf>>>> = Mutex::new(Vec::new());
52
53/// The caller retains the returned handle for the lifetime of its attachment.
54pub(crate) fn attach_table_paths(paths: Vec<PathBuf>) -> Arc<Vec<PathBuf>> {
55    let paths = Arc::new(paths);
56    let mut attachments = ATTACHMENTS.lock().unwrap_or_else(PoisonError::into_inner);
57    attachments.retain(|paths| paths.strong_count() > 0);
58    attachments.push(Arc::downgrade(&paths));
59    paths
60}
61
62pub(super) fn attached_table_paths() -> Vec<PathBuf> {
63    let attachments = ATTACHMENTS.lock().unwrap_or_else(PoisonError::into_inner);
64    let mut paths = Vec::new();
65    for attachment in attachments.iter().filter_map(Weak::upgrade) {
66        paths.extend(attachment.iter().cloned());
67    }
68    paths
69}

71/// Per-table notification of metadata writes in this process.
72/// Weak references keep the registry from retaining closed tables.
73pub(super) fn keyword_epoch(path: &Path) -> Arc<AtomicU64> {
74    static EPOCHS: OnceLock<Mutex<HashMap<PathBuf, Weak<AtomicU64>>>> = OnceLock::new();
75    let name = path.canonicalize().unwrap_or_else(|_| path.to_path_buf());
76    let mut epochs = EPOCHS
77        .get_or_init(Mutex::default)
78        .lock()
79        .unwrap_or_else(PoisonError::into_inner);
80    epochs.retain(|_, epoch| epoch.strong_count() > 0);
81    if let Some(epoch) = epochs.get(&name).and_then(Weak::upgrade) {
82        return epoch;
83    }
84    let epoch = Arc::new(AtomicU64::new(0));
85    epochs.insert(name, Arc::downgrade(&epoch));
86    epoch
87}
88
89pub(super) fn keyword_published(path: &Path) {
90    keyword_epoch(path).fetch_add(1, Ordering::Release);
91}

93/// `state` as a tool's shared state, listed with the process's tools.
94pub(super) fn track(state: State) -> Arc<Mutex<State>> {
95    let id = state.id;
96    let tool = Arc::new(Mutex::new(state));
97    let mut tools = TOOLS.lock().unwrap_or_else(PoisonError::into_inner);
98    tools.retain(|(_, other)| other.strong_count() > 0);
99    tools.push((id, Arc::downgrade(&tool)));
100    tool
101}

103/// Calls `visit` on the state of every tool but `me` (a tool's `State::id`,
104/// the caller's, whose mutex the caller holds): with `Some(state)` for a tool
105/// that is free, with `None` for one that is busy in another call.
106pub(super) fn for_each_other(me: u64, mut visit: impl FnMut(Option<&mut State>)) {
107    let tools: Vec<Arc<Mutex<State>>> = TOOLS
108        .lock()
109        .unwrap_or_else(PoisonError::into_inner)
110        .iter()
111        .filter(|(id, _)| *id != me)
112        .filter_map(|(_, tool)| tool.upgrade())
113        .collect();
114    for tool in tools {
115        match tool.try_lock() {
116            Ok(mut state) => visit(Some(&mut state)),
117            Err(TryLockError::Poisoned(poisoned)) => visit(Some(&mut poisoned.into_inner())),
118            Err(TryLockError::WouldBlock) => visit(None),
119        }
120    }
121}
