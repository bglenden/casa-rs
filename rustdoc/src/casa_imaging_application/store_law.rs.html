2//! The store law (#932, #942): the batch geometry ([`GeometryPlan`]) of an
3//! observation's samples in its MeasurementSet equals their batch geometry
4//! in another store of the same observation (an MSv4 conversion), sample
5//! for sample, to [`TOLERANCE`]: what the store alone changes. Every sample
6//! of every partition is compared, on every channel; the request gives the
7//! geometry (phase centre, image domains, pointing), not the samples.
8//!
9//! A MeasurementSet sample is its MAIN row
10//! ([`Msv2Observation::main_row`]), so padding matches nothing. A sample of
11//! the other store is matched through its partition's content, its
12//! antennas by name and its time to the millisecond, a key two of the
13//! MeasurementSet's rows may share; such rows, and rows two samples claim,
14//! are counted as duplicates and fail the law.
15

35/// The law's tolerance: every quantity relative to its scale.
36pub const TOLERANCE: f64 = 1.0e-12;
37
38/// One request's two geometries compared.
39#[derive(Clone, Debug, Default)]
40pub struct GeometryComparison {
41    /// Rows the MeasurementSet's batches delivered.
42    pub rows: usize,
43    /// Of them, rows no sample of the other store matched.
44    pub unmatched: usize,
45    /// Rows two samples of the other store matched, and MeasurementSet
46    /// rows that share another's key.
47    pub duplicates: usize,
48    /// Per quantity, the largest difference relative to its scale and the
49    /// physical row where it occurred.
50    pub differences: BTreeMap<&'static str, (f64, u64)>,
51    /// Per quantity, the values compared.
52    pub compared: BTreeMap<&'static str, usize>,
53}
54
55impl GeometryComparison {
56    /// Whether the law holds: there were rows, every one matched exactly
57    /// once, and no quantity differs by more than [`TOLERANCE`].
58    #[must_use]
59    pub fn holds(&self) -> bool {
60        self.rows > 0 && self.unmatched == 0 && self.duplicates == 0 && self.largest() <= TOLERANCE
61    }
62
63    /// The largest relative difference of any quantity.
64    #[must_use]
65    pub fn largest(&self) -> f64 {
66        self.differences
67            .values()
68            .map(|(difference, _)| *difference)
69            .fold(0.0, worst)
70    }
71
72    /// Record one difference; a NaN one (either side absent or not finite)
73    /// counts as infinite.
74    fn record(&mut self, quantity: &'static str, difference: f64, row: u64) {
75        *self.compared.entry(quantity).or_default() += 1;
76        let difference = worst(difference, 0.0);
77        let entry = self.differences.entry(quantity).or_insert((0.0, row));
78        if difference > entry.0 {
79            *entry = (difference, row);
80        }
81    }
82}
83
84/// The larger of two differences, a NaN infinite: `f64::max` would drop it.
85fn worst(a: f64, b: f64) -> f64 {
86    if a.is_nan() || b.is_nan() {
87        f64::INFINITY
88    } else {
89        a.max(b)
90    }
91}

93/// Rows by MAIN row number.
94type Rows = HashMap<u64, Row>;

96/// A row's identity in another store: data description, antennas, and
97/// time in milliseconds.
98type Key = (u32, u32, u32, i64);
99
100fn key(data_description: u32, antennas: [u32; 2], time_s: f64) -> Key {
101    (
102        data_description,
103        antennas[0],
104        antennas[1],
105        (time_s * 1.0e3).round() as i64,
106    )
107}

124/// Compare the batch geometry of every sample of `request`'s
125/// MeasurementSet with its batch geometry in `converted`, the same
126/// observation in another store (an MSv4 conversion). A converted partition
127/// is matched to the data description of the MeasurementSet's partitions of
128/// the same content
129/// ([`PartitionEntry::same_content`](casa_observation::PartitionEntry::same_content)),
130/// and a converted antenna to the MeasurementSet's of its name.
131///
132/// # Errors
133///
134/// When the request does not compile, or either store cannot be opened or
135/// read.
136pub fn compare_stores(
137    request: &ImagingRequest,
138    context: &RunContext,
139    converted: &Path,
140) -> Result<GeometryComparison, ApplicationError> {
141    let resolved = std::cell::Cell::new(None);
142    let (problem, input, _native, _resident) = compile_run(request, context, &resolved)?;
143    let OpenedObservation {
144        store,
145        measures,
146        phase_table,
147    } = input.observation;
148    let observation = &store.value;
149    let inputs = PlanInputs {
150        measures,
151        phase_table,
152    };
153    let mut keys = HashMap::new();
154    let mut shared = 0;
155    let reference = batch_rows(
156        &problem,
157        observation,
158        &observation.catalogue().partition_ids(),
159        |batch, sample| {
160            let per_time = batch.samples_per_time();
161            let baseline = batch.baselines[sample % per_time];
162            let time_s = batch.time.offsets_s()[sample / per_time];
163            let row = u64::from(observation.main_row(batch.partition, time_s, baseline)?);
164            let identity = key(
165                observation.data_description_id(batch.partition),
166                [baseline.antenna1.0, baseline.antenna2.0],
167                batch
168                    .time
169                    .seconds_since(sample / per_time, MjdHighPrec::new(0.0, 0.0)),
170            );
171            if keys
172                .insert(identity, row)
173                .is_some_and(|earlier| earlier != row)
174            {
175                shared += 1;
176            }
177            Some(row)
178        },
179        inputs.clone(),
180        None,
181    )?;
182    let other = casa_ms::observation_io::open(converted, &OpenOptions::default())?;
183    let bridge = Bridge::new(observation, other.catalogue());
184    let rows = batch_rows(
185        &problem,
186        other.as_ref(),
187        &bridge.partitions(),
188        |batch, sample| bridge.row(batch, sample, &keys),
189        inputs,
190        Some(&reference.rows),
191    )?;
192    let mut comparison = compare(&reference.rows, &rows);
193    comparison.duplicates += reference.duplicates + shared;
194    Ok(comparison)
195}

197/// What compiles a store's geometry beside its catalogue: the run's
198/// Measures provider and the ephemeris table a moving phase centre names.
199#[derive(Clone)]
200struct PlanInputs {
201    measures: Arc<dyn MeasuresProvider>,
202    phase_table: Option<Arc<FieldEphemeris>>,
203}

205/// The rows a store's batch samples matched.
206struct Found {
207    rows: Rows,
208    /// Samples that matched a row an earlier sample had.
209    duplicates: usize,
210}
211
212/// `actual`'s rows against `expected`'s.
213fn compare(expected: &Rows, actual: &Found) -> GeometryComparison {
214    let mut comparison = GeometryComparison {
215        rows: expected.len(),
216        duplicates: actual.duplicates,
217        ..GeometryComparison::default()
218    };
219    for (row, want) in expected {
220        match actual.rows.get(row) {
221            Some(got) => compare_row(want, got, &mut comparison),
222            None => comparison.unmatched += 1,
223        }
224    }
225    comparison
226}

228/// The batch geometry of the rows of `partitions` of `observation`;
229/// `row_of` gives a sample's row, `None` for padding or a sample no row
230/// matches. With `expected`, only its rows, on its channels.
231fn batch_rows(
232    problem: &CompiledProblem,
233    observation: &dyn Observation,
234    partitions: &[PartitionId],
235    mut row_of: impl FnMut(&VisibilityBatch<'_>, usize) -> Option<u64>,
236    inputs: PlanInputs,
237    expected: Option<&Rows>,
238) -> Result<Found, ApplicationError> {
239    let catalogue = Arc::clone(observation.catalogue());
240    let plan = GeometryPlan::compile(problem, &catalogue, inputs.measures, inputs.phase_table)?;
241    let mut traversal = GeometryTraversal::default();
242    let mut geometry = BatchGeometry::default();
243    let mut found = Found {
244        rows: HashMap::new(),
245        duplicates: 0,
246    };
247    let centroids = matches!(
248        problem.geometry().centres().pointing(),
249        PointingCentreLaw::Observation(law)
250            if law.time_sampling() == casa_imaging_model::PointingTimeSampling::VisibilityTimeCentroid
251    );
252    for &partition in partitions {
253        let entry = catalogue.partition(partition);
254        let data_group = entry
255            .data_groups
256            .first()
257            .ok_or_else(|| format!("partition {} has no data group", partition.0))?;
258        let request = AccessRequest {
259            partitions: vec![partition],
260            data_group: data_group.name.clone(),
261            roles: Roles {
262                values: false,
263                weights: false,
264                time_centroid: centroids,
265                ..Roles::PAYLOAD
266            },
267            channels: 0..entry.frequencies_hz.len(),
268            times: TimeSelection::All,
269            fields: FieldSelection::All,
270            weights: WeightRequest::Stored,
271            correlations: CorrelationRequest::ALL,
272            grouping: Grouping::None,
273            batch_bytes: 1 << 26,
274        };
275        let mut reader = observation.bind_read(&request)?;
276        let mut buffers = BatchBuffers::default();
277        while let Some(event) = reader.next(&mut buffers)? {
278            let ReadEvent::Batch(batch) = event else {
279                continue;
280            };
281            if batch.samples() == 0 {
282                continue;
283            }
284            let samples = (0..batch.samples())
285                .map(|sample| {
286                    row_of(&batch, sample)
287                        .filter(|row| expected.is_none_or(|rows| rows.contains_key(row)))
288                })
289                .collect::<Vec<_>>();
290            let rows = samples.iter().map(Option::is_some).collect::<Vec<_>>();
291            let pointing = plan
292                .reads_pointing()
293                .then(|| observation.pointing(&pointing_request(problem, &batch, partition)))
294                .transpose()?;
295            plan.apply(
296                &batch,
297                &rows,
298                pointing.as_ref(),
299                &mut traversal,
300                &mut geometry,
301            )?;
302            for (sample, row) in samples.iter().enumerate() {
303                let Some(row) = row else {
304                    continue;
305                };
306                let got = batch_row(
307                    &batch,
308                    &geometry,
309                    (sample, *row),
310                    plan.charts(),
311                    expected.map(|rows| &rows[row]),
312                );
313                if found.rows.insert(*row, got).is_some() {
314                    found.duplicates += 1;
315                }
316            }
317        }
318    }
319    Ok(found)
320}

322/// How another store's samples find the MeasurementSet's rows.
323struct Bridge {
324    /// Each matched converted partition's data description.
325    data_descriptions: BTreeMap<PartitionId, u32>,
326    /// Each converted antenna's MeasurementSet antenna.
327    antennas: Vec<Option<u32>>,
328}
329
330impl Bridge {
331    fn new(observation: &Msv2Observation, converted: &Catalogue) -> Self {
332        let reference = observation.catalogue();
333        let data_descriptions = converted
334            .partition_ids()
335            .into_iter()
336            .filter_map(|partition| {
337                let entry = converted.partition(partition);
338                let mut matching = reference
339                    .partition_ids()
340                    .into_iter()
341                    .filter(|candidate| reference.partition(*candidate).same_content(entry))
342                    .map(|candidate| observation.data_description_id(candidate));
343                let first = matching.next()?;
344                matching
345                    .all(|other| other == first)
346                    .then_some((partition, first))
347            })
348            .collect();
349        let antennas = converted
350            .antennas
351            .iter()
352            .map(|antenna| {
353                let mut matching = reference
354                    .antennas
355                    .iter()
356                    .enumerate()
357                    .filter(|(_, candidate)| same_antenna(candidate, antenna))
358                    .map(|(index, _)| index as u32);
359                let first = matching.next()?;
360                matching.next().is_none().then_some(first)
361            })
362            .collect();
363        Self {
364            data_descriptions,
365            antennas,
366        }
367    }
368
369    /// The converted partitions matched to a data description.
370    fn partitions(&self) -> Vec<PartitionId> {
371        self.data_descriptions.keys().copied().collect()
372    }
373
374    /// The MeasurementSet row `keys` gives `sample` of `batch`.
375    fn row(
376        &self,
377        batch: &VisibilityBatch<'_>,
378        sample: usize,
379        keys: &HashMap<Key, u64>,
380    ) -> Option<u64> {
381        let data_description = *self.data_descriptions.get(&batch.partition)?;
382        let per_time = batch.samples_per_time();
383        let baseline = batch.baselines[sample % per_time];
384        let antenna = |id: AntennaId| self.antennas.get(id.0 as usize).copied().flatten();
385        let time_s = batch
386            .time
387            .seconds_since(sample / per_time, MjdHighPrec::new(0.0, 0.0));
388        let antennas = [antenna(baseline.antenna1)?, antenna(baseline.antenna2)?];
389        keys.get(&key(data_description, antennas, time_s)).copied()
390    }
391}

393/// Whether `converted` names `reference`: as stored, or as XRADIO names an
394/// MSv4 antenna (`NAME_STATION`).
395fn same_antenna(reference: &AntennaEntry, converted: &AntennaEntry) -> bool {
396    converted.name == reference.name
397        || converted.name == format!("{}_{}", reference.name, reference.station)
398}

400/// The pointing request covering `batch`'s times for its partition's
401/// antennas.
402fn pointing_request(
403    problem: &CompiledProblem,
404    batch: &VisibilityBatch<'_>,
405    partition: PartitionId,
406) -> PointingRequest {
407    let mut antennas: Vec<AntennaId> = batch
408        .baselines
409        .iter()
410        .flat_map(|baseline| [baseline.antenna1, baseline.antenna2])
411        .collect();
412    antennas.sort_unstable();
413    antennas.dedup();
414    let times = batch.time.offsets_s();
415    let mut span = times[0]..=times[times.len() - 1];
416    if let Some(centroid) = batch.timing.centroid {
417        let offsets = centroid.offsets_s();
418        let low = offsets.iter().copied().fold(*span.start(), f64::min);
419        let high = offsets.iter().copied().fold(*span.end(), f64::max);
420        span = low..=high;
421    }
422    let direction = crate::imaging::pointing_direction(problem);
423    PointingRequest {
424        partition,
425        direction,
426        antennas,
427        times: span,
428        budget_bytes: 1 << 34,
429    }
430}

432/// Sample `sample`'s geometry as MAIN row `row`, on every channel the
433/// batch holds or, with `expected`, on its channels.
434fn batch_row(
435    batch: &VisibilityBatch<'_>,
436    geometry: &BatchGeometry,
437    (sample, row): (usize, u64),
438    charts: usize,
439    expected: Option<&Row>,
440) -> Row {
441    let time = sample / batch.samples_per_time();
442    let frequencies = geometry.frequencies_hz(time);
443    Row {
444        physical_row: row,
445        field: batch.field[time].0,
446        raw_uvw_m: batch
447            .group
448            .uvw
449            .map_or([f64::NAN; 3], |uvw| uvw.metres_at(sample)),
450        projections: (0..charts)
451            .map(|chart| {
452                let projection = geometry.projection(sample, chart);
453                (projection.uvw_m, projection.phase_shift_m)
454            })
455            .collect(),
456        aw_pixels: (0..charts)
457            .map(|chart| geometry.aw_pointing_pixel(sample, chart))
458            .collect(),
459        pointing: geometry.pointing(sample),
460        parallactic: geometry.parallactic_angles(sample),
461        responses: geometry.antenna_responses(sample),
462        frequencies: match expected {
463            Some(expected) => expected
464                .frequencies
465                .iter()
466                .map(|(channel, _)| {
467                    let index = *channel as usize - batch.first_channel;
468                    (
469                        *channel,
470                        frequencies.get(index).copied().unwrap_or(f64::NAN),
471                    )
472                })
473                .collect(),
474            None => frequencies
475                .iter()
476                .enumerate()
477                .map(|(index, frequency)| ((batch.first_channel + index) as u32, *frequency))
478                .collect(),
479        },
480    }
481}

483/// Record the differences of one row.
484fn compare_row(expected: &Row, actual: &Row, comparison: &mut GeometryComparison) {
485    let row = expected.physical_row;
486    let length = norm(expected.raw_uvw_m).max(1.0);
487    let exact = |same: bool| if same { 0.0 } else { f64::INFINITY };
488    comparison.record("field", exact(expected.field == actual.field), row);
489    comparison.record(
490        "raw uvw",
491        norm(difference(expected.raw_uvw_m, actual.raw_uvw_m)) / length,
492        row,
493    );
494    comparison.record(
495        "charts",
496        exact(expected.projections.len() == actual.projections.len()),
497        row,
498    );
499    for (want, got) in expected.projections.iter().zip(&actual.projections) {
500        comparison.record("uvw", norm(difference(want.0, got.0)) / length, row);
501        comparison.record("phase shift", (want.1 - got.1).abs() / length, row);
502    }
503    for (want, got) in expected.aw_pixels.iter().zip(&actual.aw_pixels) {
504        let difference = match (want, got) {
505            (None, None) => continue,
506            (Some(want), Some(got)) => {
507                ((want[0] - got[0]).hypot(want[1] - got[1])) / want[0].hypot(want[1]).max(1.0)
508            }
509            _ => f64::INFINITY,
510        };
511        comparison.record("AW pointing pixel", difference, row);
512    }
513    let angle = |want: SkyDirection, got: SkyDirection| {
514        worst(
515            exact(want.frame() == got.frame()),
516            worst(
517                (want.longitude_rad() - got.longitude_rad()).abs(),
518                (want.latitude_rad() - got.latitude_rad()).abs(),
519            ),
520        )
521    };
522    comparison.record(
523        "pointing",
524        worst(
525            angle(expected.pointing.antenna1, actual.pointing.antenna1),
526            angle(expected.pointing.antenna2, actual.pointing.antenna2),
527        ),
528        row,
529    );
530    match (expected.parallactic, actual.parallactic) {
531        (None, None) => {}
532        (Some(want), Some(got)) => comparison.record(
533            "parallactic angle",
534            worst((want[0] - got[0]).abs(), (want[1] - got[1]).abs()),
535            row,
536        ),
537        _ => comparison.record("parallactic angle", f64::INFINITY, row),
538    }
539    if expected.responses.is_some() || actual.responses.is_some() {
540        comparison.record(
541            "aperture classes",
542            exact(expected.responses == actual.responses),
543            row,
544        );
545    }
546    for ((_, want), (_, got)) in expected.frequencies.iter().zip(&actual.frequencies) {
547        comparison.record("frequency", (want - got).abs() / want.abs(), row);
548    }
549}

551fn difference(a: [f64; 3], b: [f64; 3]) -> [f64; 3] {
552    [a[0] - b[0], a[1] - b[1], a[2] - b[2]]
553}
554
555fn norm(vector: [f64; 3]) -> f64 {
556    vector.iter().map(|value| value * value).sum::<f64>().sqrt()
