16pub(super) enum Pixels {
17    Real(ArrayD<f64>),
18    Complex(ArrayD<Complex64>),
19}
20
21impl Pixels {
22    pub(super) fn capture(
23        py: Python<'_>,
24        value: Option<&Bound<'_, PyAny>>,
25        creating: bool,
26    ) -> PyResult<Self> {
27        let format_error = if creating {
28            "pixels is not understood, try using an array"
29        } else {
30            "Unsupported type for pixels parameter. It must be either a vector of doubles or ints"
31        };
32        if let Some(value) = value
33            && let Ok(array) = value.cast::<PyUntypedArray>()
34        {
35            let dtype = array.dtype();
36            let pixels = match (dtype.kind(), dtype.itemsize()) {
37                (b'i' | b'u', _) | (b'f', 4 | 8) => Self::Real(snapshot_array(py, array)?),
38                (b'c', 8 | 16) => Self::Complex(snapshot_array(py, array)?),
39                _ => return Err(raised(format_error)),
40            };
41            return pixels.check_shape("The pixels array cannot be empty");
42        }
43        let variant = value.map(|v| v.extract::<Variant>()).transpose()?;
44        Self::from_variant(variant, format_error)
45    }
46
47    fn from_variant(value: Option<Variant>, format_error: &str) -> PyResult<Self> {
48        let pixels = match value.unwrap_or(Variant::Empty) {
49            Variant::Numbers(values) => Self::Real(ndarray::Array1::from_vec(values).into_dyn()),
50            Variant::Complexes(values) => {
51                Self::Complex(ndarray::Array1::from_vec(values).into_dyn())
52            }
53            Variant::Array { shape, values } => {
54                Self::Real(ArrayD::from_shape_vec(IxDyn(&shape).f(), values).map_err(raised)?)
55            }
56            Variant::ComplexArray { shape, values } => {
57                Self::Complex(ArrayD::from_shape_vec(IxDyn(&shape).f(), values).map_err(raised)?)
58            }
59            Variant::Empty => return Err(raised("Exception: The pixels array cannot be empty")),
60            _ => return Err(raised(format_error)),
61        };
62        pixels.check_shape("The pixels array cannot be empty")
63    }
64
65    fn check_shape(self, empty: &str) -> PyResult<Self> {
66        if self.shape().is_empty() || self.shape().contains(&0) {
67            return Err(raised(format!("Exception: {empty}")));
68        }
69        Ok(self)
70    }
71
72    pub(super) fn shape(&self) -> &[usize] {
73        match self {
74            Self::Real(data) => data.shape(),
75            Self::Complex(data) => data.shape(),
76        }
77    }
78
79    pub(super) fn is_complex(&self) -> bool {
80        matches!(self, Self::Complex(_))
81    }
82
83    pub(super) fn f32(&self) -> ArrayD<f32> {
84        match self {
85            Self::Real(data) => data.mapv(|v| v as f32),
86            Self::Complex(_) => unreachable!(),
87        }
88    }
89
90    pub(super) fn f64(&self) -> ArrayD<f64> {
91        match self {
92            Self::Real(data) => data.clone(),
93            Self::Complex(_) => unreachable!(),
94        }
95    }
96
97    pub(super) fn c32(&self) -> ArrayD<Complex32> {
98        match self {
99            Self::Real(data) => data.mapv(|v| Complex32::new(v as f32, 0.0)),
100            Self::Complex(data) => data.mapv(|v| Complex32::new(v.re as f32, v.im as f32)),
101        }
102    }
103
104    pub(super) fn c64(&self) -> ArrayD<Complex64> {
105        match self {
106            Self::Real(data) => data.mapv(|v| Complex64::new(v, 0.0)),
107            Self::Complex(data) => data.clone(),
108        }
109    }
110
111    pub(super) fn history(&self) -> String {
112        if self.shape().iter().product::<usize>() > 100 {
113            return "\"[...]\"".into();
114        }
115        let values: Vec<String> = match self {
116            Self::Real(data) => data
117                .view()
118                .reversed_axes()
119                .iter()
120                .map(|v| format!("{v:.6}"))
121                .collect(),
122            Self::Complex(data) => data
123                .view()
124                .reversed_axes()
125                .iter()
126                .map(|v| format!("({:.6},{:.6})", v.re, v.im))
127                .collect(),
128        };
129        format!("[{}]", values.join(","))
130    }
131}

133// NumPy copies into a private contiguous buffer. No caller can mutate that
134// buffer while Rust reads it with the interpreter detached.
135pub(super) fn snapshot_array<T: Element + Copy + Send + Sync>(
136    py: Python<'_>,
137    source: &Bound<'_, PyUntypedArray>,
138) -> PyResult<ArrayD<T>> {
139    let shape = source.shape().to_vec();
140    let snapshot = PyArrayDyn::<T>::zeros(py, IxDyn(&shape), true);
141    // SAFETY: both pointers name live NumPy arrays with identical shapes and
142    // numeric dtypes. The destination is freshly allocated and cannot alias
143    // either the input or an object another Python thread can access.
144    let status = unsafe {
145        PY_ARRAY_API.PyArray_CopyInto(py, snapshot.as_array_ptr(), source.as_array_ptr())
146    };
147    if status != 0 {
148        return Err(PyErr::fetch(py));
149    }
150    let snapshot = snapshot.readonly();
151    let values = snapshot.as_slice().map_err(raised)?;
152    py.detach(|| ArrayD::from_shape_vec(IxDyn(&shape).f(), values.to_vec()).map_err(raised))
153}

155pub(super) struct Chunk {
156    pub(super) start: Vec<usize>,
157    pub(super) size: Vec<usize>,
158    pub(super) stride: Vec<usize>,
159}
160
161impl Chunk {
162    fn read(shape: &[usize], blc: &[i64], trc: &[i64], inc: &[i64]) -> PyResult<Self> {
163        let mut chunk = Self {
164            start: Vec::new(),
165            size: Vec::new(),
166            stride: Vec::new(),
167        };
168        for (axis, &length) in shape.iter().enumerate() {
169            let valid = |v| usize::try_from(v).ok().filter(|&i| i < length);
170            let first = blc.get(axis).and_then(|&v| valid(v)).unwrap_or(0);
171            let last = trc.get(axis).and_then(|&v| valid(v)).unwrap_or(length - 1);
172            let (first, last) = if first <= last {
173                (first, last)
174            } else {
175                (0, length - 1)
176            };
177            let step = inc.get(axis).copied().unwrap_or(1);
178            if step < 1 {
179                return Err(raised("Exception: stride must be positive"));
180            }
181            let step = if step as usize > length {
182                1
183            } else {
184                step as usize
185            };
186            chunk.start.push(first);
187            chunk.size.push((last - first) / step + 1);
188            chunk.stride.push(step);
189        }
190        Ok(chunk)
191    }
192
193    fn write(
194        shape: &[usize],
195        data: &[usize],
196        blc: &[i64],
197        inc: &[i64],
198        replicate: bool,
199    ) -> PyResult<Self> {
200        if data.len() > shape.len() {
201            return Err(raised(
202                "Exception: Pixel array cannot have more dimensions than the image!",
203            ));
204        }
205        let mut chunk = Self {
206            start: Vec::new(),
207            size: Vec::new(),
208            stride: Vec::new(),
209        };
210        for (axis, &length) in shape.iter().enumerate() {
211            let start = blc
212                .get(axis)
213                .and_then(|&v| usize::try_from(v).ok())
214                .filter(|&v| v < length)
215                .unwrap_or(0);
216            let stride = inc.get(axis).copied().unwrap_or(1).max(1) as usize;
217            let count = data.get(axis).copied().unwrap_or(1);
218            if start >= length || (length - 1 - start) / stride < count - 1 {
219                return Err(raised(
220                    "Exception: Pixels array, including inc, extends beyond edge of image.",
221                ));
222            }
223            let extent = if replicate {
224                (length - 1 - start) / stride + 1
225            } else {
226                count
227            };
228            if replicate && extent % count != 0 {
229                return Err(raised(
230                    "Exception: region shape is not an integer multiple of the input array shape",
231                ));
232            }
233            chunk.start.push(start);
234            chunk.size.push(extent);
235            chunk.stride.push(stride);
236        }
237        Ok(chunk)
238    }

240    pub(super) fn mask<T: ImagePixel>(
241        &self,
242        image: &impl ImageInterface<T>,
243    ) -> PyResult<Option<ArrayD<bool>>> {
244        image
245            .get_mask_slice(&self.start, &self.size, &self.stride)
246            .map_err(raised)
247    }
248
249    fn put<T: ImagePixel>(&self, storage: &mut Storage<T>, mut data: ArrayD<T>) -> PyResult<()> {
250        if let Storage::Temp { image, .. } = storage {
251            image
252                .require_writable_pixels()
253                .map_err(casa_images::ImageError::from)
254                .map_err(raised)?;
255        }
256        while data.ndim() < self.start.len() {
257            let axis = ndarray::Axis(data.ndim());
258            data = data.insert_axis(axis);
259        }
260        if data.shape() != self.size {
261            data = ArrayD::from_shape_fn(IxDyn(&self.size).f(), |index| {
262                let source: Vec<_> = index
263                    .slice()
264                    .iter()
265                    .zip(data.shape())
266                    .map(|(&i, &n)| i % n)
267                    .collect();
268                data[IxDyn(&source)]
269            });
270        }
271        if self.stride.iter().all(|&step| step == 1) {
272            with_storage!(storage, image => image.put_slice(&data, &self.start)).map_err(raised)?;
273        } else {
274            let mut pos = self.start.clone();
275            for (index, &value) in data.indexed_iter() {
276                for (axis, &i) in index.slice().iter().enumerate() {
277                    pos[axis] = self.start[axis] + i * self.stride[axis];
278                }
279                with_storage!(storage, image => image.put_at(value, &pos)).map_err(raised)?;
280            }
281        }
282        Ok(())
283    }

285    fn get<T: ImagePixel, U: Average>(
286        &self,
287        image: &impl ImageInterface<T>,
288        request: &ChunkRead<'_>,
289        expression: Option<ArrayD<bool>>,
290        wide: (impl Fn(T) -> U, impl Fn(U) -> U),
291    ) -> PyResult<CasaValue> {
292        let mask = if request.getmask || !request.axes.is_empty() {
293            and_mask(self.mask(image)?, expression)
294        } else {
295            None
296        };
297        self.format(
298            request.axes,
299            request.dropdeg,
300            request.getmask,
301            wide,
302            mask,
303            || {
304                image
305                    .get_slice(&self.start, &self.size, &self.stride)
306                    .map_err(raised)
307            },
308        )
309    }

311    fn format<T: ImagePixel, U: Average>(
312        &self,
313        axes: &[usize],
314        dropdeg: bool,
315        getmask: bool,
316        wide: (impl Fn(T) -> U, impl Fn(U) -> U),
317        mask: Option<ArrayD<bool>>,
318        read: impl FnOnce() -> PyResult<ArrayD<T>>,
319    ) -> PyResult<CasaValue> {
320        let (widen, round) = wide;
321        let mut shape = self.size.clone();
322        for &axis in axes {
323            shape[axis] = 1;
324        }
325        let output_shape = if dropdeg {
326            let axes: Vec<_> = shape.iter().copied().filter(|&n| n != 1).collect();
327            if axes.is_empty() { vec![1] } else { axes }
328        } else {
329            shape.clone()
330        };
331        let collapse_index = |index: &[usize], output: &mut [usize]| {
332            output.copy_from_slice(index);
333            for &axis in axes {
334                output[axis] = 0;
335            }
336        };
337        if !axes.is_empty() && mask.as_ref().is_some_and(|mask| !mask.iter().any(|&v| v)) {
338            return Err(raised("All selected pixels are masked."));
339        }
340        if getmask {
341            let values = match mask {
342                None => vec![true; shape.iter().product()],
343                Some(mask) if axes.is_empty() => fortran_values(mask),
344                Some(mask) => {
345                    let mut valid = ArrayD::from_elem(IxDyn(&shape).f(), false);
346                    let mut output = vec![0; shape.len()];
347                    for (index, &pixel_valid) in mask.indexed_iter() {
348                        if pixel_valid {
349                            collapse_index(index.slice(), &mut output);
350                            valid[IxDyn(&output)] = true;
351                        }
352                    }
353                    fortran_values(valid)
354                }
355            };
356            return Ok(CasaValue::Bools {
357                shape: output_shape,
358                values,
359            });
360        }
361        let pixels = read()?;
362        if axes.is_empty() {
363            return Ok(U::output(
364                output_shape,
365                pixels.reversed_axes().iter().copied().map(&widen).collect(),
366            ));
367        }
368
369        let mut sums = ArrayD::from_elem(IxDyn(&shape).f(), U::default());
370        let mut counts = ArrayD::from_elem(IxDyn(&shape).f(), 0_usize);
371        let mut out_index = vec![0; shape.len()];
372        for (index, &pixel) in pixels.indexed_iter() {
373            if mask.as_ref().is_some_and(|mask| !mask[&index]) {
374                continue;
375            }
376            collapse_index(index.slice(), &mut out_index);
377            counts[IxDyn(&out_index)] += 1;
378            sums[IxDyn(&out_index)] += widen(pixel);
379        }
380        let values = fortran_values(sums)
381            .into_iter()
382            .zip(fortran_values(counts))
383            .map(|(sum, n)| {
384                if n == 0 {
385                    U::default()
386                } else {
387                    round(sum / (n as f64))
388                }
389            })
390            .collect();
391        Ok(U::output(output_shape, values))
392    }
393}

395pub(super) fn and_mask(
396    original: Option<ArrayD<bool>>,
397    expression: Option<ArrayD<bool>>,
398) -> Option<ArrayD<bool>> {
399    match (original, expression) {
400        (Some(mut original), Some(expression)) => {
401            Zip::from(&mut original)
402                .and(&expression)
403                .for_each(|valid, &passes| *valid &= passes);
404            Some(original)
405        }
406        (original, expression) => original.or(expression),
407    }
408}

416pub(super) fn collapse_axes(axes: &[i64], rank: usize) -> PyResult<Vec<usize>> {
417    if matches!(axes, [only] if *only < 0) {
418        return Ok(Vec::new());
419    }
420    let mut collapse = Vec::with_capacity(axes.len());
421    for &axis in axes {
422        if axis < 0 || axis as usize >= rank || collapse.contains(&(axis as usize)) {
423            return Err(raised("IPosition::makeAxisPath: invalid defined axes"));
424        }
425        collapse.push(axis as usize);
426    }
427    Ok(collapse)
428}

430impl Chunk {
431    pub(super) fn read_from(
432        &self,
433        image: &ImageHandle,
434        request: &ChunkRead<'_>,
435        expression: Option<ArrayD<bool>>,
436    ) -> PyResult<CasaValue> {
437        if let Some(fits) = image.fits() {
438            let pixels = fits.read(&self.start, &self.size, &self.stride)?;
439            let mask = (request.getmask || !request.axes.is_empty())
440                .then(|| pixels.mapv(|value| !value.is_nan()));
441            return self.format(
442                request.axes,
443                request.dropdeg,
444                request.getmask,
445                (f64::from, |v| f64::from(v as f32)),
446                and_mask(mask, expression),
447                || Ok(pixels),
448            );
449        }
450        match image {
451            ImageHandle::Float(storage) => {
452                with_storage!(storage, image => self.get(image.as_ref(), request, expression, (f64::from, |v| f64::from(v as f32))))
453            }
454            ImageHandle::Double(storage) => {
455                with_storage!(storage, image => self.get(image.as_ref(), request, expression, (f64::from, std::convert::identity)))
456            }
457            ImageHandle::Complex(storage) => {
458                with_storage!(storage, image => self.get(image.as_ref(), request, expression, (widen_complex, |v: Complex64| widen_complex(Complex32::new(v.re as f32, v.im as f32)))))
459            }
460            ImageHandle::DComplex(storage) => {
461                with_storage!(storage, image => self.get(image.as_ref(), request, expression, (std::convert::identity, std::convert::identity)))
462            }
463        }
464    }
465}

467fn widen_complex(value: Complex32) -> Complex64 {
468    Complex64::new(f64::from(value.re), f64::from(value.im))

471fn fortran_values<T: Copy>(data: ArrayD<T>) -> Vec<T> {
472    data.reversed_axes().iter().copied().collect()

475trait Average: Copy + Default + std::ops::AddAssign + std::ops::Div<f64, Output = Self> {
476    fn output(shape: Vec<usize>, values: Vec<Self>) -> CasaValue;
477}
478impl Average for f64 {
479    fn output(shape: Vec<usize>, values: Vec<Self>) -> CasaValue {
480        CasaValue::Floats { shape, values }
481    }

484    fn output(shape: Vec<usize>, values: Vec<Self>) -> CasaValue {
485        CasaValue::Complexes { shape, values }
486    }

489#[pymethods]
490#[allow(clippy::too_many_arguments)]
491impl PyImageTool {
492    #[pyo3(signature = (blc=None, trc=None, inc=None, axes=None, list=false, dropdeg=false, getmask=false), text_signature = "($self, blc=[-1], trc=[-1], inc=[1], axes=[-1], list=False, dropdeg=False, getmask=False)")]
493    fn getchunk(
494        &self,
495        py: Python<'_>,
496        blc: Option<IntList>,
497        trc: Option<IntList>,
498        inc: Option<IntList>,
499        axes: Option<IntList>,
500        list: bool,
501        dropdeg: bool,
502        getmask: bool,
503    ) -> PyResult<Py<PyAny>> {
504        let _ = list;
505        let value = py.detach(|| {
506            let state = self.state();
507            let Some(attachment) = state.image.as_ref() else {
508                return Ok(CasaValue::Bools { shape: vec![0], values: Vec::new() });
509            };
510            let mut image = attachment.lock();
511            attachment.require_pixel_read_lock(&mut image)?;
512            let shape = image.shape();
513            if image.coordinates().n_pixel_axes() != shape.len() {
514                return Err(raised("LCSlicer::makeComplete - referencePixel and newLatticeShape vectors do not have same length"));
515            }
516            let chunk = Chunk::read(shape, &blc.map_or_else(Vec::new, |arg| arg.0), &trc.map_or_else(Vec::new, |arg| arg.0), &inc.map_or_else(Vec::new, |arg| arg.0))?;
517            let collapse = collapse_axes(axes.as_ref().map_or(&[], |axes| axes.0.as_slice()), shape.len())?;
518            let value = chunk.read_from(&image, &ChunkRead { axes: &collapse, dropdeg, getmask }, None)?;
519            attachment.release_requested_lock(&mut image)?;
520            Ok(value)
521        })?;
522        value.into_python(py)
523    }
524
525    #[pyo3(signature = (pixels=None, blc=None, inc=None, list=false, locking=true, replicate=false), text_signature = "($self, pixels=[], blc=[-1], inc=[1], list=False, locking=True, replicate=False)")]
526    fn putchunk(
527        &self,
528        py: Python<'_>,
529        pixels: Option<&Bound<'_, PyAny>>,
530        blc: Option<IntList>,
531        inc: Option<IntList>,
532        list: bool,
533        locking: bool,
534        replicate: bool,
535    ) -> PyResult<bool> {
536        let pixels = Pixels::capture(py, pixels, false);
537        py.detach(|| {
538            let state = self.state();
539            let history = state.history;
540            let Some(attachment) = state.image.as_ref() else { return Ok(false) };
541            let mut image = attachment.lock();
542            let pixels = pixels?;
543            let blc = blc.unwrap_or_else(|| IntList::of(&[-1])).0;
544            let inc = inc.unwrap_or_else(|| IntList::of(&[1])).0;
545            let chunk = Chunk::write(image.shape(), pixels.shape(), &blc, &inc, replicate)?;
546            if pixels.is_complex() && matches!(&*image, ImageHandle::Float(_) | ImageHandle::Double(_)) {
547                return Err(raised("Exception: Unsupported type for pixels parameter. It must be either a vector of doubles or ints"));
548            }
549            if image.fits().is_some() {
550                return Err(raised("FITSImage::putSlice - is not possible as FITSImage is not writable"));
551            }
552            attachment.require_lock(&mut image, casa_tables::LockType::Write)?;
553            match &mut *image {
554                ImageHandle::Float(storage) => chunk.put(storage, pixels.f32())?,
555                ImageHandle::Double(storage) => chunk.put(storage, pixels.f64())?,
556                ImageHandle::Complex(storage) => chunk.put(storage, pixels.c32())?,
557                ImageHandle::DComplex(storage) => chunk.put(storage, pixels.c64())?,
558            }
559            if locking && cfg!(unix) {
560                image.unlock_storage()?;
561            } else {
562                image.save().map_err(raised)?;
563                attachment.release_requested_lock(&mut image)?;
564            }
565            if history {
566                let command = format!(
567                    "ia.putchunk(pixels={}, blc={}, inc={}, list={list}, locking={locking}, replicate={replicate})",
568                    pixels.history(), format!("{blc:?}").replace(' ', ""), format!("{inc:?}").replace(' ', "")
569                );
570                image.record_history("putchunk", command)?;
571            }
572            Ok(true)
573        })
574    }
