33/// Build a FITS header for an attached image without reading its pixels.
34///
35/// The shape is supplied separately so that virtual and native attachments
36/// use the same header path.
37///
38/// # Errors
39/// Returns an error for invalid WCS, metadata or an unreadable clock.
40pub fn image_fits_header(
41    image: &dyn ImageInterface<f32>,
42    shape: &[usize],
43    name: &str,
44    history: &[ImageHistoryEntry],
45) -> Result<ImageFitsHeader, ImageError> {
46    image_fits_header_with_beams(image, shape, name, history, None)
47}

49/// Build an attached image's FITS header using its saved restoring beam view.
50///
51/// An open tool can retain beam metadata independently of shared native pixels.
52pub fn image_fits_header_with_beams(
53    image: &dyn ImageInterface<f32>,
54    shape: &[usize],
55    name: &str,
56    history: &[ImageHistoryEntry],
57    beam_info: Option<&RecordValue>,
58) -> Result<ImageFitsHeader, ImageError> {
59    let coords = image.coordinates();
60    let plan = ExportAxisPlan::build(coords, shape, false, false, false, false, false)?;
61    let standard_wcs = || {
62        let base = try_to_fits_header(coords, shape)?;
63        let mut wcs = build_export_keywords(coords, &base, &plan, true, true, None)?
64            .into_iter()
65            .collect();
66        normalize_wcs(coords, &mut wcs)?;
67        Ok::<_, ImageError>(wcs)
68    };
69    let (mut wcs, projection) = match standard_wcs() {
70        Ok(wcs) => (wcs, Some(projection_parameters(coords, Some))),
71        Err(_) => (linear_fallback(coords, shape.len())?, None),
72    };
73    let mut header = ImageFitsHeader(Vec::new());
74    header.add("SIMPLE", FitsValue::Logical(true), Some("Standard FITS"));
75    header.add(
76        "BITPIX",
77        FitsValue::Integer(-32),
78        Some("Floating point (32 bit)"),
79    );
80    header.integer("NAXIS", shape.len() as i64);
81    for (index, length) in shape.iter().enumerate() {
82        header.integer(format!("NAXIS{}", index + 1), *length as i64);
83    }
84    header.add("EXTEND", FitsValue::Logical(true), None);
85    header.add(
86        "BSCALE",
87        FitsValue::Float(1.0),
88        Some("PHYSICAL = PIXEL*BSCALE + BZERO"),
89    );
90    header.add("BZERO", FitsValue::Float(0.0), None);
91    let mut info = image.image_info()?;
92    if let Some(record) = beam_info {
93        info.beam_set = ImageInfo::from_record(record)?.beam_set;
94    }
95    header.image_metadata(image.units(), &info, beam_info)?;
96    let standard = projection.is_some();
97    header.axes(shape.len(), &mut wcs, projection);
98    for (name, comment) in [
99        ("RESTFRQ", "Rest Frequency (Hz)"),
100        ("SPECSYS", "Spectral reference frame"),
101        ("ALTRVAL", "Alternate frequency reference value"),
102        ("ALTRPIX", "Alternate frequency reference pixel"),
103        ("VELREF", "1 LSR, 2 HEL, 3 OBS, +256 Radio"),
104    ] {
105        if let Some(value) = wcs.remove(name) {
106            header.add(name, value, Some(comment));
107            if name == "VELREF" {
108                header.text(
109                    "COMMENT",
110                    "casacore non-standard usage: 4 LSD, 5 GEO, 6 SOU, 7 GAL",
111                );
112            }
113        }
114    }
115    if standard {
116        if image
117            .coordinates()
118            .obs_info()
119            .date
120            .as_ref()
121            .is_none_or(|epoch| epoch.value().as_mjd() == 0.0)
122        {
123            wcs.remove("DATE-OBS");
124            wcs.insert("TIMESYS".into(), FitsValue::String("UTC".into()));
125        }
126        header.observation(image, &mut wcs)?;
127    } else {
128        header.text("TIMESYS", "UTC");
129    }
130    header.extend_misc(image.misc_info())?;
131    header.add(
132        "DATE",
133        FitsValue::String(format_time(
134            ImageHistoryEntry::current_time()? / 86_400.0,
135            FITS,
136            12,
137        )),
138        Some("Date FITS file was written"),
139    );
140    header.text("ORIGIN", "casacore-3.8.0");
141    if info.beam_set.is_multi() {
142        header.add(
143            "CASAMBM",
144            FitsValue::Logical(true),
145            Some("CASA multiple BEAMS table present"),
146        );
147    }
148    if !history.is_empty() {
149        visit_history_cards(history, false, |text| {
150            header.text("HISTORY", text);
151            Ok(())
152        })?;
153    }
154    header.text("END", "");
155    let start = name.len().saturating_sub(68);
156    let suffix = name.get(start..).unwrap_or_else(|| {
157        let boundary = name.char_indices().find(|(i, _)| *i >= start);
158        &name[boundary.map_or(name.len(), |(i, _)| i)..]
159    });
160    header.text("IMAGENME", suffix);
161    Ok(header)
162}

164pub(super) fn image_export_cards<T: ImagePixel>(
165    image: &dyn ImageInterface<T>,
166    info: &ImageInfo,
167    wcs: &[(String, FitsValue)],
168) -> Result<Vec<String>, ImageError> {
169    let mut header = ImageFitsHeader(Vec::new());
170    header.image_metadata(image.units(), info, None)?;
171    header.0.retain(|card| match card.keyword.name.as_str() {
172        "BMAJ" | "BMIN" | "BPA" => false,
173        "BUNIT" => image.units().is_empty(),
174        "OBJECT" => info.object_name.is_empty(),
175        _ => true,
176    });
177    header.observation(image, &mut BTreeMap::new())?;
178    let mut misc = ImageFitsHeader(Vec::new());
179    misc.extend_misc(image.misc_info())?;
180    let keep_misc = |name: &str| {
181        !matches!(
182            name,
183            "SIMPLE"
184                | "BITPIX"
185                | "NAXIS"
186                | "EXTEND"
187                | "EXTNAME"
188                | "END"
189                | "BSCALE"
190                | "BZERO"
191                | "BLANK"
192                | "DATAMIN"
193                | "DATAMAX"
194                | "BMAJ"
195                | "BMIN"
196                | "BPA"
197                | "CASAMBM"
198                | "BTYPE"
199                | "BUNIT"
200                | "OBJECT"
201                | "DATE"
202                | "ORIGIN"
203        ) && !(name.starts_with("NAXIS") && name[5..].bytes().all(|byte| byte.is_ascii_digit()))
204            && !wcs.iter().any(|(key, _)| key == name)
205    };
206    header.0.extend(
207        misc.0
208            .into_iter()
209            .filter(|card| keep_misc(&card.keyword.name)),
210    );
211    header.text(
212        "DATE",
213        format_time(ImageHistoryEntry::current_time()? / 86_400.0, FITS, 12),
214    );
215    header.text("ORIGIN", "casa-rs");
216    Ok(header
217        .0
218        .iter()
219        .map(|card| render_card(&card.keyword))
220        .collect())
221}

223impl ImageFitsHeader {
224    fn add(&mut self, name: impl Into<String>, value: FitsValue, comment: Option<&str>) {
225        let name = name.into();
226        let group = group_name(&name);
227        self.0.push(HeaderCard {
228            keyword: FitsKeyword {
229                name,
230                value,
231                comment: comment.map(str::to_owned),
232            },
233            group,
234        });
235    }
236
237    fn text(&mut self, name: impl Into<String>, text: impl Into<String>) {
238        self.add(name, FitsValue::String(text.into()), None);
239    }
240
241    fn integer(&mut self, name: impl Into<String>, number: i64) {
242        self.add(name, FitsValue::Integer(number), None);
243    }
244
245    fn image_metadata(
246        &mut self,
247        units: &str,
248        info: &ImageInfo,
249        beam_info: Option<&RecordValue>,
250    ) -> Result<(), ImageError> {
251        if let Some(beam) = info.beam_set.single_beam().filter(|beam| !beam.is_null()) {
252            self.add("BMAJ", FitsValue::Float(beam.major_in("deg")?), None);
253            self.add("BMIN", FitsValue::Float(beam.minor_in("deg")?), None);
254            let angle = match beam_info.and_then(ImageInfo::first_beam_record) {
255                Some(stored) => {
256                    stored_position_angle_degrees(&StoredGaussianBeam::from_record(stored)?)?
257                }
258                _ => wrapped_beam_position_angle(beam.position_angle_in("deg")?),
259            };
260            self.add("BPA", FitsValue::Float(angle), None);
261        }
262        let image_type = if info.image_type == ImageType::Undefined {
263            ImageType::Intensity
264        } else {
265            info.image_type
266        };
267        self.text("BTYPE", image_type.to_string());
268        self.text("OBJECT", &info.object_name);
269        self.add(
270            "BUNIT",
271            FitsValue::String(units.to_owned()),
272            Some("Brightness (pixel) unit"),
273        );
274        Ok(())
275    }
276
277    fn axes(
278        &mut self,
279        rank: usize,
280        wcs: &mut BTreeMap<String, FitsValue>,
281        projection: Option<Vec<(String, FitsValue)>>,
282    ) {
283        for keyword in ["EQUINOX", "RADESYS", "LONPOLE", "LATPOLE"] {
284            if let Some(value) = wcs.remove(keyword) {
285                self.add(keyword, value, None);
286            }
287        }
288        if projection.is_some() {
289            for col in 1..=rank {
290                for row in 1..=rank {
291                    let name = format!("PC{row}_{col}");
292                    let value = wcs
293                        .remove(&name)
294                        .unwrap_or(FitsValue::Float(f64::from(row == col)));
295                    self.add(name, value, None);
296                }
297            }
298        }
299        let projection = projection.unwrap_or_default();
300        let interleave_projection = projection.len() == rank;
301        let mut projection = projection.into_iter();
302        for axis in 1..=rank {
303            for key in ["CTYPE", "CRVAL", "CDELT", "CRPIX", "CUNIT"] {
304                let name = format!("{key}{axis}");
305                if let Some(value) = wcs.remove(&name) {
306                    self.add(name, value, None);
307                } else if key == "CUNIT" {
308                    self.text(name, "");
309                }
310            }
311            if interleave_projection && let Some((name, value)) = projection.next() {
312                self.add(name, value, None);
313            }
314        }
315        for (name, value) in projection {
316            self.add(name, value, None);
317        }
318    }
319
320    fn observation<T: ImagePixel>(
321        &mut self,
322        image: &dyn ImageInterface<T>,
323        wcs: &mut BTreeMap<String, FitsValue>,
324    ) -> Result<(), ImageError> {
325        for keyword in ["TELESCOP", "OBSERVER", "DATE-OBS", "TIMESYS"] {
326            if let Some(value) = wcs.remove(keyword) {
327                self.add(keyword, value, None);
328            }
329        }
330        let obs = image.coordinates().obs_info();
331        if let Some(position) = &obs.telescope_position {
332            let xyz = position
333                .convert_to(PositionRef::ITRF)
334                .map_err(|error| ImageError::InvalidMetadata(error.to_string()))?
335                .values();
336            for (key, value) in ["OBSGEO-X", "OBSGEO-Y", "OBSGEO-Z"].into_iter().zip(xyz) {
337                self.add(key, FitsValue::Float(value), None);
338            }
339        }
340        if !obs.pointing_center_initial {
341            for (key, angle) in ["OBSRA", "OBSDEC"].into_iter().zip(obs.pointing_center_rad) {
342                self.add(key, FitsValue::Float(angle.to_degrees()), None);
343            }
344        }
345        Ok(())
346    }
347
348    fn extend_misc(&mut self, info: RecordValue) -> Result<(), ImageError> {
349        let mut fields = info.fields().iter().collect::<Vec<_>>();
350        fields.sort_by_key(|field| field.name.to_ascii_uppercase());
351        let mut cards_by_field = Vec::new();
352        for field in fields {
353            let name = field.name.to_ascii_uppercase();
354            if name.is_empty() || !name.is_ascii() {
355                continue;
356            }
357            let values = misc_values(&name, &field.value)?;
358            if values.is_empty() {
359                continue;
360            }
361            let shape = match &field.value {
362                Value::Array(array) => array.shape().to_vec(),
363                _ => Vec::new(),
364            };
365            cards_by_field.push((name, shape, values));
366        }
367        // casacore interleaves arrays of the same rank and element count,
368        // naming matrix cards using the last array's row count in the run.
369        for run in cards_by_field.chunk_by(|(_, first, values), (_, next, other)| {
370            first.len() == next.len() && values.len() == other.len()
371        }) {
372            let shape = &run.last().expect("nonempty run").1;
373            let mut fields = Vec::with_capacity(run.len());
374            for (name, _, values) in run {
375                let names = misc_keyword_names(name, shape, values.len())?;
376                fields.push(names.into_iter().zip(values));
377            }
378            for _ in 0..run[0].2.len() {
379                for cards in &mut fields {
380                    let (mut key, value) = cards.next().expect("one card per value");
381                    if key == "COMMENT"
382                        && matches!(value, FitsValue::String(text) if text.is_empty())
383                    {
384                        key.clear();
385                    }
386                    self.0.push(HeaderCard {
387                        keyword: FitsKeyword {
388                            name: key,
389                            value: value.clone(),
390                            comment: None,
391                        },
392                        group: None,
393                    });
394                }
395            }
396        }
397        Ok(())
398    }

400    /// The FITS header as consecutive 80-byte cards, with no block padding.
401    pub fn card_string(&self) -> String {
402        self.0
403            .iter()
404            .map(|card| render_card(&card.keyword))
405            .collect()
406    }

408    /// The FITS header as a CASA record with the given entry names omitted.
409    pub fn record(&self, exclude: &[String]) -> RecordValue {
410        let mut grouped = BTreeMap::<String, Vec<FitsValue>>::new();
411        for card in &self.0 {
412            let keyword = &card.keyword;
413            let name = card.group.unwrap_or(&keyword.name);
414            let name = if name.is_empty() { "        " } else { name };
415            grouped
416                .entry(name.to_owned())
417                .or_default()
418                .push(keyword.value.clone());
419        }
420        let mut record = RecordValue::default();
421        for (name, values) in grouped {
422            if !exclude.contains(&name) {
423                let axis_vector = matches!(
424                    name.as_str(),
425                    "NAXIS" | "CTYPE" | "CRVAL" | "CRPIX" | "CDELT"
426                );
427                let value = if !axis_vector && values.len() == 1 {
428                    scalar(&name, &values[0]).into()
429                } else {
430                    Value::Array(array(&name, &values))
431                };
432                record.upsert(name, value);
433            }
434        }
435        record
436    }
437}

439fn linear_fallback(
440    coords: &CoordinateSystem,
441    rank: usize,
442) -> Result<BTreeMap<String, FitsValue>, ImageError> {
443    if coords.n_world_axes() != rank
444        || (0..rank).any(|axis| coords.pixel_axis_to_world_axis(axis) != Some(axis))
445    {
446        return Err(header_creation_error());
447    }
448    let mut wcs = BTreeMap::new();
449    for index in 0..coords.n_coordinates() {
450        let coordinate = coords.coordinate(index);
451        let (reference, increment, units) = match coordinate {
452            CoordinateModel::Direction(direction) => {
453                let (reference, increment) = direction.reference_and_increment_in_world_units();
454                (reference, increment, direction.visible_world_units())
455            }
456            _ => (
457                coordinate.reference_value(),
458                coordinate.increment(),
459                coordinate.axis_units(),
460            ),
461        };
462        let names = coordinate.axis_names();
463        let pixels = coordinate.reference_pixel();
464        for (local, axis) in coords
465            .coordinate_pixel_axes(index)
466            .expect("existing coordinate")
467            .into_iter()
468            .enumerate()
469        {
470            let Some(axis) = axis else {
471                continue;
472            };
473            let axis = axis + 1;
474            let mut name = names[local].to_ascii_uppercase();
475            let mut unit = units[local].clone();
476            truncate_bytes(&mut name, 8);
477            truncate_bytes(&mut unit, 8);
478            for (key, value) in [
479                ("CTYPE", FitsValue::String(name)),
480                ("CRVAL", FitsValue::Float(reference[local])),
481                ("CDELT", FitsValue::Float(increment[local])),
482                ("CRPIX", FitsValue::Float(pixels[local] + 1.0)),
483                ("CUNIT", FitsValue::String(unit)),
484            ] {
485                wcs.insert(format!("{key}{axis}"), value);
486            }
487        }
488    }
489    Ok(wcs)
490}

492fn normalize_wcs(
493    coords: &CoordinateSystem,
494    wcs: &mut BTreeMap<String, FitsValue>,
495) -> Result<(), ImageError> {
496    if !wcs.contains_key("RESTFRQ") {
497        wcs.remove("SPECSYS");
498    }
499    for index in 0..coords.n_coordinates() {
500        let coordinate = coords.coordinate(index);
501        let axes = coords
502            .coordinate_world_axes(index)
503            .expect("existing coordinate");
504        match coordinate {
505            CoordinateModel::Direction(direction) => {
506                let frame = direction.direction_ref();
507                let (longitude, latitude) = match frame {
508                    DirectionRef::GALACTIC => ("GLON", "GLAT"),
509                    DirectionRef::SUPERGAL => ("SLON", "SLAT"),
510                    DirectionRef::ECLIPTIC | DirectionRef::MECLIPTIC | DirectionRef::TECLIPTIC => {
511                        ("ELON", "ELAT")
512                    }
513                    DirectionRef::J2000
514                    | DirectionRef::JMEAN
515                    | DirectionRef::APP
516                    | DirectionRef::B1950
517                    | DirectionRef::B1950Vla
518                    | DirectionRef::BMEAN
519                    | DirectionRef::BTRUE
520                    | DirectionRef::ICRS
521                    | DirectionRef::HADEC => ("RA--", "DEC-"),
522                    _ => ("??LN", "??LT"),
523                };
524                let projection = direction.projection().name();
525                for (axis, name) in axes.iter().zip([longitude, latitude]) {
526                    let Some(axis) = axis else {
527                        continue;
528                    };
529                    wcs.insert(
530                        format!("CTYPE{}", axis + 1),
531                        FitsValue::String(format!("{name}-{projection}")),
532                    );
533                }
534                wcs.remove("EQUINOX");
535                wcs.remove("RADESYS");
536                let reference = match frame {
537                    DirectionRef::J2000 => Some(("FK5", Some(2000.0))),
538                    DirectionRef::B1950 => Some(("FK4", Some(1950.0))),
539                    DirectionRef::B1950Vla => Some(("FK4", Some(1979.9))),
540                    DirectionRef::ICRS => Some(("ICRS", None)),
541                    _ => None,
542                };
543                if let Some((system, equinox)) = reference {
544                    wcs.insert("RADESYS".into(), FitsValue::String(system.into()));
545                    if let Some(equinox) = equinox {
546                        wcs.insert("EQUINOX".into(), FitsValue::Float(equinox));
547                    }
548                }
549            }
550            CoordinateModel::Stokes(stokes) => {
551                if let [Some(axis)] = axes.as_slice()
552                    && let [Some(pixel)] = coords
553                        .coordinate_pixel_axes(index)
554                        .expect("existing coordinate")
555                        .as_slice()
556                {
557                    wcs.extend(stokes_axis_keywords(stokes, axis + 1, pixel + 1)?);
558                }
559            }
560            _ => {}
561        }
562    }
563    for (key, value) in wcs.iter_mut() {
564        if matches!(split_axis_keyword(key), Some(("CTYPE" | "CUNIT", _)))
565            && let FitsValue::String(text) = value
566        {
567            truncate_bytes(text, 8);
568        }
569    }
570    Ok(())
571}

573pub(super) fn wrapped_beam_position_angle(angle: f64) -> f64 {
574    let angle = angle % 180.0;
575    if angle > 90.0 {
576        angle - 180.0
577    } else if angle <= -90.0 {
578        angle + 180.0
579    } else {
580        angle
581    }
582}

584pub(super) fn stokes_axis_keywords(
585    stokes: &StokesCoordinate,
586    axis: usize,
587    pixel: usize,
588) -> Result<[(String, FitsValue); 3], ImageError> {
589    let (reference, increment) =
590        fits_stokes_axis(stokes.stokes()).ok_or_else(header_creation_error)?;
591    Ok([
592        (format!("CRVAL{axis}"), FitsValue::Float(reference)),
593        (format!("CDELT{axis}"), FitsValue::Float(increment)),
594        (format!("CRPIX{pixel}"), FitsValue::Float(1.0)),
595    ])
596}

598pub(super) fn projection_parameters(
599    coords: &CoordinateSystem,
600    output_axis: impl Fn(usize) -> Option<usize>,
601) -> Vec<(String, FitsValue)> {
602    let mut result = Vec::new();
603    for index in 0..coords.n_coordinates() {
604        let axis = coords.coordinate(index);
605        if let CoordinateModel::Direction(direction) = axis
606            && let [_, Some(latitude)] = coords
607                .coordinate_world_axes(index)
608                .expect("existing coordinate")
609                .as_slice()
610            && let Some(latitude_axis) = output_axis(latitude + 1)
611        {
612            let projection = direction.projection();
613            let params = projection.parameters();
614            let start = projection.projection_type().fits_parameter_start();
615            let count = match projection.projection_type() {
616                ProjectionType::SIN => 2,
617                ProjectionType::ZPN => params
618                    .len()
619                    .min(projection.projection_type().fits_parameter_defaults().len()),
620                _ => params.len(),
621            };
622            for index in 0..count {
623                result.push((
624                    format!("PV{}_{}", latitude_axis, index + start),
625                    FitsValue::Float(params.get(index).copied().unwrap_or(0.0)),
626                ));
627            }
628        }
629    }
630    result
631}

633fn header_creation_error() -> ImageError {
634    ImageError::InvalidMetadata("header creation failed: Error creating initial FITS header".into())

637fn misc_keyword_names(
638    name: &str,
639    shape: &[usize],
640    count: usize,
641) -> Result<Vec<String>, ImageError> {
642    match shape {
643        [] => Ok(vec![if name.starts_with("COMMENT") {
644            "COMMENT".into()
645        } else if name.starts_with("HISTORY") {
646            "HISTORY".into()
647        } else {
648            name[..name.len().min(8)].to_owned()
649        }]),
650        [length] if *length <= 999 && name.len() + length.to_string().len() <= 8 => {
651            Ok((1..=count).map(|index| format!("{name}{index}")).collect())
652        }
653        [rows, cols] if name.len() <= 2 && *rows <= 99 && *cols <= 99 => {
654            let width = if *rows > 9 { 2 } else { 1 };
655            Ok((0..count)
656                .map(|index| {
657                    let row = index % rows + 1;
658                    let col = index / rows + 1;
659                    format!("{name}{row:0width$}_{col:0width$}")
660                })
661                .collect())
662        }
663        _ => Err(header_creation_error()),
664    }
665}

667fn misc_values(name: &str, value: &Value) -> Result<Vec<FitsValue>, ImageError> {
668    let card_integer = |value| FitsValue::Integer(i64::from(value as i32));
669    macro_rules! array {
670        ($values:expr, $variant:ident, $map:expr) => {
671            $values
672                .t()
673                .iter()
674                .map(|value| FitsValue::$variant($map(value)))
675                .collect()
676        };
677    }
678    let values = match value {
679        Value::Scalar(value) => {
680            let scalar = match value {
681                ScalarValue::Bool(value) => FitsValue::Logical(*value),
682                ScalarValue::UInt8(value) => FitsValue::Integer(i64::from(*value)),
683                ScalarValue::UInt16(value) => FitsValue::Integer(i64::from(*value)),
684                ScalarValue::UInt32(value) => card_integer(i64::from(*value)),
685                ScalarValue::Int16(value) => FitsValue::Integer(i64::from(*value)),
686                ScalarValue::Int32(value) => FitsValue::Integer(i64::from(*value)),
687                ScalarValue::Int64(value) => card_integer(*value),
688                ScalarValue::Float32(value) => FitsValue::Float(f64::from(*value)),
689                ScalarValue::Float64(value) => FitsValue::Float(*value),
690                ScalarValue::String(value) => {
691                    let mut text = value.clone();
692                    if name.starts_with("COMMENT") || name.starts_with("HISTORY") {
693                        truncate_bytes(&mut text, 72);
694                        return Ok(vec![FitsValue::String(text)]);
695                    }
696                    FitsValue::String(text)
697                }
698                ScalarValue::Complex32(_) | ScalarValue::Complex64(_) => {
699                    return Err(header_creation_error());
700                }
701            };
702            vec![scalar]
703        }
704        Value::Array(ArrayValue::Complex32(_) | ArrayValue::Complex64(_)) => Vec::new(),
705        Value::Array(value) if value.shape().len() > 2 => return Err(header_creation_error()),
706        Value::Array(value) => match value {
707            ArrayValue::Bool(values) => array!(values, Logical, |value: &bool| *value),
708            ArrayValue::UInt8(values) => array!(values, Integer, |value: &u8| i64::from(*value)),
709            ArrayValue::UInt16(values) => array!(values, Integer, |value: &u16| i64::from(*value)),
710            ArrayValue::UInt32(values) => {
711                array!(values, Integer, |value: &u32| i64::from(*value as i32))
712            }
713            ArrayValue::Int16(values) => array!(values, Integer, |value: &i16| i64::from(*value)),
714            ArrayValue::Int32(values) => array!(values, Integer, |value: &i32| i64::from(*value)),
715            ArrayValue::Int64(values) => {
716                array!(values, Integer, |value: &i64| i64::from(*value as i32))
717            }
718            ArrayValue::Float32(values) => array!(values, Float, |value: &f32| f64::from(*value)),
719            ArrayValue::Float64(values) => array!(values, Float, |value: &f64| *value),
720            ArrayValue::String(values) => array!(values, String, Clone::clone),
721            _ => Vec::new(),
722        },
723        _ => Vec::new(),
724    };
725    if values
726        .iter()
727        .any(|value| matches!(value, FitsValue::String(text) if text.len() > 68))
728    {
729        return Err(header_creation_error());
730    }
731    Ok(values)
732}

734fn group_name(name: &str) -> Option<&'static str> {
735    if name == "NAXIS" {
736        return Some("NAXIS");
737    }
738    if let Some((base, _)) = split_axis_keyword(name) {
739        return (base != "CUNIT").then_some(base);
740    }
741    name.strip_prefix("NAXIS")
742        .filter(|axis| !axis.is_empty() && axis.bytes().all(|ch| ch.is_ascii_digit()))
743        .map(|_| "NAXIS")
744}

746fn padded_text(name: &str, value: &str) -> String {
747    if matches!(name, "        " | "END" | "HISTORY" | "COMMENT") {
748        value.to_owned()
749    } else {
750        let mut text = value.to_owned();
751        text.extend(std::iter::repeat_n(' ', 8_usize.saturating_sub(text.len())));
752        text
753    }
754}
755
756fn scalar(name: &str, value: &FitsValue) -> ScalarValue {
757    match value {
758        FitsValue::String(value) => ScalarValue::String(padded_text(name, value)),
759        FitsValue::Integer(value) => ScalarValue::Int64(*value),
760        FitsValue::Float(value) => ScalarValue::Float64(*value),
761        FitsValue::Logical(value) => ScalarValue::Bool(*value),
762    }
763}

765fn array(name: &str, values: &[FitsValue]) -> ArrayValue {
766    match values.first().expect("at least one FITS card in a group") {
767        FitsValue::String(_) => ArrayValue::from_string_vec(
768            values
769                .iter()
770                .filter_map(|value| match value {
771                    FitsValue::String(text) => Some(padded_text(name, text)),
772                    _ => None,
773                })
774                .collect(),
775        ),
776        FitsValue::Integer(_) => ArrayValue::Int64(
777            ndarray::Array1::from_iter(values.iter().filter_map(|value| match value {
778                FitsValue::Integer(integer) => Some(*integer),
779                _ => None,
780            }))
781            .into_dyn(),
782        ),
783        FitsValue::Float(_) => ArrayValue::from_f64_vec(
784            values
785                .iter()
786                .filter_map(|value| match value {
787                    FitsValue::Float(float) => Some(*float),
788                    _ => None,
789                })
790                .collect(),
791        ),
792        FitsValue::Logical(_) => ArrayValue::Bool(
793            ndarray::Array1::from_iter(values.iter().filter_map(|value| match value {
794                FitsValue::Logical(flag) => Some(*flag),
795                _ => None,
796            }))
797            .into_dyn(),
798        ),
799    }
800}

802fn render_card(keyword: &FitsKeyword) -> String {
803    let name = &keyword.name;
804    let mut card = match &keyword.value {
805        FitsValue::String(text) if matches!(name.as_str(), "" | "END" | "HISTORY" | "COMMENT") => {
806            format!("{name:<8}{text}")
807        }
808        value => {
809            let field = match value {
810                FitsValue::String(text) => {
811                    format!("'{}'", padded_text(name, text).replace('\'', "''"))
812                }
813                FitsValue::Integer(integer) => format!("{integer:>20}"),
814                FitsValue::Logical(flag) => format!("{:>20}", if *flag { 'T' } else { 'F' }),
815                FitsValue::Float(number) => {
816                    let token = format!("{number:.12E}");
817                    let field = token.split_once('E').map_or_else(
818                        || token.clone(),
819                        |(mantissa, exponent)| {
820                            format!(
821                                "{mantissa}E{:+03}",
822                                exponent.parse::<i32>().expect("decimal exponent")
823                            )
824                        },
825                    );
826                    format!("{field:>20}")
827                }
828            };
829            format!("{name:<8}= {field}")
830        }
831    };
832    if let Some(comment) = &keyword.comment {
833        card.extend(std::iter::repeat_n(
834            ' ',
835            30_usize.saturating_sub(card.len()),
836        ));
837        card.push_str(" /");
838        card.push_str(comment);
839    }
840    truncate_bytes(&mut card, 80);
841    card.extend(std::iter::repeat_n(' ', 80 - card.len()));
842    card
843}
844
845fn truncate_bytes(text: &mut String, limit: usize) {
846    if text.len() > limit {
847        let boundary = (0..=limit)
848            .rev()
849            .find(|index| text.is_char_boundary(*index))
850            .expect("zero is a boundary");
851        text.truncate(boundary);
852    }
853}
