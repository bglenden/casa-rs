2//! The string buckets of a StandardStMan file being written: where long
3//! scalar strings and string arrays are stored (casacore's `SSMStringHandler`).
4
5use super::StorageError;
6
7/// Bytes of a string bucket's header: four canonical (big-endian) ints,
8/// the free link, the used length, the deleted count and the next bucket.
9const HEADER_BYTES: usize = 16;
10
11/// String buckets as they are filled.
12///
13/// Strings are only ever appended to the last bucket, so the ones before it
14/// are final. A writer that streams its file writes them out as the next
15/// one starts and releases them ([`Self::release`]), holding one bucket
16/// whatever the row count; a writer that assembles the file in memory keeps
17/// them all. Bucket numbers count the released buckets too.
18#[derive(Debug, Default)]
19pub(super) struct StringBuckets {
20    /// Buckets written out and dropped.
21    released: usize,
22    /// Buckets still held, numbered from `released`; the last one is open.
23    held: Vec<Vec<u8>>,
24}
25
26impl StringBuckets {
27    /// Buckets made so far, released or held.
28    pub(super) fn len(&self) -> usize {
29        self.released + self.held.len()
30    }
31
32    /// The buckets still held, numbered from the count of released ones.
33    pub(super) fn held(&self) -> &[Vec<u8>] {
34        &self.held
35    }
36
37    /// Store `data` in the open bucket, or in a new one of `bucket_size`
38    /// bytes when it does not fit, and return the bucket's number and the
39    /// offset of `data` in its data area.
40    pub(super) fn allocate(&mut self, data: &[u8], bucket_size: usize) -> (usize, usize) {
41        let capacity = bucket_size - HEADER_BYTES;
42        if let Some(open) = self.held.last_mut() {
43            let used = i32::from_be_bytes([open[4], open[5], open[6], open[7]]) as usize;
44            if used + data.len() <= capacity {
45                open[HEADER_BYTES + used..HEADER_BYTES + used + data.len()].copy_from_slice(data);
46                open[4..8].copy_from_slice(&((used + data.len()) as i32).to_be_bytes());
47                return (self.len() - 1, used);
48            }
49        }
50        let mut bucket = vec![0u8; bucket_size];
51        // No free link; the data's length; nothing deleted; no next bucket.
52        bucket[0..4].copy_from_slice(&(-1i32).to_be_bytes());
53        bucket[4..8].copy_from_slice(&(data.len() as i32).to_be_bytes());
54        bucket[8..12].copy_from_slice(&0i32.to_be_bytes());
55        bucket[12..16].copy_from_slice(&(-1i32).to_be_bytes());
56        bucket[HEADER_BYTES..HEADER_BYTES + data.len()].copy_from_slice(data);
57        self.held.push(bucket);
58        (self.len() - 1, 0)
59    }
60
61    /// Pass each finished bucket, and with `all` the open one too, to
62    /// `write` with its number, in order, then drop them.
63    pub(super) fn release(
64        &mut self,
65        all: bool,
66        mut write: impl FnMut(usize, &[u8]) -> Result<(), StorageError>,
67    ) -> Result<(), StorageError> {
68        let finished = if all {
69            self.held.len()
70        } else {
71            self.held.len().saturating_sub(1)
72        };
73        for (index, bucket) in self.held.drain(..finished).enumerate() {
74            write(self.released + index, &bucket)?;
75        }
76        self.released += finished;
77        Ok(())
78    }
79}

81#[cfg(test)]
82mod tests {
83    use super::*;
84
85    #[test]
86    fn released_buckets_keep_their_numbers() {
87        let bucket_size = HEADER_BYTES + 8;
88        let mut kept = StringBuckets::default();
89        let mut streamed = StringBuckets::default();
90        let mut written = Vec::new();
91        for string in [&b"abcde"[..], b"fgh", b"ijklmn", b"op", b"qrstuvwx"] {
92            let place = kept.allocate(string, bucket_size);
93            assert_eq!(streamed.allocate(string, bucket_size), place);
94            streamed
95                .release(false, |number, bucket| {
96                    written.push((number, bucket.to_vec()));
97                    Ok(())
98                })
99                .unwrap();
100            assert!(streamed.held().len() <= 1);
101        }
102        streamed
103            .release(true, |number, bucket| {
104                written.push((number, bucket.to_vec()));
105                Ok(())
106            })
107            .unwrap();
108        assert_eq!(kept.len(), 3);
109        assert_eq!(streamed.len(), 3);
110        assert!(streamed.held().is_empty());
111        let expected = kept.held().iter().cloned().enumerate().collect::<Vec<_>>();
112        assert_eq!(written, expected);
113    }
114}
