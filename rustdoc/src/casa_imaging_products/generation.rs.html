62impl ContinuumProductControls {
63    /// Construct validated controls.
64    ///
65    /// # Errors
66    ///
67    /// Rejects a cutoff outside `(0, 1)` or non-finite values.
68    pub fn new(psf_cutoff: f32) -> Result<Self, ProductsError> {
69        if !psf_cutoff.is_finite() || psf_cutoff <= 0.0 || psf_cutoff >= 1.0 {
70            return Err(ProductsError::InvalidControls);
71        }
72        Ok(Self {
73            psf_cutoff,
74            primary_beam_model: None,
75        })
76    }
77
78    /// Return the main-lobe cutoff fraction used for beam fitting.
79    #[must_use]
80    pub const fn psf_cutoff(self) -> f32 {
81        self.psf_cutoff
82    }
83
84    /// Select the exact analytic primary-beam law used by requested PB products.
85    #[must_use]
86    pub const fn with_primary_beam_model(mut self, model: AnalyticPrimaryBeamModel) -> Self {
87        self.primary_beam_model = Some(model);
88        self
89    }
90
91    /// Return the explicitly selected analytic primary-beam law, if any.
92    #[must_use]
93    pub const fn primary_beam_model(self) -> Option<AnalyticPrimaryBeamModel> {
94        self.primary_beam_model
95    }
96
97    /// Check beam-frequency coverage before visibility processing or production.
98    ///
99    /// Uses compiled output frequencies, including frame conversion and channel
100    /// ordering. Taylor products evaluate the beam at output channel zero;
101    /// channel-local products require coverage of every output channel. Unused
102    /// models impose no additional restriction on non-Taylor products.
103    ///
104    /// # Errors
105    ///
106    /// Returns the first unsupported output channel and its frequency without
107    /// constructing any beam planes or reading visibility payloads.
108    pub fn validate_for_problem(&self, problem: &CompiledProblem) -> Result<(), ProductsError> {
109        let Some(model) = self.primary_beam_model else {
110            return Ok(());
111        };
112        let taylor = matches!(
113            problem.reconstruction().basis(),
114            ReconstructionBasis::Taylor { .. }
115        );
116        let graph = problem.product_graph();
117        let needs_beam = taylor || graph.publication().members().iter().any(|ordinal| {
118            let node = &graph.nodes()[ordinal.ordinal()];
119            let needs_validity = |rule| matches!(rule,
120                ProductValidityRule::PrimaryBeam(_) | ProductValidityRule::TaylorAndPrimaryBeam { .. });
121            matches!(node.role(), ProductRole::PrimaryBeam(_) | ProductRole::PbCorrectedImage(_))
122                || needs_validity(node.validity())
123                || matches!(node.storage().pixel_mask(), ProductPixelMask::Explicit(rule) if needs_validity(rule))
124        });
125        if !needs_beam {
126            return Ok(());
127        }
128        let spectral = problem.geometry().spectral();
129        let channels = if taylor {
130            1
131        } else {
132            spectral.output_channels()
133        };
134        for output_channel in 0..channels {
135            let frequency_hz = spectral
136                .channel_centre_hz(output_channel)
137                .ok_or(ProductsError::UnsupportedProblem)?;
138            if !primary_beam_frequency_supported(model, frequency_hz) {
139                return Err(ProductsError::UnsupportedPrimaryBeamFrequency {
140                    model,
141                    output_channel,
142                    frequency_hz,
143                });
144            }
145        }
146        Ok(())
147    }
148}

159fn ensure_producible(role: ProductRole) -> Result<(), ProductsError> {
160    match role {
161        ProductRole::Psf(_)
162        | ProductRole::Residual(_)
163        | ProductRole::Model(_)
164        | ProductRole::Weight(_)
165        | ProductRole::RestoredImage(_)
166        | ProductRole::SumWeights(_)
167        | ProductRole::PrimaryBeam(_)
168        | ProductRole::PbCorrectedImage(_)
169        | ProductRole::SpectralIndex
170        | ProductRole::SpectralIndexError
171        | ProductRole::PbCorrectedSpectralIndex
172        | ProductRole::Sensitivity
173        | ProductRole::CleanMask
174        | ProductRole::ContinuumCleanMask
175        | ProductRole::LineCleanMask => Ok(()),
176        role => Err(ProductsError::UnsupportedProductRole {
177            role,
178            catalog: CONTINUUM_ALGORITHM_CATALOG_VERSION,
179        }),
180    }

201impl PlannedContinuumGeneration {
202    /// Compile the exact publication inventory for these scientific inputs.
203    pub fn new(
204        inputs: &ContinuumProductInputs<'_>,
205        controls: &ContinuumProductControls,
206    ) -> Result<Self, ProductsError> {
207        controls.validate_for_problem(inputs.problem())?;
208        let graph = inputs.problem().product_graph();
209        let mut members = Vec::with_capacity(graph.publication().members().len());
210        for node_ordinal in graph.publication().members() {
211            let node = graph
212                .nodes()
213                .get(node_ordinal.ordinal())
214                .ok_or(ProductsError::UnsupportedProblem)?;
215            ensure_producible(node.role())?;
216            let needs_primary_beam = |rule| {
217                matches!(
218                    rule,
219                    ProductValidityRule::PrimaryBeam(_)
220                        | ProductValidityRule::TaylorAndPrimaryBeam { .. }
221                )
222            };
223            let requires_primary_beam = needs_primary_beam(node.validity())
224                || matches!(node.storage().pixel_mask(), ProductPixelMask::Explicit(rule)
225                    if needs_primary_beam(rule));
226            if requires_primary_beam
227                && (controls.primary_beam_model.is_none()
228                    || matches!(
229                        inputs.problem().reconstruction().basis(),
230                        ReconstructionBasis::JointContinuumLine { .. }
231                    ))
232            {
233                return Err(ProductsError::UnsupportedProblem);
234            }
235            let axes = node.axes();
236            let shape = axes.shape();
237            let payload_values = shape
238                .iter()
239                .copied()
240                .try_fold(1usize, |total, extent| total.checked_mul(extent))
241                .ok_or(ProductsError::ResourceDemandOverflow("product shape"))?;
242            members.push(PlannedMember {
243                node: node.node_id(),
244                role: node.role(),
245                name: node.name().unwrap_or_default().to_string(),
246                shape,
247                payload_values,
248                unit: node.unit(),
249                schema: node.schema(),
250                axes: axes.clone(),
251                normalization: node.normalization(),
252                beam_rule: node.beam(),
253                validity: node.validity(),
254                storage: node.storage(),
255                dependencies: node.dependencies().to_vec().into_boxed_slice(),
256            });
257        }
258        Ok(Self {
259            problem_id: inputs.problem().problem_id(),
260            graph_id: graph.graph_id(),
261            major_cycle_completion: inputs.major_cycle_completion(),
262            normal_state_completion: inputs.normal_state_completion(),
263            psf_cutoff: controls.psf_cutoff(),
264            primary_beam_model: controls.primary_beam_model(),
265            members: members.into_boxed_slice(),
266            final_model_generation: inputs.final_model().generation_id(),
267            reconstruction_mask_generation: inputs.reconstruction_mask_generation(),
268            line_reconstruction_mask_generation: inputs
269                .coupled_reconstruction_masks()
270                .map(|masks| masks.line().generation_id()),
271        })
272    }

513/// Produce every planned member through the continuum algorithm catalog.
514///
515/// Runs restoring-beam fitting, restoration, residual scaling, normalization,
516/// validity, and metadata exactly once per member, in the planned publication
517/// order.
518///
519/// # Errors
520///
521/// Rejects source/run mismatches, unsupported roles, failed beam fits, and
522/// generated non-finite payloads.
523pub fn produce_continuum_members(
524    planned: &PlannedContinuumGeneration,
525    inputs: &ContinuumProductInputs<'_>,
526    storage_plan: ProductStoragePlan,
527    execution: &impl crate::ProductWindowExecutor,
528    output: &dyn ProductOutput,
529) -> Result<PublishedContinuumGeneration, ProductsError> {
530    if inputs.problem().problem_id() != planned.problem_id
531        || inputs.problem().product_graph().graph_id() != planned.graph_id
532        || inputs.major_cycle_completion() != planned.major_cycle_completion
533        || inputs.normal_state_completion() != planned.normal_state_completion
534        || inputs.final_model().generation_id() != planned.final_model_generation
535    {
536        return Err(ProductsError::SourceLineageMismatch);
537    }
538    if inputs.reconstruction_mask_generation() != planned.reconstruction_mask_generation {
539        return Err(ProductsError::SourceLineageMismatch);
540    }
541    if inputs
542        .coupled_reconstruction_masks()
543        .map(|masks| masks.line().generation_id())
544        != planned.line_reconstruction_mask_generation
545    {
546        return Err(ProductsError::SourceLineageMismatch);
547    }
548    if inputs.normal_state().catalog() == NormalStateCatalog::UnnormalizedTaylorBlockV1 {
549        return produce_taylor_members(planned, inputs, storage_plan, output);
550    }
551    if inputs.normal_state().catalog() == NormalStateCatalog::UnnormalizedJointBlockV1 {
552        return produce_joint_members(planned, inputs, storage_plan, output);
553    }
554    let normal_state = inputs.normal_state();
555    let channel_count = normal_state.channel_count();
556    if normal_state.slab().core_range() != (0..normal_state.slab().total_channels())
557        || channel_count != normal_state.slab().total_channels()
558        || inputs.final_model().shape().coefficients() != channel_count
559        || inputs.final_model().shape().polarizations() != normal_state.polarization_count()
560        || normal_state.domain_count() != inputs.problem().geometry().domains().len()
561        || inputs.final_model().shape().domains().len()
562            != inputs.problem().geometry().domains().len()
563    {
564        return Err(ProductsError::SourceLineageMismatch);
565    }
566    let requires_beam = planned
567        .members
568        .iter()
569        .any(|member| member.beam_rule != ProductBeamRule::None);
570    let fitted_beams = if requires_beam {
571        let beam_count = normal_state
572            .domain_count()
573            .checked_mul(channel_count)
574            .and_then(|count| count.checked_mul(normal_state.polarization_count()))
575            .ok_or(ProductsError::ResourceDemandOverflow(
576                "generation beam count",
577            ))?;
578        let mut fitted = Vec::with_capacity(beam_count);
579        for start in (0..beam_count).step_by(storage_plan.maximum_workers()) {
580            let count = storage_plan.maximum_workers().min(beam_count - start);
581            let mut slots = vec![None; count];
582            execution.prepare(&mut slots, &|index| {
583                let ordinal = start + index;
584                let polarization = ordinal % normal_state.polarization_count();
585                let local_channel = (ordinal / normal_state.polarization_count()) % channel_count;
586                let domain = &inputs.problem().geometry().domains()
587                    [ordinal / (channel_count * normal_state.polarization_count())];
588                let channel = normal_state.slab().core_range().start + local_channel;
589                let plane = normal_state.read_plane(
590                    inputs.model_domain_ordinal(domain.role())?,
591                    channel,
592                    polarization,
593                )?;
594                if plane.validity() == SpectralChannelValidity::Valid {
595                    fit_restoring_beam(
596                        &psf_real_plane(&plane)?,
597                        plane.shape(),
598                        inputs.cell_size_rad_for_domain(domain.role())?,
599                        planned.psf_cutoff(),
600                    )
601                    .map(Some)
602                } else {
603                    Ok(None)
604                }
605            })?;
606            for slot in slots {
607                fitted.push(slot.ok_or(ProductsError::SourceLineageMismatch)?);
608            }
609        }
610        fitted.into_boxed_slice()
611    } else {
612        Box::new([])
613    };
614    let restoring_beams = match inputs.problem().products().restoring_beam() {
615        RestoringBeamPolicy::None => Box::new([]),
616        RestoringBeamPolicy::PerPlane => fitted_beams.clone(),
617        RestoringBeamPolicy::Common => {
618            let mut selected = Vec::with_capacity(fitted_beams.len());
619            for domain_beams in
620                fitted_beams.chunks(channel_count * normal_state.polarization_count())
621            {
622                let mut valid = Vec::with_capacity(domain_beams.len());
623                valid.extend(domain_beams.iter().flatten().copied());
624                if valid.is_empty() {
625                    selected.extend_from_slice(domain_beams);
626                } else {
627                    let common = RestoringBeam::common_enclosing(&valid)
628                        .map_err(|error| ProductsError::BeamFitFailed(error.to_string()))?;
629                    selected.extend(domain_beams.iter().map(|fitted| fitted.map(|_| common)));
630                }
631            }
632            selected.into_boxed_slice()
633        }
634    };
635
636    for member in &planned.members {
637        let domain_ordinal = inputs.model_domain_ordinal(member.axes().domain())?;
638        let plane_shape = inputs
639            .final_model()
640            .shape()
641            .domains()
642            .get(domain_ordinal)
643            .ok_or(ProductsError::SourceLineageMismatch)?
644            .pixels();
645        let reconstruction_only = matches!(
646            member.role,
647            ProductRole::Model(
648                casa_imaging_model::ProductTerm::Single
649                    | casa_imaging_model::ProductTerm::Taylor(0)
650            ) | ProductRole::CleanMask
651        );
652        if normal_state.domain_shape(domain_ordinal) != Some(plane_shape)
653            || (reconstruction_only
654                && (member.validity != ProductValidityRule::All
655                    || member.storage.pixel_mask() != ProductPixelMask::Absent))
656        {
657            return Err(ProductsError::SourceLineageMismatch);
658        }
659        let beam_offset = domain_ordinal
660            .checked_mul(channel_count)
661            .and_then(|offset| offset.checked_mul(normal_state.polarization_count()))
662            .ok_or(ProductsError::SourceLineageMismatch)?;
663        let layout = storage_plan.layout(member.axes())?;
664        let member_beams = beams_for_member(
665            member,
666            &planned.members,
667            member.beam_rule,
668            &fitted_beams,
669            &restoring_beams,
670        )?;
671        let writer = output.begin_member(member, layout, &member_beams)?;
672        let mut writer = ProductMemberWriter::new(layout, writer)?;
673        let window_count = channel_count.div_ceil(layout.maximum_channels());
674        let workers = storage_plan.maximum_workers().min(window_count);
675        for wave_start in (0..window_count).step_by(workers) {
676            let mut slots: Vec<Option<crate::ProductWindow>> = (0..workers
677                .min(window_count - wave_start))
678                .map(|_| None)
679                .collect();
680            execution.prepare(&mut slots, &|index| {
681                let window_start = (wave_start + index) * layout.maximum_channels();
682                let window_end = (window_start + layout.maximum_channels()).min(channel_count);
683                let mut output = layout.window(window_start..window_end)?;
684                for local_channel in window_start..window_end {
685                    let channel = normal_state.slab().core_range().start + local_channel;
686                    if reconstruction_only {
687                        for polarization in 0..normal_state.polarization_count() {
688                            let payload = if member.role == ProductRole::CleanMask {
689                                reconstruction_support_plane(
690                                    inputs,
691                                    member.axes().domain(),
692                                    plane_shape[0] * plane_shape[1],
693                                )?
694                            } else {
695                                model_real_plane(
696                                    inputs.final_model(),
697                                    domain_ordinal,
698                                    channel,
699                                    polarization,
700                                    plane_shape,
701                                )?
702                            };
703                            scatter_image_polarization_plane(
704                                &mut output.payload,
705                                member.axes().order(),
706                                output.shape,
707                                polarization,
708                                channel - window_start,
709                                plane_shape,
710                                &payload,
711                            )?;
712                        }
713                        continue;
714                    }
715                    for polarization in 0..normal_state.polarization_count() {
716                        let plane =
717                            normal_state.read_plane(domain_ordinal, channel, polarization)?;
718                        if plane.shape() != plane_shape {
719                            return Err(ProductsError::SourceLineageMismatch);
720                        }
721                        let output_channel = plane.output_channel() - window_start;
722                        if matches!(member.role, ProductRole::SumWeights(_)) {
723                            scatter_polarization_plane_state(
724                                &mut output.payload,
725                                member.axes(),
726                                output.shape,
727                                polarization,
728                                output_channel,
729                                plane.published_sum_weight() as f32,
730                            )?;
731                            continue;
732                        }
733                        let beam_index = beam_offset
734                            + local_channel * normal_state.polarization_count()
735                            + polarization;
736                        let mut plane_payload = produce_plane_member(PlaneMemberRequest {
737                            member,
738                            inputs,
739                            plane: &plane,
740                            domain_ordinal,
741                            polarization,
742                            fitted_beam: fitted_beams.get(beam_index).copied().flatten(),
743                            restoring_beam: restoring_beams.get(beam_index).copied().flatten(),
744                            primary_beam_model: planned.primary_beam_model,
745                        })?;
746                        if member.validity != ProductValidityRule::All {
747                            let support = product_plane_validity(
748                                member.validity,
749                                &plane,
750                                planned.primary_beam_model,
751                                inputs,
752                                member.axes().domain(),
753                            )?;
754                            zero_invalid_plane_values(&mut plane_payload, &support)?;
755                        }
756                        if let ProductPixelMask::Explicit(rule) = member.storage.pixel_mask() {
757                            let support = product_plane_validity(
758                                rule,
759                                &plane,
760                                planned.primary_beam_model,
761                                inputs,
762                                member.axes().domain(),
763                            )?;
764                            scatter_image_polarization_plane(
765                                &mut output.validity,
766                                member.axes().order(),
767                                output.shape,
768                                polarization,
769                                output_channel,
770                                plane_shape,
771                                &support,
772                            )?;
773                        }
774                        scatter_image_polarization_plane(
775                            &mut output.payload,
776                            member.axes().order(),
777                            output.shape,
778                            polarization,
779                            output_channel,
780                            plane_shape,
781                            &plane_payload,
782                        )?;
783                    }
784                }
785                Ok(output)
786            })?;
787            for slot in slots {
788                writer.write(slot.ok_or(ProductsError::InvalidWindow)?)?;
789            }
790        }
791        writer.finish()?;
792    }
793
794    published_generation(planned, fitted_beams, restoring_beams)
795}

852fn produce_joint_members(
853    planned: &PlannedContinuumGeneration,
854    inputs: &ContinuumProductInputs<'_>,
855    storage_plan: ProductStoragePlan,
856    output: &dyn ProductOutput,
857) -> Result<PublishedContinuumGeneration, ProductsError> {
858    let normal = inputs.normal_state();
859    let normal = &normal.read_window(normal.slab().core_range())?;
860    if normal.domain_count() != 1 || inputs.final_model().shape().domains().len() != 1 {
861        return Err(ProductsError::SourceLineageMismatch);
862    }
863    let domain_role = normal
864        .domain(0)
865        .ok_or(ProductsError::SourceLineageMismatch)?
866        .role()
867        .clone();
868    let shape = normal.shape();
869    let channels = normal.slab().total_channels();
870    if normal.slab().core_range() != (0..channels)
871        || inputs.coupled_reconstruction_masks().is_none()
872    {
873        return Err(ProductsError::SourceLineageMismatch);
874    }
875    let ReconstructionBasis::JointContinuumLine {
876        continuum_terms,
877        line_terms,
878    } = inputs.problem().reconstruction().basis()
879    else {
880        return Err(ProductsError::SourceLineageMismatch);
881    };
882    if normal.coefficient_term_count() != continuum_terms + line_terms
883        || normal.normal_moment_count() != (continuum_terms + line_terms).pow(2)
884        || inputs.final_model().shape().coefficients() != continuum_terms + line_terms
885        || normal.channel_sum_weights().len() != channels
886    {
887        return Err(ProductsError::SourceLineageMismatch);
888    }
889    let h00 = normal
890        .normal_block(0, 0)
891        .ok_or(ProductsError::SourceLineageMismatch)?;
892    let normalization_weight = h00.sum_weight();
893    if !normalization_weight.is_finite() || normalization_weight <= 0.0 {
894        return Err(ProductsError::SourceLineageMismatch);
895    }
896    let requires_beam = planned
897        .members
898        .iter()
899        .any(|member| member.beam_rule != ProductBeamRule::None);
900    let fitted_beam = requires_beam
901        .then(|| {
902            fit_restoring_beam(
903                &h00.normal_approximation()
904                    .iter()
905                    .map(|value| value.re as f32)
906                    .collect::<Vec<_>>(),
907                shape,
908                inputs.cell_size_rad_for_domain(&domain_role)?,
909                planned.psf_cutoff(),
910            )
911        })
912        .transpose()?;
913    let restoring_beam = match inputs.problem().products().restoring_beam() {
914        RestoringBeamPolicy::None => None,
915        RestoringBeamPolicy::PerPlane | RestoringBeamPolicy::Common => fitted_beam,
916    };
917    let fitted_beams = fitted_beam.map_or_else(Box::default, |beam| {
918        vec![Some(beam); channels].into_boxed_slice()
919    });
920    let restoring_beams = restoring_beam.map_or_else(Box::default, |beam| {
921        vec![Some(beam); channels].into_boxed_slice()
922    });
923    for member in &planned.members {
924        let (payload, mut validity) = produce_joint_member(
925            member,
926            inputs,
927            shape,
928            channels,
929            continuum_terms,
930            normalization_weight,
931            fitted_beam,
932            restoring_beam,
933        )?;
934        if member.storage.pixel_mask() == ProductPixelMask::Absent {
935            validity.fill(true);
936        }
937        if payload.len() != member.payload_values || validity.len() != member.payload_values {
938            return Err(ProductsError::PayloadLengthMismatch {
939                expected: member.payload_values,
940                actual: payload.len(),
941            });
942        }
943        let layout = storage_plan.layout(member.axes())?;
944        let member_beams = beams_for_member(
945            member,
946            &planned.members,
947            member.beam_rule,
948            &fitted_beams,
949            &restoring_beams,
950        )?;
951        let writer = output.begin_member(member, layout, &member_beams)?;
952        let mut writer = ProductMemberWriter::new(layout, writer)?;
953        writer.write_coupled(&payload, &validity)?;
954        writer.finish()?;
955    }
956    published_generation(planned, fitted_beams, restoring_beams)
957}

959#[allow(clippy::too_many_arguments)]
960fn produce_joint_member(
961    member: &PlannedMember,
962    inputs: &ContinuumProductInputs<'_>,
963    shape: [usize; 2],
964    channels: usize,
965    continuum_terms: usize,
966    normalization_weight: f64,
967    fitted_beam: Option<RestoringBeam>,
968    restoring_beam: Option<RestoringBeam>,
969) -> Result<(Vec<f32>, Vec<bool>), ProductsError> {
970    let normal = inputs.normal_state();
971    let normal = &normal.read_window(normal.slab().core_range())?;
972    let mut payload = vec![0.0_f32; member.payload_values];
973    let mut validity = vec![true; member.payload_values];
974    match member.role {
975        ProductRole::Psf(ProductTerm::JointNormal { row, column }) => {
976            let block = normal
977                .normal_block(row, column)
978                .ok_or(ProductsError::SourceLineageMismatch)?;
979            payload = normalize_plane(
980                &block
981                    .normal_approximation()
982                    .iter()
983                    .map(|value| value.re as f32)
984                    .collect::<Vec<_>>(),
985                member
986                    .normalization
987                    .unwrap_or(ProductNormalization::UnitResponse),
988                normalization_weight,
989            )?;
990        }
991        ProductRole::SumWeights(ProductTerm::JointNormal { row, column }) => {
992            let block = normal
993                .normal_block(row, column)
994                .ok_or(ProductsError::SourceLineageMismatch)?;
995            scatter_plane_state(&mut payload, member.axes(), 0, block.sum_weight() as f32)?;
996        }
997        ProductRole::Weight(ProductTerm::JointNormal { row, column }) => {
998            let block = normal
999                .normal_block(row, column)
1000                .ok_or(ProductsError::SourceLineageMismatch)?;
1001            payload = block
1002                .sensitivity()
1003                .iter()
1004                .map(|value| *value as f32)
1005                .collect();
1006        }
1007        ProductRole::Model(ProductTerm::Continuum(term)) => {
1008            payload = model_real_plane(inputs.final_model(), 0, term, 0, shape)?;
1009        }
1010        ProductRole::Model(ProductTerm::Line) | ProductRole::Model(ProductTerm::Total) => {
1011            for channel in 0..channels {
1012                let plane = evaluate_joint_model_plane(
1013                    inputs,
1014                    shape,
1015                    channel,
1016                    continuum_terms,
1017                    matches!(member.role, ProductRole::Model(ProductTerm::Line)),
1018                )?;
1019                scatter_image_plane(&mut payload, member.axes(), channel, shape, &plane)?;
1020            }
1021        }
1022        ProductRole::Residual(ProductTerm::Total) => {
1023            for channel in 0..channels {
1024                let plane = evaluate_joint_residual_plane(
1025                    inputs,
1026                    shape,
1027                    channel,
1028                    required_normalization(member)?,
1029                )?;
1030                scatter_image_plane(&mut payload, member.axes(), channel, shape, &plane)?;
1031                if inputs.normal_state().channel_validity()[channel]
1032                    != SpectralChannelValidity::Valid
1033                {
1034                    let blank = vec![false; shape[0] * shape[1]];
1035                    scatter_image_plane(&mut validity, member.axes(), channel, shape, &blank)?;
1036                }
1037            }
1038        }
1039        ProductRole::RestoredImage(ProductTerm::Line | ProductTerm::Total) => {
1040            let fitted_beam = fitted_beam.ok_or_else(|| {
1041                ProductsError::BeamFitFailed("joint restoration requires a fitted beam".to_string())
1042            })?;
1043            let restoring_beam = restoring_beam.ok_or_else(|| {
1044                ProductsError::BeamFitFailed(
1045                    "joint restoration requires a restoring beam".to_string(),
1046                )
1047            })?;
1048            let line_only = member.role == ProductRole::RestoredImage(ProductTerm::Line);
1049            let cell_size = inputs.cell_size_rad_for_domain(member.axes().domain())?;
1050            let kernel = gaussian_beam_image(shape, &restoring_beam, cell_size);
1051            for channel in 0..channels {
1052                let model =
1053                    evaluate_joint_model_plane(inputs, shape, channel, continuum_terms, line_only)?;
1054                let mut restored =
1055                    fft_convolve(&model, kernel.as_slice().expect("contiguous"), shape);
1056                let residual = evaluate_joint_residual_plane(
1057                    inputs,
1058                    shape,
1059                    channel,
1060                    required_normalization(member)?,
1061                )?;
1062                let residual = rescale_residual_to_beam(
1063                    &residual,
1064                    shape,
1065                    cell_size,
1066                    fitted_beam,
1067                    restoring_beam,
1068                )?
1069                .into_values();
1070                for (restored, residual) in restored.iter_mut().zip(residual) {
1071                    *restored += residual;
1072                }
1073                scatter_image_plane(&mut payload, member.axes(), channel, shape, &restored)?;
1074                if inputs.normal_state().channel_validity()[channel]
1075                    != SpectralChannelValidity::Valid
1076                {
1077                    let blank = vec![false; shape[0] * shape[1]];
1078                    scatter_image_plane(&mut validity, member.axes(), channel, shape, &blank)?;
1079                }
1080            }
1081        }
1082        ProductRole::ContinuumCleanMask | ProductRole::LineCleanMask => {
1083            let masks = inputs
1084                .coupled_reconstruction_masks()
1085                .ok_or(ProductsError::SourceLineageMismatch)?;
1086            let mask = if member.role == ProductRole::ContinuumCleanMask {
1087                masks.continuum()
1088            } else {
1089                masks.line()
1090            };
1091            for channel in 0..channels {
1092                let plane = mask
1093                    .support()
1094                    .iter()
1095                    .map(|selected| if *selected { 1.0 } else { 0.0 })
1096                    .collect::<Vec<_>>();
1097                scatter_image_plane(&mut payload, member.axes(), channel, shape, &plane)?;
1098            }
1099        }
1100        ProductRole::Sensitivity => {
1101            for channel in 0..channels {
1102                let plane = h00_sensitivity(inputs)?;
1103                scatter_image_plane(&mut payload, member.axes(), channel, shape, &plane)?;
1104            }
1105        }
1106        role => {
1107            return Err(ProductsError::UnsupportedProductRole {
1108                role,
1109                catalog: CONTINUUM_ALGORITHM_CATALOG_VERSION,
1110            });
1111        }
1112    }
1113    Ok((payload, validity))
1114}

1116fn evaluate_joint_model_plane(
1117    inputs: &ContinuumProductInputs<'_>,
1118    shape: [usize; 2],
1119    channel: usize,
1120    continuum_terms: usize,
1121    line_only: bool,
1122) -> Result<Vec<f32>, ProductsError> {
1123    let mut output = vec![0.0_f32; shape[0] * shape[1]];
1124    if !line_only {
1125        let reference = inputs
1126            .normal_state()
1127            .reference_frequency_hz()
1128            .ok_or(ProductsError::SourceLineageMismatch)?;
1129        let frequency = inputs
1130            .problem()
1131            .geometry()
1132            .spectral()
1133            .channel_centre_hz(channel)
1134            .ok_or(ProductsError::SourceLineageMismatch)?;
1135        let x = (frequency - reference) / reference;
1136        for term in 0..continuum_terms {
1137            let plane = model_real_plane(inputs.final_model(), 0, term, 0, shape)?;
1138            let factor =
1139                x.powi(i32::try_from(term).map_err(|_| ProductsError::UnsupportedProblem)?);
1140            for (output, value) in output.iter_mut().zip(plane) {
1141                *output += (factor * f64::from(value)) as f32;
1142            }
1143        }
1144    }
1145    if let Some(line) = joint_line_term(inputs.problem(), channel)? {
1146        let plane = model_real_plane(inputs.final_model(), 0, continuum_terms + line, 0, shape)?;
1147        for (output, value) in output.iter_mut().zip(plane) {
1148            *output += value;
1149        }
1150    }
1151    Ok(output)
1152}

1174        .channel_sum_weights()

1203struct PlaneMemberRequest<'request, 'inputs, 'plane> {
1204    member: &'request PlannedMember,
1205    inputs: &'request ContinuumProductInputs<'inputs>,
1206    plane: &'request FinalNormalPlaneReader<'plane>,
1207    domain_ordinal: usize,
1208    polarization: usize,
1209    fitted_beam: Option<RestoringBeam>,
1210    restoring_beam: Option<RestoringBeam>,
1211    primary_beam_model: Option<AnalyticPrimaryBeamModel>,
1212}
1213
1214fn produce_plane_member(
1215    request: PlaneMemberRequest<'_, '_, '_>,
1216) -> Result<Vec<f32>, ProductsError> {
1217    let PlaneMemberRequest {
1218        member,
1219        inputs,
1220        plane,
1221        domain_ordinal,
1222        polarization,
1223        fitted_beam,
1224        restoring_beam,
1225        primary_beam_model,
1226    } = request;
1227    let scalar_sensitivity = plane.sum_weight();
1228    let valid = plane.validity() == SpectralChannelValidity::Valid
1229        && scalar_sensitivity.is_finite()
1230        && scalar_sensitivity > 0.0;
1231    let shape = plane.shape();
1232    let cells = shape[0] * shape[1];
1233    let invalid_residual = || vec![0.0; cells];
1234    match member.role {
1235        ProductRole::Psf(casa_imaging_model::ProductTerm::Single)
1236        | ProductRole::Psf(casa_imaging_model::ProductTerm::Taylor(0)) => {
1237            if valid {
1238                normalize_domain_plane(
1239                    &psf_real_plane(plane)?,
1240                    member
1241                        .normalization
1242                        .unwrap_or(ProductNormalization::UnitResponse),
1243                    plane,
1244                )
1245            } else {
1246                Ok(vec![0.0; cells])
1247            }
1248        }
1249        ProductRole::Residual(
1250            casa_imaging_model::ProductTerm::Single | casa_imaging_model::ProductTerm::Taylor(0),
1251        ) => {
1252            if valid {
1253                normalize_domain_plane(
1254                    &residual_real_plane(plane)?,
1255                    required_normalization(member)?,
1256                    plane,
1257                )
1258            } else {
1259                Ok(invalid_residual())
1260            }
1261        }
1262        ProductRole::Weight(
1263            casa_imaging_model::ProductTerm::Single | casa_imaging_model::ProductTerm::Taylor(0),
1264        ) => {
1265            let scale = if primary_beam_model == Some(AnalyticPrimaryBeamModel::MosaicSensitivity) {
1266                if !plane.sum_weight().is_finite() || plane.sum_weight() <= 0.0 {
1267                    return Err(ProductsError::GeneratedNonfinite);
1268                }
1269                plane.sum_weight()
1270            } else {
1271                1.0
1272            };
1273            Ok(plane
1274                .read_sensitivity()?
1275                .iter()
1276                .map(|value| (*value / scale) as f32)
1277                .collect())
1278        }
1279        ProductRole::Sensitivity => Ok(plane
1280            .read_sensitivity()?
1281            .iter()
1282            .map(|value| *value as f32)
1283            .collect()),
1284        ProductRole::RestoredImage(
1285            casa_imaging_model::ProductTerm::Single | casa_imaging_model::ProductTerm::Taylor(0),
1286        ) => {
1287            if !valid {
1288                return Ok(invalid_residual());
1289            }
1290            let Some(beam) = restoring_beam.as_ref() else {
1291                return Err(ProductsError::BeamFitFailed(
1292                    "restoration requires a fitted beam for every valid plane".to_string(),
1293                ));
1294            };
1295            let model = model_real_plane(
1296                inputs.final_model(),
1297                domain_ordinal,
1298                plane.output_channel(),
1299                polarization,
1300                shape,
1301            )?;
1302            let cell_size = inputs.cell_size_rad_for_domain(member.axes().domain())?;
1303            let residual = normalize_domain_plane(
1304                &residual_real_plane(plane)?,
1305                required_normalization(member)?,
1306                plane,
1307            )?;
1308            let fitted_beam = fitted_beam.ok_or_else(|| {
1309                ProductsError::BeamFitFailed(
1310                    "restoration requires a fitted beam for every valid plane".to_string(),
1311                )
1312            })?;
1313            let residual =
1314                rescale_residual_to_beam(&residual, shape, cell_size, fitted_beam, *beam)?
1315                    .into_values();
1316            Ok(restore_model_plane(
1317                &model, residual, shape, beam, cell_size,
1318            ))
1319        }
1320        ProductRole::PrimaryBeam(
1321            casa_imaging_model::ProductTerm::Single | casa_imaging_model::ProductTerm::Taylor(0),
1322        ) => primary_beam_plane(primary_beam_model, inputs, member.axes().domain(), plane),
1323        ProductRole::PbCorrectedImage(
1324            casa_imaging_model::ProductTerm::Single | casa_imaging_model::ProductTerm::Taylor(0),
1325        ) => {
1326            if !valid {
1327                return Ok(invalid_residual());
1328            }
1329            let restored = restored_plane(
1330                member,
1331                inputs,
1332                plane,
1333                domain_ordinal,
1334                polarization,
1335                fitted_beam,
1336                restoring_beam,
1337            )?;
1338            let primary_beam =
1339                primary_beam_plane(primary_beam_model, inputs, member.axes().domain(), plane)?;
1340            correct_primary_beam(
1341                &restored,
1342                &primary_beam,
1343                inputs.problem().products().validity().primary_beam(),
1344            )
1345        }
1346        role => Err(ProductsError::UnsupportedProductRole {
1347            role,
1348            catalog: CONTINUUM_ALGORITHM_CATALOG_VERSION,
1349        }),
1350    }
1351}

1481fn product_plane_validity(
1482    rule: ProductValidityRule,
1483    plane: &FinalNormalPlaneReader<'_>,
1484    primary_beam_model: Option<AnalyticPrimaryBeamModel>,
1485    inputs: &ContinuumProductInputs<'_>,
1486    domain_role: &ImageDomainRole,
1487) -> Result<Vec<bool>, ProductsError> {
1488    let shape = plane.shape();
1489    match rule {
1490        ProductValidityRule::All => Ok(vec![true; shape[0] * shape[1]]),
1491        ProductValidityRule::FinalNormalState => {
1492            let valid = plane.validity() == SpectralChannelValidity::Valid
1493                && plane.sum_weight().is_finite()
1494                && plane.sum_weight() > 0.0;
1495            Ok(vec![valid; shape[0] * shape[1]])
1496        }
1497        ProductValidityRule::PrimaryBeam(policy) => {
1498            Ok(
1499                primary_beam_plane(primary_beam_model, inputs, domain_role, plane)?
1500                    .into_iter()
1501                    .map(|value| match policy.comparison() {
1502                        ProductSupportComparison::StrictlyGreater => value > policy.cutoff(),
1503                    })
1504                    .collect(),
1505            )
1506        }
1507        ProductValidityRule::Taylor(_) | ProductValidityRule::TaylorAndPrimaryBeam { .. } => {
1508            Err(ProductsError::UnsupportedProblem)
1509        }
1510    }
1511}

1547fn correct_primary_beam(
1548    values: &[f32],
1549    primary_beam: &[f32],
1550    policy: casa_imaging_model::PrimaryBeamValidityPolicy,
1551) -> Result<Vec<f32>, ProductsError> {
1552    if values.len() != primary_beam.len() {
1553        return Err(ProductsError::SourceLineageMismatch);
1554    }
1555    values
1556        .iter()
1557        .zip(primary_beam)
1558        .map(|(value, beam)| {
1559            let valid = match policy.comparison() {
1560                ProductSupportComparison::StrictlyGreater => *beam > policy.cutoff(),
1561            };
1562            if !valid {
1563                return Ok(0.0);
1564            }
1565            let corrected = *value / *beam;
1566            corrected
1567                .is_finite()
1568                .then_some(corrected)
1569                .ok_or(ProductsError::GeneratedNonfinite)
1570        })
1571        .collect()
1572}

1586fn model_real_plane(
1587    model: &ModelGeneration,
1588    domain_ordinal: usize,
1589    output_channel: usize,
1590    polarization: usize,
1591    plane_shape: [usize; 2],
1592) -> Result<Vec<f32>, ProductsError> {
1593    let [width, height] = plane_shape;
1594    if output_channel >= model.shape().coefficients()
1595        || polarization >= model.shape().polarizations()
1596        || model
1597            .shape()
1598            .domains()
1599            .get(domain_ordinal)
1600            .map(|shape| shape.pixels())
1601            != Some(plane_shape)
1602        || model.sample_count() != model.shape().sample_count()
1603    {
1604        return Err(ProductsError::SourceLineageMismatch);
1605    }
1606    // Canonical model order is y-major (`flat = y * W + x`); product planes
1607    // are stored x-major like every normal-state primitive.
1608    let mut plane = vec![0.0_f32; width * height];
1609    let samples = model.read_plane(domain_ordinal, output_channel, polarization)?;
1610    for y in 0..height {
1611        for x in 0..width {
1612            plane[x * height + y] = samples[y * width + x].value().value() as f32;
1613        }
1614    }
1615    Ok(plane)
1616}

1636fn scatter_image_polarization_plane<T: Copy>(
1637    payload: &mut [T],
1638    order: &AxisOrder,
1639    storage_shape: [usize; 4],
1640    polarization: usize,
1641    output_channel: usize,
1642    plane_shape: [usize; 2],
1643    plane: &[T],
1644) -> Result<(), ProductsError> {
1645    let [width, height] = plane_shape;
1646    if width == 0 || height == 0 || width.checked_mul(height) != Some(plane.len()) {
1647        return Err(ProductsError::SourceLineageMismatch);
1648    }
1649    let last = product_offset(
1650        order,
1651        storage_shape,
1652        width - 1,
1653        height - 1,
1654        polarization,
1655        output_channel,
1656    )?;
1657    if last >= payload.len() {
1658        return Err(ProductsError::SourceLineageMismatch);
1659    }
1660    let base = product_offset(order, storage_shape, 0, 0, polarization, output_channel)?;
1661    let (mut longitude_stride, mut latitude_stride) = (0, 0);
1662    let mut stride = 1usize;
1663    for (axis, extent) in order.positions().iter().zip(storage_shape).rev() {
1664        match axis {
1665            ImageAxis::DirectionLongitude => longitude_stride = stride,
1666            ImageAxis::DirectionLatitude => latitude_stride = stride,
1667            _ => {}
1668        }
1669        stride = stride
1670            .checked_mul(extent)
1671            .ok_or(ProductsError::SourceLineageMismatch)?;
1672    }
1673    for (x, column) in plane.chunks_exact(height).enumerate() {
1674        let start = base + x * longitude_stride;
1675        for (y, value) in column.iter().enumerate() {
1676            payload[start + y * latitude_stride] = *value;
1677        }
1678    }
1679    Ok(())
1680}

1711fn product_offset(
1712    order: &AxisOrder,
1713    storage_shape: [usize; 4],
1714    longitude: usize,
1715    latitude: usize,
1716    polarization: usize,
1717    spectral: usize,
1718) -> Result<usize, ProductsError> {
1719    let mut offset = 0usize;
1720    for (position, axis) in order.positions().iter().enumerate() {
1721        let coordinate = match axis {
1722            ImageAxis::DirectionLongitude => longitude,
1723            ImageAxis::DirectionLatitude => latitude,
1724            ImageAxis::Polarization => polarization,
1725            ImageAxis::Spectral => spectral,
1726        };
1727        let extent = storage_shape[position];
1728        if coordinate >= extent {
1729            return Err(ProductsError::SourceLineageMismatch);
1730        }
1731        offset = offset
1732            .checked_mul(extent)
1733            .and_then(|offset| offset.checked_add(coordinate))
1734            .ok_or(ProductsError::SourceLineageMismatch)?;
1735    }
1736    Ok(offset)
1737}

1739/// Resolve one member's compiled beam rule against the fitted generation beam.
1740fn beams_for_member(
1741    member: &PlannedMember,
1742    members: &[PlannedMember],
1743    rule: ProductBeamRule,
1744    fitted: &[Option<RestoringBeam>],
1745    restoring: &[Option<RestoringBeam>],
1746) -> Result<Box<[Option<RestoringBeam>]>, ProductsError> {
1747    let mut domain_count = 0;
1748    let mut domain_ordinal = None;
1749    for (index, planned) in members.iter().enumerate() {
1750        let role = planned.axes().domain();
1751        if members[..index]
1752            .iter()
1753            .any(|prior| prior.axes().domain() == role)
1754        {
1755            continue;
1756        }
1757        if role == member.axes().domain() {
1758            domain_ordinal = Some(domain_count);
1759        }
1760        domain_count += 1;
1761    }
1762    let domain_ordinal = domain_ordinal.ok_or(ProductsError::SourceLineageMismatch)?;
1763    Ok(resolve_beams(
1764        rule,
1765        domain_beam_slice(fitted, domain_ordinal, domain_count)?,
1766        domain_beam_slice(restoring, domain_ordinal, domain_count)?,
1767    ))
1768}
1769
1770fn domain_beam_slice(
1771    beams: &[Option<RestoringBeam>],
1772    domain_ordinal: usize,
1773    domain_count: usize,
1774) -> Result<&[Option<RestoringBeam>], ProductsError> {
1775    if beams.is_empty() {
1776        return Ok(beams);
1777    }
1778    if domain_count == 0 || beams.len() % domain_count != 0 {
1779        return Err(ProductsError::SourceLineageMismatch);
1780    }
1781    let channels = beams.len() / domain_count;
1782    let start = domain_ordinal
1783        .checked_mul(channels)
1784        .ok_or(ProductsError::SourceLineageMismatch)?;
1785    beams
1786        .get(start..start + channels)
1787        .ok_or(ProductsError::SourceLineageMismatch)
1788}

1790fn resolve_beams(
1791    rule: ProductBeamRule,
1792    fitted: &[Option<RestoringBeam>],
1793    restoring: &[Option<RestoringBeam>],
1794) -> Box<[Option<RestoringBeam>]> {
1795    match rule {
1796        ProductBeamRule::None => Box::new([]),
1797        ProductBeamRule::Restoring(RestoringBeamPolicy::None)
1798        | ProductBeamRule::Metadata(RestoringBeamPolicy::None) => Box::new([]),
1799        ProductBeamRule::Fitted => fitted.into(),
1800        ProductBeamRule::Restoring(RestoringBeamPolicy::PerPlane)
1801        | ProductBeamRule::Metadata(RestoringBeamPolicy::PerPlane) => restoring.into(),
1802        ProductBeamRule::Restoring(RestoringBeamPolicy::Common)
1803        | ProductBeamRule::Metadata(RestoringBeamPolicy::Common) => {
1804            vec![restoring.iter().flatten().next().copied()].into_boxed_slice()
1805        }
1806        ProductBeamRule::Inherit(_) => restoring.into(),
1807    }
1808}
