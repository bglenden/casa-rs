2//! Apply execution for the first calibration workflow slice.
3//!
4//! This module consumes an [`ApplyPlan`](crate::apply::ApplyPlan) and applies
5//! diagonal complex antenna-based calibration solutions to MS `DATA`,
6//! writing `CORRECTED_DATA` and optionally propagating calibration flags.
7//!
8//! Supported in this wave:
9//!
10//! - chained complex `CPARAM` tables
11//! - float `FPARAM` support for `K Jones` and `Kcross Jones` delay tables
12//! - legacy `BPOLY` bandpass tables expanded onto the target SPW grid
13//! - diagonal `G` / `T`-style application to `RR/RL/LR/LL` and `XX/XY/YX/YY`
14//! - all six apply modes, including strict and flag-only application
15//! - circular-feed and full four-correlation linear-feed parallactic angle
16//! - linearized and full-matrix instrumental polarization leakage
17//! - automatic `CORRECTED_DATA` creation when absent
18//!
19//! Explicitly deferred from this executor cut:
20//!
21//! - solver output beyond the existing planner surface
22

107fn calibration_profile_enabled() -> bool {
108    static ENABLED: OnceLock<bool> = OnceLock::new();
109    *ENABLED.get_or_init(|| match std::env::var("CASA_RS_CALIBRATION_PROFILE") {
110        Ok(value) => {
111            let trimmed = value.trim();
112            !trimmed.is_empty()
113                && trimmed != "0"
114                && !trimmed.eq_ignore_ascii_case("false")
115                && !trimmed.eq_ignore_ascii_case("off")
116        }
117        Err(_) => false,
118    })
119}

243struct CalibrationPerfTracer {
244    started_at: Option<Instant>,
245    json_file: Option<File>,
246    log_file: Option<File>,
247}
248
249impl CalibrationPerfTracer {
250    fn from_env() -> Self {
251        if std::env::var_os(PERF_ENV).is_none() {
252            return Self::disabled();
253        }
254        let output_dir = std::env::var_os(PERF_DIR_ENV)
255            .map(PathBuf::from)
256            .unwrap_or_else(|| PathBuf::from("/tmp"));
257        if create_dir_all(&output_dir).is_err() {
258            return Self::disabled();
259        }
260        let pid = std::process::id();
261        let json_path = output_dir.join(format!("casa-calibration-perf-{pid}.jsonl"));
262        let log_path = output_dir.join(format!("casa-calibration-perf-{pid}.log"));
263        let json_file = open_append_file(&json_path);
264        let log_file = open_append_file(&log_path);
265        if json_file.is_none() && log_file.is_none() {
266            return Self::disabled();
267        }
268        Self {
269            started_at: Some(Instant::now()),
270            json_file,
271            log_file,
272        }
273    }
274
275    const fn disabled() -> Self {
276        Self {
277            started_at: None,
278            json_file: None,
279            log_file: None,
280        }
281    }
282
283    fn is_enabled(&self) -> bool {
284        self.started_at.is_some()
285    }
286
287    fn monotonic_ns(&self) -> u64 {
288        self.started_at
289            .map(|started| started.elapsed().as_nanos() as u64)
290            .unwrap_or_default()
291    }
292
293    fn write_event(&mut self, event: &CalibrationPerfEvent) {
294        if let Some(file) = self.json_file.as_mut() {
295            let _ = serde_json::to_writer(&mut *file, event);
296            let _ = writeln!(file);
297            let _ = file.flush();
298        }
299        if let Some(file) = self.log_file.as_mut() {
300            let _ = writeln!(
301                file,
302                "[+{:>7} ms] kind={:?} rows={} total_ms={:.2} planning_ms={:.2} row_field_index_ms={:.2} row_read_ms={:.2} row_read_overhead_ms={:.2} row_write_ms={:.2} save_ms={:.2} unattributed_ms={:.2}",
303                event.monotonic_ns / 1_000_000,
304                event.kind,
305                event.selected_row_count,
306                event.total_ns as f64 / 1_000_000.0,
307                event.planning_ns as f64 / 1_000_000.0,
308                event.row_field_index_lookup_ns as f64 / 1_000_000.0,
309                event.row_read_total_ns as f64 / 1_000_000.0,
310                event.row_read_overhead_ns as f64 / 1_000_000.0,
311                event.row_writeback_ns as f64 / 1_000_000.0,
312                event.save_ns as f64 / 1_000_000.0,
313                event.execute_apply_plan_unattributed_ns as f64 / 1_000_000.0
314            );
315            let _ = file.flush();
316        }
317    }

512/// Errors returned while executing an apply plan.
513#[derive(Debug, Error)]
514pub enum ApplyExecutionError {
515    /// Planning failed before any mutation occurred.
516    #[error(transparent)]
517    Plan(Box<ApplyPlanError>),
518
519    /// Opening the MeasurementSet failed.
520    #[error("failed to open MeasurementSet {path}: {source}")]
521    OpenMeasurementSet {
522        /// Path that was being opened.
523        path: String,
524        /// Underlying MS error.
525        #[source]
526        source: MsError,
527    },
528
529    /// Saving the flag backup failed; nothing was written to the MeasurementSet.
530    #[error("failed to back up the flags of {path}: {source}")]
531    FlagBackup {
532        /// MeasurementSet path.
533        path: String,
534        /// Underlying flag-version error.
535        #[source]
536        source: Box<FlaggingError>,
537    },
538
539    /// Opening an auxiliary calibration subtable failed.
540    #[error("failed to open calibration subtable {subtable} in {path}: {source}")]
541    OpenCalibrationAuxiliaryTable {
542        /// Calibration-table path.
543        path: String,
544        /// Auxiliary subtable name.
545        subtable: &'static str,
546        /// Underlying table error.
547        #[source]
548        source: TableError,
549    },
550
551    /// The on-disk calibration metadata is outside the supported executor surface.
552    #[error("unsupported calibration metadata in {path}: {reason}")]
553    UnsupportedCalibrationTable {
554        /// Calibration-table path.
555        path: String,
556        /// Human-readable explanation.
557        reason: String,
558    },
559
560    /// The row/table combination requested an interpolation mode not yet supported.
561    #[error(
562        "interpolation mode {interp:?} is not supported for calibration table {path}: {reason}"
563    )]
564    UnsupportedInterpolation {
565        /// Table path.
566        path: String,
567        /// Interpolation mode.
568        interp: ApplyInterpolationMode,
569        /// Additional context.
570        reason: String,
571    },
572
573    /// The executor only supports diagonal correlation layouts for now.
574    #[error(
575        "unsupported correlation layout for DATA_DESC_ID {data_desc_id}: {correlation_types:?}"
576    )]
577    UnsupportedCorrelationLayout {
578        /// Data description id.
579        data_desc_id: i32,
580        /// Correlation names derived from POLARIZATION.CORR_TYPE.
581        correlation_types: Vec<String>,
582    },
583
584    /// Linear-feed `parang` requires all four correlation products.
585    #[error(
586        "parallactic-angle application requires circular-feed correlations or the full XX/XY/YX/YY linear-feed layout for DATA_DESC_ID {data_desc_id}: {correlation_types:?}"
587    )]
588    UnsupportedParallacticAngleBasis {
589        /// Data description id.
590        data_desc_id: i32,
591        /// Correlation names derived from POLARIZATION.CORR_TYPE.
592        correlation_types: Vec<String>,
593    },

595    /// The caltable payload shape is outside the supported diagonal surface.

600        /// Shape discovered in the caltable.

602    },

604    /// Creating or populating `CORRECTED_DATA` failed.

609        /// Underlying table error.

660fn opacity_gain(tau: f32, elevation: f64) -> f32 {
661    let zenith_angle = std::f64::consts::FRAC_PI_2 - elevation;
662    if zenith_angle < std::f64::consts::FRAC_PI_2 {
663        (-f64::from(tau) / zenith_angle.cos()).exp().sqrt() as f32
664    } else {
665        1.0
666    }
667}
