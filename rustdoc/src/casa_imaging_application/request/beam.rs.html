12/// `none` (each plane's fitted beam), `common`, or a beam's
13/// `[major, minor, angle]`, read as `SynthesisParamsDeconv::fromRecord`
14/// reads `restoringbeam`'s quantities: each with casacore's quantity
15/// reader, as casacore's `GaussianBeam::setMajorMinor` and `setPA` accept
16/// them (angular units, axes neither negative nor inverted). A zero axis is
17/// casacore's null beam, which restores each plane with its fitted beam.
18pub(super) fn restoring_beam<'de, D: Deserializer<'de>>(
19    deserializer: D,
20) -> Result<RestoringBeamPolicy, D::Error> {
21    match Value::deserialize(deserializer)? {
22        Value::String(text) if unset(&text) => Ok(RestoringBeamPolicy::PerPlane),
23        Value::String(text) if text == "common" => Ok(RestoringBeamPolicy::Common),
24        Value::Array(items) => {
25            let texts = items
26                .iter()
27                .map(|item| item.as_str().ok_or_else(|| unexpected(item, "text")))
28                .collect::<Result<Vec<_>, D::Error>>()?;
29            let [major, minor, angle] = texts.as_slice() else {
30                return Err(D::Error::custom(format!(
31                    "restoringbeam {texts:?}: expected its major and minor axes and angle"
32                )));
33            };
34            given_restoring_beam([major, minor, angle]).map_err(|_| {
35                D::Error::custom(format!(
36                    "Cannot construct a restoringbeam from supplied parameters {texts:?}: give \
37                     angular axes, the major axis at least the minor, and an angular position \
38                     angle, as CASA's tclean requires"
39                ))
40            })
41        }
42        value => Err(unexpected(
43            &value,
44            "none, common or a beam's axes and angle",
45        )),
46    }
47}

49/// A beam's `[major, minor, angle]` quantities, or the fitted beams for
50/// casacore's null beam.
51fn given_restoring_beam(texts: [&str; 3]) -> Result<RestoringBeamPolicy, ()> {
52    let radians = Unit::from_str("rad").expect("a casacore unit");
53    let [major, minor, angle] = texts.map(|text| {
54        Quantity::read(text)
55            .ok()
56            .and_then(|quantity| quantity.get_value_in(&radians).ok())
57            .filter(|value| value.is_finite())
58            .ok_or(())
59    });
60    let [major, minor, angle] = [major?, minor?, angle?];
61    if major < 0.0 || minor < 0.0 || major < minor {
62        return Err(());
63    }
64    if major == 0.0 || minor == 0.0 {
65        return Ok(RestoringBeamPolicy::PerPlane);
66    }
67    FixedRestoringBeam::new(major, minor, angle)
68        .map(RestoringBeamPolicy::Fixed)
69        .map_err(|_| ())
70}
