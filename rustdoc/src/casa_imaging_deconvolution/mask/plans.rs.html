4//! materializes from them: a static mask (every pixel, pixel boxes or a
5//! reprojected mask image, its planes as CASA expands it), such a mask cut

7//! boundary, or CASA's automatic mask of every channel and polarization,

25/// Deferred mask construction evaluated at an exact major-cycle boundary.
26#[derive(Debug, Clone)]
27pub enum ReconstructionMaskPlan {
28    /// Admit every model-support pixel.
29    FullPlane {
30        /// Target model direction coordinate.
31        coordinate: DirectionCoordinateSpec,
32    },
33    /// Admit the union of explicit target-grid boxes.
34    Boxes {
35        /// Target model direction coordinate.
36        coordinate: DirectionCoordinateSpec,
37        /// Inclusive target-grid regions.
38        boxes: Vec<MaskBox>,
39    },
40    /// Admit exact target-grid support reprojected from a source mask's WCS.
41    Reprojected {
42        /// Target model direction coordinate.
43        coordinate: DirectionCoordinateSpec,
44        /// The planes the support holds, shared along an axis of one.
45        planes: MaskPlanes,
46        /// Canonical target-grid support produced by exact WCS reprojection,
47        /// plane after plane (channel-major), each x-major; every mask
48        /// materialized from the plan shares it.
49        support: Arc<[bool]>,
50    },
51    /// A static mask cut to the primary beam at the run's first boundary:
52    /// tclean's `usemask='pb'` (every pixel) and a named user mask under a
53    /// `pbmask` above zero (`SDMaskHandler::makePBMask`), each kept in
54    /// every channel and polarization where that plane's `.pb` exceeds
55    /// `level`. It holds a plane per channel and polarization, and the next
56    /// boundary keeps the mask it made, as tclean makes it once
57    /// (`SynthesisDeconvolver::setupMask`).
58    WithinPrimaryBeam {
59        /// The static mask the beam cuts: every pixel, boxes or a
60        /// reprojected support.
61        within: Box<ReconstructionMaskPlan>,
62        /// tclean's `pbmask`, in `[0, 1)`: the mask keeps the pixels whose
63        /// `.pb` exceeds it.
64        level: f64,
65    },
66    /// Generate CASA auto-multithreshold support of every plane from the
67    /// current Normal State.
68    AutoMultithresh {
69        /// Target model direction coordinate.
70        coordinate: DirectionCoordinateSpec,
71        /// Thresholding, pruning, smoothing, and growth controls.
72        controls: AutoMultithreshControls,
73        /// Number of completed major cycles before this boundary.
74        completed_major_cycles: usize,
75        /// Whether the previous cycle reached its stopping threshold.
76        cycle_threshold_reached: bool,
77        /// Prior immutable mask generation, when another cycle already ran.
78        previous: Option<Box<ReconstructionMask>>,
79        /// The channels CASA's channel-stop rule froze
80        /// (`SDMaskHandler::skipChannels`); empty before the first boundary.
81        stopped_channels: Box<[bool]>,
82    },
83}

85impl ReconstructionMaskPlan {
86    /// A static `within` cut to the primary beam at `level`
87    /// ([`Self::WithinPrimaryBeam`]).
88    ///
89    /// # Errors
90    ///
91    /// [`MaskError::InvalidPrimaryBeamMask`] when `within` is not a static
92    /// mask or `level` is not in `[0, 1)`, which tclean refuses for
93    /// `pbmask`.
94    pub fn within_primary_beam(within: Self, level: f64) -> Result<Self, MaskError> {
95        let plan = Self::WithinPrimaryBeam {
96            within: Box::new(within),
97            level,
98        };
99        plan.validate()?;
100        Ok(plan)
101    }
102
103    /// Check a [`Self::WithinPrimaryBeam`] plan: it cuts a static mask at a
104    /// level in `[0, 1)`.
105    fn validate(&self) -> Result<(), MaskError> {
106        match self {
107            Self::WithinPrimaryBeam { within, level }
108                if !valid_level(*level)
109                    || !matches!(
110                        **within,
111                        Self::FullPlane { .. } | Self::Boxes { .. } | Self::Reprojected { .. }
112                    ) =>
113            {
114                Err(MaskError::InvalidPrimaryBeamMask)
115            }
116            _ => Ok(()),
117        }
118    }
119
120    /// Advance live automasking to the next major-cycle boundary from the
121    /// mask `current` and its automatic evidence. A mask cut to the primary
122    /// beam keeps `current`, whose support it shares.
123    #[must_use]
124    pub fn next_cycle(
125        &self,
126        current: &ReconstructionMask,
127        evidence: Option<&AutoMaskEvidence>,
128        completed_major_cycles: usize,
129        cycle_threshold_reached: bool,
130    ) -> Self {
131        match self {
132            Self::AutoMultithresh {
133                coordinate,
134                controls,
135                ..
136            } => Self::AutoMultithresh {
137                coordinate: *coordinate,
138                controls: *controls,
139                completed_major_cycles,
140                cycle_threshold_reached,
141                previous: Some(Box::new(current.clone())),
142                stopped_channels: evidence
143                    .map(|evidence| evidence.stopped_channels().collect())
144                    .unwrap_or_default(),
145            },
146            Self::WithinPrimaryBeam { .. } => Self::Reprojected {
147                coordinate: current.coordinate,
148                planes: current.planes,
149                support: Arc::clone(&current.support),
150            },
151            other => other.clone(),
152        }
153    }
154
155    /// The planes this plan's masks hold for an image of `channels` by
156    /// `polarizations` planes: an automatic mask, or one cut to the
157    /// primary beam, has one per plane.
158    #[must_use]
159    pub fn mask_planes(&self, channels: usize, polarizations: usize) -> MaskPlanes {
160        match self {
161            Self::FullPlane { .. } | Self::Boxes { .. } => MaskPlanes::SHARED,
162            Self::Reprojected { planes, .. } => *planes,
163            Self::AutoMultithresh { .. } | Self::WithinPrimaryBeam { .. } => {
164                MaskPlanes::new(channels, polarizations)
165                    .expect("an image has at least one channel and one polarization")
166            }
167        }
168    }
169
170    /// Materialize the mask of domain `domain`, `shape` pixels with
171    /// `planes` planes, at this boundary.
172    fn materialize<E>(
173        &self,
174        domain: usize,
175        (shape, planes): ([usize; 2], MaskPlanes),
176        sources: &MaskSources<'_, E>,
177    ) -> Result<(ReconstructionMask, Option<AutoMaskEvidence>), E>
178    where
179        E: From<MaskError> + Send,
180    {
181        let mask = match self {
182            Self::FullPlane { coordinate } => ReconstructionMask::full_plane(*coordinate, shape)?,
183            Self::Boxes { coordinate, boxes } => {
184                ReconstructionMask::from_boxes(*coordinate, shape, boxes.iter().copied())?
185            }
186            Self::Reprojected {
187                coordinate,
188                planes: held,
189                support,
190            } => {
191                if !held.covers(planes.channels(), planes.polarizations()) {
192                    return Err(MaskError::ShapeMismatch.into());
193                }
194                ReconstructionMask::from_plane_support(
195                    *coordinate,
196                    shape,
197                    *held,
198                    Arc::clone(support),
199                )?
200            }
201            Self::WithinPrimaryBeam { within, level } => {
202                self.validate()?;
203                let static_sources = MaskSources {
204                    auto: None,
205                    beam: None,
206                };
207                let (within, _) = within.materialize(domain, (shape, planes), &static_sources)?;
208                cut_to_beam(&within, (domain, planes), *level, sources.beam)?
209            }
210            Self::AutoMultithresh {
211                coordinate,
212                controls,
213                completed_major_cycles,
214                cycle_threshold_reached,
215                previous,
216                stopped_channels,
217            } => {
218                let source = sources.auto.ok_or(MaskError::InvalidBeamArea)?;
219                let state = AutoMaskState {
220                    coordinate: *coordinate,
221                    controls,
222                    completed_major_cycles: *completed_major_cycles,
223                    threshold_reached: *cycle_threshold_reached,
224                    previous: previous.as_deref(),
225                    stopped_channels,
226                };
227                let (mask, evidence) = auto_mask_domain(&state, domain, shape, planes, source)?;
228                return Ok((mask, Some(evidence)));
229            }
230        };
231        Ok((mask, None))
232    }
233}

267/// The clean mask tclean's robust noise estimate (`fastnoise=False`)
268/// leaves out of one plane's statistics at a boundary
269/// (`SDMaskHandler::calcRobustImageStatistics`).
270#[derive(Debug, Clone, Copy, PartialEq, Eq)]
271pub enum NoiseMaskExclusion<'a> {
272    /// No mask yet: Chauvenet statistics of the valid pixels. tclean's
273    /// first noise estimate reads the mask image before it is filled.
274    Unmasked,
275    /// The plane's clean mask, in canonical x-major order: the classical
276    /// statistics outside it, or Chauvenet's when it is empty or `full`.
277    Mask {
278        /// The plane's support.
279        support: &'a [bool],
280        /// Whether the mask covers the whole first plane CASA's statistics
281        /// read it from: the automatic mask's channel block, or else the
282        /// plane itself.
283        full: bool,
284    },
285    /// A plane whose automatic mask stopped evolving: tclean records its
286    /// statistics as zero.
287    Stopped,
288}

332/// The channel and polarization planes of `normal`: a Taylor family's
333/// terms share one plane per polarization.
334fn normal_planes(normal: &FinalNormalState) -> MaskPlanes {
335    let channels = if normal.basis() == NormalBasis::Taylor {
336        1
337    } else {
338        normal.channel_count()
339    };
340    MaskPlanes::new(channels, normal.polarization_count())
341        .expect("a normal state has at least one channel and one polarization")
342}

344impl ImageDomainReconstructionMaskPlans {
345    /// Adopt one non-empty plan for each image domain in compiled order.
346    ///
347    /// # Errors
348    ///
349    /// No plan, or a mask cut to the primary beam that does not cut a
350    /// static mask at a level in `[0, 1)`.
351    pub fn new(plans: impl IntoIterator<Item = ReconstructionMaskPlan>) -> Result<Self, MaskError> {
352        let plans = plans.into_iter().collect::<Vec<_>>().into_boxed_slice();
353        if plans.is_empty() {
354            return Err(MaskError::DomainCardinalityMismatch);
355        }
356        for plan in &plans {
357            plan.validate()?;
358        }
359        Ok(Self {
360            plans,
361            completed_major_cycles: 0,
362        })
363    }

365    /// Return the number of canonical domain plans.
366    #[must_use]
367    pub const fn len(&self) -> usize {
368        self.plans.len()
369    }
370
371    /// Return whether no domain plans are present (always false for valid values).
372    #[must_use]
373    pub const fn is_empty(&self) -> bool {
374        self.plans.is_empty()
375    }

377    /// Major cycles completed before the boundary these plans are for: zero
378    /// at a run's first boundary, before any minor cycle (CASA's `iterdone`
379    /// of 0).
380    #[must_use]
381    pub const fn completed_major_cycles(&self) -> usize {
382        self.completed_major_cycles
383    }

385    /// Borrow the primary-domain plan.
386    #[must_use]
387    pub const fn primary(&self) -> &ReconstructionMaskPlan {
388        &self.plans[0]
389    }

391    /// Whether some domain's mask is CASA's automatic mask.

393    pub fn automatic(&self) -> bool {
394        self.plans
395            .iter()
396            .any(|plan| matches!(plan, ReconstructionMaskPlan::AutoMultithresh { .. }))

399    /// Whether some domain's mask is cut to the primary beam at this
400    /// boundary, so that [`Self::materialize`] reads its planes' `.pb`.
401    #[must_use]
402    pub fn reads_primary_beam(&self) -> bool {
403        self.plans
404            .iter()
405            .any(|plan| matches!(plan, ReconstructionMaskPlan::WithinPrimaryBeam { .. }))
406    }

408    /// Advance every domain plan to the next shared major-cycle boundary
409    /// from the masks `current` and their automatic `evidence`, both in
410    /// domain order.
411    pub fn next_cycle(
412        &self,
413        current: &ImageDomainReconstructionMasks,
414        evidence: &[Option<AutoMaskEvidence>],
415        completed_major_cycles: usize,
416        cycle_threshold_reached: bool,
417    ) -> Result<Self, MaskError> {
418        if self.plans.len() != current.len() || self.plans.len() != evidence.len() {
419            return Err(MaskError::DomainCardinalityMismatch);
420        }
421        let plans =
422            self.plans
423                .iter()
424                .zip(current.iter())
425                .zip(evidence)
426                .map(|((plan, mask), evidence)| {
427                    plan.next_cycle(
428                        mask,
429                        evidence.as_ref(),
430                        completed_major_cycles,
431                        cycle_threshold_reached,
432                    )
433                });
434        let mut next = Self::new(plans)?;
435        next.completed_major_cycles = completed_major_cycles;
436        Ok(next)
437    }

439    /// What tclean's robust noise estimate leaves out of the statistics of
440    /// domain `ordinal`'s plane `channel` and `polarization` at this
441    /// boundary, `current` being the masks materialized for it: an
442    /// automatic mask's inherited support, or a static mask once the first
443    /// boundary has passed.
444    #[must_use]
445    pub fn noise_exclusion<'a>(
446        &'a self,
447        ordinal: usize,
448        (channel, polarization): (usize, usize),
449        current: &'a ImageDomainReconstructionMasks,
450    ) -> NoiseMaskExclusion<'a> {
451        let full = |plane: &[bool]| plane.iter().all(|set| *set);
452        match self.plans.get(ordinal) {
453            Some(ReconstructionMaskPlan::AutoMultithresh {
454                stopped_channels, ..
455            }) if stopped_channels.get(channel).copied().unwrap_or(false) => {
456                NoiseMaskExclusion::Stopped
457            }
458            Some(ReconstructionMaskPlan::AutoMultithresh { previous, .. }) => previous
459                .as_deref()
460                .and_then(|mask| {
461                    let support = mask.plane(channel, polarization)?;
462                    // The statistics of a channel block read the mask's
463                    // first plane in it (`calcRobustImageStatistics`).
464                    let first = channel_blocks(mask.planes().channels(), mask.shape())
465                        .into_iter()
466                        .find(|block| block.contains(&channel))
467                        .map_or(0, |block| block.start);
468                    Some(NoiseMaskExclusion::Mask {
469                        support,
470                        full: mask.plane(first, 0).is_some_and(full),
471                    })
472                })
473                .unwrap_or(NoiseMaskExclusion::Unmasked),
474            _ if self.completed_major_cycles == 0 => NoiseMaskExclusion::Unmasked,
475            _ => current
476                .get(ordinal)
477                .and_then(|mask| mask.plane(channel, polarization))
478                .map_or(NoiseMaskExclusion::Unmasked, |support| {
479                    NoiseMaskExclusion::Mask {
480                        support,
481                        full: full(support),
482                    }
483                }),
484        }
485    }

487    /// Heap bytes the masks materialized from these plans hold against
488    /// domains of the shapes `domain_shape` gives by ordinal, with
489    /// `channels` by `polarizations` planes: each domain's support, one
490    /// per plane its plan holds ([`ReconstructionMaskPlan::mask_planes`]),
491    /// and an automatic mask's summary row per plane.
492    #[must_use]
493    pub fn mask_bytes_for(
494        &self,
495        domain_shape: impl Fn(usize) -> Option<[usize; 2]>,
496        (channels, polarizations): (usize, usize),
497    ) -> u64 {
498        // The returned masks and evidence also hold their domain entries,
499        // and each Arc support owns its strong and weak counters.
500        let containers = self.plans.len() as u64
501            * (size_of::<ReconstructionMask>()
502                + size_of::<Option<AutoMaskEvidence>>()
503                + 2 * size_of::<usize>()) as u64;
504        containers
505            + self
506                .plans
507                .iter()
508                .enumerate()
509                .map(|(ordinal, plan)| {
510                    let cells =
511                        domain_shape(ordinal).map_or(0, |[width, height]| (width * height) as u64);
512                    let planes = plan.mask_planes(channels, polarizations).count() as u64;
513                    let rows = match plan {
514                        ReconstructionMaskPlan::AutoMultithresh { .. } => {
515                            planes * size_of::<AutoMultithreshEvidence>() as u64
516                        }
517                        _ => 0,
518                    };
519                    cells * planes * size_of::<bool>() as u64 + rows
520                })
521                .sum::<u64>()
522    }

524    /// Heap bytes [`Self::materialize`] holds at most against `normal`
525    /// with `workers` channels masked at once.
526    #[must_use]
527    pub fn materialize_bytes(&self, normal: &FinalNormalState, workers: usize) -> u64 {
528        let planes = normal_planes(normal);
529        self.materialize_bytes_for(
530            |ordinal| normal.domain_shape(ordinal),
531            (planes.channels(), planes.polarizations()),
532            workers,
533        )
534    }

536    /// [`Self::materialize_bytes`] against image domains of the shapes
537    /// `domain_shape` gives by ordinal, with `channels` by `polarizations`
538    /// planes: what a run plans before its first major cycle forms the
539    /// Normal State (#737). Each domain's new masks
540    /// ([`Self::mask_bytes_for`]) and, at most, one domain's working
541    /// memory. For an automatic mask, on each of the `workers` channels
542    /// masked at once, what one plane's mask holds
543    /// ([`auto_multithresh_bytes`]), the plane's read (as `Complex64`) and
544    /// the model support it reads, and the channel's new supports; and every
545    /// plane's PSF measurements, which the mask lists. For a mask cut to the
546    /// primary beam, the static plane it cuts and one plane's beam pixels;
547    /// the owner charges what it forms a beam plane from.
548    #[must_use]
549    pub fn materialize_bytes_for(
550        &self,
551        domain_shape: impl Fn(usize) -> Option<[usize; 2]>,
552        (channels, polarizations): (usize, usize),
553        workers: usize,
554    ) -> u64 {
555        let working = self
556            .plans
557            .iter()
558            .enumerate()
559            .map(|(ordinal, plan)| {
560                let Some(shape) = domain_shape(ordinal) else {
561                    return 0;
562                };
563                let cells = (shape[0] * shape[1]) as u64;
564                match plan {
565                    ReconstructionMaskPlan::AutoMultithresh { .. } => {}
566                    ReconstructionMaskPlan::WithinPrimaryBeam { .. } => {
567                        return 2 * cells * size_of::<bool>() as u64;
568                    }
569                    _ => return 0,
570                }
571                let read = size_of::<num_complex::Complex64>() as u64
572                    + size_of::<ModelSample>() as u64
573                    + polarizations as u64 * size_of::<bool>() as u64;
574                // Every plane's PSF measurements and the empty plane a mask
575                // inherits before the first boundary, beside the workers'.
576                let measurements = (channels * polarizations) as u64
577                    * size_of::<Option<PsfSummary>>() as u64
578                    + cells * size_of::<bool>() as u64;
579                measurements
580                    + block_beam_bytes(channels, polarizations, shape)
581                    + workers.min(channels).max(1) as u64
582                        * (auto_multithresh_bytes(shape) + cells * read)
583            })
584            .max()
585            .unwrap_or(0);
586        self.mask_bytes_for(&domain_shape, (channels, polarizations)) + working
587    }

589    /// Materialize every domain's mask against one shared Normal State;
590    /// `sources` reads the planes an automatic mask is made from and the
591    /// `.pb` a mask cut to the primary beam reads, each required by such a
592    /// mask. Channels of an automatic mask are masked side by side on the
593    /// current rayon pool.
594    ///
595    /// # Errors
596    ///
597    /// A plan does not fit the state, a static mask is invalid, the
598    /// automatic mask fails or cannot read a plane, or a mask cut to the
599    /// primary beam has no `.pb` to read ([`MaskError::NoPrimaryBeam`]).
600    pub fn materialize<E>(
601        &self,
602        base: &ModelGeneration,
603        normal: &FinalNormalState,
604        sources: &MaskSources<'_, E>,
605    ) -> Result<ImageDomainMaskMaterialization, E>
606    where
607        E: From<MaskError> + Send,
608    {
609        if self.plans.len() != normal.domain_count()
610            || self.plans.len() != base.shape().domains().len()
611        {
612            return Err(MaskError::DomainCardinalityMismatch.into());
613        }
614        let planes = normal_planes(normal);
615        let mut masks = Vec::with_capacity(self.plans.len());
616        let mut evidence = Vec::with_capacity(self.plans.len());
617        for (ordinal, plan) in self.plans.iter().enumerate() {
618            let shape = normal
619                .domain_shape(ordinal)
620                .ok_or(MaskError::DomainCardinalityMismatch)?;
621            let (mask, auto_evidence) = plan.materialize(ordinal, (shape, planes), sources)?;
622            masks.push(mask);
623            evidence.push(auto_evidence);
624        }
625        Ok(ImageDomainMaskMaterialization {
626            masks: ImageDomainReconstructionMasks::new(masks)?,
627            auto_mask_evidence: evidence.into_boxed_slice(),
628        })
629    }
630}

645impl ImageDomainReconstructionMasks {

648        if masks.is_empty() {

651        Ok(Self { masks })

654    /// Return the canonical domain count.
655    #[must_use]
656    pub const fn len(&self) -> usize {
657        self.masks.len()
658    }
659
660    /// Return whether no domain supports are present (always false for valid values).
661    #[must_use]
662    pub const fn is_empty(&self) -> bool {
663        self.masks.is_empty()
664    }
665
666    /// Borrow the primary-domain support.
667    #[must_use]
668    pub const fn primary(&self) -> &ReconstructionMask {
669        &self.masks[0]
670    }
671
672    /// Borrow one support by canonical domain ordinal.
673    #[must_use]
674    pub fn get(&self, domain_ordinal: usize) -> Option<&ReconstructionMask> {
675        self.masks.get(domain_ordinal)
676    }
677
678    /// Iterate supports in canonical domain order.
679    pub fn iter(&self) -> impl ExactSizeIterator<Item = &ReconstructionMask> {
680        self.masks.iter()
681    }

718    fn same(left: &ReconstructionMask, right: &[bool]) -> bool {
719        std::ptr::eq(left.support().as_ptr(), right.as_ptr())
720    }

722    /// A generation of masks is held once, whoever holds it: the plan the
723    /// next boundary evolves from, the run's outcome (a clone) and the
724    /// minor cycle share its support, and a static plan's masks share the
725    /// plan's. Each holder once copied a cube's every plane.
726    #[test]
727    fn every_holder_shares_a_generation_of_masks() {
728        let planes = MaskPlanes::new(3, 2).unwrap();
729        let used =
730            ReconstructionMask::from_plane_support(grid(), [2, 2], planes, vec![true; 24]).unwrap();
731        let masks = ImageDomainReconstructionMasks::new([used]).unwrap();
732        let plans =
733            ImageDomainReconstructionMaskPlans::new([ReconstructionMaskPlan::AutoMultithresh {
734                coordinate: grid(),
735                controls: controls(),
736                completed_major_cycles: 0,
737                cycle_threshold_reached: false,
738                previous: None,
739                stopped_channels: Box::new([]),
740            }])
741            .unwrap();
742        let next = plans.next_cycle(&masks, &[None], 1, false).unwrap();
743        let ReconstructionMaskPlan::AutoMultithresh {
744            previous: Some(previous),
745            ..
746        } = next.primary()
747        else {
748            panic!("the next boundary evolves from the masks used");
749        };
750        assert!(same(previous, masks.primary().support()));
751        let outcome = masks.clone();
752        assert!(same(outcome.primary(), masks.primary().support()));
753
754        let support: Arc<[bool]> = vec![true, false, false, true].into();
755        let plan = ReconstructionMaskPlan::Reprojected {
756            coordinate: grid(),
757            planes: MaskPlanes::SHARED,
758            support: Arc::clone(&support),
759        };
760        let (made, _) = plan
761            .materialize::<MaskError>(0, ([2, 2], planes), &MaskSources::default())
762            .unwrap();
763        assert!(same(&made, &support));
764        assert!(same(&made.clone(), &support));
765    }

767    /// A mask cut to the primary beam cuts a static mask at a `pbmask` in
768    /// `[0, 1)`, holds a plane per channel and polarization, and is kept by
769    /// the next boundary, which shares its support (tclean makes it once).
770    #[test]
771    fn a_mask_cut_to_the_beam_is_made_once_and_kept() {
772        let full = || ReconstructionMaskPlan::FullPlane { coordinate: grid() };
773        let auto = ReconstructionMaskPlan::AutoMultithresh {
774            coordinate: grid(),
775            controls: controls(),
776            completed_major_cycles: 0,
777            cycle_threshold_reached: false,
778            previous: None,
779            stopped_channels: Box::new([]),
780        };
781        for (within, level) in [(full(), 1.0), (full(), -0.1), (auto, 0.2)] {
782            assert_eq!(
783                ReconstructionMaskPlan::within_primary_beam(within, level).unwrap_err(),
784                MaskError::InvalidPrimaryBeamMask
785            );
786        }
787        let plan = ReconstructionMaskPlan::within_primary_beam(full(), 0.2).unwrap();
788        let planes = MaskPlanes::new(3, 2).unwrap();
789        assert_eq!(plan.mask_planes(3, 2), planes);
790        let beam = |key: MaskPlaneKey, _: f32| {
791            Ok::<_, MaskError>(Some(vec![key.channel == 1, true, false, true]))
792        };
793        let sources = MaskSources {
794            auto: None,
795            beam: Some(&beam),
796        };
797        let (made, _) = plan
798            .materialize::<MaskError>(0, ([2, 2], planes), &sources)
799            .unwrap();
800        assert_eq!(made.plane(1, 1).unwrap(), [true, true, false, true]);
801        assert_eq!(made.plane(2, 0).unwrap(), [false, true, false, true]);
802        let next = plan.next_cycle(&made, None, 1, false);
803        let (kept, _) = next
804            .materialize::<MaskError>(0, ([2, 2], planes), &MaskSources::default())
805            .unwrap();
806        assert!(same(&kept, made.support()));
807        // Without a `.pb`, tclean's makePBMask refuses the mask.
808        assert_eq!(
809            plan.materialize::<MaskError>(0, ([2, 2], planes), &MaskSources::default())
810                .unwrap_err(),
811            MaskError::NoPrimaryBeam
812        );
813    }
