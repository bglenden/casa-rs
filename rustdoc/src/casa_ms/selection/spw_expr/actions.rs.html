11/// What one expression resolves to, before it becomes a
12/// [`super::SpwSelection`].
13pub(super) struct Resolved {
14    pub(super) spw_ids: Vec<usize>,
15    pub(super) data_description_ids: Vec<usize>,
16    pub(super) channels: Vec<ChannelRange>,
17    pub(super) rows: Option<Vec<usize>>,
18    pub(super) warnings: Vec<String>,
19}

21/// The reports an `MSSelectionErrorHandler` collects during the parse
22/// (`MSSelectionErrorHandler.cc:56-77`). The handler `MSSelection`
23/// installs is a base-class clone of `MSSSpwErrorHandler`, which does not
24/// override `clone()`, so a report refuses the expression once parsing
25/// ends (`MSSelection.cc:528-536`, `859-883`), and its message replaces
26/// any error that stopped the parse.
27#[derive(Default)]
28struct Reports {
29    message: Option<String>,
30    tokens: Vec<String>,
31}
32
33impl Reports {
34    fn report(&mut self, message: &str, token: String) {
35        self.message.get_or_insert_with(|| message.to_string());
36        self.tokens.push(token);
37    }
38
39    /// `constructMessage`: the first message, then every token followed by
40    /// a space.
41    fn refusal(&self) -> Option<String> {
42        self.message.as_ref().map(|first| {
43            let mut message = first.clone();
44            for token in &self.tokens {
45                message.push_str(token);
46                message.push(' ');
47            }
48            message.trim_end().to_string()
49        })
50    }
51
52    fn check_empty(&mut self, ids: &[usize], message: &str, token: String) {
53        if ids.is_empty() {
54            self.report(
55                &format!("Spw Expression: {message}"),
56                format!("\"{token}\""),
57            );
58        }
59    }
60}

62/// Resolve `expression` against the SPECTRAL_WINDOW and DATA_DESCRIPTION
63/// rows, or return CASA's refusal.
64pub(super) fn resolve(
65    expression: &Expression,
66    windows: &[SpectralWindowRow<'_>],
67    descriptions: &[DataDescriptionRow],
68) -> Result<Resolved, String> {
69    let mut reports = Reports::default();
70    let resolved = resolve_specs(expression, windows, descriptions, &mut reports);
71    if let Some(refusal) = reports.refusal() {
72        return Err(refusal);
73    }
74    let resolved = resolved?;
75    if !expression.parenthesized && resolved.spw_ids.is_empty() {
76        return Err("No Spw ID(s) matched specifications".to_string());
77    }
78    if resolved.spw_ids.is_empty() || resolved.channels.is_empty() {
79        return Err("No valid SPW & Chan combination found".to_string());
80    }
81    Ok(resolved)
82}
83
84fn resolve_specs(
85    expression: &Expression,
86    windows: &[SpectralWindowRow<'_>],
87    descriptions: &[DataDescriptionRow],
88    reports: &mut Reports,
89) -> Result<Resolved, String> {
90    let index = SpwIndex::new(windows);
91    let mut resolved = Resolved {
92        spw_ids: Vec::new(),
93        data_description_ids: Vec::new(),
94        channels: Vec::new(),
95        rows: None,
96        warnings: Vec::new(),
97    };
98    for spec in &expression.specs {
99        let spws = windows_of(&spec.spw, &index, reports)?;
100        let elements = spec.channels.as_deref();
101        let triples = index.channel_triples(&spws, elements, &mut resolved.warnings)?;
102        select_channels(
103            &spws,
104            &triples,
105            elements.map_or(1, <[_]>::len),
106            &mut resolved.channels,
107        );
108        for &spw in &spws {
109            // `selectSpwIdsFromIDList(ids, False)` (`MSSpwParse.cc:83-155`):
110            // a window once per unflagged DATA_DESCRIPTION row naming it,
111            // none when the window is flagged.
112            if index.window(spw).flag_row {
113                continue;
114            }
115            for (ddid, description) in descriptions.iter().enumerate() {
116                if !description.flag_row
117                    && usize::try_from(description.spectral_window_id).ok() == Some(spw)
118                {
119                    resolved.spw_ids.push(spw);
120                    resolved.data_description_ids.push(ddid);
121                }
122            }
123        }
124    }
125    if !expression.parenthesized {
126        // `endOfCeremony` (`MSSpwParse.cc:291-317`): the rows of the data
127        // descriptions of the unique windows, the same rule again.
128        let mut rows = resolved.data_description_ids.clone();
129        rows.sort_unstable();
130        rows.dedup();
131        resolved.rows = Some(rows);
132    }
133    Ok(resolved)
134}

136/// The windows of one `Spw` form, after `FullSpec`'s `matchId`.
137fn windows_of(
138    spw: &Spw,
139    index: &SpwIndex<'_, '_>,
140    reports: &mut Reports,
141) -> Result<Vec<usize>, String> {
142    Ok(match spw {
143        Spw::Name(name) => {
144            let ids = index.match_name(name);
145            reports.check_empty(&ids, "No match found for ", name.clone());
146            ids
147        }
148        Spw::Pattern(pattern) => {
149            let ids = index.match_pattern(pattern)?;
150            reports.check_empty(&ids, "No match found for ", pattern.clone());
151            ids
152        }
153        Spw::Greater(bound) => {
154            let ids = compare(index, *bound, None, |value, bound| value > bound)?;
155            reports.check_empty(&ids, "No spw ID found for > ", shown(*bound));
156            ids
157        }
158        Spw::Less(bound) => {
159            let ids = compare(index, *bound, None, |value, bound| value < bound)?;
160            reports.check_empty(&ids, "No spw ID found for < ", shown(*bound));
161            ids
162        }
163        Spw::Between(low, high) => {
164            let ids = compare(index, *low, Some(*high), |_, _| false)?;
165            reports.check_empty(
166                &ids,
167                "No spw ID found ",
168                format!("{}<>{}", shown(*low), shown(*high)),
169            );
170            ids
171        }
172        Spw::Near(frequency) => {
173            // `matchFrequencyRange(f, f, True)` refuses when nothing matches.
174            index.match_frequency_range((frequency.value, frequency.value), true, 0.0)?
175        }
176        Spw::List(elements) => {
177            let list = index.window_list(elements)?;
178            if list.missing_count > 0 {
179                let mut token = list
180                    .missing
181                    .iter()
182                    .map(|id| format!("{id},"))
183                    .collect::<String>();
184                if list.missing_count > list.missing.len() as u64 {
185                    token.push_str(&format!("... ({} indices in all)", list.missing_count));
186                }
187                reports.report("Spw Expression: No match found for ", token);
188            }
189            list.ids
190        }
191    })
192}

194/// `matchGT`, `matchLT` (with `test`) and `matchGTAndLT` (with `high`): on
195/// window identifiers when the (first) bound is an index, on REF_FREQUENCY
196/// when it is a frequency; the code of the first bound decides for both.
197fn compare(
198    index: &SpwIndex<'_, '_>,
199    bound: OneFreq,
200    high: Option<OneFreq>,
201    test: impl Fn(f64, f64) -> bool,
202) -> Result<Vec<usize>, String> {
203    let low = f64::from(bound.value);
204    Ok(match (bound.unit, high) {
205        (Unit::Index, None) => index.ids_where(|id| test(id as f64, low)),
206        (Unit::Hertz, None) => index.references_where(|reference| test(reference, low)),
207        (Unit::Index, Some(high)) => {
208            // `matchGTAndLT((Int)$1[0], (Int)$3[0])`.
209            let high = f64::from(integer(high.value)?);
210            index.ids_where(|id| (id as f64) > low && (id as f64) < high)
211        }
212        (Unit::Hertz, Some(high)) => {
213            let high = f64::from(high.value);
214            index.references_where(|reference| reference > low && reference < high)
215        }
216    })
217}
218
219fn shown(bound: OneFreq) -> String {
220    match bound.unit {
221        Unit::Index => format!("{}", bound.value),
222        Unit::Hertz => format!("{}Hz", bound.value),
223    }
224}

226/// `selectChannelsFromIDList` (`MSSpwParse.cc:228-257`): each window's
227/// `n_specs` triples, kept only when the first of them starts and stops on
228/// a channel. A window whose first triple does not is skipped by one triple
229/// only, so the next window reads from the skipped window's second triple:
230/// a CASA quirk kept for parity.
231fn select_channels(
232    spws: &[usize],
233    triples: &[[i32; 3]],
234    n_specs: usize,
235    channels: &mut Vec<ChannelRange>,
236) {
237    let mut next = 0;
238    for &spw in spws {
239        let [start, stop, _] = triples[next];
240        if start != -1 && stop != -1 {
241            for &[start, stop, step] in &triples[next..next + n_specs] {
242                channels.push(ChannelRange {
243                    spw_id: spw,
244                    start,
245                    stop,
246                    step,
247                });
248            }
249            next += n_specs;
250        } else {
251            next += 1;
252        }
253    }
