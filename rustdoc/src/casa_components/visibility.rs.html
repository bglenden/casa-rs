2//! A component's visibility, as CASA's `SkyCompRep::visibility` forms it
3//! (`components/ComponentModels`): the flux in Stokes, scaled by the spectral
4//! model at the frequency, converted to the data's polarisation basis, and
5//! scaled by the shape's visibility at the baseline.
6//!
7//! The baseline is in metres and already rotated to the component's
8//! reference direction, with CASA's sign: `SimpleComponentFTMachine::get`
9//! negates u and v before it rotates them. The frequency is the data's,
10//! taken in the frame of the spectral model's reference frequency, as CASA
11//! takes it ("the frequency is expected to be in the frame of the
12//! component").
13//!
14//! What casa-rs does not evaluate is refused rather than approximated:
15//! - a tabular spectrum whose reference frequency is in another frame than
16//!   its tabulation, which CASA converts through a frame casa-rs would have
17//!   to build, and a table of fewer than two points, which CASA's linear
18//!   interpolation reads out of bounds;
19//! - a flux unit that is not a flux density (`Jy.km/s`, `K.rad2`), which
20//!   CASA's `convertUnit(Unit("Jy"))` cannot convert either.
21

34/// The speed of light, m/s (casacore `C::c`).
35const SPEED_OF_LIGHT_M_PER_S: f64 = 299_792_458.0;

37impl Flux {
38    /// Whether the flux is physical in CASA's Stokes convention.
39    ///
40    /// CASA requires imaginary Stokes values to be within 1e-13 of zero and
41    /// squared Stokes I to be at least the sum of squared Q, U, and V. It does
42    /// not reject a negative I on its own.
43    #[must_use]
44    pub fn is_physical(&self) -> bool {
45        let [i, q, u, v] = to_stokes(*self.values(), self.polarisation());
46        if ![i, q, u, v].iter().all(|value| value.im.abs() <= 1e-13) {
47            return false;
48        }
49        (i.re * i.re).partial_cmp(&(q.re * q.re + u.re * u.re + v.re * v.re))
50            != Some(std::cmp::Ordering::Less)
51    }
52
53    /// The flux as real Stokes I, Q, U and V in Jy: `convertUnit("Jy")`, then
54    /// `FluxRep::value(Vector<Double>&)`, which converts to Stokes and keeps
55    /// the real parts.
56    ///
57    /// # Errors
58    ///
59    /// [`ComponentError::FluxUnit`] when the unit is not a flux density.
60    pub fn stokes_jy_values(&self) -> Result<[f64; 4], ComponentError> {
61        let jy = Unit::new("Jy").expect("Jy is a unit");
62        let scale = Quantity::with_unit(1.0, self.unit().clone())
63            .get_value_in(&jy)
64            .map_err(|_| ComponentError::FluxUnit(self.unit().name().to_string()))?;
65        let stokes = to_stokes(*self.values(), self.polarisation());
66        Ok(stokes.map(|value| value.re * scale))
67    }
68}

70/// `values` in `from`'s representation, as complex Stokes I, Q, U and V
71/// (`Flux::linearToStokes`, `Flux::circularToStokes`).
72pub(crate) fn to_stokes(values: [Complex64; 4], from: Polarisation) -> [Complex64; 4] {
73    let [a, b, c, d] = values;
74    match from {
75        Polarisation::Stokes => values,
76        Polarisation::Circular => [
77            (a + d) / 2.0,
78            (b + c) / 2.0,
79            Complex64::new((b.im - c.im) / 2.0, (c.re - b.re) / 2.0),
80            (a - d) / 2.0,
81        ],
82        Polarisation::Linear => [
83            (a + d) / 2.0,
84            (a - d) / 2.0,
85            (b + c) / 2.0,
86            Complex64::new((b.im - c.im) / 2.0, (c.re - b.re) / 2.0),
87        ],
88    }
89}

91/// Real Stokes I, Q, U and V in `basis`'s order (`Flux::stokesToCircular`,
92/// `Flux::stokesToLinear`): RR = I + V, RL = Q + iU, LR = Q − iU, LL = I − V,
93/// or XX = I + Q, XY = U + iV, YX = U − iV, YY = I − Q.
94#[must_use]
95pub fn stokes_in(basis: Polarisation, iquv: [f64; 4]) -> [Complex64; 4] {
96    let [i, q, u, v] = iquv;
97    match basis {
98        Polarisation::Stokes => iquv.map(|value| Complex64::new(value, 0.0)),
99        Polarisation::Circular => [
100            Complex64::new(i + v, 0.0),
101            Complex64::new(q, u),
102            Complex64::new(q, -u),
103            Complex64::new(i - v, 0.0),
104        ],
105        Polarisation::Linear => [
106            Complex64::new(i + q, 0.0),
107            Complex64::new(u, v),
108            Complex64::new(u, -v),
109            Complex64::new(i - q, 0.0),
110        ],
111    }
112}

114impl SpectralModel {
115    /// The Stokes flux at `frequency_hz` of a component whose flux at the
116    /// reference frequency is `iquv`: CASA's matrix `sampleStokes`, which
117    /// `SkyCompRep::visibility` uses.
118    /// - Constant: `iquv` unchanged.
119    /// - Spectral index: I scales as `(ν/ν₀)^α₀`, Q and U as
120    ///   `(ν/ν₀)^(α₀+α₁)` and V as `(ν/ν₀)^(α₀+α₃)`, then Q and U rotate by
121    ///   twice `α₂ c² (ν₀² − ν²)/(ν² ν₀²)` (a rotation measure). A single
122    ///   index is `[α, 0, 0, 0]`: CASA's matrix sampler has no branch for it
123    ///   and reads the missing indices out of bounds (`docs/CASA (C++)
124    ///   bugs.md`), which for an unpolarised component multiplies zeros.
125    /// - Power-log-polynomial: I scales as `x^(c₀ + c₁ ln x + c₂ ln²x + …)`
126    ///   with `x = ν/ν₀`; Q, U and V are unchanged.
127    /// - Tabular: the tabulated Stokes values, interpolated linearly in
128    ///   frequency and extrapolated from the end points
129    ///   (`InterpolateArray1D::linear`); the component's own flux does not
130    ///   enter.
131    ///
132    /// # Errors
133    ///
134    /// [`ComponentError::Spectrum`] for a tabular spectrum casa-rs does not
135    /// evaluate (see the module documentation) or a reference frequency that
136    /// is not positive.
137    pub fn sample_stokes(
138        &self,
139        frequency_hz: f64,
140        iquv: [f64; 4],
141    ) -> Result<[f64; 4], ComponentError> {
142        let nu0 = self.reference_frequency().hz();
143        let positive = |spectrum: &'static str| {
144            if nu0 > 0.0 {
145                Ok(nu0)
146            } else {
147                Err(ComponentError::Spectrum {
148                    spectrum,
149                    reason: format!("the reference frequency {nu0} Hz is not positive"),
150                })
151            }
152        };
153        Ok(match self.kind() {
154            SpectralKind::Constant => iquv,
155            SpectralKind::SpectralIndex(index) => {
156                let nu0 = positive("spectral index")?;
157                let [a0, a1, a2, a3] =
158                    index
159                        .stokes_indices()
160                        .unwrap_or([index.index(), 0.0, 0.0, 0.0]);
161                let ratio = frequency_hz / nu0;
162                let [i, q, u, v] = iquv;
163                let (mut q, mut u) = (q * ratio.powf(a0 + a1), u * ratio.powf(a0 + a1));
164                if a2 != 0.0 {
165                    let nu = frequency_hz;
166                    let two_alpha = 2.0
167                        * a2
168                        * SPEED_OF_LIGHT_M_PER_S
169                        * SPEED_OF_LIGHT_M_PER_S
170                        * (nu0 * nu0 - nu * nu)
171                        / (nu * nu * nu0 * nu0);
172                    let (sin, cos) = two_alpha.sin_cos();
173                    (q, u) = (cos * q - sin * u, sin * q + cos * u);
174                }
175                [i * ratio.powf(a0), q, u, v * ratio.powf(a0 + a3)]
176            }
177            SpectralKind::PowerLogPolynomial(polynomial) => {
178                let nu0 = positive("power-log-polynomial")?;
179                let x = frequency_hz / nu0;
180                let log_x = x.ln();
181                let mut exponent = 0.0;
182                for (power, coefficient) in polynomial.coefficients().iter().enumerate() {
183                    exponent += match power {
184                        0 => *coefficient,
185                        1 => coefficient * log_x,
186                        2 => coefficient * log_x * log_x,
187                        _ => coefficient * log_x.powi(power as i32),
188                    };
189                }
190                let [i, q, u, v] = iquv;
191                [i * x.powf(exponent), q, u, v]
192            }
193            SpectralKind::Tabular(table) => {
194                tabular_stokes(table, self.reference_frequency().refer(), frequency_hz)?
195            }
196        })
197    }
198}

200/// `table`'s Stokes values at `frequency_hz` (see
201/// [`SpectralModel::sample_stokes`]).
202fn tabular_stokes(
203    table: &TabularSpectrum,
204    reference_frame: casa_types::measures::frequency::FrequencyRef,
205    frequency_hz: f64,
206) -> Result<[f64; 4], ComponentError> {
207    let refused = |reason: String| ComponentError::Spectrum {
208        spectrum: "tabular",
209        reason,
210    };
211    if table.frame() != reference_frame {
212        return Err(refused(format!(
213            "a reference frequency in {reference_frame:?} and a tabulation in {:?}: casa-rs \
214             does not convert the data's frequencies between them",
215            table.frame()
216        )));
217    }
218    let frequencies = table.frequencies_hz();
219    if frequencies.len() < 2 {
220        return Err(refused(format!(
221            "{} tabulated frequencies; linear interpolation needs two",
222            frequencies.len()
223        )));
224    }
225    // `binarySearchBrackets`, which takes the tabulation in either order,
226    // then the bracketing pair, or the end pair.
227    let ascending = frequencies[0] <= frequencies[frequencies.len() - 1];
228    let upper = frequencies
229        .partition_point(|frequency| {
230            if ascending {
231                *frequency < frequency_hz
232            } else {
233                *frequency > frequency_hz
234            }
235        })
236        .clamp(1, frequencies.len() - 1);
237    let (x1, x2) = (frequencies[upper - 1], frequencies[upper]);
238    let fraction = (frequency_hz - x1) / (x2 - x1);
239    let (y1, y2) = (table.stokes_flux()[upper - 1], table.stokes_flux()[upper]);
240    Ok(std::array::from_fn(|stokes| {
241        y1[stokes] + fraction * (y2[stokes] - y1[stokes])
242    }))
243}

245impl ComponentShape {
246    /// The shape's visibility at `uvw_m` and `frequency_hz` (`visibility`
247    /// of `PointShape`, `GaussianShape`, `DiskShape` and
248    /// `LimbDarkenedDiskShape`), real.
249    ///
250    /// # Errors
251    ///
252    /// [`ComponentError::Visibility`] if the Fourier transform cannot be
253    /// evaluated. A shape alone uses the default limb-darkening exponent;
254    /// [`SkyComponent::transform`] includes its optional parameters.
255    pub fn visibility_scale(
256        &self,
257        uvw_m: [f64; 3],
258        frequency_hz: f64,
259    ) -> Result<f64, ComponentError> {
260        self.transform()?
261            .at_baseline(uvw_m)
262            .at_frequency(frequency_hz)
263    }
264
265    /// The shape's Fourier transform, prepared from its axes for evaluation
266    /// at many baselines and frequencies, as `GaussianShape::updateFT` caches
267    /// the Gaussian's.
268    ///
269    /// # Errors
270    ///
271    /// [`ComponentError::Visibility`] for invalid limb-darkening parameters.
272    pub fn transform(&self) -> Result<ShapeTransform, ComponentError> {
273        self.transform_with_limb_darkening(crate::DEFAULT_LIMB_DARKENING_EXPONENT)
274    }
275
276    fn transform_with_limb_darkening(
277        &self,
278        exponent: f64,
279    ) -> Result<ShapeTransform, ComponentError> {
280        Ok(ShapeTransform(match self.kind() {
281            ShapeKind::Point => Transform::Point,
282            ShapeKind::Gaussian { axes, .. } => Transform::gaussian(axes),
283            ShapeKind::Disk { axes, .. } | ShapeKind::LimbDarkenedDisk { axes, .. } => {
284                Transform::Disk {
285                    rotation: (axes.position_angle().abs() > NEAR_ABS)
286                        .then(|| axes.position_angle().sin_cos()),
287                    minor_axis: axes.minor_axis(),
288                    major_axis: axes.major_axis(),
289                    limb: matches!(self.kind(), ShapeKind::LimbDarkenedDisk { .. })
290                        .then(|| LimbDarkening::new(exponent))
291                        .transpose()?,
292                }
293            }
294        }))
295    }
296}

298/// A shape's Fourier transform ([`ComponentShape::transform`]), holding no
299/// heap. [`Self::at_baseline`] does the work that depends on the baseline,
300/// so a baseline's channels each evaluate a function of frequency alone.
301#[derive(Debug, Clone, Copy, PartialEq)]
302pub struct ShapeTransform(Transform);
303
304#[derive(Debug, Clone, Copy, PartialEq)]
305enum Transform {
306    Point,
307    /// `GaussianShape`'s cached transform: casacore's `Gaussian2D` of unit
308    /// height evaluated at `(x, y) = (−u, v) ν/c` (`Gaussian2D::eval`), as
309    /// `exp(−(x′² + y′²))`. Its rotated and scaled coordinates are rows of
310    /// coefficients on `(u, v)`, so that `x′ = (ν/c) (x_row · (u, v))` and
311    /// likewise `y′`.
312    Gaussian {
313        x_row: [f64; 2],
314        y_row: [f64; 2],
315    },
316    /// `DiskShape::visibility`: the baseline rotated by the position angle
317    /// when it is not near zero, and scaled by the axes.
318    Disk {
319        rotation: Option<(f64, f64)>,
320        minor_axis: f64,
321        major_axis: f64,
322        limb: Option<LimbDarkening>,
323    },
324}
325
326impl Transform {
327    /// The Gaussian of `axes`: widths `4 ln 2/(π · minor)` and `4 ln 2/(π ·
328    /// major)` and the shape's position angle plus π/2
329    /// (`GaussianShape::updateFT`).
330    fn gaussian(axes: &Ellipse) -> Self {
331        let factor = 4.0 * std::f64::consts::LN_2 / std::f64::consts::PI;
332        // `setWidth`: the larger width is the major one, held as the y
333        // width, and the ratio is x over y; `setPA` keeps the angle when y is
334        // the larger width, as here.
335        let y_width = factor / axes.minor_axis();
336        let ratio = (factor / axes.major_axis()) / y_width;
337        let angle = axes.position_angle() + std::f64::consts::FRAC_PI_2;
338        let (sin, cos) = angle.sin_cos();
339        let fwhm_to_sigma = 1.0 / 16.0_f64.ln().sqrt();
340        // x′ = (cos x + sin y)/σx and y′ = (−sin x + cos y)/σy, with x = −u
341        // and y = v.
342        let (x_sigma, y_sigma) = (y_width * ratio * fwhm_to_sigma, y_width * fwhm_to_sigma);
343        Self::Gaussian {
344            x_row: [-cos / x_sigma, sin / x_sigma],
345            y_row: [sin / y_sigma, cos / y_sigma],
346        }
347    }
348}

350impl ShapeTransform {
351    /// The transform at `uvw_m` (rotated to the shape's reference direction,
352    /// CASA's sign), as a function of frequency.
353    #[must_use]
354    pub fn at_baseline(&self, uvw_m: [f64; 3]) -> BaselineTransform {
355        let [u, v, _] = uvw_m;
356        let wavenumber_squared = 1.0 / (SPEED_OF_LIGHT_M_PER_S * SPEED_OF_LIGHT_M_PER_S);
357        BaselineTransform(match self.0 {
358            Transform::Point => AtBaseline::One,
359            Transform::Gaussian { x_row, y_row } => {
360                let x = x_row[0] * u + x_row[1] * v;
361                let y = y_row[0] * u + y_row[1] * v;
362                AtBaseline::Gaussian((x * x + y * y) * wavenumber_squared)
363            }
364            // 1 where `u + v` is near zero, as CASA tests it (casacore
365            // `near`, relative to 1e-13), and otherwise `2 J₁(r)/r` of the
366            // rotated baseline scaled by the axes (the minor axis on u, the
367            // major on v) and by `π ν/c`.
368            Transform::Disk { .. } if near_zero(u + v) => AtBaseline::One,
369            Transform::Disk {
370                rotation,
371                minor_axis,
372                major_axis,
373                limb,
374            } => {
375                let (u, v) =
376                    rotation.map_or((u, v), |(sin, cos)| (u * cos - v * sin, u * sin + v * cos));
377                AtBaseline::Disk {
378                    per_hz: (u * minor_axis).hypot(v * major_axis) * std::f64::consts::PI
379                        / SPEED_OF_LIGHT_M_PER_S,
380                    limb,
381                }
382            }
383        })
384    }
385}

387/// A shape's transform at one baseline ([`ShapeTransform::at_baseline`]).
388#[derive(Debug, Clone, Copy, PartialEq)]
389pub struct BaselineTransform(AtBaseline);
390
391#[derive(Debug, Clone, Copy, PartialEq)]
392enum AtBaseline {
393    One,
394    /// `exp(−g ν²)`.
395    Gaussian(f64),
396    /// A uniform or limb-darkened disk at `r = R ν`.
397    Disk {
398        per_hz: f64,
399        limb: Option<LimbDarkening>,
400    },
401}
402
403impl BaselineTransform {
404    /// The shape's visibility at `frequency_hz`, real.
405    ///
406    /// # Errors
407    ///
408    /// [`ComponentError::Visibility`] when a real-order Bessel evaluation
409    /// fails or cannot be normalized without losing its result.
410    pub fn at_frequency(&self, frequency_hz: f64) -> Result<f64, ComponentError> {
411        Ok(match self.0 {
412            AtBaseline::One => 1.0,
413            AtBaseline::Gaussian(per_hz_squared) => {
414                (-per_hz_squared * frequency_hz * frequency_hz).exp()
415            }
416            AtBaseline::Disk { per_hz, limb } => {
417                let r = per_hz * frequency_hz;
418                match limb {
419                    Some(limb) => limb.at_radius(r)?,
420                    None => 2.0 * libm::j1(r) / r,
421                }
422            }
423        })
424    }
425}

427/// casacore's default `nearAbs` tolerance.
428const NEAR_ABS: f64 = 1.0e-13;
429
430/// casacore `near(x, 0.0)` with its default 1e-13: relative to zero, only
431/// values within `DBL_MIN` of it.
432fn near_zero(x: f64) -> bool {
433    x == 0.0 || x.abs() <= (1.0 + 1.0e-13) * f64::MIN_POSITIVE
434}

436impl SkyComponent {
437    /// Prepare this component's shape transform, including its optional
438    /// limb-darkening exponent rounded to CASA's stored single precision.
439    ///
440    /// # Errors
441    ///
442    /// [`ComponentError::Visibility`] for invalid limb-darkening parameters.
443    pub fn transform(&self) -> Result<ShapeTransform, ComponentError> {
444        self.shape().transform_with_limb_darkening(
445            self.limb_darkening_exponent()
446                .unwrap_or(crate::DEFAULT_LIMB_DARKENING_EXPONENT),
447        )
448    }
449
450    /// The component's visibility at `uvw_m` (rotated to its reference
451    /// direction, CASA's sign) and `frequency_hz`, in `basis`'s order
452    /// (`SkyCompRep::visibility`): the Stokes flux in Jy sampled by the
453    /// spectral model, converted to `basis`, times the shape's visibility.
454    ///
455    /// # Errors
456    ///
457    /// The flux unit, spectral model or shape casa-rs does not evaluate.
458    pub fn visibility(
459        &self,
460        uvw_m: [f64; 3],
461        frequency_hz: f64,
462        basis: Polarisation,
463    ) -> Result<[Complex64; 4], ComponentError> {
464        let flux = self.flux_in(frequency_hz, basis)?;
465        let scale = self
466            .transform()?
467            .at_baseline(uvw_m)
468            .at_frequency(frequency_hz)?;
469        Ok(flux.map(|value| value * scale))
470    }
471
472    /// The component's flux at `frequency_hz` in `basis`'s order, in Jy: the
473    /// part of [`Self::visibility`] that does not depend on the baseline.
474    ///
475    /// # Errors
476    ///
477    /// The flux unit or spectral model casa-rs does not evaluate.
478    pub fn flux_in(
479        &self,
480        frequency_hz: f64,
481        basis: Polarisation,
482    ) -> Result<[Complex64; 4], ComponentError> {
483        let iquv = self.flux().stokes_jy_values()?;
484        let sampled = self.spectrum().sample_stokes(frequency_hz, iquv)?;
485        Ok(stokes_in(basis, sampled))
486    }
487}
