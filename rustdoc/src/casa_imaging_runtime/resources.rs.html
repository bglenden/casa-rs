2//! Host resources, the run's resource policy and the admission of each
3//! phase's memory against them (plan section 5.4).
4//!
5//! The host is detected once per process. A policy turns it into the
6//! workers and memory a run may use; each phase then admits the memory its
7//! owners say it holds ([`admit`]) and keeps it until the returned
8//! [`Reservation`] drops. Reservations of every run in the process count
9//! against each run's policy.
10
11#[cfg(target_os = "macos")]
12use std::process::Command;
13use std::sync::atomic::{AtomicU64, Ordering};
14use std::sync::{Mutex, MutexGuard, OnceLock, PoisonError};

16/// The host's capacity, detected once per process by
17/// [`HostResources::detect`]; tests construct it with pinned values.
18#[derive(Clone, Copy, Debug, PartialEq, Eq)]
19pub struct HostResources {
20    /// Logical CPU threads.
21    pub threads: usize,
22    /// Performance-class cores; `threads` on a host that reports no core
23    /// classes.
24    pub performance_cores: usize,
25    /// Bytes of memory free for new allocations when the host was detected:
26    /// free, inactive and speculative pages on macOS, `MemAvailable` on
27    /// Linux.
28    pub available_memory: u64,
29    /// Whether a unified-memory Metal 3 device is present.
30    pub metal: bool,
31}
32
33/// Why the host could not be detected: one variant per detector, each with
34/// what it reported.
35#[derive(Clone, Debug, thiserror::Error)]
36pub enum HostError {
37    /// `std::thread::available_parallelism` could not count the threads.
38    #[error("host detection: thread count: {0}")]
39    Threads(String),
40    /// macOS `vm_stat` did not run or its report had no page size.
41    #[error("host detection: vm_stat: {0}")]
42    VmStat(String),
43    /// Linux `/proc/meminfo` could not be read or has no `MemAvailable`.
44    #[error("host detection: /proc/meminfo: {0}")]
45    MemInfo(String),
46    /// macOS `sysctl` did not run or did not report a count for `name`.
47    #[error("host detection: sysctl {name}: {detail}")]
48    Sysctl {
49        /// The queried name.
50        name: String,
51        /// What went wrong.
52        detail: String,
53    },
54    /// This platform has no free-memory detector.
55    #[error("host detection: free memory is unknown on this platform")]
56    UnsupportedPlatform,
57}
58
59impl HostResources {
60    /// The host, detected on the first call and the same thereafter.
61    ///
62    /// # Errors
63    ///
64    /// Fails when the thread count or the free memory cannot be read.
65    pub fn detect() -> Result<Self, HostError> {
66        static HOST: OnceLock<Result<HostResources, HostError>> = OnceLock::new();
67        HOST.get_or_init(|| {
68            let threads = std::thread::available_parallelism()
69                .map_err(|error| HostError::Threads(error.to_string()))?
70                .get();
71            Ok(Self {
72                threads,
73                performance_cores: performance_cores().map_or(threads, |cores| {
74                    usize::try_from(cores).unwrap_or(threads).clamp(1, threads)
75                }),
76                available_memory: available_memory()?,
77                metal: casa_imaging_metal::available(),
78            })
79        })
80        .clone()
81    }
82}

84/// Threads a streamed phase keeps busy beside its workers: the producer
85/// that fills the source's next block while the workers place the current
86/// one (`imaging-source`).
87pub const STREAM_PRODUCERS: usize = 1;
88
89/// How one run's phases use the host's cores ([`ResourcePolicy::cores`]).
90///
91/// A phase that computes on data in memory (minor cycles, products) runs
92/// `workers` workers. A phase that streams the source (the major-cycle and
93/// imaging-weight passes) also keeps the source's producer busy
94/// ([`STREAM_PRODUCERS`]), so it runs `streaming_workers`: as many workers
95/// as fit beside the producer within the policy's core budget. Threads past
96/// that budget run on efficiency cores or contend for the performance
97/// cores' caches and memory bandwidth, which costs more cycles for the same
98/// work (#659: four placing workers and the producer on four performance
99/// cores took 1.13× the cycles of three and the producer).
100#[derive(Clone, Copy, Debug, PartialEq, Eq)]
101pub struct CoreAllocation {
102    /// Workers of a phase that computes on data in memory.
103    pub workers: usize,
104    /// Workers of a phase that streams the source; at least one, at most
105    /// `workers`.
106    pub streaming_workers: usize,
107}
108
109impl CoreAllocation {
110    /// This allocation with at most `workers` workers in each phase, and at
111    /// least one: what a run uses when the policy's workers do not fit its
112    /// memory (#737).
113    #[must_use]
114    pub fn at_most(self, workers: usize) -> Self {
115        let workers = workers.clamp(1, self.workers);
116        Self {
117            workers,
118            streaming_workers: self.streaming_workers.min(workers),
119        }
120    }
121}

123/// How much of the host one run may use.
124///
125/// Each policy sets a budget of busy threads: its workers and, while a
126/// phase streams the source, the source's producer ([`CoreAllocation`]).
127#[derive(Clone, Copy, Debug, PartialEq, Eq)]
128pub enum ResourcePolicy {
129    /// Half the threads, rounded up, and half the free memory.
130    Interactive,
131    /// Three quarters of the threads, rounded up and at most the
132    /// performance cores, and three quarters of the free memory. A streamed
133    /// phase's producer shares the performance cores: its workers are as
134    /// many as fit beside it there.
135    Balanced,
136    /// Every thread and all the free memory.
137    Exclusive,
138    /// At most `workers` threads, a streamed phase's producer among them,
139    /// and `memory` bytes, each capped at what the host has.
140    Explicit {
141        /// Worker ceiling.
142        workers: usize,
143        /// Memory ceiling in bytes.
144        memory: u64,
145    },
146}
147
148impl ResourcePolicy {
149    /// Workers a run may use on `host`; at least one.
150    #[must_use]
151    pub fn workers(&self, host: &HostResources) -> usize {
152        let threads = host.threads;
153        match *self {
154            Self::Interactive => threads.div_ceil(2),
155            Self::Balanced => (threads * 3).div_ceil(4).min(host.performance_cores),
156            Self::Exclusive => threads,
157            Self::Explicit { workers, .. } => workers.min(threads),
158        }
159        .max(1)
160    }
161
162    /// How a run's phases use `host`: [`Self::workers`] for a phase that
163    /// computes in memory, and for a phase that streams the source the
164    /// workers that fit beside its producer within the policy's budget of
165    /// busy threads. That budget is the performance cores for
166    /// [`Self::Balanced`] and the workers for the other policies.
167    #[must_use]
168    pub fn cores(&self, host: &HostResources) -> CoreAllocation {
169        let workers = self.workers(host);
170        let budget = match *self {
171            Self::Balanced => host.performance_cores,
172            Self::Interactive | Self::Exclusive | Self::Explicit { .. } => workers,
173        };
174        CoreAllocation {
175            workers,
176            streaming_workers: budget.saturating_sub(STREAM_PRODUCERS).min(workers).max(1),
177        }
178    }
179
180    /// Bytes of memory a run may use on `host`, before reservations.
181    #[must_use]
182    pub fn memory(&self, host: &HostResources) -> u64 {
183        let free = u128::from(host.available_memory);
184        let scaled = |numerator: u128, denominator: u128| {
185            u64::try_from(free * numerator / denominator).expect("a fraction of a u64")
186        };
187        match *self {
188            Self::Interactive => scaled(1, 2),
189            Self::Balanced => scaled(3, 4),
190            Self::Exclusive => host.available_memory,
191            Self::Explicit { memory, .. } => memory.min(host.available_memory),
192        }
193    }
194}

196/// The memory one phase holds while it runs, as its owners report it.
197#[derive(Clone, Copy, Debug, PartialEq, Eq)]
198pub struct Demand {
199    /// The phase, for the rejection message.
200    pub phase: &'static str,
201    /// Bytes held.
202    pub memory: u64,

205/// A phase's demand did not fit what the policy leaves free.
206#[derive(Clone, Copy, Debug, PartialEq, Eq, thiserror::Error)]
207#[error("{phase} needs {required} bytes but the resource policy leaves {available} free")]
208pub struct Admission {
209    /// The rejected phase.
210    pub phase: &'static str,
211    /// Bytes the phase needs.
212    pub required: u64,
213    /// Bytes the policy leaves free.
214    pub available: u64,

217/// Memory admitted to one phase; it is released when this drops.
218#[must_use = "the memory is released when the reservation drops"]
219#[derive(Debug)]
220pub struct Reservation {
221    memory: u64,
222}
223
224impl Reservation {
225    /// A reservation holding nothing, for state not yet formed.
226    pub const fn none() -> Self {
227        Self { memory: 0 }
228    }
229
230    /// Bytes held.
231    #[must_use]
232    pub const fn memory(&self) -> u64 {
233        self.memory
234    }
235
236    /// Keep `memory` of the held bytes and release the rest: what a phase
237    /// hands on stays charged with that state when its scratch is released.
238    ///
239    /// # Panics
240    ///
241    /// When `memory` exceeds the bytes held.
242    pub fn retain(&mut self, memory: u64) {
243        assert!(
244            memory <= self.memory,
245            "a reservation retains at most what it holds"
246        );
247        RESERVED.fetch_sub(self.memory - memory, Ordering::AcqRel);
248        self.memory = memory;
249    }
250
251    /// Move `memory` of the held bytes into a reservation of their own, for
252    /// state that outlives the phase that admitted it.
253    ///
254    /// # Panics
255    ///
256    /// When `memory` exceeds the bytes held.
257    pub fn split(&mut self, memory: u64) -> Self {
258        assert!(
259            memory <= self.memory,
260            "a reservation splits off at most what it holds"
261        );
262        self.memory -= memory;
263        Self { memory }
264    }
265
266    /// Hold `other`'s bytes too, until this reservation drops.
267    pub fn join(&mut self, mut other: Self) {
268        self.memory += std::mem::take(&mut other.memory);
269    }
270}
271
272impl Drop for Reservation {
273    fn drop(&mut self) {
274        RESERVED.fetch_sub(self.memory, Ordering::AcqRel);
275    }
276}

279/// its charge, until the reservation drops.

284    /// The memory.

288/// Bytes held by the live reservations of this process. Only [`admit`]
289/// raises it, holding [`WINDOWS`]; releases lower it without the lock.
290static RESERVED: AtomicU64 = AtomicU64::new(0);
291
292/// The open [`ReservationWindow`]s, each with the most [`RESERVED`] has
293/// held since it last closed.
294static WINDOWS: Mutex<Windows> = Mutex::new(Windows {
295    next: 0,
296    open: Vec::new(),
297});
298
299struct Windows {
300    next: u64,
301    open: Vec<(u64, u64)>,
302}
303
304fn windows() -> MutexGuard<'static, Windows> {
305    WINDOWS.lock().unwrap_or_else(PoisonError::into_inner)
306}
307
308/// One observer's view of the process's reservations: the most they held
309/// at once over each interval it closes. Every run summary has its own, so
310/// runs sharing a process neither reset nor take each other's intervals.
311/// Admission raises every open window under the lock that a window closes
312/// under, so an admission falls wholly within one interval.
313#[derive(Debug)]
314pub struct ReservationWindow {
315    id: u64,
316}
317
318impl ReservationWindow {
319    /// A window whose first interval starts at what the reservations hold
320    /// now.
321    #[must_use]
322    pub fn open() -> Self {
323        let mut windows = windows();
324        let id = windows.next;
325        windows.next += 1;
326        windows.open.push((id, RESERVED.load(Ordering::Acquire)));
327        Self { id }
328    }
329
330    /// The most the reservations held at once in the interval this call
331    /// closes; the next starts at what they hold now, so a reservation that
332    /// outlives an interval counts in the next too.
333    pub fn close(&self) -> u64 {
334        let mut windows = windows();
335        let now = RESERVED.load(Ordering::Acquire);
336        let (_, peak) = windows
337            .open
338            .iter_mut()
339            .find(|(id, _)| *id == self.id)
340            .expect("a window is open until it drops");
341        std::mem::replace(peak, now).max(now)
342    }
343}
344
345impl Drop for ReservationWindow {
346    fn drop(&mut self) {
347        windows().open.retain(|(id, _)| *id != self.id);
348    }
349}

351/// Bytes `policy` leaves free on `host` beside the live reservations.
352#[must_use]
353pub fn free_memory(host: &HostResources, policy: &ResourcePolicy) -> u64 {
354    policy
355        .memory(host)
356        .saturating_sub(RESERVED.load(Ordering::Acquire))
357}

359/// Bytes the live reservations of this process hold.
360#[must_use]
361pub fn reserved_memory() -> u64 {
362    RESERVED.load(Ordering::Acquire)
363}

365/// Admit `demand` if it fits what `policy` leaves free on `host`, and raise
366/// every open [`ReservationWindow`] to what the process then holds.
367///
368/// # Errors
369///
370/// [`Admission`] when the demand exceeds the free memory; nothing is held.
371pub fn admit(
372    host: &HostResources,
373    policy: &ResourcePolicy,
374    demand: &Demand,
375) -> Result<Reservation, Admission> {
376    let capacity = policy.memory(host);
377    let mut windows = windows();
378    // Only admissions raise the total, and they hold the lock, so releases
379    // between this read and the add only leave more free.
380    let available = capacity.saturating_sub(RESERVED.load(Ordering::Acquire));
381    if demand.memory > available {
382        return Err(Admission {
383            phase: demand.phase,
384            required: demand.memory,
385            available,
386        });
387    }
388    let held = RESERVED.fetch_add(demand.memory, Ordering::AcqRel) + demand.memory;
389    for (_, peak) in &mut windows.open {
390        *peak = (*peak).max(held);
391    }
392    Ok(Reservation {
393        memory: demand.memory,
394    })
395}

397/// Bytes the process-wide caches hold at most: the table-read cache, with a
398/// MeasurementSet's tiles, and FFTW's plan cache, which every imaging run
399/// and simulator session fills and which keep what they hold after it.
400#[must_use]
401pub fn process_cache_bytes() -> u64 {
402    (casa_ms::table_read_cache_bytes() + casa_imaging_operator::FFT_PLAN_CACHE_BYTES) as u64
403}
404
405/// The process caches' charge, taken by the first caller admitted.
406static PROCESS_CACHES: OnceLock<Reservation> = OnceLock::new();
407
408/// Charge the process-wide caches ([`process_cache_bytes`]) against
409/// `policy` on `host`, unless the process already holds their charge: it is
410/// taken once, by the first caller admitted, and held for the process. Of
411/// callers admitted at once, one keeps its charge and the others release
412/// theirs. A run or session calls this before it fills the caches.
413///
414/// # Errors
415///
416/// [`Admission`] when the caches do not fit what `policy` leaves free.
417pub fn charge_process_caches(
418    host: &HostResources,
419    policy: &ResourcePolicy,
420) -> Result<(), Admission> {
421    if PROCESS_CACHES.get().is_none() {
422        let charge = admit(
423            host,
424            policy,
425            &Demand {
426                phase: "process caches",
427                memory: process_cache_bytes(),
428            },
429        )?;
430        // A concurrent caller that set it first holds the charge, and this
431        // one's is released.
432        let _ = PROCESS_CACHES.set(charge);
433    }
434    Ok(())
435}

447#[cfg(target_os = "macos")]
448fn available_memory() -> Result<u64, HostError> {
449    let output = Command::new("/usr/bin/vm_stat")
450        .output()
451        .map_err(|error| HostError::VmStat(error.to_string()))?;
452    if !output.status.success() {
453        return Err(HostError::VmStat(format!("exited with {}", output.status)));
454    }
455    let text =
456        String::from_utf8(output.stdout).map_err(|error| HostError::VmStat(error.to_string()))?;
457    let page_size = text
458        .lines()
459        .next()
460        .and_then(|line| line.split("page size of ").nth(1))
461        .and_then(|value| value.split_ascii_whitespace().next())
462        .and_then(|value| value.parse::<u64>().ok())
463        .ok_or_else(|| HostError::VmStat("reports no page size".to_string()))?;
464    let pages = ["Pages free", "Pages inactive", "Pages speculative"]
465        .into_iter()
466        .map(|name| {
467            text.lines()
468                .find_map(|line| {
469                    let value = line.strip_prefix(name)?.strip_prefix(':')?;
470                    value.trim().trim_end_matches('.').parse::<u64>().ok()
471                })
472                .unwrap_or(0)
473        })
474        .sum::<u64>();
475    Ok(pages.saturating_mul(page_size))
476}

478#[cfg(target_os = "linux")]
479fn available_memory() -> Result<u64, HostError> {
480    let contents = std::fs::read_to_string("/proc/meminfo")
481        .map_err(|error| HostError::MemInfo(error.to_string()))?;
482    contents
483        .lines()
484        .find_map(|line| {
485            line.strip_prefix("MemAvailable:")?
486                .split_ascii_whitespace()
487                .next()?
488                .parse::<u64>()
489                .ok()
490        })
491        .map(|kib| kib.saturating_mul(1024))
492        .ok_or_else(|| HostError::MemInfo("has no MemAvailable".to_string()))
493}

500#[cfg(target_os = "macos")]
501fn sysctl_u64(name: &str) -> Result<u64, HostError> {
502    let failed = |detail: String| HostError::Sysctl {
503        name: name.to_string(),
504        detail,
505    };
506    let output = Command::new("/usr/sbin/sysctl")
507        .args(["-n", name])
508        .output()
509        .map_err(|error| failed(error.to_string()))?;
510    if !output.status.success() {
511        return Err(failed(format!("exited with {}", output.status)));
512    }
513    String::from_utf8_lossy(&output.stdout)
514        .trim()
515        .parse()
516        .map_err(|_| failed("is not a count".to_string()))
517}
