26/// Layout and rendering options for prepared bandpass panels.
27pub struct DisplayOptions<'a> {
28    /// Positive grid dimensions as rows followed by columns.
29    pub layout: [u32; 2],
30    /// Base output filename requested by the caller.
31    pub figfile: &'a str,
32    /// Whether to number page files sequentially instead of using selection suffixes.
33    pub sequential_names: bool,
34    /// Raster output resolution in dots per inch.
35    pub density: u32,
36    /// Plotting marker size in points.
37    pub markersize: f64,
38    /// Point visibility override, or None to use the panel default.
39    pub show_points: Option<bool>,
40    /// Line visibility override, or None to use the panel default.
41    pub show_lines: Option<bool>,
42    /// Whether the caller can pause for per-page interaction.
43    pub interactive: bool,
44    /// Explicit x-min, x-max, y-min, and y-max in plot units. Paired zeros request automatic
45    /// bounds.
46    pub limits: [f64; 4],
47    /// Optional vertical bounds in degrees for phase plots.
48    pub phase_limits: Option<[f64; 2]>,
49    /// Optional subset of channels used to choose the plot bounds.
50    pub channel_range: Option<ChannelRange>,
51    /// Whether the selected channels also restrict the horizontal range.
52    pub limit_channels: bool,
53    /// Resampling factor for placing plotted samples on the calibrated channel grid.
54    pub resample: u32,
55}

57#[derive(Clone, Copy)]
58/// Spectral span used to compute panel limits.
59pub enum ChannelRange {
60    /// Inclusive zero-based first and last channel indices.
61    Indices([usize; 2]),
62    /// Central fraction of the spectral span, expressed as a percentage.
63    CentralPercent(f64),
64}
65
66impl ChannelRange {
67    fn frequencies(self, coordinates: &[f64]) -> Option<[f64; 2]> {
68        let last = coordinates.len().checked_sub(1)?;
69        let [start, end] = match self {
70            Self::Indices(bounds) => bounds,
71            Self::CentralPercent(percent) => {
72                let outer = (100.0 - percent) * 0.005;
73                [
74                    (last as f64 * outer).round_ties_even() as usize,
75                    (last as f64 * (1.0 - outer)).round_ties_even() as usize,
76                ]
77            }
78        };
79        Some([*coordinates.get(start)?, *coordinates.get(end)?])
80    }
81}

119/// Paginates prepared panels into the requested positive grid and attaches legends and annotations.
120/// Invalid layouts return a diagnostic.
121pub fn pages(
122    plot: BandpassPlots,
123    display: &DisplayOptions<'_>,
124) -> Result<Vec<BandpassPage>, String> {
125    let [rows, columns] = display.layout;
126    let capacity = (rows as usize)
127        .checked_mul(columns as usize)
128        .filter(|&capacity| capacity > 0)
129        .ok_or("bandpass subplot needs a positive grid size")?;
130    let subtitle = plot
131        .measurement_set
132        .as_deref()
133        .and_then(|path| path.file_name())
134        .map_or_else(String::new, |name| name.to_string_lossy().into_owned());
135    let with_phase_page = capacity == 1
136        && plot
137            .panels
138            .iter()
139            .any(|panel| panel.ordinate == super::Ordinate::Phase);
140    plot.panels
141        .chunks(capacity)
142        .enumerate()
143        .map(|(index, source)| {
144            let (panels, legends) = source
145                .iter()
146                .enumerate()
147                .map(|(index, panel)| document_panel(panel, index, display))
148                .unzip();
149            let y_limits = source
150                .iter()
151                .map(|panel| y_limits(panel, display))
152                .collect();
153            let document = PlotDocument {
154                id: format!("plotbandpass-{index:03}"),
155                title: String::new(),
156                subtitle: subtitle.clone(),
157                header_lines: vec![plot.header.clone()],
158                axes: Vec::new(),
159                layers: Vec::new(),
160                annotations: Vec::new(),
161                panels,
162                show_legend: false,
163                layout: Some(PlotLayout {
164                    rows,
165                    columns,
166                    first_panel: None,
167                    aspect: None,
168                    margins: None,
169                    font_size: None,
170                }),
171            };
172            document
173                .validate_payload(0)
174                .map_err(|error| error.to_string())?;
175            let first = &source[0];
176            let last = &source[source.len() - 1];
177            // CASA records the page cursor before trying to draw a panel, so
178            // a multi-panel filename can refer to a skipped flagged group.
179            let suffix = if capacity == 1 {
180                first.suffix.as_str()
181            } else {
182                first
183                    .first_skipped_suffix
184                    .as_deref()
185                    .unwrap_or(&first.suffix)
186            };
187            let output_file = output_file(display, suffix, index);
188            let date = format_time(last.epoch / 86_400.0, YMD, 0);
189            let date = date.get(..10).unwrap_or(&date).replace('/', "-");
190            let mut footer = format!("{subtitle}  ObsDate={date}");
191            let regression = std::path::Path::new(display.figfile)
192                .file_name()
193                .is_some_and(|name| name.to_string_lossy().starts_with("regression"));
194            if !regression {
195                footer.push_str("   plotbandpass v2.20 = 2024/09/04 16:00:03");
196            }
197            Ok(BandpassPage {
198                figure: PlotFigure {
199                    document,
200                    output_file,
201                    show: display.interactive,
202                },
203                density: display.density,
204                y_limits,
205                atmosphere: source
206                    .iter()
207                    .map(|panel| panel.atmosphere.clone())
208                    .collect(),
209                fdm: source.iter().map(|panel| panel.fdm.clone()).collect(),
210                prompt: display.interactive && source.len() == capacity,
211                quit_immediately: plot.polynomial
212                    || plot.immediate_combined_overlay
213                    || (with_phase_page && last.ordinate != super::Ordinate::Phase),
214                footer,
215                diagnostics: source
216                    .iter()
217                    .flat_map(|panel| panel.diagnostics.iter().cloned())
218                    .collect(),
219                legends,
220            })
221        })
222        .collect()
223}

225fn output_file(display: &DisplayOptions<'_>, suffix: &str, page: usize) -> Option<PathBuf> {
226    if display.figfile.is_empty() {
227        return None;
228    }
229    let name = display
230        .figfile
231        .split_once(".png")
232        .map_or(display.figfile, |(base, _)| base);
233    let name = if display.sequential_names {
234        format!("{name}.{page:03}")
235    } else {
236        format!("{name}.{suffix}")
237    };
238    if display.resample > 1 {
239        Some(format!("{name}.resample{}.png", display.resample).into())
240    } else {
241        Some(format!("{name}.png").into())
242    }
243}

245fn document_panel(
246    bandpass: &BandpassPanel,
247    panel_number: usize,
248    display: &DisplayOptions<'_>,
249) -> (PlotDocumentPanel, OverlayLegend) {
250    let phase = bandpass.ordinate == super::Ordinate::Phase;
251    let (points, lines) = match (display.show_points, display.show_lines) {
252        (None, None) => (phase, !phase),
253        (None, Some(lines)) => (!lines, lines),
254        (Some(false), _) => (false, true),
255        (Some(true), lines) => (true, lines.unwrap_or(!phase)),
256    };
257    let delay = bandpass.ordinate == super::Ordinate::Delay;
258    let markersize = if delay {
259        display.markersize.max(8.0)
260    } else {
261        display.markersize
262    };
263    let (points, lines) = if delay {
264        (true, false)
265    } else {
266        (points, lines)
267    };
268    let mut paint = PanelColors::new(bandpass, display.layout);
269    let identity_overlay = bandpass.overlay.antenna || bandpass.overlay.time;
270    let combined = bandpass.overlay.antenna && bandpass.overlay.time;
271    let b_comparison = bandpass
272        .series
273        .iter()
274        .any(|series| !series.polynomial && series.table_index > 0);
275    let dual_receptor = bandpass.series.iter().any(|series| series.receptor == 0)
276        && bandpass.series.iter().any(|series| series.receptor == 1);
277    let mut annotations = Vec::new();
278    let mut labels = std::collections::BTreeSet::new();
279    let layers = bandpass
280        .series
281        .iter()
282        .enumerate()
283        .map(|(index, data)| {
284            let color = paint.color(data, index);
285            let points = points && !data.polynomial;
286            let lines = lines || data.polynomial;
287            let dashed = identity_overlay && dual_receptor && data.receptor == 1 && lines;
288            let marker = if points {
289                if data.table_index > 0 || (identity_overlay && data.receptor == 1) {
290                    "o"
291                } else {
292                    "."
293                }
294            } else {
295                "None"
296            };
297            let table_id = if b_comparison { data.table_index } else { 0 };
298            if !combined && labels.insert((&data.label, table_id)) {
299                let rows = f64::from(display.layout[0]);
300                let text = if b_comparison {
301                    format!("{}-c{}", data.label, table_id + 1)
302                } else if identity_overlay && lines {
303                    format!("{} {}", data.label, if dashed { "dashed" } else { "solid" })
304                } else {
305                    data.label.clone()
306                };
307                let y = 1.0 - (0.04 + 0.03 * (data.receptor + 2 * table_id) as f64) * rows;
308                if identity_overlay && points && !lines {
309                    let offset = if phase {
310                        0.015
311                            * if bandpass.overlay.antenna && data.receptor == 0 {
312                                1.0
313                            } else {
314                                rows
315                            }
316                    } else {
317                        0.0
318                    };
319                    paint.legend.markers.push(LegendMarker {
320                        marker,
321                        x: if phase { 0.05 } else { 0.04 },
322                        y: y + offset,
323                    });
324                }
325                annotations.push(PlotDocumentAnnotation {
326                    id: format!("polarization-{table_id}-{}", data.receptor),
327                    x: if identity_overlay && points && !lines {
328                        0.07
329                    } else {
330                        0.05
331                    },
332                    y,
333                    text,
334                    style: Some(PlotTextStyle {
335                        axes_relative: true,
336                        horizontal_alignment: Some("left".into()),
337                        vertical_alignment: Some("top".into()),
338                        font_size: None,
339                        background: None,
340                    }),
341                });
342            }
343            PlotDocumentLayer {
344                id: format!("series-{index}"),
345                title: String::new(),
346                kind: PlotLayerKind::Line,
347                x_axis_id: "x".into(),
348                y_axis_id: "y".into(),
349                x_values: data.curve.x.clone(),
350                y_values: data.curve.y.clone(),
351                point_labels: Vec::new(),
352                point_symbol_sizes: Vec::new(),
353                interval_x_start: Vec::new(),
354                interval_x_end: Vec::new(),
355                interval_y: Vec::new(),
356                interval_height: Vec::new(),
357                interval_labels: Vec::new(),
358                provenance: Vec::new(),
359                color_group: color.into(),
360                symbol_size: markersize,
361                line_width: 1.0,
362                opacity: 1.0,
363                source_sample_count: data.curve.x.len() as u64,
364                payload_strategy: PlotPayloadStrategy::PointCloud,
365                provenance_summary: String::new(),
366                data: None,
367                style: Some(PlotLayerStyle {
368                    color: Some(color.into()),
369                    marker: Some(marker.into()),
370                    line_style: Some(
371                        (if dashed {
372                            "--"
373                        } else if lines {
374                            "-"
375                        } else {
376                            "None"
377                        })
378                        .into(),
379                    ),
380                }),
381            }
382        })
383        .collect();
384    (
385        PlotDocumentPanel {
386            id: format!("panel-{panel_number}"),
387            title: bandpass.title.clone(),
388            title_font_size: None,
389            axes: vec![
390                axis("x", &bandpass.x_label, x_limits(bandpass, display)),
391                axis(
392                    "y",
393                    bandpass.y_label,
394                    y_limits(bandpass, display).unwrap_or([0.0, 1.0]),
395                ),
396            ],
397            layers,
398            annotations,
399            aspect: None,
400            frame_visible: None,
401        },
402        paint.legend,
403    )
404}

406struct PanelColors<'a> {
407    bandpass: &'a BandpassPanel,
408    antennas: Vec<&'a str>,
409    first_time: usize,
410    legend: OverlayLegend,
411}
412
413impl<'a> PanelColors<'a> {
414    fn new(bandpass: &'a BandpassPanel, layout: [u32; 2]) -> Self {
415        let mut antennas = Vec::new();
416        for series in &bandpass.series {
417            let name = series.antenna_name.as_str();
418            if !antennas.contains(&name) {
419                antennas.push(name);
420            }
421        }
422        let mut legend = OverlayLegend::default();
423        if bandpass.overlay.antenna && !bandpass.overlay.time {
424            legend
425                .identities
426                .extend(
427                    antennas
428                        .iter()
429                        .enumerate()
430                        .map(|(index, name)| LegendIdentity {
431                            text: (*name).into(),
432                            color: identity_color(index),
433                            x: 0.07 + 0.05 * index.min(17) as f64,
434                            y: 0.931 - 0.018 * index.saturating_sub(17) as f64,
435                        }),
436                );
437        } else if bandpass.overlay.time {
438            legend.identities.push(LegendIdentity {
439                text: "UT".into(),
440                color: "k",
441                x: 0.04,
442                y: 0.931,
443            });
444            let spacing = if layout[1] == 1 { 0.078 } else { 0.065 };
445            legend
446                .identities
447                .extend(bandpass.legend_times.iter().enumerate().map(
448                    |(position, &(index, epoch))| {
449                        let date = format_time(epoch / 86_400.0, YMD, 0);
450                        LegendIdentity {
451                            text: date.get(11..).unwrap_or(&date).into(),
452                            color: if bandpass.overlay.antenna {
453                                "k"
454                            } else {
455                                identity_color(index)
456                            },
457                            x: 0.07 + spacing * position.min(12) as f64,
458                            y: 0.931 - 0.018 * position.saturating_sub(12) as f64,
459                        }
460                    },
461                ));
462        }
463        let first_time = bandpass.legend_times.first().map_or(0, |(index, _)| *index);
464        Self {
465            bandpass,
466            antennas,
467            first_time,
468            legend,
469        }
470    }
471
472    fn color(&self, series: &BandpassSeries, index: usize) -> &'static str {
473        let overlay = self.bandpass.overlay;
474        let antenna = || {
475            self.antennas
476                .iter()
477                .position(|name| *name == series.antenna_name)
478                .unwrap_or(0)
479        };
480        if overlay.antenna && overlay.time {
481            if series.time_index == self.first_time {
482                return ["b", "g", "r", "c"][index % 4];
483            }
484            return identity_color(antenna());
485        }
486        if overlay.antenna {
487            return identity_color(antenna());
488        }
489        if overlay.time {
490            return identity_color(series.time_index);
491        }
492        [["b", "g"], ["k", "c"], ["m", "r"]][series.table_index][series.receptor % 2]
493    }
494}

509fn x_limits(panel: &BandpassPanel, display: &DisplayOptions<'_>) -> [f64; 2] {
510    let requested = display
511        .limit_channels
512        .then(|| {
513            display
514                .channel_range
515                .and_then(|range| range.frequencies(&panel.frequency_channels))
516        })
517        .flatten();
518    let [lower, upper] = requested.unwrap_or([display.limits[0], display.limits[1]]);
519    if lower == 0.0 && upper == 0.0 {
520        return panel.x_limits;
521    }
522    let range = (upper - lower).abs();
523    let pad = if range == 0.0 { 1e-5 } else { 0.01 * range };
524    [lower.min(upper) - pad, lower.max(upper) + pad]
525}
526
527fn y_limits(panel: &BandpassPanel, display: &DisplayOptions<'_>) -> Option<[f64; 2]> {
528    let [_, _, lower, upper] = display.limits;
529    if lower < upper {
530        return Some([lower, upper]);
531    }
532    if panel.y_label == "Phase (deg)" {
533        if let Some(limits) = display.phase_limits {
534            return Some(limits);
535        }
536        if let Some(first) = panel.series.first()
537            && first.curve.y.iter().all(|sample| sample.abs() < 0.2)
538        {
539            return Some([-0.2, 0.2]);
540        }
541    }
542    let channel_limits = display
543        .channel_range
544        .and_then(|range| range.frequencies(&panel.frequency_channels));
545    let requested = channel_limits.or_else(|| {
546        let [lower, upper, ..] = display.limits;
547        (!panel.frequency_channels.is_empty() && (lower != 0.0 || upper != 0.0))
548            .then_some([lower, upper])
549    })?;
550    let lower = requested[0].min(requested[1]);
551    let upper = requested[0].max(requested[1]);
552    let mut values = panel.series.iter().flat_map(|series| {
553        series
554            .curve
555            .x
556            .iter()
557            .zip(&series.curve.y)
558            .filter_map(move |(&x, &y)| (x >= lower && x <= upper && y.is_finite()).then_some(y))
559    });
560    let first = values.next()?;
561    let [min, max] = values.fold([first, first], |[min, max], y| [min.min(y), max.max(y)]);
562    (min < max).then_some([min, max])
563}
