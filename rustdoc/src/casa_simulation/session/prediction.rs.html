2//! `sm.predict`, `sm.setoptions` and `sm.setvp`: the model DATA is predicted
3//! from.
4//!
5//! As in CASA's `Simulator`:
6//! - `setoptions` and `setvp` only record; `predict` reads them.
7//! - `predict` reads the model image when it is called and predicts it into
8//!   the rows observed so far: replacing their DATA, or adding to it when
9//!   `incremental`. Scans observed later are not predicted by it.
10//! - The primary beam (`setvp(dovp=True)`) is the default voltage pattern of
11//!   the telescope of the first `observe` (OBSERVATION row 0), centred on
12//!   each field.
13//!
14//! The model is read when `predict` is called, so a model changed or
15//! deleted afterwards predicts as in CASA and errors surface at the call:
16//! - Without the beam, or under it with one field observed, it is prepared
17//!   then (multiplied by the beam and transformed into one set of grids).
18//!   The grids are admitted against the session's memory budget and held
19//!   until the step is replaced or the session closes.
20//! - Under the beam with several fields, each field's grids have the beam
21//!   centred on that field, and one set per field may not fit the budget
22//!   (a 106-pointing mosaic of a 1873² model needed 5.3 GB). The step holds
23//!   the model's planes instead, and `close` prepares each field's grids as
24//!   its write reaches the field, keeping as many as its budget allows and
25//!   preparing a field again when it returns, as CASA's `GridFT` transforms
26//!   the model again whenever the field changes.
27//!
28//! `close` degrids the grids into the rows it writes.
29//!
30//! Divergences: CASA rewrites the model image's spectral units and frame in
31//! place; casa-rs never writes it (`docs/CASA (C++) bugs.md`). MODEL_DATA
32//! and CORRECTED_DATA are not created (plan decision 6), even for an
33//! incremental predict, whose sum CASA forms through MODEL_DATA; a
34//! MeasurementSet `openfromms` opened keeps those it has in step with DATA.

97    /// `incremental`: add the prediction to DATA rather than replace it.

112    /// The check casatools' binding makes before `Simulator::predict`, even
113    /// before `open`: every named image exists and opens as an image
114    /// (`ImageFactory::fromFile`). Here it must open as the simulator reads
115    /// it, a CASA image of `Float` pixels: CASA's binding also accepts FITS
116    /// and complex images, which its simulator then refuses with `False`
117    /// (`docs/CASA (C++) bugs.md`).
118    ///
119    /// # Errors
120    ///
121    /// Raised: "File … does not exist." or "Unable to open image …".
122    pub fn check_images(&self) -> SimulatorResult<()> {
123        for path in self.named_images() {
124            if !path.exists() {
125                return Err(SimulatorError::raised(format!(
126                    "File {} does not exist.",
127                    path.display()
128                )));
129            }
130            if SkyModel::open(path).is_err_and(|error| matches!(error, PredictError::Image { .. }))
131            {
132                return Err(SimulatorError::raised(format!(
133                    "Unable to open image {}.",
134                    path.display()
135                )));
136            }
137        }
138        Ok(())
139    }

142/// The arguments of `sm.setoptions` that casa-rs reads. `cache`, `tile`,
143/// `facets` and `maxdata` do not change the supported prediction path;
144/// the session's memory budget stands for `cache`.
145#[derive(Debug, Clone, PartialEq)]
146pub struct PredictionOptions {
147    /// `ftmachine`: `WProjectFT` for `wproject`, `SDGrid` for `sd`, and
148    /// `GridFT` unless `both`, `mosaic` or `pbwproject` (case-insensitive).
149    pub ftmachine: String,
150    /// `gridfunction`: `SF`, the spheroid, for `GridFT` and `WProjectFT`;
151    /// `PB`, `BOX` or `SF` for `SDGrid`.
152    pub grid_function: String,
153    /// `padding`, `GridFT`'s; factors below 1 are ignored, as CASA ignores
154    /// them. `WProjectFT` pads by 1 whatever it is.
155    pub padding: f64,
156    /// `wprojplanes`, `WProjectFT`'s planes.
157    pub w_planes: i64,
158    /// `location`: the FTMachine's reference position in ITRF metres,
159    /// used when converting native image direction frames. Defaults to
160    /// the geocentre, as CASA's unset `MPosition` does.
161    pub location_itrf_m: [f64; 3],
162}

240/// The grids a step predicts from.
241enum StepGrids {
242    /// Prepared at `predict`: one set for every field without a beam
243    /// (`None`), or for the one field observed under it.
244    Prepared(Option<usize>, Box<PreparedModelGrids>),
245    /// One set per observed field under the beam, centred on the field,
246    /// which `close` prepares ([`Prediction::prepare_field`]).
247    PerField(Box<FieldModel>),
248}
249
250/// What `close` prepares a field's grids from: the model with its planes
251/// held, the beam's power pattern, and each observed field's beam centre
252/// on the model's pixels, by field.
253struct FieldModel {
254    model: SkyModel,
255    pattern: BeamPattern,
256    centres: Vec<([u64; 2], [f64; 2])>,
257}

259/// Where the grids of a field come from: a set prepared at `predict`, or
260/// one `close` prepares ([`Prediction::prepare_field`]).
261pub(crate) enum FieldGrids<'p> {
262    Prepared(&'p PreparedModelGrids),
263    PerField,
264}
265
266impl std::fmt::Debug for Prediction {
267    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
268        let fields: Vec<Option<usize>> = match &self.grids {
269            StepGrids::Prepared(field, _) => vec![*field],
270            StepGrids::PerField(held) => held
271                .centres
272                .iter()
273                .enumerate()
274                .map(|(index, _)| Some(index))
275                .collect(),
276        };
277        formatter
278            .debug_struct("Prediction")
279            .field("fields", &fields)
280            .field("per_field", &matches!(self.grids, StepGrids::PerField(_)))
281            .finish_non_exhaustive()
282    }
283}

295    pub(crate) fn has_selected_chunk(&self, scan: &super::observe::Scan) -> bool {
296        self.tangent.refer() == casa_types::measures::direction::DirectionRef::J2000
297            || super::from_ms::polynomial::epoch(self.phase_centres.as_ref(), scan).is_some()

300    pub(crate) fn frame(
301        &self,
302        scan: &super::observe::Scan,
303    ) -> SimulatorResult<casa_types::measures::MeasFrame> {
304        use casa_types::measures::{EpochRef, MEpoch, MeasFrame};
305        let frame = MeasFrame::new()
306            .with_measures(self.frame_measures.clone())
307            .with_position({
308                let [x, y, z] = self.frame_position_itrf_m;
309                casa_types::measures::MPosition::new_itrf(x, y, z)
310            });
311        if self.tangent.refer() == casa_types::measures::direction::DirectionRef::J2000 {
312            return Ok(frame);
313        }
314        let time = super::from_ms::polynomial::epoch(self.phase_centres.as_ref(), scan)
315            .ok_or_else(|| SimulatorError::raised("missing selected image-frame epoch"))?;
316        Ok(frame.with_epoch(MEpoch::from_mjd(time / 86400., EpochRef::UTC)))
317    }

323    /// Index the grids by antenna beam direction, independently of FIELD.

325        match &self.grids {
326            StepGrids::Prepared(..) => 0,

336    /// Where the grids rows of `field` are predicted from come from; `None`
337    /// for a field the step did not observe.
338    pub(crate) fn grids_for(&self, field: usize) -> Option<FieldGrids<'_>> {
339        match &self.grids {
340            StepGrids::Prepared(grid_field, grids) => grid_field
341                .is_none_or(|grid_field| grid_field == field)
342                .then_some(FieldGrids::Prepared(grids)),
343            StepGrids::PerField(held) => held.centres.get(field).map(|_| FieldGrids::PerField),
344        }

362    /// Bytes one set of grids holds ([`PreparedModelGrids::bytes`]).

367    /// Bytes preparing a field's grids holds beside them on `workers`
368    /// workers: a plane's scratch for each worker that has a plane
369    /// ([`ModelOperator::plane_scratch_bytes`]); none when `close` prepares
370    /// no grids.
371    pub(crate) fn preparation_bytes(&self, workers: usize) -> u64 {
372        match &self.grids {
373            StepGrids::Prepared(..) => 0,
374            StepGrids::PerField(held) => {
375                workers.clamp(1, held.model.planes().max(1)) as u64
376                    * self.model.plane_scratch_bytes(&held.model)
377            }
378        }
379    }

381    /// Prepare the grids of `field`, which the step observed under the
382    /// beam, on `team`: the planes held, multiplied by the beam's power
383    /// centred on the field and transformed, as `predict` prepares one
384    /// field's ([`ModelOperator::prepare`]); the same grids whatever the
385    /// workers.
386    ///
387    /// # Errors
388    ///
389    /// A raised error for a field the step did not observe or a step whose
390    /// grids were prepared at `predict`, and the operator's refusal, which
391    /// for a model `predict` read is only a transform's failure.
392    pub(crate) fn prepare_field(
393        &self,
394        field: usize,
395        team: &WorkerTeam,
396    ) -> SimulatorResult<PreparedModelGrids> {
397        let StepGrids::PerField(held) = &self.grids else {
398            return Err(SimulatorError::raised(
399                "the step's grids were prepared at predict",
400            ));
401        };
402        let centre = held
403            .centres
404            .get(field)
405            .map(|(_, centre)| *centre)
406            .ok_or_else(|| {
407                SimulatorError::raised(format!("predict did not observe field {field}"))
408            })?;
409        self.model
410            .prepare(&held.model, Some((&held.pattern, centre)), Some(team))
411            .map_err(model_error)
412    }

414    /// The offset of rows pointed at `centre_rad`, which the mosaic
415    /// machine's kernels ramp by; zero for the other machines.
416    pub(crate) fn pointing_offset_rad(&self, centre_rad: [f64; 2]) -> [f64; 2] {
417        let key = centre_rad.map(f64::to_bits);
418        let found = self
419            .pointings
420            .binary_search_by_key(&key, |(known, _)| *known)
421            .ok()
422            .map(|index| &self.pointings[index]);
423        // A mosaic step measured every scan it covers.
424        debug_assert!(
425            found.is_some() || self.pointings.is_empty(),
426            "a mosaic step predicts rows pointed at {centre_rad:?}, which it did not measure"
427        );
428        found.map_or([0.0; 2], |(_, offset)| *offset)
429    }

438impl Simulator {
439    /// `sm.setoptions(...)`: record the FTMachine `predict` degrids through,
440    /// `GridFT`'s padding and `WProjectFT`'s planes. Each call sets the
441    /// FTMachine anew, as CASA's does.
442    ///
443    /// # Errors
444    ///
445    /// Not yet implemented: an FTMachine other than `GridFT`, `WProjectFT`,
446    /// `MosaicFTNew` and `SDGrid` (`both`, `pbwproject`), a grid function
447    /// other than `SF` for the interferometric machines, or other than `PB`,
448    /// `BOX` and `SF` for `SDGrid`. Nothing is recorded then.
449    pub fn set_options(&mut self, options: &PredictionOptions) -> SimulatorResult<()> {
450        let ftmachine = options.ftmachine.to_lowercase();
451        if OTHER_FTMACHINES.contains(&ftmachine.as_str()) {
452            return Err(SimulatorError::not_yet_implemented(format!(
453                "setoptions(ftmachine='{ftmachine}') is not yet implemented in casa-rs (#1053)"
454            )));
455        }
456        let single_dish = if ftmachine == "sd" {
457            Some(
458                SingleDishKernel::from_grid_function(&options.grid_function).ok_or_else(|| {
459                    SimulatorError::not_yet_implemented(format!(
460                        "setoptions(ftmachine='sd', gridfunction='{}') is not yet implemented \
461                         in casa-rs (#1053); 'PB', 'BOX' and 'SF' are",
462                        options.grid_function
463                    ))
464                })?,
465            )
466        } else {
467            if options.grid_function.to_lowercase() != "sf" {
468                return Err(SimulatorError::not_yet_implemented(format!(
469                    "setoptions(gridfunction='{}') is not yet implemented in casa-rs (#1053); \
470                     'SF' is",
471                    options.grid_function
472                )));
473            }
474            None
475        };
476        tracing::info!("Setting processing options");
477        // CASA keeps a padding of at least 1 and ignores the rest.
478        let padding = options.padding as f32;
479        if padding >= 1.0 {
480            self.padding = PaddingFactor::new(padding)
481                .map_err(|error| SimulatorError::raised(format!("padding {padding}: {error}")))?;
482        }
483        self.w_projection_planes = (ftmachine == "wproject").then_some(options.w_planes);
484        self.mosaic = ftmachine == "mosaic";
485        self.single_dish = single_dish;
486        self.prediction_location_itrf_m = options.location_itrf_m;
487        Ok(())
488    }
489
490    /// The FTMachine `predict` degrids through, as `setoptions` chose it.
491    ///
492    /// # Errors
493    ///
494    /// - CASA's `False`: `WProjectFT` with one plane, whose convolution
495    ///   function CASA's `WPConvFunc` cannot normalise.
496    /// - Raised: `WProjectFT` with no planes or fewer, where CASA's
497    ///   simulator crashes (`docs/CASA (C++) bugs.md`), or more than
498    ///   casa-rs's [`MAX_W_PLANES`] (65,536).
499    /// - Not yet implemented: `WProjectFT` on the Metal backend.
500    fn ft_machine(&self) -> SimulatorResult<FtMachine> {
501        if self.mosaic {
502            return Ok(FtMachine::Mosaic);
503        }
504        let Some(planes) = self.w_projection_planes else {
505            return Ok(FtMachine::Grid(self.padding));
506        };
507        match planes {
508            ..=0 => Err(SimulatorError::raised(format!(
509                "setoptions(ftmachine='wproject', wprojplanes={planes}): W projection needs at \
510                 least two planes"
511            ))),
512            1 => Err(SimulatorError::refused(
513                "Caught exception: Convolution function integral is not positive",
514            )),
515            _ => {
516                let planes = u32::try_from(planes)
517                    .ok()
518                    .filter(|&planes| planes as usize <= MAX_W_PLANES)
519                    .ok_or_else(|| {
520                        SimulatorError::raised(format!(
521                            "setoptions(ftmachine='wproject', wprojplanes={planes}): casa-rs \
522                             builds at most 65,536 W planes"
523                        ))
524                    })?;
525                if self.options.resources.backend() == BackendRequest::Metal {
526                    return Err(SimulatorError::not_yet_implemented(
527                        "predict with setoptions(ftmachine='wproject') on the Metal backend is \
528                         not yet implemented in casa-rs (#1053); backend 'auto' or 'cpu' \
529                         predicts it on the CPU",
530                    ));
531                }
532                Ok(FtMachine::WProject(planes))
533            }
534        }
535    }

537    /// `sm.setvp(...)`: record whether `predict` applies the primary beam.
538    ///
539    /// # Errors
540    ///
541    /// Admission fails before retaining the optional voltage-pattern path.
542    /// Like CASA, the table itself is read by `predict`.
543    pub fn set_vp(&mut self, spec: &VoltagePatternSpec) -> SimulatorResult<()> {
544        let table = super::voltage_patterns::capture_path(self, spec)?;
545        tracing::info!("Setting voltage pattern parameters");
546        if spec.use_default_vp {
547            tracing::info!("Using system default voltage patterns for each telescope");
548        }
549        if spec.do_squint {
550            tracing::info!("Beam Squint will be included in the VP model");
551            tracing::info!(
552                "and the parallactic angle increment is {} degrees",
553                spec.parallactic_angle_increment_deg
554            );
555        }
556        self.voltage_pattern = spec.dovp;
557        self.voltage_pattern_table = table;
558        self.squint = spec.do_squint;
559        self.beam_angle_increment_rad = spec.parallactic_angle_increment_deg.to_radians();
560        Ok(())
561    }

563    /// `sm.predict(imagename, complist, incremental)`: predict the model
564    /// into the rows observed so far, replacing their DATA or, when
565    /// `incremental`, adding to it.
566    ///
567    /// # Errors
568    ///
569    /// - Raised, as casatools' binding raises before predicting: a named
570    ///   image that does not exist or does not open
571    ///   ([`PredictSpec::check_images`]).
572    /// - Not yet implemented: a model layout CASA reads
573    ///   and casa-rs does not (other than Stokes I/Q/U/V, J2000,
574    ///   direction–Stokes–spectral axes), a component casa-rs does not
575    ///   predict (a direction not in J2000, or a flux, spectrum or shape
576    ///   casa-components does not evaluate), scans whose windows differ in
577    ///   correlations, or
578    ///   `setvp(dovp=True)` for a telescope whose default pattern casa-rs
579    ///   does not have (it supports ALMA, ACA, VLA, EVLA, ngVLA, ATA and SMA), or
580    ///   a fresh total-power prediction with a displaced beam pointing.
581    /// - CASA's `False`: no model image ("No model images found"), or no
582    ///   rows observed yet.
583    /// - Raised: a scan of 3 correlations ("No such VisVector type"), or
584    ///   model grids or kernels, or a component list's read or preparation,
585    ///   that do not fit the session's memory budget
586    ///   ([`SimulatorError::admission_refused`]).
587    /// - As the FTMachine `setoptions` chose refuses: `WProjectFT` with
588    ///   fewer than two planes, or on the Metal backend.
589    pub fn predict(&mut self, spec: &PredictSpec) -> SimulatorResult<()> {
590        spec.check_images()?;
591        let images = spec.named_images();
592        let lists = spec.named_lists();
593        tracing::info!(
594            "Predicting visibilities using model: {images:?} and componentList: {}",
595            list_label(&lists)
596        );
597        if spec.incremental {
598            tracing::info!("The data column will be incremented");
599        } else {
600            tracing::info!("The data column will be replaced");
601        }
602        if images.is_empty() && lists.is_empty() {
603            return Err(SimulatorError::refused(
604                "Failed to create SkyEquation: No model images found",
605            ));
606        }
607        // `createSkyEquation` reads the component list before the images.
608        let list = read_component_lists(&lists, &self.options.resources)?;
609        let model = images
610            .first()
611            .map(|path| SkyModel::open(path).map_err(model_error))
612            .transpose()?;
613        let mut frame_epoch = model.as_ref().is_some_and(|model| {
614            model.tangent_direction().refer()
615                != casa_types::measures::direction::DirectionRef::J2000
616        });
617        // Validate every header before changing the DATA plan. Later
618        // images are reopened for preparation, one at a time, so the
619        // input count does not keep that many open images or tile caches.
620        for path in images.iter().skip(1) {
621            frame_epoch |= SkyModel::open(path)
622                .map_err(model_error)?
623                .tangent_direction()
624                .refer()
625                != casa_types::measures::direction::DirectionRef::J2000;
626        }
627        if self.planned_rows() == 0 {
628            return Err(SimulatorError::refused(
629                "Failed to create SkyEquation: Caught exception: MSIter::construct - Input \
630                 MeasurementSet.has zero selected rows",
631            ));
632        }
633        let correlations = self.predicted_correlations()?;
634        if self.single_dish.is_some()
635            && self.scans.iter().any(|scan| {
636                self.correlations_for_scan(scan) != self.correlations_for_scan(&self.scans[0])
637            })
638        {
639            return Err(SimulatorError::not_yet_implemented(
640                "total-power prediction over different correlation setups is not yet implemented in casa-rs (#1299)",
641            ));
642        }
643        let selected_beam = self.voltage_pattern_beam()?;
644        let component_beam = selected_beam.as_ref().filter(|_| self.voltage_pattern);
645        let beam = component_beam.map(|beam| &beam.pattern);
646        if (self.mosaic && model.is_some()) || beam.is_some() || self.single_dish.is_some() {
647            self.load_pointings()?;
648        }
649        // Interferometric images/components and mosaic kernels keep phase
650        // and pointing separately. Fresh total-power beam offsets await
651        // their dedicated integration controls.
652        if beam.is_some() && self.single_dish.is_some() {
653            self.check_pointings_on_fields("setvp(dovp=True)")?;
654        }
655        if let Some(kernel) = self.single_dish {
656            // Total power samples a model image; `SDGrid` with a component
657            // list is not yet here.
658            if model.is_none() || list.is_some() {
659                return Err(SimulatorError::not_yet_implemented(
660                    "predict(complist=…) after setoptions(ftmachine='sd') is not yet \
661                     implemented in casa-rs (#1053); a model image is",
662                ));
663            }
664            let beam = beam
665                .map(|beam| {
666                    beam.configured().ok_or_else(|| {
667                        SimulatorError::not_yet_implemented(
668                            "single-dish consumption of saved radial curves (#1299)",
669                        )
670                    })
671                })
672                .transpose()?;
673            let predictions = opened_models(model, &images)
674                .map(|model| self.prepare_single_dish(model?, kernel, beam))
675                .collect::<SimulatorResult<Vec<_>>>()?;
676            self.data_plan.begin_prediction(spec.incremental);
677            for prediction in predictions {
678                self.data_plan
679                    .predict_single_dish(Arc::new(prediction), self.scans.len());
680            }
681            return Ok(());
682        }
683        let component_frame_epoch = (self.squint
684            && beam.is_some_and(|beam| beam.configured().is_some_and(PrimaryBeam::squints)))
685            || list.as_ref().is_some_and(|(_, list, _)| {
686                list.components().iter().any(|component| {
687                    component.shape().reference_direction().refer()
688                        != casa_types::measures::direction::DirectionRef::J2000
689                })
690            });
691        let phase_centres = if list.is_some() {
692            super::from_ms::polynomial::selected_centres(self, false, component_frame_epoch)?
693        } else {
694            None
695        };
696        let components = list
697            .map(|(label, list, held)| {
698                self.prepare_components(
699                    &label,
700                    (list, held),
701                    component_beam,
702                    phase_centres.as_ref(),
703                )
704            })
705            .transpose()?;
706        let image_centres = if model.is_some() {
707            super::from_ms::polynomial::selected_centres(self, true, frame_epoch)?
708        } else {
709            None
710        };
711        let mut predictions = Vec::with_capacity(images.len() * correlations.len());
712        let mut first_model = model;
713        for (group, correlations) in correlations.iter().enumerate() {
714            let first = if group == 0 {
715                first_model.take()
716            } else {
717                images
718                    .first()
719                    .map(|path| SkyModel::open(path).map_err(model_error))
720                    .transpose()?
721            };
722            for model in opened_models(first, &images) {
723                predictions.push(self.prepare_image(
724                    model?,
725                    correlations,
726                    selected_beam.as_ref(),
727                    image_centres.clone(),
728                )?);
729            }
730        }
731        if predictions
732            .iter()
733            .any(|prediction| prediction.distance_m != 0.)
734        {
735            let focus = super::focus::capture(self)?;
736            for prediction in &mut predictions {
737                if prediction.distance_m != 0. {
738                    prediction.focus = Some(focus.clone());
739                }
740            }
741        }
742        self.data_plan.begin_prediction(spec.incremental);
743        let scans = self.scans.len();
744        if let Some(components) = components {
745            self.data_plan
746                .predict_components(Arc::new(components), scans);
747        }
748        for prediction in predictions {
749            self.data_plan.predict(Arc::new(prediction), scans);
750        }
751        Ok(())
752    }

754    fn prediction_frame(
755        &self,
756        phase_centres: Option<&super::from_ms::polynomial::PhaseCentres>,
757    ) -> casa_types::measures::MeasFrame {
758        use casa_types::measures::{EpochRef, MEpoch, MPosition, MeasFrame};
759        let mut frame = MeasFrame::new()
760            .with_measures(self.options.measures.clone())
761            .with_position({
762                let [x, y, z] = self.prediction_location_itrf_m;
763                MPosition::new_itrf(x, y, z)
764            });
765        if let Some(time) = self
766            .scans
767            .iter()
768            .find_map(|scan| super::from_ms::polynomial::epoch(phase_centres, scan))
769        {
770            frame = frame.with_epoch(MEpoch::from_mjd(time / 86400., EpochRef::UTC));
771        }
772        frame
773    }

775    /// `model` admitted and prepared for rows with `correlations` through
776    /// the FTMachine `setoptions` chose: its kernels, then its grids, with
777    /// `beam` in the image plane except through the mosaic machine, whose
778    /// kernels carry the beams and which keeps each scan's pointing and
779    /// each antenna's dish class. The model, with its image's tile cache,
780    /// is dropped once it is read ([`Self::prepare`]).
781    fn prepare_image(
782        &self,
783        model: SkyModel,
784        correlations: &[Correlation],
785        selected_beam: Option<&super::voltage_patterns::ResolvedBeam>,
786        phase_centres: Option<super::from_ms::polynomial::PhaseCentres>,
787    ) -> SimulatorResult<Prediction> {
788        use casa_types::measures::direction::DirectionRef;
789        let beam = selected_beam.map(|beam| &beam.pattern);
790        let fixed_celestial_frame = matches!(
791            model.tangent_direction().refer(),
792            DirectionRef::J2000
793                | DirectionRef::B1950
794                | DirectionRef::B1950Vla
795                | DirectionRef::GALACTIC
796                | DirectionRef::ICRS
797                | DirectionRef::SUPERGAL
798                | DirectionRef::ECLIPTIC
799        );
800        if self.mosaic && !fixed_celestial_frame {
801            return Err(SimulatorError::not_yet_implemented(
802                "mosaic prediction with a time-dependent image direction frame needs \
803                 a valid CASA reference (#1298)",
804            ));
805        }
806        if beam.is_some() && !fixed_celestial_frame {
807            return Err(SimulatorError::not_yet_implemented(
808                "primary-beam prediction with a time-dependent image direction frame needs \
809                 the beam's actual array observer context (#1297)",
810            ));
811        }
812        let frame = self.prediction_frame(phase_centres.as_ref());
813        let model = model.with_frame(frame);
814        let machine = self.ft_machine()?;
815        let mut mosaic_state = None;
816        let (beams, antenna_types, pointings) = match machine {
817            FtMachine::Mosaic => {
818                // What the machine keeps per pointing and antenna, and its
819                // beams' inputs until the kernels are built, admitted before
820                // they are formed.
821                let (kept, building) = self.mosaic_state_bytes();
822                let resources = &self.options.resources;
823                let reservation = admit(
824                    resources.host(),
825                    resources.policy(),
826                    &Demand {
827                        phase: MOSAIC_STATE_PHASE,
828                        memory: kept + building,
829                    },
830                )
831                .map_err(SimulatorError::admission)?;
832                mosaic_state = Some((reservation, kept));
833                let (beams, antenna_types) = self.mosaic_beams(
834                    selected_beam
835                        .filter(|beam| !beam.standard)
836                        .map(|beam| &beam.pattern),
837                )?;
838                (Some(beams), antenna_types, self.mosaic_pointings(&model)?)
839            }
840            FtMachine::Grid(_) | FtMachine::WProject(_) => {
841                if self
842                    .antennas
843                    .windows(2)
844                    .any(|pair| pair[0].diameter_m != pair[1].diameter_m)
845                {
846                    tracing::warn!(
847                        "Heterogeneous array is only supported for ALMA,ACA,OVRO telescopes, \
848                         and only with option ftmachine=mosaic."
849                    );
850                }
851                (None, Vec::new(), Vec::new())
852            }
853        };
854        let layout = ModelLayout::new(&model, correlations, machine, beams)
855            .map_err(|error| correlation_error(error, correlations))?;
856        let (kernels, mut kernel_reservation) = self.admitted_kernels(&layout)?;
857        // The operator drops the layout's beams.
858        let operator = ModelOperator::new(layout, kernels, GridPrecision::F32);
859        if let Some((mut reservation, kept)) = mosaic_state {
860            debug_assert_eq!(
861                (pointings.capacity() * size_of::<([u64; 2], [f64; 2])>()
862                    + antenna_types.capacity()) as u64,
863                kept,
864                "the mosaic state keeps what was admitted for it"
865            );
866            reservation.retain(kept);
867            kernel_reservation.join(reservation);
868        }
869        let image_beam = beam.filter(|_| machine != FtMachine::Mosaic);
870        let distance_m = self
871            .fields
872            .last()
873            .filter(|field| field.distance_m != 0.)
874            .map_or(model.distance_m, |field| field.distance_m);
875        let mut prediction = self.prepare(model, operator, image_beam, kernel_reservation)?;
876        prediction.distance_m = distance_m;
877        prediction.phase_centres = phase_centres;
878        prediction.pointings = pointings;
879        prediction.antenna_types = antenna_types;
880        Ok(prediction)
881    }

883    /// `list` prepared for the session's windows, with `beam` per observed
884    /// field, and admitted. The admission is made when the list has been
885    /// read: what the prediction will hold, known from the list's length,
886    /// and its preparation's scratch. The list's own share of the read's
887    /// admission, `held_list`, is released when the list is dropped here.
888    fn prepare_components(
889        &self,
890        label: &str,
891        (list, held_list): (ComponentList, Reservation),
892        selected_beam: Option<&super::voltage_patterns::ResolvedBeam>,
893        phase_centres: Option<&super::from_ms::polynomial::PhaseCentres>,
894    ) -> SimulatorResult<ComponentPrediction> {
895        let squint = selected_beam
896            .filter(|beam| self.squint && beam.standard)
897            .and_then(|beam| beam.pattern.configured())
898            .filter(|beam| beam.squints())
899            .map(|beam| -> SimulatorResult<_> {
900                Ok((
901                    beam,
902                    self.reference_frequency_hz(),
903                    self.beam_angle_increment_rad,
904                    super::voltage_patterns::squint_mount(self)?,
905                ))
906            })
907            .transpose()?;
908        let beam = selected_beam.map(|beam| &beam.pattern);
909        if self
910            .antennas
911            .windows(2)
912            .any(|pair| pair[0].diameter_m != pair[1].diameter_m)
913        {
914            tracing::warn!(
915                "Heterogeneous array is not supported for simulation from components.  \
916                 Primary beam will be applied by telescope name."
917            );
918        }
919        let resources = &self.options.resources;
920        let _target_memory = admit(
921            resources.host(),
922            resources.policy(),
923            &Demand {
924                phase: COMPONENTS_PHASE,
925                memory: (self.scans.len() as u64)
926                    .saturating_mul(128)
927                    .saturating_add((self.data_descriptions.len() as u64).saturating_mul(256))
928                    .saturating_add(128 << 10),
929            },
930        )
931        .map_err(SimulatorError::admission)?;
932        let windows: Vec<ComponentWindow> = self
933            .data_descriptions
934            .iter()
935            .map(|description| {
936                let window = &self.spectral_windows[description.window];
937                ComponentWindow {
938                    stored_frequencies: window.stored_frequencies.clone(),
939                    first_hz: window.start_frequency_hz,
940                    step_hz: window.channel_width_hz,
941                    channels: window.channel_count,
942                    correlations: self.polarizations[description.polarization].clone(),
943                }
944            })
945            .collect();
946        let pattern = beam.map(super::voltage_patterns::BeamSource::pattern);
947        // A scan is phased on its field's phase centre, as CASA's
948        // `VisBuffer` phases it, with the beam centred independently on
949        // POINTING. Only equal phase/beam pairs share prepared terms.
950        let mut pointings: Vec<(usize, [f64; 2], [f64; 2])> = Vec::new();
951        let pointing_of_scan: Vec<usize> = self
952            .scans
953            .iter()
954            .map(|scan| {
955                let direction = super::from_ms::polynomial::centre(phase_centres, scan);
956                let centre = direction.map(f64::to_bits);
957                pointings
958                    .iter()
959                    .position(|(_, known, beam)| {
960                        known.map(f64::to_bits) == centre
961                            && beam.map(f64::to_bits) == scan.pointing_rad.map(f64::to_bits)
962                    })
963                    .unwrap_or_else(|| {
964                        pointings.push((pointings.len(), direction, scan.pointing_rad));
965                        pointings.len() - 1
966                    })
967            })
968            .collect();
969        let [x, y, z] = self.antennas[0].position_itrf_m;
970        let frame = self
971            .prediction_frame(phase_centres)
972            .with_position(casa_types::measures::MPosition::new_itrf(x, y, z));
973        let targets = ComponentTargets {
974            windows: &windows,
975            pointings: &pointings,
976            pointing_of_scan: &pointing_of_scan,
977            beam: pattern.as_ref(),
978            frame: &frame,
979            squint,
980            workers: self
981                .options
982                .resources
983                .close_cores(
984                    self.options.resources.prediction_backend()
985                        == casa_imaging_runtime::pass::BackendChoice::Metal,
986                )
987                .streaming_workers,
988        };
989        let resources = &self.options.resources;
990        let preparation = ComponentTargets::preparation_bytes(list.len());
991        let mut reservation = admit(
992            resources.host(),
993            resources.policy(),
994            &Demand {
995                phase: COMPONENTS_PHASE,
996                memory: targets.bytes(list.len()) + preparation,
997            },
998        )
999        .map_err(SimulatorError::admission)?;
1000        let held_preparation = reservation.split(preparation);
1001        let prediction = ComponentPrediction::new(&list, targets, reservation).map_err(
1002            |ComponentRefusal::NotYetImplemented(reason)| {
1003                SimulatorError::not_yet_implemented(format!(
1004                    "predict(complist='{label}'): {reason} (#1053)"
1005                ))
1006            },
1007        )?;
1008        drop(held_preparation);
1009        tracing::info!(
1010            components = list.len(),
1011            bytes = prediction.bytes(),
1012            "prepared the component list {label}"
1013        );
1014        drop(list);
1015        drop(held_list);
1016        Ok(prediction)
1017    }

1019    /// The correlations of the scans observed so far, which one operator
1020    /// predicts.
1021    fn predicted_correlations(&self) -> SimulatorResult<Vec<Vec<Correlation>>> {
1022        self.check_visibility_correlations("predicted")?;
1023        let mut groups = Vec::<Vec<Correlation>>::new();
1024        for scan in &self.scans {
1025            if !self.data_plan.selection().contains_scan(scan) {
1026                continue;
1027            }
1028            let codes = self.correlations_for_scan(scan);
1029            if groups.iter().any(|group| {
1030                group
1031                    .iter()
1032                    .map(|c| c.casacore_code())
1033                    .eq(codes.iter().copied())
1034            }) {
1035                continue;
1036            }
1037            let group = codes.iter().map(|&code|Correlation::from_casacore_code(code).ok_or_else(|| {
1038                SimulatorError::not_yet_implemented(format!("predicting correlation code {code} is not yet implemented in casa-rs (#1295)"))
1039            })).collect::<SimulatorResult<Vec<_>>>()?;
1040            groups.push(group);
1041        }
1042        Ok(groups)
1043    }

1045    /// The primary beam `setvp` asks for: the default pattern of OBSERVATION
1046    /// row 0's telescope at spectral window 0's REF_FREQUENCY
1047    /// (`VPSkyJones`).
1048    fn voltage_pattern_beam(
1049        &self,
1050    ) -> SimulatorResult<Option<super::voltage_patterns::ResolvedBeam>> {
1051        if !self.voltage_pattern && !(self.mosaic && self.voltage_pattern_table.is_some()) {
1052            return Ok(None);
1053        }
1054        if let Some(path) = &self.voltage_pattern_table {
1055            return super::voltage_patterns::read(self, Path::new(&path.value)).map(Some);
1056        }
1057        let telescope = self.observing_telescope();
1058        let beam = PrimaryBeam::for_telescope(telescope, self.reference_frequency_hz())
1059            .ok_or_else(|| {
1060                SimulatorError::not_yet_implemented(format!(
1061                    "setvp(dovp=True) for telescope {telescope} is not yet implemented in \
1062                     casa-rs (#1297); the default patterns of ALMA, ACA, VLA, EVLA, NGVLA, ATA \
1063                     and SMA are"
1064                ))
1065            })?;
1066        Ok(Some(super::voltage_patterns::ResolvedBeam {
1067            pattern: super::voltage_patterns::BeamSource::Configured(beam),
1068            standard: true,
1069        }))
1070    }

1090    /// The remaining fresh total-power beam path requires pointings at
1091    /// their fields until its displaced-pointing consumer is implemented
1092    /// in SIM-12/22 (#1299). Interferometric prediction instead keeps the
1093    /// phase and beam directions separately, correcting CASA's field-only
1094    /// beam cache (documented in `docs/CASA (C++) bugs.md`).
1095    fn check_pointings_on_fields(&self, what: &str) -> SimulatorResult<()> {
1096        let moved = self.scans.iter().find(|scan| {
1097            !matches!(scan.rows, super::observe::ScanRows::OnDisk(_))
1098                && scan.pointing_rad != scan.phase_center_rad
1099        });
1100        match moved {
1101            Some(scan) => Err(SimulatorError::not_yet_implemented(format!(
1102                "predict with {what} for scan {} pointed away from its field {} is not yet \
1103                 implemented in casa-rs (#1053); pointings at their \
1104                 fields' centres are",
1105                scan.scan_number, self.fields[scan.field_id].name
1106            ))),
1107            None => Ok(()),
1108        }
1109    }

1111    /// Admit and prepare `model`'s grids: one set, without `beam` or with
1112    /// one field observed (the beam centred on it). With several fields
1113    /// under the beam, the model's planes are held for `close` to prepare
1114    /// each field's grids ([`Self::hold_for_fields`]). The model, and with
1115    /// it its image's tile cache, which the preparation's or the read's
1116    /// admission covers, is dropped before that admission is released. The
1117    /// prediction keeps `kernels`, the reservation of the operator's
1118    /// kernels.
1119    fn prepare(
1120        &self,
1121        model: SkyModel,
1122        operator: ModelOperator,
1123        beam: Option<&super::voltage_patterns::BeamSource>,
1124        kernels: Reservation,
1125    ) -> SimulatorResult<Prediction> {
1126        let resources = &self.options.resources;
1127        let (host, policy) = (resources.host(), resources.policy());
1128        charge_process_caches(host, policy).map_err(SimulatorError::admission)?;
1129        let point_memory = admit(
1130            host,
1131            policy,
1132            &Demand {
1133                phase: PREPARATION_PHASE,
1134                memory: if beam.is_some() {
1135                    (self.scans.len() as u64)
1136                        .saturating_mul(64)
1137                        .saturating_add(256)
1138                } else {
1139                    0
1140                },
1141            },
1142        )
1143        .map_err(SimulatorError::admission)?;
1144        let mut pointings: Vec<_> = if beam.is_some() {
1145            self.scans.iter().map(|scan| scan.pointing_rad).collect()
1146        } else {
1147            Vec::new()
1148        };
1149        pointings.sort_unstable_by_key(|direction| direction.map(f64::to_bits));
1150        pointings.dedup_by_key(|direction| direction.map(f64::to_bits));
1151        let _ = &point_memory;
1152        let pattern = beam.map(super::voltage_patterns::BeamSource::pattern);
1153        let tangent = model.tangent_direction().clone();
1154        let (grids, reservation) = match pattern {
1155            Some(pattern) if pointings.len() > 1 => {
1156                self.hold_for_fields(model, &operator, pattern, &pointings)?
1157            }
1158            pattern => {
1159                let field = None;
1160                let reservation = admit(
1161                    host,
1162                    policy,
1163                    &Demand {
1164                        phase: MODEL_PHASE,
1165                        memory: operator.grid_bytes()
1166                            + pattern.as_ref().map_or(0, |_| BeamPattern::BYTES),
1167                    },
1168                )
1169                .map_err(SimulatorError::admission)?;
1170                let scratch = operator.plane_scratch_bytes(&model);
1171                let free = free_memory(host, policy);
1172                let workers = (1..=resources.cores().workers.max(1))
1173                    .rev()
1174                    .find(|workers| *workers as u64 * scratch <= free)
1175                    .unwrap_or(1);
1176                let preparation = admit(
1177                    host,
1178                    policy,
1179                    &Demand {
1180                        phase: PREPARATION_PHASE,
1181                        memory: workers as u64 * scratch,
1182                    },
1183                )
1184                .map_err(SimulatorError::admission)?;
1185                let team = WorkerTeam::new(workers)
1186                    .map_err(|error| SimulatorError::raised(format!("the worker team: {error}")))?;
1187                let centre = pattern
1188                    .as_ref()
1189                    .map(|pattern| {
1190                        model
1191                            .pixel_of(pointings[0])
1192                            .map(|pixel| (pattern, pixel))
1193                            .map_err(model_error)
1194                    })
1195                    .transpose()?;
1196                let prepared = operator
1197                    .prepare(&model, centre, Some(&team))
1198                    .map_err(model_error)?;
1199                tracing::info!(
1200                    planes = model.planes(),
1201                    grids = 1,
1202                    workers,
1203                    bytes = operator.grid_bytes(),
1204                    "prepared the model {}",
1205                    model.path().display()
1206                );
1207                drop(model);
1208                drop(preparation);
1209                (StepGrids::Prepared(field, Box::new(prepared)), reservation)
1210            }
1211        };
1212        Ok(Prediction {
1213            model: operator,
1214            tangent,
1215            frame_measures: self.options.measures.clone(),
1216            frame_position_itrf_m: self.prediction_location_itrf_m,
1217            phase_centres: None,
1218            focus: None,
1219            distance_m: 0.,
1220            grids,
1221            pointings: Vec::new(),
1222            antenna_types: Vec::new(),
1223            _reservation: reservation,
1224            _kernels: kernels,
1225        })
1226    }
1227
1228    /// `model`'s planes read and held for `close` to prepare the grids of
1229    /// each of `fields` with `pattern` centred on it, each field's centre
1230    /// placed on the model's pixels: the planes and the pattern admitted
1231    /// and kept, the read's scratch (a plane, and the image's tile cache)
1232    /// admitted while it reads. One field's grids and their preparation on
1233    /// one worker, the least `close` holds, are admitted and released
1234    /// again, so a model too large for the budget is refused here, as when
1235    /// `predict` prepared every field's grids.
1236    fn hold_for_fields(
1237        &self,
1238        model: SkyModel,
1239        operator: &ModelOperator,
1240        pattern: BeamPattern,
1241        pointings: &[[f64; 2]],
1242    ) -> SimulatorResult<(StepGrids, Reservation)> {
1243        let resources = &self.options.resources;
1244        let (host, policy) = (resources.host(), resources.policy());
1245        let admitted = |phase, memory| {
1246            admit(host, policy, &Demand { phase, memory }).map_err(SimulatorError::admission)
1247        };
1248        let centres = pointings
1249            .iter()
1250            .map(|&pointing| {
1251                Ok((
1252                    pointing.map(f64::to_bits),
1253                    model.pixel_of(pointing).map_err(model_error)?,
1254                ))
1255            })
1256            .collect::<SimulatorResult<Vec<_>>>()?;
1257        let kept = model.held_bytes()
1258            + BeamPattern::BYTES
1259            + (size_of::<FieldModel>() + centres.capacity() * size_of::<([u64; 2], [f64; 2])>())
1260                as u64;
1261        let reservation = admitted(MODEL_PHASE, kept)?;
1262        drop(admitted(
1263            MODEL_PHASE,
1264            operator.grid_bytes() + operator.plane_scratch_bytes(&model),
1265        )?);
1266        let reading = admitted(PREPARATION_PHASE, model.plane_bytes() + model.cache_bytes())?;
1267        let held = model.held().map_err(model_error)?;
1268        drop(model);
1269        drop(reading);
1270        tracing::info!(
1271            planes = held.planes(),
1272            pointings = pointings.len(),
1273            bytes = held.held_bytes(),
1274            grid_bytes = operator.grid_bytes(),
1275            "held the model {} for each field's grids",
1276            held.path().display()
1277        );
1278        Ok((
1279            StepGrids::PerField(Box::new(FieldModel {
1280                model: held,
1281                pattern,
1282                centres,
1283            })),
1284            reservation,
1285        ))
1286    }
1287}

1322/// The component lists at `paths` as one list, their components in order,
1323/// with the label naming them and what [`read_component_list`] holds for
1324/// each; `None` for no lists.
1325fn read_component_lists(
1326    paths: &[&Path],
1327    resources: &SessionResources,
1328) -> SimulatorResult<Option<(String, ComponentList, Reservation)>> {
1329    let mut joined: Option<(ComponentList, Reservation)> = None;
1330    for path in paths {
1331        let (list, held) = read_component_list(path, resources)?;
1332        match &mut joined {
1333            None => joined = Some((list, held)),
1334            Some((components, reservation)) => {
1335                for component in list.into_components() {
1336                    components.push(component);
1337                }
1338                reservation.join(held);
1339            }
1340        }
1341    }
1342    Ok(joined.map(|(list, held)| (list_label(paths), list, held)))
1343}

1345/// The component list at `path`, as `createSkyEquation` reads it: CASA's
1346/// `False` when the path is not a readable table ("ComponentList … not
1347/// readable"), and when the table is not a component list CASA reads
1348/// (`ComponentList`'s constructor throws, and `predict` catches it). The
1349/// read is admitted within `resources` before it starts; once the list is
1350/// formed, the list's share of that admission is returned with it and the
1351/// rest released.
1352fn read_component_list(
1353    path: &Path,
1354    resources: &SessionResources,
1355) -> SimulatorResult<(ComponentList, Reservation)> {
1356    if !path.join("table.dat").is_file() {
1357        return Err(SimulatorError::refused(format!(
1358            "Failed to create SkyEquation: ComponentList {} not readable",
1359            path.display()
1360        )));
1361    }
1362    let refused = |error| SimulatorError::refused(format!("Caught exception: {error}"));
1363    let reading_bytes = ComponentList::read_table_bytes(path).map_err(refused)?;
1364    let mut reading = admit(
1365        resources.host(),
1366        resources.policy(),
1367        &Demand {
1368            phase: COMPONENTS_PHASE,
1369            memory: reading_bytes,
1370        },
1371    )
1372    .map_err(SimulatorError::admission)?;
1373    let list = ComponentList::read_table(path).map_err(refused)?;
1374    let held = reading.split(list.bytes().min(reading_bytes));
1375    Ok((list, held))
1376}

1378/// A model that cannot be read or prepared, as CASA reports it: a layout
1379/// CASA reads and casa-rs does not is not yet implemented; the rest is
1380/// raised.
1381fn model_error(error: PredictError) -> SimulatorError {
1382    match error {
1383        PredictError::InvalidModel(reason) => {
1384            SimulatorError::refused(format!("Caught exception: {reason}"))
1385        }
1386        PredictError::UnsupportedModel(reason) => SimulatorError::not_yet_implemented(format!(
1387            "predict from this model is not yet implemented in casa-rs (#1053): {reason}"
1388        )),
1389        PredictError::Image { path, .. } => {
1390            SimulatorError::raised(format!("Unable to open image {path}."))
1391        }
1392        other => SimulatorError::raised(format!("Caught exception: {other}")),
1393    }
1394}
