10pub(super) const SCALAR_ROWS: usize = 512;
11pub(super) const ARRAY_BYTES: usize = 256 * 1024;
12
13/// Encoded files bound strings when their manager cannot report header-only
14/// lengths. This is conservative for a table mixing strings with science data.
15pub(super) fn text_bound(path: &std::path::Path) -> SimulatorResult<usize> {
16    std::fs::read_dir(path)
17        .map_err(super::error)?
18        .try_fold(0usize, |total, entry| {
19            let metadata = entry
20                .map_err(super::error)?
21                .metadata()
22                .map_err(super::error)?;
23            if metadata.is_file() {
24                total
25                    .checked_add(usize::try_from(metadata.len()).map_err(super::error)?)
26                    .ok_or_else(|| super::error("encoded table size overflows usize"))
27            } else {
28                Ok(total)
29            }
30        })
31}

33/// The source descriptors, bounded source blocks and SSM save buffers coexist.
34/// Tiled reads can stage two tiles; scalar save's bound also covers its index
35/// and encoding buffers. Source bucket indices use at most two usize pairs per
36/// row; they are metadata, rather than copies of the old science arrays.
37pub(super) fn admit_copy(
38    template: &Table,
39    left: &Table,
40    right: &Table,
41    text_bound: usize,
42    resources: &SessionResources,
43) -> SimulatorResult<Reservation> {
44    let error = |e: casa_tables::TableError| SimulatorError::raised(e.to_string());
45    let metadata = |bytes: Option<usize>| {
46        bytes.ok_or_else(|| {
47            SimulatorError::raised("append source has unmodeled retained table state")
48        })
49    };
50    let template_bytes = metadata(template.retained_owned_metadata_bytes())?;
51    let mut bytes = template_bytes
52        + metadata(left.retained_read_metadata_bytes())?
53        + metadata(right.retained_read_metadata_bytes())?;
54    let columns = template.schema().expect("append schema").columns();
55    let text_factor = columns
56        .iter()
57        .filter(|column| column.data_type() == Some(PrimitiveType::String))
58        .map(|column| {
59            if matches!(column.column_type(), casa_tables::ColumnType::Array(_)) {
60                16
61            } else {
62                4
63            }
64        })
65        .max()
66        .unwrap_or(0);
67    bytes += text_bound
68        .checked_mul(text_factor)
69        .ok_or_else(|| super::error("append text memory overflows usize"))?;
70    bytes += casa_tables::streamed_scalar_columns_resident_bytes(columns.len())
71        .ok_or_else(|| SimulatorError::raised("append copy memory overflows usize"))?;
72    bytes += left.row_count().max(right.row_count()) * 4 * size_of::<usize>();
73    for column in columns {
74        bytes += match column.column_type() {
75            casa_tables::ColumnType::Array(_) => {
76                let tiles = left
77                    .array_column_tile_bytes(column.name())
78                    .map_err(error)?
79                    .unwrap_or(0)
80                    .max(
81                        right
82                            .array_column_tile_bytes(column.name())
83                            .map_err(error)?
84                            .unwrap_or(0),
85                    );
86                2 * tiles + 2 * ARRAY_BYTES + 128 * size_of::<Option<ArrayValue>>()
87            }
88            _ => 2 * SCALAR_ROWS * size_of::<Option<ScalarValue>>(),
89        };
90    }
91    // Save clones table descriptors while binding generated columns.
92    bytes += 2 * template_bytes;
93    admit(
94        resources.host(),
95        resources.policy(),
96        &Demand {
97            phase: "copying existing simulated observations",
98            memory: bytes as u64,
99        },
100    )
101    .map_err(SimulatorError::admission)
102}

104pub(super) struct Block<T> {
105    owner: usize,
106    first: usize,
107    values: Vec<Option<T>>,
108    extra: Reservation,
109}
110
111impl<T> Default for Block<T> {
112    fn default() -> Self {
113        Self {
114            owner: 0,
115            first: 0,
116            values: Vec::new(),
117            extra: Reservation::none(),
118        }
119    }
120}
121
122impl<T: Clone> Block<T> {
123    fn contains(&self, table: &Table, row: usize) -> bool {
124        self.owner == std::ptr::from_ref(table) as usize
125            && row >= self.first
126            && row - self.first < self.values.len()
127    }
128    fn store(&mut self, table: &Table, first: usize, values: Vec<Option<T>>) {
129        self.owner = std::ptr::from_ref(table) as usize;
130        self.first = first;
131        self.values = values;
132    }
133    fn value(&self, row: usize) -> Option<T> {
134        self.values[row - self.first].clone()
135    }
136}

138pub(super) fn scalar(
139    table: &Table,
140    key: &str,
141    row: usize,
142    cache: &Mutex<Block<ScalarValue>>,
143) -> Result<Option<ScalarValue>, TableError> {
144    let mut cache = cache.lock().expect("append scalar block");
145    if !cache.contains(table, row) {
146        let indices: Vec<_> = (row..table.row_count().min(row + SCALAR_ROWS)).collect();
147        cache.store(
148            table,
149            row,
150            table
151                .column_accessor(key)?
152                .scalar_cells_owned_for_rows(&indices)?,
153        );
154    }
155    Ok(cache.value(row))
156}

158pub(super) fn array(
159    table: &Table,
160    key: &str,
161    row: usize,
162    cache: &Mutex<Block<ArrayValue>>,
163    resources: &SessionResources,
164) -> SimulatorResult<Option<ArrayValue>> {
165    let mut cache = cache.lock().expect("append array block");
166    if !cache.contains(table, row) {
167        let column = table.column_accessor(key).map_err(super::error)?;
168        let candidates: Vec<_> = (row..table.row_count().min(row + 64)).collect();
169        let text = table
170            .schema()
171            .and_then(|schema| schema.column(key))
172            .and_then(|column| column.data_type())
173            == Some(PrimitiveType::String);
174        let mut extra = Reservation::none();
175        let count = if text {
176            1
177        } else {
178            let size = column
179                .array_shapes_uncached(&candidates)
180                .map_err(super::error)?
181                .iter()
182                .flatten()
183                .map(|shape| shape.iter().copied().fold(16usize, usize::saturating_mul))
184                .max()
185                .unwrap_or(1)
186                .max(1);
187            // A single cell can exceed the target block size. Admit its
188            // decoded cache, returned clone and encoding before reading it.
189            let bytes = size
190                .checked_mul(3)
191                .ok_or_else(|| super::error("append cell size overflows usize"))?
192                .saturating_sub(2 * ARRAY_BYTES);
193            if bytes != 0 {
194                extra = admit(
195                    resources.host(),
196                    resources.policy(),
197                    &Demand {
198                        phase: "copying an oversized simulated cell",
199                        memory: bytes as u64,
200                    },
201                )
202                .map_err(SimulatorError::admission)?;
203            }
204            (ARRAY_BYTES / size).clamp(1, candidates.len())
205        };
206        cache.store(
207            table,
208            row,
209            column
210                .array_cells_owned_uncached(&candidates[..count])
211                .map_err(super::error)?,
212        );
213        cache.extra = extra;
214    }
215    Ok(cache.value(row))
216}
