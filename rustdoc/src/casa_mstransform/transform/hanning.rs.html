18const COEFFICIENTS: [f32; 3] = [0.25, 0.5, 0.25];

31fn smooth_visibility(
32    value: &mut ArrayValue,
33    flag_shape: &[usize],
34    average_edges: bool,
35) -> Result<(), TableError> {
36    let ranks_match = flag_shape.len() == 2 || (average_edges && flag_shape.len() == 3);
37    if !ranks_match || value.shape() != flag_shape {
38        return Err(TableError::Schema(
39            "visibility and FLAG must have matching [corr, chan] shapes for Hanning smoothing"
40                .into(),
41        ));
42    }
43    match value {
44        ArrayValue::Complex32(array) => smooth_values(array, average_edges),
45        ArrayValue::Float32(array) => smooth_values(array, average_edges),
46        _ => {
47            return Err(TableError::Schema(
48                "visibility arrays must hold Complex32 or Float32".into(),
49            ));
50        }
51    }
52    Ok(())
53}
54
55fn smooth_values<T>(array: &mut ArrayD<T>, average_edges: bool)
56where
57    T: Copy + Add<Output = T> + Mul<f32, Output = T>,
58{
59    apply_interior(
60        array,
61        |[left, center, right]| {
62            left * COEFFICIENTS[0] + center * COEFFICIENTS[1] + right * COEFFICIENTS[2]
63        },
64        average_edges.then_some(|[left, right]| left * 0.5 + right * 0.5),
65    );
66}

68pub(super) fn smooth_regrid(value: &mut ArrayValue, flags: &ArrayValue) -> Result<(), TableError> {
69    let ArrayValue::Bool(flags) = flags else {
70        return Err(TableError::Schema("Hanning FLAG must be Bool".into()));
71    };
72    if value.shape() != flags.shape() || flags.ndim() != 2 {
73        return Err(TableError::Schema(
74            "visibility and FLAG must have matching [corr, chan] shapes for Hanning smoothing"
75                .into(),
76        ));
77    }
78    match value {
79        ArrayValue::Complex32(array) => smooth_flagged_values(array, flags),
80        ArrayValue::Float32(array) => smooth_flagged_values(array, flags),
81        _ => {
82            return Err(TableError::Schema(
83                "Hanning visibility must be Complex32 or Float32".into(),
84            ));
85        }
86    }
87    Ok(())
88}

90fn smooth_flagged_values<T>(values: &mut ArrayD<T>, flags: &ArrayD<bool>)
91where
92    T: Copy + Add<Output = T> + Mul<f32, Output = T>,
93{
94    let channels = values.len_of(Axis(1));
95    if channels < 2 {
96        return;
97    }
98    let last = channels - 1;
99    for (mut row, flags) in values
100        .lanes_mut(Axis(1))
101        .into_iter()
102        .zip(flags.lanes(Axis(1)))
103    {
104        // SubMS uses Smooth::hanning with relaxed=True and TrueIsGood=False.
105        // Both endpoints remain flagged. Their stored values use the edge masks.
106        let first_value = if !flags[0] && !flags[1] {
107            row[0] * (2.0 / 3.0) + row[1] * (1.0 / 3.0)
108        } else {
109            row[0]
110        };
111        let last_value = if flags[last] {
112            row[last]
113        } else if flags[last - 1] {
114            row[last - 1]
115        } else {
116            row[last] * (2.0 / 3.0) + row[last - 1] * (1.0 / 3.0)
117        };
118        let mut left = row[0];
119        for channel in 1..last {
120            let centre = row[channel];
121            let right = row[channel + 1];
122            if !flags[channel] {
123                let weights = match (flags[channel - 1], flags[channel + 1]) {
124                    (false, false) => COEFFICIENTS,
125                    (false, true) => [1.0 / 3.0, 2.0 / 3.0, 0.0],
126                    (true, false) => [0.0, 2.0 / 3.0, 1.0 / 3.0],
127                    (true, true) => [0.0, 1.0, 0.0],
128                };
129                row[channel] = left * weights[0] + centre * weights[1] + right * weights[2];
130            }
131            left = centre;
132        }
133        row[0] = first_value;
134        row[last] = last_value;
135    }
136}

138pub(super) fn flag(value: &mut ArrayValue) {
139    let ArrayValue::Bool(flags) = value else {
140        unreachable!("FLAG has already been validated")
141    };
142    apply_interior(flags, |[left, center, right]| left || center || right, None);
143    let channels = flags.len_of(Axis(1));
144    if channels > 0 {
145        flags.index_axis_mut(Axis(1), 0).fill(true);
146        flags.index_axis_mut(Axis(1), channels - 1).fill(true);
147    }
148}

150fn apply_interior<T: Copy>(
151    array: &mut ArrayD<T>,
152    filter: impl Fn([T; 3]) -> T,
153    edges: Option<fn([T; 2]) -> T>,
154) {
155    let channels = array.len_of(Axis(1));
156    if channels < 3 {
157        return;
158    }
159    for mut row in array.lanes_mut(Axis(1)) {
160        let endpoints = edges.map(|edge| {
161            [
162                edge([row[0], row[1]]),
163                edge([row[channels - 2], row[channels - 1]]),
164            ]
165        });
166        let mut left = row[0];
167        let mut center = row[1];
168        for channel in 1..channels - 1 {
169            let right = row[channel + 1];
170            row[channel] = filter([left, center, right]);
171            left = center;
172            center = right;
173        }
174        if let Some([first, last]) = endpoints {
175            row[0] = first;
176            row[channels - 1] = last;
177        }
178    }
179}

181pub(super) fn tool_weight(
182    value: &mut ArrayValue,
183    flag_shape: &[usize],
184    spectrum: bool,
185) -> Result<(), TableError> {
186    let ArrayValue::Float32(weights) = value else {
187        return Err(TableError::Schema("Hanning weights must be Float32".into()));
188    };
189    if spectrum {
190        if weights.shape() != flag_shape || !matches!(flag_shape.len(), 2 | 3) {
191            return Err(TableError::Schema("WEIGHT_SPECTRUM must match FLAG".into()));
192        }
193        apply_interior(
194            weights,
195            |values| {
196                if values.contains(&0.0) {
197                    return 0.0;
198                }
199                let [left, center, right] = values.map(f64::from);
200                (1.0 / (1.0 / (16.0 * left) + 1.0 / (4.0 * center) + 1.0 / (16.0 * right))) as f32
201            },
202            Some(|_| 0.0),
203        );
204    } else {
205        if flag_shape.len() != 2 || weights.shape() != &flag_shape[..1] {
206            return Err(TableError::Schema(
207                "WEIGHT must have one value per correlation".into(),
208            ));
209        }
210        weights.map_inplace(|weight| *weight = (f64::from(*weight) * 8.0 / 3.0) as f32);
211    }
212    Ok(())
213}

215pub(super) fn statistics(
216    (scalar_weight, scalar_sigma): &(ArrayValue, ArrayValue),
217    spectra: SpectrumCells<'_>,
218    flags: &ArrayValue,
219    bins: &[Vec<usize>],
220    averages: Option<&ChannelAverages<'_>>,
221    visibility: &VisibilityColumns,
222    time_average: bool,
223) -> Result<AveragedStatistics, TableError> {
224    let (weight, observed_weight, flags) = if let Some(averages) = averages {
225        let (weight, observed_weight) = averages.weight_spectra();
226        (weight, observed_weight, averages.bin_flags())
227    } else {
228        let ArrayValue::Bool(flags) = flags else {
229            return Err(TableError::Schema("FLAG must hold Bool".into()));
230        };
231        let &[correlations, channels] = flags.shape() else {
232            return Err(TableError::Schema(
233                "FLAG must be rank-2 [corr, chan]".into(),
234            ));
235        };
236        let shape = [correlations, channels];
237        let weight = InputStatistic::from_cells(scalar_weight, spectra.weight, shape, "WEIGHT")?;
238        let sigma = InputStatistic::from_cells(scalar_sigma, spectra.sigma, shape, "SIGMA")?;
239        let has = |source| visibility.copies().iter().any(|copy| copy.source == source);
240        let dims = (correlations, bins.len());
241        let calibrated =
242            || Array2::from_shape_fn(dims, |(corr, bin)| weight.value(corr, bins[bin][0]));
243        let observed = || {
244            Array2::from_shape_fn(dims, |(corr, bin)| {
245                weight_from_sigma(sigma.value(corr, bins[bin][0]))
246            })
247        };
248        let (weight, observed_weight) = if has(Data) && has(CorrectedData) {
249            (calibrated(), observed())
250        } else {
251            let weight = if time_average || has(CorrectedData) {
252                calibrated()
253            } else if has(Data) {
254                observed()
255            } else if has(ModelData) {
256                Array2::ones(dims)
257            } else {
258                // Keep defined noise columns for FLOAT_DATA and LAG_DATA.
259                calibrated()
260            };
261            (weight.clone(), weight)
262        };
263        let selected = Array2::from_shape_fn(dims, |(corr, bin)| flags[[corr, bins[bin][0]]]);
264        (weight, observed_weight, selected)
265    };
266    let (weight, weight_spectrum) = filter_noise(weight, &flags, |weight| weight);
267    let (sigma, sigma_spectrum) = filter_noise(observed_weight, &flags, sigma_from_weight);
268    Ok(AveragedStatistics {
269        weight: ArrayValue::Float32(weight.into_dyn()),
270        sigma: ArrayValue::Float32(sigma.into_dyn()),
271        weight_spectrum: ArrayValue::Float32(weight_spectrum.into_dyn()),
272        sigma_spectrum: ArrayValue::Float32(sigma_spectrum.into_dyn()),
273    })
274}

276fn filter_noise(
277    mut weights: Array2<f32>,
278    flags: &Array2<bool>,
279    convert: impl Fn(f32) -> f32,
280) -> (Array1<f32>, Array2<f32>) {
281    let channels = weights.ncols();
282    let mut scalars = Array1::zeros(weights.nrows());
283    if channels == 0 {
284        return (scalars, weights);
285    }
286    for ((mut row, mask), scalar) in weights
287        .rows_mut()
288        .into_iter()
289        .zip(flags.rows())
290        .zip(&mut scalars)
291    {
292        let mut sum = 0.;
293        let mut unflagged_sum = 0.;
294        let mut unflagged = 0;
295        let mut left = 0.;
296        let mut center = row[0];
297        for channel in 0..channels {
298            let mut output = center;
299            if channel > 0 && channel + 1 < channels {
300                let right = row[channel + 1];
301                let inputs = [left, center, right];
302                let mut variance = 0.;
303                let mut flagged = false;
304                for (offset, (weight, coefficient)) in
305                    inputs.into_iter().zip(COEFFICIENTS).enumerate()
306                {
307                    if weight <= f32::MIN_POSITIVE {
308                        flagged = true;
309                    } else {
310                        // An unknown (NaN) variance must propagate through
311                        // the filter. Zero or negative weight marks a flag.
312                        variance += coefficient * coefficient / weight;
313                    }
314                    flagged |= mask[channel - 1 + offset];
315                }
316                output = if variance > f32::MIN_POSITIVE {
317                    variance.recip()
318                } else {
319                    variance
320                };
321                if !flagged {
322                    unflagged_sum += convert(output);
323                    unflagged += 1;
324                }
325            }
326            row[channel] = convert(output);
327            sum += row[channel];
328            left = center;
329            center = row.get(channel + 1).copied().unwrap_or_default();
330        }
331        *scalar = if unflagged == 0 {
332            sum / channels as f32
333        } else {
334            unflagged_sum / unflagged as f32
335        };
336    }
337    (scalars, weights)
