2//! Resolve stored DD/POL mappings, admitting their buffers before decoding.
3use std::sync::Arc;
4
5use casa_imaging_runtime::{Demand, Reservation, admit};
6use casa_ms::{MeasurementSet, SubtableId};
7use casa_tables::{RequiredScalarColumnDestination, RequiredScalarColumnValuesMut};
8use casa_types::ArrayValue;
9
10use super::super::{SessionResources, spectral::DataDescription};
11use super::{SimulatorError, SimulatorResult, index};
12
13pub(super) struct Mappings {
14    pub(super) polarizations: Arc<Vec<Vec<i32>>>,
15    pub(super) descriptions: Arc<Vec<DataDescription>>,
16    pub(super) memory: Arc<Reservation>,
17}
18
19fn error(value: impl std::fmt::Display) -> SimulatorError {
20    SimulatorError::raised(format!("openfromms data descriptions: {value}"))
21}
22
23pub(super) fn read(ms: &MeasurementSet, resources: &SessionResources) -> SimulatorResult<Mappings> {
24    let dd = ms
25        .subtable(SubtableId::DataDescription)
26        .ok_or_else(|| error("missing DATA_DESCRIPTION"))?;
27    let pol = ms
28        .subtable(SubtableId::Polarization)
29        .ok_or_else(|| error("missing POLARIZATION"))?;
30    let windows = ms.spectral_window().map_err(error)?.row_count();
31    let codes = pol.column_accessor("CORR_TYPE").map_err(error)?;
32    // One cell header at a time: the pass holds no row-count-sized scratch.
33    let mut elements = 0usize;
34    for row in 0..pol.row_count() {
35        let shapes = codes.array_shapes_uncached(&[row]).map_err(error)?;
36        let shape = shapes
37            .first()
38            .and_then(Option::as_ref)
39            .ok_or_else(|| error("undefined CORR_TYPE"))?;
40        if shape.len() != 1 {
41            return Err(error("CORR_TYPE is not a vector"));
42        }
43        elements = elements
44            .checked_add(shape[0])
45            .ok_or_else(|| error("correlation count overflows usize"))?;
46    }
47    let bytes = dd
48        .row_count()
49        .checked_mul(size_of::<DataDescription>())
50        .and_then(|v| v.checked_add(pol.row_count().checked_mul(size_of::<Vec<i32>>())?))
51        .and_then(|v| v.checked_add(elements.checked_mul(8)?))
52        .and_then(|v| v.checked_add(128 << 10))
53        .ok_or_else(|| error("mapping memory overflows usize"))?;
54    let mut reservation = admit(
55        resources.host(),
56        resources.policy(),
57        &Demand {
58            phase: "reading stored simulation data descriptions",
59            memory: bytes as u64,
60        },
61    )
62    .map_err(SimulatorError::admission)?;
63    let mut polarizations = Vec::with_capacity(pol.row_count());
64    for row in 0..pol.row_count() {
65        let mut cells = codes.array_cells_owned_uncached(&[row]).map_err(error)?;
66        let Some(ArrayValue::Int32(values)) = cells.pop().flatten() else {
67            return Err(error("CORR_TYPE is not Int32"));
68        };
69        polarizations.push(values.into_raw_vec_and_offset().0);
70    }
71    let mut descriptions = Vec::with_capacity(dd.row_count());
72    {
73        let mut ids = Vec::with_capacity(1024);
74        let mut window_ids = Vec::with_capacity(1024);
75        let mut polarization_ids = Vec::with_capacity(1024);
76        for first in (0..dd.row_count()).step_by(1024) {
77            ids.clear();
78            ids.extend(first..dd.row_count().min(first + 1024));
79            window_ids.clear();
80            polarization_ids.clear();
81            dd.required_scalar_columns_for_rows_into(
82                &ids,
83                &mut [
84                    RequiredScalarColumnDestination::new(
85                        "SPECTRAL_WINDOW_ID",
86                        RequiredScalarColumnValuesMut::Int32(&mut window_ids),
87                    ),
88                    RequiredScalarColumnDestination::new(
89                        "POLARIZATION_ID",
90                        RequiredScalarColumnValuesMut::Int32(&mut polarization_ids),
91                    ),
92                ],
93            )
94            .map_err(error)?;
95            for (offset, (&window, &polarization)) in
96                window_ids.iter().zip(&polarization_ids).enumerate()
97            {
98                descriptions.push(DataDescription {
99                    window: index(window, windows, first + offset, "SPECTRAL_WINDOW_ID")?,
100                    polarization: index(
101                        polarization,
102                        polarizations.len(),
103                        first + offset,
104                        "POLARIZATION_ID",
105                    )?,
106                });
107            }
108        }
109    }
110    let retained = descriptions.capacity() * size_of::<DataDescription>()
111        + polarizations.capacity() * size_of::<Vec<i32>>()
112        + polarizations
113            .iter()
114            .map(|v| v.capacity() * size_of::<i32>())
115            .sum::<usize>()
116        + 128;
117    reservation.retain(retained as u64);
118    Ok(Mappings {
119        polarizations: Arc::new(polarizations),
120        descriptions: Arc::new(descriptions),
121        memory: Arc::new(reservation),
122    })
123}
