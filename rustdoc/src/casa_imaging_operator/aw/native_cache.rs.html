2//! How a native AW cache directory stands against a request (#735): the
3//! request manifest generation writes beside its cells, the cells the
4//! request names, and any cells it does not.
5//!
6//! A native cell's content depends on the whole request, not only on its
7//! own (PA, frequency, w, Mueller) key. The conjugate beam is the request's
8//! listed frequency nearest `√(2f₀² − f²)` (`AWConvFunc::makeConvFunction2`),
9//! and with `conjbeams` the catalog looks each row up at the nearest listed
10//! frequency (`CFBuffer::initMaps`). Cells are named by index, so a
11//! directory generated for another SPW selection can name the same cells
12//! and hold different ones. The manifest records every input that enters a
13//! cell. A directory serves a request only when its manifest is the
14//! request's and it holds exactly the request's cells.
15

24/// The request manifest native generation writes beside its cells, last,
25/// once every cell is in place.
26const MANIFEST: &str = "CasaRsNativeRequest.txt";

28/// The lock file runs share while they read the cache and a generation
29/// holds alone while it writes ([`NativeCacheLock`]).
30const LOCK: &str = "CasaRsNativeCache.lock";
31
32/// The prefix of a cell image being written: [`super::write_cell`] builds
33/// the image under it and renames it to its cell name only when complete,
34/// so a cell name always names a complete image (#739 review).
35pub(super) const STAGING_PREFIX: &str = ".staging-";

37/// An advisory lock on a native cache directory, held until dropped
38/// (#739 review). A run holds it shared for as long as its catalog reads
39/// cells, which load lazily. Generation holds it exclusive while it clears
40/// and writes. So a cache is never replaced under a run that reads it, and
41/// a run never opens one that is being written. Both are taken without
42/// waiting: the conflicting run's error names the directory.
43#[derive(Debug)]
44pub struct NativeCacheLock {
45    _file: File,
46}
47
48impl NativeCacheLock {
49    /// The shared lock a reading run holds on the cache at `root`, created
50    /// if missing.
51    ///
52    /// # Errors
53    ///
54    /// The directory or its lock file cannot be created, or another run is
55    /// writing the cache.
56    pub fn shared(root: impl AsRef<Path>) -> Result<Self, AwCatalogError> {
57        let root = root.as_ref();
58        let file = Self::open(root)?;
59        match file.try_lock_shared() {
60            Ok(()) => Ok(Self { _file: file }),
61            Err(TryLockError::WouldBlock) => Err(cache_error(
62                root,
63                "another run is generating this native AW cache; wait for it or use another \
64                 native_cf_cache",
65            )),
66            Err(TryLockError::Error(error)) => {
67                Err(cache_error(root, format!("cannot lock the cache: {error}")))
68            }
69        }
70    }
71
72    /// The exclusive lock generation holds on the cache at `root`, created
73    /// if missing.
74    ///
75    /// # Errors
76    ///
77    /// The directory or its lock file cannot be created, or another run is
78    /// reading or writing the cache.
79    pub fn exclusive(root: impl AsRef<Path>) -> Result<Self, AwCatalogError> {
80        let root = root.as_ref();
81        let file = Self::open(root)?;
82        match file.try_lock() {
83            Ok(()) => Ok(Self { _file: file }),
84            Err(TryLockError::WouldBlock) => Err(cache_error(
85                root,
86                "another run is using this native AW cache, so it cannot be regenerated for this \
87                 request now; wait for that run or use another native_cf_cache",
88            )),
89            Err(TryLockError::Error(error)) => {
90                Err(cache_error(root, format!("cannot lock the cache: {error}")))
91            }
92        }
93    }
94
95    fn open(root: &Path) -> Result<File, AwCatalogError> {
96        std::fs::create_dir_all(root)
97            .map_err(|error| cache_error(root, format!("cannot create: {error}")))?;
98        let path = root.join(LOCK);
99        File::options()
100            .create(true)
101            .truncate(false)
102            .write(true)
103            .open(&path)
104            .map_err(|error| cache_error(&path, format!("cannot open the lock file: {error}")))
105    }
106}

108/// The suffix a cell's `CFS_` and `WTCFS_` images share,
109/// `<pa>_0_CF_<frequency>_<w>_<mueller>.im` (one baseline type).
110pub(super) fn cell_suffix(pa: usize, frequency: usize, w: usize, mueller: usize) -> String {
111    format!("{pa}_0_CF_{frequency}_{w}_{mueller}.im")
112}

114/// The number of cells `input` names: one per (parallactic angle,
115/// frequency, w plane, Mueller element).
116pub(super) const fn request_cells(input: &NativeAwRequestInput) -> usize {
117    input.pa_values.len()
118        * input.frequencies.len()
119        * input.w_values.len()
120        * input.mueller_elements.len()
121}
122
123/// Heap bytes [`AwCatalog::native_cache_state`] holds for `input`'s cells
124/// beside the manifest: a bit per cell for each of its two images.
125pub(super) const fn cache_state_bytes(input: &NativeAwRequestInput) -> usize {
126    2 * request_cells(input).div_ceil(64) * size_of::<u64>()
127}
128
129/// The position, in the order (parallactic angle, frequency, w plane,
130/// Mueller element), of the cell of `input` whose suffix is `suffix`, if it
131/// names one: exactly the name [`cell_suffix`] gives it.
132fn request_cell(input: &NativeAwRequestInput, suffix: &str) -> Option<usize> {
133    let (pa, rest) = suffix.strip_suffix(".im")?.split_once("_0_CF_")?;
134    let mut fields = rest.split('_');
135    let index =
136        |field: Option<&str>, len: usize| field?.parse::<usize>().ok().filter(|&index| index < len);
137    let pa = index(Some(pa), input.pa_values.len())?;
138    let frequency = index(fields.next(), input.frequencies.len())?;
139    let w = index(fields.next(), input.w_values.len())?;
140    let mueller = index(fields.next(), input.mueller_elements.len())?;
141    (cell_suffix(pa, frequency, w, mueller) == suffix).then(|| {
142        ((pa * input.frequencies.len() + frequency) * input.w_values.len() + w)
143            * input.mueller_elements.len()
144            + mueller
145    })
146}

148/// Heap bytes reading and comparing `input`'s manifest holds at most: the
149/// file's text (this manifest, in a directory of this request), and the
150/// manifest being built, at most twice its length while it grows and three
151/// times as it moves, and the frequencies it lists. The text is at most
152/// 1 KiB of labels and scalars, 26 bytes for each value listed in `Debug`
153/// form (at most 24 characters and a separator) and 12 for each Mueller
154/// element.
155pub(super) const fn manifest_bytes(input: &NativeAwRequestInput) -> usize {
156    let values = input.frequencies.len() + input.w_values.len() + input.pa_values.len();
157    let text = 1024 + 26 * values + 12 * input.mueller_elements.len();
158    4 * text + input.frequencies.len() * size_of::<f64>()
159}

161/// The manifest of `input`: one line per input that enters a cell, values
162/// in Rust's round-trip `Debug` form, so equal text means equal values
163/// (signed zeros differ, which can only cost a needless regeneration). The
164/// dish surface is recorded by its sample count and a 64-bit FNV-1a hash of
165/// its samples' bits, which identifies it with overwhelming but not
166/// certain probability. The channel frequencies and SPW identifiers of a
167/// frequency group route rows but do not enter its cell, and the cell bound
168/// only limits the request, so neither is recorded. A change to how cells
169/// are computed must bump the version line, which invalidates every
170/// existing cache.
171fn manifest(input: &NativeAwRequestInput) -> String {
172    let mut text = String::from("casa-rs native AW request, version 1\n");
173    let samples = input.surface.samples();
174    let surface_hash = samples
175        .iter()
176        .flatten()
177        .fold(0xcbf2_9ce4_8422_2325_u64, |hash, value| {
178            value
179                .to_bits()
180                .to_le_bytes()
181                .iter()
182                .fold(hash, |hash, byte| {
183                    (hash ^ u64::from(*byte)).wrapping_mul(0x0000_0100_0000_01b3)
184                })
185        });
186    let cf_frequencies = input
187        .frequencies
188        .iter()
189        .map(|group| group.cf_frequency_hz)
190        .collect::<Vec<_>>();
191    let grid = input.grid;
192    let terms = input.terms;
193    // Writing to a String cannot fail.
194    let _ = writeln!(
195        text,
196        "surface_samples {} fnv1a64 {surface_hash:016x}",
197        samples.len()
198    );
199    let _ = writeln!(text, "antenna_diameter_m {:?}", input.antenna_diameter_m);
200    let _ = writeln!(
201        text,
202        "reference_frequency_hz {:?}",
203        input.reference_frequency_hz
204    );
205    let _ = writeln!(text, "cf_frequencies_hz {cf_frequencies:?}");
206    let _ = writeln!(text, "w_values {:?}", input.w_values);
207    let _ = writeln!(text, "w_increment {:?}", input.w_increment);
208    let _ = writeln!(text, "pa_values_rad {:?}", input.pa_values);
209    let _ = writeln!(text, "mueller_elements {:?}", input.mueller_elements);
210    let _ = writeln!(
211        text,
212        "grid size {} sky_increment_rad {:?} oversampling {}",
213        grid.size, grid.sky_increment_rad, grid.oversampling
214    );
215    let _ = writeln!(
216        text,
217        "terms aperture {} w_term {} prolate_spheroidal {} wideband {} conjugate_beams {}",
218        terms.aperture,
219        terms.w_term,
220        terms.prolate_spheroidal,
221        terms.wideband,
222        terms.conjugate_beams
223    );
224    text
225}

227/// Remove the manifest from `root`, so the directory claims no request
228/// while its cells change (#739 review).
229pub(super) fn remove_manifest(root: &Path) -> Result<(), AwCatalogError> {
230    let path = root.join(MANIFEST);
231    match std::fs::remove_file(&path) {
232        Ok(()) => Ok(()),
233        Err(error) if error.kind() == std::io::ErrorKind::NotFound => Ok(()),
234        Err(error) => Err(cache_error(&path, format!("cannot remove: {error}"))),
235    }
236}

238/// Write `input`'s manifest into `root`.
239pub(super) fn write_manifest(
240    root: &Path,
241    input: &NativeAwRequestInput,
242) -> Result<(), AwCatalogError> {
243    let path = root.join(MANIFEST);
244    std::fs::write(&path, manifest(input)).map_err(|error| AwCatalogError::Write {
245        path,
246        detail: format!("cannot write the request manifest: {error}"),
247    })
248}

250/// Whether a native cache directory records the request it was generated
251/// for ([`NativeCacheState::manifest`]).
252#[derive(Clone, Copy, Debug, PartialEq, Eq)]
253pub enum NativeManifest {
254    /// No manifest: the directory predates manifests or was not written by
255    /// native generation.
256    Absent,
257    /// The manifest records another request.
258    Differs,
259    /// The manifest records this request.
260    Matches,
261}
262
263/// How a native cache directory stands against one request
264/// ([`AwCatalog::native_cache_state`]).
265#[derive(Clone, Copy, Debug, PartialEq, Eq)]
266pub struct NativeCacheState {
267    /// Whether the directory was generated for this request.
268    pub manifest: NativeManifest,
269    /// Cells the request names whose `CFS_` and `WTCFS_` images are both
270    /// present.
271    pub present: usize,
272    /// Cells the request names.
273    pub expected: usize,
274    /// `CFS_`/`WTCFS_` images the request does not name.
275    pub foreign: usize,
276}
277
278impl NativeCacheState {
279    /// Generated for this request, complete, and holding nothing else: the
280    /// catalog may serve the directory as it is.
281    #[must_use]
282    pub fn reusable(&self) -> bool {
283        self.manifest == NativeManifest::Matches
284            && self.present == self.expected
285            && self.foreign == 0
286    }
287
288    /// Missing cells may be generated beside the present ones: the
289    /// directory holds nothing of another request. It was generated for
290    /// this request, or it holds no cell at all.
291    #[must_use]
292    pub fn fillable(&self) -> bool {
293        self.foreign == 0
294            && match self.manifest {
295                NativeManifest::Matches => true,
296                NativeManifest::Absent => self.present == 0,
297                NativeManifest::Differs => false,
298            }
299    }
300}
301
302impl std::fmt::Display for NativeCacheState {
303    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
304        let manifest = match self.manifest {
305            NativeManifest::Absent => "no request manifest",
306            NativeManifest::Differs => "generated for another request",
307            NativeManifest::Matches => "generated for this request",
308        };
309        write!(
310            formatter,
311            "{manifest}, {} of the {} cells the request names, {} cell images it does not name",
312            self.present, self.expected, self.foreign
313        )
314    }
315}

317impl AwCatalog {
318    /// How the native cache at `root` stands against `input`: whether its
319    /// manifest records `input`, how many of `input`'s cells it holds, and
320    /// how many cell images it holds that `input` does not name. A missing
321    /// directory holds nothing.
322    ///
323    /// # Errors
324    ///
325    /// The directory or its manifest cannot be read.
326    pub fn native_cache_state(
327        root: impl AsRef<Path>,
328        input: &NativeAwRequestInput,
329    ) -> Result<NativeCacheState, AwCatalogError> {
330        let root = root.as_ref();
331        let expected = request_cells(input);
332        if !root.is_dir() {
333            return Ok(NativeCacheState {
334                manifest: NativeManifest::Absent,
335                present: 0,
336                expected,
337                foreign: 0,
338            });
339        }
340        let manifest_path = root.join(MANIFEST);
341        let recorded = match std::fs::read_to_string(&manifest_path) {
342            Ok(text) if text == manifest(input) => NativeManifest::Matches,
343            Ok(_) => NativeManifest::Differs,
344            Err(error) if error.kind() == std::io::ErrorKind::NotFound => NativeManifest::Absent,
345            Err(error) => {
346                return Err(cache_error(
347                    &manifest_path,
348                    format!("cannot read the request manifest: {error}"),
349                ));
350            }
351        };
352        let words = expected.div_ceil(64);
353        let (mut imaging, mut weight) = (vec![0_u64; words], vec![0_u64; words]);
354        let mut foreign = 0;
355        let entries = std::fs::read_dir(root)
356            .map_err(|error| cache_error(root, format!("cannot list: {error}")))?;
357        for entry in entries {
358            let entry =
359                entry.map_err(|error| cache_error(root, format!("cannot list: {error}")))?;
360            let name = entry.file_name();
361            let Some(name) = name.to_str() else {
362                continue;
363            };
364            let (held, rest) = if let Some(rest) = name.strip_prefix(WEIGHT_PREFIX) {
365                (&mut weight, rest)
366            } else if let Some(rest) = name.strip_prefix(IMAGING_PREFIX) {
367                (&mut imaging, rest)
368            } else {
369                continue;
370            };
371            match request_cell(input, rest) {
372                Some(cell) if entry.path().is_dir() => held[cell / 64] |= 1 << (cell % 64),
373                _ => foreign += 1,
374            }
375        }
376        let present = imaging
377            .iter()
378            .zip(&weight)
379            .map(|(imaging, weight)| (imaging & weight).count_ones() as usize)
380            .sum();
381        Ok(NativeCacheState {
382            manifest: recorded,
383            present,
384            expected,
385            foreign,
386        })
387    }
388
389    /// Remove the request manifest, then every `CFS_` and `WTCFS_` cell and
390    /// any cell image left half written under `root`, so a regenerated
391    /// catalog carries no member of an earlier request. The manifest goes
392    /// first, so a clear that stops part way leaves a directory that claims
393    /// no request. The lock file stays. The caller holds the exclusive
394    /// [`NativeCacheLock`].
395    ///
396    /// # Errors
397    ///
398    /// The directory cannot be listed or an entry cannot be removed.
399    pub fn clear_native(root: impl AsRef<Path>) -> Result<(), AwCatalogError> {
400        let root = root.as_ref();
401        if !root.is_dir() {
402            return Ok(());
403        }
404        remove_manifest(root)?;
405        let entries = std::fs::read_dir(root)
406            .map_err(|error| cache_error(root, format!("cannot list: {error}")))?;
407        for entry in entries {
408            let entry =
409                entry.map_err(|error| cache_error(root, format!("cannot list: {error}")))?;
410            let name = entry.file_name();
411            let Some(name) = name.to_str() else {
412                continue;
413            };
414            if name.starts_with(IMAGING_PREFIX)
415                || name.starts_with(WEIGHT_PREFIX)
416                || name.starts_with(STAGING_PREFIX)
417            {
418                std::fs::remove_dir_all(entry.path()).map_err(|error| {
419                    cache_error(entry.path(), format!("cannot remove: {error}"))
420                })?;
421            }
422        }
423        Ok(())
424    }
