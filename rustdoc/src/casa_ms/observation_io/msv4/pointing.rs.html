3//! (`POINTING_BEAM[time_pointing | time, antenna_name, local_sky_dir_label]`)

26/// XRADIO's `frame` for a pointing direction: the celestial frames, and
27/// `altaz` for casacore's `AZELGEO`.
28pub(crate) const POINTING_FRAMES: [(&str, DirectionRef); 3] = [
29    ("fk5", DirectionRef::J2000),
30    ("icrs", DirectionRef::ICRS),
31    ("altaz", DirectionRef::AZELGEO),
32];
33
34/// Pointing times read at first while searching outward for the antennas'
35/// nearest samples beyond the requested times (the casa-rs writer's chunk
36/// of times); each further read doubles, up to [`MOST_SEARCH_TIMES`]. Every
37/// antenna is searched in one read, so a chunk is decoded about once a side
38/// rather than once an antenna.
39const SEARCH_TIMES: u64 = 4096;
40
41/// The most pointing times one outward read covers.
42const MOST_SEARCH_TIMES: u64 = 16_384;
43
44/// Column indices and nearest-sample slots per requested antenna.
45const ANTENNA_WORKSPACE_BYTES: usize = 128;
46
47fn antenna_workspace(requested: usize) -> usize {
48    requested
49        .saturating_mul(ANTENNA_WORKSPACE_BYTES)
50        .saturating_add(casa_types::memory::btree_node_bytes::<
51            AntennaId,
52            AntennaPointing,
53        >(requested))
54}
55
56/// Small Zarr codec contexts and traversal state beside the charged chunk,
57/// array handles and coordinates. The conversion API grants 1 MiB for a
58/// pointing read.
59const POINTING_WORKSPACE_BYTES: usize = 64 << 10;
60
61/// The decoded coordinates, directions and final series, with the array
62/// handles, string-label decode and per-antenna bookkeeping held alongside.
63fn pointing_charge(
64    times: usize,
65    span: usize,
66    search: usize,
67    columns: usize,
68    requested: usize,
69    metadata: usize,
70) -> usize {
71    let read = times
72        .saturating_add(
73            span.saturating_add(search)
74                .saturating_mul(columns)
75                .saturating_mul(2),
76        )
77        .saturating_mul(size_of::<f64>());
78    let held = requested
79        .saturating_mul(span.saturating_add(2))
80        .saturating_mul(4 * size_of::<f64>());
81    read.saturating_add(held)
82        .saturating_add(metadata)
83        .saturating_add(antenna_workspace(requested))
84}

87    /// A pointing read's array metadata and string-coordinate workspace.
88    /// The opening projector uses the same conservative handle and UTF32
89    /// costs; its codec workspace also covers transient Zarr decode buffers.
90    fn pointing_metadata(
91        &self,
92        path: &str,
93        beam: &Array<FilesystemStore>,
94        time_axis: &str,
95    ) -> Result<usize, ObservationError> {
96        if beam.shape().len() < 3 {
97            return Err(invalid(format!(
98                "{path}/POINTING_BEAM has no pointing axes"
99            )));
100        }
101        let time = open_array(&self.store, &format!("{path}/{time_axis}"))?;
102        let names = open_array(&self.store, &format!("{path}/antenna_name"))?;
103        if time.shape() != [beam.shape()[0]] || names.shape() != [beam.shape()[1]] {
104            return Err(invalid(format!(
105                "{path}/POINTING_BEAM does not span its coordinates"
106            )));
107        }
108        let fixed = names.data_type().fixed_size().unwrap_or(0);
109        let labels = usize::try_from(names.shape()[0])
110            .unwrap_or(usize::MAX)
111            .saturating_mul(
112                fixed
113                    .saturating_mul(4)
114                    .saturating_add(size_of::<Vec<char>>() + size_of::<String>()),
115            );
116        Ok(POINTING_WORKSPACE_BYTES
117            .saturating_add(handle_bytes(beam))
118            .saturating_add(handle_bytes(&time))
119            .saturating_add(handle_bytes(&names))
120            .saturating_add(labels)
121            .saturating_add(chunk_bytes(beam)?.saturating_mul(2))
122            .saturating_add(chunk_bytes(&time)?.saturating_mul(2))
123            .saturating_add(chunk_bytes(&names)?.saturating_mul(2)))
124    }

126    /// The most bytes a pointing query reads and holds
127    /// ([`casa_observation::Observation::pointing_bytes`]): over the
128    /// datasets, a query spanning every pointing time of every antenna, or
129    /// one that also searches outward from a span, whichever reads more,
130    /// and the series it gives every antenna of the catalogue.
131    pub(super) fn pointing_bound(&self) -> Result<usize, ObservationError> {
132        let mut most = 0;
133        for dataset in &self.datasets {
134            let path = format!("{}/pointing_xds", dataset.path);
135            let beam_path = format!("{path}/POINTING_BEAM");
136            if !has_node(&self.store, &beam_path)? {
137                continue;
138            }
139            let beam = open_array(&self.store, &beam_path)?;
140            let time_axis = if has_node(&self.store, &format!("{path}/time_pointing"))? {
141                "time_pointing"
142            } else {
143                "time"
144            };
145            let metadata = self.pointing_metadata(&path, &beam, time_axis)?;
146            let shape = beam.shape();
147            let [times, antennas] =
148                [shape[0], shape[1]].map(|extent| usize::try_from(extent).unwrap_or(usize::MAX));
149            let search = times.min(MOST_SEARCH_TIMES as usize);
150            most = most.max(pointing_charge(
151                times,
152                times,
153                search,
154                antennas,
155                self.catalogue.antennas.len(),
156                metadata,
157            ));
158        }
159        Ok(most)
160    }

162    /// [`casa_observation::Observation::pointing`] on one dataset: each
163    /// antenna's samples within the span and its nearest one each side.
164    pub(super) fn read_pointing(
165        &self,
166        request: &PointingRequest,
167    ) -> Result<PointingSeries, ObservationError> {
168        let (dataset, entry) = self.dataset(request.partition)?;
169        if request.direction == PointingDirection::Target {
170            return Err(ObservationError::Unsupported {
171                what: "an MSv4 pointing target (pointing_xds holds the beam's direction only)",
172            });
173        }
174        let (start, end) = (*request.times.start(), *request.times.end());
175        if !(start.is_finite() && end.is_finite() && start <= end) {
176            return Err(ObservationError::InvalidRequest {
177                reason: format!("pointing times {start}..={end} s are not a finite span"),
178            });
179        }
180        let path = format!("{}/pointing_xds", dataset.path);
181        for (index, antenna) in request.antennas.iter().enumerate() {
182            if request.antennas[..index].contains(antenna) {
183                return Err(ObservationError::InvalidRequest {
184                    reason: format!("{antenna:?} is repeated in a pointing query"),
185                });
186            }
187            if !dataset.antennas.iter().any(|(_, id)| id == antenna) {
188                return Err(ObservationError::InvalidRequest {
189                    reason: format!("{antenna:?} is not an antenna of {:?}", request.partition),
190                });
191            }
192        }
193        if !has_node(&self.store, &format!("{path}/POINTING_BEAM"))? {
194            return Ok(PointingSeries::empty(DirectionRef::J2000));
195        }
196        let beam = open_array(&self.store, &format!("{path}/POINTING_BEAM"))?;
197        let time_axis = if has_node(&self.store, &format!("{path}/time_pointing"))? {
198            "time_pointing"
199        } else {
200            "time"
201        };
202        let shape = beam.shape().to_vec();
203        let mut dimensions = vec![time_axis, "antenna_name", "local_sky_dir_label"];
204        if shape.len() == 4 {
205            dimensions.push("n_polynomial");
206        }
207        require_layout(
208            &beam,
209            &format!("{path}/POINTING_BEAM"),
210            "float64",
211            &dimensions,
212        )?;
213        if shape.len() == 4 && shape[3] != 1 {
214            return Err(ObservationError::Unsupported {
215                what: "an MSv4 pointing polynomial of more than one term (XRADIO defines no \
216                       origin for its terms)",
217            });
218        }
219        if shape[2] != 2 {
220            return Err(invalid(format!(
221                "{path}/POINTING_BEAM does not hold two direction axes"
222            )));
223        }
224        let attributes = beam.attributes();
225        require_units(attributes, "rad", "POINTING_BEAM")?;
226        let frame_name = text(attributes, "frame", "POINTING_BEAM")?;
227        let frame = POINTING_FRAMES
228            .iter()
229            .find(|(name, _)| *name == frame_name)
230            .map(|(_, frame)| *frame)
231            .ok_or(ObservationError::Unsupported {
232                what: "an MSv4 pointing frame other than fk5, icrs or altaz",
233            })?;
234        let metadata = self.pointing_metadata(&path, &beam, time_axis)?;
235        let minimum = metadata
236            .saturating_add(antenna_workspace(request.antennas.len()))
237            .saturating_add(
238                usize::try_from(shape[0])
239                    .unwrap_or(usize::MAX)
240                    .saturating_mul(size_of::<f64>()),
241            );
242        if minimum > request.budget_bytes {
243            return Err(ObservationError::OverBudget {
244                what: "an MSv4 pointing query",
245                required: minimum,
246                budget: request.budget_bytes,
247            });
248        }
249        let time = open_array(&self.store, &format!("{path}/{time_axis}"))?;
250        let times: Vec<f64> = time
251            .retrieve_array_subset_opt(
252                &ArraySubset::new_with_shape(time.shape().to_vec()),
253                &self.options,
254            )
255            .map_err(storage)?;
256        let time_attributes = time.attributes().clone();
257        if time_reference(&time_attributes, time_axis)? != (entry.time_scale, entry.time_origin) {
258            return Err(invalid(format!(
259                "{path}'s times are not in its partition's time system"
260            )));
261        }
262        let names_array = open_array(&self.store, &format!("{path}/antenna_name"))?;
263        let names: Vec<String> = names_array
264            .retrieve_array_subset_opt::<Vec<Vec<char>>>(
265                &ArraySubset::new_with_shape(names_array.shape().to_vec()),
266                &self.options,
267            )
268            .map_err(storage)?
269            .into_iter()
270            .map(|chars| chars.into_iter().collect())
271            .collect();
272        if times.len() as u64 != shape[0] || names.len() as u64 != shape[1] {
273            return Err(invalid(format!(
274                "{path}/POINTING_BEAM does not span its coordinates"
275            )));
276        }
277        let first = times.partition_point(|time| *time < start);
278        let last = times.partition_point(|time| *time <= end);
279        // Read: the time coordinate, the span's values and the largest
280        // outward block. Held, at most: a time, an origin and a direction for
281        // each requested antenna at each time of the span and one each side.
282        // No search runs when the span holds every time.
283        let search = if first == 0 && last == times.len() {
284            0
285        } else {
286            times.len().min(MOST_SEARCH_TIMES as usize)
287        };
288        let required = pointing_charge(
289            times.len(),
290            last - first,
291            search,
292            names.len(),
293            request.antennas.len(),
294            metadata,
295        );
296        if required > request.budget_bytes {
297            return Err(ObservationError::OverBudget {
298                what: "an MSv4 pointing query",
299                required,
300                budget: request.budget_bytes,
301            });
302        }
303        let span = self.beam_values(&beam, first as u64..last as u64, 0..shape[1])?;
304        // Each requested antenna's column; an antenna the dataset's pointing
305        // does not name has no samples.
306        let columns: Vec<(AntennaId, usize)> = request
307            .antennas
308            .iter()
309            .filter_map(|antenna| {
310                let name = dataset
311                    .antennas
312                    .iter()
313                    .find(|(_, id)| id == antenna)
314                    .map(|(name, _)| name)?;
315                let column = names.iter().position(|known| known == name)?;
316                Some((*antenna, column))
317            })
318            .collect();
319        let wanted: Vec<usize> = columns.iter().map(|(_, column)| *column).collect();
320        let before = self.nearest(&beam, &wanted, first, Outward::Before)?;
321        let after = self.nearest(&beam, &wanted, last, Outward::After)?;
322        let mut series = PointingSeries {
323            store_has_pointing: shape[0] > 0,
324            ..PointingSeries::empty(frame)
325        };
326        for (index, (antenna, column)) in columns.into_iter().enumerate() {
327            let within = (first..last).filter_map(|time| {
328                let at = ((time - first) * names.len() + column) * 2;
329                let direction = [span[at], span[at + 1]];
330                (!direction[0].is_nan()).then_some((time, direction))
331            });
332            let samples = before[index].into_iter().chain(within).chain(after[index]);
333            let mut sample_times = Vec::with_capacity(last - first + 2);
334            let mut coefficients = Vec::with_capacity((last - first + 2) * 2);
335            for (time, direction) in samples {
336                sample_times.push(times[time]);
337                coefficients.extend_from_slice(&direction);
338            }
339            series.antennas.insert(
340                antenna,
341                AntennaPointing::new(sample_times.clone(), None, sample_times, 1, coefficients)?,
342            );
343        }
344        Ok(series)
345    }

347    /// `POINTING_BEAM`'s directions at `times` for `antennas`, `[time,
348    /// antenna, axis]`.
349    fn beam_values(
350        &self,
351        beam: &Array<FilesystemStore>,
352        times: Range<u64>,
353        antennas: Range<u64>,
354    ) -> Result<Vec<f64>, ObservationError> {
355        let mut ranges = vec![times, antennas, 0..2];
356        if beam.shape().len() == 4 {
357            ranges.push(0..1);
358        }
359        beam.retrieve_array_subset_opt(&ArraySubset::new_with_ranges(&ranges), &self.options)
360            .map_err(storage)
361    }

363    /// Each of `columns`' nearest sample before time index `from` (the
364    /// samples before it) or at or after it: reading every antenna's
365    /// directions outward, a block of times at a time, until each column
366    /// holds one or the times end.
367    #[expect(
368        clippy::cast_possible_truncation,
369        reason = "array extents are in-memory element counts"
370    )]
371    fn nearest(
372        &self,
373        beam: &Array<FilesystemStore>,
374        columns: &[usize],
375        from: usize,
376        outward: Outward,
377    ) -> Result<Vec<Nearest>, ObservationError> {
378        let (times, antennas) = (beam.shape()[0], beam.shape()[1]);
379        let mut found = vec![None; columns.len()];
380        let (mut edge, mut size) = (from as u64, SEARCH_TIMES);
381        while found.iter().any(Option::is_none) {
382            let block = match outward {
383                Outward::Before if edge == 0 => break,
384                Outward::Before => edge.saturating_sub(size)..edge,
385                Outward::After if edge >= times => break,
386                Outward::After => edge..(edge + size).min(times),
387            };
388            let values = self.beam_values(beam, block.clone(), 0..antennas)?;
389            let rows = (block.end - block.start) as usize;
390            for (slot, column) in columns.iter().enumerate() {
391                if found[slot].is_some() {
392                    continue;
393                }
394                let at = |row: usize| (row * antennas as usize + column) * 2;
395                let held = |row: &usize| !values[at(*row)].is_nan();
396                let row = match outward {
397                    Outward::Before => (0..rows).rev().find(held),
398                    Outward::After => (0..rows).find(held),
399                };
400                found[slot] = row.map(|row| {
401                    let at = at(row);
402                    (block.start as usize + row, [values[at], values[at + 1]])
403                });
404            }
405            edge = match outward {
406                Outward::Before => block.start,
407                Outward::After => block.end,
408            };
409            size = (2 * size).min(MOST_SEARCH_TIMES);
410        }
411        Ok(found)
412    }
413}
