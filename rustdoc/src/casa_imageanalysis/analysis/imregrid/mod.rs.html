28impl ImregridTemplate {
29    /// Reports whether the string is a built-in template selector such as
30    /// `get` or a CASA direction reference. Selectors resolve independently
31    /// of the task workspace; other strings are image paths.
32    pub fn is_selector(value: &str) -> bool {
33        value.eq_ignore_ascii_case("get") || frame_from_template(value).is_some()
34    }
35}

37/// Return an image's coordinate template, or regrid its pixels onto a template.
38///
39/// A dictionary template holds the two fields returned by `template='get'`.
40/// A template image supplies its coordinates and dimensions in the same form.
41///
42/// # Errors
43/// Returns an error for unreadable images, incompatible templates, unsupported
44/// pixel or coordinate types, and failed output publication.
45pub fn imregrid(request: &ImregridRequest) -> Result<ImregridOutcome, ImageError> {
46    imregrid_with_cancellation(request, None)
47}

49/// Regrid cooperatively until the staged image is accepted for publication.
50///
51/// # Errors
52/// Returns the same errors as [`imregrid`], or [`ImageError::Interrupted`]
53/// when cancellation is observed before publication.
54pub fn imregrid_with_cancellation(
55    request: &ImregridRequest,
56    cancellation: Option<&casa_core::cancellation::Cancellation>,
57) -> Result<ImregridOutcome, ImageError> {
58    use crate::regrid::progress::check;
59    check(cancellation)?;
60    let input = AnyPagedImage::open(&request.imagename)?;
61    if matches!(&request.template, ImregridTemplate::Image(path) if path.as_os_str().eq_ignore_ascii_case("get"))
62    {
63        let shape = input
64            .shape()
65            .iter()
66            .map(|&length| i32::try_from(length))
67            .collect::<Result<Vec<_>, _>>()
68            .map_err(|_| invalid("image shape exceeds the range of CASA's template record"))?;
69        let mut coordinates = coordinate_record(&input);
70        // casacore's coordsys.torecord() returns the spectral WCS string
71        // without the fixed-width padding found in stored image records.
72        for field in coordinates.fields_mut() {
73            if field.name.starts_with("spectral")
74                && let Value::Record(spectral) = &mut field.value
75                && let Some(Value::Record(wcs)) = spectral.get_mut("wcs")
76                && let Some(Value::Scalar(ScalarValue::String(ctype))) = wcs.get_mut("ctype")
77            {
78                ctype.truncate(ctype.trim_end_matches('\0').len());
79            }
80        }
81        let template = Value::Record(RecordValue::new(vec![
82            casa_core::RecordField::new("csys", Value::Record(coordinates)),
83            casa_core::RecordField::new("shap", Value::Array(ArrayValue::from_i32_vec(shape))),
84        ]));
85        check(cancellation)?;
86        return Ok(ImregridOutcome::Template(template));
87    }
88    let template = read_template(&request.template)?;
89    match input {
90        AnyPagedImage::Float32(image) => write_product(&image, request, &template, cancellation),
91        AnyPagedImage::Float64(image) => write_product(&image, request, &template, cancellation),
92        AnyPagedImage::Complex32(image) => write_product(&image, request, &template, cancellation),
93        AnyPagedImage::Complex64(image) => write_product(&image, request, &template, cancellation),
94    }
95    .map(ImregridOutcome::Product)
96}

98fn image_coordinates<T: ImagePixel>(image: &PagedImage<T>) -> RecordValue {
99    match image.table().keywords().get("coords") {
100        Some(Value::Record(record)) => record.clone(),
101        _ => image.coordinates().to_record(),
102    }
103}
104
105fn coordinate_record(image: &AnyPagedImage) -> RecordValue {
106    match image {
107        AnyPagedImage::Float32(image) => image_coordinates(image),
108        AnyPagedImage::Float64(image) => image_coordinates(image),
109        AnyPagedImage::Complex32(image) => image_coordinates(image),
110        AnyPagedImage::Complex64(image) => image_coordinates(image),
111    }
112}

114enum Template {
115    Grid(RegridTemplate),
116    Frame(DirectionRef),
117}
118
119fn read_template(template: &ImregridTemplate) -> Result<Template, ImageError> {
120    let (shape, record) = match template {
121        ImregridTemplate::Image(path) => {
122            if let Some(reference) = path.to_str().and_then(frame_from_template) {
123                return Ok(Template::Frame(reference));
124            }
125            if !path.exists() {
126                return Err(ImageError::TaskArgumentType {
127                    parameter: "template",
128                    message: format!("Cannot read template image {}", path.display()),
129                });
130            }
131            let image = AnyPagedImage::open(path)?;
132            (image.shape().to_vec(), coordinate_record(&image))
133        }
134        ImregridTemplate::Record { csys, shape } => {
135            if shape.contains(&0) {
136                return Err(invalid("template shape must contain positive integers"));
137            }
138            (shape.clone(), csys.clone())
139        }
140    };
141    let coordinates = CoordinateSystem::from_record(&record)?;
142    if shape.len() != coordinates.n_pixel_axes() {
143        return Err(invalid(
144            "template shape must match the number of pixel axes",
145        ));
146    }
147    Ok(Template::Grid(RegridTemplate { coordinates, shape }))
148}

150fn write_product<T: RegridPixel>(
151    input: &PagedImage<T>,
152    request: &ImregridRequest,
153    template: &Template,
154    cancellation: Option<&casa_core::cancellation::Cancellation>,
155) -> Result<ImageRegridSummary, ImageError> {
156    use crate::regrid::progress::{self, check};
157    let output = match (&request.template, request.output.as_os_str().is_empty()) {
158        (ImregridTemplate::Image(_), true) => {
159            let mut name = request.imagename.as_os_str().to_os_string();
160            name.push(".regridded");
161            PathBuf::from(name)
162        }
163        _ => request.output.clone(),
164    };
165    // The same-frame CASA path uses image.subimage with its overwrite default.
166    let same_frame = match template {
167        Template::Frame(reference) => input
168            .coordinates()
169            .find_coordinate(CoordinateType::Direction)
170            .is_some_and(|index| {
171                matches!(input.coordinates().coordinate(index),
172                CoordinateModel::Direction(direction) if direction.direction_ref() == *reference)
173            }),
174        Template::Grid(_) => false,
175    };
176    let overwrite = request.overwrite && !same_frame;
177    let destination = if output.as_os_str().is_empty() {
178        None
179    } else {
180        let path = std::path::absolute(&output).map_err(output_io)?;
181        check_output(request, &path, overwrite)?;
182        Some(path)
183    };
184    let method = RegridMethod::from_casa(&request.interpolation)?;
185    let (regridded, beam_rotation) = match template {
186        Template::Grid(grid) => {
187            let record_shape;
188            let shape = if matches!(request.template, ImregridTemplate::Record { .. }) {
189                record_shape = grid
190                    .shape
191                    .iter()
192                    .map(|&length| length as i64)
193                    .collect::<Vec<_>>();
194                &record_shape
195            } else {
196                &request.shape
197            };
198            (
199                regrid_task(
200                    input,
201                    grid,
202                    TaskRegridOptions {
203                        axes: &request.axes,
204                        shape,
205                        as_velocity: request.asvelocity,
206                        decimate: request.decimate,
207                        replicate: request.replicate,
208                        method,
209                        image_tool: None,
210                        cancellation,
211                    },
212                )?,
213                0.0,
214            )
215        }
216        Template::Frame(reference) => {
217            regrid_direction_frame(input, *reference, method, request.decimate, cancellation)?
218        }
219    };
220    let Regridded {
221        data,
222        mask,
223        coordinates,
224        ..
225    } = regridded;
226    let shape = data.shape().to_vec();
227    let valid_pixels = progress::count_valid(&mask, cancellation)?;
228    if let Some(destination) = destination {
229        let input_info = input.image_info()?;
230        let info = output_image_info(
231            input_info.clone(),
232            input.coordinates(),
233            &coordinates,
234            beam_rotation,
235        );
236        let stage = tempfile::tempdir_in(destination.parent().ok_or(ImageError::NotPersistent)?)
237            .map_err(output_io)?;
238        let product = stage.path().join("image");
239        let mut image = PagedImage::<T>::create(shape.clone(), coordinates, &product)?;
240        image.set_units(input.units())?;
241        if info == input_info
242            && let Some(Value::Record(record)) = input.table().keywords().get("imageinfo")
243        {
244            image.set_image_info_record(record.clone())?;
245        } else {
246            image.set_image_info(&info)?;
247        }
248        image.set_misc_info(input.misc_info())?;
249        check(cancellation)?;
250        image.put_slice(&data, &vec![0; shape.len()])?;
251        check(cancellation)?;
252        if valid_pixels < data.len() {
253            image.put_mask("mask0", &mask)?;
254            image.set_default_mask("mask0")?;
255        }
256        image.append_history_entries(&casa_images::image_history_rows(&request.imagename)?)?;
257        image.append_history_at(
258            "::imregrid",
259            &[
260                concat!("version: casa-rs ", env!("CARGO_PKG_VERSION")).into(),
261                format!(
262                    "imregrid(imagename={:?}, output={:?}, interpolation={:?})",
263                    request.imagename, output, request.interpolation
264                ),
265            ],
266        )?;
267        check(cancellation)?;
268        image.save()?;
269        drop(image);
270        #[cfg(feature = "imregrid-test-hooks")]
271        crate::imregrid_test_hooks::observe(
272            crate::imregrid_test_hooks::Checkpoint::BeforePublication,
273        );
274
275        // Cancellation before acceptance only drops the private image. Once
276        // accepted, finish the established rename or locked exchange.
277        check(cancellation)?;
278        #[cfg(feature = "imregrid-test-hooks")]
279        crate::imregrid_test_hooks::observe(
280            crate::imregrid_test_hooks::Checkpoint::PublicationAccepted,
281        );
282
283        match rename_without_replacing(&product, &destination) {
284            Err(error) if error.kind() == io::ErrorKind::AlreadyExists && overwrite => {
285                check_output(request, &destination, overwrite)?;
286                let _locks = lock_table_tree_for_replacement(&destination)?;
287                exchange_directories(&product, &destination).map_err(output_io)?;
288            }
289            result => result.map_err(output_io)?,
290        }
291    } else {
292        check(cancellation)?;
293    }
294    Ok(ImageRegridSummary {
295        imagename: request.imagename.display().to_string(),
296        template: match &request.template {
297            ImregridTemplate::Image(path) => path.display().to_string(),
298            ImregridTemplate::Record { .. } => "coordinate record".into(),
299        },
300        output: output.display().to_string(),
301        shape,
302        valid_pixels,
303    })
304}

306fn output_image_info(
307    mut info: ImageInfo,
308    source: &CoordinateSystem,
309    output: &CoordinateSystem,
310    rotation: f64,
311) -> ImageInfo {
312    if info.beam_set.is_empty() {
313        return info;
314    }
315    let stokes = |system: &CoordinateSystem| {
316        system
317            .find_coordinate(CoordinateType::Stokes)
318            .and_then(|index| match system.coordinate(index) {
319                CoordinateModel::Stokes(coordinate) => Some(coordinate.stokes().to_vec()),
320                _ => None,
321            })
322    };
323    let selections = if info.beam_set.n_stokes() > 1 {
324        stokes(source).zip(stokes(output)).map(|(source, target)| {
325            target
326                .iter()
327                .filter_map(|value| source.iter().position(|candidate| candidate == value))
328                .collect::<Vec<_>>()
329        })
330    } else {
331        None
332    };
333    if rotation == 0.0
334        && selections
335            .as_ref()
336            .is_none_or(|planes| planes.len() == info.beam_set.n_stokes())
337    {
338        return info;
339    }
340    let planes = selections.unwrap_or_else(|| (0..info.beam_set.n_stokes()).collect());
341    info.beam_set = ImageBeamSet::from_grid(
342        (0..info.beam_set.n_channels())
343            .map(|channel| {
344                planes
345                    .iter()
346                    .map(|&pol| {
347                        let mut beam = *info.beam_set.beam(channel, pol);
348                        beam.position_angle += rotation;
349                        beam
350                    })
351                    .collect()
352            })
353            .collect(),
354    );
355    info
356}

358fn check_output(
359    request: &ImregridRequest,
360    destination: &Path,
361    overwrite: bool,
362) -> Result<(), ImageError> {
363    ensure_outfile_does_not_alias_input(destination, &request.imagename)?;
364    if let ImregridTemplate::Image(path) = &request.template {
365        ensure_outfile_does_not_alias_input(destination, path)?;
366    }
367    match destination.symlink_metadata() {
368        Err(error) if error.kind() == io::ErrorKind::NotFound => Ok(()),
369        Err(error) => Err(output_io(error)),
370        Ok(_) if !overwrite => Err(invalid(format!(
371            "imregrid output already exists: {}",
372            destination.display()
373        ))),
374        Ok(metadata)
375            if metadata.is_dir()
376                && Table::open_metadata_only(TableOptions::new(destination))
377                    .is_ok_and(|table| table.info().table_type == "Image") =>
378        {
379            Ok(())
380        }
381        Ok(_) => Err(invalid(format!(
382            "{} is not a CASA image directory",
383            destination.display()
384        ))),
385    }
386}

388fn invalid(message: impl Into<String>) -> ImageError {
389    ImageError::InvalidMetadata(message.into())
390}
