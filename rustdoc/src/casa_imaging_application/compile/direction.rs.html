51/// Resolve the request's image centre over the surveyed fields.
52pub(super) fn resolve_centre(
53    request: &ImagingRequest,
54    catalogue: &Catalogue,
55    survey: &Survey,
56    engine: &casa_ms::derived::engine::MsCalEngine,
57) -> Result<Centre, PrepareError> {
58    let selected_field = request
59        .phasecenter_field
60        .unwrap_or_else(|| *survey.fields.first().expect("nonempty selection"));
61    if !survey.fields.contains(&selected_field) {
62        return Err(PrepareError::PhaseCentreField {
63            field: selected_field,
64            selected: survey.fields.clone(),
65        });
66    }
67    let field_id = usize::try_from(selected_field).expect("a surveyed field is a FIELD row");
68    let mut stored_phase = catalogue.fields[field_id].phase_centre.clone();
69    let anchor_time = survey.first_time_mjd_seconds;
70    let ephemeris_direction = |name: &str, ephemeris: &PhaseCentreEphemeris| {
71        let direction = engine.ephemeris_direction_j2000(anchor_time, field_id, name, ephemeris)?;
72        let (longitude, latitude) = direction.as_angles();
73        Ok::<_, PrepareError>(SkyDirection::new(
74            DirectionFrame::J2000,
75            longitude,
76            latitude,
77        ))
78    };
79    let (law, ephemeris, main_direction) = match request.phasecenter.as_deref() {
80        None => {
81            let field_direction = field_reference_direction(
82                survey,
83                field_id,
84                engine,
85                anchor_time,
86                &mut stored_phase,
87            )?;
88            (
89                PhaseCentreLaw::Fixed(field_direction),
90                None,
91                field_direction,
92            )
93        }
94        Some(text) if text.split_ascii_whitespace().next() == Some("J2000") => {
95            let direction = parse_phase_center_direction(text)?;
96            (PhaseCentreLaw::Fixed(direction), None, direction)
97        }
98        Some("TRACKFIELD") => {
99            let ephemeris = PhaseCentreEphemeris::tracked_fields(
100                catalogue,
101                survey
102                    .fields
103                    .iter()
104                    .map(|field| usize::try_from(*field).expect("a surveyed field is a FIELD row")),
105            )?;
106            let direction = ephemeris_direction("TRACKFIELD", &ephemeris)?;
107            (
108                PhaseCentreLaw::Ephemeris("TRACKFIELD".to_string()),
109                Some(ephemeris),
110                direction,
111            )
112        }
113        Some(text) => {
114            let ephemeris = named_ephemeris(text, survey, engine)?;
115            let direction = ephemeris_direction(text, &ephemeris)?;
116            (
117                PhaseCentreLaw::Ephemeris(text.to_string()),
118                Some(ephemeris),
119                direction,
120            )
121        }
122    };
123    let phase = MDirection::from_angles(
124        main_direction.longitude_rad(),
125        main_direction.latitude_rad(),
126        DirectionRef::J2000,
127    );
128    let (image_centre, frame) = image_centre(
129        request,
130        engine,
131        anchor_time,
132        &phase,
133        stored_phase,
134        ephemeris.is_some(),
135    )?;
136    let (longitude, latitude) = image_centre.as_angles();
137    Ok(Centre {
138        law,
139        ephemeris,
140        phase,
141        direction: direction_spec(request.imsize, request.cell, frame, longitude, latitude),
142        image_centre,
143        field_id,
144    })
145}

147/// Where an analytic primary beam is centred on each image domain: the
148/// pixel of every selected field's phase direction, in the domain's frame.
149///
150/// CASA centres one beam per field on its `PHASE_DIR` and sums them on every
151/// image, outliers included (`SynthesisImagerVi2::makePrimaryBeam`,
152/// `SIMapper::addPB`, `PBMath1D::apply`'s `toPixel`). A pointing 90° or more
153/// from a domain's centre adds nothing there; CASA's `toPixel` fails for it
154/// and `PBMath1D::apply` uses whatever pixel the failed call left.
155///
156/// # Errors
157///
158/// A moving phase centre or a selected field that follows an ephemeris,
159/// whose beams CASA shifts with the source
160/// (`SynthesisImagerVi2::getMovingDirection`); a selected field whose
161/// `PHASE_DIR` polynomial has a nonzero rate, which CASA evaluates at the
162/// field's first selected time; unreadable field directions, frame
163/// conversions or projections.
164pub(super) fn primary_beam_centres(
165    domains: &[PreparedImageDomain],
166    store: &OpenedStore,
167    survey: &Survey,
168    centre: &Centre,
169    engine: &casa_ms::derived::engine::MsCalEngine,
170) -> Result<PrimaryBeamCentres, PrepareError> {
171    if centre.ephemeris.is_some() {
172        return Err(PrepareError::MovingAnalyticPrimaryBeam);
173    }
174    // Only an MSv2 states a field's direction as a polynomial in time; the
175    // catalogue keeps its constant term.
176    let field = store
177        .msv2()
178        .map(|observation| observation.measurement_set().field())
179        .transpose()?;
180    let pointings = survey
181        .fields
182        .iter()
183        .map(|field_id| {
184            let field_id = usize::try_from(*field_id).expect("selected FIELD_IDs are nonnegative");
185            if engine.field_moves(field_id) {
186                return Err(PrepareError::MovingAnalyticPrimaryBeam);
187            }
188            if let Some(field) = &field
189                && phase_direction_varies(field.phase_dir(field_id)?)
190            {
191                return Err(PrepareError::PolynomialFieldPrimaryBeam { field_id });
192            }
193            Ok((field_id, engine.field_direction_j2000(field_id)?.clone()))
194        })
195        .collect::<Result<Vec<(usize, MDirection)>, PrepareError>>()?;
196    domains
197        .iter()
198        .map(|domain| {
199            let frame = domain
200                .direction
201                .reference_direction()
202                .frame()
203                .direction_ref();
204            let coordinate = domain.coordinates.coordinate(
205                domain
206                    .coordinates
207                    .find_coordinate(CoordinateType::Direction)
208                    .expect("an image domain has a direction coordinate"),
209            );
210            let mut centres = Vec::with_capacity(pointings.len());
211            for (field_id, pointing) in &pointings {
212                let pointing = if frame == DirectionRef::J2000 {
213                    pointing.clone()
214                } else {
215                    let measures = engine.spectral_frame_observatory_direction(
216                        survey.first_time_mjd_seconds,
217                        pointing.clone(),
218                    )?;
219                    pointing.convert_to(frame, &measures)?
220                };
221                let (longitude, latitude) = pointing.as_angles();
222                let pixel =
223                    pointing_pixel(coordinate, [longitude, latitude]).map_err(|source| {
224                        PrepareError::PrimaryBeamPointing {
225                            field_id: *field_id,
226                            source,
227                        }
228                    })?;
229                centres.extend(pixel);
230            }
231            Ok((domain.role.clone(), centres))
232        })
233        .collect()
234}

236/// Whether a `PHASE_DIR` polynomial, shaped `[2, NUM_POLY + 1]`, has a
237/// nonzero rate term. A zero rate is the constant direction exactly.
238fn phase_direction_varies(coefficients: &casa_types::ArrayValue) -> bool {
239    match coefficients {
240        casa_types::ArrayValue::Float64(coefficients) => coefficients
241            .indexed_iter()
242            .any(|(index, value)| index[1] > 0 && *value != 0.0),
243        _ => false,
244    }
245}

247/// The pixel of `pointing`, `[longitude, latitude]` in radians, on the
248/// direction coordinate `coordinate`, or `None` when it lies 90° or more
249/// from the coordinate's reference direction. The SIN projection would
250/// otherwise mirror such a pointing onto the near hemisphere.
251fn pointing_pixel(
252    coordinate: &casa_coordinates::CoordinateModel,
253    pointing: [f64; 2],
254) -> Result<Option<[f64; 2]>, casa_coordinates::CoordinateError> {
255    let reference = coordinate.reference_value();
256    let cosine = reference[1].sin() * pointing[1].sin()
257        + reference[1].cos() * pointing[1].cos() * (pointing[0] - reference[0]).cos();
258    if cosine <= 0.0 {
259        return Ok(None);
260    }
261    let pixel = coordinate.to_pixel(&pointing)?;
262    Ok(Some([pixel[0], pixel[1]]))
263}

265/// casacore's `phaseDirMeas(field_id)` asked at no time, in J2000: the
266/// field's `PHASE_DIR`, or, for a field that follows an ephemeris, the
267/// body's direction at its FIELD row's `TIME` (an MSv4's: the field's first
268/// selected time) offset by `PHASE_DIR`
269/// (`MsCalEngine::moving_field_reference_direction`), which then also
270/// replaces `stored`, the direction the image records. CASA's `tclean`
271/// centres an image without a phase centre there and holds it fixed, also
272/// over a field that moves (#987).
273fn field_reference_direction(
274    survey: &Survey,
275    field_id: usize,
276    engine: &casa_ms::derived::engine::MsCalEngine,
277    anchor_time: f64,
278    stored: &mut MDirection,
279) -> Result<SkyDirection, PrepareError> {
280    let first_time = survey
281        .first_time_of(i32::try_from(field_id).expect("a FIELD row is an i32 id"))
282        .unwrap_or(anchor_time);
283    let j2000 = match engine.moving_field_reference_direction(field_id, first_time)? {
284        Some(reference) => {
285            let frame =
286                engine.spectral_frame_observatory_direction(anchor_time, reference.clone())?;
287            let j2000 = reference.convert_to(DirectionRef::J2000, &frame)?;
288            *stored = reference;
289            j2000
290        }
291        None => engine.field_direction_j2000(field_id)?.clone(),
292    };
293    let (longitude, latitude) = j2000.as_angles();
294    Ok(SkyDirection::new(
295        DirectionFrame::J2000,
296        longitude,
297        latitude,
298    ))
299}

301/// The image's reference direction and the frame it records: ICRS at the
302/// anchor time for a moving source, the field's stored direction in its
303/// own frame when the request names no centre, else the J2000 phase
304/// centre.
305fn image_centre(
306    request: &ImagingRequest,
307    engine: &casa_ms::derived::engine::MsCalEngine,
308    anchor_time: f64,
309    phase: &MDirection,
310    stored_phase: MDirection,
311    moving: bool,
312) -> Result<(MDirection, DirectionFrame), PrepareError> {
313    if moving {
314        let frame = engine.spectral_frame_observatory_direction(anchor_time, phase.clone())?;
315        return Ok((
316            phase.convert_to(DirectionRef::ICRS, &frame)?,
317            DirectionFrame::Icrs,
318        ));
319    }
320    let centre = if request.phasecenter.is_none()
321        && matches!(
322            stored_phase.refer(),
323            DirectionRef::J2000 | DirectionRef::ICRS | DirectionRef::B1950 | DirectionRef::GALACTIC
324        ) {
325        stored_phase
326    } else {
327        phase.clone()
328    };
329    let frame = match centre.refer() {
330        DirectionRef::ICRS => DirectionFrame::Icrs,
331        DirectionRef::B1950 => DirectionFrame::B1950,
332        DirectionRef::GALACTIC => DirectionFrame::Galactic,
333        _ => DirectionFrame::J2000,
334    };
335    Ok((centre, frame))
336}

338/// A named ephemeris (or an external table).
339fn named_ephemeris(
340    text: &str,
341    survey: &Survey,
342    engine: &casa_ms::derived::engine::MsCalEngine,
343) -> Result<PhaseCentreEphemeris, PrepareError> {
344    let ephemeris = if Path::new(text).is_dir() {
345        PhaseCentreEphemeris::external(text, EPHEMERIS_BUDGET)?
346    } else {
347        PhaseCentreEphemeris::named(text, EPHEMERIS_BUDGET)?
348    };
349    // The engine observes each attached field towards its own body
350    // (`MsCalEngine::with_field_ephemerides`); a body followed over fields
351    // of which only some move is refused.
352    let mut moving = 0;
353    for field_id in &survey.fields {
354        let field_id = usize::try_from(*field_id).expect("a surveyed field is a FIELD row");
355        moving += usize::from(engine.field_moves(field_id));
356    }
357    if moving != 0 && moving != survey.fields.len() {
358        return Err(PrepareError::MixedEphemerisFields);
359    }
360    Ok(ephemeris)
361}

363/// The image's observation record: telescope and observer of the selected
364/// observations, which must agree, the first selected time and the image
365/// centre.
366pub(super) fn observation_info(
367    catalogue: &Catalogue,
368    survey: &Survey,
369    centre: &Centre,
370    engine: &casa_ms::derived::engine::MsCalEngine,
371) -> Result<ObsInfo, PrepareError> {
372    let mut labels = std::collections::BTreeSet::new();
373    for id in &survey.observation_ids {
374        let id = usize::try_from(*id).expect("a surveyed observation is an OBSERVATION row");
375        labels.insert(match catalogue.observations.get(id) {
376            Some(observation) => (
377                observation.telescope.clone(),
378                observation.observers.join(", "),
379            ),
380            None => (String::new(), String::new()),
381        });
382    }
383    if labels.len() != 1 {
384        return Err(PrepareError::ObservationLabels {
385            observation_ids: survey.observation_ids.clone(),
386            labels,
387        });
388    }
389    let (telescope_name, observer) = labels.pop_first().expect("one label pair");
390    // ObsInfo::toRecord uses MVDirection::get, preserving the signed atan2
391    // endpoint rather than mapping an exactly positive pi to negative pi.
392    let [pointing_x, pointing_y, _] = centre.image_centre.cosines();
393    let pointing_longitude = if pointing_x == 0.0 && pointing_y == 0.0 {
394        0.0
395    } else {
396        pointing_y.atan2(pointing_x)
397    };
398    Ok(ObsInfo::new(telescope_name)
399        .with_observer(observer)
400        .with_date(MEpoch::from_mjd(
401            survey.first_time_mjd_seconds / 86_400.0,
402            engine.time_reference(),
403        ))
404        .with_telescope_position(engine.observatory_position().clone())
405        .with_pointing_center(pointing_longitude, centre.image_centre.as_angles().1))
406}

408/// A SIN direction coordinate of `image_size` (x, y) cells of `cell_arcsec`
409/// (x, y), centred on `(longitude, latitude)` in `frame`.
410pub(super) fn direction_spec(
411    image_size: [usize; 2],
412    cell_arcsec: [f64; 2],
413    frame: DirectionFrame,
414    longitude: f64,
415    latitude: f64,
416) -> DirectionCoordinateSpec {
417    let [cell_x, cell_y] = cell_arcsec.map(|cell| cell * std::f64::consts::PI / (180.0 * 3600.0));
418    DirectionCoordinateSpec::new(
419        Projection::Sin,
420        SkyDirection::new(frame, longitude, latitude),
421        image_reference_pixel(image_size),
422        [-cell_x, cell_y],
423        [[1.0, 0.0], [0.0, 1.0]],
424        [180.0, 0.0],
425    )
426}

437/// The coordinate system of one image domain's products.
438pub(super) fn image_coordinates(
439    direction: DirectionCoordinateSpec,
440    image_size: [usize; 2],
441    polarizations: &[PolarizationCoordinate],
442    spectral: ImageSpectralCoordinate,
443    observation: &ObsInfo,
444) -> CoordinateSystem {
445    let reference = direction.reference_direction();
446    let mut coordinates = CoordinateSystem::new();
447    coordinates.add_coordinate(DirectionCoordinate::new(
448        reference.frame().direction_ref(),
449        CoordinateProjection::new(ProjectionType::SIN),
450        [reference.longitude_rad(), reference.latitude_rad()],
451        direction.increment_rad(),
452        image_reference_pixel(image_size),
453    ));
454    coordinates.add_coordinate(StokesCoordinate::new(
455        polarizations.iter().copied().map(stokes_type).collect(),
456    ));
457    coordinates.add_coordinate(SpectralCoordinate::new(
458        spectral.frequency_reference,
459        spectral.reference_frequency_hz,
460        spectral.increment_hz,
461        0.0,
462        spectral.rest_frequency_hz,
463    ));
464    *coordinates.obs_info_mut() = observation.clone();
465    coordinates
466}

490/// A CASA phase-centre literal `J2000 <lon> <lat>`, with sexagesimal,
491/// degree or radian angles.
492pub(super) fn parse_phase_center_direction(text: &str) -> Result<SkyDirection, PrepareError> {
493    let parts = text.split_whitespace().collect::<Vec<_>>();
494    if parts.len() != 3 || !parts[0].eq_ignore_ascii_case("J2000") {
495        return Err(PrepareError::PhaseCentreSyntax {
496            text: text.to_string(),
497        });
498    }
499    Ok(SkyDirection::new(
500        DirectionFrame::J2000,
501        parse_phase_center_angle(parts[1], true)?,
502        parse_phase_center_angle(parts[2], false)?,
503    ))
504}
505
506fn parse_phase_center_angle(text: &str, longitude: bool) -> Result<f64, PrepareError> {
507    let lower = text.to_ascii_lowercase();
508    let angle = if let Some(radians) = lower.strip_suffix("rad") {
509        radians.trim().parse::<f64>().ok()
510    } else if let Some(degrees) = lower.strip_suffix("deg") {
511        degrees.trim().parse::<f64>().ok().map(f64::to_radians)
512    } else if longitude {
513        parse_sexagesimal(text, true).map(|hours| hours * std::f64::consts::PI / 12.0)
514    } else {
515        parse_sexagesimal(text, false).map(f64::to_radians)
516    };
517    angle.ok_or_else(|| PrepareError::PhaseCentreAngle {
518        angle: text.to_string(),
519    })
520}
521
522fn parse_sexagesimal(text: &str, hours: bool) -> Option<f64> {
523    let trimmed = text.trim();
524    let sign = if trimmed.starts_with('-') { -1.0 } else { 1.0 };
525    let body = trimmed.trim_start_matches(['+', '-']);
526    let fields = if body.contains(':') {
527        body.split(':').map(str::to_owned).collect::<Vec<_>>()
528    } else if body.contains('h') || body.contains('d') || body.contains('m') || body.contains('s') {
529        body.replace(['h', 'd', 'm', 's'], " ")
530            .split_whitespace()
531            .map(str::to_owned)
532            .collect()
533    } else if !hours && body.matches('.').count() >= 2 {
534        let mut split = body.split('.');
535        let major = split.next()?.to_owned();
536        let minutes = split.next()?.to_owned();
537        let seconds = split.collect::<Vec<_>>().join(".");
538        vec![major, minutes, seconds]
539    } else {
540        return None;
541    };
542    let [major, minutes, seconds] = fields.as_slice() else {
543        return None;
544    };
545    let major = major.parse::<f64>().ok()?;
546    let minutes = minutes.parse::<f64>().ok()?;
547    let seconds = seconds.parse::<f64>().ok()?;
548    if !(major.is_finite()
549        && minutes.is_finite()
550        && seconds.is_finite()
551        && (0.0..60.0).contains(&minutes)
552        && (0.0..60.0).contains(&seconds))
553    {
554        return None;
555    }
556    Some(sign * (major.abs() + minutes / 60.0 + seconds / 3600.0))
557}

627    /// A pointing on the far hemisphere adds no beam. The SIN projection
628    /// alone would mirror the antipode onto the reference pixel.
629    #[test]
630    fn a_pointing_beyond_ninety_degrees_has_no_pixel() {
631        let cell = 1.0e-5;
632        let coordinate = CoordinateModel::from(DirectionCoordinate::new(
633            DirectionRef::J2000,
634            CoordinateProjection::new(ProjectionType::SIN),
635            [1.0, 0.4],
636            [-cell, cell],
637            [32.0, 32.0],
638        ));
639        let near = pointing_pixel(&coordinate, [1.0 + 10.0 * cell / 0.4_f64.cos(), 0.4])
640            .expect("projection")
641            .expect("a near pointing has a pixel");
642        assert!((near[0] - 22.0).abs() < 1.0e-3, "{near:?}");
643        assert!((near[1] - 32.0).abs() < 1.0e-3, "{near:?}");
644        let antipode = [1.0 + std::f64::consts::PI, -0.4];
645        let mirrored = coordinate.to_pixel(&antipode).expect("SIN never fails");
646        assert!((mirrored[0] - 32.0).abs() < 1.0e-6, "{mirrored:?}");
647        assert_eq!(
648            pointing_pixel(&coordinate, antipode).expect("projection"),
649            None
650        );
651        let beyond = [1.0 + 1.6, 0.0];
652        assert_eq!(
653            pointing_pixel(&coordinate, beyond).expect("projection"),
654            None
655        );
656    }

660    fn a_phase_direction_varies_only_with_a_nonzero_rate() {
661        use ndarray::ShapeBuilder as _;
662        // Fortran order, as casacore stores it: lon0, lat0, lon1, lat1, ...
663        let polynomial = |coefficients: &[f64]| {
664            casa_types::ArrayValue::Float64(
665                ndarray::ArrayD::from_shape_vec(
666                    ndarray::IxDyn(&[2, coefficients.len() / 2]).f(),

671        };

674        assert!(phase_direction_varies(&polynomial(&[
675            1.0, 0.4, 1.0e-9, 0.0

678            1.0, 0.4, 0.0, -1.0e-9
