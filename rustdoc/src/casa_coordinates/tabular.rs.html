2//! One-axis tabular coordinate with lookup table interpolation.
3//!
4//! [`TabularCoordinate`] maps between pixel and world values using a pair of
5//! lookup tables. Intermediate values are found by linear interpolation.
6//! This is useful for non-linear axes (e.g. irregularly-spaced frequency
7//! channels or a velocity axis with non-uniform spacing).
8//!
9//! Corresponds to C++ `TabularCoordinate`.
10
11use casa_core::{ArrayValue, RecordValue, Value};
12
13use crate::coordinate::{Coordinate, CoordinateType};
14use crate::error::CoordinateError;
15use crate::record_utils::{get_optional_string, get_optional_vec_string, get_required_vec_f64};

17const CASACORE_PIXEL_TOLERANCE: f64 = 1e-13;
18
19/// An edit to one of a tabular coordinate's live reference parameters.
20#[derive(Clone, Copy)]
21pub enum TabularEdit {
22    /// Move the live reference pixel.
23    ReferencePixel(f64),
24    /// Change the live reference world value.
25    ReferenceValue(f64),
26    /// Change the live world-axis increment.
27    Increment(f64),
28}
29
30impl TabularEdit {
31    pub(crate) fn apply(self, [mut crval, mut cdelt, mut crpix]: [f64; 3]) -> [f64; 3] {
32        match self {
33            Self::ReferencePixel(value) => crpix = value,
34            Self::ReferenceValue(value) => crval = value,
35            Self::Increment(value) => cdelt = value,
36        }
37        [crval, cdelt, crpix]
38    }
39}
40
41pub(crate) fn scaled_world(value: f64, reference: f64, transform: f64) -> f64 {
42    if transform == 1.0 {
43        value
44    } else {
45        reference + transform * (value - reference)
46    }
47}

49pub(crate) fn validate_transform_samples(
50    values: &[f64],
51    reference: f64,
52    transform: f64,
53) -> Result<(), CoordinateError> {
54    let invalid = || {
55        CoordinateError::InvalidRecord(
56            "linear transform would produce invalid tabular samples".into(),
57        )
58    };
59    if !transform.is_finite() || transform == 0.0 || values.len() < 2 {
60        return Err(invalid());
61    }
62    let overall_order = values[values.len() - 1].partial_cmp(&values[0]);
63    if !matches!(
64        overall_order,
65        Some(std::cmp::Ordering::Less | std::cmp::Ordering::Greater)
66    ) {
67        return Err(invalid());
68    }
69    let mut previous = scaled_world(values[0], reference, transform);
70    for (interval, pair) in values.windows(2).enumerate() {
71        let next = scaled_world(pair[1], reference, transform);
72        let order = if interval == 0 {
73            pair[1].partial_cmp(&pair[0])
74        } else {
75            overall_order
76        };
77        let expected = order.map(|order| {
78            if transform < 0.0 {
79                order.reverse()
80            } else {
81                order
82            }
83        });
84        if !previous.is_finite() || !next.is_finite() || next.partial_cmp(&previous) != expected {
85            return Err(invalid());
86        }
87        previous = next;
88    }
89    Ok(())
90}

92pub(crate) fn sample_reference(record: &RecordValue) -> Result<[f64; 3], CoordinateError> {
93    let read = |field| {
94        let values = get_required_vec_f64(record, field)?;
95        match values.as_slice() {
96            [value] if value.is_finite() => Ok(*value),
97            _ => Err(CoordinateError::InvalidRecord(format!(
98                "invalid tabular {field}"
99            ))),
100        }
101    };
102    Ok([read("crval")?, read("cdelt")?, read("crpix")?])
103}

105pub(crate) fn validate_live_edit(edit: TabularEdit) -> Result<(), CoordinateError> {
106    let value = match edit {
107        TabularEdit::ReferencePixel(value)
108        | TabularEdit::ReferenceValue(value)
109        | TabularEdit::Increment(value) => value,
110    };
111    if !value.is_finite() || (value == 0.0 && matches!(edit, TabularEdit::Increment(_))) {
112        return Err(CoordinateError::InvalidRecord(
113            "invalid tabular reference parameter".into(),
114        ));
115    }
116    Ok(())
117}

125/// Apply casacore's tabular affine edit to its sampled world values.
126pub(crate) fn edit_table(
127    pixel_values: &[f64],
128    world_values: &[f64],
129    reference_value: f64,
130    increment: f64,
131    reference_pixel: f64,
132    edit: TabularEdit,
133) -> Result<EditedTable, CoordinateError> {
134    if pixel_values.len() < 2 || pixel_values.len() != world_values.len() || !increment.is_normal()
135    {
136        return Err(CoordinateError::InvalidRecord(
137            "invalid tabular coordinate or subnormal increment (#977)".into(),
138        ));
139    }
140    let (old, values) = match edit {
141        TabularEdit::ReferencePixel(value) => (
142            reference_pixel,
143            world_values
144                .iter()
145                .map(|world| *world - increment * (value - reference_pixel))
146                .collect::<Vec<_>>(),
147        ),
148        TabularEdit::ReferenceValue(value) => (
149            reference_value,
150            world_values
151                .iter()
152                .map(|world| *world + (value - reference_value))
153                .collect::<Vec<_>>(),
154        ),
155        TabularEdit::Increment(value) => (
156            increment,
157            world_values
158                .iter()
159                .map(|world| reference_value + (*world - reference_value) * (value / increment))
160                .collect::<Vec<_>>(),
161        ),
162    };
163    if values.iter().any(|value| !value.is_finite()) {
164        return Err(CoordinateError::InvalidRecord(
165            "tabular edit produces non-finite values (#977)".into(),
166        ));
167    }
168    let first = values[0];
169    let last = values[values.len() - 1];
170    let span = pixel_values[pixel_values.len() - 1] - pixel_values[0];
171    let increment = (last - first) / span;
172    let increasing = last > first;
173    let reversed = increasing != (world_values[world_values.len() - 1] > world_values[0]);
174    let first_order = world_values[1]
175        .partial_cmp(&world_values[0])
176        .map(|order| if reversed { order.reverse() } else { order });
177    if !increment.is_normal()
178        || values[1].partial_cmp(&first) != first_order
179        || values
180            .windows(2)
181            .skip(1)
182            .any(|pair| pair[0] == pair[1] || (pair[1] > pair[0]) != increasing)
183    {
184        return Err(CoordinateError::InvalidRecord(
185            "tabular edit produces an invalid increment (#977)".into(),
186        ));
187    }
188    Ok(EditedTable {
189        old,
190        values,
191        increment,
192    })
193}

195/// A one-axis coordinate defined by a pair of lookup tables.
196///
197/// The `pixel_values` and `world_values` vectors have the same length.
198/// Each table can increase or decrease independently. A single sample
199/// defines a constant mapping in both directions.
200///
201/// Corresponds to C++ `TabularCoordinate`.
202#[derive(Debug, Clone)]
203pub struct TabularCoordinate {
204    pixel_values: Vec<f64>,
205    world_values: Vec<f64>,
206    /// Linear tabular coordinates carry no sampled lookup table.
207    linear: Option<(f64, f64, f64)>,
208    sample_reference: Option<[f64; 3]>,
209    linear_transform: f64,
210    name: String,
211    unit: String,
212}
213
214impl TabularCoordinate {
215    /// Creates a new tabular coordinate.
216    ///
217    /// Both vectors must have the same length (at least one). The pixel values
218    /// should be monotonically increasing or decreasing.
219    ///
220    /// # Panics
221    ///
222    /// Panics if the vectors have different lengths or are empty.
223    pub fn new(
224        pixel_values: Vec<f64>,
225        world_values: Vec<f64>,
226        name: impl Into<String>,
227        unit: impl Into<String>,
228    ) -> Self {
229        assert_eq!(
230            pixel_values.len(),
231            world_values.len(),
232            "pixel and world tables must have the same length"
233        );
234        assert!(
235            !pixel_values.is_empty(),
236            "tabular coordinate requires at least one entry"
237        );
238        Self {
239            pixel_values,
240            world_values,
241            linear: None,
242            sample_reference: None,
243            linear_transform: 1.0,
244            name: name.into(),
245            unit: unit.into(),
246        }
247    }
248
249    /// Create a tabular coordinate from user-supplied samples.
250    ///
251    /// # Errors
252    /// Returns an error for empty, unmatched, nonfinite, or unusable tables.
253    pub fn from_samples(
254        pixel: Vec<f64>,
255        world: Vec<f64>,
256        name: impl Into<String>,
257        unit: impl Into<String>,
258    ) -> Result<Self, CoordinateError> {
259        Self::validate_samples(&pixel, &world)?;
260        Ok(Self::new(pixel, world, name, unit))
261    }
262
263    pub(crate) fn validate_samples(pixel: &[f64], world: &[f64]) -> Result<(), CoordinateError> {
264        let invalid = |message: &str| CoordinateError::InvalidRecord(message.into());
265        if pixel.is_empty() || pixel.len() != world.len() {
266            return Err(invalid(
267                "tabular pixel and world tables must have the same nonzero length",
268            ));
269        }
270        if pixel.iter().chain(world).any(|x| !x.is_finite()) {
271            return Err(invalid("tabular samples must be finite"));
272        }
273        if pixel.len() > 1 {
274            let last = pixel.len() - 1;
275            let dp = pixel[last] - pixel[0];
276            let dw = world[last] - world[0];
277            if dp == 0.0 {
278                return Err(invalid(
279                    "TabularCoordinate::TabularCoordinate - illegal table first and last pixel values are the same",
280                ));
281            }
282            if dw / dp == 0.0 {
283                return Err(invalid(
284                    "TabularCoordinate - start and end values in table must differ",
285                ));
286            }
287            if !dp.is_finite() || !dw.is_finite() || !(dw / dp).is_finite() {
288                return Err(invalid("tabular sample span must be finite"));
289            }
290            // Interpolate1D compares adjacent pixels with nearAbs before it
291            // can evaluate or serialize the world samples.
292            if pixel
293                .windows(2)
294                .any(|pair| (pair[1] - pair[0]).abs() <= CASACORE_PIXEL_TOLERANCE)
295            {
296                return Err(invalid("tabular table has repeated pixel values"));
297            }
298            // Casacore allows the first interval to run in either direction.
299            if pixel
300                .windows(2)
301                .skip(1)
302                .any(|pair| (pair[1] - pair[0]) * dp.signum() <= 0.0)
303                || world
304                    .windows(2)
305                    .skip(1)
306                    .any(|pair| (pair[1] - pair[0]) * dw.signum() <= 0.0)
307            {
308                return Err(invalid(
309                    "TabularCoordinate - pixel and world values must increase or decrease monotonically",
310                ));
311            }
312        }
313        Ok(())
314    }

345    fn has_reversed_pixel_interval(&self) -> bool {
346        self.pixel_values.len() > 2
347            && (self.pixel_values[1] > self.pixel_values[0])
348                != (self.pixel_values.last().unwrap() > &self.pixel_values[0])
349    }

351    /// The multiplier applied to offsets from the reference world value.
352    pub fn linear_transform(&self) -> f64 {
353        self.linear_transform
354    }

356    /// Set the live linear transform of this coordinate.
357    ///
358    /// # Errors
359    /// Returns an error for a nonfinite transform or collapsed lookup table.
360    pub fn with_linear_transform(mut self, transform: f64) -> Result<Self, CoordinateError> {
361        if self.world_values.len() > 1 {
362            validate_transform_samples(&self.world_values, self.reference_world(), transform)?;
363        } else if !transform.is_finite() {
364            return Err(CoordinateError::InvalidRecord(
365                "linear transform would produce invalid tabular samples".into(),
366            ));
367        }
368        self.linear_transform = transform;
369        if self.pixel_values.len() == 1 && !self.constant_world().is_finite() {
370            return Err(CoordinateError::InvalidRecord(
371                "linear transform would produce invalid tabular samples".into(),
372            ));
373        }
374        Ok(self)
375    }
376
377    fn scaled_world(&self, value: f64) -> f64 {
378        scaled_world(value, self.reference_world(), self.linear_transform)
379    }
380
381    fn reference_world(&self) -> f64 {
382        if let Some((value, _, _)) = self.linear {
383            value
384        } else if let Some([value, _, _]) = self.sample_reference {
385            value
386        } else {
387            self.world_values[0]
388        }
389    }
390
391    fn constant_world(&self) -> f64 {
392        match self.sample_reference {
393            Some([value, increment, pixel]) => {
394                value + increment * self.linear_transform * (self.pixel_values[0] - pixel)
395            }
396            None => self.world_values[0],
397        }
398    }

400    /// Restore live reference parameters from an in-memory record containing
401    /// unscaled lookup values. Public record restoration normalizes them.
402    ///
403    /// # Errors
404    /// Returns an error if a required reference parameter is missing.
405    pub fn with_live_reference(mut self, record: &RecordValue) -> Result<Self, CoordinateError> {
406        let [crval, cdelt, crpix] = sample_reference(record)?;
407        if self.linear.is_some() {
408            self.linear = Some((crval, cdelt, crpix));
409        } else {
410            if self.has_reversed_pixel_interval() {
411                // The stored first sample was evaluated in a later interval.
412                // Recover the initial endpoint of the live interpolation.
413                self.world_values[0] = crval + cdelt * (self.pixel_values[0] - crpix);
414            }
415            self.sample_reference = Some([crval, cdelt, crpix]);
416        }
417        Ok(self)
418    }

420    /// Edit the live reference without renormalizing a sampled coordinate.
421    ///
422    /// # Errors
423    /// Returns an error if the edit produces an invalid mapping.
424    pub fn edit_live_reference(&mut self, edit: TabularEdit) -> Result<(), CoordinateError> {
425        if self.pixel_values.len() == 1 {
426            self.edit(edit)?;
427            return Ok(());
428        }
429        validate_live_edit(edit)?;
430        let reference = [
431            self.reference_value()[0],
432            self.increment()[0],
433            self.reference_pixel()[0],
434        ];
435        self.edit(edit)?;
436        if self.linear.is_none() {
437            self.sample_reference = Some(edit.apply(reference));
438        }
439        Ok(())
440    }

442    pub(crate) fn edit(&mut self, edit: TabularEdit) -> Result<f64, CoordinateError> {
443        if let Some((ref mut crval, ref mut cdelt, ref mut crpix)) = self.linear {
444            return Ok(match edit {
445                TabularEdit::ReferencePixel(value) => std::mem::replace(crpix, value),
446                TabularEdit::ReferenceValue(value) => std::mem::replace(crval, value),
447                TabularEdit::Increment(value) => std::mem::replace(cdelt, value),
448            });
449        }
450        if self.pixel_values.len() == 1 {
451            let [mut world, mut increment, mut pixel] =
452                self.sample_reference
453                    .unwrap_or([self.world_values[0], 0.0, self.pixel_values[0]]);
454            let (slot, value) = match edit {
455                TabularEdit::ReferencePixel(value) => (&mut pixel, value),
456                TabularEdit::ReferenceValue(value) => (&mut world, value),
457                TabularEdit::Increment(value) => (&mut increment, value),
458            };
459            let old = std::mem::replace(slot, value);
460            if !value.is_finite()
461                || !(world + increment * self.linear_transform * (self.pixel_values[0] - pixel))
462                    .is_finite()
463            {
464                return Err(CoordinateError::InvalidRecord(
465                    "invalid tabular reference parameter".into(),
466                ));
467            }
468            self.sample_reference = Some([world, increment, pixel]);
469            return Ok(old);
470        }
471        let updated = edit_table(
472            &self.pixel_values,
473            &self.world_values,
474            self.reference_value()[0],
475            self.increment()[0],
476            self.reference_pixel()[0],
477            edit,
478        )?;
479        self.world_values = updated.values;
480        Ok(updated.old)
481    }

483    /// Reconstructs a tabular coordinate from a serialized record.
484    pub fn from_record(rec: &RecordValue) -> Result<Self, CoordinateError> {
485        let pixel_values = get_required_vec_f64(rec, "pixelvalues")?;
486        let world_values = get_required_vec_f64(rec, "worldvalues")?;
487        let name = get_optional_string(rec, "name")
488            .or_else(|| get_optional_vec_string(rec, "axes").and_then(|v| v.into_iter().next()))
489            .unwrap_or_else(|| "Tabular".into());
490        let unit = get_optional_string(rec, "unit")
491            .or_else(|| get_optional_vec_string(rec, "units").and_then(|v| v.into_iter().next()))
492            .unwrap_or_default();
493        if pixel_values.is_empty() && world_values.is_empty() {
494            let read = |field| {
495                let values = get_required_vec_f64(rec, field)?;
496                if values.len() == 1 {
497                    Ok(values[0])
498                } else {
499                    Err(CoordinateError::InvalidRecord(format!(
500                        "invalid tabular {field}"
501                    )))
502                }
503            };
504            return Ok(Self::linear(
505                read("crval")?,
506                read("cdelt")?,
507                read("crpix")?,
508                name,
509                unit,
510            ));
511        }
512        if pixel_values.len() != world_values.len() {
513            return Err(CoordinateError::InvalidRecord(
514                "tabular coordinate requires matching pixel/world tables".into(),
515            ));
516        }
517        Ok(Self::new(pixel_values, world_values, name, unit))
518    }
519}

521/// Linear interpolation: given tables `xs` and `ys`, find the `y` value
522/// corresponding to `x` by interpolating (or extrapolating) between the
523/// nearest bracketing entries.
524fn interpolate(
525    xs: impl Fn(usize) -> f64,
526    ys: &[f64],
527    x: f64,
528    tolerance: f64,
529) -> Result<f64, CoordinateError> {
530    let n = ys.len();
531    if n < 2 || x.is_nan() {
532        return Err(CoordinateError::ConversionFailed(
533            "tabular interpolation requires a valid argument and at least 2 entries".into(),
534        ));
535    }
536
537    // Determine if xs is increasing or decreasing
538    let increasing = xs(n - 1) > xs(0);
539
540    // Search in sample order with the lower middle position, as casacore
541    // does. The first interval can run opposite to the others.
542    let mut lo = 0;
543    let mut hi = n;
544    while lo < hi {
545        let mid = lo + (hi - lo - 1) / 2;
546        let sample = xs(mid);
547        if (increasing && sample < x) || (!increasing && sample > x) {
548            lo = mid + 1;
549        } else {
550            hi = mid;
551        }
552    }
553
554    // Clamp to valid interval for interpolation/extrapolation
555    let lo = lo.saturating_sub(1).min(n - 2);
556    let hi = lo + 1;
557
558    let lower = xs(lo);
559    let upper = xs(hi);
560    let dx = upper - lower;
561    if !dx.is_finite() || dx.abs() <= tolerance {
562        return Err(CoordinateError::ConversionFailed(
563            "Interpolate1D::operator() data has repeated x values".into(),
564        ));
565    }
566    if x == lower {
567        return Ok(ys[lo]);
568    }
569    if x == upper {
570        return Ok(ys[hi]);
571    }
572    let t = (x - lower) / dx;
573    Ok(ys[lo] + t * (ys[hi] - ys[lo]))
574}
575
576pub(crate) fn interpolate_samples(xs: &[f64], ys: &[f64], x: f64) -> Result<f64, CoordinateError> {
577    if xs.is_empty() || xs.len() != ys.len() {
578        return Err(CoordinateError::ConversionFailed(
579            "tabular interpolation requires matching lookup tables with at least one entry".into(),
580        ));
581    }
582    if xs.len() == 1 {
583        return Ok(ys[0]);
584    }
585    interpolate(|index| xs[index], ys, x, 0.0)
586}

588pub(crate) fn inverse_sampled_axis(
589    pixels: &[f64],
590    worlds: &[f64],
591    reference: f64,
592    transform: f64,
593    world: f64,
594) -> Result<f64, CoordinateError> {
595    if pixels.len() < 2 || pixels.len() != worlds.len() {
596        return interpolate_samples(worlds, pixels, world);
597    }
598    // Casacore's inverse channel corrector interpolates in the average
599    // pixel domain, so its duplicate tolerance is independent of world units.
600    let last = pixels.len() - 1;
601    let origin = worlds[0];
602    let increment = (worlds[last] - origin) / (pixels[last] - pixels[0]);
603    let corrected_pixel = |value| (value - origin) / increment + pixels[0];
604    let target = corrected_pixel(reference) + (world - reference) / (increment * transform);
605    interpolate(
606        |index| corrected_pixel(worlds[index]),
607        pixels,
608        target,
609        CASACORE_PIXEL_TOLERANCE,
610    )
611}

613impl Coordinate for TabularCoordinate {
614    fn coordinate_type(&self) -> CoordinateType {
615        CoordinateType::Tabular
616    }
617
618    fn n_pixel_axes(&self) -> usize {
619        1
620    }
621
622    fn n_world_axes(&self) -> usize {
623        1
624    }
625
626    fn to_world(&self, pixel: &[f64]) -> Result<Vec<f64>, CoordinateError> {
627        if pixel.len() != 1 {
628            return Err(CoordinateError::DimensionMismatch {
629                expected: 1,
630                got: pixel.len(),
631            });
632        }
633        let world = if let Some((crval, cdelt, crpix)) = self.linear {
634            crval + cdelt * self.linear_transform * (pixel[0] - crpix)
635        } else if self.pixel_values.len() == 1 {
636            self.constant_world()
637        } else {
638            self.scaled_world(interpolate_samples(
639                &self.pixel_values,
640                &self.world_values,
641                pixel[0],
642            )?)
643        };
644        Ok(vec![world])
645    }
646
647    fn to_pixel(&self, world: &[f64]) -> Result<Vec<f64>, CoordinateError> {
648        if world.len() != 1 {
649            return Err(CoordinateError::DimensionMismatch {
650                expected: 1,
651                got: world.len(),
652            });
653        }
654        let pixel = if let Some((crval, cdelt, crpix)) = self.linear {
655            if cdelt.abs() < 1e-300 {
656                return Err(CoordinateError::ConversionFailed(
657                    "zero tabular increment".into(),
658                ));
659            }
660            crpix + (world[0] - crval) / (cdelt * self.linear_transform)
661        } else {
662            inverse_sampled_axis(
663                &self.pixel_values,
664                &self.world_values,
665                self.reference_world(),
666                self.linear_transform,
667                world[0],
668            )?
669        };
670        Ok(vec![pixel])
671    }
672
673    fn reference_value(&self) -> Vec<f64> {
674        vec![self.reference_world()]
675    }
676
677    fn reference_pixel(&self) -> Vec<f64> {
678        let value = if let Some((_, _, value)) = self.linear {
679            value
680        } else if let Some([_, _, value]) = self.sample_reference {
681            value
682        } else {
683            self.pixel_values[0]
684        };
685        vec![value]
686    }
687
688    fn increment(&self) -> Vec<f64> {
689        if let Some((_, cdelt, _)) = self.linear {
690            return vec![cdelt];
691        }
692        if let Some([_, cdelt, _]) = self.sample_reference {
693            return vec![cdelt];
694        }
695        // Casacore derives the global increment from the table endpoints.
696        let last = self.pixel_values.len() - 1;
697        let dp = self.pixel_values[last] - self.pixel_values[0];
698        let dw = self.world_values[last] - self.world_values[0];
699        if dp.abs() < 1e-300 {
700            vec![0.0]
701        } else {
702            vec![dw / dp]
703        }
704    }
705
706    fn axis_names(&self) -> Vec<String> {
707        vec![self.name.clone()]
708    }
709
710    fn axis_units(&self) -> Vec<String> {
711        vec![self.unit.clone()]
712    }
713}
714
715impl TabularCoordinate {
716    pub(crate) fn to_record(&self) -> RecordValue {
717        let mut rec = RecordValue::default();
718        rec.upsert(
719            "crval",
720            Value::Array(ArrayValue::from_f64_vec(self.reference_value())),
721        );
722        rec.upsert(
723            "crpix",
724            Value::Array(ArrayValue::from_f64_vec(self.reference_pixel())),
725        );
726        rec.upsert(
727            "cdelt",
728            Value::Array(ArrayValue::from_f64_vec(self.increment())),
729        );
730        rec.upsert(
731            "pc",
732            Value::Array(ArrayValue::Float64(
733                ndarray::ArrayD::from_shape_vec(
734                    ndarray::IxDyn(&[1, 1]),
735                    vec![self.linear_transform],
736                )
737                .expect("1x1 tabular pc matrix"),
738            )),
739        );
740        rec.upsert(
741            "axes",
742            Value::Array(ArrayValue::from_string_vec(self.axis_names())),
743        );
744        rec.upsert(
745            "units",
746            Value::Array(ArrayValue::from_string_vec(self.axis_units())),
747        );
748        rec.upsert(
749            "pixelvalues",
750            Value::Array(ArrayValue::from_f64_vec(self.pixel_values.clone())),
751        );
752        let world_values = if self.world_values.len() == 1 {
753            vec![self.constant_world()]
754        } else if self.has_reversed_pixel_interval() {
755            // Casacore saves evaluated samples, including when the first
756            // supplied pixel falls in a later interval of the table.
757            self.pixel_values
758                .iter()
759                .zip(&self.world_values)
760                .map(|(&pixel, &world)| {
761                    self.scaled_world(
762                        interpolate_samples(&self.pixel_values, &self.world_values, pixel)
763                            .unwrap_or(world),
764                    )
765                })
766                .collect()
767        } else {
768            self.world_values
769                .iter()
770                .map(|&value| self.scaled_world(value))
771                .collect()
772        };
773        rec.upsert(
774            "worldvalues",
775            Value::Array(ArrayValue::from_f64_vec(world_values)),
776        );
777        rec
778    }
779}

785    fn make_linear_tab() -> TabularCoordinate {
786        // Linear table: pixel 0..4 -> world 100..500
787        TabularCoordinate::new(
788            vec![0.0, 1.0, 2.0, 3.0, 4.0],

795    #[test]
796    fn exact_table_values() {
797        let coord = make_linear_tab();
798        assert_eq!(coord.to_world(&[0.0]).unwrap(), vec![100.0]);
799        assert_eq!(coord.to_world(&[2.0]).unwrap(), vec![300.0]);
800        assert_eq!(coord.to_world(&[4.0]).unwrap(), vec![500.0]);
801    }
802
803    #[test]
804    fn interpolation() {
805        let coord = make_linear_tab();
806        let world = coord.to_world(&[1.5]).unwrap();
807        assert!((world[0] - 250.0).abs() < 1e-10);
808    }
809
810    #[test]
811    fn reverse_interpolation() {
812        let coord = make_linear_tab();
813        let pixel = coord.to_pixel(&[250.0]).unwrap();
814        assert!((pixel[0] - 1.5).abs() < 1e-10);
815    }
816
817    #[test]
818    fn roundtrip() {
819        let coord = make_linear_tab();
820        let pixel = vec![2.7];
821        let world = coord.to_world(&pixel).unwrap();
822        let back = coord.to_pixel(&world).unwrap();
823        assert!((back[0] - pixel[0]).abs() < 1e-10);
824    }
825
826    #[test]
827    fn non_linear_table() {
828        // Non-linear spacing
829        let coord = TabularCoordinate::new(
830            vec![0.0, 1.0, 2.0, 3.0],
831            vec![0.0, 1.0, 4.0, 9.0], // quadratic-like
832            "X",
833            "",
834        );
835        // At pixel 1.5, world should be interpolated between 1.0 and 4.0 = 2.5
836        let world = coord.to_world(&[1.5]).unwrap();
837        assert!((world[0] - 2.5).abs() < 1e-10);
838    }

840    #[test]
841    fn extrapolation() {
842        let coord = make_linear_tab();
843        // Extrapolate beyond table

846    }
