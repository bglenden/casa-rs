12pub(super) struct FitSelection {
13    pub(super) region: SubimageSelection,
14    pub(super) sky_axes: [usize; 2],
15    pub(super) sky_shape: [usize; 2],
16    pub(super) sky_stride: [usize; 2],
17    pub(super) channels: Vec<usize>,
18    pub(super) stokes_pixel: usize,
19    pub(super) stokes: StokesType,
20    pub(super) ancillary: Vec<(usize, usize)>,
21    pub(super) can_fit_plane: bool,
22}
23
24impl FitSelection {
25    pub(super) fn resolve<T: ImagePixel>(
26        image: &dyn ImageInterface<T>,
27        pixel_box: Option<&str>,
28        region: Option<SubimageRegion<'_>>,
29        channels: Option<&str>,
30        stokes: Option<&str>,
31    ) -> Result<Self, ImageError> {
32        if image
33            .coordinates()
34            .find_coordinate(CoordinateType::Direction)
35            .is_none()
36        {
37            return Err(ImageError::CasaSelection(
38                "Image does not have required coordinate Direction".into(),
39            ));
40        }
41        let sky_axes = direction_plane_axes(image)?;
42        let region =
43            SubimageSelection::resolve_section(image, pixel_box, region, channels, stokes)?;
44        if stokes.is_some_and(|value| !value.trim().is_empty())
45            && region
46                .selected_stokes
47                .as_ref()
48                .is_some_and(|planes| planes.len() > 1)
49        {
50            return Err(ImageError::CasaSelection(
51                "Only a single Stokes plane is supported".into(),
52            ));
53        }
54        let sky_stride =
55            sky_axes.map(|axis| region.stride.as_ref().map_or(1, |stride| stride[axis]));
56        let sky_shape = std::array::from_fn(|i| region.shape[sky_axes[i]].div_ceil(sky_stride[i]));
57        let channels = if let Some(axis) = region.channel_axis {
58            let stride = region.stride.as_ref().map_or(1, |stride| stride[axis]);
59            let first = region.start[axis];
60            let selected: Vec<_> = (first..first + region.shape[axis])
61                .step_by(stride)
62                .filter(|&channel| {
63                    region.channel_spans.is_empty()
64                        || region
65                            .channel_spans
66                            .iter()
67                            .any(|span| span.contains(channel))
68                })
69                .collect();
70            if selected.windows(2).any(|pair| pair[1] != pair[0] + 1) {
71                return Err(ImageError::CasaSelection(
72                    "Only a single contiguous channel range is supported".into(),
73                ));
74            }
75            selected
76        } else {
77            vec![0]
78        };
79        let stokes_pixel = region.stokes_axis.map_or(0, |axis| {
80            region
81                .stokes_indices
82                .as_ref()
83                .and_then(|indices| indices.first().copied())
84                .unwrap_or(region.start[axis])
85        });
86        let stokes = image
87            .coordinates()
88            .find_coordinate(CoordinateType::Stokes)
89            .and_then(|index| match image.coordinates().coordinate(index) {
90                CoordinateModel::Stokes(coordinate) => {
91                    coordinate.stokes().get(stokes_pixel).copied()
92                }
93                _ => None,
94            })
95            .unwrap_or(StokesType::I);
96        let mut can_fit_plane = sky_shape.iter().all(|&axis| axis >= 2);
97        let ancillary = (0..image.ndim())
98            .filter(|axis| {
99                !sky_axes.contains(axis)
100                    && Some(*axis) != region.stokes_axis
101                    && Some(*axis) != region.channel_axis
102            })
103            .map(|axis| {
104                let step = region.stride.as_ref().map_or(1, |stride| stride[axis]);
105                can_fit_plane &= region.shape[axis].div_ceil(step) == 1;
106                (axis, region.start[axis])
107            })
108            .collect();
109        Ok(Self {
110            region,
111            sky_axes,
112            sky_shape,
113            sky_stride,
114            channels,
115            stokes_pixel,
116            stokes,
117            ancillary,
118            can_fit_plane,
119        })
120    }
121
122    pub(super) fn read<T: ImagePixel>(
123        &self,
124        reader: &PixelSections<'_, T>,
125        channel: usize,
126        to_float: impl Fn(T) -> f32,
127    ) -> Result<FitPlane, ImageError> {
128        let selection = &self.region;
129        let [width, height] = self.sky_shape;
130        let [x, y] = self.sky_axes;
131        let [x_step, y_step] = self.sky_stride;
132        let mut start = selection.start.clone();
133        let mut shape = vec![1; start.len()];
134        let stride = selection
135            .stride
136            .clone()
137            .unwrap_or_else(|| vec![1; start.len()]);
138        if let Some(axis) = selection.channel_axis {
139            start[axis] = channel;
140        }
141        if let Some(axis) = selection.stokes_axis {
142            start[axis] = self.stokes_pixel;
143        }
144        let mut values = Vec::new();
145        let mut validity = Vec::new();
146        let block = (64 * 1024_usize).div_ceil(x_step);
147        for row in 0..height {
148            start[y] = selection.start[y] + row * y_step;
149            for column in (0..width).step_by(block) {
150                start[x] = selection.start[x] + column * x_step;
151                shape[x] = block.min(width - column);
152                let (pixels, valid) = reader.read(&start, &shape, &stride)?;
153                values.extend(pixels.iter().copied().map(&to_float));
154                if let Some(valid) = valid {
155                    validity.extend(valid.iter().copied());
156                } else {
157                    validity.resize(values.len(), true);
158                }
159            }
160        }
161        Ok(FitPlane {
162            shape: [width, height],
163            values,
164            validity,
165        })
166    }
167}
