2//! A component's visibility, as CASA's `SkyCompRep::visibility` forms it
3//! (`components/ComponentModels`): the flux in Stokes, scaled by the spectral
4//! model at the frequency, converted to the data's polarisation basis, and
5//! scaled by the shape's visibility at the baseline.
6//!
7//! The baseline is in metres and already rotated to the component's
8//! reference direction, with CASA's sign: `SimpleComponentFTMachine::get`
9//! negates u and v before it rotates them. The frequency is the data's,
10//! taken in the frame of the spectral model's reference frequency, as CASA
11//! takes it ("the frequency is expected to be in the frame of the
12//! component").
13//!
14//! What casa-rs does not evaluate is refused rather than approximated:
15//! - a limb-darkened disk, whose visibility needs a Bessel function of real
16//!   order (`gsl_sf_bessel_Jnu`);
17//! - a tabular spectrum whose reference frequency is in another frame than
18//!   its tabulation, which CASA converts through a frame casa-rs would have
19//!   to build, and a table of fewer than two points, which CASA's linear
20//!   interpolation reads out of bounds;
21//! - a flux unit that is not a flux density (`Jy.km/s`, `K.rad2`), which
22//!   CASA's `convertUnit(Unit("Jy"))` cannot convert either.
23

33/// The speed of light, m/s (casacore `C::c`).
34const SPEED_OF_LIGHT_M_PER_S: f64 = 299_792_458.0;

36impl Flux {
37    /// The flux as real Stokes I, Q, U and V in Jy: `convertUnit("Jy")`, then
38    /// `FluxRep::value(Vector<Double>&)`, which converts to Stokes and keeps
39    /// the real parts.
40    ///
41    /// # Errors
42    ///
43    /// [`ComponentError::FluxUnit`] when the unit is not a flux density.
44    pub fn stokes_jy_values(&self) -> Result<[f64; 4], ComponentError> {
45        let jy = Unit::new("Jy").expect("Jy is a unit");
46        let scale = Quantity::with_unit(1.0, self.unit().clone())
47            .get_value_in(&jy)
48            .map_err(|_| ComponentError::FluxUnit(self.unit().name().to_string()))?;
49        let stokes = to_stokes(*self.values(), self.polarisation());
50        Ok(stokes.map(|value| value.re * scale))
51    }
52}

54/// `values` in `from`'s representation, as complex Stokes I, Q, U and V
55/// (`Flux::linearToStokes`, `Flux::circularToStokes`).
56fn to_stokes(values: [Complex64; 4], from: Polarisation) -> [Complex64; 4] {
57    let [a, b, c, d] = values;
58    match from {
59        Polarisation::Stokes => values,
60        Polarisation::Circular => [
61            (a + d) / 2.0,
62            (b + c) / 2.0,
63            Complex64::new((b.im - c.im) / 2.0, (c.re - b.re) / 2.0),
64            (a - d) / 2.0,
65        ],
66        Polarisation::Linear => [
67            (a + d) / 2.0,
68            (a - d) / 2.0,
69            (b + c) / 2.0,
70            Complex64::new((b.im - c.im) / 2.0, (c.re - b.re) / 2.0),
71        ],
72    }
73}

75/// Real Stokes I, Q, U and V in `basis`'s order (`Flux::stokesToCircular`,
76/// `Flux::stokesToLinear`): RR = I + V, RL = Q + iU, LR = Q − iU, LL = I − V,
77/// or XX = I + Q, XY = U + iV, YX = U − iV, YY = I − Q.
78#[must_use]
79pub fn stokes_in(basis: Polarisation, iquv: [f64; 4]) -> [Complex64; 4] {
80    let [i, q, u, v] = iquv;
81    match basis {
82        Polarisation::Stokes => iquv.map(|value| Complex64::new(value, 0.0)),
83        Polarisation::Circular => [
84            Complex64::new(i + v, 0.0),
85            Complex64::new(q, u),
86            Complex64::new(q, -u),
87            Complex64::new(i - v, 0.0),
88        ],
89        Polarisation::Linear => [
90            Complex64::new(i + q, 0.0),
91            Complex64::new(u, v),
92            Complex64::new(u, -v),
93            Complex64::new(i - q, 0.0),
94        ],
95    }
96}

98impl SpectralModel {
99    /// The Stokes flux at `frequency_hz` of a component whose flux at the
100    /// reference frequency is `iquv`: CASA's matrix `sampleStokes`, which
101    /// `SkyCompRep::visibility` uses.
102    /// - Constant: `iquv` unchanged.
103    /// - Spectral index: I scales as `(ν/ν₀)^α₀`, Q and U as
104    ///   `(ν/ν₀)^(α₀+α₁)` and V as `(ν/ν₀)^(α₀+α₃)`, then Q and U rotate by
105    ///   twice `α₂ c² (ν₀² − ν²)/(ν² ν₀²)` (a rotation measure). A single
106    ///   index is `[α, 0, 0, 0]`: CASA's matrix sampler has no branch for it
107    ///   and reads the missing indices out of bounds (`docs/CASA (C++)
108    ///   bugs.md`), which for an unpolarised component multiplies zeros.
109    /// - Power-log-polynomial: I scales as `x^(c₀ + c₁ ln x + c₂ ln²x + …)`
110    ///   with `x = ν/ν₀`; Q, U and V are unchanged.
111    /// - Tabular: the tabulated Stokes values, interpolated linearly in
112    ///   frequency and extrapolated from the end points
113    ///   (`InterpolateArray1D::linear`); the component's own flux does not
114    ///   enter.
115    ///
116    /// # Errors
117    ///
118    /// [`ComponentError::Spectrum`] for a tabular spectrum casa-rs does not
119    /// evaluate (see the module documentation) or a reference frequency that
120    /// is not positive.
121    pub fn sample_stokes(
122        &self,
123        frequency_hz: f64,
124        iquv: [f64; 4],
125    ) -> Result<[f64; 4], ComponentError> {
126        let nu0 = self.reference_frequency().hz();
127        let positive = |spectrum: &'static str| {
128            if nu0 > 0.0 {
129                Ok(nu0)
130            } else {
131                Err(ComponentError::Spectrum {
132                    spectrum,
133                    reason: format!("the reference frequency {nu0} Hz is not positive"),
134                })
135            }
136        };
137        Ok(match self.kind() {
138            SpectralKind::Constant => iquv,
139            SpectralKind::SpectralIndex(index) => {
140                let nu0 = positive("spectral index")?;
141                let [a0, a1, a2, a3] =
142                    index
143                        .stokes_indices()
144                        .unwrap_or([index.index(), 0.0, 0.0, 0.0]);
145                let ratio = frequency_hz / nu0;
146                let [i, q, u, v] = iquv;
147                let (mut q, mut u) = (q * ratio.powf(a0 + a1), u * ratio.powf(a0 + a1));
148                if a2 != 0.0 {
149                    let nu = frequency_hz;
150                    let two_alpha = 2.0
151                        * a2
152                        * SPEED_OF_LIGHT_M_PER_S
153                        * SPEED_OF_LIGHT_M_PER_S
154                        * (nu0 * nu0 - nu * nu)
155                        / (nu * nu * nu0 * nu0);
156                    let (sin, cos) = two_alpha.sin_cos();
157                    (q, u) = (cos * q - sin * u, sin * q + cos * u);
158                }
159                [i * ratio.powf(a0), q, u, v * ratio.powf(a0 + a3)]
160            }
161            SpectralKind::PowerLogPolynomial(polynomial) => {
162                let nu0 = positive("power-log-polynomial")?;
163                let x = frequency_hz / nu0;
164                let log_x = x.ln();
165                let mut exponent = 0.0;
166                for (power, coefficient) in polynomial.coefficients().iter().enumerate() {
167                    exponent += match power {
168                        0 => *coefficient,
169                        1 => coefficient * log_x,
170                        2 => coefficient * log_x * log_x,
171                        _ => coefficient * log_x.powi(power as i32),
172                    };
173                }
174                let [i, q, u, v] = iquv;
175                [i * x.powf(exponent), q, u, v]
176            }
177            SpectralKind::Tabular(table) => {
178                tabular_stokes(table, self.reference_frequency().refer(), frequency_hz)?
179            }
180        })
181    }
182}

184/// `table`'s Stokes values at `frequency_hz` (see
185/// [`SpectralModel::sample_stokes`]).
186fn tabular_stokes(
187    table: &TabularSpectrum,
188    reference_frame: casa_types::measures::frequency::FrequencyRef,
189    frequency_hz: f64,
190) -> Result<[f64; 4], ComponentError> {
191    let refused = |reason: String| ComponentError::Spectrum {
192        spectrum: "tabular",
193        reason,
194    };
195    if table.frame() != reference_frame {
196        return Err(refused(format!(
197            "a reference frequency in {reference_frame:?} and a tabulation in {:?}: casa-rs \
198             does not convert the data's frequencies between them",
199            table.frame()
200        )));
201    }
202    let frequencies = table.frequencies_hz();
203    if frequencies.len() < 2 {
204        return Err(refused(format!(
205            "{} tabulated frequencies; linear interpolation needs two",
206            frequencies.len()
207        )));
208    }
209    // `binarySearchBrackets`, which takes the tabulation in either order,
210    // then the bracketing pair, or the end pair.
211    let ascending = frequencies[0] <= frequencies[frequencies.len() - 1];
212    let upper = frequencies
213        .partition_point(|frequency| {
214            if ascending {
215                *frequency < frequency_hz
216            } else {
217                *frequency > frequency_hz
218            }
219        })
220        .clamp(1, frequencies.len() - 1);
221    let (x1, x2) = (frequencies[upper - 1], frequencies[upper]);
222    let fraction = (frequency_hz - x1) / (x2 - x1);
223    let (y1, y2) = (table.stokes_flux()[upper - 1], table.stokes_flux()[upper]);
224    Ok(std::array::from_fn(|stokes| {
225        y1[stokes] + fraction * (y2[stokes] - y1[stokes])
226    }))
227}

229impl ComponentShape {
230    /// The shape's visibility at `uvw_m` and `frequency_hz` (`visibility`
231    /// of `PointShape`, `GaussianShape` and `DiskShape`), real.
232    ///
233    /// # Errors
234    ///
235    /// [`ComponentError::Visibility`] for a limb-darkened disk.
236    pub fn visibility_scale(
237        &self,
238        uvw_m: [f64; 3],
239        frequency_hz: f64,
240    ) -> Result<f64, ComponentError> {
241        Ok(self
242            .transform()?
243            .at_baseline(uvw_m)
244            .at_frequency(frequency_hz))
245    }
246
247    /// The shape's Fourier transform, prepared from its axes for evaluation
248    /// at many baselines and frequencies, as `GaussianShape::updateFT` caches
249    /// the Gaussian's.
250    ///
251    /// # Errors
252    ///
253    /// [`ComponentError::Visibility`] for a limb-darkened disk.
254    pub fn transform(&self) -> Result<ShapeTransform, ComponentError> {
255        Ok(ShapeTransform(match self.kind() {
256            ShapeKind::Point => Transform::Point,
257            ShapeKind::Gaussian { axes, .. } => Transform::gaussian(axes),
258            ShapeKind::Disk { axes, .. } => Transform::Disk {
259                rotation: (axes.position_angle().abs() > NEAR_ABS)
260                    .then(|| axes.position_angle().sin_cos()),
261                minor_axis: axes.minor_axis(),
262                major_axis: axes.major_axis(),
263            },
264            ShapeKind::LimbDarkenedDisk { .. } => {
265                return Err(ComponentError::Visibility {
266                    shape: "limb-darkened disk",
267                    reason: "its visibility needs a Bessel function of real order".to_string(),
268                });
269            }
270        }))
271    }
272}

274/// A shape's Fourier transform ([`ComponentShape::transform`]), holding no
275/// heap. [`Self::at_baseline`] does the work that depends on the baseline,
276/// so a baseline's channels each evaluate a function of frequency alone.
277#[derive(Debug, Clone, Copy, PartialEq)]
278pub struct ShapeTransform(Transform);
279
280#[derive(Debug, Clone, Copy, PartialEq)]
281enum Transform {
282    Point,
283    /// `GaussianShape`'s cached transform: casacore's `Gaussian2D` of unit
284    /// height evaluated at `(x, y) = (−u, v) ν/c` (`Gaussian2D::eval`), as
285    /// `exp(−(x′² + y′²))`. Its rotated and scaled coordinates are rows of
286    /// coefficients on `(u, v)`, so that `x′ = (ν/c) (x_row · (u, v))` and
287    /// likewise `y′`.
288    Gaussian {
289        x_row: [f64; 2],
290        y_row: [f64; 2],
291    },
292    /// `DiskShape::visibility`: the baseline rotated by the position angle
293    /// when it is not near zero, and scaled by the axes.
294    Disk {
295        rotation: Option<(f64, f64)>,
296        minor_axis: f64,
297        major_axis: f64,
298    },
299}
300
301impl Transform {
302    /// The Gaussian of `axes`: widths `4 ln 2/(π · minor)` and `4 ln 2/(π ·
303    /// major)` and the shape's position angle plus π/2
304    /// (`GaussianShape::updateFT`).
305    fn gaussian(axes: &Ellipse) -> Self {
306        let factor = 4.0 * std::f64::consts::LN_2 / std::f64::consts::PI;
307        // `setWidth`: the larger width is the major one, held as the y
308        // width, and the ratio is x over y; `setPA` keeps the angle when y is
309        // the larger width, as here.
310        let y_width = factor / axes.minor_axis();
311        let ratio = (factor / axes.major_axis()) / y_width;
312        let angle = axes.position_angle() + std::f64::consts::FRAC_PI_2;
313        let (sin, cos) = angle.sin_cos();
314        let fwhm_to_sigma = 1.0 / 16.0_f64.ln().sqrt();
315        // x′ = (cos x + sin y)/σx and y′ = (−sin x + cos y)/σy, with x = −u
316        // and y = v.
317        let (x_sigma, y_sigma) = (y_width * ratio * fwhm_to_sigma, y_width * fwhm_to_sigma);
318        Self::Gaussian {
319            x_row: [-cos / x_sigma, sin / x_sigma],
320            y_row: [sin / y_sigma, cos / y_sigma],
321        }
322    }
323}

325impl ShapeTransform {
326    /// The transform at `uvw_m` (rotated to the shape's reference direction,
327    /// CASA's sign), as a function of frequency.
328    #[must_use]
329    pub fn at_baseline(&self, uvw_m: [f64; 3]) -> BaselineTransform {
330        let [u, v, _] = uvw_m;
331        let wavenumber_squared = 1.0 / (SPEED_OF_LIGHT_M_PER_S * SPEED_OF_LIGHT_M_PER_S);
332        BaselineTransform(match self.0 {
333            Transform::Point => AtBaseline::One,
334            Transform::Gaussian { x_row, y_row } => {
335                let x = x_row[0] * u + x_row[1] * v;
336                let y = y_row[0] * u + y_row[1] * v;
337                AtBaseline::Gaussian((x * x + y * y) * wavenumber_squared)
338            }
339            // 1 where `u + v` is near zero, as CASA tests it (casacore
340            // `near`, relative to 1e-13), and otherwise `2 J₁(r)/r` of the
341            // rotated baseline scaled by the axes (the minor axis on u, the
342            // major on v) and by `π ν/c`.
343            Transform::Disk { .. } if near_zero(u + v) => AtBaseline::One,
344            Transform::Disk {
345                rotation,
346                minor_axis,
347                major_axis,
348            } => {
349                let (u, v) =
350                    rotation.map_or((u, v), |(sin, cos)| (u * cos - v * sin, u * sin + v * cos));
351                AtBaseline::Disk(
352                    (u * minor_axis).hypot(v * major_axis) * std::f64::consts::PI
353                        / SPEED_OF_LIGHT_M_PER_S,
354                )
355            }
356        })
357    }
358}

360/// A shape's transform at one baseline ([`ShapeTransform::at_baseline`]).
361#[derive(Debug, Clone, Copy, PartialEq)]
362pub struct BaselineTransform(AtBaseline);
363
364#[derive(Debug, Clone, Copy, PartialEq)]
365enum AtBaseline {
366    One,
367    /// `exp(−g ν²)`.
368    Gaussian(f64),
369    /// `2 J₁(r)/r` with `r = R ν`.
370    Disk(f64),
371}
372
373impl BaselineTransform {
374    /// The shape's visibility at `frequency_hz`, real.
375    #[must_use]
376    pub fn at_frequency(&self, frequency_hz: f64) -> f64 {
377        match self.0 {
378            AtBaseline::One => 1.0,
379            AtBaseline::Gaussian(per_hz_squared) => {
380                (-per_hz_squared * frequency_hz * frequency_hz).exp()
381            }
382            AtBaseline::Disk(per_hz) => {
383                let r = per_hz * frequency_hz;
384                2.0 * libm::j1(r) / r
385            }
386        }
387    }
388}

390/// casacore's default `nearAbs` tolerance.
391const NEAR_ABS: f64 = 1.0e-13;
392
393/// casacore `near(x, 0.0)` with its default 1e-13: relative to zero, only
394/// values within `DBL_MIN` of it.
395fn near_zero(x: f64) -> bool {
396    x == 0.0 || x.abs() <= (1.0 + 1.0e-13) * f64::MIN_POSITIVE
397}

399impl SkyComponent {
400    /// The component's visibility at `uvw_m` (rotated to its reference
401    /// direction, CASA's sign) and `frequency_hz`, in `basis`'s order
402    /// (`SkyCompRep::visibility`): the Stokes flux in Jy sampled by the
403    /// spectral model, converted to `basis`, times the shape's visibility.
404    ///
405    /// # Errors
406    ///
407    /// The flux unit, spectral model or shape casa-rs does not evaluate.
408    pub fn visibility(
409        &self,
410        uvw_m: [f64; 3],
411        frequency_hz: f64,
412        basis: Polarisation,
413    ) -> Result<[Complex64; 4], ComponentError> {
414        let flux = self.flux_in(frequency_hz, basis)?;
415        let scale = self.shape().visibility_scale(uvw_m, frequency_hz)?;
416        Ok(flux.map(|value| value * scale))
417    }
418
419    /// The component's flux at `frequency_hz` in `basis`'s order, in Jy: the
420    /// part of [`Self::visibility`] that does not depend on the baseline.
421    ///
422    /// # Errors
423    ///
424    /// The flux unit or spectral model casa-rs does not evaluate.
425    pub fn flux_in(
426        &self,
427        frequency_hz: f64,
428        basis: Polarisation,
429    ) -> Result<[Complex64; 4], ComponentError> {
430        let iquv = self.flux().stokes_jy_values()?;
431        let sampled = self.spectrum().sample_stokes(frequency_hz, iquv)?;
432        Ok(stokes_in(basis, sampled))
433    }
434}
