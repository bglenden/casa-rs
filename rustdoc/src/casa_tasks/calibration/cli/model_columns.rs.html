14pub(super) fn run(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
15    let path = task.input_path("vis")?;
16    if task.surface() == "uvsub" {
17        casa_calibration::uvsub::subtract_visibility_model(&path, task.flag("reverse")?)
18            .map_err(|error| TaskFailure::execution(error.to_string()))?;
19        super::history::record_task(task, &path);
20        return TaskSuccess::new(CasaValue::None, &());
21    }
22    if let Some(parts) = super::multi_ms::partition_paths(&path)? {
23        let mut session = task.session().clone();
24        for part in parts {
25            session
26                .set(
27                    "vis",
28                    ParameterValue::String(part.to_string_lossy().into_owned()),
29                )
30                .map_err(|error| TaskFailure::execution(error.to_string()))?;
31            let partition = task.with_session(&session);
32            if let Err(error) = run_one(&partition, &part, true) {
33                if error.kind == casa_contracts::TaskFailureKind::Interrupted {
34                    return Err(error);
35                }
36                tracing::warn!(task = task.surface(), partition = %part.display(), error = %error.message, "Multi-MS partition failed");
37            }
38        }
39    } else {
40        run_one(task, &path, false)?;
41    }
42    TaskSuccess::new(CasaValue::None, &())
43}

45fn run_one(task: &TaskInvocation<'_>, path: &Path, partition: bool) -> Result<(), TaskFailure> {
46    let mut lock = acquire_write(task, path).map_err(|error| {
47        if task
48            .native_options()
49            .is_some_and(crate::NativeTaskOptions::is_cancelled)
50        {
51            TaskFailure::interrupted(error.to_string()).at("vis")
52        } else {
53            TaskFailure::execution(error.to_string()).at("vis")
54        }
55    })?;
56    let mut ms = MeasurementSet::open(path)
57        .map_err(|error| TaskFailure::execution(error.to_string()).at("vis"))?;
58    if !partition {
59        // MMS sends native booleans to its workers. The direct tool call
60        // rejects a NumPy Boolean before changing its MeasurementSet.
61        let flags: &[(&str, &str, u8)] = if task.surface() == "clearcal" {
62            &[("addmodel", "calibrater_open", 5)]
63        } else {
64            &[
65                ("otf", "calibrater_delmod", 2),
66                ("scr", "calibrater_delmod", 5),
67            ]
68        };
69        for &(name, method, index) in flags {
70            if matches!(
71                task.session().given(name),
72                Some(ParameterValue::NumpyScalar {
73                    dtype: NumpyScalarType::Bool,
74                    ..
75                })
76            ) {
77                return Err(TaskFailure::execution(format!(
78                    "in method '{method}', argument {index} of type 'bool'"
79                ))
80                .at(name));
81            }
82        }
83    }
84    let field = task.text("field")?;
85    if task.surface() == "clearcal" {
86        let selection = MsSelection {
87            selectdata: true,
88            field: selectvis_text(field),
89            spw: selectvis_text(task.text("spw")?),
90            intent: selectvis_text(task.text("intent")?),
91            ..MsSelection::default()
92        };
93        lock = reset_scratch_columns(&mut ms, &selection, task.flag("addmodel")?, lock)
94            .map_err(|error| TaskFailure::execution(error.to_string()))?;
95    } else {
96        delete_calibration_model(
97            &mut ms,
98            selectvis_text(field).as_deref(),
99            None,
100            task.flag("otf")?,
101            task.flag("scr")?,
102            &mut lock,
103        )
104        .map_err(|error| TaskFailure::execution(error.to_string()))?;
105    }
106    drop(ms);
107    lock.release()
108        .map_err(|error| TaskFailure::execution(error.to_string()).at("vis"))?;
109    super::history::record_task(task, path);
110    Ok(())
111}

113fn acquire_write(
114    task: &TaskInvocation<'_>,
115    path: &Path,
116) -> Result<TableWriteLock, casa_tables::TableError> {
117    #[cfg(unix)]
118    if let Some(options) = task.native_options()
119        && let Some(reservations) = options.table_writes.as_ref()
120    {
121        return TableWriteLock::acquire_reserved_interruptibly(path, reservations, &|| {
122            options.is_cancelled()
123        });
124    }
125    let _ = task;
126    TableWriteLock::acquire_to_reread(path, 0)
127}

129fn selectvis_text(value: Option<String>) -> Option<String> {
130    let text = value?;
131    let selector = text
132        .strip_prefix('[')
133        .and_then(|inner| inner.strip_suffix(']'))
134        .unwrap_or(&text);
135    (!selector.is_empty()).then(|| selector.to_owned())
136}
