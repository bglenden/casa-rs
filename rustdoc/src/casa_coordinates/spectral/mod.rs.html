53/// Index of casacore's nearest listed rest frequency (first on ties).
54pub(crate) fn closest_rest_frequency(values: &[f64], target: f64) -> Option<usize> {
55    values
56        .iter()
57        .copied()
58        .enumerate()
59        .min_by(|(_, a), (_, b)| (a - target).abs().total_cmp(&(b - target).abs()))
60        .map(|(index, _)| index)
61}

63fn frequency_unit_scale(unit: &str) -> Option<f64> {
64    if unit == "Hz" {
65        return Some(1.0);
66    }
67    let unit = Unit::new(unit).ok()?;
68    let hz = Unit::new("Hz").ok()?;
69    let scale = unit.val().factor / hz.val().factor;
70    (unit.conformant(&hz) && scale.is_normal() && scale > 0.0).then_some(scale)
71}

79/// A one-axis spectral coordinate with linear or tabulated pixel-to-world mapping.
80///
81/// Stores the frequency reference frame, a rest frequency for velocity
82/// conversions, axis metadata, and either the standard FITS WCS linear
83/// parameters or an explicit spectral lookup table. This corresponds to C++
84/// `SpectralCoordinate`, including its tabulated non-linear form.
85///
86/// Corresponds to C++ `SpectralCoordinate`.
87#[derive(Debug, Clone)]
88pub struct SpectralCoordinate {
89    /// The velocity reference frame (LSRK, BARY, TOPO, etc.).
90    frequency_ref: FrequencyRef,
91    /// Whether the record declares no frame: C++ `MFrequency::Undefined`,
92    /// which CASA writes for a `specmode='cubedata'` image. Its frequencies
93    /// are the data's as observed, and arithmetic treats them as `frequency_ref`.
94    undefined_frame: bool,
95    /// Optional CASA conversion-layer state for world-coordinate display.
96    conversion: Option<SpectralConversion>,
97    /// The active entry in the rest-frequency list.
98    rest_frequency_index: usize,
99    /// Every rest frequency the coordinate lists, the active one among them
100    /// (C++ `restFrequencies()`).
101    rest_frequencies: Vec<f64>,
102    /// Doppler convention used by this coordinate when expressing velocities.
103    velocity_convention: DopplerRef,
104    /// Preferred spectral representation written to CASA coordinate records.
105    native_type: SpectralNativeType,
106    /// Pixel-to-world mapping state.
107    mapping: SpectralMapping,
108    /// Native WCS (or live sampled) linear transform.
109    linear_transform: f64,
110    /// The world axis name.
111    name: String,
112    /// The world axis unit string.
113    unit: String,
114    /// The stored unit's scale to Hz, if it is a supported frequency unit.
115    unit_scale_hz: Option<f64>,
116}

153    /// Creates a tabulated spectral coordinate from explicit lookup tables.
154    ///
155    /// This mirrors the non-linear record form used by C++
156    /// `SpectralCoordinate` when a `tabular` sub-record contains
157    /// `pixelvalues/worldvalues`.
158    pub fn from_tabular(
159        frequency_ref: FrequencyRef,
160        pixel_values: Vec<f64>,
161        world_values: Vec<f64>,
162        crval: f64,
163        cdelt: f64,

171            ));
172        }
173        Ok(Self {
174            frequency_ref,
175            undefined_frame: frequency_ref == FrequencyRef::Undefined,
176            conversion: None,
177            rest_frequency_index: 0,
178            rest_frequencies: vec![rest_frequency],
179            velocity_convention: DopplerRef::RADIO,
180            native_type: SpectralNativeType::Frequency,
181            mapping: SpectralMapping::Tabular {
182                pixel_values,
183                world_values,
184                crval,
185                cdelt,
186                crpix,
187            },
188            linear_transform: 1.0,
189            name: "Frequency".into(),
190            unit: "Hz".into(),
191            unit_scale_hz: Some(1.0),
192        })
193    }

251    pub(crate) fn world_at_pixel(&self, pixel: f64) -> Result<f64, CoordinateError> {
252        let world = match &self.mapping {
253            SpectralMapping::Linear {
254                crval,
255                cdelt,
256                crpix,
257            } => crval + cdelt * self.linear_transform * (pixel - crpix),
258            SpectralMapping::Tabular {
259                pixel_values,
260                world_values,
261                crval,
262                ..
263            } => {
264                let world = interpolate(pixel_values, world_values, pixel)?;
265                scaled_world(world, *crval, self.linear_transform)
266            }
267        };
268        self.convert_native_to_world(world)
269    }

280    /// Linear FITS frequency axis `(reference pixel, reference Hz, increment Hz)`.
281    ///
282    /// Casacore represents a non-linear table in FITS by the line through
283    /// its first two sampled frequencies, anchored at the first pixel.
284    ///
285    /// # Errors
286    ///
287    /// Returns an error if the unit or pixel-to-world conversion is invalid.
288    pub fn fits_linearized_axis_hz(&self) -> Result<(f64, f64, f64), CoordinateError> {
289        if let SpectralMapping::Tabular { pixel_values, .. } = &self.mapping {
290            let first = self.world_frequency_hz(pixel_values[0])?;
291            let second = match pixel_values.get(1) {
292                Some(&pixel) => self.world_frequency_hz(pixel)?,
293                None => first,
294            };
295            Ok((pixel_values[0], first, second - first))
296        } else {
297            let pixel = self.reference_pixel()[0];
298            Ok((
299                pixel,
300                self.world_frequency_hz(pixel)?,
301                self.world_increment_hz()?,
302            ))
303        }
304    }

321    /// Change the frequency reference value and translate any sampled grid
322    /// by the same amount. Increments and native channel offsets are retained.
323    ///
324    /// # Errors
325    /// Returns an error if the frequency or resulting sampled grid is invalid.
326    pub fn with_reference_frequency_hz(mut self, frequency: f64) -> Result<Self, CoordinateError> {
327        let value = self.value_from_hz(frequency)?;
328        let edit = TabularEdit::ReferenceValue(value);
329        validate_live_edit(edit)?;
330        match &mut self.mapping {
331            SpectralMapping::Linear { crval, .. } => *crval = value,
332            SpectralMapping::Tabular {
333                crval,
334                world_values,
335                ..
336            } if world_values.len() == 1 => {
337                *crval = value;
338                world_values[0] = value;
339            }
340            SpectralMapping::Tabular { .. } => self.apply_table_edit(edit, true).map(|_| ())?,
341        }
342        Ok(self)
343    }
344
345    fn apply_table_edit(&mut self, edit: TabularEdit, live: bool) -> Result<f64, CoordinateError> {
346        let SpectralMapping::Tabular {
347            pixel_values,
348            world_values,
349            crval,
350            cdelt,
351            crpix,
352        } = &mut self.mapping
353        else {
354            return Err(CoordinateError::InvalidRecord(
355                "spectral coordinate is not tabular".into(),
356            ));
357        };
358        let updated = edit_table(pixel_values, world_values, *crval, *cdelt, *crpix, edit)?;
359        [*crval, *cdelt, *crpix] = if live {
360            edit.apply([*crval, *cdelt, *crpix])
361        } else {
362            [updated.values[0], updated.increment, pixel_values[0]]
363        };
364        *world_values = updated.values;
365        Ok(updated.old)
366    }

453    /// Read the live reference and unscaled samples from an in-memory record.
454    /// Public record restoration normalizes the tabulated mapping.
455    ///
456    /// # Errors
457    /// Returns an error for missing reference parameters or invalid samples.
458    pub fn with_live_reference(mut self, record: &RecordValue) -> Result<Self, CoordinateError> {
459        let Some(Value::Record(table)) = record.get("tabular") else {
460            return Err(CoordinateError::InvalidRecord(
461                "missing spectral table".into(),
462            ));
463        };
464        let [crval, cdelt, crpix] = sample_reference(table)?;
465        let pixels = get_required_vec_f64(table, "pixelvalues")?;
466        let mut worlds = get_required_vec_f64(table, "worldvalues")?;
467        let unit = get_optional_vec_string(table, "units")
468            .and_then(|units| units.into_iter().next())
469            .unwrap_or_else(|| "Hz".into());
470        let to_hz = frequency_unit_scale(&unit).ok_or_else(|| {
471            CoordinateError::InvalidRecord(format!("invalid tabular spectral unit {unit:?}"))
472        })?;
473        let scale = to_hz / self.unit_scale_hz.unwrap_or(1.0);
474        for world in &mut worlds {
475            *world *= scale;
476        }
477        self.mapping = Self::from_tabular(
478            self.frequency_ref,
479            pixels,
480            worlds,
481            crval * scale,
482            cdelt * scale,
483            crpix,
484            self.rest_frequency(),
485        )?
486        .mapping;
487        Ok(self)
488    }

544    /// Reconstructs a spectral coordinate from a serialized record.
545    pub fn from_record(rec: &RecordValue) -> Result<Self, CoordinateError> {
546        let undefined_frame;
547        let frequency_ref = if let Some(name) =
548            get_optional_string(rec, "frequency_ref").or_else(|| get_optional_string(rec, "system"))
549        {
550            undefined_frame = name.eq_ignore_ascii_case("Undefined");
551            parse_frequency_ref_name(&name).ok_or_else(|| {
552                CoordinateError::InvalidRecord(format!("invalid spectral frequency_ref: {name}"))
553            })?
554        } else if let Some(code) = get_optional_i32(rec, "frequency_ref") {
555            undefined_frame = code == FrequencyRef::Undefined.casacore_code();
556            FrequencyRef::from_casacore_code(code).ok_or_else(|| {
557                CoordinateError::InvalidRecord(format!(
558                    "invalid spectral frequency_ref code {code}"
559                ))
560            })?
561        } else {
562            return Err(CoordinateError::InvalidRecord(
563                "missing or invalid frequency_ref".into(),
564            ));
565        };
566
567        let mut parameter_record = rec;
568        let mut lookup_tables = None;
569        if let Some(Value::Record(wcs)) = rec.get("wcs") {
570            parameter_record = wcs;
571        } else if let Some(Value::Record(tabular)) = rec.get("tabular") {
572            let pixelvalues = get_optional_vec_f64(tabular, "pixelvalues").unwrap_or_default();
573            let worldvalues = get_optional_vec_f64(tabular, "worldvalues").unwrap_or_default();
574            if !pixelvalues.is_empty() || !worldvalues.is_empty() {
575                lookup_tables = Some((pixelvalues, worldvalues));
576            }
577            parameter_record = tabular;
578        }
579
580        let mut crval = get_required_vec_f64(parameter_record, "crval")?;
581        let mut cdelt = get_required_vec_f64(parameter_record, "cdelt")?;
582        let mut crpix = get_required_vec_f64(parameter_record, "crpix")?;
583        if crval.len() != 1 || cdelt.len() != 1 || crpix.len() != 1 {
584            return Err(CoordinateError::InvalidRecord(
585                "spectral coordinate expects single-valued crval/cdelt/crpix".into(),
586            ));
587        }
588
589        let rest_frequency = get_required_f64(rec, "restfreq")?;
590        if !rest_frequency.is_finite() || rest_frequency < 0.0 {
591            return Err(CoordinateError::InvalidRecord(
592                "invalid rest frequency".into(),
593            ));
594        }
595        let axis_unit = get_optional_string(rec, "unit").or_else(|| {
596            get_optional_vec_string(parameter_record, "units")
597                .and_then(|units| units.into_iter().next())
598        });
599        let mut coord = if let Some((mut pixel_values, mut world_values)) = lookup_tables {
600            // Casacore's nested tabular coordinate is in Hz, while the
601            // surrounding spectral coordinate may display another unit.
602            let table_unit = get_optional_vec_string(parameter_record, "units")
603                .and_then(|units| units.into_iter().next())
604                .unwrap_or_else(|| "Hz".into());
605            let from = frequency_unit_scale(&table_unit).ok_or_else(|| {
606                CoordinateError::InvalidRecord(format!(
607                    "invalid tabular spectral unit {table_unit:?}"
608                ))
609            })?;
610            let to = frequency_unit_scale(axis_unit.as_deref().unwrap_or("Hz"))
611                .ok_or_else(|| CoordinateError::InvalidRecord("invalid spectral unit".into()))?;
612            let factor = from / to;
613            crval[0] *= factor;
614            cdelt[0] *= factor;
615            for value in &mut world_values {
616                *value *= factor;
617            }
618            if pixel_values.len() == 1 && world_values.len() == 1 {
619                crval[0] = world_values[0];
620                cdelt[0] = 0.0;
621                crpix[0] = 0.0;
622                pixel_values[0] = 0.0;
623            } else if pixel_values.len() >= 2 && world_values.len() == pixel_values.len() {
624                let last = pixel_values.len() - 1;
625                if pixel_values[last] == pixel_values[0] {
626                    return Err(CoordinateError::InvalidRecord(
627                        "tabular spectral pixel endpoints coincide".into(),
628                    ));
629                }
630                // Casacore rebuilds a spectral lookup with consecutive
631                // channel indices, then restores the table's reference pixel.
632                let original_reference_pixel = pixel_values[0];
633                crval[0] = world_values[0];
634                cdelt[0] = (world_values[last] - world_values[0]) / last as f64;
635                crpix[0] = original_reference_pixel;
636                for (index, (pixel, world)) in
637                    pixel_values.iter_mut().zip(&mut world_values).enumerate()
638                {
639                    *pixel = index as f64;
640                    *world -= cdelt[0] * original_reference_pixel;
641                }
642            }
643            Self::from_tabular(
644                frequency_ref,
645                pixel_values,
646                world_values,
647                crval[0],
648                cdelt[0],
649                crpix[0],
650                rest_frequency,
651            )?
652        } else {
653            let mut result = Self::new(frequency_ref, crval[0], cdelt[0], crpix[0], rest_frequency);
654            if let Some(Value::Record(wcs)) = rec.get("wcs")
655                && let Some(pc) = get_optional_f64(wcs, "pc")
656            {
657                result = result.with_linear_transform(pc)?;
658            }
659            result
660        };
661        if let Some(convention) = get_optional_i32(rec, "velType")
662            .or_else(|| get_optional_i32(rec, "prefVelType"))
663            .and_then(DopplerRef::from_casacore_code)
664        {
665            coord = coord.with_velocity_convention(convention);
666        }
667        if let Some(name) = get_optional_string(rec, "name").or_else(|| {
668            get_optional_vec_string(parameter_record, "axes")
669                .and_then(|names| names.into_iter().next())
670        }) {
671            coord = coord.with_name(name);
672        }
673        if let Some(unit) = axis_unit {
674            coord = coord.with_unit(unit);
675        }
676        if let Some(conversion) = parse_conversion(rec) {
677            coord = coord.with_conversion(conversion.frequency_ref, conversion.frame);
678        }
679        coord.undefined_frame = undefined_frame;
680        if let Some(code) = get_optional_i32(rec, "nativeType") {
681            coord.native_type = SpectralNativeType::from_code(code);
682        }
683        // C++ `restoreRestFrequencies` clamps the optional list and selects
684        // the listed frequency nearest the saved `restfreq` selector.
685        if let Some(field) = rec.get("restfreqs") {
686            if matches!(field, Value::Array(array) if array.shape().len() != 1) {
687                return Err(CoordinateError::InvalidRecord(
688                    "invalid rest frequency list".into(),
689                ));
690            }
691            let list = get_optional_vec_f64(rec, "restfreqs").ok_or_else(|| {
692                CoordinateError::InvalidRecord("invalid rest frequency list".into())
693            })?;
694            if list.is_empty() || list.iter().any(|value| !value.is_finite()) {
695                return Err(CoordinateError::InvalidRecord(
696                    "invalid rest frequency list".into(),
697                ));
698            }
699            coord.rest_frequencies = list.into_iter().map(|value| value.max(0.0)).collect();
700            if let Some(index) = closest_rest_frequency(&coord.rest_frequencies, rest_frequency) {
701                coord.rest_frequency_index = index;
702            }
703        }
704        Ok(coord)
705    }
706}

708fn parse_frequency_ref_name(name: &str) -> Option<FrequencyRef> {
709    if name == "LSR" {
710        // Older casacore image records used LSR for the kinematic frame.
711        Some(FrequencyRef::LSRK)
712    } else if name.eq_ignore_ascii_case("Undefined") {
713        // CASA `specmode='cubedata'` serializes a valid native-frequency
714        // axis with `system="Undefined"`; retain its established handling.
715        Some(FrequencyRef::TOPO)
716    } else {
717        FrequencyRef::from_str(name).ok()
718    }
719}

721impl Coordinate for SpectralCoordinate {
722    fn coordinate_type(&self) -> CoordinateType {
723        CoordinateType::Spectral
724    }
725
726    fn n_pixel_axes(&self) -> usize {
727        1
728    }
729
730    fn n_world_axes(&self) -> usize {
731        1
732    }
733
734    fn to_world(&self, pixel: &[f64]) -> Result<Vec<f64>, CoordinateError> {
735        if pixel.len() != 1 {
736            return Err(CoordinateError::DimensionMismatch {
737                expected: 1,
738                got: pixel.len(),
739            });
740        }
741        Ok(vec![self.world_at_pixel(pixel[0])?])
742    }
743
744    fn to_pixel(&self, world: &[f64]) -> Result<Vec<f64>, CoordinateError> {
745        if world.len() != 1 {
746            return Err(CoordinateError::DimensionMismatch {
747                expected: 1,
748                got: world.len(),
749            });
750        }
751        let native_world = self.convert_world_to_native(world[0])?;
752        let pixel = match &self.mapping {
753            SpectralMapping::Linear {
754                crval,
755                cdelt,
756                crpix,
757            } => {
758                let increment = cdelt * self.linear_transform;
759                if increment.abs() < 1e-300 {
760                    return Err(CoordinateError::ConversionFailed(
761                        "zero spectral increment".into(),
762                    ));
763                }
764                crpix + (native_world - crval) / increment
765            }
766            SpectralMapping::Tabular {
767                pixel_values,
768                world_values,
769                crval,
770                ..
771            } => inverse_sampled_axis(
772                pixel_values,
773                world_values,
774                *crval,
775                self.linear_transform,
776                native_world,
777            )?,
778        };
779        Ok(vec![pixel])
780    }
781
782    fn reference_value(&self) -> Vec<f64> {
783        vec![match &self.mapping {
784            SpectralMapping::Linear { crval, .. } | SpectralMapping::Tabular { crval, .. } => {
785                *crval
786            }
787        }]
788    }
789
790    fn reference_pixel(&self) -> Vec<f64> {
791        vec![match &self.mapping {
792            SpectralMapping::Linear { crpix, .. } | SpectralMapping::Tabular { crpix, .. } => {
793                *crpix
794            }
795        }]
796    }
797
798    fn increment(&self) -> Vec<f64> {
799        vec![match &self.mapping {
800            SpectralMapping::Linear { cdelt, .. } | SpectralMapping::Tabular { cdelt, .. } => {
801                *cdelt
802            }
803        }]
804    }
805
806    fn axis_names(&self) -> Vec<String> {
807        vec![self.name.clone()]
808    }
809
810    fn axis_units(&self) -> Vec<String> {
811        vec![self.unit.clone()]
812    }
813}

815impl SpectralCoordinate {
816    /// The record's `system`: the frame, or `"Undefined"` for an axis that
817    /// declares none.
818    fn system_name(&self) -> &'static str {
819        if self.undefined_frame {
820            "Undefined"
821        } else {
822            self.frequency_ref.as_str()
823        }

825
826    pub(crate) fn to_record(&self) -> RecordValue {
827        let mut rec = RecordValue::default();
828        rec.upsert("version", Value::Scalar(ScalarValue::Int32(2)));
829        rec.upsert(
830            "system",
831            Value::Scalar(ScalarValue::String(self.system_name().into())),
832        );
833        rec.upsert(
834            "restfreq",
835            Value::Scalar(ScalarValue::Float64(self.rest_frequency())),
836        );
837        rec.upsert(
838            "restfreqs",
839            Value::Array(ArrayValue::from_f64_vec(self.rest_frequencies.clone())),
840        );
841        rec.upsert(
842            "velType",
843            Value::Scalar(ScalarValue::Int32(self.velocity_convention.casacore_code())),
844        );
845        rec.upsert(
846            "nativeType",
847            Value::Scalar(ScalarValue::Int32(self.native_type as i32)),
848        );
849        rec.upsert("velUnit", Value::Scalar(ScalarValue::String("km/s".into())));
850        rec.upsert("waveUnit", Value::Scalar(ScalarValue::String("mm".into())));
851        rec.upsert(
852            "formatUnit",
853            Value::Scalar(ScalarValue::String(String::new())),
854        );
855        rec.upsert(
856            "unit",
857            Value::Scalar(ScalarValue::String(self.unit.clone())),
858        );
859        rec.upsert(
860            "name",
861            Value::Scalar(ScalarValue::String(self.name.clone())),
862        );
863
864        match &self.mapping {
865            SpectralMapping::Linear {
866                crval,
867                cdelt,
868                crpix,
869            } => {
870                let mut wcs = RecordValue::default();
871                wcs.upsert("crval", Value::Scalar(ScalarValue::Float64(*crval)));
872                wcs.upsert("crpix", Value::Scalar(ScalarValue::Float64(*crpix)));
873                wcs.upsert("cdelt", Value::Scalar(ScalarValue::Float64(*cdelt)));
874                wcs.upsert(
875                    "pc",
876                    Value::Scalar(ScalarValue::Float64(self.linear_transform)),
877                );
878                wcs.upsert("ctype", Value::Scalar(ScalarValue::String("FREQ".into())));
879                rec.upsert("wcs", Value::Record(wcs));
880            }
881            SpectralMapping::Tabular {
882                pixel_values,
883                world_values,
884                crval,
885                cdelt,
886                crpix,
887            } => {
888                let to_hz = self.unit_scale_hz.unwrap_or(1.0);
889                let mut tabular = RecordValue::default();
890                tabular.upsert(
891                    "crval",
892                    Value::Array(ArrayValue::from_f64_vec(vec![*crval * to_hz])),
893                );
894                tabular.upsert(
895                    "cdelt",
896                    Value::Array(ArrayValue::from_f64_vec(vec![*cdelt * to_hz])),
897                );
898                tabular.upsert(
899                    "crpix",
900                    Value::Array(ArrayValue::from_f64_vec(vec![*crpix])),
901                );
902                tabular.upsert(
903                    "pc",
904                    Value::Array(ArrayValue::Float64(
905                        ndarray::ArrayD::from_shape_vec(
906                            ndarray::IxDyn(&[1, 1]),
907                            vec![self.linear_transform],
908                        )
909                        .expect("1x1 spectral tabular PC matrix"),
910                    )),
911                );
912                tabular.upsert(
913                    "axes",
914                    Value::Array(ArrayValue::from_string_vec(vec!["Frequency".into()])),
915                );
916                tabular.upsert(
917                    "units",
918                    Value::Array(ArrayValue::from_string_vec(vec!["Hz".into()])),
919                );
920                tabular.upsert(
921                    "pixelvalues",
922                    Value::Array(ArrayValue::from_f64_vec(pixel_values.clone())),
923                );
924                tabular.upsert(
925                    "worldvalues",
926                    Value::Array(ArrayValue::from_f64_vec(
927                        world_values
928                            .iter()
929                            .map(|value| {
930                                let world = scaled_world(*value, *crval, self.linear_transform);
931                                world * to_hz
932                            })
933                            .collect(),
934                    )),
935                );
936                rec.upsert("tabular", Value::Record(tabular));
937            }
938        }
939
940        if let Some(conversion) = &self.conversion
941            && let (Some(direction), Some(position), Some(epoch)) = (
942                conversion.frame.direction(),
943                conversion.frame.position(),
944                conversion.frame.epoch(),
945            )
946        {
947            let mut conversion_record = RecordValue::default();
948            conversion_record.upsert(
949                "system",
950                Value::Scalar(ScalarValue::String(
951                    conversion.frequency_ref.as_str().into(),
952                )),
953            );
954            conversion_record.upsert("direction", Value::Record(direction_to_record(direction)));
955            conversion_record.upsert("position", Value::Record(position_to_record(position)));
956            conversion_record.upsert("epoch", Value::Record(epoch_to_record(epoch)));
957            rec.upsert("conversion", Value::Record(conversion_record));
958        }
959
960        rec
961    }
962}

964impl SpectralCoordinate {
965    fn convert_native_to_world(&self, native: f64) -> Result<f64, CoordinateError> {
966        let Some(conversion) = &self.conversion else {
967            return Ok(native);
968        };
969        let native_hz = self.value_in_hz(native)?;
970        let converted_hz = MFrequency::new(native_hz, self.frequency_ref)
971            .convert_to(conversion.frequency_ref, &conversion.frame)
972            .map(|frequency| frequency.hz())
973            .map_err(|err| {
974                CoordinateError::ConversionFailed(format!(
975                    "spectral frame conversion {} -> {} failed: {err}",
976                    self.frequency_ref, conversion.frequency_ref
977                ))
978            })?;
979        self.value_from_hz(converted_hz)
980    }
981
982    fn convert_world_to_native(&self, world: f64) -> Result<f64, CoordinateError> {
983        let Some(conversion) = &self.conversion else {
984            return Ok(world);
985        };
986        let world_hz = self.value_in_hz(world)?;
987        let converted_hz = MFrequency::new(world_hz, conversion.frequency_ref)
988            .convert_to(self.frequency_ref, &conversion.frame)
989            .map(|frequency| frequency.hz())
990            .map_err(|err| {
991                CoordinateError::ConversionFailed(format!(
992                    "spectral frame conversion {} -> {} failed: {err}",
993                    conversion.frequency_ref, self.frequency_ref
994                ))
995            })?;
996        self.value_from_hz(converted_hz)
997    }
998}

1000fn parse_conversion(rec: &RecordValue) -> Option<SpectralConversion> {
1001    let Value::Record(conversion) = rec.get("conversion")? else {
1002        return None;
1003    };
1004    let target = FrequencyRef::from_str(&get_optional_string(conversion, "system")?).ok()?;
1005    // An undefined conversion frame is a declaration, not a physical
1006    // conversion. It was ignored before Undefined became a FrequencyRef;
1007    // preserve the native coordinate rather than routing TOPO to Undefined.
1008    if target == FrequencyRef::Undefined {
1009        return None;
1010    }
1011    let Value::Record(direction_rec) = conversion.get("direction")? else {
1012        return None;
1013    };
1014    let Value::Record(position_rec) = conversion.get("position")? else {
1015        return None;
1016    };
1017    let Value::Record(epoch_rec) = conversion.get("epoch")? else {
1018        return None;
1019    };
1020    let direction = direction_from_record(direction_rec).ok()?;
1021    let position = position_from_record(position_rec).ok()?;
1022    let epoch = epoch_from_record(epoch_rec).ok()?;
1023    Some(SpectralConversion {
1024        frequency_ref: target,
1025        frame: MeasFrame::new()
1026            .with_direction(direction)
1027            .with_position(position)
1028            .with_epoch(epoch),
1029    })
1030}
