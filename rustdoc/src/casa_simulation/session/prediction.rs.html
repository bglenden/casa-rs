2//! `sm.predict`, `sm.setoptions` and `sm.setvp`: the model DATA is predicted
3//! from.
4//!
5//! As in CASA's `Simulator`:
6//! - `setoptions` and `setvp` only record; `predict` reads them.
7//! - `predict` reads the model image when it is called and predicts it into
8//!   the rows observed so far: replacing their DATA, or adding to it when
9//!   `incremental`. Scans observed later are not predicted by it.
10//! - The primary beam (`setvp(dovp=True)`) is the default voltage pattern of
11//!   the telescope of the first `observe` (OBSERVATION row 0), centred on
12//!   each field.
13//!
14//! The model is prepared when `predict` is called (read, multiplied by the
15//! beam and transformed, one set of grids per field when the beam is on), so
16//! a model changed or deleted afterwards predicts as in CASA and errors
17//! surface at the call. The grids are admitted against the session's memory
18//! budget and held until the step is replaced or the session closes; `close`
19//! degrids them into the rows it writes.
20//!
21//! Divergences: CASA rewrites the model image's spectral units and frame in
22//! place; casa-rs never writes it (`docs/CASA (C++) bugs.md`). MODEL_DATA
23//! and CORRECTED_DATA are not created (plan decision 6), even for an
24//! incremental predict, whose sum CASA forms through MODEL_DATA; a
25//! MeasurementSet `openfromms` opened keeps those it has in step with DATA.
26

75/// The arguments of `sm.predict`.
76#[derive(Debug, Clone, Default, PartialEq, Eq)]
77pub struct PredictSpec {
78    /// `imagename`: the model images; empty names are skipped, as CASA skips
79    /// them.
80    pub images: Vec<PathBuf>,
81    /// `complist`: a component list, or `None` for `''`.
82    pub component_list: Option<PathBuf>,
83    /// `incremental`: add the prediction to DATA rather than replace it.
84    pub incremental: bool,
85}
86
87impl PredictSpec {
88    /// The model images named, without the empty names.
89    fn named_images(&self) -> Vec<&Path> {
90        self.images
91            .iter()
92            .map(PathBuf::as_path)
93            .filter(|path| !path.as_os_str().is_empty())
94            .collect()
95    }
96
97    /// The check casatools' binding makes before `Simulator::predict`, even
98    /// before `open`: every named image exists and opens as an image
99    /// (`ImageFactory::fromFile`). Here it must open as the simulator reads
100    /// it, a CASA image of `Float` pixels: CASA's binding also accepts FITS
101    /// and complex images, which its simulator then refuses with `False`
102    /// (`docs/CASA (C++) bugs.md`).
103    ///
104    /// # Errors
105    ///
106    /// Raised: "File … does not exist." or "Unable to open image …".
107    pub fn check_images(&self) -> SimulatorResult<()> {
108        for path in self.named_images() {
109            if !path.exists() {
110                return Err(SimulatorError::raised(format!(
111                    "File {} does not exist.",
112                    path.display()
113                )));
114            }
115            if SkyModel::open(path).is_err_and(|error| matches!(error, PredictError::Image { .. }))
116            {
117                return Err(SimulatorError::raised(format!(
118                    "Unable to open image {}.",
119                    path.display()
120                )));
121            }
122        }
123        Ok(())
124    }
125}

127/// The arguments of `sm.setoptions` that casa-rs reads. `cache`, `tile`,
128/// `location`, `facets` and `maxdata` do not change what `GridFT` and
129/// `WProjectFT` predict; the session's memory budget stands for `cache`.
130#[derive(Debug, Clone, PartialEq)]
131pub struct PredictionOptions {
132    /// `ftmachine`: `WProjectFT` for `wproject`, `SDGrid` for `sd`, and
133    /// `GridFT` unless `both`, `mosaic` or `pbwproject` (case-insensitive).
134    pub ftmachine: String,
135    /// `gridfunction`: `SF`, the spheroid, for `GridFT` and `WProjectFT`;
136    /// `PB`, `BOX` or `SF` for `SDGrid`.
137    pub grid_function: String,
138    /// `padding`, `GridFT`'s; factors below 1 are ignored, as CASA ignores
139    /// them. `WProjectFT` pads by 1 whatever it is.
140    pub padding: f64,
141    /// `wprojplanes`, `WProjectFT`'s planes.
142    pub w_planes: i64,
143}

191/// A predict step's prepared model: the operator and grids `close` degrids
192/// the covered scans' rows from.
193pub(crate) struct Prediction {
194    pub(crate) model: ModelOperator,
195    /// The model's tangent point (pixel `(nx/2, ny/2)`), J2000 radians.
196    pub(crate) tangent_rad: [f64; 2],
197    /// The grids: one set for every field without a beam, else one set per
198    /// field, the beam centred on it.
199    grids: Vec<(Option<usize>, PreparedModelGrids)>,
200    /// The mosaic machine's pointing offsets, from the model's pixel
201    /// `(nx/2, ny/2)`, by phase centre (its bits), sorted by the bits; empty
202    /// for the others.
203    pointings: Vec<([u64; 2], [f64; 2])>,
204    /// The mosaic machine's dish class of each antenna; empty for the
205    /// others.
206    antenna_types: Vec<u8>,
207    /// The grids' memory, admitted at `predict` and released with them.
208    _reservation: Reservation,
209    /// What the operator's kernels keep, and the mosaic machine's pointings
210    /// and dish classes, admitted as they were built and released with
211    /// them.
212    _kernels: Reservation,
213}

232impl Prediction {
233    /// The grids rows of `field` are predicted from.
234    pub(crate) fn grids_for(&self, field: usize) -> Option<&PreparedModelGrids> {
235        self.grids
236            .iter()
237            .find(|(grid_field, _)| grid_field.is_none_or(|grid_field| grid_field == field))
238            .map(|(_, grids)| grids)
239    }
240
241    /// Bytes every set of grids holds.
242    pub(crate) fn grid_bytes(&self) -> u64 {
243        self.grids.iter().map(|(_, grids)| grids.bytes()).sum()
244    }
245
246    /// The offset of rows pointed at `centre_rad`, which the mosaic
247    /// machine's kernels ramp by; zero for the other machines.
248    pub(crate) fn pointing_offset_rad(&self, centre_rad: [f64; 2]) -> [f64; 2] {
249        let key = centre_rad.map(f64::to_bits);
250        let found = self
251            .pointings
252            .binary_search_by_key(&key, |(known, _)| *known)
253            .ok()
254            .map(|index| &self.pointings[index]);
255        // A mosaic step measured every scan it covers.
256        debug_assert!(
257            found.is_some() || self.pointings.is_empty(),
258            "a mosaic step predicts rows pointed at {centre_rad:?}, which it did not measure"
259        );
260        found.map_or([0.0; 2], |(_, offset)| *offset)
261    }
262
263    /// Each antenna's dish class, which keys the mosaic machine's kernels;
264    /// empty for the other machines.
265    pub(crate) fn antenna_types(&self) -> &[u8] {
266        &self.antenna_types
267    }
268}

270impl Simulator {
271    /// `sm.setoptions(...)`: record the FTMachine `predict` degrids through,
272    /// `GridFT`'s padding and `WProjectFT`'s planes. Each call sets the
273    /// FTMachine anew, as CASA's does.
274    ///
275    /// # Errors
276    ///
277    /// Not yet implemented: an FTMachine other than `GridFT`, `WProjectFT`,
278    /// `MosaicFTNew` and `SDGrid` (`both`, `pbwproject`), a grid function
279    /// other than `SF` for the interferometric machines, or other than `PB`,
280    /// `BOX` and `SF` for `SDGrid`. Nothing is recorded then.
281    pub fn set_options(&mut self, options: &PredictionOptions) -> SimulatorResult<()> {
282        let ftmachine = options.ftmachine.to_lowercase();
283        if OTHER_FTMACHINES.contains(&ftmachine.as_str()) {
284            return Err(SimulatorError::not_yet_implemented(format!(
285                "setoptions(ftmachine='{ftmachine}') is not yet implemented in casa-rs (#841)"
286            )));
287        }
288        let single_dish = if ftmachine == "sd" {
289            Some(
290                SingleDishKernel::from_grid_function(&options.grid_function).ok_or_else(|| {
291                    SimulatorError::not_yet_implemented(format!(
292                        "setoptions(ftmachine='sd', gridfunction='{}') is not yet implemented \
293                         in casa-rs (#841); 'PB', 'BOX' and 'SF' are",
294                        options.grid_function
295                    ))
296                })?,
297            )
298        } else {
299            if options.grid_function.to_lowercase() != "sf" {
300                return Err(SimulatorError::not_yet_implemented(format!(
301                    "setoptions(gridfunction='{}') is not yet implemented in casa-rs (#841); \
302                     'SF' is",
303                    options.grid_function
304                )));
305            }
306            None
307        };
308        tracing::info!("Setting processing options");
309        // CASA keeps a padding of at least 1 and ignores the rest.
310        let padding = options.padding as f32;
311        if padding >= 1.0 {
312            self.padding = PaddingFactor::new(padding)
313                .map_err(|error| SimulatorError::raised(format!("padding {padding}: {error}")))?;
314        }
315        self.w_projection_planes = (ftmachine == "wproject").then_some(options.w_planes);
316        self.mosaic = ftmachine == "mosaic";
317        self.single_dish = single_dish;
318        Ok(())
319    }

321    /// The FTMachine `predict` degrids through, as `setoptions` chose it.
322    ///
323    /// # Errors
324    ///
325    /// - CASA's `False`: `WProjectFT` with one plane, whose convolution
326    ///   function CASA's `WPConvFunc` cannot normalise.
327    /// - Raised: `WProjectFT` with no planes or fewer, where CASA's
328    ///   simulator crashes (`docs/CASA (C++) bugs.md`), or more than
329    ///   casa-rs's [`MAX_W_PLANES`] (65,536).
330    /// - Not yet implemented: `WProjectFT` on the Metal backend.
331    fn ft_machine(&self) -> SimulatorResult<FtMachine> {
332        if self.mosaic {
333            return Ok(FtMachine::Mosaic);
334        }
335        let Some(planes) = self.w_projection_planes else {
336            return Ok(FtMachine::Grid(self.padding));
337        };
338        match planes {
339            ..=0 => Err(SimulatorError::raised(format!(
340                "setoptions(ftmachine='wproject', wprojplanes={planes}): W projection needs at \
341                 least two planes"
342            ))),
343            1 => Err(SimulatorError::refused(
344                "Caught exception: Convolution function integral is not positive",
345            )),
346            _ => {
347                let planes = u32::try_from(planes)
348                    .ok()
349                    .filter(|&planes| planes as usize <= MAX_W_PLANES)
350                    .ok_or_else(|| {
351                        SimulatorError::raised(format!(
352                            "setoptions(ftmachine='wproject', wprojplanes={planes}): casa-rs \
353                             builds at most 65,536 W planes"
354                        ))
355                    })?;
356                if self.options.resources.backend() == BackendRequest::Metal {
357                    return Err(SimulatorError::not_yet_implemented(
358                        "predict with setoptions(ftmachine='wproject') on the Metal backend is \
359                         not yet implemented in casa-rs (#841); backend 'auto' or 'cpu' \
360                         predicts it on the CPU",
361                    ));
362                }
363                Ok(FtMachine::WProject(planes))
364            }
365        }
366    }

368    /// `sm.setvp(...)`: record whether `predict` applies the primary beam.
369    ///
370    /// # Errors
371    ///
372    /// Not yet implemented: a voltage-pattern table (`vptable` without
373    /// `usedefaultvp`). Nothing is recorded then.
374    pub fn set_vp(&mut self, spec: &VoltagePatternSpec) -> SimulatorResult<()> {
375        if !spec.use_default_vp && !spec.vp_table.is_empty() {
376            return Err(SimulatorError::not_yet_implemented(format!(
377                "setvp(vptable='{}') is not yet implemented in casa-rs (#841); the default \
378                 voltage patterns are",
379                spec.vp_table
380            )));
381        }
382        tracing::info!("Setting voltage pattern parameters");
383        if spec.use_default_vp {
384            tracing::info!("Using system default voltage patterns for each telescope");
385        }
386        if spec.do_squint {
387            tracing::info!("Beam Squint will be included in the VP model");
388            tracing::info!(
389                "and the parallactic angle increment is {} degrees",
390                spec.parallactic_angle_increment_deg
391            );
392        }
393        self.voltage_pattern = spec.dovp;
394        self.squint = spec.do_squint;
395        Ok(())
396    }

398    /// `sm.predict(imagename, complist, incremental)`: predict the model
399    /// into the rows observed so far, replacing their DATA or, when
400    /// `incremental`, adding to it.
401    ///
402    /// # Errors
403    ///
404    /// - Raised, as casatools' binding raises before predicting: a named
405    ///   image that does not exist or does not open
406    ///   ([`PredictSpec::check_images`]).
407    /// - Not yet implemented: more than one image, a model layout CASA reads
408    ///   and casa-rs does not (other than Stokes I, J2000,
409    ///   direction–Stokes–spectral axes), a component casa-rs does not
410    ///   predict (a direction not in J2000, or a flux, spectrum or shape
411    ///   casa-components does not evaluate), a near-field distance on the
412    ///   last field, scans whose windows differ in correlations, or
413    ///   `setvp(dovp=True)` for a telescope whose default pattern casa-rs
414    ///   does not have (ALMA's, ACA's, the VLA's and the EVLA's it has) or
415    ///   for pointings `observemany` moved off their fields.
416    /// - CASA's `False`: no model image ("No model images found"), or no
417    ///   rows observed yet.
418    /// - Raised: a scan of 3 correlations ("No such VisVector type"), or
419    ///   model grids or kernels, or a component list's read or preparation,
420    ///   that do not fit the session's memory budget
421    ///   ([`SimulatorError::admission_refused`]).
422    /// - As the FTMachine `setoptions` chose refuses: `WProjectFT` with
423    ///   fewer than two planes, or on the Metal backend.
424    pub fn predict(&mut self, spec: &PredictSpec) -> SimulatorResult<()> {
425        spec.check_images()?;
426        let images = spec.named_images();
427        tracing::info!(
428            "Predicting visibilities using model: {images:?} and componentList: {}",
429            spec.component_list
430                .as_deref()
431                .map_or(String::new(), |path| path.display().to_string())
432        );
433        if spec.incremental {
434            tracing::info!("The data column will be incremented");
435        } else {
436            tracing::info!("The data column will be replaced");
437        }
438        if images.len() > 1 {
439            return Err(SimulatorError::not_yet_implemented(format!(
440                "predict from {} model images is not yet implemented in casa-rs (simulation \
441                 Phase 2); one image is",
442                images.len()
443            )));
444        }
445        let list = spec
446            .component_list
447            .as_deref()
448            .filter(|path| !path.as_os_str().is_empty());
449        if images.is_empty() && list.is_none() {
450            return Err(SimulatorError::refused(
451                "Failed to create SkyEquation: No model images found",
452            ));
453        }
454        if let Some(field) = self.fields.last()
455            && field.distance_m != 0.0
456        {
457            return Err(SimulatorError::not_yet_implemented(format!(
458                "predict refocused to the distance {} m of field {} is not yet implemented in \
459                 casa-rs (#841)",
460                field.distance_m, field.name
461            )));
462        }
463        // `createSkyEquation` reads the component list before the images.
464        let list = list
465            .map(|path| read_component_list(path, &self.options.resources))
466            .transpose()?;
467        let model = images
468            .first()
469            .map(|path| SkyModel::open(path).map_err(model_error))
470            .transpose()?;
471        if self.planned_rows() == 0 {
472            return Err(SimulatorError::refused(
473                "Failed to create SkyEquation: Caught exception: MSIter::construct - Input \
474                 MeasurementSet.has zero selected rows",
475            ));
476        }
477        let correlations = self.predicted_correlations()?;
478        let beam = self.voltage_pattern_beam()?;
479        // Each of these points a scan at its phase centre, which on an
480        // opened MeasurementSet POINTING must agree with.
481        if self.mosaic && model.is_some() {
482            self.require_pointings_at_phase_centres("setoptions(ftmachine='mosaic')")?;
483        }
484        if beam.is_some() {
485            self.require_pointings_at_phase_centres("setvp(dovp=True)")?;
486        }
487        if self.single_dish.is_some() {
488            self.require_pointings_at_phase_centres("setoptions(ftmachine='sd')")?;
489        }
490        // A primary beam, over a model image, a component list or total
491        // power, and the mosaic machine's kernels are centred on each
492        // scan's field here, which a pointing `observemany` moved does not
493        // point at (#884).
494        if model.is_some() && self.mosaic {
495            self.check_pointings_on_fields("setoptions(ftmachine='mosaic')")?;
496        }
497        if beam.is_some() {
498            self.check_pointings_on_fields("setvp(dovp=True)")?;
499        }
500        if let Some(kernel) = self.single_dish {
501            // Total power samples a model image; `SDGrid` with a component
502            // list is not yet here.
503            let Some(model) = model.filter(|_| list.is_none()) else {
504                return Err(SimulatorError::not_yet_implemented(
505                    "predict(complist=…) after setoptions(ftmachine='sd') is not yet \
506                     implemented in casa-rs (#841); a model image is",
507                ));
508            };
509            return self.predict_single_dish(model, kernel, beam, spec.incremental);
510        }
511        let components = list
512            .map(|(path, list, held)| self.prepare_components(&path, (list, held), beam))
513            .transpose()?;
514        let prediction = model
515            .map(|model| self.prepare_image(&model, &correlations, beam))
516            .transpose()?;
517        if !spec.incremental {
518            // Every step so far covers a prefix of the scans this one covers.
519            self.data_plan.clear();
520        }
521        let scans = self.scans.len();
522        if let Some(components) = components {
523            self.data_plan
524                .predict_components(Arc::new(components), scans);
525        }
526        if let Some(prediction) = prediction {
527            self.data_plan.predict(Arc::new(prediction), scans);
528        }
529        Ok(())
530    }

532    /// `model` admitted and prepared for rows with `correlations` through
533    /// the FTMachine `setoptions` chose: its kernels, then its grids, with
534    /// `beam` in the image plane except through the mosaic machine, whose
535    /// kernels carry the beams and which keeps each scan's pointing and
536    /// each antenna's dish class.
537    fn prepare_image(
538        &self,
539        model: &SkyModel,
540        correlations: &[CorrelationType],
541        beam: Option<PrimaryBeam>,
542    ) -> SimulatorResult<Prediction> {
543        let machine = self.ft_machine()?;
544        let mut mosaic_state = None;
545        let (beams, antenna_types, pointings) = match machine {
546            FtMachine::Mosaic => {
547                // What the machine keeps per pointing and antenna, and its
548                // beams' inputs until the kernels are built, admitted before
549                // they are formed.
550                let (kept, building) = self.mosaic_state_bytes();
551                let resources = &self.options.resources;
552                let reservation = admit(
553                    resources.host(),
554                    resources.policy(),
555                    &Demand {
556                        phase: MOSAIC_STATE_PHASE,
557                        memory: kept + building,
558                    },
559                )
560                .map_err(SimulatorError::admission)?;
561                mosaic_state = Some((reservation, kept));
562                let (beams, antenna_types) = self.mosaic_beams()?;
563                (Some(beams), antenna_types, self.mosaic_pointings(model)?)
564            }
565            FtMachine::Grid(_) | FtMachine::WProject(_) => {
566                if self
567                    .antennas
568                    .windows(2)
569                    .any(|pair| pair[0].diameter_m != pair[1].diameter_m)
570                {
571                    tracing::warn!(
572                        "Heterogeneous array is only supported for ALMA,ACA,OVRO telescopes, \
573                         and only with option ftmachine=mosaic."
574                    );
575                }
576                (None, Vec::new(), Vec::new())
577            }
578        };
579        let layout = ModelLayout::new(model, correlations, machine, beams)
580            .map_err(|error| correlation_error(error, correlations))?;
581        let (kernels, mut kernel_reservation) = self.admitted_kernels(&layout)?;
582        // The operator drops the layout's beams.
583        let operator = ModelOperator::new(layout, kernels, GridPrecision::F32);
584        if let Some((mut reservation, kept)) = mosaic_state {
585            debug_assert_eq!(
586                (pointings.capacity() * size_of::<([u64; 2], [f64; 2])>()
587                    + antenna_types.capacity()) as u64,
588                kept,
589                "the mosaic state keeps what was admitted for it"
590            );
591            reservation.retain(kept);
592            kernel_reservation.join(reservation);
593        }
594        let image_beam = beam.filter(|_| machine != FtMachine::Mosaic);
595        let mut prediction = self.prepare(model, operator, image_beam, kernel_reservation)?;
596        prediction.pointings = pointings;
597        prediction.antenna_types = antenna_types;
598        Ok(prediction)
599    }

601    /// `list` prepared for the session's windows, with `beam` per observed
602    /// field, and admitted. The admission is made when the list has been
603    /// read: what the prediction will hold, known from the list's length,
604    /// and its preparation's scratch. The list's own share of the read's
605    /// admission, `held_list`, is released when the list is dropped here.
606    fn prepare_components(
607        &self,
608        path: &Path,
609        (list, held_list): (ComponentList, Reservation),
610        beam: Option<PrimaryBeam>,
611    ) -> SimulatorResult<ComponentPrediction> {
612        // CASA squints the VLA's and EVLA's beams for a component list (RR
613        // and LL beams apart); casa-rs's patterns are unsquinted.
614        if self.squint
615            && let Some(beam) = beam.filter(|beam| beam.squints())
616        {
617            return Err(SimulatorError::not_yet_implemented(format!(
618                "predict(complist='{}') under the {beam:?} primary beam with beam squint is \
619                 not yet implemented in casa-rs (#841): CASA squints the RR and LL beams for \
620                 components; setvp(dosquint=False) predicts them unsquinted",
621                path.display()
622            )));
623        }
624        if self
625            .antennas
626            .windows(2)
627            .any(|pair| pair[0].diameter_m != pair[1].diameter_m)
628        {
629            tracing::warn!(
630                "Heterogeneous array is not supported for simulation from components.  \
631                 Primary beam will be applied by telescope name."
632            );
633        }
634        let windows: Vec<ComponentWindow> = self
635            .spectral_windows
636            .iter()
637            .map(|window| ComponentWindow {
638                first_hz: window.start_frequency_hz,
639                step_hz: window.channel_width_hz,
640                channels: window.channel_count,
641                correlations: window.correlations.clone(),
642            })
643            .collect();
644        let pattern = beam.map(PrimaryBeam::pattern);
645        // A scan is phased on its field's phase centre, as CASA's
646        // `VisBuffer` phases it, and the beam centred there: a beam on a
647        // pointing `observemany` moved was refused above (#884). Scans at
648        // one centre share a pointing.
649        let mut pointings: Vec<(usize, [f64; 2])> = Vec::new();
650        let pointing_of_scan: Vec<usize> = self
651            .scans
652            .iter()
653            .map(|scan| {
654                let centre = scan.phase_center_rad.map(f64::to_bits);
655                pointings
656                    .iter()
657                    .position(|(_, known)| known.map(f64::to_bits) == centre)
658                    .unwrap_or_else(|| {
659                        pointings.push((pointings.len(), scan.phase_center_rad));
660                        pointings.len() - 1
661                    })
662            })
663            .collect();
664        let targets = ComponentTargets {
665            windows: &windows,
666            pointings: &pointings,
667            pointing_of_scan: &pointing_of_scan,
668            beam: pattern.as_ref(),
669        };
670        let resources = &self.options.resources;
671        let preparation = ComponentTargets::preparation_bytes(list.len());
672        let mut reservation = admit(
673            resources.host(),
674            resources.policy(),
675            &Demand {
676                phase: COMPONENTS_PHASE,
677                memory: targets.bytes(list.len()) + preparation,
678            },
679        )
680        .map_err(SimulatorError::admission)?;
681        let held_preparation = reservation.split(preparation);
682        let prediction = ComponentPrediction::new(&list, targets, reservation).map_err(
683            |ComponentRefusal::NotYetImplemented(reason)| {
684                SimulatorError::not_yet_implemented(format!(
685                    "predict(complist='{}'): {reason} (#841)",
686                    path.display()
687                ))
688            },
689        )?;
690        drop(held_preparation);
691        tracing::info!(
692            components = list.len(),
693            bytes = prediction.bytes(),
694            "prepared the component list {}",
695            path.display()
696        );
697        drop(list);
698        drop(held_list);
699        Ok(prediction)
700    }

702    /// The correlations of the scans observed so far, which one operator
703    /// predicts.
704    fn predicted_correlations(&self) -> SimulatorResult<Vec<CorrelationType>> {
705        self.check_visibility_correlations("predicted")?;
706        let window = |scan: &super::observe::Scan| &self.spectral_windows[scan.spectral_window_id];
707        let first = &window(&self.scans[0]).correlations;
708        if self
709            .scans
710            .iter()
711            .any(|scan| window(scan).correlations != *first)
712        {
713            return Err(SimulatorError::not_yet_implemented(
714                "predicting scans whose spectral windows have different correlations is not yet \
715                 implemented in casa-rs (#841)",
716            ));
717        }
718        first
719            .iter()
720            .map(|&code| {
721                CorrelationType::from_casa_code(code).ok_or_else(|| {
722                    SimulatorError::not_yet_implemented(format!(
723                        "predicting correlation code {code} is not yet implemented in casa-rs \
724                         (#841)"
725                    ))
726                })
727            })
728            .collect()
729    }

732    /// row 0's telescope at spectral window 0's REF_FREQUENCY
733    /// (`VPSkyJones`).
734    fn voltage_pattern_beam(&self) -> SimulatorResult<Option<PrimaryBeam>> {
735        if !self.voltage_pattern {
736            return Ok(None);
737        }
738        let telescope = self.observing_telescope();
739        let beam = PrimaryBeam::for_telescope(telescope, self.reference_frequency_hz())

743                     casa-rs (#841); the default patterns of ALMA, ACA, VLA and EVLA are"
744                ))
745            })?;
746        Ok(Some(beam))
747    }

767    /// That every scan points at its field, as `what` needs here: a primary
768    /// beam, or the mosaic machine's kernels, centred on each scan's field.
769    /// CASA's beam on a pointing `observemany` moved off its field is
770    /// centred where the first row of the field it visits points, which
771    /// `BeamSkyJones` keeps for the field's other rows (#884, `docs/CASA
772    /// (C++) bugs.md`); casa-rs does not follow it yet.
773    fn check_pointings_on_fields(&self, what: &str) -> SimulatorResult<()> {
774        let moved = self
775            .scans
776            .iter()
777            .find(|scan| scan.pointing_rad != scan.phase_center_rad);
778        match moved {
779            Some(scan) => Err(SimulatorError::not_yet_implemented(format!(
780                "predict with {what} for scan {} pointed away from its field {} is not yet \
781                 implemented in casa-rs (simulation Phase 2, #884); pointings at their \
782                 fields' centres are",
783                scan.scan_number, self.fields[scan.field_id].name
784            ))),
785            None => Ok(()),
786        }
787    }

801    /// Build `layout`'s kernels, admitted as they are built, as imaging
802    /// admits them: the spheroidal set's bytes, or the W-projection screens
803    /// ([`WPlanes::screens_bytes`]) and then the kernels cropped from them
804    /// ([`WScreens::kernel_bytes`](casa_imaging_operator::WScreens::kernel_bytes)),
805    /// the screens released once the kernels are built. The reservation
806    /// returned keeps what the kernels hold.
807    fn admitted_kernels(
808        &self,
809        layout: &ModelLayout,
810    ) -> SimulatorResult<(Box<dyn ConvolutionFunctionSet>, Reservation)> {
811        let resources = &self.options.resources;
812        let (host, policy) = (resources.host(), resources.policy());
813        charge_process_caches(host, policy).map_err(SimulatorError::admission)?;
814        let admitted = |phase, memory| {
815            admit(host, policy, &Demand { phase, memory }).map_err(SimulatorError::admission)
816        };
817        let (geometry, polarization) = (layout.geometry(), layout.polarization());
818        let raised =
819            |error: OperatorError| SimulatorError::raised(format!("Caught exception: {error}"));
820        Ok(match layout.machine() {
821            FtMachine::Grid(_) => (
822                Box::new(Spheroidal::new(geometry, polarization)),
823                admitted(KERNEL_PHASE, Spheroidal::bytes(geometry))?,
824            ),
825            FtMachine::WProject(planes) => {
826                let count = WPlaneCount::Fixed(planes);
827                let screening = admitted(
828                    W_SCREENS_PHASE,
829                    WPlanes::screens_bytes(geometry, count).map_err(raised)?,
830                )?;
831                let screens = WPlanes::screens(geometry, polarization, count).map_err(raised)?;
832                let mut kernels = admitted(KERNEL_PHASE, screens.kernel_bytes())?;
833                let set = screens.finish().map_err(raised)?;
834                drop(screening);
835                kernels.retain(set.resident_bytes());
836                tracing::info!(
837                    planes,
838                    screen = set.screen_size(),
839                    bytes = set.resident_bytes(),
840                    "W projection using {planes} planes"
841                );
842                (Box::new(set), kernels)
843            }
844            FtMachine::Mosaic => {
845                let beams = layout.beams().expect("a mosaic layout has its beams");
846                let (dishes, windows) = (&beams.dishes, &beams.windows);
847                let (rules, frequency_hz) = (MosaicRules::Simulator, layout.image_frequency_hz());
848                tracing::info!("Performing Mosaic gridding for model images (uv convolution)");
849                let screening = admitted(
850                    MOSAIC_SCREENS_PHASE,
851                    MosaicPb::screens_bytes(rules, geometry, frequency_hz, dishes, windows)
852                        .map_err(raised)?,
853                )?;
854                let screens =
855                    MosaicPb::screens(rules, geometry, polarization, frequency_hz, dishes, windows)
856                        .map_err(raised)?;
857                let mut kernels = admitted(KERNEL_PHASE, screens.kernel_bytes())?;
858                let set = screens.finish().map_err(raised)?;
859                drop(screening);
860                kernels.retain(set.resident_bytes());
861                tracing::info!(
862                    dishes = set.dishes(),
863                    screen = set.screen_size(),
864                    bytes = set.resident_bytes(),
865                    "convolution function support: {}",
866                    set.max_half_support()[0]
867                );
868                (Box::new(set), kernels)
869            }
870        })
871    }

873    /// Admit and prepare `model`'s grids: one set, or one per observed field
874    /// with `beam` centred on it. The planes are prepared on as many of the
875    /// policy's workers as fit their scratch beside the grids. The
876    /// prediction keeps `kernels`, the reservation of the operator's
877    /// kernels.
878    fn prepare(
879        &self,
880        model: &SkyModel,
881        operator: ModelOperator,
882        beam: Option<PrimaryBeam>,
883        kernels: Reservation,
884    ) -> SimulatorResult<Prediction> {
885        let resources = &self.options.resources;
886        let (host, policy) = (resources.host(), resources.policy());
887        charge_process_caches(host, policy).map_err(SimulatorError::admission)?;
888        let fields: Vec<Option<usize>> = match beam {
889            Some(_) => self.observed_fields().into_iter().map(Some).collect(),
890            None => vec![None],
891        };
892        let pattern_bytes = if beam.is_some() {
893            BeamPattern::BYTES
894        } else {
895            0
896        };
897        let reservation = admit(
898            host,
899            policy,
900            &Demand {
901                phase: MODEL_PHASE,
902                memory: operator.grid_bytes() * fields.len() as u64 + pattern_bytes,
903            },
904        )
905        .map_err(SimulatorError::admission)?;
906        let scratch = operator.plane_scratch_bytes(model);
907        let free = free_memory(host, policy);
908        let workers = (1..=resources.cores().workers.max(1))
909            .rev()
910            .find(|workers| *workers as u64 * scratch <= free)
911            .unwrap_or(1);
912        let _preparation = admit(
913            host,
914            policy,
915            &Demand {
916                phase: PREPARATION_PHASE,
917                memory: workers as u64 * scratch,
918            },
919        )
920        .map_err(SimulatorError::admission)?;
921        let team = WorkerTeam::new(workers)
922            .map_err(|error| SimulatorError::raised(format!("the worker team: {error}")))?;
923        let pattern = beam.map(PrimaryBeam::pattern);
924        let mut grids = Vec::with_capacity(fields.len());
925        for field in fields {
926            let centre = match (&pattern, field) {
927                (Some(pattern), Some(field)) => {
928                    let pointing = self.fields[field].direction_rad;
929                    Some((pattern, model.pixel_of(pointing).map_err(model_error)?))
930                }
931                _ => None,
932            };
933            let prepared = operator
934                .prepare(model, centre, Some(&team))
935                .map_err(model_error)?;
936            grids.push((field, prepared));
937        }
938        tracing::info!(
939            planes = model.planes(),
940            grids = grids.len(),
941            workers,
942            bytes = operator.grid_bytes() * grids.len() as u64,
943            "prepared the model {}",
944            model.path().display()
945        );
946        Ok(Prediction {
947            model: operator,
948            tangent_rad: model.centre_rad(),
949            grids,
950            pointings: Vec::new(),
951            antenna_types: Vec::new(),
952            _reservation: reservation,
953            _kernels: kernels,
954        })
955    }
956}

958/// The component list at `path`, as `createSkyEquation` reads it: CASA's
959/// `False` when the path is not a readable table ("ComponentList … not
960/// readable"), and when the table is not a component list CASA reads
961/// (`ComponentList`'s constructor throws, and `predict` catches it). The
962/// read is admitted within `resources` before it starts; once the list is
963/// formed, the list's share of that admission is returned with it and the
964/// rest released.
965fn read_component_list(
966    path: &Path,
967    resources: &SessionResources,
968) -> SimulatorResult<(PathBuf, ComponentList, Reservation)> {
969    if !path.join("table.dat").is_file() {
970        return Err(SimulatorError::refused(format!(
971            "Failed to create SkyEquation: ComponentList {} not readable",
972            path.display()
973        )));
974    }
975    let refused = |error| SimulatorError::refused(format!("Caught exception: {error}"));
976    let reading_bytes = ComponentList::read_table_bytes(path).map_err(refused)?;
977    let mut reading = admit(
978        resources.host(),
979        resources.policy(),
980        &Demand {
981            phase: COMPONENTS_PHASE,
982            memory: reading_bytes,
983        },
984    )
985    .map_err(SimulatorError::admission)?;
986    let list = ComponentList::read_table(path).map_err(refused)?;
987    let held = reading.split(list.bytes().min(reading_bytes));
988    Ok((path.to_path_buf(), list, held))
989}

991/// A model that cannot be read or prepared, as CASA reports it: a layout
992/// CASA reads and casa-rs does not is not yet implemented; the rest is
993/// raised.
994fn model_error(error: PredictError) -> SimulatorError {
995    match error {
996        PredictError::UnsupportedModel(reason) => SimulatorError::not_yet_implemented(format!(
997            "predict from this model is not yet implemented in casa-rs (#841): {reason}"
998        )),
999        PredictError::Image { path, .. } => {
1000            SimulatorError::raised(format!("Unable to open image {path}."))
1001        }
1002        other => SimulatorError::raised(format!("Caught exception: {other}")),
1003    }
1004}
