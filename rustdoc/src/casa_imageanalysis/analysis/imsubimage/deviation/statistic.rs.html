3use casa_lattices::statistics::RunningStats;
4use casa_lattices::{
5    RobustStats, SamplePoint, Statistic, StatisticsAlgorithm, compute_robust_stats,
6};
7
8use casa_images::ImageError;
9
10use super::invalid;
11
12pub(super) struct LocalStatistic {
13    kind: Statistic,
14    scale: f64,
15    algorithm: StatisticsAlgorithm,
16}
17
18pub(super) enum Samples {
19    Running(RunningStats),
20    Retained(Vec<SamplePoint>),
21}
22
23impl Samples {
24    pub(super) fn clear(&mut self) {
25        match self {
26            Self::Running(stats) => *stats = RunningStats::default(),
27            Self::Retained(points) => points.clear(),
28        }
29    }
30
31    pub(super) fn push(&mut self, value: f32) {
32        let value = f64::from(value);
33        match self {
34            Self::Running(stats) => stats.push(value),
35            Self::Retained(points) => points.push(SamplePoint {
36                value,
37                position: points.len() as u64,
38            }),
39        }
40    }
41}
42
43impl LocalStatistic {
44    pub(super) fn new(
45        name: &str,
46        algorithm: &str,
47        zscore: f64,
48        maxiter: i64,
49    ) -> Result<Self, ImageError> {
50        use Statistic as S;
51        let kind = name.to_ascii_lowercase();
52        let stat = [
53            ("i", S::Quartile),
54            ("max", S::Max),
55            ("mea", S::Mean),
56            ("meda", S::MedAbsDevMed),
57            ("mad", S::MedAbsDevMed),
58            ("x", S::MedAbsDevMed),
59            ("medi", S::Median),
60            ("mi", S::Min),
61            ("n", S::Npts),
62            ("q1", S::Q1),
63            ("q3", S::Q3),
64            ("r", S::Rms),
65            ("si", S::Sigma),
66            ("st", S::Sigma),
67            ("sums", S::SumSq),
68            ("sum", S::Sum),
69            ("v", S::Variance),
70        ]
71        .into_iter()
72        .find_map(|(prefix, stat)| kind.starts_with(prefix).then_some(stat))
73        .ok_or_else(|| invalid(format!("Statistic {name} not supported.")))?;
74        let normalized = algorithm.to_ascii_lowercase();
75        let algorithm = if normalized.starts_with("cl") {
76            StatisticsAlgorithm::Classic
77        } else if normalized.starts_with("ch") {
78            StatisticsAlgorithm::Chauvenet { zscore, maxiter }
79        } else {
80            return Err(invalid(format!("Unsupported stats algorithm {algorithm}")));
81        };
82        Ok(Self {
83            kind: stat,
84            scale: if kind.starts_with('x') {
85                1.482602218505602
86            } else {
87                1.
88            },
89            algorithm,
90        })
91    }
92
93    pub(super) fn samples(&self) -> Samples {
94        use Statistic as S;
95        match (self.algorithm, self.kind) {
96            (
97                StatisticsAlgorithm::Classic,
98                S::Median | S::Q1 | S::Q3 | S::Quartile | S::MedAbsDevMed,
99            )
100            | (StatisticsAlgorithm::Chauvenet { .. }, _) => Samples::Retained(Vec::new()),
101            _ => Samples::Running(RunningStats::default()),
102        }
103    }
104
105    pub(super) fn evaluate(&self, samples: &Samples) -> Result<Option<f32>, ImageError> {
106        use Statistic as S;
107        let points = match samples {
108            Samples::Running(stats) => {
109                return Ok(stats.statistic(self.kind).map(|value| value as f32));
110            }
111            Samples::Retained(points) => points,
112        };
113        if points.is_empty() {
114            return Ok(None);
115        }
116        if let &[point] = points.as_slice() {
117            return Ok(Some(self.single(point.value as f32)));
118        }
119        let RobustStats::Sample(stats) = compute_robust_stats(points, &self.algorithm)
120            .map_err(|error| invalid(error.to_string()))?
121        else {
122            unreachable!("the local image statistics algorithms return samples");
123        };
124        if stats.npts == 0. {
125            return match self.kind {
126                S::Min => Err(invalid(
127                    "Logic Error: stat min is not defined. Please file a defect report.",
128                )),
129                S::Max => Err(invalid(
130                    "Logic Error: stat max is not defined. Please file a defect report.",
131                )),
132                S::Median | S::MedAbsDevMed | S::Q1 | S::Q3 | S::Quartile => {
133                    Err(invalid("No valid data found"))
134                }
135                _ => Ok(Some(0.)),
136            };
137        }
138        let value = match self.kind {
139            S::Npts => stats.npts,
140            S::Min => stats.min,
141            S::Max => stats.max,
142            S::Sum => stats.sum,
143            S::SumSq => stats.sumsq,
144            S::Mean => stats.mean,
145            S::Rms => stats.rms,
146            S::Variance => stats.sigma * stats.sigma,
147            S::Sigma => stats.sigma,
148            S::Median => stats.median,
149            S::MedAbsDevMed => stats.medabsdevmed,
150            S::Q1 => stats.q1,
151            S::Q3 => stats.q3,
152            S::Quartile => stats.quartile,
153        };
154        Ok(Some((f64::from(value as f32) * self.scale) as f32))
155    }
156
157    pub(super) fn single(&self, value: f32) -> f32 {
158        use Statistic as S;
159        match self.kind {
160            S::Npts => 1.,
161            S::SumSq => value * value,
162            S::Rms => value.abs(),
163            S::Variance | S::Sigma | S::Quartile | S::MedAbsDevMed => 0.,
164            _ => value,
165        }
166    }
167}
