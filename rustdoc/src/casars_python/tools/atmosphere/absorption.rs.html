388impl PyAtmosphere {
389    pub(super) fn with_spectrum<T: Send>(
390        &self,
391        py: Python<'_>,
392        read: impl FnOnce(&AtmosphereSpectrum) -> Result<T, &'static str> + Send,
393    ) -> PyResult<T> {
394        self.run(py, |session| read(session.spectrum().ok_or(NO_WINDOW)?))
395            .map_err(error)
396    }
397
398    pub(super) fn channel_value(
399        &self,
400        py: Python<'_>,
401        nc: i64,
402        spwid: i64,
403        read: impl FnOnce(&AtmosphereSpectrum, usize) -> f64 + Send,
404    ) -> PyResult<f64> {
405        self.with_spectrum(py, |spectrum| {
406            let index = channel_index(spectrum, spwid, nc)?;
407            Ok(index.map_or(INVALID_CHANNEL, |channel| read(spectrum, channel)))
408        })
409    }
410
411    fn channel_quantity(
412        &self,
413        py: Python<'_>,
414        nc: i64,
415        spwid: i64,
416        unit: &str,
417        read: impl FnOnce(&AtmosphereSpectrum, usize) -> f64 + Send,
418    ) -> PyResult<Py<PyAny>> {
419        let value = self.channel_value(py, nc, spwid, read)?;
420        reply_value(py, &quantity::scalar(unit, value))
421    }
422
423    fn spectral_values(
424        &self,
425        py: Python<'_>,
426        spwid: i64,
427        read: impl Fn(&AtmosphereSpectrum, usize) -> f64 + Send,
428    ) -> PyResult<Vec<f64>> {
429        self.with_spectrum(py, move |spectrum| {
430            Ok(channel_range(spectrum, spwid)?
431                .map(|channel| read(spectrum, channel))
432                .collect())
433        })
434    }
435
436    fn layer_quantity(
437        &self,
438        py: Python<'_>,
439        nl: i64,
440        nf: i64,
441        spwid: i64,
442        read: impl FnOnce(&AtmosphereSpectrum, usize, usize) -> f64 + Send,
443    ) -> PyResult<Py<PyAny>> {
444        let layer = usize::try_from(nl).map_err(|_| error("Failed AlwaysAssert value>=0"))?;
445        let absorption = self.with_spectrum(py, |spectrum| {
446            let window = spectral_window(spectrum, spwid)?;
447            if nf < 0 || nf as usize >= window.num_channels() {
448                return Err("Exception: Channel ID out of range..");
449            }
450            let channel = spectrum
451                .absorption
452                .grid()
453                .flat_index(spwid as usize, nf as usize)
454                .expect("validated channel");
455            if layer >= spectrum.absorption.atmosphere().layers().len() {
456                return Err("Exception: Layer ID out of range..");
457            }
458            Ok(read(spectrum, channel, layer))
459        })?;
460        reply_value(py, &quantity::scalar("m-1", absorption))
461    }
462}
463
464pub(super) fn channel_index(
465    spectrum: &AtmosphereSpectrum,
466    spwid: i64,
467    channel: i64,
468) -> Result<Option<usize>, &'static str> {
469    let window = spectral_window(spectrum, spwid)?;
470    let index = usize::try_from(channel).unwrap_or_else(|_| window.reference_channel());
471    Ok(spectrum.absorption.grid().flat_index(spwid as usize, index))
472}
473
474pub(super) fn channel_range(
475    spectrum: &AtmosphereSpectrum,
476    spwid: i64,
477) -> Result<Range<usize>, &'static str> {
478    let window = spectral_window(spectrum, spwid)?;
479    let start = spectrum
480        .absorption
481        .grid()
482        .flat_index(spwid as usize, 0)
483        .expect("configured window has channels");
484    Ok(start..start + window.num_channels())
485}
