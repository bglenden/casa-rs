11const AVOGADRO: f64 = 6.022045e23;
12const AIR_MOLAR_MASS: f64 = 28.964;
13
14/// Minor-gas number densities in molecules/cm³.
15#[derive(Clone, Copy, Debug, Default, PartialEq)]
16pub(super) struct MinorGases {
17    pub(super) o3: f64,
18    pub(super) n2o: f64,
19    pub(super) co: f64,
20    pub(super) no2: f64,
21    pub(super) so2: f64,
22    pub(super) hcl: f64,
23    pub(super) hcn: f64,
24}

26/// The minor gases at `height_km` above sea level for reference profile
27/// `reference` (1 to 6) and seasonal/diurnal ozone model `ozone_model`
28/// (1 to 4): ATM's `st76`.
29///
30/// Each profile is interpolated by the parabola through the three tabulated
31/// levels around the height. The total air density converts the mixing
32/// ratios (ppmv) to number densities. ATM then scales ozone by layer for its
33/// Chajnantor models, N2O above 30 km, and HCl and HCN above fixed levels
34/// (ATM's 2024 fits to Chajnantor spectra).
35pub(super) fn minor_gases(height_km: f64, reference: usize, ozone_model: u32) -> MinorGases {
36    if !(0.0..=120.0).contains(&height_km) {
37        return MinorGases::default();
38    }
39    // The first tabulated level above the height, and its neighbours.
40    let Some(above) = ALT.iter().position(|&level| height_km < level) else {
41        return MinorGases::default();
42    };
43    let (i1, i2, i3) = match above {
44        0 => (0, 1, 2),
45        49 => (47, 48, 49),
46        k => (k - 1, k, k + 1),
47    };
48    let (x1, x2, x3) = (ALT[i1], ALT[i2], ALT[i3]);
49    if x1 == 0.0 && x2 == 0.0 && x3 == 0.0 {
50        return MinorGases::default();
51    }
52    let at = |table: &[f64; 50]| parabola(height_km, x1, x2, x3, table[i1], table[i2], table[i3]);
53    let r = reference - 1;
54    // Air density in g/m³.
55    let density = at(&DEN[r]) * AIR_MOLAR_MASS * 1e6 / AVOGADRO;
56    let per_cm3 = |mixing_ratio: f64| mixing_ratio * 1e-12 * density * AVOGADRO / AIR_MOLAR_MASS;
57
58    let mut o3 = per_cm3(at(&OZONE[r]));
59    o3 *= ozone_factor(ozone_model, i2);
60    let mut n2o = per_cm3(at(&N2O[r]));
61    if height_km > 36.0 {
62        n2o *= 1.5;
63    } else if height_km > 30.0 {
64        n2o *= 2.5;
65    }
66    let no2 = per_cm3(at(&NO2));
67    let mut hcl = per_cm3(at(&HCL));
68    if i2 >= 30 {
69        hcl *= 3.5;
70    }
71    let mut hcn = per_cm3(at(&HCN));
72    hcn *= if i2 < 25 { 0.9 } else { 2.5 };
73    let so2 = per_cm3(at(&SO2));
74    let co = per_cm3(at(&CO[r]));
75    MinorGases {
76        o3,
77        n2o,
78        co,
79        no2,
80        so2,
81        hcl,
82        hcn,
83    }
84}

86/// ATM's ozone scaling for its seasonal/diurnal models, by the middle of the
87/// three interpolation levels.
88fn ozone_factor(ozone_model: u32, i2: usize) -> f64 {
89    // (first upper level, factor below level 24, factor up to the split,
90    // factor below level 32, below level 35, above)
91    let (split, low, low_upper, mid, high, top) = match ozone_model {
92        1 => (29, 0.650, 0.860, 1.35, 1.60, 2.45),
93        2 => (29, 0.650, 0.890, 1.45, 1.6, 2.55),
94        3 => (30, 0.500, 0.800, 1.65, 1.70, 2.40),
95        4 => (30, 0.500, 0.800, 1.65, 1.70, 2.50),
96        _ => return 1.0,
97    };
98    if i2 < split {
99        if i2 < 24 { low } else { low_upper }
100    } else if i2 < 32 {
101        mid
102    } else if i2 < 35 {
103        high
104    } else {
105        top
106    }
107}

109/// The parabola through (x1, y1), (x2, y2), (x3, y3), evaluated at `x`:
110/// ATM's `poli2`.
111fn parabola(x: f64, x1: f64, x2: f64, x3: f64, y1: f64, y2: f64, y3: f64) -> f64 {
112    let mut c = (y3 - y2) * (x2 - x1) - (y2 - y1) * (x3 - x2);
113    let mut b = (x2 - x1) * (x3 * x3 - x2 * x2) - (x2 * x2 - x1 * x1) * (x3 - x2);
114    c /= b;
115    b = (y2 - y1) - c * (x2 * x2 - x1 * x1);
116    b /= x2 - x1;
117    let a = y1 - c * x1 * x1 - b * x1;
118    a + b * x + c * x.powf(2.0)
119}
