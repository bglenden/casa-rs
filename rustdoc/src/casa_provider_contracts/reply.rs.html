2//! The shared task front end's wire types (#931): the `--json-run` request a
3//! caller sends a task binary, and the one typed reply the binary writes to
4//! stdout.
5//!
6//! The reply is the whole of stdout. Log lines go to stderr, and nothing is
7//! ever inferred from them. The exit status agrees with the reply:
8//! [`EXIT_SUCCEEDED`] for [`TaskReply::Succeeded`], and
9//! [`TaskFailureKind::exit_code`] for [`TaskReply::Failed`]. A caller
10//! classifies a finished process from its exit status and stdout alone; a
11//! process killed by a signal, or one whose stdout is not a reply that agrees
12//! with its exit status, is a different outcome from either failure kind.
13
14use schemars::JsonSchema;
15use schemars::schema::RootSchema;
16use serde::{Deserialize, Serialize};
17use serde_json::{Map, Value};
18
19/// Exit status of a task whose reply is [`TaskReply::Succeeded`].
20pub const EXIT_SUCCEEDED: i32 = 0;
21/// Exit status of a task that started and failed ([`TaskFailureKind::Execution`]).
22pub const EXIT_EXECUTION_FAILED: i32 = 1;
23/// Exit status of a task whose parameters were refused before it changed
24/// anything ([`TaskFailureKind::Validation`]).
25pub const EXIT_VALIDATION_FAILED: i32 = 2;
26
27/// One `--json-run` request: the surface to run and the CASA-named values the
28/// caller set.
29///
30/// Every parameter missing from `values` takes its current default. The task
31/// binary resolves the values against the surface's contract itself, as it
32/// resolves command-line flags, so a Python call and a command line share one
33/// resolution. A `--json-run` call neither reads nor writes Last state.
34#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
35#[serde(deny_unknown_fields)]
36pub struct TaskRunRequest {
37    /// Surface id, such as `"split"`; one the binary hosts.
38    pub surface: String,
39    /// The values the caller set, keyed by CASA parameter name, as plain JSON.
40    #[serde(default)]
41    pub values: Map<String, Value>,
42}
43
44/// Why a task did not succeed.
45#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
46#[serde(rename_all = "snake_case")]
47pub enum TaskFailureKind {
48    /// The parameters were refused before the task changed anything. Python
49    /// raises `ValueError`.
50    Validation,
51    /// The task started and failed. Python raises `RuntimeError`.
52    Execution,
53}
54
55impl TaskFailureKind {
56    /// The exit status that accompanies a reply of this kind.
57    pub fn exit_code(self) -> i32 {
58        match self {
59            Self::Validation => EXIT_VALIDATION_FAILED,
60            Self::Execution => EXIT_EXECUTION_FAILED,
61        }
62    }
63}
64
65/// The typed reply a task binary writes to stdout for `--json-run`.
66#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
67#[serde(tag = "status", rename_all = "snake_case")]
68pub enum TaskReply {
69    /// The task succeeded.
70    Succeeded {
71        /// Surface that ran.
72        surface: String,
73        /// The value CASA's task returns, such as `imstat`'s statistics
74        /// record, or `null` where CASA returns `None`. It is kept ahead of
75        /// the generic product projection, which cannot recover it.
76        value: Value,
77        /// The provider's full typed result, from which the surface's declared
78        /// products are projected.
79        result: Value,
80    },
81    /// The task failed.
82    Failed {
83        /// Surface that was asked to run.
84        surface: String,
85        /// Whether the parameters were refused or the run failed.
86        kind: TaskFailureKind,
87        /// The reason, as the task reports it.
88        message: String,
89        /// The parameter at fault, when the failure names one.
90        #[serde(default, skip_serializing_if = "Option::is_none")]
91        parameter: Option<String>,
92    },
93}
94
95impl TaskReply {
96    /// The surface the reply is about.
97    pub fn surface(&self) -> &str {
98        match self {
99            Self::Succeeded { surface, .. } | Self::Failed { surface, .. } => surface,
100        }
101    }
102
103    /// The exit status that must accompany this reply.
104    pub fn exit_code(&self) -> i32 {
105        match self {
106            Self::Succeeded { .. } => EXIT_SUCCEEDED,
107            Self::Failed { kind, .. } => kind.exit_code(),
108        }
109    }
110}

112/// The shared front end's `--json-run` request and reply schemas, carried by
113/// a provider contract whose binary runs on the front end.
114#[derive(Debug, Clone, Serialize)]
115pub struct TaskFrontEndSchemas {
116    /// Schema of [`TaskRunRequest`].
117    pub run_request_schema: RootSchema,
118    /// Schema of [`TaskReply`].
119    pub reply_schema: RootSchema,
120}
121
122impl Default for TaskFrontEndSchemas {
123    fn default() -> Self {
124        Self {
125            run_request_schema: schemars::schema_for!(TaskRunRequest),
126            reply_schema: schemars::schema_for!(TaskReply),
127        }
128    }
129}

131#[cfg(test)]
132mod tests {
133    use super::*;
134    use serde_json::json;
135
136    #[test]
137    fn a_success_carries_casas_return_value_ahead_of_the_result() {
138        let reply = TaskReply::Succeeded {
139            surface: "imstat".into(),
140            value: json!({"npts": [12.0]}),
141            result: json!({"kind": "statistics"}),
142        };
143        let text = serde_json::to_string(&reply).unwrap();
144        assert_eq!(
145            text,
146            r#"{"status":"succeeded","surface":"imstat","value":{"npts":[12.0]},"result":{"kind":"statistics"}}"#
147        );
148        assert_eq!(serde_json::from_str::<TaskReply>(&text).unwrap(), reply);
149        assert_eq!(reply.exit_code(), EXIT_SUCCEEDED);
150    }
151
152    #[test]
153    fn failure_kinds_keep_distinct_exit_statuses() {
154        let refused: TaskReply = serde_json::from_value(json!({
155            "status": "failed",
156            "surface": "split",
157            "kind": "validation",
158            "message": "unknown parameter",
159            "parameter": "widht",
160        }))
161        .unwrap();
162        assert_eq!(refused.exit_code(), EXIT_VALIDATION_FAILED);
163        assert_eq!(refused.surface(), "split");
164        let failed = TaskReply::Failed {
165            surface: "split".into(),
166            kind: TaskFailureKind::Execution,
167            message: "disk full".into(),
168            parameter: None,
169        };
170        assert_eq!(failed.exit_code(), EXIT_EXECUTION_FAILED);
171        assert!(
172            !serde_json::to_string(&failed)
173                .unwrap()
174                .contains("parameter")
175        );
176    }
177
178    #[test]
179    fn a_run_request_names_its_surface_and_only_the_values_set() {
180        let request: TaskRunRequest =
181            serde_json::from_value(json!({"surface": "split", "values": {"width": 2}})).unwrap();
182        assert_eq!(request.values["width"], 2);
183        let bare: TaskRunRequest = serde_json::from_value(json!({"surface": "split"})).unwrap();
184        assert!(bare.values.is_empty());
185        assert!(
186            serde_json::from_value::<TaskRunRequest>(json!({"surface": "split", "vals": {}}))
187                .is_err()
188        );
189    }
190}
