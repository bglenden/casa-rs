2//! The observation a request names, opened once for compile and the run: a
3//! MeasurementSet (MSv2) or an MSv4 processing set, through the observation
4//! boundary. Each store admits its opening before allocating and keeps
5//! the charge for its retained tables, index, catalogue and coordinates.
6
7use std::path::Path;
8use std::sync::Arc;
9
10use casa_imaging_runtime::{Admitted, Demand, HostResources, ResourcePolicy, admit};
11use casa_ms::observation_io::{Msv2Observation, Msv4Observation, OpenOptions};
12use casa_observation::{Catalogue, Observation, ObservationError};
13
14use super::PrepareError;

16/// The observation compile and the run read, in its store.
17pub(crate) enum OpenedStore {
18    /// A MeasurementSet v2, assembled into the boundary's partitions at
19    /// opening (boxed: it holds MAIN's index and subtables).
20    V2(Box<Msv2Observation>),
21    /// An MSv4 processing set (XRADIO's schema on local Zarr v3).
22    V4(Msv4Observation),
23}
24
25impl OpenedStore {
26    /// The observation, whatever its store.
27    pub(crate) fn observation(&self) -> &dyn Observation {
28        match self {
29            Self::V2(observation) => observation.as_ref(),
30            Self::V4(observation) => observation,
31        }
32    }
33
34    /// Its catalogue.
35    pub(crate) fn catalogue(&self) -> &Arc<Catalogue> {
36        self.observation().catalogue()
37    }
38
39    /// The observation, owned, for a run that only reads it.
40    pub(crate) fn into_observation(self) -> Box<dyn Observation> {
41        match self {
42            Self::V2(observation) => observation,
43            Self::V4(observation) => Box::new(observation),
44        }
45    }
46
47    /// The MeasurementSet, for what only an MSv2 states: its data
48    /// descriptions and its `PHASE_DIR` polynomials.
49    pub(crate) fn msv2(&self) -> Option<&Msv2Observation> {
50        match self {
51            Self::V2(observation) => Some(observation.as_ref()),
52            Self::V4(_) => None,
53        }
54    }
55}

57/// The observation at `path`, an MSv4 processing set when it is one (its
58/// root holds `zarr.json`), else a MeasurementSet,
59/// with the charge of what it holds while it is open.
60pub(crate) fn open_observation(
61    path: &Path,
62    (host, policy): (&HostResources, &ResourcePolicy),
63) -> Result<Admitted<OpenedStore>, PrepareError> {
64    if path.join("zarr.json").is_file() {
65        let Admitted { value, reservation } = admitted_opening(
66            (0, "the MSv4 opening"),
67            |grant| Msv4Observation::open_granted(path, &OpenOptions::default(), grant),
68            (
69                |opened: &Msv4Observation| opened.held_bytes() as u64,
70                "observation opening",
71            ),
72            (host, policy),
73        )?;
74        return Ok(Admitted {
75            value: OpenedStore::V4(value),
76            reservation,
77        });
78    }
79    let Admitted { value, reservation } = admitted_opening(
80        (0, "the MSv2 opening"),
81        |grant| Msv2Observation::open_granted(path, &OpenOptions::default(), grant),
82        (
83            |opened: &Msv2Observation| opened.retained_bytes() as u64,
84            "observation retained",
85        ),
86        (host, policy),
87    )?;
88    Ok(Admitted {
89        value: OpenedStore::V2(Box::new(value)),
90        reservation,
91    })
92}

94/// The store `open` opens, admitted on `host` under `policy`: `initial`
95/// before it opens, and each increase its opening asks the grant for beyond
96/// what is admitted before it is allocated, so that one that does not fit
97/// refuses the opening (`what`), as an admission. The charge is then
98/// trimmed to what the opened store holds (`held`), or raised to it
99/// (`held_phase`).
100fn admitted_opening<T>(
101    (initial, what): (u64, &'static str),
102    open: impl FnOnce(
103        &mut dyn FnMut(usize) -> Result<(), ObservationError>,
104    ) -> Result<T, ObservationError>,
105    (held, held_phase): (impl FnOnce(&T) -> u64, &'static str),
106    (host, policy): (&HostResources, &ResourcePolicy),
107) -> Result<Admitted<T>, PrepareError> {
108    let mut reservation = admit(
109        host,
110        policy,
111        &Demand {
112            phase: "observation opening",
113            memory: initial,
114        },
115    )?;
116    let mut refusal = None;
117    let opened = open(&mut |total| {
118        let total = total as u64;
119        if total > reservation.memory() {
120            let demand = Demand {
121                phase: "observation opening",
122                memory: total - reservation.memory(),
123            };
124            match admit(host, policy, &demand) {
125                Ok(more) => reservation.join(more),
126                Err(admission) => {
127                    refusal = Some(admission);
128                    return Err(ObservationError::OverBudget {
129                        what,
130                        required: total as usize,
131                        budget: reservation.memory() as usize,
132                    });
133                }
134            }
135        }
136        Ok(())
137    });
138    let opened = opened.map_err(|error| match refusal.take() {
139        Some(admission) => PrepareError::Admission(admission),
140        None => PrepareError::Opening(error),
141    })?;
142    let held = held(&opened);
143    if held <= reservation.memory() {
144        reservation.retain(held);
145    } else {
146        reservation.join(admit(
147            host,
148            policy,
149            &Demand {
150                phase: held_phase,
151                memory: held - reservation.memory(),
152            },
153        )?);
154    }
155    Ok(Admitted {
156        value: opened,
157        reservation,
158    })
159}
