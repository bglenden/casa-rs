2//! CASA table-column statistics (#1028).
3
4use casa_tables::statistics::{ColumnStatisticsError, column_statistics};
5use pyo3::prelude::*;
6
7use super::state::{Fail, State, no_table_specified};
8use super::{PyTableTool, text};
9use crate::convert::value::CasaValue;
10use crate::tools::LogLine;
11
12impl State {
13    fn statistics(
14        &self,
15        column: &str,
16        mode: &str,
17        log: &mut Vec<LogLine>,
18    ) -> Result<CasaValue, Fail> {
19        let Some(open) = self.open.as_ref() else {
20            no_table_specified(log);
21            return Ok(CasaValue::empty_record());
22        };
23        if open.row_map.as_ref().is_some_and(Vec::is_empty) {
24            return Err(Fail::Runtime("Table has zero rows, cannot continue".into()));
25        }
26        if !open.column_allowed(column) {
27            return Err(Fail::Runtime(format!("Table column {column} is unknown")));
28        }
29        let root = open.root();
30        let result =
31            column_statistics(&root.table, open.row_map.as_deref(), column, mode, |line| {
32                log.push(LogLine::info(line));
33            })
34            .map_err(|error| match error {
35                ColumnStatisticsError::Unsupported(message) => Fail::Unsupported(message),
36                error => Fail::Runtime(error.to_string()),
37            })?;
38        Ok(CasaValue::record(result.into_iter().map(
39            |(column, stats)| {
40                (
41                    column,
42                    CasaValue::record(
43                        stats
44                            .into_iter()
45                            .map(|(key, value)| (key, CasaValue::Float(value))),
46                    ),
47                )
48            },
49        )))
50    }
51}
52
53#[pymethods]
54impl PyTableTool {
55    /// Compute CASA statistics for a table column.
56    #[pyo3(signature = (column, complex_value = None), text_signature = "($self, column, complex_value='')")]
57    fn statistics(
58        &self,
59        py: Python<'_>,
60        column: &Bound<'_, PyAny>,
61        complex_value: Option<&Bound<'_, PyAny>>,
62    ) -> PyResult<Py<PyAny>> {
63        let column = text(column, "column")?;
64        let mode = complex_value.map_or(Ok(String::new()), |value| text(value, "complex_value"))?;
65        self.read(py, move |state, log| state.statistics(&column, &mode, log))
66    }
67}
