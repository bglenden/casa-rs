25pub(super) fn run(request: &ImpbcorRequest) -> Result<PbcorSummary, ImageError> {
26    let correction = Correction::parse(&request.mode)?;
27    let destination = destination(request)?;
28    let input = import_fits_for_reading(&request.imagename)?;
29    let image = AnyPagedImage::open(
30        input
31            .as_ref()
32            .map_or(request.imagename.as_path(), |(_, path)| path),
33    )?;
34    let beam_path = match &request.pbimage {
35        PbcorBeam::Image(path) => Some(path.as_path()),
36        PbcorBeam::Pixels(_) => None,
37    };
38    let converted = beam_path
39        .map(import_fits_for_reading)
40        .transpose()?
41        .flatten();
42    let beam = beam_path
43        .map(|path| AnyPagedImage::open(converted.as_ref().map_or(path, |(_, path)| path)))
44        .transpose()?;
45    match (&image, &beam) {
46        (AnyPagedImage::Float32(image), Some(AnyPagedImage::Float32(beam))) => {
47            run_typed(image, Some(beam), request, correction, &destination)
48        }
49        (AnyPagedImage::Float64(image), Some(AnyPagedImage::Float64(beam))) => {
50            run_typed(image, Some(beam), request, correction, &destination)
51        }
52        (AnyPagedImage::Float32(image), None) => {
53            run_typed(image, None, request, correction, &destination)
54        }
55        (AnyPagedImage::Float64(image), None) => {
56            run_typed(image, None, request, correction, &destination)
57        }
58        _ => Err(ImageError::InvalidMetadata(
59            "impbcor requires matching real-valued image pixel types".into(),
60        )),
61    }
62}

64fn import_fits_for_reading(
65    path: &Path,
66) -> Result<Option<(tempfile::TempDir, PathBuf)>, ImageError> {
67    if !path.is_file() || !is_fits_file(path)? {
68        return Ok(None);
69    }
70    let stage = tempfile::tempdir().map_err(output_error)?;
71    let imagename = stage.path().join("image");
72    importfits(&ImportFitsRequest {
73        fitsimage: path.to_path_buf(),
74        imagename: imagename.clone(),
75        zeroblanks: false,
76        ..ImportFitsRequest::default()
77    })?;
78    Ok(Some((stage, imagename)))
79}

81fn run_typed<T>(
82    image: &PagedImage<T>,
83    pb: Option<&PagedImage<T>>,
84    request: &ImpbcorRequest,
85    correction: Correction,
86    destination: &Path,
87) -> Result<PbcorSummary, ImageError>
88where
89    T: ImagePixel + Into<f64> + From<f32> + Copy + Mul<Output = T> + Div<Output = T>,
90{
91    let record = region_record(image, request.region.as_ref())?;
92    let region = match request.region.as_ref() {
93        Some(PbcorRegion::Text(text)) if record.is_none() && !text.trim().is_empty() => {
94            Some(SubimageRegion::Text(text))
95        }
96        _ => record.as_ref().map(SubimageRegion::Record),
97    };
98    if region.is_some() {
99        let supplied = |value: Option<&str>| value.is_some_and(|text| !text.trim().is_empty());
100        if supplied(request.box_selection.as_deref()) {
101            return Err(ImageError::CasaSelection(
102                "box and regionPtr cannot be simultaneously specified".into(),
103            ));
104        }
105        if supplied(request.chans.as_deref()) || supplied(request.stokes.as_deref()) {
106            return Err(ImageError::CasaSelection(
107                "regionPtr and regionName, chans, and/or stokes cannot be simultaneously specified"
108                    .into(),
109            ));
110        }
111    }
112    let chans = request.chans.as_deref().map(str::trim).filter(|text| {
113        !text.eq_ignore_ascii_case("ALL") && image.find_axis(CoordinateType::Spectral).is_some()
114    });
115    let stokes = normalize_stokes(image, request.stokes.as_deref());
116    let box_pixels = request
117        .box_selection
118        .as_deref()
119        .filter(|text| !text.trim().is_empty())
120        .map(|text| normalize_box(image, text))
121        .transpose()?;
122    let selection = SubimageSelection::resolve(
123        image,
124        box_pixels.as_deref(),
125        region,
126        chans,
127        stokes.as_deref(),
128    )?;
129    if selection.has_multiple_regions {
130        return Err(ImageError::CasaSelection(
131            "Only a single n-dimensional rectangular region is supported".into(),
132        ));
133    }
134    let mut data = image.get_slice(&selection.start, &selection.shape)?;
135    let mut mask =
136        combined_selection_mask(image, &selection, request.mask.as_deref(), request.stretch)?;
137    selection.apply_selection_masks(&mut mask);
138    let beam = match (&request.pbimage, pb) {
139        (PbcorBeam::Pixels(values), _) => beam::array(image, values, &selection)?,
140        (PbcorBeam::Image(_), Some(beam)) => beam::image(image, beam, &selection)?,
141        _ => unreachable!("a named primary beam was opened before execution"),
142    };
143    let mut valid = mask.unwrap_or_else(|| ArrayD::from_elem(IxDyn(&selection.shape), true));
144    correction.apply_to_window(
145        &mut data,
146        &mut valid,
147        &beam.pixels,
148        beam.mask.as_ref(),
149        request.cutoff as f32,
150    )?;
151    let coords = match image.table().keywords().get("coords") {
152        Some(Value::Record(record)) => Some(record.clone()),
153        _ => None,
154    };
155    let plan = SubimagePlan::from_selection(image, coords, selection, false, &[], None)?;
156    let valid_pixels = valid.iter().filter(|&&good| good).count();
157    let parent = destination.parent().ok_or(ImageError::NotPersistent)?;
158    let stage = tempfile::tempdir_in(parent).map_err(output_error)?;
159    let product = stage.path().join("image");
160    let mut output = PagedImage::<T>::create(plan.shape.clone(), plan.coordinates, &product)?;
161    if let Some(record) = plan.coordinate_record {
162        output.set_coords_record_keyword(record);
163    }
164    output.set_units(image.units())?;
165    output.set_image_info(&plan.image_info)?;
166    output.set_misc_info(image.misc_info())?;
167    output.put_slice(&data, &vec![0; plan.shape.len()])?;
168    if valid_pixels != valid.len() {
169        output.put_mask("mask0", &valid)?;
170        output.set_default_mask("mask0")?;
171    }
172    let pb_label = match &request.pbimage {
173        PbcorBeam::Image(path) => path.display().to_string(),
174        PbcorBeam::Pixels(_) => format!("floating-point beam array {:?}", beam.input_shape),
175    };
176    write_history(&mut output, request, &pb_label)?;
177    output.save()?;
178    drop(output);
179    match rename_without_replacing(&product, destination) {
180        Err(error) if error.kind() == io::ErrorKind::AlreadyExists && request.overwrite => {
181            check_destination(destination, request)?;
182            exchange_directories(&product, destination).map_err(output_error)?;
183        }
184        result => result.map_err(output_error)?,
185    }
186    Ok(PbcorSummary {
187        imagename: request.imagename.display().to_string(),
188        pbimage: pb_label,
189        outfile: request.outfile.display().to_string(),
190        mode: request.mode.clone(),
191        cutoff: request.cutoff,
192        shape: plan.shape,
193        units: image.units().to_owned(),
194        valid_pixels,
195    })
196}

198pub(super) fn normalize_stokes<T: ImagePixel>(
199    image: &dyn crate::ImageInterface<T>,
200    text: Option<&str>,
201) -> Option<String> {
202    text.map(str::trim)
203        .map(|text| text.trim_start_matches('[').trim_end_matches(']').trim())
204        .filter(|text| {
205            !text.eq_ignore_ascii_case("ALL") && image.find_axis(CoordinateType::Stokes).is_some()
206        })
207        .map(|text| text.replace(';', ","))
208}

210pub(super) fn normalize_box<T: ImagePixel>(
211    image: &dyn crate::ImageInterface<T>,
212    text: &str,
213) -> Result<String, ImageError> {
214    let plane = super::axes::direction_plane_axes(image)?;
215    text.split(',')
216        .enumerate()
217        .map(|(i, value)| {
218            let axis_len = image.shape()[plane[i % 2]];
219            let coordinate = value.trim().parse::<f64>().map_err(|_| {
220                ImageError::CasaSelection(format!("invalid numeric box corner {value:?}"))
221            })?;
222            if !coordinate.is_finite() || !(0.0..axis_len as f64).contains(&coordinate) {
223                return Err(ImageError::CasaSelection(format!(
224                    "box corner {coordinate} is outside the image"
225                )));
226            }
227            Ok((coordinate.round() as usize).min(axis_len - 1).to_string())
228        })
229        .collect::<Result<Vec<_>, _>>()
230        .map(|values| values.join(","))
231}

233fn region_record<T: ImagePixel>(
234    image: &PagedImage<T>,
235    region: Option<&PbcorRegion>,
236) -> Result<Option<RecordValue>, ImageError> {
237    match region {
238        Some(PbcorRegion::Record(fields)) if !fields.is_empty() => {
239            let value = parameter_value_to_casacore(&ParameterValue::Table(fields.clone()))
240                .map_err(|error| {
241                    ImageError::CasaSelection(format!("invalid region record: {error}"))
242                })?;
243            let Value::Record(record) = value else {
244                unreachable!("a parameter table converts to a record")
245            };
246            Ok(Some(record))
247        }
248        Some(PbcorRegion::Text(name))
249            if !name.trim().is_empty()
250                && !name.contains(['[', ':', '\n'])
251                && !Path::new(name).exists() =>
252        {
253            Ok(image.get_region_record(name.trim()).ok())
254        }
255        _ => Ok(None),
256    }
257}

259#[derive(Clone, Copy)]
260pub(super) enum Correction {
261    Divide,
262    Multiply,
263}
264
265impl Correction {
266    pub(super) fn parse(mode: &str) -> Result<Self, ImageError> {
267        match mode.trim().as_bytes().first().map(u8::to_ascii_lowercase) {
268            Some(b'd') => Ok(Self::Divide),
269            Some(b'm') => Ok(Self::Multiply),
270            _ => Err(ImageError::CasaSelection(format!("Unknown mode {mode}"))),
271        }
272    }
273
274    fn keeps(self, beam: f64, cutoff: f32) -> bool {
275        cutoff.is_nan()
276            || cutoff < 0.0
277            || match self {
278                Self::Divide => beam >= f64::from(cutoff),
279                Self::Multiply => beam <= f64::from(cutoff),
280            }
281    }
282
283    pub(super) fn apply_to_window<T>(
284        self,
285        data: &mut ArrayD<T>,
286        validity: &mut ArrayD<bool>,
287        beam: &ArrayD<T>,
288        beam_mask: Option<&ArrayD<bool>>,
289        cutoff: f32,
290    ) -> Result<(), ImageError>
291    where
292        T: Copy + Into<f64> + Div<Output = T> + Mul<Output = T>,
293    {
294        let shape = data.raw_dim();
295        let pixels = beam
296            .broadcast(shape.clone())
297            .ok_or_else(|| ImageError::ShapeMismatch {
298                expected: data.shape().to_vec(),
299                got: beam.shape().to_vec(),
300            })?;
301        Zip::from(data)
302            .and(&mut *validity)
303            .and(pixels)
304            .for_each(|pixel, good, &pb| {
305                *good &= self.keeps(pb.into(), cutoff);
306                *pixel = self.apply(*pixel, pb);
307            });
308        if let Some(mask) = beam_mask {
309            let mask = mask.broadcast(shape).expect("beam and mask share a shape");
310            Zip::from(validity)
311                .and(mask)
312                .for_each(|good, &mask| *good &= mask);
313        }
314        Ok(())
315    }
316
317    fn apply<T: Div<Output = T> + Mul<Output = T>>(self, image: T, beam: T) -> T {
318        match self {
319            Self::Divide => image / beam,
320            Self::Multiply => image * beam,
321        }
322    }
323}

325fn write_history<T: ImagePixel>(
326    output: &mut PagedImage<T>,
327    request: &ImpbcorRequest,
328    beam: &str,
329) -> Result<(), ImageError> {
330    let source = &request.imagename;
331    if let PbcorBeam::Image(path) = &request.pbimage {
332        for input in [source, path] {
333            output.append_history_at(
334                "::impbcor",
335                &[format!("History of image {}", input.display())],
336            )?;
337            output.append_history_entries(&crate::image::image_history_rows(input)?)?;
338            output.append_history_at(
339                "::impbcor",
340                &[format!("End history of image {}", input.display())],
341            )?;
342        }
343    } else {
344        output.append_history_entries(&crate::image::image_history_rows(source)?)?;
345    }
346    output.append_history_at("::impbcor", &[
347        concat!("version: casa-rs ", env!("CARGO_PKG_VERSION")).into(),
348        format!(
349        "impbcor(imagename={:?}, pbimage={beam:?}, outfile={:?}, overwrite={}, box={:?}, region={:?}, chans={:?}, stokes={:?}, mask={:?}, mode={:?}, cutoff={}, stretch={})",
350        source.display().to_string(), request.outfile.display().to_string(), request.overwrite,
351        request.box_selection, request.region, request.chans, request.stokes, request.mask, request.mode,
352        request.cutoff, request.stretch,
353    )])
354}

356fn destination(request: &ImpbcorRequest) -> Result<PathBuf, ImageError> {
357    let name = request
358        .outfile
359        .file_name()
360        .ok_or_else(|| ImageError::CasaSelection("outfile must be specified".into()))?;
361    let output = std::path::absolute(&request.outfile).map_err(output_error)?;
362    let parent = output
363        .parent()
364        .ok_or(ImageError::NotPersistent)?
365        .canonicalize()
366        .map_err(output_error)?;
367    let destination = parent.join(name);
368    check_destination(&destination, request)?;
369    Ok(destination)
370}
371
372fn check_destination(destination: &Path, request: &ImpbcorRequest) -> Result<(), ImageError> {
373    let beam = match &request.pbimage {
374        PbcorBeam::Image(path) => Some(path.as_path()),
375        PbcorBeam::Pixels(_) => None,
376    };
377    for input in [Some(request.imagename.as_path()), beam]
378        .into_iter()
379        .flatten()
380    {
381        super::ensure_outfile_does_not_alias_input(destination, input)?;
382        if input
383            .canonicalize()
384            .is_ok_and(|input| destination.starts_with(input))
385        {
386            return Err(ImageError::CasaSelection(
387                "outfile must be outside the input image".into(),
388            ));
389        }
390    }
391    let metadata = match destination.symlink_metadata() {
392        Ok(metadata) => metadata,
393        Err(error) if error.kind() == io::ErrorKind::NotFound => return Ok(()),
394        Err(error) => return Err(output_error(error)),
395    };
396    if !request.overwrite {
397        return Err(ImageError::CasaSelection(format!(
398            "impbcor output already exists: {}",
399            destination.display()
400        )));
401    }
402    if !metadata.is_dir()
403        || !(Table::open_metadata_only(TableOptions::new(destination))
404            .is_ok_and(|table| table.info().table_type == "Image")
405            || fs::read_dir(destination).is_ok_and(|mut entries| entries.next().is_none()))
406    {
407        return Err(ImageError::CasaSelection(format!(
408            "{} is not a CASA image or an empty directory",
409            destination.display()
410        )));
411    }
412    Ok(())
413}

415fn output_error(error: io::Error) -> ImageError {
416    ImageError::Io(error.to_string())
