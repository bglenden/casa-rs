2//! The paged state of a channel-local run: model and normal planes in one
3//! bounded block cache over files in the run's paged-state directory (plan
4//! decision D10).

8use casa_imaging_reconstruction::runtime_adapter::NormalStoragePlan;
9use casa_imaging_reconstruction::{ModelLifecycleError, ModelStoragePlan, SpectralOperatorError};

16/// Planes of one channel-local run kept in a bounded cache and paged to
17/// disk: two model generations, an invariant PSF and two residual epochs of
18/// every image domain, each domain at its own size.
19pub struct CubeState {
20    residency: Arc<CubeResidency>,
21    model: Arc<ManagedModelFactory>,
22    normal: Arc<ManagedNormalFactory>,
23    window_samples: usize,
24}
25
26/// Payload, fixed and per-access bytes of the paged state of `domains`.
27struct Footprint {
28    /// Bytes below which no plane access can proceed.
29    minimum: usize,
30    /// Cache bytes of every block of every array.
31    payload: usize,
32    /// Bytes the cache needs beyond its payload.
33    overhead: usize,
34    /// File bytes of every array once every block is paged out.
35    disk: usize,
36}
37
38fn footprint(
39    directory: &Path,
40    domains: &[[usize; 2]],
41    planes: usize,
42    workers: usize,
43) -> io::Result<Footprint> {
44    let mut owners = 0;
45    let mut active = 0;
46    let mut creation = 0;
47    let mut payload = 0;
48    let mut disk = 0;
49    for &[width, height] in domains {
50        let value = ManagedPlaneArray::<f32>::footprint(directory, height, width, planes)?;
51        let support = ManagedPlaneArray::<bool>::footprint(directory, height, width, planes)?;
52        owners += 5 * (value.owner_bytes + value.registry_bytes + value.staging_bytes)
53            + 2 * (support.owner_bytes + support.registry_bytes + support.staging_bytes);
54        active = usize::max(
55            active,
56            workers * (3 * value.block_bytes + 2 * support.block_bytes),
57        );
58        creation = usize::max(creation, value.creation_bytes.max(support.creation_bytes));
59        payload += planes * (5 * value.block_bytes + 2 * support.block_bytes);
60        disk += 5 * value.storage_bytes + 2 * support.storage_bytes;
61    }
62    let operation = CubeResidency::operation_overhead(5 * workers)?;
63    let overhead = CubeResidency::fixed_owner_bytes() + owners + operation;
64    Ok(Footprint {
65        minimum: overhead + active.max(creation),
66        payload,
67        overhead,
68        disk,
69    })
70}

72impl CubeState {
73    /// Cache bytes below which a run of `planes` planes of each
74    /// `[width, height]` in `domains` cannot proceed, and above which every
75    /// plane stays resident, for `workers` concurrent plane accesses.
76    pub fn cache_limits(
77        directory: &Path,
78        domains: &[[usize; 2]],
79        planes: usize,
80        workers: usize,
81    ) -> io::Result<(usize, usize)> {
82        let footprint = footprint(directory, domains, planes, workers)?;
83        Ok((
84            footprint.minimum,
85            (footprint.overhead + footprint.payload).max(footprint.minimum),
86        ))
87    }
88
89    /// A cache of the bytes `cache` holds over files in `directory` for
90    /// `planes` planes of each `[width, height]` in `domains` (each domain's
91    /// planes are its channels × polarizations).
92    ///
93    /// The reservation is released when the last storage this state hands
94    /// out drops, so a completed run's model and normal state stay charged
95    /// for as long as anything holds them.
96    ///
97    /// Every plane may be paged out, so the directory's file system must have
98    /// room for the whole payload; otherwise the error is
99    /// [`io::ErrorKind::StorageFull`].
100    pub fn new(
101        directory: &Path,
102        domains: &[[usize; 2]],
103        planes: usize,
104        cache: Reservation,
105    ) -> io::Result<Self> {
106        let cache_bytes = usize::try_from(cache.memory())
107            .map_err(|_| io::Error::other("the cube cache exceeds the address space"))?;
108        let disk = footprint(directory, domains, planes, 1)?.disk;
109        let available = fs2::available_space(directory)?;
110        if available < disk as u64 {
111            return Err(io::Error::new(
112                io::ErrorKind::StorageFull,
113                format!(
114                    "the paged cube state needs {disk} bytes but {} has {available} free",
115                    directory.display()
116                ),
117            ));
118        }
119        let axes = domains
120            .iter()
121            .map(|[width, height]| [*height, *width])
122            .collect::<Vec<_>>();
123        let residency = CubeResidency::new(cache_bytes)?;
124        let retention: Arc<dyn std::fmt::Debug + Send + Sync> = Arc::new(cache);
125        Ok(Self {
126            model: Arc::new(ManagedModelFactory::new(
127                residency.clone(),
128                retention.clone(),
129                directory,
130                &axes,
131                planes,
132            )?),
133            normal: Arc::new(ManagedNormalFactory::new(
134                residency.clone(),
135                retention,
136                directory,
137                &axes,
138            )?),
139            residency,
140            window_samples: domains
141                .iter()
142                .map(|[width, height]| width * height)
143                .max()
144                .unwrap_or(0),
145        })
146    }
147
148    /// Model storage reading and writing at most one plane of the largest
149    /// domain at a time.
150    pub fn model_storage(&self) -> Result<ModelStoragePlan, ModelLifecycleError> {
151        ModelStoragePlan::new(self.model.clone(), self.window_samples)
152    }
153
154    /// Normal storage writing at most `window_channels` channels at a time.
155    pub fn normal_storage(
156        &self,
157        window_channels: usize,
158    ) -> Result<NormalStoragePlan, SpectralOperatorError> {
159        NormalStoragePlan::new(self.normal.clone(), window_channels)
160    }
161
162    /// Bytes of plane payload resident now.
163    #[must_use]
164    pub fn live_payload_bytes(&self) -> usize {
165        self.residency.live_payload_bytes()
166    }
167}
168
169impl std::fmt::Debug for CubeState {
170    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
171        f.debug_struct("CubeState")
172            .field("window_samples", &self.window_samples)
173            .finish_non_exhaustive()
174    }
175}
