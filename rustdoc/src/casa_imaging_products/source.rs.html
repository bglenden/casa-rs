3//! Direct scientific inputs for continuum products.
4//!
5//! The inputs borrow one whole Major-Cycle completion, so the final normal
6//! state and model they carry come from the same reconciliation.
7
8use std::borrow::Cow;
9use std::collections::BTreeMap;
10
11use casa_imaging_model::{CompiledProblem, ImageDomainRole, ReconstructionBasis};
12use casa_imaging_reconstruction::{
13    FinalNormalState, ImageDomainReconstructionMasks, MajorCycleCompletion, ModelGeneration,
14    NormalStateCatalog, ReconstructionMask,
15};
16
17/// Where an analytic primary beam is centred on each image domain: the
18/// pixel position `[x, y]` of every selected field's phase direction.
19///
20/// CASA centres one beam on each field's `PHASE_DIR` and sums them on every
21/// image, outliers included (`SynthesisImagerVi2::makePrimaryBeam`,
22/// `SIMapper::addPB`).
23pub type PrimaryBeamCentres = BTreeMap<ImageDomainRole, Vec<[f64; 2]>>;
24
25/// Borrowed scientific payloads for producing planned members.
26///
27/// This value borrows one whole completion and its compiled problem, so the
28/// final normal state and model it carries come from the same reconciliation.
29/// Binding them checks once that they have the problem's shape; product
30/// generation then indexes them without checking again.
31#[derive(Debug)]
32pub struct ContinuumProductInputs<'a> {
33    problem: &'a CompiledProblem,
34    normal_state: &'a FinalNormalState,
35    final_model: &'a ModelGeneration,
36    reconstruction_mask: Option<&'a ReconstructionMask>,
37    domain_reconstruction_masks: Option<&'a ImageDomainReconstructionMasks>,
38    primary_beam_centres: Option<&'a PrimaryBeamCentres>,
39    facet_sum_weights: Option<&'a FacetSumWeights>,
40}
41
42/// Each faceted image domain's facets' own sums of weights, by domain
43/// ordinal: per facet in CASA's order (facet `row·N + column`, facet 0 at
44/// the bottom left), per polarization. CASA's `.sumwt` of a faceted image
45/// holds each facet store's own sum (`SIImageStore::getSubImageStore`).
46pub type FacetSumWeights = std::collections::BTreeMap<usize, Vec<Vec<f64>>>;
47
48impl<'a> ContinuumProductInputs<'a> {
49    /// Bind the completion of a run of `problem`.
50    ///
51    /// # Panics
52    ///
53    /// If the completion does not have `problem`'s image domains, their
54    /// shapes, planes and Taylor terms: it came from another problem's run.
55    pub fn from_major_cycle(problem: &'a CompiledProblem, join: &'a MajorCycleCompletion) -> Self {
56        let normal = join.normal_state();
57        let model = join.final_model().shape();
58        let domains = problem.geometry().domains();
59        assert!(
60            normal.domain_count() == domains.len()
61                && model.domains().len() == domains.len()
62                && domains.iter().all(|domain| {
63                    let pixels = domain.shape().pixels();
64                    model
65                        .domain_roles()
66                        .iter()
67                        .position(|role| role == domain.role())
68                        .is_some_and(|ordinal| {
69                            model.domains()[ordinal].pixels() == pixels
70                                && normal.domain_shape(ordinal) == Some(pixels)
71                        })
72                }),
73            "a run's completion has its problem's image domains and their shapes"
74        );
75        match normal.catalog() {
76            NormalStateCatalog::UnnormalizedTaylorBlockV1 => {
77                let ReconstructionBasis::Taylor { terms } = problem.reconstruction().basis() else {
78                    panic!("a Taylor normal state comes from a Taylor-basis problem");
79                };
80                assert!(
81                    terms >= 2
82                        && normal.coefficient_term_count() == terms
83                        && normal.normal_moment_count() == 2 * terms - 1
84                        && model.coefficients() == terms
85                        && domains.len() == 1,
86                    "a Taylor completion has its problem's terms, 2·terms − 1 moments and one domain"
87                );
88            }
89            NormalStateCatalog::UnnormalizedPlaneV1
90            | NormalStateCatalog::UnnormalizedChannelSlabV1 => {
91                let slab = normal.slab();
92                assert!(
93                    slab.core_range() == (0..slab.total_channels())
94                        && normal.channel_count() == slab.total_channels()
95                        && model.coefficients() == normal.channel_count()
96                        && model.polarizations() == normal.polarization_count(),
97                    "a run's final completion covers every output channel and polarization"
98                );
99            }
100        }
101        Self {
102            problem,
103            normal_state: normal,
104            final_model: join.final_model(),
105            reconstruction_mask: None,
106            domain_reconstruction_masks: None,
107            primary_beam_centres: None,
108            facet_sum_weights: None,
109        }
110    }
111
112    /// Bind each faceted domain's facets' own sums of weights, which its
113    /// `.sumwt` publishes one per facet pixel; an unfaceted domain publishes
114    /// its normal state's sum.
115    ///
116    /// # Panics
117    ///
118    /// If an entry is not a faceted domain of the problem with one sum per
119    /// facet and polarization.
120    #[must_use]
121    pub fn with_facet_sum_weights(mut self, sums: &'a FacetSumWeights) -> Self {
122        let domains = self.problem.geometry().domains();
123        let polarizations = self.normal_state.polarization_count();
124        assert!(
125            sums.iter().all(|(ordinal, facets)| {
126                domains.get(*ordinal).is_some_and(|domain| {
127                    domain.facets().len() > 1
128                        && facets.len() == domain.facets().len()
129                        && facets.iter().all(|facet| facet.len() == polarizations)
130                })
131            }),
132            "facet sums of weights cover faceted domains, one per facet and polarization"
133        );
134        self.facet_sum_weights = Some(sums);
135        self
136    }
137
138    /// Domain `ordinal`'s facets' own sums of weights when it is faceted.
139    #[must_use]
140    pub fn facet_sum_weights(&self, ordinal: usize) -> Option<&'a [Vec<f64>]> {
141        self.facet_sum_weights
142            .and_then(|sums| sums.get(&ordinal))
143            .map(Vec::as_slice)
144    }
145
146    /// Bind the exact reconstruction mask used by the final bounded solve.
147    ///
148    /// # Panics
149    ///
150    /// If the mask does not cover the main domain's grid.
151    #[must_use]
152    pub fn with_reconstruction_mask(mut self, mask: &'a ReconstructionMask) -> Self {
153        let main = self.domain(&ImageDomainRole::Main);
154        assert!(
155            mask.shape() == main.shape().pixels() && mask.coordinate() == main.direction(),
156            "the run's reconstruction mask covers its main domain"
157        );
158        self.reconstruction_mask = Some(mask);
159        self.domain_reconstruction_masks = None;
160        self
161    }
162
163    /// Bind the exact spatial support for every canonical image domain.
164    ///
165    /// # Panics
166    ///
167    /// If the masks are not one per domain, each covering its domain.
168    #[must_use]
169    pub fn with_domain_reconstruction_masks(
170        mut self,
171        masks: &'a ImageDomainReconstructionMasks,
172    ) -> Self {
173        let domains = self.problem.geometry().domains();
174        assert!(
175            masks.len() == domains.len()
176                && masks.iter().zip(domains).all(|(mask, domain)| {
177                    mask.shape() == domain.shape().pixels()
178                        && mask.coordinate() == domain.direction()
179                }),
180            "the run's reconstruction masks cover its domains, one each"
181        );
182        self.reconstruction_mask = None;
183        self.domain_reconstruction_masks = Some(masks);
184        self
185    }
186
187    /// Centre an analytic primary beam on the selected fields' pointings.
188    /// Without them it is centred on each domain's reference pixel. A
189    /// domain with no pointing has a zero PB.
190    ///
191    /// # Panics
192    ///
193    /// If a domain of the problem has no entry.
194    #[must_use]
195    pub fn with_primary_beam_centres(mut self, centres: &'a PrimaryBeamCentres) -> Self {
196        assert!(
197            self.problem
198                .geometry()
199                .domains()
200                .iter()
201                .all(|domain| centres.contains_key(domain.role())),
202            "the pointings cover every image domain"
203        );
204        self.primary_beam_centres = Some(centres);
205        self
206    }
207
208    /// The pixel positions an analytic primary beam is centred on in the
209    /// domain `role`, whose reference pixel is `reference_pixel`: borrowed
210    /// from the bound pointings, so a plane holds no copy of them.
211    pub(crate) fn primary_beam_centres(
212        &self,
213        role: &ImageDomainRole,
214        reference_pixel: [f64; 2],
215    ) -> Cow<'a, [[f64; 2]]> {
216        self.primary_beam_centres.map_or_else(
217            || Cow::Owned(vec![reference_pixel]),
218            |centres| Cow::Borrowed(centres[role].as_slice()),
219        )
220    }
221
222    /// Borrow the exact compiled problem behind these payloads.
223    #[must_use]
224    pub const fn problem(&self) -> &CompiledProblem {
225        self.problem
226    }
227
228    fn domain(&self, role: &ImageDomainRole) -> &casa_imaging_model::CompiledImageDomain {
229        self.problem
230            .geometry()
231            .domains()
232            .iter()
233            .find(|domain| domain.role() == role)
234            .expect("a product's domain is one of its problem's domains")
235    }
236
237    /// Return radians-per-pixel on each direction axis of the main domain.
238    #[must_use]
239    pub fn cell_size_rad(&self) -> [f64; 2] {
240        self.cell_size_rad_for_domain(&ImageDomainRole::Main)
241    }
242
243    /// Return radians-per-pixel on each direction axis of one image domain.
244    pub(crate) fn cell_size_rad_for_domain(&self, role: &ImageDomainRole) -> [f64; 2] {
245        let increment = self.domain(role).direction().increment_rad();
246        [increment[0].abs(), increment[1].abs()]
247    }
248
249    /// The ordinal of a domain in the normal state and the model, which
250    /// share the problem's domain order.
251    pub(crate) fn model_domain_ordinal(&self, role: &ImageDomainRole) -> usize {
252        self.final_model
253            .shape()
254            .domain_roles()
255            .iter()
256            .position(|candidate| candidate == role)
257            .expect("a product's domain is one of its problem's domains")
258    }
259
260    /// Borrow the authoritative normal state (residual, PSF, sensitivity).
261    #[must_use]
262    pub const fn normal_state(&self) -> &FinalNormalState {
263        self.normal_state
264    }
265
266    /// Borrow the authoritative final model generation.
267    #[must_use]
268    pub const fn final_model(&self) -> &ModelGeneration {
269        self.final_model
270    }
271
272    /// Borrow the exact CLEAN mask, when a bounded solve supplied one.
273    #[must_use]
274    pub const fn reconstruction_mask(&self) -> Option<&ReconstructionMask> {
275        self.reconstruction_mask
276    }
277
278    /// Borrow canonical per-domain CLEAN masks, when supplied.
279    #[must_use]
280    pub const fn domain_reconstruction_masks(&self) -> Option<&ImageDomainReconstructionMasks> {
281        self.domain_reconstruction_masks
282    }
