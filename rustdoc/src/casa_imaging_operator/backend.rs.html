19/// Model grids ready for degridding: the forward FFT of the corrected,
20/// polarization-expanded model, one full-grid block per
21/// `(plane, grid pol, Taylor term)` in the operator's precision.
22///
23/// The grids are immutable. A device backend keeps its copy of them in the
24/// grids themselves ([`PreparedModelGrids::device_copy`]), made once and
25/// dropped with them.
26pub struct PreparedModelGrids {
27    layout: AccumulatorLayout,
28    storage: GridStorage,
29    device: OnceLock<Box<dyn Any + Send + Sync>>,
30}
31
32impl PreparedModelGrids {
33    pub(crate) fn new(layout: AccumulatorLayout, storage: GridStorage) -> Self {
34        debug_assert_eq!(storage.len(), layout.cells());
35        debug_assert!(layout.is_full_grid());
36        Self {
37            layout,
38            storage,
39            device: OnceLock::new(),
40        }
41    }
42
43    /// A device backend's copy of the grids, made by `copy` from the cells
44    /// the first time any caller asks; later calls, from any thread, return
45    /// the same copy. A failed copy is kept like a successful one, so a
46    /// backend stores a `Result` when copying can fail.
47    ///
48    /// Panics when an earlier caller stored a copy of another type: one
49    /// backend type serves a model.
50    pub fn device_copy<C: Any + Send + Sync>(&self, copy: impl FnOnce(&GridStorage) -> C) -> &C {
51        self.device
52            .get_or_init(|| Box::new(copy(&self.storage)))
53            .downcast_ref()
54            .expect("one device backend type serves a model")
55    }
56
57    /// Layout of the grids (data terms only, full grid).
58    #[must_use]
59    pub const fn layout(&self) -> &AccumulatorLayout {
60        &self.layout
61    }
62
63    /// Cell precision.
64    #[must_use]
65    pub const fn precision(&self) -> GridPrecision {
66        self.storage.precision()
67    }
68
69    /// The cells.
70    #[must_use]
71    pub const fn storage(&self) -> &GridStorage {
72        &self.storage
73    }
74
75    /// Cells of block `(plane_local, pol, term)` at precision `T`.
76    #[must_use]
77    pub fn block<T: GridScalar>(
78        &self,
79        plane_local: usize,
80        pol: usize,
81        term: usize,
82    ) -> &[Complex<T>] {
83        let offset = self.layout.block_offset(plane_local, pol, term);
84        &T::cells(&self.storage)[offset..offset + self.layout.block_cells()]
85    }
86}
87
88impl Clone for PreparedModelGrids {
89    /// The clone holds no device copy.
90    fn clone(&self) -> Self {
91        Self::new(self.layout.clone(), self.storage.clone())
92    }
93}
94
95impl PartialEq for PreparedModelGrids {
96    fn eq(&self, other: &Self) -> bool {
97        self.layout == other.layout && self.storage == other.storage
98    }
99}
100
101impl std::fmt::Debug for PreparedModelGrids {
102    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
103        formatter
104            .debug_struct("PreparedModelGrids")
105            .field("layout", &self.layout)
106            .field("storage", &self.storage)
107            .finish_non_exhaustive()
108    }
109}

111/// What one dispatch does with a sample block.
112pub enum Work<'a> {
113    /// Accumulate the block into `acc` in `mode`.
114    Grid {
115        /// Data, PSF or weight terms.
116        mode: Mode,
117        /// Target accumulator; it must hold `mode`.
118        acc: &'a mut GridAccumulator,
119    },
120    /// Predict raw visibilities (`npol` per placement, sample-major,
121    /// phase-centre phasor applied) from `model`. Reads only the block's
122    /// placements, so its values and weights may be empty.
123    Predict {
124        /// Prepared model grids covering every plane in the block.
125        model: &'a PreparedModelGrids,
126        /// `npol × placements` predicted visibilities.
127        out: &'a mut [Complex32],
128    },
129    /// Predict, subtract from the block and grid the residual data terms in
130    /// one dispatch; optionally hand back the raw residual samples for a
131    /// final pass that writes the corrected or model column.
132    ResidualGrid {
133        /// Prepared model grids covering every plane in the block.
134        model: &'a PreparedModelGrids,
135        /// Target accumulator; it must hold the data terms.
136        acc: &'a mut GridAccumulator,
137        /// `npol × placements` raw residual visibilities, when wanted.
138        residual_out: Option<&'a mut [Complex32]>,
139    },
140    /// Predict the block's placements from `grids`, phase-centre phasor
141    /// applied, as [`Work::Predict`] does, and interpolate the predictions
142    /// to native cells ([`NativeTerm`]): the model of each term's cell is
143    /// written into `model`, or with `add` added to it, so that several
144    /// image domains' predictions sum before the residual is formed
145    /// (`SIMapperCollection::degrid`). Reads only the block's placements.
146    ///
147    /// A device backend may leave the work running when `apply` returns;
148    /// later device work that reads `model` runs after it, and the host
149    /// reads device cells only once the backend has completed it.
150    PredictNative {
151        /// Prepared model grids covering every plane in the block.
152        grids: &'a PreparedModelGrids,
153        /// One term per native cell to form.
154        terms: &'a [NativeTerm],
155        /// Where the model at native cells goes.
156        model: NativeTarget<'a>,
157        /// Add to the cells instead of writing them.
158        add: bool,
159    },
160    /// Grid the residual at native channels into the data terms of `acc`:
161    /// each placement's values (`W·V·e^{iφ}` interpolated between its native
162    /// endpoints) less `scale · Σ factors · M[cells]` of its endpoints
163    /// ([`SampleBlock::sources`], [`NativeSource`]) in `model`.
164    NativeResidualGrid {
165        /// The model at native cells, from [`Work::PredictNative`].
166        model: NativeModel<'a>,
167        /// Target accumulator; it must hold the data terms.
168        acc: &'a mut GridAccumulator,
169    },
170}

172/// A gridding backend: CPU or device. One implementation of the kernel
173/// contract in plan section 5.3, selected by the runtime.
174pub trait GridBackend: Send {
175    /// Apply `work` to one block with kernel set `cf`; the work is complete
176    /// when this returns, except that a device backend may leave grid-only
177    /// work running on device cells, which the host's next access to the
178    /// cells waits for ([`DeviceCells::defer`](crate::DeviceCells::defer)).
179    ///
180    /// Every placement must lie on a plane the target holds and have its
181    /// whole support inside the target's tile; the block's polarization
182    /// count must equal the kernel set's visibility polarizations.
183    fn apply(
184        &mut self,
185        block: &SampleBlock<'_>,
186        cf: &dyn ConvolutionFunctionSet,
187        work: Work<'_>,
188    ) -> Result<(), OperatorError>;
189}
