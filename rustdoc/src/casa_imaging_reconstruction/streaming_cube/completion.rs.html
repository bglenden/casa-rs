11impl SpectralOperatorSpecification {
12    pub(crate) fn cube_geometry(&self) -> Result<SpectralOperatorGeometry, SpectralOperatorError> {
13        if self.basis != SpectralBasisPlan::ChannelLocal
14            || self.domains.len() != 1
15            || self.charts.len() != 1
16            || self.charts[0].window.origin() != [0, 0]
17            || self.charts[0].geometry.image_shape != self.image_shape
18            || self.polarization_coordinates.as_ref() != [PolarizationCoordinate::StokesI]
19            || self.w_projection.is_some()
20            || self.aw_projection.is_some()
21            || self.instrument_model.is_some()
22            || self.mosaic
23            || self.spectral_kernel != SpectralKernel::Linear
24        {
25            return Err(SpectralOperatorError::UnsupportedProblem);
26        }
27        Ok(self.charts[0].geometry)
28    }
29}

31impl SpectralOperatorPrimitives {
32    #[cfg(test)]
33    pub(crate) fn cube_owned_bytes(
34        &self,
35        include_residual: bool,
36    ) -> Result<usize, SpectralOperatorError> {
37        let fields = [
38            if include_residual {
39                size_of_val(self.dirty.as_ref())
40            } else {
41                0
42            },
43            self.invariant_dirty.as_deref().map_or(0, size_of_val),
44            size_of_val(self.psf.as_ref()),
45            size_of_val(self.sensitivity.as_ref()),
46            size_of_val(self.sum_weights.as_ref()),
47            size_of_val(self.published_sum_weights.as_ref()),
48            size_of_val(self.validity.as_ref()),
49            size_of_val(self.joint_line_term_by_channel.as_ref()),
50        ];
51        fields
52            .into_iter()
53            .try_fold(size_of::<Self>(), |sum, bytes| {
54                sum.checked_add(bytes)
55                    .ok_or(SpectralOperatorError::ResidencyOverflow)
56            })
57    }
58
59    pub(crate) fn validate_cube_layout(
60        &self,
61        shape: [usize; 2],
62        core: std::ops::Range<usize>,
63        total_channels: usize,
64    ) -> Result<(), SpectralOperatorError> {
65        let values = checked_cells(shape)?
66            .checked_mul(core.len())
67            .ok_or(SpectralOperatorError::ResidencyOverflow)?;
68        if self.shape != shape
69            || core.is_empty()
70            || core.end > total_channels
71            || self.slab.core_range() != core
72            || self.slab.total_channels() != total_channels
73            || self.basis != SpectralBasisPlan::ChannelLocal
74            || self.polarizations != 1
75            || !self.major_cycle_residual_promoted
76            || self.residual_model.is_none()
77            || self.dirty.len() != values
78            || self.psf.len() != values
79            || self.sensitivity.len() != values
80            || self.sum_weights.len() != core.len()
81            || self.published_sum_weights.len() != core.len()
82            || self.validity.len() != core.len()
83            || self
84                .invariant_dirty
85                .as_ref()
86                .is_some_and(|v| v.len() != values)
87            || self.major_cycle_residual.is_some()
88            || self.primary_beam_weighted_sum.is_some()
89            || self.common_residual.is_some()
90            || self.invariant_common_dirty.is_some()
91            || !self.channel_sum_weights.is_empty()
92            || self.joint_line_term_by_channel.len() != total_channels
93            || self.joint_line_term_by_channel.iter().any(Option::is_some)
94        {
95            return Err(SpectralOperatorError::ReusableNormalStateMismatch);
96        }
97        Ok(())
98    }

100    pub(crate) fn from_cube_band(
101        images: BandImages,
102        total_channels: usize,
103        model: ModelGenerationId,
104    ) -> Result<Self, SpectralOperatorError> {
105        let BandImages {
106            phase,
107            shape,
108            core,
109            dirty,
110            residual,
111            psf,
112            sum_weight,
113            mapped,
114        } = images;
115        let cells = checked_cells(shape)?;
116        let values = cells
117            .checked_mul(core.len())
118            .ok_or(SpectralOperatorError::ResidencyOverflow)?;
119        if phase == BandPhase::Residual
120            || core.is_empty()
121            || core.end > total_channels
122            || dirty.len() != values
123            || residual.len() != if phase == BandPhase::Full { values } else { 0 }
124            || psf.len() != values
125            || sum_weight.len() != core.len()
126            || mapped.len() != core.len()
127        {
128            return Err(SpectralOperatorError::InvalidSlab);
129        }
130        if sum_weight
131            .iter()
132            .any(|weight| !weight.is_finite() || *weight < 0.0)
133        {
134            return Err(SpectralOperatorError::GeneratedNonfinite);
135        }
136        let mut sensitivity = Vec::with_capacity(values);
137        for &weight in &sum_weight {
138            sensitivity.extend(std::iter::repeat_n(weight, cells));
139        }
140        let sensitivity = sensitivity.into_boxed_slice();
141        let validity = mapped
142            .iter()
143            .zip(&sum_weight)
144            .map(|(&mapped, &weight)| validity_from_support(mapped, weight))
145            .collect();
146        let (dirty, invariant_dirty) = if phase == BandPhase::InitialZero {
147            // A direct-native refresh does not consume the old data-side dirty
148            // image. The initial residual is already dirty, so move it once.
149            (dirty.into_boxed_slice(), None)
150        } else {
151            (residual.into_boxed_slice(), Some(dirty.into_boxed_slice()))
152        };
153        Ok(Self {
154            shape,
155            slab: SpectralSlabPlan {
156                total_channels,
157                core_start: core.start,
158                core_end: core.end,
159                resident_start: core.start,
160                resident_end: core.end,
161            },
162            basis: SpectralBasisPlan::ChannelLocal,
163            polarizations: 1,
164            joint_line_term_by_channel: vec![None; total_channels].into(),
165            // The controller consumes the exact residual. Retain the data-side
166            // dirty image by moving it, avoiding the old clone-then-promotion copy.
167            dirty,
168            invariant_dirty,
169            common_residual: None,
170            invariant_common_dirty: None,
171            psf: psf.into_boxed_slice(),
172            sensitivity,
173            primary_beam_weighted_sum: None,
174            published_sum_weights: sum_weight.clone().into_boxed_slice(),
175            sum_weights: sum_weight.into_boxed_slice(),
176            channel_sum_weights: Box::new([]),
177            validity,
178            major_cycle_residual: None,
179            major_cycle_residual_promoted: true,
180            residual_model: Some(model),
181            #[cfg(test)]
182            measurements: SpectralOperatorMeasurements::default(),
183        })
184    }
185}

187impl CompleteDataOwnerResult {
188    /// Transfer a completed native band into the existing normal-state fold.
189    ///
190    /// The runtime must first finish its native-input writer and authoritative
191    /// weighted traversal, then match the problem/attempt/node/lease and frozen
192    /// source association. This binds that terminal replay to an owned numerical
193    /// result; it does not manufacture source completion or inspect image values.
194    #[doc(hidden)]
195    pub fn from_streaming_cube(
196        specification: &SpectralOperatorSpecification,
197        primitives: SpectralOperatorPrimitives,
198        replay: &WeightingReplaySummary,
199        selected_generation: SelectedObservationGenerationId,
200        continuum_transform_generation: Option<ContinuumTransformGenerationId>,
201    ) -> Result<Self, SpectralOperatorError> {
202        let geometry = specification.cube_geometry()?;
203        primitives.validate_cube_layout(
204            geometry.image_shape,
205            specification.slab.core_range(),
206            specification.slab.total_channels(),
207        )?;
208        let domains =
209            combine_initial_chart_primitives(specification, std::iter::once(Ok(primitives)))?;
210        Ok(Self {
211            domains,
212            completion: CompleteDataOwnerCompletion::from_streaming_cube(
213                specification,
214                replay,
215                selected_generation,
216                continuum_transform_generation,
217            )?,
218        })
219    }
220}

222impl CompleteDataOwnerCompletion {
223    pub(crate) fn from_streaming_cube(
224        specification: &SpectralOperatorSpecification,
225        replay: &WeightingReplaySummary,
226        selected_generation: SelectedObservationGenerationId,
227        continuum_transform_generation: Option<ContinuumTransformGenerationId>,
228    ) -> Result<Self, SpectralOperatorError> {
229        specification.cube_geometry()?;
230        if replay.sample_count() == 0 || replay.block_count() == 0 {
231            return Err(SpectralOperatorError::IncompleteCoverage);
232        }
233        Ok(Self {
234            problem: specification.problem,
235            geometry: specification.geometry,
236            numerics: specification.numerics,
237            weighting_commitment: specification.weighting_commitment,
238            weighting_generation: replay.weighting_generation(),
239            replay: replay.replay_id(),
240            coverage: replay.coverage(),
241            // Reuse terminal source coverage; no second encoding pass.
242            coverage_proof_bytes: 0,
243            coverage_proof_hash_calls: 0,
244            primitives: SpectralPrimitiveCatalog::UnnormalizedChannelSlabV1,
245            selected_generation,
246            continuum_transform_generation,
247            sample_count: replay.sample_count(),
248            block_count: replay.block_count(),
249        })
250    }
251}

253#[test]
254fn cube_phase_handoff_moves_initial_dirty_and_preserves_explicit_invariants() {
255    let model = ModelGenerationId(LogicalIdentity::from_sha256([47; 32]));
256    let make = |phase| BandImages {
257        phase,
258        shape: [2, 1],
259        core: 0..1,
260        dirty: vec![Complex64::new(3.0, 0.0); 2],
261        residual: if phase == BandPhase::Full {
262            vec![Complex64::new(1.0, 0.0); 2]
263        } else {
264            Vec::new()
265        },
266        psf: vec![Complex64::new(2.0, 0.0); 2],
267        sum_weight: vec![2.0],
268        mapped: vec![3],
269    };
270    let images = make(BandPhase::InitialZero);
271    let pointer = images.dirty.as_ptr();
272    let initial = SpectralOperatorPrimitives::from_cube_band(images, 1, model).unwrap();
273    assert_eq!(initial.dirty.as_ptr(), pointer);
274    assert!(initial.invariant_dirty.is_none());
275    assert!(initial.major_cycle_residual.is_none());
276    let images = make(BandPhase::Full);
277    let dirty_pointer = images.dirty.as_ptr();
278    let residual_pointer = images.residual.as_ptr();
279    let full = SpectralOperatorPrimitives::from_cube_band(images, 1, model).unwrap();
280    assert_eq!(full.dirty.as_ptr(), residual_pointer);
281    assert_eq!(
282        full.invariant_dirty.as_ref().unwrap().as_ptr(),
283        dirty_pointer
284    );
285    assert_eq!(
286        full.invariant_dirty.as_deref().unwrap(),
287        &[Complex64::new(3.0, 0.0); 2]
288    );
289    assert!(full.validate_cube_layout([2, 1], 0..1, 2).is_err());
290}
