19/// Structured imaging run report consumed by the `casars` workflow shell.
20#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
21pub struct ManagedImagingOutput {
22    /// High-level request summary used by the shell overview.
23    pub request: ManagedImagingRequest,
24    /// Structured run metrics and per-channel diagnostics.
25    pub run: ManagedImagingRun,
26    /// Expected CASA image products written under the configured prefix.
27    pub artifacts: Vec<ManagedImagingArtifact>,
28}
29
30/// Launcher-facing view of the requested imaging configuration.
31#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
32pub struct ManagedImagingRequest {
33    /// MeasurementSet path supplied for the run.
34    pub measurement_set: String,
35    /// Output CASA image prefix.
36    pub imagename: String,
37    /// Spectral imaging mode.
38    pub spectral_mode: String,
39    /// Requested visibility weighting.
40    pub weighting: String,
41    /// Requested minor-cycle deconvolver.
42    pub deconvolver: String,
43    /// Effective visibility-gridder family.
44    pub gridder: String,
45    /// Hogbom minor-cycle iteration accounting policy.
46    pub hogbom_iteration_mode: String,
47    /// Requested `w`-term handling mode.
48    pub w_term_mode: String,
49    /// Optional data-column override.
50    pub data_column: Option<String>,
51    /// Requested model persistence mode.
52    pub save_model: String,
53    /// Image size in pixels.
54    pub imsize: usize,
55    /// Cell size in arcseconds.
56    pub cell_arcsec: f64,
57    /// Image direction-coordinate projection.
58    pub projection: String,
59    /// Whether the run skipped CLEAN.
60    pub dirty_only: bool,
61    /// Whether preview PNG sidecars were requested.
62    pub write_preview_pngs: bool,
63    /// Whether the primary-beam product was requested.
64    pub write_pb: bool,
65    /// Whether per-channel density estimation was requested for cube weighting.
66    pub per_channel_weight_density: bool,
67    /// Requested MTMFS Taylor-term count.
68    pub nterms: usize,
69    /// Output channel count for cube-like runs.
70    pub output_channels: usize,
71    /// Requested raw-correlation or Stokes plane, when explicitly selected.
72    pub correlation: Option<String>,
73    /// Requested restoring-beam mode for restored products.
74    pub restoring_beam_mode: String,
75}

77/// Structured run metrics emitted after one successful imaging run.
78#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
79pub struct ManagedImagingRun {
80    /// Warnings emitted by the imaging run.
81    pub warnings: Vec<String>,
82    /// Number of scalar samples that reached the gridder.
83    pub gridded_samples: usize,
84    /// Total major-cycle count reported by the run.
85    pub major_cycles: usize,
86    /// Total minor-cycle component updates reported by the run.
87    pub minor_iterations: usize,
88    /// Final CLEAN stop reason when deconvolution ran.
89    pub clean_stop_reason: Option<String>,
90    /// Measured end-to-end application wall time.
91    pub elapsed_ns: u64,
92}

94/// One expected output artifact written by the imaging run.
95#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
96pub struct ManagedImagingArtifact {
97    /// Stable artifact kind identifier such as `psf`, `image`, or `alpha`.
98    pub kind: String,
99    /// Human-readable artifact label.
100    pub label: String,
101    /// On-disk path for the CASA image product.
102    pub path: String,
103    /// Whether that product exists after the run.
104    pub exists: bool,
105    /// Optional preview sidecar path.
106    pub preview_png_path: Option<String>,
107    /// Whether the preview sidecar exists after the run.
108    pub preview_png_exists: bool,
109}

111impl ManagedImagingOutput {
112    /// Build the structured launcher report from one completed run.
113    pub fn from_run(config: &CliConfig, summary: &RunSummary) -> Self {
114        Self {
115            request: ManagedImagingRequest {
116                measurement_set: config.ms.display().to_string(),
117                imagename: config.imagename.display().to_string(),
118                spectral_mode: canonical_spectral_mode_name(config.spectral_mode).to_string(),
119                weighting: canonical_weighting_name(config.weighting),
120                deconvolver: canonical_deconvolver_name(config.deconvolver).to_string(),
121                gridder: managed_gridder_from_config(config).to_string(),
122                hogbom_iteration_mode: canonical_hogbom_iteration_mode_name(
123                    config.hogbom_iteration_mode,
124                )
125                .to_string(),
126                w_term_mode: if config.aw_project.is_some() {
127                    "awproject".to_string()
128                } else {
129                    canonical_w_term_mode_name(config.w_term_mode).to_string()
130                },
131                data_column: config.datacolumn.clone(),
132                save_model: match config.save_model {
133                    crate::SaveModelMode::None => "none",
134                    crate::SaveModelMode::ModelColumn => "modelcolumn",
135                }
136                .to_string(),
137                imsize: config.imsize,
138                cell_arcsec: config.cell_arcsec,
139                projection: "SIN".to_string(),
140                dirty_only: config.dirty_only,
141                write_preview_pngs: config.write_preview_pngs,
142                write_pb: config.write_pb,
143                per_channel_weight_density: config.per_channel_weight_density,
144                nterms: config.nterms,
145                output_channels: 1,
146                correlation: config.correlation.clone(),
147                restoring_beam_mode: canonical_restoring_beam_mode_name(config.restoring_beam_mode)
148                    .to_string(),
149            },
150            run: ManagedImagingRun {
151                warnings: summary.warnings.clone(),
152                gridded_samples: summary.gridded_samples,
153                major_cycles: summary.major_cycles,
154                minor_iterations: summary.minor_iterations,
155                clean_stop_reason: summary
156                    .clean_stop_reason
157                    .map(|reason| format!("{reason:?}")),
158                elapsed_ns: summary.elapsed.as_nanos() as u64,
159            },
160            artifacts: imaging_artifacts(config, &summary.output_products),
161        }
162    }

164    /// Build the launcher report from the canonical task result.
165    pub fn from_task_result(result: &ImagerRunTaskResult) -> Self {
166        let request = &result.request;
167        Self {
168            request: ManagedImagingRequest {
169                measurement_set: request.measurement_set.display().to_string(),
170                imagename: request.image_name.display().to_string(),
171                spectral_mode: match request.spectral_mode {
172                    ImagerSpectralMode::Mfs => "mfs".to_string(),
173                    ImagerSpectralMode::Cube => "cube".to_string(),
174                    ImagerSpectralMode::Cubedata => "cubedata".to_string(),
175                },
176                weighting: match &request.weighting {
177                    ImagerWeighting::Natural => "natural".to_string(),
178                    ImagerWeighting::Uniform => "uniform".to_string(),
179                    ImagerWeighting::Briggs { robust } => format!("briggs:{robust}"),
180                    ImagerWeighting::BriggsBwTaper { robust } => {
181                        format!("briggsbwtaper:{robust}")
182                    }
183                },
184                deconvolver: match request.deconvolver {
185                    ImagerDeconvolver::Hogbom => "hogbom".to_string(),
186                    ImagerDeconvolver::Mtmfs => "mtmfs".to_string(),
187                    ImagerDeconvolver::Clark => "clark".to_string(),
188                    ImagerDeconvolver::Multiscale => "multiscale".to_string(),
189                },
190                gridder: managed_gridder_from_request(request).to_string(),
191                hogbom_iteration_mode: match request.hogbom_iteration_mode {
192                    ImagerHogbomIterationMode::Strict => "strict".to_string(),
193                    ImagerHogbomIterationMode::CasaInclusive => "casa".to_string(),
194                },
195                w_term_mode: if request.aw_project.is_some() {
196                    "awproject".to_string()
197                } else {
198                    match request.w_term_mode {
199                        ImagerWTermMode::None => "none".to_string(),
200                        ImagerWTermMode::Direct => "direct".to_string(),
201                        ImagerWTermMode::Wproject => "wproject".to_string(),
202                    }
203                },
204                data_column: request.data_column.clone(),
205                save_model: match request.save_model {
206                    ImagerSaveModel::None => "none",
207                    ImagerSaveModel::ModelColumn => "modelcolumn",
208                }
209                .to_string(),
210                imsize: request.image_size,
211                cell_arcsec: request.cell_arcsec,
212                projection: request.projection.as_cli_text().to_string(),
213                dirty_only: request.dirty_only,
214                write_preview_pngs: request.write_preview_pngs,
215                write_pb: request.write_pb,
216                per_channel_weight_density: managed_request_per_channel_weight_density(request),
217                nterms: request.nterms,
218                output_channels: 1,
219                correlation: request
220                    .correlation
221                    .map(|value| value.as_cli_text().to_string()),
222                restoring_beam_mode: match request.restoring_beam_mode {
223                    ImagerRestoringBeamMode::PerPlane => "per_plane".to_string(),
224                    ImagerRestoringBeamMode::Common => "common".to_string(),
225                },
226            },
227            run: ManagedImagingRun {
228                warnings: result.run.warnings.clone(),
229                gridded_samples: result.run.gridded_samples,
230                major_cycles: result.run.major_cycles,
231                minor_iterations: result.run.minor_iterations,
232                clean_stop_reason: result
233                    .run
234                    .clean_stop_reason
235                    .map(|reason| format!("{reason:?}")),
236                elapsed_ns: result.run.elapsed_ns,
237            },
238            artifacts: result
239                .artifacts
240                .iter()
241                .map(|artifact| ManagedImagingArtifact {
242                    kind: match artifact.kind {
243                        ImagerArtifactKind::Psf => "psf".to_string(),
244                        ImagerArtifactKind::Residual => "residual".to_string(),
245                        ImagerArtifactKind::Model => "model".to_string(),
246                        ImagerArtifactKind::Image => "image".to_string(),
247                        ImagerArtifactKind::Mask => "mask".to_string(),
248                        ImagerArtifactKind::Weight => "weight".to_string(),
249                        ImagerArtifactKind::Sumwt => "sumwt".to_string(),
250                        ImagerArtifactKind::PrimaryBeam => "pb".to_string(),
251                        ImagerArtifactKind::ImagePbcor => "image.pbcor".to_string(),
252                        ImagerArtifactKind::Alpha => "alpha".to_string(),
253                        ImagerArtifactKind::AlphaError => "alpha.error".to_string(),
254                    },
255                    label: artifact.label.clone(),
256                    path: artifact.path.clone(),
257                    exists: artifact.exists,
258                    preview_png_path: artifact.preview_png_path.clone(),
259                    preview_png_exists: artifact.preview_png_exists,
260                })
261                .collect(),
262        }
263    }

266fn managed_request_per_channel_weight_density(
267    request: &crate::task_contract::ImagerRunTaskRequest,
268) -> bool {
269    request
270        .per_channel_weight_density
271        .unwrap_or(matches!(request.spectral_mode, ImagerSpectralMode::Cube))
272}

274fn managed_gridder_from_config(config: &CliConfig) -> &'static str {
275    if config.aw_project.is_some() {
276        "awproject"
277    } else if config.force_standard_gridder {
278        "standard"
279    } else if matches!(
280        config.w_term_mode,
281        crate::WTermMode::WProject | crate::WTermMode::Direct
282    ) {
283        if matches!(config.w_term_mode, crate::WTermMode::Direct) {
284            "widefield"
285        } else {
286            "wproject"
287        }
288    } else if config.use_pointing
289        || config
290            .field_ids
291            .as_ref()
292            .is_some_and(|field_ids| field_ids.len() > 1)
293    {
294        "mosaic"
295    } else {
296        "standard"
297    }
298}
299
300fn managed_gridder_from_request(
301    request: &crate::task_contract::ImagerRunTaskRequest,
302) -> &'static str {
303    if request.aw_project.is_some() {
304        "awproject"
305    } else if matches!(request.w_term_mode, ImagerWTermMode::Wproject) {
306        "wproject"
307    } else if request.use_pointing
308        || request
309            .field_ids
310            .as_ref()
311            .is_some_and(|field_ids| field_ids.len() > 1)
312    {
313        "mosaic"
314    } else {
315        "standard"
316    }
317}

319fn imaging_artifacts(config: &CliConfig, products: &[String]) -> Vec<ManagedImagingArtifact> {
320    let request = crate::task_contract::ImagerRunTaskRequest::from_cli_config(config);
321    super::task_contract::build_artifacts_for_products(&request, products)
322        .into_iter()
323        .map(|artifact| ManagedImagingArtifact {
324            kind: artifact.kind.as_suffix().to_string(),
325            label: artifact.label,
326            path: artifact.path,
327            exists: artifact.exists,
328            preview_png_path: artifact.preview_png_path,
329            preview_png_exists: artifact.preview_png_exists,
330        })
331        .collect()
332}
