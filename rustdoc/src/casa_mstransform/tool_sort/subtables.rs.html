13pub(super) fn check_named_output(
14    input: &Path,
15    output: &Path,
16    replace_keyword: Option<(&Path, &str)>,
17) -> MsResult<PathBuf> {
18    let mut staging_parent = output
19        .parent()
20        .ok_or_else(|| invalid(output, "output has no parent"))?
21        .to_path_buf();
22    let replacement = replace_keyword.map(|(owner, name)| (output, owner, name));
23    visit_source_tables(input, replacement, |path, subtable| {
24        if path.starts_with(output) {
25            return Err(invalid(output, "output overlaps a source table"));
26        }
27        if subtable && staging_parent.starts_with(path) {
28            if replace_keyword.is_none() {
29                return Err(invalid(output, "output overlaps a source table"));
30            }
31            staging_parent = path
32                .parent()
33                .ok_or_else(|| invalid(output, "no staging parent outside the source table"))?
34                .to_path_buf();
35        }
36        Ok(())
37    })?;
38    Ok(staging_parent)
39}

41pub(super) fn visit_source_tables(
42    input: &Path,
43    replacement: Option<(&Path, &Path, &str)>,
44    mut visit: impl FnMut(&Path, bool) -> MsResult<()>,
45) -> MsResult<()> {
46    let root = reference_root(input)?;
47    let mut visited = BTreeSet::new();
48    let mut pending = vec![(input.to_path_buf(), false)];
49    let standard = SubtableId::ALL_REQUIRED
50        .iter()
51        .chain(SubtableId::ALL_OPTIONAL);
52    while let Some((path, subtable)) = pending.pop() {
53        let path = path.canonicalize().map_err(|error| invalid(&path, error))?;
54        visit(&path, subtable && path != root)?;
55        if !visited.insert(path.clone()) {
56            continue;
57        }
58        match Table::immediate_part_names(&path, &path)? {
59            TablePartNames::Reference(parent) => pending.push((parent, subtable)),
60            TablePartNames::Concatenation { parts, .. } => {
61                pending.extend(parts.into_iter().map(|part| (part, subtable)));
62            }
63            TablePartNames::Plain(_) => {}
64        }
65        let table = Table::open(TableOptions::new(&path))?;
66        for field in table.keywords().fields() {
67            let standard = standard.clone().any(|id| id.name() == field.name);
68            let link = match &field.value {
69                Value::TableRef(link) if field.name != "SORTED_TABLE" => Some(link),
70                Value::Scalar(ScalarValue::String(link)) if standard => Some(link),
71                Value::Record(nested) => {
72                    protect_keyword_links(nested, &path, &mut pending);
73                    None
74                }
75                _ => None,
76            };
77            if let Some(link) = link {
78                let linked = add_directory(link, &path);
79                // The designated result keyword is republished with the new table.
80                // Protect standard MS subtables and all other names for that table.
81                if standard
82                    || !matches!(replacement,
83                    Some((output, owner, name)) if linked == output && name == field.name
84                        && Table::keyword_descriptor_path(&path)? == owner)
85                {
86                    pending.push((linked, true));
87                }
88            }
89        }
90        for column in table
91            .schema()
92            .into_iter()
93            .flat_map(|schema| schema.columns())
94        {
95            if let Some(keys) = table.column_keywords(column.name()) {
96                protect_keyword_links(keys, &path, &mut pending);
97            }
98        }
99    }
100    Ok(())
101}
102
103fn protect_keyword_links(record: &RecordValue, path: &Path, pending: &mut Vec<(PathBuf, bool)>) {
104    for field in record.fields() {
105        match &field.value {
106            Value::TableRef(link) => pending.push((add_directory(link, path), true)),
107            Value::Record(nested) => protect_keyword_links(nested, path, pending),
108            _ => {}
109        }
110    }
111}

113pub(super) fn copy_links(
114    input: &Table,
115    keywords: &mut RecordValue,
116    copy: &mut Table,
117    output: &Path,
118    held: &BTreeSet<PathBuf>,
119) -> MsResult<()> {
120    let input_path = table_path(input)?;
121    for id in SubtableId::ALL_REQUIRED
122        .iter()
123        .chain(SubtableId::ALL_OPTIONAL)
124    {
125        if let Some(Value::Scalar(ScalarValue::String(reference))) = keywords.get(id.name()) {
126            keywords.upsert(id.name(), Value::TableRef(reference.clone()));
127        }
128    }
129    let mut copies = Copies {
130        by_output: BTreeMap::new(),
131        copying: Vec::new(),
132        held,
133    };
134    copies.rewrite(keywords, input_path, output)?;
135    copies.rewrite_columns(input, copy, input_path, output)
136}

138/// Copy the remaining table and column links after the standard subtables
139/// have been installed under their MeasurementSet names.
140pub(crate) fn copy_additional_links(
141    input: &Path,
142    copy: &mut Table,
143    output: &Path,
144    by_output: BTreeMap<PathBuf, PathBuf>,
145) -> MsResult<()> {
146    let mut copies = Copies {
147        by_output,
148        copying: Vec::new(),
149        held: &BTreeSet::new(),
150    };
151    let standard = SubtableId::ALL_REQUIRED
152        .iter()
153        .chain(SubtableId::ALL_OPTIONAL);
154    let mut extra = RecordValue::new(
155        copy.keywords()
156            .fields()
157            .iter()
158            .filter(|field| {
159                matches!(&field.value, Value::TableRef(_))
160                    && !standard.clone().any(|id| id.name() == field.name)
161            })
162            .cloned()
163            .collect(),
164    );
165    for field in extra.fields() {
166        copy.keywords_mut().remove(&field.name);
167    }
168    copies.rewrite(&mut extra, input, output)?;
169    for field in extra.fields() {
170        copy.keywords_mut().upsert(&field.name, field.value.clone());
171    }
172    let column_links: Vec<_> = copy
173        .schema()
174        .into_iter()
175        .flat_map(|schema| schema.columns())
176        .filter_map(|column| {
177            let name = column.name();
178            let record = copy.column_keywords(name)?;
179            record
180                .fields()
181                .iter()
182                .any(|field| matches!(&field.value, Value::TableRef(_)))
183                .then(|| (name.to_owned(), record.clone()))
184        })
185        .collect();
186    for (name, mut record) in column_links {
187        copies.rewrite(&mut record, input, output)?;
188        copy.set_column_keywords(&name, record);
189    }
190    Ok(())
191}

193/// Keep the donor's ordinary keywords and carry over independently copied
194/// table links from the reconciled attached MS.
195pub(crate) fn copy_concatenation_links(
196    attached: &Table,
197    donor: &Table,
198    output_table: &mut Table,
199    output: &Path,
200    held: &BTreeSet<PathBuf>,
201) -> MsResult<()> {
202    let attached_path = table_path(attached)?;
203    let donor_path = table_path(donor)?;
204    let mut copies = Copies {
205        by_output: BTreeMap::new(),
206        copying: Vec::new(),
207        held,
208    };
209    let mut keywords = standard_link_keywords(donor, donor_path);
210    keywords.remove("SORTED_TABLE");
211    let attached_keywords = standard_link_keywords(attached, attached_path);
212    overlay_copied_links(
213        &mut copies,
214        &mut keywords,
215        &attached_keywords,
216        attached_path,
217        donor_path,
218        output,
219    )?;
220    *output_table.keywords_mut() = keywords;
221    for column in donor
222        .schema()
223        .into_iter()
224        .flat_map(|schema| schema.columns())
225    {
226        let name = column.name();
227        let mut keywords = donor.column_keywords(name).cloned().unwrap_or_default();
228        if let Some(attached_keywords) = attached.column_keywords(name) {
229            overlay_copied_links(
230                &mut copies,
231                &mut keywords,
232                attached_keywords,
233                attached_path,
234                donor_path,
235                output,
236            )?;
237        } else {
238            anchor_links(&mut keywords, donor_path);
239        }
240        output_table.set_column_keywords(name, keywords);
241    }
242    Ok(())
243}

245fn standard_link_keywords(table: &Table, root: &Path) -> RecordValue {
246    let mut keywords = table.keywords().clone();
247    for &id in SubtableId::ALL_REQUIRED
248        .iter()
249        .chain(SubtableId::ALL_OPTIONAL)
250    {
251        if let Some(path) = subtable_path_from_main(table, root, id) {
252            keywords.upsert(
253                id.name(),
254                Value::TableRef(path.to_string_lossy().into_owned()),
255            );
256        }
257    }
258    keywords
259}

261fn anchor_links(keywords: &mut RecordValue, input: &Path) {
262    for field in keywords.fields_mut() {
263        if let Value::TableRef(link) = &mut field.value {
264            *link = add_directory(link, input).to_string_lossy().into_owned();
265        }
266    }
267}

269fn overlay_copied_links(
270    copies: &mut Copies<'_>,
271    output_keys: &mut RecordValue,
272    attached: &RecordValue,
273    attached_path: &Path,
274    donor_path: &Path,
275    output: &Path,
276) -> MsResult<()> {
277    anchor_links(output_keys, donor_path);
278    let mut links = RecordValue::new(
279        attached
280            .fields()
281            .iter()
282            .filter(|field| matches!(&field.value, Value::TableRef(_)))
283            .cloned()
284            .collect(),
285    );
286    for field in links.fields() {
287        output_keys.remove(&field.name);
288    }
289    copies.rewrite(&mut links, attached_path, output)?;
290    for field in links.fields() {
291        output_keys.upsert(&field.name, field.value.clone());
292    }
293    Ok(())
294}

296struct Copies<'a> {
297    by_output: BTreeMap<PathBuf, PathBuf>,
298    copying: Vec<PathBuf>,
299    held: &'a BTreeSet<PathBuf>,
300}
301
302impl Copies<'_> {
303    fn rewrite(&mut self, record: &mut RecordValue, input: &Path, output: &Path) -> MsResult<()> {
304        let root = reference_root(input)?;
305        let mut omitted = Vec::new();
306        for field in record.fields_mut() {
307            let Value::TableRef(reference) = &mut field.value else {
308                continue;
309            };
310            let declared = add_directory(reference, input);
311            let source = declared
312                .canonicalize()
313                .map_err(|error| invalid(&declared, error))?;
314            if reference_root(&source)? == root {
315                omitted.push(field.name.clone());
316                continue;
317            }
318            let name = declared
319                .file_name()
320                .ok_or_else(|| invalid(&declared, "linked table has no name"))?;
321            let destination = output.join(name);
322            match self.by_output.get(&destination) {
323                Some(previous) if previous != &source => {
324                    return Err(invalid(
325                        &destination,
326                        "different table links have the same name",
327                    ));
328                }
329                Some(_) => {}
330                None => self.copy(&source, &destination)?,
331            }
332            *reference = strip_directory(&destination, output);
333        }
334        for name in omitted {
335            record.remove(&name);
336        }
337        Ok(())
338    }
339
340    fn rewrite_columns(
341        &mut self,
342        input_table: &Table,
343        copy: &mut Table,
344        input: &Path,
345        output: &Path,
346    ) -> MsResult<()> {
347        for column in input_table
348            .schema()
349            .into_iter()
350            .flat_map(|schema| schema.columns())
351        {
352            let name = column.name();
353            if let Some(record) = input_table.column_keywords(name) {
354                let mut record = record.clone();
355                self.rewrite(&mut record, input, output)?;
356                copy.set_column_keywords(name, record);
357            }
358        }
359        Ok(())
360    }
361
362    fn copy(&mut self, source: &Path, output: &Path) -> MsResult<()> {
363        if self.copying.iter().any(|path| path == source) {
364            return Err(invalid(source, "cyclic table link"));
365        }
366        let table = open_source(source, self.held.contains(source))?;
367        if matches!(
368            Table::immediate_part_names(source, source)?,
369            TablePartNames::Plain(_)
370        ) {
371            // A physical copy retains column metadata and nested table
372            // directories, as casacore does for plain subtable deep copies.
373            std::fs::create_dir_all(output.parent().expect("staging table parent"))
374                .map_err(|error| invalid(output, error))?;
375            casa_tables::publication::copy_locked_table_tree(source, output)
376                .map_err(|error| invalid(source, error))?;
377        } else {
378            self.copying.push(source.to_path_buf());
379            let mut copy = Table::open(TableOptions::new(source))?;
380            let mut keywords = table.keywords().clone();
381            self.rewrite(&mut keywords, source, output)?;
382            *copy.keywords_mut() = keywords;
383            self.rewrite_columns(&table, &mut copy, source, output)?;
384            copy.deep_copy(TableOptions::new(output))?;
385            self.copying.pop();
386        }
387        self.by_output
388            .insert(output.to_path_buf(), source.to_path_buf());
389        Ok(())
390    }
391}

393pub(super) fn table_path(table: &Table) -> MsResult<&Path> {
394    table
395        .path()
396        .ok_or_else(|| MsError::InvalidInput("the sorted source has no table path".into()))
397}

399pub(super) fn reference_root(path: &Path) -> MsResult<PathBuf> {
400    let mut path = path.canonicalize().map_err(|error| invalid(path, error))?;
401    for _ in 0..64 {
402        let TablePartNames::Reference(parent) = Table::immediate_part_names(&path, &path)? else {
403            return Ok(path);
404        };
405        path = parent
406            .canonicalize()
407            .map_err(|error| invalid(&parent, error))?;
408    }
409    Err(invalid(&path, "reference table chain is too deep"))
410}
411
412fn open_source(path: &Path, already_held: bool) -> MsResult<Table> {
413    let options = TableOptions::new(path);
414    #[cfg(unix)]
415    if !already_held {
416        return Ok(Table::open_with_lock(
417            options,
418            casa_tables::LockOptions::new(casa_tables::LockMode::PermanentLockingWait)
419                .read_only(true),
420        )?);
421    }
422    #[cfg(not(unix))]
423    let _ = already_held;
424    Ok(Table::open(options)?)
425}
426
427fn invalid(path: &Path, error: impl std::fmt::Display) -> MsError {
428    MsError::InvalidInput(format!("copy linked table {}: {error}", path.display()))
