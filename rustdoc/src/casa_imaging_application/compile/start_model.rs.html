3//! CASA's `startmodel` (#841 Phase 6b): the image the model starts from
4//! (`SIImageStore::setModelImageOne`). As CASA takes it:
5//! - it is in Jy/pixel (`SynthesisImagerVi2::defineImage` refuses others);
6//! - its masked pixels are zero;
7//! - it is regridded linearly onto the main image's grid, along direction
8//!   and frequency, and scaled by the ratio of the pixel areas, so its flux
9//!   is kept (`regridToModelImage`: `ImageRegrid` with `Interpolate2D::LINEAR`
10//!   over the direction and spectral axes). A channel beyond the model's
11//!   takes its end channel's planes, as casacore's linear interpolation
12//!   writes them into a model without a mask. CASA copies a start model
13//!   already on that grid; casa-rs regrids it too, which changes it only by
14//!   rounding;
15//! - for `deconvolver='mtmfs'` it is a list of at most `nterms` images, the
16//!   k-th the start of Taylor term k, and an empty entry leaves its term
17//!   zero (`SIImageStoreMultiTerm::setModelImage`);
18//! - its direction frame is the image's: casa-rs does not convert frames;
19//! - its Stokes planes are copied by position, as CASA leaves that axis
20//!   out of the regrid. Different labels produce a warning, not a refusal
21//!   or a polarization conversion (owner decision, #980, 2026-10-31);
22//! - its spectral frame is the image's: casa-rs does not convert frames. An
23//!   MFS image's start model has one plane, at any frequency: CASA's regrid
24//!   replicates a degenerate spectral axis, and an MFS model holds one plane
25//!   whatever frequency it is stated at.
26//!
27//! It is the model's first generation, so the first major cycle subtracts
28//! its prediction and the restored image includes it. Compile reads its
29//! header only; its pixels are read and regridded once, when the initial
30//! pass seeds the model ([`StartModel::main_samples`]).
31

44/// An explicit start model or same-grid restart model checked against the main domain.
45#[derive(Clone, Debug)]
46pub(crate) struct StartModel {
47    source: Source,
48    template: RegridTemplate,
49}

51/// One start-model image: its path and shape, the template it is regridded
52/// onto, and whether its spectral axis is regridded.
53#[derive(Clone, Debug)]
54struct Regrid {
55    path: PathBuf,
56    shape: Vec<usize>,
57    template: RegridTemplate,
58    spectral: bool,
59}
60
61/// Where a start model's pixels come from.
62#[derive(Clone, Debug)]
63enum Source {
64    /// An explicit `startmodel`: an image per term, `None` for a term that
65    /// starts from zero, each regridded onto its term's grid (one Taylor
66    /// term, or a cube's every channel).
67    Regrid(Box<[Option<Regrid>]>),
68    /// An earlier run's model on the image's own grid, read as it is: one
69    /// image, or one per Taylor term, `None` for a term with none (CASA
70    /// makes it zero).
71    Direct(Box<[Option<PathBuf>]>),
72}
73
74impl StartModel {
75    /// Check `paths` as the start model of `main`, an image of
76    /// `polarizations` Stokes planes and `planes` reconstruction planes in
77    /// `terms` terms (a cube's channels in one term, or one Taylor term
78    /// each): the k-th path, when it is given, is term k's.
79    pub(super) fn prepare(
80        paths: &[Option<PathBuf>],
81        main: &PreparedImageDomain,
82        polarizations: usize,
83        (planes, terms): (usize, usize),
84    ) -> Result<Self, PrepareError> {
85        let [nx, ny] = main.image_size;
86        let term = RegridTemplate {
87            coordinates: main.coordinates.clone(),
88            shape: vec![nx, ny, polarizations, planes / terms],
89        };
90        let mut sources = Vec::with_capacity(terms);
91        for index in 0..terms {
92            let source = match paths.get(index).cloned().flatten() {
93                Some(path) => Some(Self::check(&path, term.clone())?),
94                None => None,
95            };
96            sources.push(source);
97        }
98        Ok(Self {
99            source: Source::Regrid(sources.into_boxed_slice()),
100            template: RegridTemplate {
101                coordinates: main.coordinates.clone(),
102                shape: vec![nx, ny, polarizations, planes],
103            },
104        })
105    }
106
107    /// Check the image at `path` as a start model of `template`'s image.
108    pub(crate) fn onto(path: &Path, template: RegridTemplate) -> Result<Self, PrepareError> {
109        let checked = Self::check(path, template.clone())?;
110        Ok(Self {
111            source: Source::Regrid(Box::new([Some(checked)])),
112            template,
113        })
114    }
115
116    /// The image at `path` checked as a start model of `template`'s image,
117    /// with the template it is regridded onto, whose Stokes are relabelled
118    /// to the image's so its planes are copied by position.
119    fn check(path: &Path, mut template: RegridTemplate) -> Result<Regrid, PrepareError> {
120        let refuse = |reason: String| PrepareError::StartModel {
121            path: path.to_path_buf(),
122            reason,
123        };
124        let opened = AnyPagedImage::open(path).map_err(|error| refuse(error.to_string()))?;
125        let (units, coordinates, shape) = match &opened {
126            AnyPagedImage::Float32(image) => header(image),
127            AnyPagedImage::Float64(image) => header(image),
128            AnyPagedImage::Complex32(_) | AnyPagedImage::Complex64(_) => {
129                return Err(refuse("a start model is a real image".into()));
130            }
131        };
132        if units != "Jy/pixel" {
133            return Err(refuse(format!(
134                "its unit is {units:?}; a start model is in Jy/pixel, as CASA's tclean requires"
135            )));
136        }
137        let (ours, theirs) = (stokes(&coordinates), stokes(&template.coordinates));
138        // SIImageStore::regridToModelImage regrids only direction and
139        // spectral axes. Match its positional Stokes copying even when
140        // the labels differ. This template is private to seeding pixels;
141        // published coordinates still come from the compiled image domain.
142        if ours != theirs
143            && let (Some(input), Some(output)) = (
144                coordinates.find_coordinate(CoordinateType::Stokes),
145                template.coordinates.find_coordinate(CoordinateType::Stokes),
146            )
147        {
148            template
149                .coordinates
150                .replace_coordinate(output, coordinates.coordinate(input).clone());
151        }
152        let spectral =
153            check_regrid_along_frequency(&coordinates, &shape, &template).map_err(|error| {
154                refuse(format!(
155                    "{error}; casa-rs regrids a start model's direction and spectral axes, in \
156                     the image's frames"
157                ))
158            })?;
159        if ours != theirs {
160            tracing::warn!(startmodel = ?path, input_stokes = ?ours, image_stokes = ?theirs,
161                "startmodel Stokes labels differ: copying planes by position as CASA does; no polarization conversion is performed");
162        }
163        Ok(Regrid {
164            path: path.to_path_buf(),
165            shape,
166            template,
167            spectral,
168        })
169    }
170
171    /// Reuse an earlier run's model, whose images `terms` (one, or one per
172    /// Taylor term with `None` for a missing term) the caller checked on
173    /// the requested grid, without regridding. `template` is the model's
174    /// whole grid: `[nx, ny, stokes, planes]`, its planes the channels of
175    /// the one image or the Taylor terms.
176    pub(super) fn restart(terms: Box<[Option<PathBuf>]>, template: RegridTemplate) -> Self {
177        Self {
178            source: Source::Direct(terms),
179            template,
180        }
181    }
182
183    /// Bytes [`Self::main_samples`] holds at most besides the model: the
184    /// image read (up to 8 bytes a pixel) with its mask, its copy with the
185    /// masked pixels zero, `regrid`'s own read of that copy, the regridded
186    /// image with its mask and the samples copied out of it, and `regrid`'s
187    /// planes: an input plane of `f64` with its mask, and an output plane of
188    /// `f64`, a flag and an input position each. The direction axes are the
189    /// first two of both images, as `check_regrid` requires. A spectral
190    /// regrid holds the direction pass's image, of the output's planes and
191    /// the input's channels, with its mask, beside its own output. Terms
192    /// are regridded one at a time, each onto its own term's grid, the
193    /// samples of all of them kept: the largest term's charge. Direct
194    /// restart holds only the input image and the emitted samples, within
195    /// this charge.
196    pub(crate) fn seeding_bytes(&self) -> u64 {
197        let samples: usize = self.template.shape.iter().product::<usize>() * 4;
198        let term_bytes = |input_shape: &[usize], output_shape: &[usize], spectral: bool| {
199            let input: usize = input_shape.iter().product();
200            let output: usize = output_shape.iter().product();
201            let input_plane = input_shape[0] * input_shape[1];
202            let output_plane = output_shape[0] * output_shape[1];
203            let position = size_of::<Option<[f64; 2]>>();
204            let direction_pass = if spectral {
205                output_plane * (input / input_plane) * (4 + 1)
206            } else {
207                0
208            };
209            input * (8 + 1 + 4 + 4)
210                + output * (4 + 1)
211                + direction_pass
212                + input_plane * (8 + 1)
213                + output_plane * (8 + 1 + position)
214        };
215        let largest = match &self.source {
216            Source::Regrid(terms) => terms
217                .iter()
218                .flatten()
219                .map(|term| term_bytes(&term.shape, &term.template.shape, term.spectral))
220                .max()
221                .unwrap_or(0),
222            Source::Direct(_) => term_bytes(&self.template.shape, &self.template.shape, false),
223        };
224        (samples + largest) as u64
225    }

227    /// The image's pixels along x and y.
228    pub(crate) fn image_size(&self) -> [usize; 2] {
229        [self.template.shape[0], self.template.shape[1]]
230    }

232    /// The image's polarizations.
233    pub(crate) fn polarizations(&self) -> usize {
234        self.template.shape[2]
235    }

237    /// Bytes making the samples physical holds besides
238    /// [`Self::seeding_bytes`]: the physical samples, and one plane's
239    /// sensitivity read in `f64`.
240    pub(crate) fn conversion_bytes(&self) -> u64 {
241        let output: usize = self.template.shape.iter().product();
242        let output_plane = self.template.shape[0] * self.template.shape[1];
243        (output * 4 + output_plane * 8) as u64
244    }

246    /// The main domain's model samples in the model's canonical order
247    /// (spectral plane, polarization, y, x, x fastest), either read directly
248    /// for restart or regridded for an explicit start model.
249    pub(crate) fn main_samples(&self) -> Result<Vec<f32>, ImageError> {
250        let terms = match &self.source {
251            Source::Regrid(terms) => terms,
252            Source::Direct(terms) => {
253                let total: usize = self.template.shape.iter().product();
254                let per_term = total / terms.len();
255                let mut samples = Vec::with_capacity(total);
256                for term in terms {
257                    match term {
258                        Some(path) => {
259                            let image = casa_images::PagedImage::<f32>::open(path)?;
260                            // CASA's restart prediction uses StokesImageUtil::From's raw
261                            // lattice iterator; unlike explicit startmodel, it does not
262                            // zero a pixel mask.
263                            let values = image.get()?;
264                            // Iterate logical axes in model order (term or channel,
265                            // Stokes, y, x), independent of image storage order.
266                            samples.extend(values.reversed_axes().iter().copied());
267                        }
268                        None => samples.resize(samples.len() + per_term, 0.0),
269                    }
270                }
271                return Ok(samples);
272            }
273        };
274        let total: usize = self.template.shape.iter().product();
275        let per_term = total / terms.len();
276        let mut samples = Vec::with_capacity(total);
277        for term in terms {
278            let Some(term) = term else {
279                samples.resize(samples.len() + per_term, 0.0);
280                continue;
281            };
282            let zeroed = match AnyPagedImage::open(&term.path)? {
283                AnyPagedImage::Float32(image) => zeroed_copy(&image)?,
284                AnyPagedImage::Float64(image) => zeroed_copy(&image)?,
285                AnyPagedImage::Complex32(_) | AnyPagedImage::Complex64(_) => {
286                    return Err(ImageError::InvalidMetadata(
287                        "a start model is a real image".into(),
288                    ));
289                }
290            };
291            let regridded = regrid_along_frequency(&zeroed, &term.template, RegridMethod::Linear)?;
292            samples.extend_from_slice(
293                regridded
294                    .data
295                    .as_slice_memory_order()
296                    .expect("regrid writes a contiguous Fortran-order image"),
297            );
298        }
299        Ok(samples)
300    }
301}

303/// The Stokes of `coordinates`' Stokes coordinate, in plane order; `None`
304/// without one.
305fn stokes(coordinates: &CoordinateSystem) -> Option<Vec<StokesType>> {
306    (0..coordinates.n_coordinates()).find_map(|index| match coordinates.coordinate(index) {
307        CoordinateModel::Stokes(stokes) => Some(stokes.stokes().to_vec()),
308        _ => None,
309    })
310}

312/// The unit, coordinates and shape [`StartModel::prepare`] reads.
313fn header<T: ImagePixel>(
314    image: &casa_images::PagedImage<T>,
315) -> (String, CoordinateSystem, Vec<usize>) {
316    (
317        image.units().to_string(),
318        image.coordinates().clone(),
319        image.shape().to_vec(),
320    )
321}

323/// `image` in memory, in single precision, with its masked pixels zero and
324/// no mask, as `setModelImageOne` copies it before regridding.
325fn zeroed_copy<T>(image: &casa_images::PagedImage<T>) -> Result<TempImage<f32>, ImageError>
326where
327    T: ImagePixel + Into<f64> + Copy,
328{
329    let mut values: ArrayD<f32> = image.get()?.mapv(|value| Into::<f64>::into(value) as f32);
330    if let Some(mask) = image.default_mask()? {
331        Zip::from(&mut values).and(&mask).for_each(|value, &valid| {
332            if !valid {
333                *value = 0.0;
334            }
335        });
336    }
337    let mut copy = TempImage::<f32>::new(
338        image.shape().to_vec(),
339        image.coordinates().clone(),
340        TempStoragePolicy::Memory,
341    )?;
342    copy.set_units(image.units())?;
343    copy.put_slice(&values, &vec![0; values.ndim()])?;
344    Ok(copy)
345}

347#[cfg(test)]
348mod tests {
349    use super::*;
350    use casa_coordinates::{
351        DirectionCoordinate, Projection, ProjectionType, SpectralCoordinate, StokesCoordinate,
352    };
353    use casa_types::measures::{direction::DirectionRef, frequency::FrequencyRef};
354    use ndarray::IxDyn;
355
356    #[test]
357    fn direct_restart_reads_raw_x_fastest_pixels_while_startmodel_zeros_the_mask() {
358        let root = tempfile::tempdir().unwrap();
359        let path = root.path().join("existing.model");
360        let shape = vec![4, 3, 1, 1];
361        let mut coordinates = CoordinateSystem::new();
362        coordinates.add_coordinate(DirectionCoordinate::new(
363            DirectionRef::J2000,
364            Projection::new(ProjectionType::SIN),
365            [1.0, 0.5],
366            [-1e-5, 1e-5],
367            [2.0, 1.0],
368        ));
369        coordinates.add_coordinate(StokesCoordinate::new(vec![StokesType::I]));
370        coordinates.add_coordinate(SpectralCoordinate::new(
371            FrequencyRef::LSRK,
372            1e9,
373            1e6,
374            0.0,
375            1e9,
376        ));
377        let mut image =
378            casa_images::PagedImage::<f32>::create(shape.clone(), coordinates.clone(), &path)
379                .unwrap();
380        image.set_units("Jy/pixel").unwrap();
381        let pixels =
382            ArrayD::from_shape_fn(IxDyn(&shape), |index| (index[0] + 10 * index[1]) as f32);
383        image.put_slice(&pixels, &[0; 4]).unwrap();
384        image.make_mask("external", true, false).unwrap();
385        image.save().unwrap();
386        drop(image);
387        let template = RegridTemplate {
388            coordinates,
389            shape: shape.clone(),
390        };
391        let explicit = StartModel::onto(&path, template.clone()).unwrap();
392        assert!(
393            explicit
394                .main_samples()
395                .unwrap()
396                .iter()
397                .all(|&value| value == 0.0)
398        );
399        // The task gate still refuses masked-model continuation until publication
400        // can preserve the mask; the seed reader must nevertheless match CASA.
401        let direct = StartModel::restart(Box::new([Some(path.clone())]), template.clone());
402        assert_eq!(
403            direct.main_samples().unwrap(),
404            vec![
405                0.0, 1.0, 2.0, 3.0, 10.0, 11.0, 12.0, 13.0, 20.0, 21.0, 22.0, 23.0
406            ]
407        );
408        // A Taylor family reads each term in order; a missing term is zero.
409        let terms = StartModel::restart(
410            Box::new([None, Some(path)]),
411            RegridTemplate {
412                shape: vec![4, 3, 1, 2],
413                ..template
414            },
415        );
416        let samples = terms.main_samples().unwrap();
417        assert_eq!(samples[..12], [0.0; 12]);
418        assert_eq!(samples[12..], direct.main_samples().unwrap()[..]);
419    }
