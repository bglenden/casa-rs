3use libm::j1;
4
5const SAMPLE_COUNT: usize = 10_000;
6const SAMPLE_COUNT_MINUS_ONE: f64 = (SAMPLE_COUNT - 1) as f64;
7
8/// Cached CASA-compatible voltage table for an annular aperture.
9///
10/// The caller owns the physical meaning and units of `maximum_radius`; values
11/// passed to [`Self::evaluate`] must use the same radial coordinate. The table
12/// preserves CASA's 10,000 `Float` samples and truncated lookup indexing.
13#[derive(Clone)]
14pub struct AnnularApertureVoltageTable {
15    maximum_radius: f64,
16    values: Box<[f32]>,
17}
18
19impl AnnularApertureVoltageTable {
20    /// Tabulate an aperture with the supplied diameter, blockage, and support.
21    #[must_use]
22    pub fn new(aperture_diameter: f64, blockage_diameter: f64, maximum_radius: f64) -> Self {
23        let dimensionless_max_radius =
24            maximum_radius * 7.016 / (1.566 * 60.0) * aperture_diameter / 24.5;
25        let area_ratio = (aperture_diameter / blockage_diameter).powi(2);
26        let length_ratio = aperture_diameter / blockage_diameter;
27        let values = (0..SAMPLE_COUNT)
28            .map(|index| {
29                voltage_at_index(
30                    index as f64,
31                    dimensionless_max_radius,
32                    area_ratio,
33                    length_ratio,
34                    blockage_diameter,
35                )
36            })
37            .collect();
38        Self {
39            maximum_radius,
40            values,
41        }
42    }
43
44    /// Evaluate the tabulated voltage at a nonnegative radial coordinate.
45    ///
46    /// Coordinates beyond the caller-supplied support return zero. Values
47    /// inside the support use CASA's lower-sample (floor) selection rather
48    /// than interpolation.
49    #[must_use]
50    pub fn evaluate(&self, radius: f64) -> f32 {
51        if !radius.is_finite() || radius < 0.0 || radius > self.maximum_radius {
52            return 0.0;
53        }
54        let index = (radius * SAMPLE_COUNT_MINUS_ONE / self.maximum_radius)
55            .floor()
56            .clamp(0.0, SAMPLE_COUNT_MINUS_ONE) as usize;
57        self.values[index]
58    }
59
60    /// Return the radial support supplied by the scientific owner.
61    #[must_use]
62    pub fn maximum_radius(&self) -> f64 {
63        self.maximum_radius
64    }
65}
66
67fn voltage_at_index(
68    index: f64,
69    dimensionless_max_radius: f64,
70    area_ratio: f64,
71    length_ratio: f64,
72    blockage_diameter: f64,
73) -> f32 {
74    let x = index * dimensionless_max_radius / SAMPLE_COUNT_MINUS_ONE;
75    if x.abs() <= f64::EPSILON {
76        return 1.0;
77    }
78    if blockage_diameter <= 0.0 {
79        return (2.0 * j1(x) / x) as f32;
80    }
81    ((area_ratio * 2.0 * j1(x) / x - 2.0 * j1(x * length_ratio) / (x * length_ratio))
82        / (area_ratio - 1.0)) as f32
83}

85#[cfg(test)]
86mod tests {
87    use super::*;
88
89    #[test]
90    fn table_uses_explicit_support_and_casa_truncation() {
91        let table = AnnularApertureVoltageTable::new(10.7, 0.75, 107.04);
92        let bin_width = table.maximum_radius() / SAMPLE_COUNT_MINUS_ONE;
93
94        assert_eq!(
95            [0.0, 0.01, 1.0, 50.0, 107.04].map(|radius| table.evaluate(radius).to_bits()),
96            [
97                1_065_353_216,
98                1_065_353_216,
99                1_065_353_196,
100                1_060_422_159,
101                1_034_343_953
102            ]
103        );
104
105        assert_eq!(table.evaluate(0.0), 1.0);
106        assert_eq!(table.evaluate(bin_width * 12.25), table.values[12]);
107        assert_eq!(table.evaluate(bin_width * 12.75), table.values[12]);
108        assert_eq!(table.evaluate(bin_width * 13.25), table.values[13]);
109        assert_eq!(table.evaluate(107.04 + bin_width), 0.0);
110    }
111
112    #[test]
113    fn unblocked_aperture_is_supported() {
114        let table = AnnularApertureVoltageTable::new(12.0, 0.0, 107.04);
115        assert_eq!(table.evaluate(0.0), 1.0);
116        assert!(table.evaluate(50.0).is_finite());
117    }
118}
