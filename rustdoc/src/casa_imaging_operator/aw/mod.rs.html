147/// One (PA, frequency, w) group's supports; its cells, one per Mueller
148/// element in the catalog's element order, are [`AwCatalog::group_cells`].
149#[derive(Clone, Debug)]
150struct Group {
151    /// The largest declared half support over the group's imaging cells:
152    /// the imaging dense cell is padded to it, and `AWVisResampler`'s
153    /// `onGrid` test for a data row uses it (`CF Support: 4 (6)` in CASA's
154    /// log is the imaging support with the weight support in brackets).
155    imaging_half_support: [u16; 2],
156    /// The same over the weight cells, which the PSF and weight image use.
157    weight_half_support: [u16; 2],
158    /// The smallest declared half support over both kinds: the row filter
159    /// keeps every row some cell of the group fits.
160    smallest_half_support: [u16; 2],
161}
162
163impl Group {
164    fn new(cells: &[CellFiles]) -> Self {
165        let largest = |select: fn(&CellFiles) -> &CellHeader| {
166            cells.iter().map(select).fold([0_u16; 2], |half, header| {
167                [
168                    half[0].max(header.support[0]),
169                    half[1].max(header.support[1]),
170                ]
171            })
172        };
173        let imaging_half_support = largest(|cell| &cell.imaging);
174        let weight_half_support = largest(|cell| &cell.weight);
175        let smallest_half_support = cells
176            .iter()
177            .flat_map(|cell| [&cell.imaging, &cell.weight])
178            .fold([u16::MAX; 2], |half, header| {
179                [
180                    half[0].min(header.support[0]),
181                    half[1].min(header.support[1]),
182                ]
183            });
184        Self {
185            imaging_half_support,
186            weight_half_support,
187            smallest_half_support,
188        }
189    }
190}

192/// The support `locate_sample` tests a row against: the smallest cell the
193/// gridding group or the prediction group can select. Every mode re-tests
194/// the support of the cell it reads (`AWVisResampler::DataToGrid` and
195/// `GridToData` call `onGrid` with the selected cell's support), so the
196/// filter only has to keep every row CASA grids with any of them.
197fn placement_half_support(gridding: &Group, prediction: &Group) -> [u16; 2] {
198    let (g, p) = (
199        gridding.smallest_half_support,
200        prediction.smallest_half_support,
201    );
202    [g[0].min(p[0]), g[1].min(p[1])]
203}

223/// The bounded cell cache: least recently used eviction once the loaded
224/// cells exceed the bound; a cell a worker holds stays alive through its

226struct Lru {

228    slots: Vec<Slot>,
229    /// Use counter stamped on every fetch.
230    clock: u64,

237/// A group's place in the cache.
238enum Slot {
239    /// Not loaded.
240    Empty,
241    /// A fetcher is reading it from disk, outside the cache's lock; another
242    /// fetcher of it waits for that load rather than reading the group again

245    /// Its loaded taps.
246    Loaded(Loaded),
247}

249/// The AW-projection kernel set over a CASA convolution-function cache.
250///
251/// Cells are keyed on the parallactic-angle cell, the frequency cell and
252/// the w-plane (`CfKey::group`); the Mueller elements live inside the cell
253/// as planes, routed by [`MuellerRouting`] with CASA's selection of both
254/// the cell and its input polarization (`AWVisResampler`). Prediction
255/// divides once by the kernel sum and `sumwt += W·|N|`
256/// ([`KernelNormalisation::KernelSum`]); the pointing ramp applies to
257/// every row ([`ConvolutionFunctionSet::pointing_ramp`]). A row whose
258/// parallactic angle lies between cells takes the nearest cell; CASA would
259/// rotate it (`AWConvFunc::rotateCF`, `rotatepastep`), which the frozen
260/// 360° steps of the EVLA cache make the same choice.
261///
262/// # Panics
263///
264/// [`ConvolutionFunctionSet::taps`] and [`ConvolutionFunctionSet::weight_taps`]
265/// are infallible: a cell image that was validated at open but cannot be
266/// read when a row first needs it is a catalog broken under the run, and
267/// the catalog panics with the cell's path rather than gridding without it.
268pub struct AwCatalog {
269    root: PathBuf,
270    indexing: AwIndexing,
271    pa_deg: Vec<f64>,
272    frequencies_hz: Vec<f64>,
273    w_values: Vec<f64>,
274    w_increment: f64,
275    mueller_elements: Vec<u32>,
276    /// Every cell, by group and then Mueller element: group `g`'s are
277    /// `cells[g·E..(g + 1)·E]` with `E` the number of elements.
278    cells: Vec<CellFiles>,
279    groups: Vec<Group>,
280    oversampling: u16,
281    max_half_support: [u16; 2],
282    mueller: MuellerRouting,
283    correction: ImageCorrection,
284    lru: Mutex<Lru>,
285    /// Signalled whenever a group's load ends, loaded or not, for the
286    /// fetchers waiting on it ([`Slot::Loading`]).
287    load_ended: Condvar,
288}

303impl AwCatalog {
304    /// Open the cache at `root`: read every `CFS_*`/`WTCFS_*` image's
305    /// coordinates and misc info (`CFCache::fillCFListFromDisk`), sort the
306    /// unique parallactic angles, frequencies, w values and Mueller
307    /// elements, and build the routing of `polarization`'s correlations
308    /// over `geometry`'s grid. Pixels are read on demand, at most
309    /// `bound_bytes` of cells resident.
310    ///
311    /// # Errors
312    ///
313    /// The directory cannot be listed, a cell is unreadable or malformed,
314    /// a cell lacks its partner, a (PA, frequency, w) group lacks a Mueller
315    /// element, or the catalog has no element for a selected correlation.
316    pub fn open_casa(
317        root: impl AsRef<Path>,
318        indexing: AwIndexing,
319        geometry: &GridGeometry,
320        polarization: &PolarizationRouting,
321        bound_bytes: usize,
322    ) -> Result<Self, AwCatalogError> {
323        let root = root.as_ref().to_path_buf();
324        let mut cells = read_cells(&root)?;
325        // A native cell's working field is the image field times the
326        // oversampling: `working_size · working_increment = sampling · n · Δx`.
327        let [grid_n, _] = geometry.grid_shape();
328        let image_increment = geometry.image().increment_rad[0].abs();
329        for header in cells.iter().map(|cell| &cell.imaging) {
330            let Some(working_field) = header.working_field_rad else {
331                continue;
332            };
333            let expected = image_increment * f64::from(header.sampling) * grid_n as f64;
334            if (working_field - expected).abs() > 1.0e-6 * expected {
335                return Err(cache_error(
336                    &header.path,
337                    format!(
338                        "generated for a working field of {working_field} rad, the image needs \
339                         {expected} rad: the cache belongs to another image geometry"
340                    ),
341                ));
342            }
343        }
344        // Sorted unique values, held at their length: the catalog keeps them
345        // and its charge counts their capacity (#745 review). Values equal
346        // under `total_cmp` share their bits, so the order among them is moot.
347        let unique = |values: Vec<f64>| {
348            let mut values = values;
349            values.sort_unstable_by(f64::total_cmp);
350            values.dedup_by(|a, b| a.to_bits() == b.to_bits());
351            values.shrink_to_fit();
352            values
353        };
354        let pa_deg = unique(cells.iter().map(|cell| cell.imaging.pa_deg).collect());
355        let frequencies_hz = unique(cells.iter().map(|cell| cell.imaging.frequency_hz).collect());
356        let w_values = unique(cells.iter().map(|cell| cell.imaging.w_value).collect());
357        let mut mueller_elements = cells
358            .iter()
359            .map(|cell| cell.imaging.mueller)
360            .collect::<Vec<_>>();
361        mueller_elements.sort_unstable();
362        mueller_elements.dedup();
363        mueller_elements.shrink_to_fit();
364        let w_increment = cells[0].imaging.w_increment;
365        let oversampling = cells[0].imaging.sampling;
366        if cells.iter().any(|cell| {
367            cell.imaging.w_increment.to_bits() != w_increment.to_bits()
368                || cell.imaging.sampling != oversampling
369        }) {
370            return Err(cache_error(&root, "cells disagree on WIncr or Sampling"));
371        }
372        // Order the cells in place by group and element: distinct keys over
373        // every (group, element) pair leave cell `i` at key `i`.
374        let index_of = |values: &[f64], value: f64| {
375            values
376                .binary_search_by(|listed| listed.total_cmp(&value))
377                .expect("listed value")
378        };
379        let elements = mueller_elements.len();
380        let key = |cell: &CellFiles| {
381            let header = &cell.imaging;
382            let group = (index_of(&pa_deg, header.pa_deg) * frequencies_hz.len()
383                + index_of(&frequencies_hz, header.frequency_hz))
384                * w_values.len()
385                + index_of(&w_values, header.w_value);
386            let plane = mueller_elements
387                .binary_search(&header.mueller)
388                .expect("listed element");
389            group * elements + plane
390        };
391        cells.sort_unstable_by_key(key);
392        if let Some(pair) = cells.windows(2).find(|pair| key(&pair[0]) == key(&pair[1])) {
393            return Err(cache_error(&pair[1].imaging.path, "duplicate cell key"));
394        }
395        if cells.len() != pa_deg.len() * frequencies_hz.len() * w_values.len() * elements {
396            return Err(cache_error(
397                &root,
398                "a (PA, frequency, w) group lacks a Mueller element",
399            ));
400        }
401        let max_half_support = cells.iter().fold([0_u16; 2], |largest, cell| {
402            [0, 1].map(|axis| {
403                largest[axis]
404                    .max(cell.imaging.support[axis])
405                    .max(cell.weight.support[axis])
406            })
407        });
408        let groups = cells
409            .chunks_exact(elements)
410            .map(Group::new)
411            .collect::<Vec<_>>();
412        let mueller = routing(&mueller_elements, polarization).ok_or_else(|| {
413            cache_error(
414                &root,
415                "selected correlations have missing or overlapping Mueller routes",
416            )
417        })?;
418        let [nx, ny] = geometry.grid_shape();
419        // `AWProjectFT::getImage` and `getWeightImage` divide every image
420        // by the sampling sinc `sin(x)/x`, `x = π(i − n/2)/(n·sampling)`;
421        // the spheroidal lives in the cells under `psterm`, and
422        // `initializeToVis` leaves the model uncorrected (its `sincConv`
423        // is reset to one).
424        let sampling_correction = |len: usize| {
425            (0..len)
426                .map(|index| {
427                    if index == len / 2 {
428                        return 1.0;
429                    }
430                    let x = std::f64::consts::PI * (index as f64 - (len / 2) as f64)
431                        / (len as f64 * f64::from(oversampling));
432                    x / x.sin()
433                })
434                .collect::<Vec<_>>()
435        };
436        Ok(Self {
437            root,
438            indexing,
439            pa_deg,
440            frequencies_hz,
441            w_values,
442            w_increment,
443            mueller_elements,
444            lru: Mutex::new(Lru {
445                slots: (0..groups.len()).map(|_| Slot::Empty).collect(),
446                clock: 0,
447                bytes: 0,
448                bound: bound_bytes,
449                loads: 0,
450            }),
451            cells,
452            groups,
453            oversampling,
454            max_half_support,
455            mueller,
456            correction: ImageCorrection::split(
457                sampling_correction(nx),
458                sampling_correction(ny),
459                vec![1.0; nx],
460                vec![1.0; ny],
461            ),
462            load_ended: Condvar::new(),
463        })
464    }

466    /// Heap bytes [`Self::generate_native`] holds at most for `input` on
467    /// `workers` workers (#738, #708): for each worker, the working grid
468    /// ([`EvlaAwWorkspace::bytes`]) and the cell image being written (a crop
469    /// of the working grid at most as wide, as its pixel array and the copy
470    /// the image takes of it, and the image's table and records, 16 KiB);
471    /// planning the grid's transform, which the process does one at a time
472    /// and then caches for every workspace; and the bookkeeping that grows
473    /// with the request (#747 review): the work list, two presence bits per
474    /// cell for the state a fill checks first, and the request manifest.
475    #[must_use]
476    pub const fn generation_bytes(input: &NativeAwRequestInput, workers: usize) -> u64 {
477        let workers = if workers > 1 { workers } else { 1 };
478        let cells = input.grid.size * input.grid.size;
479        let worker = EvlaAwWorkspace::bytes(input.grid.size)
480            + 2 * cells * size_of::<Complex32>()
481            + CELL_IMAGE_RECORD_BYTES;
482        let apertures =
483            input.pa_values.len() * input.frequencies.len() * input.mueller_elements.len();
484        let bookkeeping = generation::units_bytes(apertures, workers)
485            + native_cache::cache_state_bytes(input)
486            + native_cache::manifest_bytes(input);
487        (workers * worker + bookkeeping) as u64 + casa_fft::planning_bytes::<f32>(cells)
488    }

490    /// Generate every cell of `input` natively into `root` as CASA-format
491    /// images (`CFS_<pa>_0_CF_<f>_<w>_<m>.im` and the `WTCFS_` partner, one
492    /// baseline type) so [`AwCatalog::open_casa`] serves them, and record
493    /// `input` in the directory's request manifest
494    /// ([`AwCatalog::native_cache_state`]).
495    ///
496    /// Without `only_missing` the directory is cleared first
497    /// ([`AwCatalog::clear_native`]). With it, the cells already present
498    /// are kept, and the directory must hold nothing of another request
499    /// ([`NativeCacheState::fillable`]). Each cell image is published only
500    /// once complete, and the manifest is written last, so generation that
501    /// fails or stops part way never leaves a directory that claims a
502    /// request it does not hold (#739 review). The caller holds the
503    /// exclusive [`NativeCacheLock`].
504    ///
505    /// The cells are generated on up to `workers` threads, each with its
506    /// own workspace, and are the cells one worker generates (#708).
507    ///
508    /// # Errors
509    ///
510    /// Generation fails or an image cannot be written, or `only_missing`
511    /// is asked of a directory that holds cells of another request.
512    pub fn generate_native(
513        root: impl AsRef<Path>,
514        input: &NativeAwRequestInput,
515        only_missing: bool,
516        workers: usize,
517    ) -> Result<(), AwCatalogError> {
518        let root = root.as_ref();
519        if only_missing {
520            let state = Self::native_cache_state(root, input)?;
521            if !state.fillable() {
522                return Err(cache_error(
523                    root,
524                    format!("cannot fill in a cache of another request: {state}"),
525                ));
526            }
527        } else {
528            Self::clear_native(root)?;
529        }
530        std::fs::create_dir_all(root)
531            .map_err(|error| cache_error(root, format!("cannot create: {error}")))?;
532        let units = generation::units(root, input, only_missing, workers);
533        generation::generate(root, input, &units, only_missing, workers)?;
534        write_manifest(root, input)
535    }

603    /// stages. Opening reads one header at a time and lists one axis's
604    /// values before it is reduced. Serving holds its cells and, for each
605    /// holder, the planes it reads to build one group (each read as an image
606    /// array and its copy) with that read's records.
607    ///
608    /// When every group fits the cache's bound nothing is evicted and each
609    /// group is loaded once, so the cells are at most every group (#738).
610    /// Otherwise they are the bound and one group past it, which a fetch
611    /// inserts before the cache evicts, and for each holder the largest
612    /// group twice. A CPU residual holds a gridding cell and a prediction
613    /// cell (#449), and either keeps both alive after eviction, or one
614    /// beside the group it builds outside the cache's lock for the other
615    /// (#744): a [`CellHold`] releases its own cell before it fetches
616    /// (#773 review). Every loaded cell is also its [`Arc`] (#745 review).
617    #[must_use]
618    pub fn charge_bytes(&self, holders: usize) -> u64 {
619        let lru = self.lru.lock().expect("cache lock");
620        let paths = self
621            .cells
622            .iter()
623            .map(|cell| cell.imaging.path.capacity() + cell.weight.path.capacity());
624        let index = self.root.capacity()
625            + self.cells.capacity() * size_of::<CellFiles>()
626            + paths.clone().sum::<usize>()
627            + self.groups.capacity() * size_of::<Group>()
628            + (self.pa_deg.capacity() + self.frequencies_hz.capacity() + self.w_values.capacity())
629                * size_of::<f64>()
630            + self.mueller_elements.capacity() * size_of::<u32>()
631            + lru.slots.capacity() * size_of::<Slot>();
632        let opening = (self.cells.len() * size_of::<f64>())
633            .max(CELL_IMAGE_RECORD_BYTES + 4 * paths.max().unwrap_or(0))
634            as u64;
635        let elements = self.mueller_elements.len();
636        let group = |group: &Group| {
637            dense_cell_bytes(elements, self.oversampling, group.imaging_half_support)
638                + dense_cell_bytes(elements, self.oversampling, group.weight_half_support)
639        };
640        let read = |cells: &[CellFiles]| {
641            let plane = |header: &CellHeader| header.shape[0] * header.shape[1];
642            let imaging = cells.iter().map(|cell| plane(&cell.imaging)).sum::<usize>();
643            let weight = cells.iter().map(|cell| plane(&cell.weight)).sum::<usize>();
644            // `load_group`'s header and plane lists.
645            let lists = elements
646                * (size_of::<&CellHeader>() + size_of::<(Vec<Complex32>, [usize; 2], [u16; 2])>());
647            (2 * imaging.max(weight) * size_of::<Complex32>() + lists + CELL_IMAGE_RECORD_BYTES)
648                as u64
649        };
650        let largest = self.groups.iter().map(group).max().unwrap_or(0);
651        let every = self.groups.iter().map(group).sum::<u64>();
652        let reading = self
653            .cells
654            .chunks_exact(elements)
655            .map(read)
656            .max()
657            .unwrap_or(0);
658        let bound = lru.bound as u64;
659        let groups = self.groups.len() as u64;
660        let (taps, pairs) = if every <= bound {
661            (every, groups)
662        } else {
663            (
664                bound + (2 * holders as u64 + 1) * largest,
665                groups + 2 * holders as u64 + 1,
666            )
667        };
668        let serving = taps + pairs * 2 * ARC_CELL_BYTES as u64 + holders as u64 * reading;
669        index as u64
670            + self.mueller.bytes()
671            + self.correction.resident_bytes()
672            + opening.max(serving)
673    }

675    /// Group `group`'s cells, one per Mueller element.
676    fn group_cells(&self, group: usize) -> &[CellFiles] {
677        let elements = self.mueller_elements.len();
678        &self.cells[group * elements..(group + 1) * elements]
679    }

681    /// The parallactic-angle cell of `pa_deg`: the nearest listed angle on
682    /// the circle (`CFCache::fillCFListFromDisk` selects cells within `dPA`
683    /// and `SynthesisUtils::stdNearestValue` picks among them).
684    #[must_use]
685    pub fn pa_cell(&self, pa_deg: f64) -> usize {
686        self.pa_deg
687            .iter()
688            .enumerate()
689            .fold((0, f64::INFINITY), |best, (index, listed)| {
690                let distance = circular_degrees(pa_deg - listed).abs();
691                if distance < best.1 {
692                    (index, distance)
693                } else {
694                    best
695                }
696            })
697            .0
698    }

700    /// The frequency cell of a row at `freq_hz`: the nearest listed
701    /// frequency to `√(2f₀² − f²)` under `conjbeams`, to `f` otherwise
702    /// (`CFBuffer::initMaps`, `nearestFreqNdx`).
703    #[must_use]
704    pub fn frequency_cell(&self, freq_hz: f64) -> usize {
705        let target = if self.indexing.conjugate_beams {
706            conjugate_frequency(freq_hz, self.indexing.image_reference_hz)
707        } else {
708            freq_hz
709        };
710        nearest(&self.frequencies_hz, target).0
711    }

713    /// The w-plane of `w_lambda` (`CFBuffer::nearestWNdx`, CAS-13191).
714    #[must_use]
715    pub fn w_cell(&self, w_lambda: f64) -> usize {
716        let index = (self.w_increment * w_lambda.abs()).sqrt().round() as usize;
717        index.min(self.w_values.len() - 1)
718    }

720    /// The group of a (PA, frequency, w) cell triple.
721    #[must_use]
722    pub fn group_index(&self, pa: usize, frequency: usize, w: usize) -> usize {
723        (pa * self.frequencies_hz.len() + frequency) * self.w_values.len() + w
724    }

726    /// Fetch group `group`'s cells from the cache, or load them.
727    ///
728    /// The cache's lock is held only to look a group up and to insert it,
729    /// never while a group is read from disk, so a load does not stall the
730    /// other workers' fetches, hits included (#744). One fetcher at a time
731    /// reads a group: it marks the group loading, and another fetcher of
732    /// that group waits for the load to end and then finds it cached. A
733    /// load that panics unmarks its group as it unwinds, so a waiter reads
734    /// the group itself.
735    fn loaded(&self, group: usize) -> (Arc<DenseCell>, Arc<DenseCell>) {
736        let mut lru = self.lru.lock().expect("cache lock");
737        loop {
738            lru.clock += 1;
739            let now = lru.clock;
740            match &mut lru.slots[group] {
741                Slot::Loaded(loaded) => {
742                    loaded.last_use = now;
743                    return (loaded.imaging.clone(), loaded.weight.clone());
744                }
745                Slot::Empty => break,
746                Slot::Loading => {}
747            }
748            lru = self.load_ended.wait(lru).expect("cache lock");
749        }
750        lru.slots[group] = Slot::Loading;
751        lru.loads += 1;
752        drop(lru);
753        let loading = Loading {
754            catalog: self,
755            group,
756        };
757        let mut loaded = load_group(
758            self.group_cells(group),
759            &self.groups[group],
760            self.oversampling,
761        )
762        .unwrap_or_else(|error| panic!("AW cell group {group} cannot be read: {error}"));
763        let mut lru = self.lru.lock().expect("cache lock");
764        lru.clock += 1;
765        loaded.last_use = lru.clock;
766        let pair = (loaded.imaging.clone(), loaded.weight.clone());
767        lru.bytes += loaded.bytes();
768        // Caching the group ends its load in the same lock hold, so no
769        // other fetcher can mark it loading in between.
770        lru.slots[group] = Slot::Loaded(loaded);
771        // Evict the least recently used loaded group but the one just loaded.
772        while lru.bytes > lru.bound {
773            let Some(oldest) = lru
774                .slots
775                .iter()
776                .enumerate()
777                .filter(|(other, _)| *other != group)
778                .filter_map(|(other, slot)| match slot {
779                    Slot::Loaded(loaded) => Some((other, loaded.last_use)),
780                    Slot::Empty | Slot::Loading => None,
781                })
782                .min_by_key(|(_, last_use)| *last_use)
783                .map(|(other, _)| other)
784            else {
785                break;
786            };
787            if let Slot::Loaded(cell) = std::mem::replace(&mut lru.slots[oldest], Slot::Empty) {
788                lru.bytes -= cell.bytes();
789            }
790        }
791        loading.cached(lru);
792        pair
793    }
794}

796/// A group one fetcher is reading ([`AwCatalog::loaded`]). Its load ends
797/// with the group cached ([`Self::cached`]) or, dropped as a failed load
798/// unwinds, with the group unmarked; either way the fetchers waiting on it
799/// wake.
800struct Loading<'c> {
801    catalog: &'c AwCatalog,
802    group: usize,
803}
804
805impl Loading<'_> {
806    /// The group is cached and unmarked in `lru`, one lock hold, which this
807    /// releases before it wakes the waiters.
808    fn cached(self, lru: MutexGuard<'_, Lru>) {
809        drop(lru);
810        self.catalog.load_ended.notify_all();
811        std::mem::forget(self);
812    }
813}
814
815impl Drop for Loading<'_> {
816    fn drop(&mut self) {
817        // Only a failed load drops its mark here: a cached group's mark
818        // went with its insert ([`Self::cached`]). The load holds no lock,
819        // so it cannot have poisoned this one; a poisoned lock is another
820        // thread's panic, which ends the pass.
821        if let Ok(mut lru) = self.catalog.lru.lock() {
822            lru.slots[self.group] = Slot::Empty;
823        }
824        self.catalog.load_ended.notify_all();
825    }
826}

933/// Every cell of the cache at `root` (`CFCache::fillCFListFromDisk`): each
934/// `CFS_` image's header with its `WTCFS_` partner's, which shares the name
935/// after the prefix. The directory is listed first to count the cells, so
936/// the list is allocated once, at its length, and opening holds no table of
937/// names beside it (#745 review).
938fn read_cells(root: &Path) -> Result<Vec<CellFiles>, AwCatalogError> {
939    let entries =
940        || {
941            let listing = std::fs::read_dir(root)
942                .map_err(|error| cache_error(root, format!("cannot list: {error}")))?;
943            Ok::<_, AwCatalogError>(listing.map(|entry| {
944                entry.map_err(|error| cache_error(root, format!("cannot list: {error}")))
945            }))
946        };
947    let named = |entry: &std::fs::DirEntry, prefix: &str| {
948        let name = entry.file_name();
949        name.to_str()
950            .and_then(|name| name.strip_prefix(prefix))
951            .map(str::to_owned)
952    };
953    let (mut imaging, mut weight) = (0_usize, 0_usize);
954    for entry in entries()? {
955        let entry = entry?;
956        if named(&entry, WEIGHT_PREFIX).is_some() {
957            weight += 1;
958        } else if named(&entry, IMAGING_PREFIX).is_some() {
959            imaging += 1;
960        }
961    }
962    if imaging == 0 {
963        return Err(cache_error(root, "no CFS_ cell images"));
964    }
965    let mut cells = Vec::with_capacity(imaging);
966    for entry in entries()? {
967        let entry = entry?;
968        let Some(rest) = named(&entry, IMAGING_PREFIX) else {
969            continue;
970        };
971        let header = read_header(&entry.path())?;
972        let partner = root.join(format!("{WEIGHT_PREFIX}{rest}"));
973        if !partner.exists() {
974            return Err(cache_error(&header.path, "paired WTCFS_ cell is missing"));
975        }
976        let partner = read_header(&partner)?;
977        if partner.mueller != header.mueller
978            || partner.w_value.to_bits() != header.w_value.to_bits()
979            || partner.pa_deg.to_bits() != header.pa_deg.to_bits()
980            || partner.sampling != header.sampling
981        {
982            return Err(cache_error(
983                &partner.path,
984                "WTCFS_ cell disagrees with its CFS_ cell",
985            ));
986        }
987        cells.push(CellFiles {
988            imaging: header,
989            weight: partner,
990        });
991    }
992    // Every CFS_ cell found its partner, so a WTCFS_ cell over their count
993    // has none.
994    if weight > cells.len() {
995        for entry in entries()? {
996            let entry = entry?;
997            if let Some(rest) = named(&entry, WEIGHT_PREFIX)
998                && !root.join(format!("{IMAGING_PREFIX}{rest}")).exists()
999            {
1000                return Err(cache_error(entry.path(), "paired CFS_ cell is missing"));
1001            }
1002        }
1003    }
1004    Ok(cells)
1005}

1007/// `AWVisResampler`'s routes through diagonal Mueller cells. Each selected
1008/// correlation (the `outer` loop of `DataToGrid` and `GridToData`) has the
1009/// element `e` of its own hand (`RR, RL, LR, LL` → `0, 5, 10, 15`,
1010/// `AWProjectFT::makeCFPolMap`) and the partner `15 − e` (`RR ↔ LL`,
1011/// `RL ↔ LR`), or `e` again when the partner hand is not a selected
1012/// correlation (`PolOuterProduct::makeConjPol2CFMat` keeps the direct
1013/// entry). `getConvFunc_p` selects the own cell for `w > 0` and the
1014/// partner's otherwise in the adjoint, the reverse in the forward
1015/// transform (`GridToData` passes `mNdx` and `conjMNdx` swapped); the hand
1016/// of the selected cell is `element % npol`, CASA's index into the
1017/// visibility axis, so
1018///
1019/// `DataToGrid`: `grid[pol_map[outer]] += vis[hand(cell)] ⋆ cell`,
1020/// `GridToData`: `vis[outer] += model[pol_map[hand(cell)]] ⋆ cell`.
1021///
1022/// `None` when a cell is missing or two outer correlations route one
1023/// (grid, visibility) pair through different cells (a non-canonical
1024/// three-correlation selection, where `element % npol` collides).
1025fn routing(elements: &[u32], polarization: &PolarizationRouting) -> Option<MuellerRouting> {
1026    let pol_map = polarization.pol_map();
1027    let correlations = polarization.correlations();
1028    let plane_of = |element: u32| {
1029        elements
1030            .binary_search(&element)
1031            .ok()
1032            .map(|plane| u8::try_from(plane).expect("planes fit u8"))
1033    };
1034    let selected = |element: u32| {
1035        correlations
1036            .iter()
1037            .zip(pol_map)
1038            .any(|(correlation, target)| {
1039                target.is_some() && mueller_element(*correlation) == Some(element)
1040            })
1041    };
1042    let hand = |element: u32| element as usize % pol_map.len();
1043    let empty = || vec![vec![None; pol_map.len()]; polarization.grid_pols()];
1044    let mut adjoint = [empty(), empty()];
1045    let mut forward = [empty(), empty()];
1046    for (outer, (correlation, target)) in correlations.iter().zip(pol_map).enumerate() {
1047        let Some(gpol) = *target else {
1048            continue;
1049        };
1050        let own = mueller_element(*correlation)?;
1051        let partner = Some(MUELLER_ELEMENTS - 1 - own)
1052            .filter(|partner| selected(*partner))
1053            .unwrap_or(own);
1054        for positive in [false, true] {
1055            let cell = if positive { own } else { partner };
1056            let plane = plane_of(cell)?;
1057            if adjoint[usize::from(positive)][usize::from(gpol)][hand(cell)]
1058                .replace(plane)
1059                .is_some()
1060            {
1061                return None;
1062            }
1063            if let Some(input_grid) = pol_map[hand(cell)] {
1064                forward[usize::from(!positive)][usize::from(input_grid)][outer] = Some(plane);
1065            }
1066        }
1067    }
1068    Some(MuellerRouting { adjoint, forward })
1069}
1070
1071/// The diagonal Mueller element of a parallel or cross hand.
1072fn mueller_element(correlation: CorrelationType) -> Option<u32> {
1073    Some(match correlation {
1074        CorrelationType::CircularRr | CorrelationType::LinearXx => 0,
1075        CorrelationType::CircularRl | CorrelationType::LinearXy => 5,
1076        CorrelationType::CircularLr | CorrelationType::LinearYx => 10,
1077        CorrelationType::CircularLl | CorrelationType::LinearYy => 15,
1078        _ => return None,
1079    })
1080}

1082/// `SynthesisUtils::conjFreq`: `√(2f₀² − f²)`.
1083fn conjugate_frequency(frequency_hz: f64, reference_hz: f64) -> f64 {
1084    (2.0 * reference_hz * reference_hz - frequency_hz * frequency_hz).sqrt()
1085}

1087/// `SynthesisUtils::nearestValue`: the first listed value at the smallest
1088/// distance.
1089fn nearest(values: &[f64], target: f64) -> (usize, f64) {
1090    let mut best = (0, f64::INFINITY);
1091    for (index, value) in values.iter().enumerate() {
1092        let distance = (value - target).abs();
1093        if distance < best.1 {
1094            best = (index, distance);
1095        }
1096    }
1097    (best.0, values[best.0])
1098}

1100fn circular_degrees(value: f64) -> f64 {
1101    (value + 180.0).rem_euclid(360.0) - 180.0
1102}
