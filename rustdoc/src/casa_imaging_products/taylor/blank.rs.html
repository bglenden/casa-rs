3//! CASA MVC's unrestored, all-zero Float Taylor family (#1069).
4
5use casa_imaging_deconvolution::FinalNormalStateWindow;
6use casa_imaging_model::ProductRole;
7
8use super::{TaylorFamily, TaylorPlane, TaylorProducts};
9use crate::{AnalyticPrimaryBeamModel, ContinuumProductInputs, ProductsError};
10
11/// Publish only the already-blank Taylor products and common beam. This path
12/// never inverts a singular Hessian or supplies a restoring beam. The generic
13/// direct-Taylor nonpositive-weight refusal remains in `SolvedFamily::read`.
14pub(super) fn build(
15    inputs: &ContinuumProductInputs<'_>,
16    state: &FinalNormalStateWindow<'_>,
17    primary_beam_model: Option<AnalyticPrimaryBeamModel>,
18) -> Result<TaylorProducts, ProductsError> {
19    let list = inputs.problem().product_list();
20    if state.coefficient_term_count() != 2
21        || list.publication().members().iter().any(|id| {
22            !matches!(
23                list.entries()[id.ordinal()].role(),
24                ProductRole::Psf(_)
25                    | ProductRole::Residual(_)
26                    | ProductRole::SumWeights(_)
27                    | ProductRole::PrimaryBeam(_)
28            )
29        })
30    {
31        return Err(ProductsError::InvalidTaylorMomentReduction);
32    }
33    let family = TaylorFamily::read(inputs, (state, 0), primary_beam_model, None, 1.0)?;
34    if family.sum_weights.iter().any(|value| *value != 0.0)
35        || family.psf.iter().flatten().any(|value| *value != 0.0)
36        || family.residual.iter().flatten().any(|value| *value != 0.0)
37        || family.model.iter().flatten().any(|value| *value != 0.0)
38    {
39        return Err(ProductsError::InvalidTaylorMomentReduction);
40    }
41    let beam = inputs
42        .taylor_primary_beam()
43        .expect("MVC beam was checked")
44        .to_vec();
45    let cutoff = inputs
46        .problem()
47        .products()
48        .validity()
49        .primary_beam()
50        .cutoff();
51    let validity = beam
52        .iter()
53        .map(|value| value.is_finite() && *value > cutoff)
54        .collect();
55    Ok(TaylorProducts {
56        shape: state.shape(),
57        planes: vec![TaylorPlane {
58            psf: family.psf,
59            principal_psf_peak: 0.0,
60            residual: family.residual,
61            model: family.model,
62            restored: Vec::new(),
63            weight: family.weight,
64            sensitivity: Vec::new(),
65            sum_weights: family.sum_weights,
66            primary_beam: vec![beam, Vec::new()],
67            pb_corrected: Vec::new(),
68            spectral_index: None,
69            primary_beam_validity: validity,
70            clean_mask: Vec::new(),
71            fitted_beam: None,
72            restoring_beam: None,
73        }],
74    })
75}
