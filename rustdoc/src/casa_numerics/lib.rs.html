7/// Solve a weighted linear least-squares system with an SVD-backed solver.
8///
9/// Each row is `(basis_values, observed_value, weight)`. The implementation
10/// applies the usual `sqrt(weight)` row scaling before solving `A x = b`.
11#[must_use]
12pub fn solve_weighted_least_squares(
13    rows: &[(Vec<f64>, f64, f64)],
14    coefficient_count: usize,
15) -> Option<Vec<f64>> {
16    if coefficient_count == 0 || rows.len() < coefficient_count {
17        return None;
18    }
19
20    let mut design = Vec::with_capacity(rows.len() * coefficient_count);
21    let mut observed = Vec::with_capacity(rows.len());
22    for (basis, value, weight) in rows {
23        if basis.len() != coefficient_count
24            || !value.is_finite()
25            || !weight.is_finite()
26            || *weight <= 0.0
27        {
28            return None;
29        }
30        let scale = weight.sqrt();
31        design.extend(basis.iter().map(|entry| entry * scale));
32        observed.push(value * scale);
33    }
34
35    let design = DMatrix::from_row_slice(rows.len(), coefficient_count, &design);
36    let observed = DVector::from_row_slice(&observed);
37    let solution = design.svd(true, true).solve(&observed, 1.0e-12).ok()?;
38    Some(solution.as_slice().to_vec())
39}

41/// Solve a symmetric system with casacore `LSQFit` ordering.
42///
43/// The input matrix must contain the symmetric system in its upper triangle.
44/// This preserves casacore's LDLT factorization and arithmetic order for
45/// algorithms whose serialized results depend on exact rounding.
46#[must_use]
47pub fn solve_symmetric_ldlt_casacore<const N: usize>(
48    mut normal: [[f64; N]; N],
49    known: [f64; N],
50) -> Option<[f64; N]> {
51    for row in 0..N {
52        let (prior_rows, current_and_after) = normal.split_at_mut(row);
53        let current_row = &mut current_and_after[0];
54        let original_diagonal = current_row[row];
55        if original_diagonal == 0.0 {
56            return None;
57        }
58        let mut diagonal = original_diagonal;
59        for (prior, prior_row) in prior_rows.iter().enumerate() {
60            diagonal -= prior_row[row] * prior_row[row] / prior_row[prior];
61        }
62        if !diagonal.is_finite() || diagonal * diagonal / original_diagonal <= 1.0e-12 {
63            return None;
64        }
65        current_row[row] = diagonal;
66        for column in row + 1..N {
67            for (prior, prior_row) in prior_rows.iter().enumerate() {
68                current_row[column] -= prior_row[row] * prior_row[column] / prior_row[prior];
69            }
70        }
71    }
72
73    let mut solution = [0.0; N];
74    for row in 0..N {
75        solution[row] = known[row];
76        for prior in 0..row {
77            solution[row] -= normal[prior][row] * solution[prior] / normal[prior][prior];
78        }
79    }
80    for row in (0..N).rev() {
81        for later in row + 1..N {
82            solution[row] -= normal[row][later] * solution[later];
83        }
84        solution[row] /= normal[row][row];
85    }
86    Some(solution)
87}

89#[cfg(test)]
90mod tests {
91    use super::{solve_symmetric_ldlt_casacore, solve_weighted_least_squares};
92
93    #[test]
94    fn weighted_linear_fit_recovers_line() {
95        let rows = vec![
96            (vec![1.0, 0.0], 2.0, 1.0),
97            (vec![1.0, 1.0], 5.0, 1.0),
98            (vec![1.0, 2.0], 8.0, 1.0),
99            (vec![1.0, 3.0], 11.0, 1.0),
100        ];
101        let solution = solve_weighted_least_squares(&rows, 2).expect("nonsingular system");
102        assert!((solution[0] - 2.0).abs() < 1.0e-10);
103        assert!((solution[1] - 3.0).abs() < 1.0e-10);
104    }
105
106    #[test]
107    fn invalid_weighted_rows_are_rejected() {
108        assert!(solve_weighted_least_squares(&[], 1).is_none());
109        assert!(solve_weighted_least_squares(&[(vec![1.0], 1.0, 0.0)], 1).is_none());
110        assert!(solve_weighted_least_squares(&[(vec![1.0], f64::NAN, 1.0)], 1).is_none());
111    }
112
113    #[test]
114    fn casacore_ldlt_solves_symmetric_system() {
115        let normal = [[4.0, 1.0], [0.0, 3.0]];
116        let solution =
117            solve_symmetric_ldlt_casacore(normal, [9.0, 7.0]).expect("positive-definite system");
118        assert!((solution[0] - 20.0 / 11.0).abs() < 1.0e-12);
119        assert!((solution[1] - 19.0 / 11.0).abs() < 1.0e-12);
120    }
121
122    #[test]
123    fn casacore_ldlt_rejects_singular_system() {
124        assert!(solve_symmetric_ldlt_casacore([[0.0]], [1.0]).is_none());
125    }
126}
