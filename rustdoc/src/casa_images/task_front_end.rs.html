2//! The image task binaries on the shared task front end (#931).
3//!
4//! Each binary's `main` builds its front end here, passing the surfaces it
5//! hosts and a callback that reads the resolved run into its typed request.
6//! The front end owns the command line, `--json-run`, Last and logging.

16#[cfg(casa_prebuilt_image_provider)]
17fn compiled_provider_matches_target() -> bool {
18    static MATCHES: std::sync::OnceLock<bool> = std::sync::OnceLock::new();
19    *MATCHES
20        .get_or_init(|| provider_matches_target(IMAGE_PROVIDER_SCHEMA_ORDER, IMAGE_PROVIDER_PROBES))
21}
22
23#[cfg(casa_prebuilt_image_provider)]
24fn provider_matches_target(order: bool, probes: &[(&str, u64, u64, u64)]) -> bool {
25    crate::analysis::schema_maps_preserve_order() == order
26        && casa_provider_contracts::builtin_provider_parser_matches(probes)
27}

29/// The front end of the image binary `program`, which hosts `surfaces`.
30pub fn image_task_front_end<F>(program: &str, surfaces: &[&str], execute: F) -> TaskFrontEnd<F>
31where
32    F: Fn(&TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure>,
33{
34    #[cfg(casa_prebuilt_image_provider)]
35    let compiled = compiled_provider_matches_target();
36    #[cfg(not(casa_prebuilt_image_provider))]
37    let compiled = false;
38    image_task_front_end_using(program, surfaces, execute, compiled)
39}
40
41fn image_task_front_end_using<F>(
42    program: &str,
43    surfaces: &[&str],
44    execute: F,
45    compiled: bool,
46) -> TaskFrontEnd<F>
47where
48    F: Fn(&TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure>,
49{
50    #[cfg(not(casa_prebuilt_image_provider))]
51    let _ = compiled;
52    #[cfg(casa_prebuilt_image_provider)]
53    if compiled {
54        let hosted = surfaces
55            .iter()
56            .map(|id| {
57                if !crate::analysis::IMAGE_ANALYSIS_TASK_SURFACES.contains(id) {
58                    return Err(format!("the provider contract carries no surface {id:?}"));
59                }
60                casa_provider_contracts::builtin_surface_bundle(id).map_err(|error| {
61                    format!("built-in image-analysis parameter surface {id:?}: {error}")
62                })
63            })
64            .collect::<Result<Vec<_>, _>>()
65            .expect("the image-analysis contract carries every image surface");
66        return TaskFrontEnd::new_with_compiled_contract(
67            program,
68            crate::image_analysis_protocol_descriptor(),
69            hosted,
70            include_str!(concat!(env!("OUT_DIR"), "/image-provider.json")),
71            execute,
72        )
73        .expect("the image-analysis contract carries every image surface");
74    }
75    TaskFrontEnd::new_with_deferred_schema(
76        program,
77        image_analysis_task_schema_bundle(),
78        surfaces,
79        execute,
80    )
81    .expect("the image-analysis contract carries every image surface")
82}

84/// A product-writing image task's reply: CASA's return value `None`, and
85/// its summary as the result, which a command line prints as JSON.
86pub fn image_task_success(
87    result: Result<ImageAnalysisTaskResult, ImageError>,
88) -> Result<TaskSuccess, TaskFailure> {
89    image_task_returning(CasaValue::None, result)
90}
91
92/// A product-writing image task's reply that returns CASA's `value`, as
93/// `immath` returns `True`.
94pub fn image_task_returning(
95    value: CasaValue,
96    result: Result<ImageAnalysisTaskResult, ImageError>,
97) -> Result<TaskSuccess, TaskFailure> {
98    let result = result.map_err(|error| TaskFailure::execution(error.to_string()))?;
99    let text = serde_json::to_string_pretty(&result)
100        .map_err(|error| TaskFailure::execution(format!("serialize the result: {error}")))?;
101    Ok(TaskSuccess::new(value, &result)?.with_text(text + "\n"))
102}

104#[cfg(test)]
105mod tests {
106    use std::ffi::OsString;
107
108    use casa_provider_contracts::CasaValue;
109    use casa_task_runtime::{TaskFrontEnd, TaskSuccess};
110
111    use super::{image_task_front_end, image_task_front_end_using};
112    use crate::image_analysis_task_schema_bundle;
113
114    #[test]
115    fn compiled_image_provider_matches_the_runtime_contract() {
116        let old = TaskFrontEnd::new_with_deferred_schema(
117            "imexplore",
118            image_analysis_task_schema_bundle(),
119            &["imhead", "imstat"],
120            |_| TaskSuccess::new(CasaValue::None, &()),
121        )
122        .unwrap();
123        let current = image_task_front_end("imexplore", &["imhead", "imstat"], |_| {
124            TaskSuccess::new(CasaValue::None, &())
125        });
126        let fallback = image_task_front_end_using(
127            "imexplore",
128            &["imhead", "imstat"],
129            |_| TaskSuccess::new(CasaValue::None, &()),
130            false,
131        );
132        #[cfg(casa_prebuilt_image_provider)]
133        let forced_compiled = super::compiled_provider_matches_target().then(|| {
134            image_task_front_end_using(
135                "imexplore",
136                &["imhead", "imstat"],
137                |_| TaskSuccess::new(CasaValue::None, &()),
138                true,
139            )
140        });
141        for option in ["--help", "--json-schema", "--protocol-info", "--show"] {
142            let args = [OsString::from("imhead"), OsString::from(option)];
143            assert_eq!(current.dispatch(&args), old.dispatch(&args), "{option}");
144            assert_eq!(fallback.dispatch(&args), old.dispatch(&args), "{option}");
145            #[cfg(casa_prebuilt_image_provider)]
146            if let Some(compiled) = &forced_compiled {
147                assert_eq!(compiled.dispatch(&args), old.dispatch(&args), "{option}");
148            }
149        }
150        let workspace = tempfile::tempdir().unwrap();
151        let request = workspace.path().join("invalid.json");
152        std::fs::write(
153            &request,
154            r#"{"surface":"imhead","values":{"imagename":123}}"#,
155        )
156        .unwrap();
157        let args = [
158            OsString::from("imhead"),
159            OsString::from("--json-run"),
160            request.into_os_string(),
161        ];
162        let refused = current.dispatch(&args);
163        assert_ne!(refused.code, 0);
164        assert_eq!(refused, old.dispatch(&args));
165        assert_eq!(fallback.dispatch(&args), refused);
166        #[cfg(casa_prebuilt_image_provider)]
167        if let Some(compiled) = &forced_compiled {
168            assert_eq!(compiled.dispatch(&args), refused);
169        }
170    }
171
172    #[cfg(casa_prebuilt_image_provider)]
173    #[test]
174    fn incompatible_target_parser_requires_runtime_contract() {
175        let order = super::IMAGE_PROVIDER_SCHEMA_ORDER;
176        let (token, direct, value, casa) = super::IMAGE_PROVIDER_PROBES[0];
177        assert!(!super::provider_matches_target(!order, &[]));
178        assert!(!super::provider_matches_target(
179            order,
180            &[(token, direct ^ 1, value, casa)]
181        ));
182    }
