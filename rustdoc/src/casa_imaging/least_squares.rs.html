4/// Solve a symmetric normal-equation system with casacore `LSQFit` ordering.
5///
6/// The input matrix must contain the symmetric system in its upper triangle.
7/// This deliberately preserves casacore's LDLᵀ factorization and arithmetic
8/// order so algorithms whose serialized results depend on exact rounding can
9/// share one implementation.
10pub(crate) fn solve_symmetric_ldlt_casacore<const N: usize>(
11    mut normal: [[f64; N]; N],
12    known: [f64; N],
13) -> Option<[f64; N]> {
14    for row in 0..N {
15        let (prior_rows, current_and_after) = normal.split_at_mut(row);
16        let current_row = &mut current_and_after[0];
17        let original_diagonal = current_row[row];
18        let mut diagonal = original_diagonal;
19        for (prior, prior_row) in prior_rows.iter().enumerate() {
20            diagonal -= prior_row[row] * prior_row[row] / prior_row[prior];
21        }
22        if !diagonal.is_finite() || diagonal * diagonal / original_diagonal <= 1.0e-12 {
23            return None;
24        }
25        current_row[row] = diagonal;
26        for column in row + 1..N {
27            for (prior, prior_row) in prior_rows.iter().enumerate() {
28                current_row[column] -= prior_row[row] * prior_row[column] / prior_row[prior];
29            }
30        }
31    }
32
33    let mut solution = [0.0; N];
34    for row in 0..N {
35        let (prior_solutions, current_and_after) = solution.split_at_mut(row);
36        let current_solution = &mut current_and_after[0];
37        *current_solution = known[row];
38        for (prior, prior_solution) in prior_solutions.iter().enumerate() {
39            *current_solution -= normal[prior][row] * *prior_solution / normal[prior][prior];
40        }
41    }
42    for row in (0..N).rev() {
43        let (_, current_and_after) = solution.split_at_mut(row);
44        let (current, later_solutions) = current_and_after.split_at_mut(1);
45        for (offset, later_solution) in later_solutions.iter().enumerate() {
46            current[0] -= normal[row][row + offset + 1] * *later_solution;
47        }
48        current[0] /= normal[row][row];
49    }
50    Some(solution)
51}

53#[cfg(test)]
54mod tests {
55    use super::solve_symmetric_ldlt_casacore;
56
57    #[test]
58    fn solves_symmetric_positive_definite_system() {
59        let normal = [[4.0, 1.0, 2.0], [1.0, 3.0, 0.0], [2.0, 0.0, 5.0]];
60        let known = [12.0, 7.0, 17.0];
61
62        let solution =
63            solve_symmetric_ldlt_casacore(normal, known).expect("nonsingular system should solve");
64
65        for (actual, expected) in solution.into_iter().zip([1.0, 2.0, 3.0]) {
66            assert!((actual - expected).abs() <= 3.0 * f64::EPSILON);
67        }
68    }
69
70    #[test]
71    fn rejects_singular_system() {
72        let normal = [[1.0, 1.0], [1.0, 1.0]];
73
74        assert!(solve_symmetric_ldlt_casacore(normal, [1.0, 1.0]).is_none());
75    }
76}
