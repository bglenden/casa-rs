13/// Grid arguments of the legacy in-place MeasurementSet spectral tools.
14#[derive(Clone, Copy, Debug)]
15pub enum LegacySpectralGrid {
16    /// The mode, start and channel count established by `cvel()`.
17    Cvel,
18    /// Convert frequency metadata without resampling.
19    FrameOnly,
20    /// The numeric band in frequency, velocity or wavelength units.
21    CenteredNumeric {
22        /// Units for the centre, bandwidth and channel width.
23        quantity: LegacyRegridQuantity,
24        /// Centre of the new spectral window in the requested units.
25        centre: f64,
26        /// Width of the new spectral window in the requested units.
27        bandwidth: f64,
28        /// Width of one new channel in the requested units.
29        channel_width: f64,
30        /// Rest frequency in Hz, used for the two velocity conventions.
31        rest_hz: f64,
32    },
33    /// The numeric band supplied to `regridspw(mode="chan")`.
34    CenteredChannels {
35        /// Centre expressed as an original channel index.
36        centre: f64,
37        /// Bandwidth in units of original channels.
38        bandwidth: f64,
39        /// Width of a full output channel in units of original channels.
40        channel_width: f64,
41    },
42}
43
44impl LegacySpectralGrid {
45    /// Whether this `regridspw` grid leaves the input channel centres unchanged.
46    /// Callers compare the frequency references separately.
47    pub fn leaves_band_unchanged(self, centres: &[f64], widths: &[f64]) -> bool {
48        match self {
49            Self::FrameOnly => true,
50            Self::Cvel => false,
51            _ => self
52                .centered_frequencies(centres, widths, centres.len())
53                .is_ok_and(|axis| axis.centres_hz == centres),
54        }
55    }
56
57    pub(super) fn centered_frequencies(
58        self,
59        centres: &[f64],
60        widths: &[f64],
61        maximum_channels: usize,
62    ) -> Result<CasaChannelFrequencies, MsTransformError> {
63        match self {
64            Self::CenteredChannels {
65                centre,
66                bandwidth,
67                channel_width,
68            } => centered_channel_bounds(centres, widths, centre, bandwidth, channel_width),
69            Self::CenteredNumeric {
70                quantity,
71                centre,
72                bandwidth,
73                channel_width,
74                rest_hz,
75            } => casa_ms::centered_numeric_frequencies(
76                quantity,
77                (centre, bandwidth, channel_width, rest_hz),
78                centres,
79                widths,
80                maximum_channels,
81            )
82            .map_err(super::ms_error),
83            _ => unreachable!("only centered grids use these bounds"),
84        }
85    }
86}

88pub(super) fn output_axis(
89    options: &MsRegridOptions,
90    centres: &[f64],
91    widths: &[f64],
92    maximum_channels: usize,
93) -> Result<CasaChannelFrequencies, MsTransformError> {
94    let error = MsTransformError::Spectrum;
95    let mode = options.mode.as_str();
96    let length = centres.len();
97    if length == 0 || length != widths.len() {
98        return Err(error(
99            "Regridding needs matching non-empty frequency and width arrays.".into(),
100        ));
101    }
102    let count = usize::try_from(options.nchan)
103        .ok()
104        .filter(|&n| n > 0)
105        .map(|channels| {
106            channels
107                .checked_mul(options.nspw.max(1))
108                .filter(|&count| count <= i32::MAX as usize)
109                .ok_or_else(|| {
110                    error("Regridding channel count exceeds the MeasurementSet limit.".into())
111                })
112        })
113        .transpose()?;
114    let rest_frequency_hz = if options.restfreq.trim().is_empty() {
115        if mode == "velocity" {
116            return Err(MsTransformError::RegridParameter {
117                parameter: "restfreq",
118                reason: "Parameter restfreq must be set when mode='velocity'".into(),
119            });
120        }
121        0.0
122    } else {
123        parse_rest_frequency_hz(&options.restfreq).map_err(|source| error(source.to_string()))?
124    };
125    let mut request = RegridRequest {
126        quantity: RegridQuantity::Frequency,
127        start: None,
128        width: None,
129        nchan: count,
130        rest_frequency_hz,
131    };
132    match mode {
133        "channel" => {
134            let start = channel_arg(&options.start, 0);
135            let step = channel_arg(&options.width, 1);
136            let descending = centres.first() > centres.last();
137            let first = if descending { length - 1 } else { 0 };
138            let start = usize::try_from(start).ok();
139            let selected = start.filter(|&value| value < length).unwrap_or(first);
140            let mut width = widths[first].abs() * f64::from(step.unsigned_abs().max(1));
141            let mut start_is_end = (step < 0) ^ descending;
142            if start.is_some_and(|index| index >= length) {
143                width = width.abs();
144                start_is_end = false;
145            }
146            let edge = if start_is_end {
147                centres[selected] + widths[selected].abs() / 2.
148            } else {
149                centres[selected] - widths[selected].abs() / 2.
150            };
151            request.start = start.map(|_| {
152                if start_is_end {
153                    edge - width / 2.
154                } else {
155                    edge + width / 2.
156                }
157            });
158            request.width = Some(if start_is_end { -width } else { width });
159        }
160        "channel_b" => return channel_bounds(options, centres, widths, count),
161        "frequency" => {
162            request.start = quantity_arg("start", &options.start, "Hz", "0")?;
163            request.width = quantity_arg("width", &options.width, "Hz", "1")?;
164        }
165        "velocity" => {
166            request.quantity = match options.veltype.to_ascii_lowercase().as_str() {
167                "optical" => RegridQuantity::OpticalVelocity,
168                "radio" => RegridQuantity::RadioVelocity,
169                _ => return Err(super::invalid_choice("veltype", &options.veltype)),
170            };
171            request.start = quantity_arg("start", &options.start, "m/s", "0")?;
172            request.width = quantity_arg("width", &options.width, "m/s", "1")?;
173        }
174        _ => return Err(super::invalid_choice("mode", mode)),
175    }
176    casa_data_channel_frequencies(&request, centres, widths, maximum_channels)
177        .map_err(|source| error(source.to_string()))
178}

180fn channel_arg(value: &str, default: i32) -> i32 {
181    let value = value.trim();
182    if value.is_empty() || value == "[]" {
183        default
184    } else {
185        value.parse().unwrap_or_default()
186    }
187}
188
189fn quantity_arg(
190    parameter: &'static str,
191    value: &str,
192    unit: &str,
193    default: &str,
194) -> Result<Option<f64>, MsTransformError> {
195    let value = value.trim();
196    if value.is_empty() || value == "[]" || value == default {
197        return Ok(None);
198    }
199    let unit = Unit::new(unit).expect("regridding uses a known frequency or velocity unit");
200    let valid = Quantity::read(value)
201        .ok()
202        .and_then(|quantity| quantity.get_value_in(&unit).ok())
203        .filter(|number| number.is_finite());
204    valid.map(Some).ok_or_else(|| {
205        let dimension = if unit.name() == "Hz" {
206            "frequency"
207        } else {
208            "velocity"
209        };
210        MsTransformError::RegridParameter {
211            parameter,
212            reason: format!("Parameter {parameter} does not have valid {dimension} units"),
213        }
214    })
215}

217fn channel_bounds(
218    options: &MsRegridOptions,
219    frequencies: &[f64],
220    widths: &[f64],
221    count: Option<usize>,
222) -> Result<CasaChannelFrequencies, MsTransformError> {
223    let channel_count = frequencies.len() as i64;
224    let step = channel_arg(&options.width, 1);
225    let requested_step = i64::from(step.unsigned_abs().max(1));
226    let requested_start = channel_arg(&options.start, 0);
227    let descending = frequencies.first() > frequencies.last();
228    let starts_high = (step < 0) ^ descending;
229    let mut span = count.map_or(channel_count, |channels| channels as i64 * requested_step);
230    let from = i64::from(requested_start);
231    if from < 0 || from >= channel_count {
232        return Err(MsTransformError::Spectrum(format!(
233            "SPW start {requested_start} outside valid range 0–{}.",
234            channel_count - 1
235        )));
236    }
237    let from_ascending = if descending {
238        channel_count - 1 - from
239    } else {
240        from
241    };
242    let centre = if starts_high {
243        from_ascending - span / 2
244    } else {
245        from_ascending + span / 2
246    };
247    if centre - span / 2 < 0 {
248        span = 2 * centre + 1;
249    }
250    if centre + span / 2 > channel_count {
251        span = 2 * (channel_count - centre);
252    }
253    if span <= 0 {
254        return Err(MsTransformError::Spectrum(
255            "Requested channel bounds leave an empty SPW.".into(),
256        ));
257    }
258    channel_groups(frequencies, widths, centre, span, requested_step.min(span))
259}

261/// Bound the legacy centre-based channel grid of SubMS::regridChanBounds.
262/// A partial endpoint remains a channel instead of being dropped.
263pub(super) fn centered_channel_bounds(
264    frequencies: &[f64],
265    widths: &[f64],
266    centre: f64,
267    bandwidth: f64,
268    channel_width: f64,
269) -> Result<CasaChannelFrequencies, MsTransformError> {
270    let len = frequencies.len() as i64;
271    let error = MsTransformError::Spectrum;
272    if len == 0 || frequencies.len() != widths.len() {
273        return Err(error(
274            "Regridding needs matching non-empty frequency and width arrays.".into(),
275        ));
276    }
277    let descending = frequencies.first() > frequencies.last();
278    let centre = if centre < -1e30 {
279        let midpoint = (frequencies[0] + frequencies[frequencies.len() - 1]) / 2.0;
280        let midpoint_row = if descending {
281            frequencies
282                .iter()
283                .rev()
284                .position(|frequency| *frequency >= midpoint)
285        } else {
286            frequencies
287                .iter()
288                .position(|frequency| *frequency >= midpoint)
289        };
290        midpoint_row.ok_or_else(|| error("Invalid spectral window centre.".into()))? as i64
291    } else if (0.0..len as f64).contains(&centre) {
292        centre.floor() as i64
293    } else {
294        return Err(error(format!(
295            "SPW center {centre} outside valid range 0–{}.",
296            len - 1
297        )));
298    };
299    let mut span = if bandwidth <= 0.0 {
300        len
301    } else {
302        bandwidth.floor() as i64
303    };
304    if centre - span / 2 < 0 {
305        span = 2 * centre + 1;
306    }
307    if centre + span / 2 > len {
308        span = 2 * (len - centre);
309    }
310    if span <= 0 {
311        return Err(error("Requested channel bounds leave an empty SPW.".into()));
312    }
313    let step = (channel_width.floor() as i64).clamp(1, span);
314    channel_groups(frequencies, widths, centre, span, step)
315}

317fn channel_groups(
318    frequencies: &[f64],
319    widths: &[f64],
320    centre: i64,
321    span: i64,
322    step: i64,
323) -> Result<CasaChannelFrequencies, MsTransformError> {
324    let channel_count = frequencies.len() as i64;
325    let descending = frequencies.first() > frequencies.last();
326    let phys = |index: i64| {
327        if descending {
328            (channel_count - 1 - index) as usize
329        } else {
330            index as usize
331        }
332    };
333    let lower = centre - span / 2;
334    let upper = (lower + span).min(channel_count);
335    if lower < 0 || lower >= upper {
336        return Err(MsTransformError::Spectrum(
337            "Requested channel bounds leave an empty SPW.".into(),
338        ));
339    }
340    let central_bin = if (span / step) % 2 == 1 {
341        centre - step / 2
342    } else {
343        centre
344    };
345    let central_bin = central_bin.clamp(lower, upper);
346    let mut bins = Vec::new();
347    let mut right = central_bin;
348    while right > lower {
349        let left = lower.max(right - step);
350        bins.push((left, right - 1));
351        right = left;
352    }
353    bins.reverse();
354    for left in (central_bin..upper).step_by(step as usize) {
355        bins.push((left, upper.min(left + step) - 1));
356    }
357    if descending {
358        bins.reverse();
359    }
360    let (centres_hz, steps_hz) = bins
361        .into_iter()
362        .map(|(low, high)| {
363            let lo = frequencies[phys(low)] - widths[phys(low)].abs() / 2.;
364            let hi = frequencies[phys(high)] + widths[phys(high)].abs() / 2.;
365            ((hi + lo) / 2., hi - lo)
366        })
367        .unzip();
368    Ok(CasaChannelFrequencies {
369        centres_hz,
370        steps_hz,
371    })
372}

374#[cfg(test)]
375mod tests {
376    use super::centered_channel_bounds;
377
378    #[test]
379    fn a_centered_band_keeps_partial_endpoints_in_the_input_order() {
380        let centres = [0., 1., 2., 3., 4., 5., 6.];
381        let widths = [1.; 7];
382        let ascending = centered_channel_bounds(&centres, &widths, -3e30, -1., 2.).unwrap();
383        assert_eq!(ascending.centres_hz, [0.5, 2.5, 4.5, 6.]);
384        assert_eq!(ascending.steps_hz, [2., 2., 2., 1.]);
385
386        let descending =
387            centered_channel_bounds(&[6., 5., 4., 3., 2., 1., 0.], &widths, -3e30, -1., 2.)
388                .unwrap();
389        assert_eq!(descending.centres_hz, [6., 4.5, 2.5, 0.5]);
390        assert_eq!(descending.steps_hz, [1., 2., 2., 2.]);
391    }
392
393    #[test]
394    fn a_shifted_midpoint_never_selects_an_out_of_bounds_channel() {
395        let centres = [0., 1., 2., 2.875, 4., 5., 6.];
396        // The first centre at or above 3 Hz is index 4. A full seven-channel
397        // span centered there reaches one past the input's high edge.
398        let grid = centered_channel_bounds(&centres, &[1.; 7], -3e30, -1., 2.).unwrap();
399        assert_eq!(grid.centres_hz, [1.5, 3.4375, 5.5]);
400        assert_eq!(grid.steps_hz, [2., 2.125, 2.]);
401    }
402
403    #[test]
404    fn the_vendored_regridspw_band_has_sixteen_two_channel_bins() {
405        let channels = std::array::from_fn::<_, 63, _>(|slot| slot as f64);
406        let axis = centered_channel_bounds(&channels, &[1.; 63], 24., 32., 2.).unwrap();
407        assert_eq!(axis.centres_hz.len(), 16);
408        assert_eq!(axis.centres_hz[0], 8.5);
409        assert_eq!(axis.centres_hz[15], 38.5);
410        assert_eq!(axis.steps_hz, [2.; 16]);
411    }
