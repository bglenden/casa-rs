2//! Visibility-equation counters from the terms executed by the Python tool.
3
4use std::ops::Range;
5
6use casa_types::{RecordField, RecordValue, ScalarValue, Value};
7
8use super::ApplyPlan;
9use super::chain::viscal_order;
10
11pub(crate) struct LibraryApplyRegistration {
12    pub(crate) name: String,
13    pub(crate) subtype: String,
14    pub(crate) tables: Range<usize>,
15}
16
17#[derive(Clone, Copy, Default)]
18pub(crate) struct ApplyTermCounts {
19    data: usize,
20    flags_in: usize,
21    flags_out: usize,
22}
23
24struct ApplyTermActivity {
25    name: Option<String>,
26    subtype: String,
27    position: (usize, u8),
28    counts: ApplyTermCounts,
29}
30
31pub(crate) struct ToolApplyActivity {
32    terms: Vec<ApplyTermActivity>,
33    table_terms: Vec<usize>,
34    first_parang_term: usize,
35    visibility_count: usize,
36}
37
38impl ToolApplyActivity {
39    pub(crate) fn from_plan(
40        plan: &ApplyPlan,
41        table_names: &[String],
42        library_terms: &[LibraryApplyRegistration],
43        parang_terms: usize,
44        opacity: &[(&'static str, usize)],
45        previous: ApplyTermCounts,
46    ) -> Self {
47        let mut terms = Vec::new();
48        let mut table_terms = vec![None; plan.calibration_tables.len()];
49        for registration in library_terms {
50            let term = terms.len();
51            for index in registration.tables.clone() {
52                table_terms[index] = Some(term);
53            }
54            terms.push(ApplyTermActivity {
55                name: Some(registration.name.clone()),
56                subtype: registration.subtype.clone(),
57                position: (registration.tables.start, 1),
58                counts: ApplyTermCounts::default(),
59            });
60        }
61        for (index, (table, name)) in plan.calibration_tables.iter().zip(table_names).enumerate() {
62            if table_terms[index].is_none() {
63                table_terms[index] = Some(terms.len());
64                terms.push(ApplyTermActivity {
65                    name: Some(name.clone()),
66                    subtype: table.summary.table_subtype.clone(),
67                    position: (index, 1),
68                    counts: ApplyTermCounts::default(),
69                });
70            }
71        }
72        terms.extend(opacity.iter().map(|&(subtype, index)| ApplyTermActivity {
73            name: Some(String::new()),
74            subtype: subtype.into(),
75            position: (index, 0),
76            counts: ApplyTermCounts::default(),
77        }));
78        let first_parang_term = terms.len();
79        terms.extend((0..parang_terms).map(|index| ApplyTermActivity {
80            name: None,
81            subtype: "P Jones".into(),
82            position: (plan.calibration_tables.len() + index, 1),
83            counts: ApplyTermCounts::default(),
84        }));
85        if let [term] = terms.as_mut_slice() {
86            // VisEquation resets term counters when it sorts a chain with
87            // multiple entries. A single term keeps accumulating its flags.
88            term.counts = previous;
89        }
90        Self {
91            terms,
92            table_terms: table_terms
93                .into_iter()
94                .map(|term| term.expect("planned apply term"))
95                .collect(),
96            first_parang_term,
97            visibility_count: 0,
98        }
99    }
100
101    pub(super) fn begin_row(&mut self, visibilities: usize) {
102        self.visibility_count += visibilities;
103    }
104
105    pub(super) fn table_term(&self, table: usize) -> usize {
106        self.table_terms[table]
107    }
108
109    pub(super) fn parallactic_term(&self, registration: usize) -> usize {
110        self.first_parang_term + registration
111    }
112
113    pub(super) fn count_term(
114        &mut self,
115        index: usize,
116        visibilities: usize,
117        flags_in: usize,
118        flags_out: usize,
119    ) {
120        let counts = &mut self.terms[index].counts;
121        counts.data += visibilities;
122        counts.flags_in += flags_in;
123        counts.flags_out += flags_out;
124    }
125
126    pub(crate) fn single_term_counts(&self) -> ApplyTermCounts {
127        match self.terms.as_slice() {
128            [term] => term.counts,
129            _ => ApplyTermCounts::default(),
130        }
131    }
132
133    pub(crate) fn into_record(mut self, origin: &str) -> RecordValue {
134        self.terms
135            .sort_by_key(|term| (viscal_order(&term.subtype), term.position));
136        let mut entries = self
137            .terms
138            .into_iter()
139            .enumerate()
140            .map(|(index, term)| {
141                let counts = term.counts;
142                let mut fields = vec![
143                    RecordField::new(
144                        "ndata",
145                        Value::Scalar(ScalarValue::Int64(counts.data as i64)),
146                    ),
147                    RecordField::new(
148                        "nflagIn",
149                        Value::Scalar(ScalarValue::Int64(counts.flags_in as i64)),
150                    ),
151                    RecordField::new(
152                        "nflagOut",
153                        Value::Scalar(ScalarValue::Int64(counts.flags_out as i64)),
154                    ),
155                ];
156                if let Some(table) = term.name {
157                    fields.push(RecordField::new(
158                        "table",
159                        Value::Scalar(ScalarValue::String(table)),
160                    ));
161                }
162                fields.push(RecordField::new(
163                    "type",
164                    Value::Scalar(ScalarValue::String(term.subtype)),
165                ));
166                RecordField::new(
167                    format!("*{}", index + 1),
168                    Value::Record(RecordValue::new(fields)),
169                )
170            })
171            .collect::<Vec<_>>();
172        entries.sort_by(|left, right| left.name.cmp(&right.name));
173        entries.push(RecordField::new(
174            "nVisTotal",
175            Value::Scalar(ScalarValue::Int64(self.visibility_count as i64)),
176        ));
177        RecordValue::new(vec![
178            RecordField::new("VisEquation", Value::Record(RecordValue::new(entries))),
179            RecordField::new(
180                "origin",
181                Value::Scalar(ScalarValue::String(origin.to_string())),
182            ),
183        ])
184    }
185}
