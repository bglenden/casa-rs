2//! The requests the batch route binds and estimates: a partition's payload
3//! for a wave, and its payload-free survey.
4
5use std::ops::Range;
6
7use casa_imaging_model::GeometryPlan;
8use casa_observation::{
9    AccessRequest, CorrelationRequest, Deliver, FlagsFrom, Grouping, PartitionEntry, Roles,
10    TimeSelection, WeightRequest,
11};
12
13use super::selection::{PartitionSelection, SelectedPartition};
14
15/// The request reading `partition`'s (`entry` in the catalogue) native
16/// channels `channels`: its selected fields' times, its routed products
17/// with their own flags, the stored weights, UVW, the time centroids when
18/// `plan` reads them and, with `values`, the visibilities, at most
19/// `batch_bytes` a batch.
20pub(super) fn payload(
21    (selection, plan): (&PartitionSelection, &GeometryPlan),
22    entry: &PartitionEntry,
23    partition: &SelectedPartition,
24    channels: Range<usize>,
25    values: bool,
26    batch_bytes: usize,
27) -> AccessRequest {
28    let deliver = if partition.correlations == entry.correlations {
29        Deliver::All
30    } else {
31        Deliver::These(partition.correlations.clone())
32    };
33    AccessRequest {
34        partitions: vec![partition.id],
35        data_group: selection.data_group.clone(),
36        roles: Roles {
37            values,
38            time_centroid: plan.reads_time_centroids(),
39            ..Roles::PAYLOAD
40        },
41        weights: WeightRequest::Stored,
42        channels,
43        times: TimeSelection::All,
44        fields: selection.fields.clone(),
45        correlations: CorrelationRequest {
46            deliver,
47            flags_from: FlagsFrom::Delivered,
48        },
49        grouping: Grouping::None,
50        batch_bytes,
51    }
52}
53
54/// The request surveying `partition`: its selected fields' times, and
55/// their time centroids when `plan` reads them, of one channel and no
56/// payload, at most `batch_bytes` a batch.
57pub(super) fn survey(
58    (selection, plan): (&PartitionSelection, &GeometryPlan),
59    entry: &PartitionEntry,
60    partition: &SelectedPartition,
61    batch_bytes: usize,
62) -> AccessRequest {
63    let first = partition.channels[0] as usize;
64    let request = payload(
65        (selection, plan),
66        entry,
67        partition,
68        first..first + 1,
69        false,
70        batch_bytes,
71    );
72    AccessRequest {
73        roles: Roles {
74            values: false,
75            flags: false,
76            weights: false,
77            uvw: false,
78            ..request.roles
79        },
80        ..request
81    }
82}
