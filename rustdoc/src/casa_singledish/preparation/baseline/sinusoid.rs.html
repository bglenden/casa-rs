12/// Invalid wave selector error message.
13pub const INVALID_WAVE: &str = "wrong value given for addwn/rejwn";
14const NO_WAVES: &str = "No effective wave number given for sinusoidal fitting.";
15pub(super) const INVALID_FIT: &str = "sakura_LSQFitSinusoidFloat() failed -- InvalidArgument";
16
17/// Wave-number selection for sinusoidal baseline fitting.
18#[derive(Clone)]
19pub enum WaveSelection {
20    /// Explicit 1-based/0-based integer list.
21    Explicit(Vec<usize>),
22    /// Exact wave indices.
23    Exact(Vec<usize>),
24    /// Closed `[first, last]` wave interval.
25    Interval {
26        /// Lower bound.
27        first: usize,
28        /// Upper bound.
29        last: usize,
30    },
31    /// All waves from this threshold through Nyquist.
32    ThroughNyquist(usize),
33}
34
35impl WaveSelection {
36    /// Construct an explicit wave list from integer arguments.
37    pub fn explicit(numbers: impl IntoIterator<Item = i64>) -> Result<Self, String> {
38        numbers
39            .into_iter()
40            .map(|number| usize::try_from(number).map_err(|_| INVALID_WAVE.into()))
41            .collect::<Result<Vec<_>, _>>()
42            .map(Self::Explicit)
43    }
44
45    /// Parse wave-number specification strings into a selection.
46    pub fn from_task_text(text: &str) -> Result<Self, String> {
47        let index = |value: &str| value.trim().parse::<usize>().map_err(|_| INVALID_WAVE);
48        if text.contains(',') {
49            let indexes = text
50                .split(',')
51                .map(index)
52                .collect::<Result<BTreeSet<_>, _>>()?;
53            return Ok(Self::Explicit(indexes.into_iter().collect()));
54        }
55        if let Some(delimiter) = ['-', '~']
56            .into_iter()
57            .find(|&delimiter| text.contains(delimiter))
58        {
59            let bounds = text
60                .split(delimiter)
61                .map(index)
62                .collect::<Result<Vec<_>, _>>()?;
63            let [first, last, ..] = bounds.as_slice() else {
64                return Err(INVALID_WAVE.into());
65            };
66            let (first, last) = (*first, *last);
67            return Ok(Self::Interval {
68                first: first.min(last),
69                last: first.max(last),
70            });
71        }
72        if let Some(last) = text
73            .strip_prefix("<=")
74            .or_else(|| text.strip_prefix("=<"))
75            .or_else(|| text.strip_suffix(">="))
76            .or_else(|| text.strip_suffix("=>"))
77        {
78            return Ok(Self::Interval {
79                first: 0,
80                last: index(last)?,
81            });
82        }
83        if let Some(last) = text.strip_prefix('<').or_else(|| text.strip_suffix('>')) {
84            return Ok(Self::Interval {
85                first: 0,
86                last: index(last)?.checked_sub(1).ok_or(INVALID_WAVE)?,
87            });
88        }
89        if let Some(first) = text
90            .strip_prefix(">=")
91            .or_else(|| text.strip_prefix("=>"))
92            .or_else(|| text.strip_suffix("<="))
93            .or_else(|| text.strip_suffix("=<"))
94        {
95            return Ok(Self::ThroughNyquist(index(first)?));
96        }
97        if let Some(first) = text.strip_prefix('>').or_else(|| text.strip_suffix('<')) {
98            return Ok(Self::ThroughNyquist(
99                index(first)?.checked_add(1).ok_or(INVALID_WAVE)?,
100            ));
101        }
102        Ok(Self::Explicit(vec![index(text)?]))
103    }
104
105    fn from_tool_text(text: &str) -> Self {
106        let values = text
107            .split(',')
108            .filter(|term| !term.is_empty())
109            .map(c_integer)
110            .collect::<Vec<_>>();
111        if let [first, -999] = values.as_slice() {
112            return Self::ThroughNyquist(usize::try_from(*first).unwrap_or_default());
113        }
114        Self::Explicit(
115            values
116                .into_iter()
117                .filter_map(|value| usize::try_from(value).ok())
118                .collect(),
119        )
120    }
121
122    fn includes(&self, wave: usize) -> bool {
123        match self {
124            Self::Explicit(values) | Self::Exact(values) => values.contains(&wave),
125            Self::Interval { first, last } => (*first..=*last).contains(&wave),
126            Self::ThroughNyquist(first) => wave >= *first,
127        }
128    }
129
130    fn rejects_fft(&self, wave: usize) -> bool {
131        match self {
132            // CASA's merge sees the original two-element sentinel representation.
133            Self::ThroughNyquist(first) => wave == *first,
134            _ => self.includes(wave),
135        }
136    }
137}

139// The direct binding passes each comma-delimited token through atoi.
140fn c_integer(text: &str) -> i32 {
141    parse_integer_prefix(text).unwrap_or_default() as i32
142}
143
144fn parse_integer_prefix(text: &str) -> Option<i64> {
145    let text = text.trim_start();
146    let digits = text.strip_prefix(['-', '+']).unwrap_or(text);
147    let count = digits.bytes().take_while(u8::is_ascii_digit).count();
148    let sign = text.len() - digits.len();
149    text[..sign + count].parse().ok()
150}

152#[derive(Clone, Copy)]
153/// FFT peak detection threshold for automatic sinusoidal wave selection.
154pub enum FftThreshold {
155    /// Signal-to-noise ratio threshold.
156    Sigma(f32),
157    /// Fixed number of top Fourier components.
158    Top(usize),
159}
160
161impl FftThreshold {
162    /// Validate an `fftthresh` parameter string.
163    pub fn validate(text: &str) -> Result<(), String> {
164        const FORMAT: &str = "fftthresh has a wrong format.";
165        const POSITIVE: &str = "threshold given to fftthresh must be positive.";
166        let not_positive = if let Some(top) = text.strip_prefix("top") {
167            top.trim().parse::<i64>().map_err(|_| FORMAT)? <= 0
168        } else {
169            text.strip_suffix("sigma")
170                .unwrap_or(text)
171                .trim()
172                .parse::<f64>()
173                .map_err(|_| FORMAT)?
174                <= 0.
175        };
176        if not_positive {
177            Err(POSITIVE.into())
178        } else {
179            Ok(())
180        }
181    }
182
183    fn from_tool(text: &str) -> Result<Self, &'static str> {
184        const INVALID: &str = "fftthresh has a wrong value";
185        if let Some(count) = text.strip_prefix("top") {
186            return parse_integer_prefix(count)
187                .and_then(|count| i32::try_from(count).ok())
188                .map(|count| Self::Top(count.max(0) as usize))
189                .ok_or(INVALID);
190        }
191        let sigma = if let Some(sigma) = text.strip_suffix("sigma") {
192            sigma
193        } else if text.is_empty()
194            || !text
195                .bytes()
196                .take(text.len() - 1)
197                .all(|byte| byte.is_ascii_digit() || byte == b'.')
198        {
199            return Err(INVALID);
200        } else {
201            text
202        };
203        let sigma = sigma.trim_start();
204        let prefix = sigma
205            .bytes()
206            .take_while(|&byte| {
207                byte.is_ascii_digit() || matches!(byte, b'+' | b'-' | b'.' | b'e' | b'E')
208            })
209            .count();
210        sigma[..prefix]
211            .parse::<f32>()
212            .ok()
213            .filter(|value| value.is_finite())
214            .map(Self::Sigma)
215            .ok_or(INVALID)
216    }
217}

219/// Wave selection for the sinusoidal single-dish baseline.
220pub struct SingleDishSinusoid {
221    include: WaveSelection,
222    exclude: WaveSelection,
223    fft: Option<Result<FftThreshold, &'static str>>,
224}
225
226impl SingleDishSinusoid {
227    /// Interpret the comma-delimited wavenumbers from the direct CASA tool.
228    /// A final sentinel of `-999` denotes an interval through Nyquist.
229    pub fn from_tool(
230        addwn: &str,
231        rejwn: &str,
232        applyfft: bool,
233        fftmethod: &str,
234        fftthresh: &str,
235    ) -> Result<Self, String> {
236        if addwn.split(',').all(str::is_empty) {
237            return Err("addwn must contain at least one element.".into());
238        }
239        Ok(Self::new(
240            WaveSelection::from_tool_text(addwn),
241            WaveSelection::from_tool_text(rejwn),
242            applyfft,
243            fftmethod,
244            fftthresh,
245        ))
246    }
247
248    pub(super) fn from_variable_waves(waves: Vec<usize>) -> Self {
249        Self {
250            include: WaveSelection::Exact(waves),
251            exclude: WaveSelection::Explicit(Vec::new()),
252            fft: None,
253        }
254    }
255
256    pub(super) fn variable_waves(&self) -> &[usize] {
257        match &self.include {
258            WaveSelection::Exact(waves) => waves,
259            _ => &[],
260        }
261    }
262
263    /// Construct a sinusoidal baseline model from parsed wave selectors and FFT options.
264    pub fn new(
265        include: WaveSelection,
266        exclude: WaveSelection,
267        applyfft: bool,
268        fftmethod: &str,
269        fftthresh: &str,
270    ) -> Self {
271        let fft = applyfft.then(|| {
272            if fftmethod != "fft" {
273                return Err("fftmethod must be 'fft' for now.");
274            }
275            FftThreshold::from_tool(fftthresh)
276        });
277        Self {
278            include,
279            exclude,
280            fft,
281        }
282    }

284    pub(super) fn fit(
285        &self,
286        spectrum: &[f32],
287        included: &[bool],
288        clip_sigma: f64,
289        max_fits: usize,
290        has_unflagged_data: bool,
291    ) -> Result<Option<(Fit, Vec<usize>)>, String> {
292        let maximum = (spectrum.len() / 2).checked_sub(1).ok_or(NO_WAVES)?;
293        let extended = matches!(self.exclude, WaveSelection::ThroughNyquist(_));
294        let first_rejected = match self.exclude {
295            WaveSelection::ThroughNyquist(first) => Some(first),
296            _ => None,
297        };
298        let mut waves = match (&self.include, first_rejected) {
299            (WaveSelection::Exact(waves), _) => waves.clone(),
300            (WaveSelection::Explicit(values), _) => values
301                .iter()
302                .copied()
303                .filter(|&wave| (extended || wave <= maximum) && !self.exclude.includes(wave))
304                .collect::<Vec<_>>(),
305            (WaveSelection::Interval { first, last }, Some(rejected)) => {
306                let last = (*last).min(rejected.checked_sub(1).ok_or(NO_WAVES)?);
307                if last < *first {
308                    return Err(NO_WAVES.into());
309                }
310                validate_context(last, spectrum.len())?;
311                (*first..=last).collect()
312            }
313            _ => (0..=maximum)
314                .filter(|&wave| self.include.includes(wave) && !self.exclude.includes(wave))
315                .collect(),
316        };
317        let base_maximum = *waves.last().ok_or(NO_WAVES)?;
318        if self.fft.is_none() || extended {
319            validate_context(base_maximum, spectrum.len())?;
320        }
321        if !has_unflagged_data {
322            return Ok(None);
323        }
324        let base_len = waves.len();
325        if let Some(choice) = self.fft {
326            let choice = choice?;
327            let selected = peaks(spectrum, included, maximum, choice)?;
328            for wave in selected {
329                if !self.exclude.rejects_fft(wave) && !waves[..base_len].contains(&wave) {
330                    waves.push(wave);
331                }
332            }
333            waves.sort_unstable();
334        }
335        let terms = terms(&waves);
336        if included.iter().filter(|&&value| value).count() < terms {
337            return Ok(None);
338        }
339        let largest = *waves.last().expect("nonempty waves");
340        if self.fft.is_some() && !extended {
341            validate_context(largest, spectrum.len())?;
342        }
343        if (extended && largest > base_maximum) || !waves.windows(2).all(|pair| pair[0] < pair[1]) {
344            return Err(INVALID_FIT.into());
345        }
346        let size = spectrum.len();
347        let basis = (0..size)
348            .map(|channel| basis(channel, size, &waves))
349            .collect();
350        let linear = LinearFit {
351            basis,
352            terms,
353            segment_ends: vec![size],
354            failure: "sakura_LSQFitSinusoidFloat() failed -- NG",
355        };
356        Ok(linear
357            .solve(spectrum, included, clip_sigma, max_fits, None)?
358            .map(|fit| (fit, waves)))
359    }
360}

362pub(super) fn validate_context(wave: usize, channels: usize) -> Result<(), String> {
363    if wave.saturating_mul(2).saturating_add(2) > channels {
364        Err("sakura_CreateLSQFitContextSinusoidFloat() failed -- InvalidArgument".into())
365    } else {
366        Ok(())
367    }
368}

370fn peaks(
371    spectrum: &[f32],
372    included: &[bool],
373    maximum: usize,
374    threshold: FftThreshold,
375) -> Result<Vec<usize>, String> {
376    let samples = interpolate(spectrum, included)?;
377    let mut fft = RealFft2::<f32>::with_threads([1, spectrum.len()], 1)
378        .map_err(|error| error.to_string())?
379        .with_estimated_plan();
380    let mut storage = vec![Complex32::new(0., 0.); fft.storage_len()];
381    for (i, sample) in samples.into_iter().enumerate() {
382        if i.is_multiple_of(2) {
383            storage[i / 2].re = sample;
384        } else {
385            storage[i / 2].im = sample;
386        }
387    }
388    fft.forward(&mut storage)
389        .map_err(|error| error.to_string())?;
390    let scale = (2. / spectrum.len() as f64) as f32;
391    let amplitudes = storage
392        .into_iter()
393        .map(|amplitude| amplitude.norm() * scale)
394        .collect::<Vec<_>>();
395    match threshold {
396        FftThreshold::Sigma(sigma) => {
397            let count = amplitudes.len() as f32;
398            let sum = amplitudes.iter().sum::<f32>();
399            let squared = amplitudes
400                .iter()
401                .map(|&amplitude| amplitude * amplitude)
402                .sum::<f32>();
403            let mean = sum / count;
404            let cutoff = mean + sigma * (squared / count - mean * mean).sqrt();
405            Ok(amplitudes
406                .iter()
407                .take(maximum + 1)
408                .enumerate()
409                .filter_map(|(wave, &amplitude)| (cutoff <= amplitude).then_some(wave))
410                .collect())
411        }
412        FftThreshold::Top(count) => {
413            let mut indexes = (0..=maximum)
414                .filter(|&index| amplitudes[index] > 0.)
415                .collect::<Vec<_>>();
416            indexes.sort_by(|&left, &right| amplitudes[right].total_cmp(&amplitudes[left]));
417            indexes.truncate(count);
418            indexes.extend(std::iter::repeat_n(
419                0,
420                count
421                    .saturating_sub(indexes.len())
422                    .min(spectrum.len().saturating_add(1)),
423            ));
424            Ok(indexes)
425        }
426    }
427}

429fn interpolate(spectrum: &[f32], included: &[bool]) -> Result<Vec<f32>, String> {
430    let mut values = spectrum.to_vec();
431    let mut previous = None;
432    let mut from = 0;
433    for (channel, &valid) in included.iter().enumerate() {
434        if !valid {
435            continue;
436        }
437        if from < channel {
438            let value = previous.map_or(spectrum[channel], |before| {
439                (before + spectrum[channel]) / 2.
440            });
441            values[from..channel].fill(value);
442        }
443        previous = Some(spectrum[channel]);
444        from = channel + 1;
445    }
446    let last = previous.ok_or("Bad data. all channels are masked.")?;
447    values[from..].fill(last);
448    Ok(values)
449}

451pub(super) fn terms(waves: &[usize]) -> usize {
452    2 * waves.len() - usize::from(waves.first() == Some(&0))
453}
454
455fn basis(channel: usize, channels: usize, waves: &[usize]) -> Vec<f64> {
456    let norm = TAU / channels.saturating_sub(1).max(1) as f64;
457    let mut result = Vec::with_capacity(terms(waves));
458    for &wave in waves {
459        if wave == 0 {
460            result.push(1.);
461        } else {
462            let (sine, cosine) = (channel as f64 * wave as f64 * norm).sin_cos();
463            result.extend([sine, cosine]);
464        }
465    }
466    result
467}

469pub(super) fn subtract(
470    spectrum: &mut [f32],
471    coefficients: &[f64],
472    waves: &[usize],
473) -> Result<(), String> {
474    if waves.is_empty() || coefficients.len() != terms(waves) {
475        return Err("Invalid sinusoidal baseline coefficients.".into());
476    }
477    let channels = spectrum.len();
478    for (channel, sample) in spectrum.iter_mut().enumerate() {
479        let terms = basis(channel, channels, waves);
480        let fitted = terms
481            .iter()
482            .zip(coefficients)
483            .map(|(basis, coefficient)| basis * coefficient)
484            .sum::<f64>();
485        *sample -= fitted as f32;
486    }
487    Ok(())
