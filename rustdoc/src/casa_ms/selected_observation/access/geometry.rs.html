41pub(super) fn evaluate_row_geometry(
42    source: &BoundObservationSource,
43    problem: &CompiledProblem,
44    stored: SelectedStoredRow,
45    observation_pointing: Option<SelectedPointingDirections>,
46) -> Result<EvaluatedRowGeometry, BoundObservationSourceError> {
47    let field_id = usize::try_from(stored.field_id())
48        .map_err(|_| BoundObservationSourceError::InvalidRowGeometry)?;
49    let (longitude_rad, latitude_rad) = source
50        .geometry_engine
51        .observation_direction_j2000(stored.time_mjd_seconds(), field_id)?
52        .as_angles();
53    let observation_direction =
54        SkyDirection::new(DirectionFrame::J2000, longitude_rad, latitude_rad);
55    let centres = problem.geometry().centres();
56    let phase_direction = match centres.phase_tracking() {
57        PhaseCentreLaw::Observation => observation_direction,
58        PhaseCentreLaw::Fixed(direction) => require_fixed_j2000(*direction)?,
59        PhaseCentreLaw::Ephemeris(target) => {
60            let direction = source.geometry_engine.moving_direction_j2000(
61                stored.time_mjd_seconds(),
62                field_id,
63                target,
64            )?;
65            let (longitude_rad, latitude_rad) = direction.as_angles();
66            SkyDirection::new(DirectionFrame::J2000, longitude_rad, latitude_rad)
67        }
68    };
69    let pointing_directions = match centres.pointing() {
70        PointingCentreLaw::PhaseTrackingCentre => SelectedPointingDirections {
71            antenna1: phase_direction,
72            antenna2: phase_direction,
73        },
74        PointingCentreLaw::FieldCentre => SelectedPointingDirections {
75            antenna1: observation_direction,
76            antenna2: observation_direction,
77        },
78        PointingCentreLaw::Fixed(direction) => {
79            let direction = require_fixed_j2000(*direction)?;
80            SelectedPointingDirections {
81                antenna1: direction,
82                antenna2: direction,
83            }
84        }
85        PointingCentreLaw::Observation(_) => observation_pointing
86            .ok_or(BoundObservationSourceError::MissingEvaluatedPointingDirections)?,
87    };
88    let domains = problem.geometry().domains();
89    let moving_direction_shift = if matches!(centres.phase_tracking(), PhaseCentreLaw::Ephemeris(_))
90    {
91        let anchor = domains
92            .first()
93            .ok_or(BoundObservationSourceError::InvalidRowGeometry)?
94            .model_phase_centre();
95        let anchor_j2000 = source.geometry_engine.direction_angles_j2000(
96            stored.time_mjd_seconds(),
97            [anchor.longitude_rad(), anchor.latitude_rad()],
98            direction_ref(anchor.frame()),
99        )?;
100        Some([
101            phase_direction.longitude_rad() - anchor_j2000[0],
102            phase_direction.latitude_rad() - anchor_j2000[1],
103        ])
104    } else {
105        None
106    };
107    let chart_count = domains.iter().try_fold(0_usize, |count, domain| {
108        count.checked_add(domain.facets().len())
109    });
110    let mut domain_projections = Vec::new();
111    domain_projections
112        .try_reserve_exact(chart_count.ok_or(BoundObservationSourceError::InvalidRowGeometry)?)
113        .map_err(|_| BoundObservationSourceError::InvalidRowGeometry)?;
114    for (domain_ordinal, domain) in domains.iter().enumerate() {
115        let domain_ordinal = u32::try_from(domain_ordinal)
116            .map_err(|_| BoundObservationSourceError::InvalidRowGeometry)?;
117        let shared_psf = domain.psf_phase_centre() == domain.model_phase_centre();
118        let distinct_psf = if shared_psf {
119            None
120        } else {
121            Some(evaluate_phase_centre_projection(
122                source,
123                stored,
124                observation_direction,
125                PhaseCentreProjectionRequest {
126                    field_id,
127                    target_direction: domain.psf_phase_centre(),
128                    project_to_observation_plane: domain.facets().len() > 1,
129                    moving_direction_shift,
130                    uvw_law: problem.geometry().uvw(),
131                },
132            )?)
133        };
134        for (facet_ordinal, facet) in domain.facets().iter().enumerate() {
135            let facet_ordinal = u32::try_from(facet_ordinal)
136                .map_err(|_| BoundObservationSourceError::InvalidRowGeometry)?;
137            let model_phase_centre = if domain.facets().len() == 1 {
138                domain.model_phase_centre()
139            } else {
140                facet.phase_centre()
141            };
142            let model = evaluate_phase_centre_projection(
143                source,
144                stored,
145                observation_direction,
146                PhaseCentreProjectionRequest {
147                    field_id,
148                    target_direction: model_phase_centre,
149                    project_to_observation_plane: domain.facets().len() > 1,
150                    moving_direction_shift,
151                    uvw_law: problem.geometry().uvw(),
152                },
153            )?;
154            let projection = match distinct_psf {
155                Some(psf) => SelectedImageDomainProjection::new_facet(
156                    domain_ordinal,
157                    facet_ordinal,
158                    model,
159                    psf,
160                ),
161                None => SelectedImageDomainProjection::facet_with_shared_psf(
162                    domain_ordinal,
163                    facet_ordinal,
164                    model,
165                ),
166            };
167            let projection = if let Some(plan) = &source.aw_pointing_plan {
168                let epoch = AwPointingEpochKey {
169                    data_description_id: stored.data_description_id(),
170                    field_id: stored.field_id(),
171                    time_bits: stored.time_mjd_seconds().to_bits(),
172                };
173                let chart = domain_projections.len();
174                let pixels = plan
175                    .chart_pixels_by_epoch
176                    .get(&epoch)
177                    .and_then(|charts| charts.get(chart))
178                    .ok_or(BoundObservationSourceError::InvalidRowGeometry)?;
179                let antenna1 = pixels
180                    .get(&stored.antenna1())
181                    .ok_or(BoundObservationSourceError::InvalidRowGeometry)?;
182                let antenna2 = pixels
183                    .get(&stored.antenna2())
184                    .ok_or(BoundObservationSourceError::InvalidRowGeometry)?;
185                projection
186                    .with_aw_pointing_pixel([
187                        0.5 * (antenna1[0] + antenna2[0]),
188                        0.5 * (antenna1[1] + antenna2[1]),
189                    ])
190                    .ok_or(BoundObservationSourceError::InvalidRowGeometry)?
191            } else {
192                projection
193            };
194            domain_projections.push(projection);
195        }
196    }
197    let domain_projections = SelectedImageDomainProjections::new(domain_projections)
198        .ok_or(BoundObservationSourceError::InvalidRowGeometry)?;
199    Ok(EvaluatedRowGeometry {
200        domain_projections,
201        pointing_directions,
202    })
203}

205struct PhaseCentreProjectionRequest {
206    field_id: usize,
207    target_direction: SkyDirection,
208    project_to_observation_plane: bool,
209    moving_direction_shift: Option<[f64; 2]>,
210    uvw_law: casa_imaging_model::UvwCoordinateLaw,
211}
212
213fn evaluate_phase_centre_projection(
214    source: &BoundObservationSource,
215    stored: SelectedStoredRow,
216    observation_direction: SkyDirection,
217    request: PhaseCentreProjectionRequest,
218) -> Result<SelectedPhaseCentreProjection, BoundObservationSourceError> {
219    let mut target_angles = source.geometry_engine.direction_angles_j2000(
220        stored.time_mjd_seconds(),
221        [
222            request.target_direction.longitude_rad(),
223            request.target_direction.latitude_rad(),
224        ],
225        direction_ref(request.target_direction.frame()),
226    )?;
227    if let Some([longitude, latitude]) = request.moving_direction_shift {
228        target_angles[0] += longitude;
229        target_angles[1] += latitude;
230    }
231    let target_j2000 = SkyDirection::new(DirectionFrame::J2000, target_angles[0], target_angles[1]);
232    let (transformed_uvw_m, phase_shift_m) = if matches!(
233        request.uvw_law,
234        casa_imaging_model::UvwCoordinateLaw::MosaicPhaseTrackingCentre
235    ) {
236        if request.project_to_observation_plane {
237            return Err(BoundObservationSourceError::InvalidRowGeometry);
238        }
239        let target = MDirection::from_angles(
240            target_j2000.longitude_rad(),
241            target_j2000.latitude_rad(),
242            DirectionRef::J2000,
243        );
244        let (uvw_m, casa_dphase_m) = source
245            .geometry_engine
246            .reproject_raw_uvw_for_mosaic_to_direction(stored.uvw_m(), request.field_id, &target)?;
247        (uvw_m, -casa_dphase_m)
248    } else if target_j2000 == observation_direction && !request.project_to_observation_plane {
249        (stored.uvw_m(), 0.0)
250    } else if request.project_to_observation_plane {
251        source
252            .geometry_engine
253            .reproject_raw_uvw_for_faceted_gridft_between_j2000_directions(
254                stored.uvw_m(),
255                [
256                    observation_direction.longitude_rad(),
257                    observation_direction.latitude_rad(),
258                ],
259                target_angles,
260            )?
261    } else {
262        source
263            .geometry_engine
264            .reproject_raw_uvw_for_gridft_between_j2000_directions(
265                stored.uvw_m(),
266                [
267                    observation_direction.longitude_rad(),
268                    observation_direction.latitude_rad(),
269                ],
270                target_angles,
271            )?
272    };
273    SelectedPhaseCentreProjection::new(transformed_uvw_m, phase_shift_m)
274        .ok_or(BoundObservationSourceError::InvalidRowGeometry)
275}

277const fn direction_ref(frame: DirectionFrame) -> DirectionRef {
278    match frame {
279        DirectionFrame::Icrs => DirectionRef::ICRS,
280        DirectionFrame::J2000 => DirectionRef::J2000,
281        DirectionFrame::B1950 => DirectionRef::B1950,
282        DirectionFrame::Galactic => DirectionRef::GALACTIC,
283    }

286pub(super) fn evaluate_observation_pointings(
287    source: &BoundObservationSource,
288    problem: &CompiledProblem,
289    buffer: &SelectedObservationBuffer,
290) -> Result<Option<Vec<SelectedPointingDirections>>, BoundObservationSourceError> {
291    let PointingCentreLaw::Observation(law) = problem.geometry().centres().pointing() else {
292        return Ok(None);
293    };
294    let mut queries = Vec::with_capacity(buffer.row_count().saturating_mul(2));
295    let mut phase_directions = Vec::with_capacity(buffer.row_count());
296    for row in 0..buffer.row_count() {
297        let stored = buffer
298            .row(row)
299            .ok_or(BoundObservationSourceError::StoredSampleShapeMismatch)?;
300        let time_mjd_seconds = match law.time_sampling() {
301            PointingTimeSampling::VisibilityTime => stored.time_mjd_seconds(),
302            PointingTimeSampling::VisibilityTimeCentroid => stored.time_centroid_mjd_seconds(),
303        };
304        queries.push(PointingDirectionQuery::new(
305            stored.antenna1(),
306            time_mjd_seconds,
307        )?);
308        queries.push(PointingDirectionQuery::new(
309            stored.antenna2(),
310            time_mjd_seconds,
311        )?);
312        phase_directions.push(evaluate_phase_direction(source, problem, stored)?);
313    }
314    let brackets = source
315        .pointing_catalog
316        .as_ref()
317        .ok_or(BoundObservationSourceError::MissingPointingQueryDomain)?
318        .direction_brackets(&source.geometry_engine, &queries)?;
319    let mut pointings = Vec::with_capacity(buffer.row_count());
320    for ((row, antenna_brackets), fallback) in brackets
321        .as_chunks::<2>()
322        .0
323        .iter()
324        .enumerate()
325        .zip(phase_directions)
326    {
327        pointings.push(SelectedPointingDirections {
328            antenna1: resolve_pointing_direction(
329                antenna_brackets[0],
330                queries[2 * row],
331                *law,
332                fallback,
333            )?,
334            antenna2: resolve_pointing_direction(
335                antenna_brackets[1],
336                queries[2 * row + 1],
337                *law,
338                fallback,
339            )?,
340        });
341    }
342    Ok(Some(pointings))
343}

345pub(super) fn build_aw_pointing_epoch_plan(
346    problem: &CompiledProblem,
347    source: &BoundObservationSource,
348) -> Result<AwPointingEpochPlan, BoundObservationSourceError> {
349    let contract = problem
350        .science()
351        .measurement_equation()
352        .aw_projection()
353        .filter(|contract| contract.use_pointing())
354        .ok_or(BoundObservationSourceError::UnsupportedCentreLaw)?;
355    let law = match problem.geometry().centres().pointing() {
356        PointingCentreLaw::Observation(law) => *law,
357        _ => return Err(BoundObservationSourceError::UnsupportedCentreLaw),
358    };
359    let catalog = source
360        .pointing_catalog
361        .as_ref()
362        .ok_or(BoundObservationSourceError::MissingPointingQueryDomain)?;
363    let available_bytes = source
364        .content_plan
365        .rows_per_block()
366        .checked_mul(crate::SelectedObservationRow::STORAGE_BYTES_PER_ROW)
367        .ok_or(BoundObservationSourceError::MeasurementOverflow)?;
368    let scan_plan = MsReadPlan::new(
369        source.measurement_set.row_count(),
370        MsSelectionIoBudget {
371            available_bytes,
372            maximum_live_blocks: 1,
373            requested_bytes_per_row: crate::SelectedObservationRow::STORAGE_BYTES_PER_ROW,
374            storage_alignment_rows: Some(source.content_plan.rows_per_block()),
375        },
376    )
377    .map_err(|error| MsError::InvalidInput(error.to_string()))?;
378    let mut cursor = source
379        .measurement_set
380        .main_row_selection_cursor(scan_plan)?;
381    let mut epochs = Vec::<AwPointingEpochInput>::new();
382    while let Some(row) = cursor.next(&source.measurement_set)? {
383        if !source.row_predicate.matches(row) {
384            continue;
385        }
386        let key = AwPointingEpochKey {
387            data_description_id: row.data_description_id(),
388            field_id: row.field_id(),
389            time_bits: row.time_mjd_seconds().to_bits(),
390        };
391        if epochs.last().is_none_or(|epoch| epoch.key != key) {
392            epochs.push(AwPointingEpochInput {
393                key,
394                row_count: 0,
395                antennas: BTreeSet::new(),
396            });
397        }
398        let epoch = epochs.last_mut().expect("AW pointing epoch was appended");
399        epoch.row_count = epoch
400            .row_count
401            .checked_add(1)
402            .ok_or(BoundObservationSourceError::MeasurementOverflow)?;
403        epoch.antennas.insert(row.antenna1());
404        epoch.antennas.insert(row.antenna2());
405    }
406
407    let charts = problem
408        .geometry()
409        .domains()
410        .iter()
411        .flat_map(|domain| domain.facets().iter().map(|facet| facet.direction()))
412        .map(aw_direction_coordinate)
413        .collect::<Result<Vec<_>, _>>()?;
414    let [grouping_arcsec, refresh_arcsec] = contract.pointing_offset_sigdev_arcsec();
415    let grouping_arcsec = if grouping_arcsec > 0.0 {
416        grouping_arcsec
417    } else {
418        1.0e-3
419    };
420    let mut chart_pixels_by_epoch = BTreeMap::new();
421    let mut cached_field = None;
422    let mut cached_row_count = 0_usize;
423    let mut cached_antenna_pixels = BTreeMap::<i32, [f64; 2]>::new();
424    let mut cached_grouped = AwPointingChartPixels::from([]);
425    let primary_increment = problem
426        .geometry()
427        .domains()
428        .first()
429        .ok_or(BoundObservationSourceError::InvalidRowGeometry)?
430        .direction()
431        .increment_rad();
432    let primary_increment_norm = primary_increment[0].hypot(primary_increment[1]);
433    let refresh_threshold_rad = refresh_arcsec.to_radians() / 3600.0;
434
435    for epoch in epochs {
436        let time = f64::from_bits(epoch.key.time_bits);
437        let fallback = pointing_epoch_phase_direction(source, problem, epoch.key.field_id, time)?;
438        let queries = epoch
439            .antennas
440            .iter()
441            .copied()
442            .map(|antenna| PointingDirectionQuery::new(antenna, time))
443            .collect::<Result<Vec<_>, _>>()?;
444        let brackets = catalog.direction_brackets(&source.geometry_engine, &queries)?;
445        let directions = epoch
446            .antennas
447            .iter()
448            .copied()
449            .zip(brackets)
450            .zip(queries.iter().copied())
451            .map(|((antenna, bracket), query)| {
452                resolve_pointing_direction(bracket, query, law, fallback)
453                    .map(|direction| (antenna, direction))
454            })
455            .collect::<Result<BTreeMap<_, _>, _>>()?;
456        let pixels_by_chart = charts
457            .iter()
458            .map(|coordinate| {
459                directions
460                    .iter()
461                    .map(|(&antenna, direction)| {
462                        coordinate
463                            .to_pixel(&[direction.longitude_rad(), direction.latitude_rad()])
464                            .map_err(|_| BoundObservationSourceError::InvalidRowGeometry)
465                            .and_then(|pixel| {
466                                pixel
467                                    .get(0..2)
468                                    .filter(|pixel| pixel.iter().all(|value| value.is_finite()))
469                                    .map(|pixel| (antenna, [pixel[0], pixel[1]]))
470                                    .ok_or(BoundObservationSourceError::InvalidRowGeometry)
471                            })
472                    })
473                    .collect::<Result<BTreeMap<_, _>, _>>()
474            })
475            .collect::<Result<Vec<_>, _>>()?;
476        let current_primary = pixels_by_chart
477            .first()
478            .ok_or(BoundObservationSourceError::InvalidRowGeometry)?;
479        let same_antennas = cached_antenna_pixels.len() == current_primary.len()
480            && cached_antenna_pixels.keys().eq(current_primary.keys());
481        let refresh = if cached_field != Some(epoch.key.field_id)
482            || cached_row_count != epoch.row_count
483            || !same_antennas
484        {
485            true
486        } else {
487            let residual = cached_antenna_pixels.iter().zip(current_primary).fold(
488                [0.0, 0.0],
489                |[sum_x, sum_y], ((_, cached), (_, current))| {
490                    [
491                        sum_x + cached[0] - current[0],
492                        sum_y + cached[1] - current[1],
493                    ]
494                },
495            );
496            residual[0].hypot(residual[1]) / current_primary.len().max(1) as f64
497                * primary_increment_norm
498                >= refresh_threshold_rad
499        };
500        if refresh {
501            let grouped =
502                pixels_by_chart
503                    .iter()
504                    .zip(
505                        problem.geometry().domains().iter().flat_map(|domain| {
506                            domain.facets().iter().map(|facet| facet.direction())
507                        }),
508                    )
509                    .map(|(pixels, direction)| {
510                        casa_aw_grouped_pixels(pixels, grouping_arcsec, direction.increment_rad())
511                    })
512                    .collect::<Result<Vec<_>, _>>()?;
513            cached_grouped = Arc::from(grouped);
514        }
515        cached_field = Some(epoch.key.field_id);
516        cached_row_count = epoch.row_count;
517        cached_antenna_pixels = current_primary.clone();
518        if chart_pixels_by_epoch
519            .insert(epoch.key, Arc::clone(&cached_grouped))
520            .is_some()
521        {
522            return Err(BoundObservationSourceError::InvalidRowGeometry);
523        }
524    }
525    let retained_byte_ceiling = size_of::<AwPointingEpochPlan>()
526        .checked_add(
527            chart_pixels_by_epoch
528                .values()
529                .try_fold(0_usize, |bytes, charts| {
530                    charts.iter().try_fold(bytes, |bytes, chart| {
531                        chart
532                            .len()
533                            .checked_mul(usize::BITS as usize)
534                            .and_then(|entries| bytes.checked_add(entries))
535                    })
536                })
537                .ok_or(BoundObservationSourceError::MeasurementOverflow)?,
538        )
539        .ok_or(BoundObservationSourceError::MeasurementOverflow)?;
540    Ok(AwPointingEpochPlan {
541        chart_pixels_by_epoch,
542        retained_byte_ceiling,
543    })
544}

546fn aw_direction_coordinate(
547    spec: casa_imaging_model::DirectionCoordinateSpec,
548) -> Result<DirectionCoordinate, BoundObservationSourceError> {
549    if spec.projection() != ModelProjection::Sin {
550        return Err(BoundObservationSourceError::InvalidRowGeometry);
551    }
552    let reference = spec.reference_direction();
553    let pc = spec.pc();
554    Ok(DirectionCoordinate::new(
555        direction_ref(reference.frame()),
556        CoordinateProjection::new(ProjectionType::SIN),
557        [reference.longitude_rad(), reference.latitude_rad()],
558        spec.increment_rad(),
559        spec.reference_pixel(),
560    )
561    .with_pc_matrix(arr2(&pc))
562    .with_longpole(spec.pole_deg()[0].to_radians())
563    .with_latpole(spec.pole_deg()[1].to_radians()))
564}

566fn casa_aw_grouped_pixels(
567    pixels: &BTreeMap<i32, [f64; 2]>,
568    grouping_arcsec: f64,
569    increment_rad: [f64; 2],
570) -> Result<BTreeMap<i32, [f64; 2]>, BoundObservationSourceError> {
571    let bin_width =
572        grouping_arcsec.to_radians() / 3600.0 / increment_rad[0].hypot(increment_rad[1]);
573    if !(bin_width.is_finite() && bin_width > 0.0) || pixels.is_empty() {
574        return Err(BoundObservationSourceError::InvalidRowGeometry);
575    }
576    let min_x = pixels
577        .values()
578        .map(|pixel| pixel[0])
579        .fold(f64::INFINITY, f64::min);
580    let min_y = pixels
581        .values()
582        .map(|pixel| pixel[1])
583        .fold(f64::INFINITY, f64::min);
584    let mut bins = BTreeMap::<i32, (i64, i64)>::new();
585    let mut sums = BTreeMap::<(i64, i64), (f32, f32, usize)>::new();
586    for (&antenna, &pixel) in pixels {
587        let bin = (
588            ((pixel[0] - min_x) / bin_width).floor() as i64,
589            ((pixel[1] - min_y) / bin_width).floor() as i64,
590        );
591        bins.insert(antenna, bin);
592        let sum = sums.entry(bin).or_insert((0.0, 0.0, 0));
593        sum.0 = (f64::from(sum.0) + pixel[0]) as f32;
594        sum.1 = (f64::from(sum.1) + pixel[1]) as f32;
595        sum.2 = sum
596            .2
597            .checked_add(1)
598            .ok_or(BoundObservationSourceError::MeasurementOverflow)?;
599    }
600    let means = sums
601        .into_iter()
602        .map(|(bin, (x, y, count))| (bin, [(x / count as f32) as f64, (y / count as f32) as f64]))
603        .collect::<BTreeMap<_, _>>();
604    Ok(bins
605        .into_iter()
606        .map(|(antenna, bin)| (antenna, means[&bin]))
607        .collect())
608}

610fn pointing_epoch_phase_direction(
611    source: &BoundObservationSource,
612    problem: &CompiledProblem,
613    field_id: i32,
614    time_mjd_seconds: f64,
615) -> Result<SkyDirection, BoundObservationSourceError> {
616    let field =
617        usize::try_from(field_id).map_err(|_| BoundObservationSourceError::InvalidRowGeometry)?;
618    let observation = source
619        .geometry_engine
620        .observation_direction_j2000(time_mjd_seconds, field)?;
621    let (longitude, latitude) = observation.as_angles();
622    let observation = SkyDirection::new(DirectionFrame::J2000, longitude, latitude);
623    match problem.geometry().centres().phase_tracking() {
624        PhaseCentreLaw::Observation => Ok(observation),
625        PhaseCentreLaw::Fixed(direction) => require_fixed_j2000(*direction),
626        PhaseCentreLaw::Ephemeris(target) => {
627            let direction =
628                source
629                    .geometry_engine
630                    .moving_direction_j2000(time_mjd_seconds, field, target)?;
631            let (longitude, latitude) = direction.as_angles();
632            Ok(SkyDirection::new(
633                DirectionFrame::J2000,
634                longitude,
635                latitude,
636            ))
637        }
638    }
639}

642    source: &BoundObservationSource,
643    problem: &CompiledProblem,
644    stored: SelectedStoredRow,
645) -> Result<SkyDirection, BoundObservationSourceError> {
646    let field_id = usize::try_from(stored.field_id())
647        .map_err(|_| BoundObservationSourceError::InvalidRowGeometry)?;
648    let (longitude_rad, latitude_rad) = source
649        .geometry_engine
650        .observation_direction_j2000(stored.time_mjd_seconds(), field_id)?
651        .as_angles();
652    let observation = SkyDirection::new(DirectionFrame::J2000, longitude_rad, latitude_rad);
653    match problem.geometry().centres().phase_tracking() {
654        PhaseCentreLaw::Observation => Ok(observation),
655        PhaseCentreLaw::Fixed(direction) => require_fixed_j2000(*direction),
656        PhaseCentreLaw::Ephemeris(target) => {
657            let direction = source.geometry_engine.moving_direction_j2000(
658                stored.time_mjd_seconds(),
659                field_id,
660                target,
661            )?;
662            let (longitude_rad, latitude_rad) = direction.as_angles();
663            Ok(SkyDirection::new(
664                DirectionFrame::J2000,
665                longitude_rad,
666                latitude_rad,
667            ))
668        }
669    }

672fn resolve_pointing_direction(
673    bracket: PointingDirectionBracket,
674    query: PointingDirectionQuery,
675    law: casa_imaging_model::ObservationPointingLaw,
676    fallback: SkyDirection,
677) -> Result<SkyDirection, BoundObservationSourceError> {
678    if let Some(covering) = bracket.covering() {
679        return Ok(j2000_direction(covering.direction_j2000_rad()));
680    }
681    let direction = match (bracket.before(), bracket.after()) {
682        (Some(before), Some(after)) if before.row_index() == after.row_index() => {
683            before.direction_j2000_rad()
684        }
685        (Some(before), Some(after)) => match law.interpolation() {
686            PointingInterpolation::Nearest => {
687                let before_distance = query.time_mjd_seconds() - before.row_time_mjd_seconds();
688                let after_distance = after.row_time_mjd_seconds() - query.time_mjd_seconds();
689                if before_distance <= after_distance {
690                    before.direction_j2000_rad()
691                } else {
692                    after.direction_j2000_rad()
693                }
694            }
695            PointingInterpolation::GreatCircleShortestArc => interpolate_direction(
696                before.direction_j2000_rad(),
697                after.direction_j2000_rad(),
698                (query.time_mjd_seconds() - before.row_time_mjd_seconds())
699                    / (after.row_time_mjd_seconds() - before.row_time_mjd_seconds()),
700            )?,
701        },
702        (Some(endpoint), None) | (None, Some(endpoint))
703            if matches!(law.extrapolation(), PointingExtrapolation::HoldNearest) =>
704        {
705            endpoint.direction_j2000_rad()
706        }
707        (Some(_), None) | (None, Some(_)) => {
708            return Err(BoundObservationSourceError::PointingOutsideCoverage {
709                antenna_id: query.antenna_id(),
710                time_mjd_seconds: query.time_mjd_seconds(),
711            });
712        }
713        (None, None) if matches!(law.missing(), MissingPointingPolicy::UsePhaseTrackingCentre) => {
714            return Ok(fallback);
715        }
716        (None, None) => {
717            return Err(BoundObservationSourceError::MissingPointingDirection {
718                antenna_id: query.antenna_id(),
719                time_mjd_seconds: query.time_mjd_seconds(),
720            });
721        }
722    };
723    Ok(j2000_direction(direction))
724}
725
726fn j2000_direction(direction_rad: [f64; 2]) -> SkyDirection {
727    SkyDirection::new(DirectionFrame::J2000, direction_rad[0], direction_rad[1])
728}
729
730fn interpolate_direction(
731    before: [f64; 2],
732    after: [f64; 2],
733    fraction: f64,
734) -> Result<[f64; 2], BoundObservationSourceError> {
735    if !(fraction.is_finite() && (0.0..=1.0).contains(&fraction)) {
736        return Err(BoundObservationSourceError::InvalidPointingInterpolation);
737    }
738    let left = unit_direction(before);
739    let right = unit_direction(after);
740    let cosine = left
741        .iter()
742        .zip(right)
743        .map(|(left, right)| left * right)
744        .sum::<f64>()
745        .clamp(-1.0, 1.0);
746    let angle = cosine.acos();
747    let vector = if angle.abs() < 1.0e-15 {
748        left
749    } else {
750        let denominator = angle.sin();
751        let before_weight = ((1.0 - fraction) * angle).sin() / denominator;
752        let after_weight = (fraction * angle).sin() / denominator;
753        [
754            before_weight * left[0] + after_weight * right[0],
755            before_weight * left[1] + after_weight * right[1],
756            before_weight * left[2] + after_weight * right[2],
757        ]
758    };
759    let norm = vector.iter().map(|value| value * value).sum::<f64>().sqrt();
760    if !(norm.is_finite() && norm > 0.0) {
761        return Err(BoundObservationSourceError::InvalidPointingInterpolation);
762    }
763    let [x, y, z] = vector.map(|value| value / norm);
764    Ok([y.atan2(x).rem_euclid(std::f64::consts::TAU), z.asin()])
765}
766
767fn unit_direction(direction: [f64; 2]) -> [f64; 3] {
768    [
769        direction[1].cos() * direction[0].cos(),
770        direction[1].cos() * direction[0].sin(),
771        direction[1].sin(),
772    ]
773}
774
775fn require_fixed_j2000(
776    direction: SkyDirection,
777) -> Result<SkyDirection, BoundObservationSourceError> {
778    if direction.frame() != DirectionFrame::J2000 {
779        return Err(BoundObservationSourceError::UnsupportedFixedCentreFrame {
780            frame: direction.frame(),
781        });
782    }
783    Ok(direction)
784}
