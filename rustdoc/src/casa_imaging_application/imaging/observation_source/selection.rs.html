2//! What the batch route reads: the compiled selection resolved to the
3//! observation's partitions, and the per-sample predicate the request
4//! cannot state.
5
6use casa_imaging_model::UvSelection;
7use casa_observation::{Correlation, FieldSelection, PartitionId};
8
9/// One partition the selection reads.
10#[derive(Clone, Debug, PartialEq)]
11pub(crate) struct SelectedPartition {
12    /// The partition.
13    pub(crate) id: PartitionId,
14    /// The spectral window number tasks select by (MSv2 `SPECTRAL_WINDOW`
15    /// row): what kernel sets and continuum rules are keyed on.
16    pub(crate) spectral_window: u32,
17    /// The selected native channels, ascending.
18    pub(crate) channels: Vec<u32>,
19    /// The correlation products the operator routes, in its order.
20    pub(crate) correlations: Vec<Correlation>,
21    /// The window's reference wavelength, when a UV-distance interval is
22    /// in wavelengths.
23    pub(crate) wavelength_m: Option<f64>,
24}
25
26/// The compiled selection resolved to the observation's partitions: what
27/// [`super::ObservationSource`] binds and tests.
28///
29/// Windows, channels, correlations, the data group and fields map onto
30/// each partition's [`AccessRequest`](casa_observation::AccessRequest);
31/// intents onto the partitions chosen, since a partition's samples share
32/// theirs; the UV distance is tested per sample as the source packs it.
33#[derive(Clone, Debug, PartialEq)]
34pub(crate) struct PartitionSelection {
35    /// The data group whose visibilities are imaged.
36    pub(crate) data_group: String,
37    /// The partitions, in traversal order.
38    pub(crate) partitions: Vec<SelectedPartition>,
39    /// The fields.
40    pub(crate) fields: FieldSelection,
41    /// The UV-distance intervals.
42    pub(crate) uv: UvSelection,
43}
44
45impl PartitionSelection {
46    /// Whether a sample of partition `partition` with stored UVW `uvw_m` is
47    /// selected by UV distance.
48    pub(crate) fn uv_holds(&self, partition: &SelectedPartition, uvw_m: [f64; 3]) -> bool {
49        self.uv.holds(uvw_m, partition.wavelength_m)
50    }
51}
