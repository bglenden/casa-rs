42    Array(ArrayD<f64>),
43    /// Path to a kernel image.
44    ImagePath(PathBuf),

47/// Linear n-dimensional convolution (`ia.convolve`).
48#[allow(clippy::too_many_arguments)]
49pub fn convolve_image(
50    attached: &AttachedImage,
51    outfile: &str,
52    kernel_in: &ConvolveKernelInput,
53    scale: f64,
54    region: Option<SubimageRegion<'_>>,
55    mask: &str,
56    overwrite: bool,
57    stretch: bool,
58    history_command: Option<String>,
59    is_external_open: ExternalPathCheck<'_>,
60) -> ImageToolResult<AttachedImage> {
61    if !matches!(
62        &*attached.lock(),
63        ImageHandle::Float(_) | ImageHandle::Double(_)
64    ) {
65        return Err(ImageToolError::new(
66            "This method only supports real-valued images",
67        ));
68    }
69    let kernel_path = match kernel_in {
70        ConvolveKernelInput::ImagePath(path) => {
71            if !path.is_dir() {
72                return Err(ImageToolError::new(format!(
73                    "Table {} does not exist",
74                    path.display()
75                )));
76            }
77            Some(path.as_path())
78        }
79        ConvolveKernelInput::Array(_) => None,
80    };
81    let same = kernel_path
82        .and_then(|path| path.canonicalize().ok())
83        .is_some_and(|path| attached.native_path() == Some(&path));
84    let kernel = kernel_path
85        .filter(|_| !same)
86        .map(AttachedImage::open_for_ordered_read)
87        .transpose()?;
88    let mut coordinates = None;
89    let mut out = publish_image(outfile, overwrite, is_external_open, |dest| {
90        let run_build = |image: &ImageHandle, k: Option<&ImageHandle>| {
91            convolve_build(
92                attached,
93                image,
94                k,
95                outfile,
96                kernel_in,
97                scale,
98                region,
99                mask,
100                stretch,
101                dest,
102                history_command.clone(),
103            )
104        };
105        let product = if let Some(k) = &kernel {
106            let source_first = attached
107                .native_path()
108                .zip(k.native_path())
109                .is_none_or(|(left, right)| left < right);
110            if source_first {
111                read_locked(attached, |image| {
112                    read_locked(k, |k_img| run_build(image, Some(k_img)))
113                })?
114            } else {
115                read_locked(k, |k_img| {
116                    read_locked(attached, |image| run_build(image, Some(k_img)))
117                })?
118            }
119        } else {
120            read_locked(attached, |image| run_build(image, same.then_some(image)))?
121        };
122        coordinates = AttachedImage::capture_live_coordinates(&product)?;
123        Ok(product)
124    })?;
125    if let Some((record, system)) = coordinates {
126        out.keep_live_coordinates(record, system);
127    }
128    Ok(out)
129}

131#[allow(clippy::too_many_arguments)]
132fn convolve_build(
133    attachment: &AttachedImage,
134    image: &ImageHandle,
135    kernel: Option<&ImageHandle>,
136    outfile: &str,
137    kernel_in: &ConvolveKernelInput,
138    scale: f64,
139    region: Option<SubimageRegion<'_>>,
140    mask: &str,
141    stretch: bool,
142    destination: Option<&Path>,
143    history_command: Option<String>,
144) -> ImageToolResult<ImageHandle> {
145    if !outfile.is_empty() {
146        check_handle_aliasing(image, attachment, Path::new(outfile), kernel)?;
147    }
148    let record = image.coordinate_record();
149    let (mut product, changed, dimensions) = match image {
150        ImageHandle::Float(storage) => {
151            let native = match kernel {
152                Some(ImageHandle::Float(Storage::Paged(native))) => {
153                    Some(native.as_ref() as &dyn ImageInterface<f32>)
154                }
155                Some(_) => return Err(ImageToolError::new(KERNEL_PRECISION)),
156                None => None,
157            };
158            let (out, changed, dims) = storage.with_float_view(|src| {
159                convolve_typed(
160                    src,
161                    record,
162                    native,
163                    kernel_in,
164                    scale,
165                    region,
166                    mask,
167                    stretch,
168                    outfile,
169                    destination,
170                )
171            })?;
172            (ImageHandle::Float(out), changed, dims)
173        }
174        ImageHandle::Double(storage) => {
175            let native = match kernel {
176                Some(ImageHandle::Double(Storage::Paged(native))) => {
177                    Some(native.as_ref() as &dyn ImageInterface<f64>)
178                }
179                Some(_) => return Err(ImageToolError::new(KERNEL_PRECISION)),
180                None => None,
181            };
182            let (out, changed, dims) = with_storage!(storage, src => {
183                convolve_typed(src.as_ref(), record, native, kernel_in, scale, region, mask, stretch, outfile, destination)
184            })?;
185            (ImageHandle::Double(out), changed, dims)
186        }
187        _ => {
188            return Err(ImageToolError::new(
189                "This method only supports real-valued images",
190            ));
191        }
192    };
193    if let Some(record) = changed {
194        product.set_coordinate_record(record)?;
195    }
196    let extra_lines = [
197        "NOTE: kernel with fewer dimensions than the input image applied. Degenerate axes added to the kernel".to_owned(),
198    ];
199    let extra = (dimensions < image.shape().len())
200        .then_some(("ImageConvolverTask::convolve", extra_lines.as_slice()));
201    let entries = image.history_entries()?;
202    finalize_output_history(&mut product, &entries, "convolve", history_command, extra)?;
203    Ok(product)
204}

206#[allow(clippy::too_many_arguments)]
207fn convolve_typed<T: ConvolutionPixel>(
208    source: &dyn ImageInterface<T>,
209    coordinates: RecordValue,
210    native: Option<&dyn ImageInterface<T>>,
211    kernel_in: &ConvolveKernelInput,
212    scale: f64,
213    region: Option<SubimageRegion<'_>>,
214    mask: &str,
215    stretch: bool,
216    outfile: &str,
217    path: Option<&Path>,
218) -> ImageToolResult<(Storage<T>, Option<RecordValue>, usize)> {
219    let data;
220    let (kernel, rank) = if let Some(native) = native {
221        (ConvolutionKernel::Image(native), native.ndim())
222    } else {
223        let ConvolveKernelInput::Array(pixels) = kernel_in else {
224            unreachable!("validated convolution kernel");
225        };
226        data = pixels.mapv(T::from_f64);
227        (ConvolutionKernel::Array(&data), data.ndim())
228    };
229    let plan = ImageConvolution::new(source, coordinates, region)?;
230    let mut output = Storage::create(
231        plan.section.shape.clone(),
232        plan.section.coordinates.clone(),
233        path,
234        None,
235    )?;
236    let options = ConvolutionOptions {
237        scale,
238        mask: Some(mask),
239        stretch,
240        destination: (!outfile.is_empty()).then(|| Path::new(outfile)),
241    };
242    with_storage!(&mut output, dest => plan.copy_to(source, kernel, options, dest.as_mut()))?;
243    Ok((output, plan.section.coordinate_record, rank))
244}

246/// 2D Gaussian/elliptical beam convolution (`ia.convolve2d`).
247#[allow(clippy::too_many_arguments)]
248pub fn convolve_2d_gaussian(
249    attached: &AttachedImage,
250    outfile: &str,
251    region: Option<SubimageRegion<'_>>,
252    mask: &str,
253    geometry: GaussianConvolutionOptions,
254    scale: f64,
255    overwrite: bool,
256    stretch: bool,
257    history_command: Option<String>,
258    is_external_open: ExternalPathCheck<'_>,
259) -> ImageToolResult<AttachedImage> {
260    let mut live_coordinates = None;
261    let mut product = publish_image(outfile, overwrite, is_external_open, |path| {
262        read_locked(attached, |source| {
263            if !matches!(source, ImageHandle::Float(_) | ImageHandle::Double(_)) {
264                return Err(ImageToolError::new(
265                    "convolve2d only supports real-valued images",
266                ));
267            }
268            if !outfile.is_empty() {
269                check_handle_aliasing(source, attached, Path::new(outfile), None)?;
270            }
271            let coordinates = source.coordinate_record();
272            let beams = attached.beam_view().clone();
273            let (mut output, updated) = match source {
274                ImageHandle::Float(storage) => {
275                    let (product, updated) = storage.with_float_view(|img| {
276                        convolve2d_typed(
277                            img,
278                            coordinates,
279                            beams.record(),
280                            region,
281                            mask,
282                            geometry,
283                            scale,
284                            stretch,
285                            outfile,
286                            path,
287                        )
288                    })?;
289                    (ImageHandle::Float(product), updated)
290                }
291                ImageHandle::Double(storage) => {
292                    let (product, updated) = with_storage!(storage, img => {
293                        convolve2d_typed(
294                            img.as_ref(),
295                            coordinates,
296                            beams.record(),
297                            region,
298                            mask,
299                            geometry,
300                            scale,
301                            stretch,
302                            outfile,
303                            path,
304                        )
305                    })?;
306                    (ImageHandle::Double(product), updated)
307                }
308                _ => unreachable!("checked real image precision"),
309            };
310            if let Some(record) = updated {
311                output.set_coordinate_record(record)?;
312            }
313            let source_name = source
314                .path()
315                .map_or_else(|| "Temporary_Image".to_owned(), |p| p.display().to_string());
316            let lines = [
317                beam_history(&format!("Original {source_name}"), Some(beams.record()))?,
318                beam_history(&format!("Output {outfile}"), image_info_record(&output))?,
319            ];
320            let entries = source.history_entries()?;
321            finalize_output_history(
322                &mut output,
323                &entries,
324                "convolve2d",
325                history_command,
326                Some(("Image2DConvolver::_dealWithRestoringBeam", &lines)),
327            )?;
328            live_coordinates = AttachedImage::capture_live_coordinates(&output)?;
329            Ok(output)
330        })
331    })?;
332    if let Some((record, system)) = live_coordinates {
333        product.keep_live_coordinates(record, system);
334    }
335    Ok(product)
336}

338#[allow(clippy::too_many_arguments)]
339fn convolve2d_typed<T: ConvolutionPixel>(
340    input: &dyn ImageInterface<T>,
341    coordinates: RecordValue,
342    beams: &RecordValue,
343    region: Option<SubimageRegion<'_>>,
344    mask: &str,
345    geometry: GaussianConvolutionOptions,
346    scale: f64,
347    stretch: bool,
348    outfile: &str,
349    destination: Option<&Path>,
350) -> ImageToolResult<(Storage<T>, Option<RecordValue>)> {
351    let plan = ImageGaussianConvolution::new(input, coordinates, beams, region, geometry)?;
352    let options = ConvolutionOptions {
353        scale,
354        mask: Some(mask),
355        stretch,
356        destination: (!outfile.is_empty()).then(|| Path::new(outfile)),
357    };
358    let section = plan.section();
359    let mut result = Storage::create(
360        section.shape.clone(),
361        section.coordinates.clone(),
362        destination,
363        None,
364    )?;
365    with_storage!(&mut result, image => plan.copy_to(input, options, image.as_mut()))?;
366    with_storage!(&mut result, image => image.set_image_info_record(plan.image_info_record().clone()))?;
367    Ok((result, section.coordinate_record.clone()))
368}

370fn beam_history(description: &str, record: Option<&RecordValue>) -> ImageToolResult<String> {
371    let Some(info) = record else {
372        return Ok(format!("{description} has no beam"));
373    };
374    let beams = ImageInfo::from_record(info)?.beam_set;
375    let format_beam = |beam: GaussianBeam| -> ImageToolResult<String> {
376        let mut stored = None;
377        for pol in 0..beams.n_stokes() {
378            for channel in 0..beams.n_channels() {
379                if *beams.beam(channel, pol) == beam {
380                    stored = Some(stored_beam_record(Some(info), &beams, channel, pol));
381                    break;
382                }
383            }
384        }
385        let rec = stored.unwrap_or_else(|| beam.to_record());
386        let s = StoredGaussianBeam::from_record(&rec)?;
387        Ok([
388            BeamComponent::Major,
389            BeamComponent::Minor,
390            BeamComponent::PositionAngle,
391        ]
392        .into_iter()
393        .zip(["major", "minor", "pa"])
394        .map(|(component, name)| {
395            let q = s.component(component);
396            format!("{name}: {} {}", stream_double(q.value()), q.unit().name())
397        })
398        .collect::<Vec<_>>()
399        .join(", "))
400    };
401    if beams.is_multi() {
402        let min = format_beam(beams.min_area_beam().copied().unwrap_or_default())?;
403        let max = format_beam(beams.max_area_beam().copied().unwrap_or_default())?;
404        let median = format_beam(beams.median_area_beam().unwrap_or_default())?;
405        Ok(format!(
406            "{description} has multiple beams. Min area beam: {min}. Max area beam: {max}. Median area beam {median}"
407        ))
408    } else if let Some(beam) = beams.single_beam().filter(|beam| !beam.is_null()) {
409        Ok(format!("{description} resolution {}", format_beam(beam)?))
410    } else {
411        Ok(format!("{description} has no beam"))
412    }
413}

415/// Computes FFT of `attached` and writes requested `[real, imag, amp, phase, complex]` outputs (`ia.fft`).
416#[allow(clippy::too_many_arguments)]
417pub fn fft_image(
418    attached: &AttachedImage,
419    outfiles: [&str; 5],
420    axes: &[i64],
421    region: Option<SubimageRegion<'_>>,
422    mask: &str,
423    stretch: bool,
424    history_command: Option<String>,
425    is_external_open: ExternalPathCheck<'_>,
426) -> ImageToolResult<bool> {
427    let destinations: Vec<_> = outfiles
428        .iter()
429        .filter(|name| !name.is_empty())
430        .map(Path::new)
431        .collect();
432    if destinations.is_empty() {
433        return Err(ImageToolError::new(
434            "No output image names have been specified",
435        ));
436    }
437    publish_image_set(outfiles, false, is_external_open, |staged| {
438        read_locked(attached, |source| {
439            for target in &destinations {
440                check_handle_aliasing(source, attached, target, None)?;
441            }
442            if let ImageHandle::Float(storage) = source {
443                storage.with_float_view(|img| {
444                    fft_compute(
445                        attached,
446                        source,
447                        img,
448                        axes,
449                        region,
450                        mask,
451                        stretch,
452                        &destinations,
453                        staged,
454                        history_command,
455                    )
456                })
457            } else {
458                with_pixels!(source, pixels => with_storage!(pixels, img => {
459                    fft_compute(
460                        attached,
461                        source,
462                        img.as_ref(),
463                        axes,
464                        region,
465                        mask,
466                        stretch,
467                        &destinations,
468                        staged,
469                        history_command,
470                    )
471                }))
472            }
473        })
474    })?;
475    Ok(true)
476}

503fn create_fft_product<T: ImagePixel>(
504    plan: &ImageFourier<'_>,
505    path: &Path,
506) -> ImageToolResult<Storage<T>> {
507    let shape = plan.shape().to_vec();
508    let tiled = TiledShape::new(shape.clone(), std::mem::size_of::<T>())?;
509    PagedImage::create_with_tile_shape_and_cache(
510        shape,
511        tiled.tile_shape().to_vec(),
512        plan.coordinates().clone(),
513        path,
514        16 * 1024 * 1024,
515    )
516    .map(|image| Storage::Paged(Box::new(image)))
517    .map_err(ImageToolError::from)
518}

520#[allow(clippy::too_many_arguments)]
521fn fft_compute<T: FourierPixel>(
522    attachment: &AttachedImage,
523    source: &ImageHandle,
524    input: &dyn ImageInterface<T>,
525    axes: &[i64],
526    region: Option<SubimageRegion<'_>>,
527    mask: &str,
528    stretch: bool,
529    destinations: &[&Path],
530    staged: [Option<&Path>; 5],
531    history_command: Option<String>,
532) -> ImageToolResult<()>
533where
534    T::Scalar: FftProductPixel,
535    Complex<T::Scalar>: FftProductPixel,
536{
537    let beams = attachment.beam_view();
538    let plan = ImageFourier::new(
539        input,
540        source.coordinate_record(),
541        beams.record(),
542        FourierOptions {
543            region,
544            axes,
545            mask: Some(mask),
546            stretch,
547            destinations,
548        },
549    )?;
550    let section = plan.section();
551    let original = beams.beam_set()?;
552    let channels = section.beam_planes(
553        input.find_axis(CoordinateType::Spectral),
554        original.n_channels(),
555    );
556    let polarizations =
557        section.beam_planes(input.find_axis(CoordinateType::Stokes), original.n_stokes());
558    let base_info = section.image_info().to_record();
559    let scratch = plan.transform(input)?;
560    type ExtractScalar<S> = fn(Complex<S>) -> S;
561    let components: [ExtractScalar<T::Scalar>; 4] = [|v| v.re, |v| v.im, T::amplitude, T::phase];
562    let history = source.history_entries()?;
563    for (index, component) in components.into_iter().enumerate() {
564        if let Some(path) = staged[index] {
565            let mut output = create_fft_product(&plan, path)?;
566            with_storage!(&mut output, img => {
567                plan.write_product(input, &scratch, img.as_mut(), index == 3, component)
568            })?;
569            apply_beam_subset(
570                &mut output,
571                base_info.clone(),
572                &beams,
573                &channels,
574                &polarizations,
575            )?;
576            let mut handle = T::Scalar::wrap_fft(output);
577            handle.set_coordinate_record(plan.coordinate_record().clone())?;
578            finalize_output_history(&mut handle, &history, "fft", history_command.clone(), None)?;
579        }
580    }
581    if let Some(path) = staged[4] {
582        let mut output = create_fft_product(&plan, path)?;
583        with_storage!(&mut output, img => {
584            plan.write_product(input, &scratch, img.as_mut(), false, |v| v)
585        })?;
586        apply_beam_subset(&mut output, base_info, &beams, &channels, &polarizations)?;
587        let mut handle = Complex::<T::Scalar>::wrap_fft(output);
588        handle.set_coordinate_record(plan.coordinate_record().clone())?;
589        finalize_output_history(&mut handle, &history, "fft", history_command, None)?;
590    }
591    Ok(())
592}

594/// Separable axis-by-axis convolution (`ia.sepconvolve`).
595pub fn sepconvolve_image(
596    attached: &AttachedImage,
597    outfile: &str,
598    overwrite: bool,
599    options: SepConvolutionOptions<'_>,
600    history_command: Option<String>,
601    is_external_open: ExternalPathCheck<'_>,
602) -> ImageToolResult<AttachedImage> {
603    let destination = options.destination;
604    let region = options.region;
605    publish_image(outfile, overwrite, is_external_open, |stage| {
606        read_locked(attached, |handle| {
607            handle.check_single_precision("sepconvolve")?;
608            let ImageHandle::Float(source) = handle else {
609                return Err(ImageToolError::new("Exception: image cannot be NULL."));
610            };
611            if let Some(dest) = destination {
612                check_handle_aliasing(handle, attached, dest, None)?;
613                check_region_overlap(outfile, region)?;
614            }
615            let beams = attached.beam_view().clone();
616            let coord_rec = handle.coordinate_record();
617            let mut output = source.with_float_view(|image| {
618                sepconvolve_float(image, coord_rec, &beams, options, stage)
619            })?;
620            let entries = handle.history_entries()?;
621            finalize_output_history(&mut output, &entries, "sepconvolve", history_command, None)?;
622            Ok(output)
623        })
624    })
625}

627fn sepconvolve_float(
628    image: &dyn ImageInterface<f32>,
629    record: RecordValue,
630    beams: &BeamView,
631    options: SepConvolutionOptions<'_>,
632    stage: Option<&Path>,
633) -> ImageToolResult<ImageHandle> {
634    let mut plan = ImageSepConvolution::new(image, record, beams.record(), options)?;
635    let beam_set = beams.beam_set()?;
636    let channels = plan.section.beam_planes(
637        image.find_axis(CoordinateType::Spectral),
638        beam_set.n_channels(),
639    );
640    let polarizations = plan
641        .section
642        .beam_planes(image.find_axis(CoordinateType::Stokes), beam_set.n_stokes());
643    let mut output = Storage::create(
644        plan.section.shape.clone(),
645        plan.section.coordinates.clone(),
646        stage,
647        None,
648    )?;
649    with_storage!(&mut output, product => plan.copy_to(image, product.as_mut()))?;
650    apply_beam_subset(
651        &mut output,
652        plan.section.image_info().to_record(),
653        beams,
654        &channels,
655        &polarizations,
656    )?;
657    set_storage_coordinate_record(&mut output, plan.section.coordinate_record.take())?;
658    Ok(ImageHandle::Float(output))
659}

661/// 1D Hanning or Boxcar smoothing along an axis (`ia.hanning`, `ia.boxcar`).
662#[allow(clippy::too_many_arguments)]
663pub fn smooth_axis(

665    outfile: &str,
666    overwrite: bool,
667    options: SmoothingOptions<'_>,
668    history_fn: Option<&dyn Fn(usize) -> String>,
669    is_external_open: ExternalPathCheck<'_>,
670) -> ImageToolResult<AttachedImage> {
671    let kernel = options.kernel;
672    let method_name = match kernel {
673        SmoothingKernel::Boxcar(_) => "boxcar",
674        SmoothingKernel::Hanning => "hanning",
675    };
676    let mut live_coordinates = None;
677    let mut product = publish_image(outfile, overwrite, is_external_open, |path| {
678        read_locked(attached, |image| {
679            image.check_single_precision(method_name)?;
680            if !outfile.is_empty() {
681                check_handle_aliasing(image, attached, Path::new(outfile), None)?;
682            }
683            let beams = attached.beam_view().clone();
684            let record = image.coordinate_record();
685            let (mut output, axis) = match image {
686                ImageHandle::Float(storage) => storage.with_float_view(|img| {
687                    let (st, ax) = smooth_typed(img, record, &beams, options, path)?;
688                    Ok::<_, ImageToolError>((ImageHandle::Float(st), ax))
689                })?,
690                ImageHandle::Complex(storage) => with_storage!(storage, img => {
691                    let (st, ax) = smooth_typed(img.as_ref(), record, &beams, options, path)?;
692                    Ok::<_, ImageToolError>((ImageHandle::Complex(st), ax))
693                })?,
694                _ => unreachable!("checked single precision"),
695            };
696            let rows = image.history_entries()?;
697            output.append_history_entries(&rows)?;
698            if let Some(build_cmd) = history_fn {
699                let cmd = build_cmd(axis);
700                match kernel {
701                    SmoothingKernel::Boxcar(_) => output.record_history(method_name, cmd)?,
702                    SmoothingKernel::Hanning => {
703                        output.append_history_at(
704                            "::image::_hanning",
705                            &["Ran ia.hanning".into(), cmd],
706                        )?;
707                    }
708                }
709            }
710            let source_name = image.path().map_or_else(
711                || "Temporary_Image".into(),
712                |name| name.display().to_string(),
713            );
714            let origin = match kernel {
715                SmoothingKernel::Boxcar(_) => "::ImageBoxcarSmoother::_reportOldNewImageShapes",
716                SmoothingKernel::Hanning => "::HanningSmoother::_reportOldNewImageShapes",
717            };
718            output.append_history_at(
719                origin,
720                &[
721                    format!("Original {source_name} size => {:?}", image.shape()),
722                    format!("New {outfile} size => {:?}", output.shape()),
723                ],
724            )?;
725            output.save()?;
726            live_coordinates = AttachedImage::capture_live_coordinates(&output)?;
727            Ok(output)
728        })
729    })?;
730    if let Some((record, system)) = live_coordinates {
731        product.keep_live_coordinates(record, system);
732    }
733    Ok(product)
734}
735
736fn smooth_typed<T: SmoothingPixel>(
737    image: &dyn ImageInterface<T>,
738    coordinates: RecordValue,
739    beams: &BeamView,
740    options: SmoothingOptions<'_>,
741    path: Option<&Path>,
742) -> ImageToolResult<(Storage<T>, usize)> {
743    let mut plan = ImageSmoothing::new(image, coordinates, options, beams.record())?;
744    let mut output = Storage::create(
745        plan.section.shape.clone(),
746        plan.section.coordinates.clone(),
747        path,
748        None,
749    )?;
750    with_storage!(&mut output, product => plan.copy_to(image, product.as_mut()))?;
751    let mut info = plan.section.image_info().to_record();
752    beams.apply_to(&mut info);
753    with_storage!(&mut output, product => product.set_image_info_record(info))?;
754    set_storage_coordinate_record(&mut output, plan.section.coordinate_record.take())?;
755    Ok((output, plan.axis()))
756}
