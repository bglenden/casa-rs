2//! The store law (#932, #942): the batch geometry ([`GeometryPlan`]) of an
3//! observation's samples in its MeasurementSet equals their batch geometry
4//! in another store of the same observation (an MSv4 conversion), sample
5//! for sample, to [`TOLERANCE`]: what the store alone changes. Every sample
6//! of every partition is compared, on every channel; the request gives the
7//! geometry (phase centre, image domains, pointing), not the samples.
8//!
9//! A MeasurementSet sample is its MAIN row
10//! ([`Msv2Observation::main_row`]), so padding matches nothing. A sample of
11//! the other store is matched through its partition's content, its
12//! antennas by name and its time to the millisecond, a key two of the
13//! MeasurementSet's rows may share; such rows, and rows two samples claim,
14//! are counted as duplicates and fail the law.
15

35/// The law's tolerance: every quantity relative to its scale.
36pub const TOLERANCE: f64 = 1.0e-12;
37
38/// One request's two geometries compared.
39#[derive(Clone, Debug, Default)]
40pub struct GeometryComparison {
41    /// Rows the MeasurementSet's batches delivered.
42    pub rows: usize,
43    /// Of them, rows no sample of the other store matched.
44    pub unmatched: usize,
45    /// Rows two samples of the other store matched, and MeasurementSet
46    /// rows that share another's key.
47    pub duplicates: usize,
48    /// Per quantity, the largest difference relative to its scale and the
49    /// physical row where it occurred.
50    pub differences: BTreeMap<&'static str, (f64, u64)>,
51    /// Per quantity, the values compared.
52    pub compared: BTreeMap<&'static str, usize>,
53}
54
55impl GeometryComparison {
56    /// Whether the law holds: there were rows, every one matched exactly
57    /// once, and no quantity differs by more than [`TOLERANCE`].
58    #[must_use]
59    pub fn holds(&self) -> bool {
60        self.rows > 0 && self.unmatched == 0 && self.duplicates == 0 && self.largest() <= TOLERANCE
61    }
62
63    /// The largest relative difference of any quantity.
64    #[must_use]
65    pub fn largest(&self) -> f64 {
66        self.differences
67            .values()
68            .map(|(difference, _)| *difference)
69            .fold(0.0, worst)
70    }
71
72    /// Record one difference; a NaN one (either side absent or not finite)
73    /// counts as infinite.
74    fn record(&mut self, quantity: &'static str, difference: f64, row: u64) {
75        *self.compared.entry(quantity).or_default() += 1;
76        let difference = worst(difference, 0.0);
77        let entry = self.differences.entry(quantity).or_insert((0.0, row));
78        if difference > entry.0 {
79            *entry = (difference, row);
80        }
81    }
82}
83
84/// The larger of two differences, a NaN infinite: `f64::max` would drop it.
85fn worst(a: f64, b: f64) -> f64 {
86    if a.is_nan() || b.is_nan() {
87        f64::INFINITY
88    } else {
89        a.max(b)
90    }
91}

96/// A row's identity in another store: data description, antennas, and
97/// time in milliseconds.
98type Key = (u32, u32, u32, i64);
99
100fn key(data_description: u32, antennas: [u32; 2], time_s: f64) -> Key {
101    (
102        data_description,
103        antennas[0],
104        antennas[1],
105        (time_s * 1.0e3).round() as i64,
106    )
107}

126/// Compare the batch geometry of every sample of `request`'s
127/// MeasurementSet with its batch geometry in `converted`, the same
128/// observation in another store (an MSv4 conversion). A converted partition
129/// is matched to the data description of the MeasurementSet's partitions of
130/// the same content
131/// ([`PartitionEntry::same_content`](casa_observation::PartitionEntry::same_content)),
132/// and a converted antenna to the MeasurementSet's of its name.
133///
134/// # Errors
135///
136/// When the request does not compile, or either store cannot be opened or
137/// read.
138pub fn compare_stores(
139    request: &ImagingRequest,
140    context: &RunContext,
141    converted: &Path,
142) -> Result<GeometryComparison, ApplicationError> {
143    let resolved = std::cell::Cell::new(None);
144    let (problem, input, _native, _resident) = compile_run(request, context, &resolved)?;
145    let OpenedObservation {
146        store,
147        measures,
148        phase_table,
149    } = input.observation;
150    let observation = &store.value;
151    let inputs = PlanInputs {
152        measures,
153        phase_table,
154    };
155    let mut keys = HashMap::new();
156    let mut shared = 0;
157    let reference = batch_rows(
158        &problem,
159        observation,
160        (
161            &observation.catalogue().partition_ids(),
162            &FieldSelection::All,
163        ),
164        |batch, sample| {
165            let per_time = batch.samples_per_time();
166            let baseline = batch.baselines[sample % per_time];
167            let row = main_row(observation, batch, sample)?;
168            let identity = key(
169                observation.data_description_id(batch.partition),
170                [baseline.antenna1.0, baseline.antenna2.0],
171                batch
172                    .time
173                    .seconds_since(sample / per_time, MjdHighPrec::new(0.0, 0.0)),
174            );
175            if keys
176                .insert(identity, row)
177                .is_some_and(|earlier| earlier != row)
178            {
179                shared += 1;
180            }
181            Some(row)
182        },
183        inputs.clone(),
184        None,
185    )?;
186    let other = casa_ms::observation_io::open(converted, &OpenOptions::default())?;
187    let bridge = Bridge::new(observation, other.catalogue());
188    let rows = batch_rows(
189        &problem,
190        other.as_ref(),
191        (&bridge.partitions(), &FieldSelection::All),
192        |batch, sample| bridge.row(batch, sample, &keys),
193        inputs,
194        Some(&reference.rows),
195    )?;
196    let mut comparison = compare(&reference.rows, &rows);
197    comparison.duplicates += reference.duplicates + shared;
198    Ok(comparison)
199}

201/// The time (MJD days) and first antenna's pointing direction of each MAIN
202/// row `request` selects (its partitions and fields; a UV-distance interval
203/// is not applied), from its batch geometry: under a
204/// pointing law that follows the phase-tracking centre, the phase centre
205/// at the row's time. What CASA oracles of a moving phase centre compare
206/// (T41's `TRACKFIELD`).
207///
208/// # Errors
209///
210/// When the request does not compile or the MeasurementSet cannot be read.
211pub fn row_pointing(
212    request: &ImagingRequest,
213    context: &RunContext,
214) -> Result<BTreeMap<u64, (f64, SkyDirection)>, ApplicationError> {
215    let resolved = std::cell::Cell::new(None);
216    let (problem, input, _native, _resident) = compile_run(request, context, &resolved)?;
217    let OpenedObservation {
218        store,
219        measures,
220        phase_table,
221    } = input.observation;
222    let observation = &store.value;
223    let selection = crate::compile::resolve_partitions(&problem, observation)?;
224    let partitions: Vec<PartitionId> = selection
225        .partitions
226        .iter()
227        .map(|partition| partition.id)
228        .collect();
229    let found = batch_rows(
230        &problem,
231        observation,
232        (&partitions, &selection.fields),
233        |batch, sample| main_row(observation, batch, sample),
234        PlanInputs {
235            measures,
236            phase_table,
237        },
238        None,
239    )?;
240    Ok(found
241        .rows
242        .into_iter()
243        .map(|(row, geometry)| (row, (geometry.time_mjd_days, geometry.pointing.antenna1)))
244        .collect())
245}

247/// The MAIN row of `sample` of `batch`; `None` for padding.
248fn main_row(
249    observation: &Msv2Observation,
250    batch: &VisibilityBatch<'_>,
251    sample: usize,
252) -> Option<u64> {
253    let per_time = batch.samples_per_time();
254    let baseline = batch.baselines[sample % per_time];
255    let time_s = batch.time.offsets_s()[sample / per_time];
256    observation
257        .main_row(batch.partition, time_s, baseline)
258        .map(u64::from)
259}

269/// The rows a store's batch samples matched.
270struct Found {
271    rows: Rows,
272    /// Samples that matched a row an earlier sample had.
273    duplicates: usize,
274}
275
276/// `actual`'s rows against `expected`'s.
277fn compare(expected: &Rows, actual: &Found) -> GeometryComparison {
278    let mut comparison = GeometryComparison {
279        rows: expected.len(),
280        duplicates: actual.duplicates,
281        ..GeometryComparison::default()
282    };
283    for (row, want) in expected {
284        match actual.rows.get(row) {
285            Some(got) => compare_row(want, got, &mut comparison),
286            None => comparison.unmatched += 1,
287        }
288    }
289    comparison
290}

292/// The batch geometry of the rows of `partitions` of `observation` of
293/// `fields`; `row_of` gives a sample's row, `None` for padding or a sample
294/// no row matches. With `expected`, only its rows, on its channels.
295fn batch_rows(
296    problem: &CompiledProblem,
297    observation: &dyn Observation,
298    (partitions, fields): (&[PartitionId], &FieldSelection),
299    mut row_of: impl FnMut(&VisibilityBatch<'_>, usize) -> Option<u64>,
300    inputs: PlanInputs,
301    expected: Option<&Rows>,
302) -> Result<Found, ApplicationError> {
303    let catalogue = Arc::clone(observation.catalogue());
304    let plan = GeometryPlan::compile(problem, &catalogue, inputs.measures, inputs.phase_table)?;
305    let mut traversal = GeometryTraversal::default();
306    let mut geometry = BatchGeometry::default();
307    let mut found = Found {
308        rows: HashMap::new(),
309        duplicates: 0,
310    };
311    let centroids = matches!(
312        problem.geometry().centres().pointing(),
313        PointingCentreLaw::Observation(law)
314            if law.time_sampling() == casa_imaging_model::PointingTimeSampling::VisibilityTimeCentroid
315    );
316    for &partition in partitions {
317        let entry = catalogue.partition(partition);
318        let data_group = entry
319            .data_groups
320            .first()
321            .ok_or_else(|| format!("partition {} has no data group", partition.0))?;
322        let request = AccessRequest {
323            partitions: vec![partition],
324            data_group: data_group.name.clone(),
325            roles: Roles {
326                values: false,
327                weights: false,
328                time_centroid: centroids,
329                ..Roles::PAYLOAD
330            },
331            channels: 0..entry.frequencies_hz.len(),
332            times: TimeSelection::All,
333            fields: fields.clone(),
334            weights: WeightRequest::Stored,
335            correlations: CorrelationRequest::ALL,
336            grouping: Grouping::None,
337            batch_bytes: 1 << 26,
338        };
339        let mut reader = observation.bind_read(&request)?;
340        let mut buffers = BatchBuffers::default();
341        while let Some(event) = reader.next(&mut buffers)? {
342            let ReadEvent::Batch(batch) = event else {
343                continue;
344            };
345            if batch.samples() == 0 {
346                continue;
347            }
348            let samples = (0..batch.samples())
349                .map(|sample| {
350                    row_of(&batch, sample)
351                        .filter(|row| expected.is_none_or(|rows| rows.contains_key(row)))
352                })
353                .collect::<Vec<_>>();
354            let rows = samples.iter().map(Option::is_some).collect::<Vec<_>>();
355            let pointing = plan
356                .reads_pointing()
357                .then(|| observation.pointing(&pointing_request(problem, &batch, partition)))
358                .transpose()?;
359            plan.apply(
360                &batch,
361                &rows,
362                pointing.as_ref(),
363                &mut traversal,
364                &mut geometry,
365            )?;
366            for (sample, row) in samples.iter().enumerate() {
367                let Some(row) = row else {
368                    continue;
369                };
370                let got = batch_row(
371                    &batch,
372                    &geometry,
373                    (sample, *row),
374                    plan.charts(),
375                    expected.map(|rows| &rows[row]),
376                );
377                if found.rows.insert(*row, got).is_some() {
378                    found.duplicates += 1;
379                }
380            }
381        }
382    }
383    Ok(found)
384}

386/// How another store's samples find the MeasurementSet's rows.
387struct Bridge {
388    /// Each matched converted partition's data description.
389    data_descriptions: BTreeMap<PartitionId, u32>,
390    /// Each converted antenna's MeasurementSet antenna.
391    antennas: Vec<Option<u32>>,
392}
393
394impl Bridge {
395    fn new(observation: &Msv2Observation, converted: &Catalogue) -> Self {
396        let reference = observation.catalogue();
397        let data_descriptions = converted
398            .partition_ids()
399            .into_iter()
400            .filter_map(|partition| {
401                let entry = converted.partition(partition);
402                let mut matching = reference
403                    .partition_ids()
404                    .into_iter()
405                    .filter(|candidate| reference.partition(*candidate).same_content(entry))
406                    .map(|candidate| observation.data_description_id(candidate));
407                let first = matching.next()?;
408                matching
409                    .all(|other| other == first)
410                    .then_some((partition, first))
411            })
412            .collect();
413        let antennas = converted
414            .antennas
415            .iter()
416            .map(|antenna| {
417                let mut matching = reference
418                    .antennas
419                    .iter()
420                    .enumerate()
421                    .filter(|(_, candidate)| same_antenna(candidate, antenna))
422                    .map(|(index, _)| index as u32);
423                let first = matching.next()?;
424                matching.next().is_none().then_some(first)
425            })
426            .collect();
427        Self {
428            data_descriptions,
429            antennas,
430        }
431    }
432
433    /// The converted partitions matched to a data description.
434    fn partitions(&self) -> Vec<PartitionId> {
435        self.data_descriptions.keys().copied().collect()
436    }
437
438    /// The MeasurementSet row `keys` gives `sample` of `batch`.
439    fn row(
440        &self,
441        batch: &VisibilityBatch<'_>,
442        sample: usize,
443        keys: &HashMap<Key, u64>,
444    ) -> Option<u64> {
445        let data_description = *self.data_descriptions.get(&batch.partition)?;
446        let per_time = batch.samples_per_time();
447        let baseline = batch.baselines[sample % per_time];
448        let antenna = |id: AntennaId| self.antennas.get(id.0 as usize).copied().flatten();
449        let time_s = batch
450            .time
451            .seconds_since(sample / per_time, MjdHighPrec::new(0.0, 0.0));
452        let antennas = [antenna(baseline.antenna1)?, antenna(baseline.antenna2)?];
453        keys.get(&key(data_description, antennas, time_s)).copied()
454    }
455}

457/// Whether `converted` names `reference`: as stored, or as XRADIO names an
458/// MSv4 antenna (`NAME_STATION`).
459fn same_antenna(reference: &AntennaEntry, converted: &AntennaEntry) -> bool {
460    converted.name == reference.name
461        || converted.name == format!("{}_{}", reference.name, reference.station)
462}

464/// The pointing request covering `batch`'s times for its partition's
465/// antennas.
466fn pointing_request(
467    problem: &CompiledProblem,
468    batch: &VisibilityBatch<'_>,
469    partition: PartitionId,
470) -> PointingRequest {
471    let mut antennas: Vec<AntennaId> = batch
472        .baselines
473        .iter()
474        .flat_map(|baseline| [baseline.antenna1, baseline.antenna2])
475        .collect();
476    antennas.sort_unstable();
477    antennas.dedup();
478    let times = batch.time.offsets_s();
479    let mut span = times[0]..=times[times.len() - 1];
480    if let Some(centroid) = batch.timing.centroid {
481        let offsets = centroid.offsets_s();
482        let low = offsets.iter().copied().fold(*span.start(), f64::min);
483        let high = offsets.iter().copied().fold(*span.end(), f64::max);
484        span = low..=high;
485    }
486    let direction = crate::imaging::pointing_direction(problem);
487    PointingRequest {
488        partition,
489        direction,
490        antennas,
491        times: span,
492        budget_bytes: 1 << 34,
493    }
494}

496/// Sample `sample`'s geometry as MAIN row `row`, on every channel the
497/// batch holds or, with `expected`, on its channels.
498fn batch_row(
499    batch: &VisibilityBatch<'_>,
500    geometry: &BatchGeometry,
501    (sample, row): (usize, u64),
502    charts: usize,
503    expected: Option<&Row>,
504) -> Row {
505    let time = sample / batch.samples_per_time();
506    let frequencies = geometry.frequencies_hz(time);
507    Row {
508        physical_row: row,
509        time_mjd_days: batch.time.seconds_since(time, MjdHighPrec::new(0.0, 0.0)) / 86_400.0,
510        field: batch.field[time].0,
511        raw_uvw_m: batch
512            .group
513            .uvw
514            .map_or([f64::NAN; 3], |uvw| uvw.metres_at(sample)),
515        projections: (0..charts)
516            .map(|chart| {
517                let projection = geometry.projection(sample, chart);
518                (projection.uvw_m, projection.phase_shift_m)
519            })
520            .collect(),
521        aw_pixels: (0..charts)
522            .map(|chart| geometry.aw_pointing_pixel(sample, chart))
523            .collect(),
524        pointing: geometry.pointing(sample),
525        parallactic: geometry.parallactic_angles(sample),
526        responses: geometry.antenna_responses(sample),
527        frequencies: match expected {
528            Some(expected) => expected
529                .frequencies
530                .iter()
531                .map(|(channel, _)| {
532                    let index = *channel as usize - batch.first_channel;
533                    (
534                        *channel,
535                        frequencies.get(index).copied().unwrap_or(f64::NAN),
536                    )
537                })
538                .collect(),
539            None => frequencies
540                .iter()
541                .enumerate()
542                .map(|(index, frequency)| ((batch.first_channel + index) as u32, *frequency))
543                .collect(),
544        },
545    }
546}

548/// Record the differences of one row.
549fn compare_row(expected: &Row, actual: &Row, comparison: &mut GeometryComparison) {
550    let row = expected.physical_row;
551    let length = norm(expected.raw_uvw_m).max(1.0);
552    let exact = |same: bool| if same { 0.0 } else { f64::INFINITY };
553    comparison.record("field", exact(expected.field == actual.field), row);
554    comparison.record(
555        "raw uvw",
556        norm(difference(expected.raw_uvw_m, actual.raw_uvw_m)) / length,
557        row,
558    );
559    comparison.record(
560        "charts",
561        exact(expected.projections.len() == actual.projections.len()),
562        row,
563    );
564    for (want, got) in expected.projections.iter().zip(&actual.projections) {
565        comparison.record("uvw", norm(difference(want.0, got.0)) / length, row);
566        comparison.record("phase shift", (want.1 - got.1).abs() / length, row);
567    }
568    for (want, got) in expected.aw_pixels.iter().zip(&actual.aw_pixels) {
569        let difference = match (want, got) {
570            (None, None) => continue,
571            (Some(want), Some(got)) => {
572                ((want[0] - got[0]).hypot(want[1] - got[1])) / want[0].hypot(want[1]).max(1.0)
573            }
574            _ => f64::INFINITY,
575        };
576        comparison.record("AW pointing pixel", difference, row);
577    }
578    let angle = |want: SkyDirection, got: SkyDirection| {
579        worst(
580            exact(want.frame() == got.frame()),
581            worst(
582                (want.longitude_rad() - got.longitude_rad()).abs(),
583                (want.latitude_rad() - got.latitude_rad()).abs(),
584            ),
585        )
586    };
587    comparison.record(
588        "pointing",
589        worst(
590            angle(expected.pointing.antenna1, actual.pointing.antenna1),
591            angle(expected.pointing.antenna2, actual.pointing.antenna2),
592        ),
593        row,
594    );
595    match (expected.parallactic, actual.parallactic) {
596        (None, None) => {}
597        (Some(want), Some(got)) => comparison.record(
598            "parallactic angle",
599            worst((want[0] - got[0]).abs(), (want[1] - got[1]).abs()),
600            row,
601        ),
602        _ => comparison.record("parallactic angle", f64::INFINITY, row),
603    }
604    if expected.responses.is_some() || actual.responses.is_some() {
605        comparison.record(
606            "aperture classes",
607            exact(expected.responses == actual.responses),
608            row,
609        );
610    }
611    for ((_, want), (_, got)) in expected.frequencies.iter().zip(&actual.frequencies) {
612        comparison.record("frequency", (want - got).abs() / want.abs(), row);
613    }
614}

616fn difference(a: [f64; 3], b: [f64; 3]) -> [f64; 3] {
617    [a[0] - b[0], a[1] - b[1], a[2] - b[2]]
618}
619
620fn norm(vector: [f64; 3]) -> f64 {
621    vector.iter().map(|value| value * value).sum::<f64>().sqrt()
