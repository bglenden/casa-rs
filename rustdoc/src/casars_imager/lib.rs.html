4//! The command-line and task surface of native CASA-RS imaging, on the
5//! shared task front end (#931).
6//!
7//! The front end resolves the imager's parameters through the provider-
8//! contracts catalog: command-line flags and a `--json-run` request become
9//! the same resolved values, and those values, and nothing else, make the
10//! [`ImagingRequest`] the application runs. A run's reply carries its
11//! [`ImagerRunTaskResult`]; the imager returns None, as CASA's `tclean` does
12//! unless asked for its history.
13
14pub mod interrupt;
15mod schema;
16mod task_contract;

30fn imager_surface() -> Result<SurfaceContractBundle, String> {
31    casa_provider_contracts::builtin_surface_bundle("imager")
32}
33
34fn workspace() -> Result<PathBuf, String> {
35    std::env::current_dir().map_err(|error| format!("the working directory: {error}"))
36}

38/// The request `resolved`, every active imager parameter, makes. A value the
39/// catalog accepts but the request's own read refuses (`scales = "0,x"`)
40/// is refused at its parameter.
41fn request_from(
42    resolved: Map<String, Value>,
43) -> Result<(ImagingParameters, ImagingRequest), TaskFailure> {
44    let request = serde_path_to_error::deserialize(Value::Object(resolved.clone())).map_err(
45        |error: serde_path_to_error::Error<serde_json::Error>| match error.path().iter().next() {
46            Some(serde_path_to_error::Segment::Map { key }) => {
47                TaskFailure::validation(format!("imager parameter {key:?}: {}", error.inner()))
48                    .at(key.clone())
49            }
50            _ => TaskFailure::validation(format!("imager parameters: {}", error.inner())),
51        },
52    )?;
53    Ok((ImagingParameters(resolved), request))
54}
55
56/// Resolve `parameters`, catalog-named and sparse, over the catalog's
57/// defaults: the resolved parameters and the request they make.
58pub fn resolve_request(
59    parameters: &ImagingParameters,
60) -> Result<(ImagingParameters, ImagingRequest), String> {
61    let resolved = resolve_plain_values(imager_surface()?, workspace()?, &parameters.0)
62        .map_err(|error| format!("resolve imager parameters: {error}"))?;
63    request_from(resolved).map_err(|failure| failure.message)
64}

66/// Run `request`, resolved as `parameters`: the run summary, with the
67/// parameters as its echo, is written beside the products
68/// (`<imagename>.summary.json`); SIGINT cancels the run.
69pub fn run(
70    parameters: ImagingParameters,
71    request: &ImagingRequest,
72) -> Result<ImagerRunTaskResult, String> {
73    let started = Instant::now();
74    let mut summary = request.imagename.clone().into_os_string();
75    summary.push(".summary.json");
76    let context = RunContext {
77        host: HostResources::detect().map_err(|error| error.to_string())?,
78        policy: request.resource_policy(),
79        cancel: interrupt::token().clone(),
80        summary: Some(SummaryTarget {
81            path: summary.into(),
82            request: Value::Object(parameters.0.clone()),
83        }),
84    };
85    let outcome = execute(request, context).map_err(|error| error.to_string())?;
86    Ok(ImagerRunTaskResult::from_outcome(
87        parameters,
88        request,
89        &outcome,
90        started.elapsed(),
91    ))
92}

94/// The front end of the `casars-imager` binary, which hosts `imager`.
95pub fn task_front_end()
96-> TaskFrontEnd<impl Fn(&TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure>> {
97    TaskFrontEnd::new(
98        "casars-imager",
99        imager_task_schema_bundle(),
100        &["imager"],
101        run_task,
102    )
103    .expect("the imager contract carries its surface")
104}
105
106/// The resolved parameters of `task`, an imager run, and the request they
107/// make; a request they cannot make is refused.
108pub fn task_request(
109    task: &TaskInvocation<'_>,
110) -> Result<(ImagingParameters, ImagingRequest), TaskFailure> {
111    request_from(task.plain_values())
112}
113
114/// Run `task`, one resolved imager request, as the `casars-imager` binary
115/// does: the execution callback of [`task_front_end`]. SIGINT cancels it:
116/// the run stops at its next safe point, publishes nothing, and replies that
117/// it was interrupted.
118pub fn run_task(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
119    let (parameters, request) = task_request(task)?;
120    interrupt::install().map_err(|error| {
121        TaskFailure::execution(format!("install the interrupt handler: {error}"))
122    })?;
123    let result = run(parameters, &request).map_err(|error| {
124        if interrupt::token().is_cancelled() {
125            TaskFailure::interrupted(format!("{error}; nothing was published"))
126        } else {
127            TaskFailure::execution(error)
128        }
129    })?;
130    let text = format!(
131        "Wrote CASA-compatible products at prefix {} ({} gridded samples, {} major cycles, \
132         {} reported minor iterations, {} actual components, stop={:?})\n",
133        request.imagename.display(),
134        result.run.gridded_samples,
135        result.run.major_cycles,
136        result.run.minor_iterations,
137        result.run.actual_minor_iterations,
138        result.run.clean_stop_reason,
139    );
140    Ok(TaskSuccess::new(&(), &result)?.with_text(text))
141}
