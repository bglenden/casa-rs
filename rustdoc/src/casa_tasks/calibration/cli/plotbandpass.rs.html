37pub(super) fn run(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
38    let caltable = task.input_path("caltable")?;
39    let axis = task.text("xaxis")?.unwrap_or_default();
40    let frequency_axis = match axis.as_str() {
41        "chan" | "channel" => false,
42        "freq" | "frequency" => true,
43        _ => {
44            return Err(
45                TaskFailure::validation("xaxis must name channel or frequency").at("xaxis"),
46            );
47        }
48    };
49    let raw_overlay = task.text("overlay")?.unwrap_or_default();
50    let overlay = overlay(&raw_overlay)?;
51    let y_axis = task.text("yaxis")?.unwrap_or_default();
52    let both = y_axis.starts_with("ap") || y_axis.starts_with("both");
53    let reference_field = selector_numeric_text(task, "showatmfield")?;
54    let image_sideband = task.flag("showimage")?;
55    let show_atmosphere = task.flag("showatm")?;
56    let transmission = show_atmosphere
57        || (!task.flag("showtsky")? && (image_sideband || !reference_field.is_empty()));
58    let sky_temperature = !show_atmosphere && task.flag("showtsky")?;
59    let lo = if image_sideband {
60        match local_oscillator(task)? {
61            Some(lo) => lo,
62            None => {
63                return plain(
64                    CasaValue::None,
65                    false,
66                    "lo1 must be a float (entered as a string or number)",
67                );
68            }
69        }
70    } else {
71        None
72    };
73    if !reference_field.is_empty() && !overlay.time {
74        return plain(
75            CasaValue::None,
76            true,
77            "The showatmfield only has meaning for overlay='time'.",
78        );
79    }
80    if both && (overlay.antenna || overlay.time || overlay.window) {
81        return plain(
82            CasaValue::None,
83            true,
84            "Option 'overlay' is incompatible with yaxis='both'. Pick either amp or phase.",
85        );
86    }
87    let show_fdm = task.flag("showfdm")?;
88    if show_fdm && !frequency_axis {
89        return plain(
90            CasaValue::None,
91            true,
92            "The option showfdm=True requires xaxis='freq'.",
93        );
94    }
95    let Some(layout) = subplot(task)? else {
96        return plain(
97            CasaValue::None,
98            true,
99            "Invalid subplot size. Use 11 through 81, or 22, 32, or 42.",
100        );
101    };
102    let limits = match plot_limits(task)? {
103        Some(limits) => limits,
104        None => {
105            return plain(
106                CasaValue::None,
107                true,
108                "plotrange must be an array: e.g. [0,1,-180,180]",
109            );
110        }
111    };
112    let phase_limits = match phase_limits(task)? {
113        Some(limits) => limits.filter(|_| both),
114        None => {
115            return plain(
116                CasaValue::None,
117                true,
118                "phase must be either '' or 2 values: [x,y]",
119            );
120        }
121    };
122    let channel_range = match channel_range(task, frequency_axis) {
123        Ok(range) if range.is_none() || limits == [0.0; 4] => range,
124        _ => {
125            return plain(
126                CasaValue::None,
127                true,
128                "Use a channel range with xaxis='freq' and default plotrange.",
129            );
130        }
131    };
132    let edge = task.integer("edge")?;
133    let resample = task.integer("resample")?;
134    if edge < 0 || resample < 1 {
135        let message = if edge < 0 {
136            "edge must be >= 0"
137        } else {
138            "resample must be an integer >= 1"
139        };
140        return plain(
141            CasaValue::String {
142                value: String::new(),
143            },
144            false,
145            message,
146        );
147    }
148    let polarization = selector(task, "poln")?;
149    let polarizations = polarization
150        .iter()
151        .flat_map(|label| label.split(',').map(str::trim))
152        .filter(|label| !label.is_empty())
153        .collect::<Vec<_>>();
154    let vis = task.text("vis")?.filter(|name| !name.is_empty());
155    let caltable2 = task.text("caltable2")?.filter(|name| !name.is_empty());
156    let caltable3 = task.text("caltable3")?.filter(|name| !name.is_empty());
157    for path in [&caltable2, &caltable3].into_iter().flatten() {
158        if !Path::new(path).exists() {
159            return plain(
160                CasaValue::None,
161                true,
162                &format!("Could not find companion calibration table {path}"),
163            );
164        }
165    }
166    let water = selector_numeric_text(task, "pwv")?;
167    if let Some(parameter) = selector_with_names_in_vector(task) {
168        return plain(
169            CasaValue::None,
170            parameter != "spw",
171            "Specify names in a comma-delimited string or pass an array of numeric indices.",
172        );
173    }
174    let parent_ms = task.text("parentms")?.filter(|name| !name.is_empty());
175    let request = PlotRequest {
176        caltable: &caltable,
177        vis: vis.as_deref().map(Path::new),
178        companion_tables: [
179            caltable2.as_deref().map(Path::new),
180            caltable3.as_deref().map(Path::new),
181        ],
182        selectors: Selectors {
183            antenna: selector(task, "antenna")?,
184            field: selector(task, "field")?,
185            spw: selector(task, "spw")?,
186            time: selector(task, "timeranges")?,
187            scan: selector(task, "scans")?,
188            baseband: selector(task, "basebands")?,
189        },
190        overlay,
191        frequency_axis,
192        y_axis: &y_axis,
193        polarizations: &polarizations,
194        threshold_seconds: task.number("solutionTimeThresholdSeconds")?,
195        show_flagged: task.flag("showflagged")?,
196        show_baseband_number: task.flag("showBasebandNumber")?,
197        derivative_sigma: numeric(task, "channeldiff")?,
198        edge: edge as usize,
199        resample: unsigned(task, "resample")?,
200        platforming_threshold: task.number("platformingThreshold")?,
201        platforming_sigma: task.number("platformingSigma")?,
202        atmosphere: AtmosphericOptions {
203            transmission,
204            sky_temperature,
205            water: &water,
206            reference_field: &reference_field,
207            image_sideband,
208            lo_ghz: lo,
209            parent_ms: parent_ms.as_deref().map(Path::new),
210            show_points: task.flag("showatmpoints")?,
211            time_overlay: overlay.time,
212        },
213        show_fdm,
214        zoom_intersect: task.text("zoom")?.as_deref() == Some("intersect"),
215    };
216    let figfile = task.text("figfile")?.unwrap_or_default();
217    let display = DisplayOptions {
218        layout,
219        figfile: &figfile,
220        sequential_names: task.flag("figfileSequential")?,
221        density: unsigned(task, "density")?,
222        markersize: task.integer("markersize")? as f64,
223        show_points: automatic_flag(task, "showpoints")?,
224        show_lines: automatic_flag(task, "showlines")?,
225        interactive: task.flag("interactive")?,
226        limits,
227        phase_limits,
228        channel_range,
229        limit_channels: task.flag("chanrangeSetXrange")?,
230        resample: unsigned(task, "resample")?,
231    };
232    run_request(task, &request, &display)
233}

235fn run_request(
236    task: &TaskInvocation<'_>,
237    request: &PlotRequest<'_>,
238    display: &DisplayOptions<'_>,
239) -> Result<TaskSuccess, TaskFailure> {
240    let figfile = display.figfile;
241    let build_pdf = task.flag("buildpdf")?;
242    if figfile.is_empty()
243        && (build_pdf || (!display.interactive && request.derivative_sigma == 0.0))
244    {
245        return plain(
246            CasaValue::None,
247            true,
248            "Specify a figfile when building a PDF or running without display or derivative statistics.",
249        );
250    }
251    let completed_pages = std::env::var("CASARS_BANDPASS_COMPLETED_PAGES")
252        .ok()
253        .and_then(|text| text.parse::<usize>().ok());
254    let panel_limit = completed_pages
255        .map(|count| count.saturating_mul(display.layout[0] as usize * display.layout[1] as usize));
256    let plan = prepare_bandpass(request, panel_limit).map_err(TaskFailure::execution)?;
257    if let Some(ChannelRange::Indices([start, end])) = display.channel_range
258        && plan.panels.iter().any(|panel| {
259            !panel.frequency_channels.is_empty()
260                && (end > panel.frequency_channels.len()
261                    || (display.limit_channels && start.max(end) >= panel.frequency_channels.len()))
262        })
263    {
264        return plain(
265            CasaValue::None,
266            true,
267            "Channel range falls outside the calibration spectral window.",
268        );
269    }
270    let value = derivative_record(&plan).unwrap_or(CasaValue::None);
271    let empty_tuple = !plan.returns_none;
272    let pdf = if build_pdf {
273        Some(PdfOutput {
274            output: format!("{figfile}.pdf"),
275            sequential: display.sequential_names,
276            density: display.density,
277            convert: task.text("convert")?.unwrap_or_default(),
278            pdftk: task.text("pdftk")?.unwrap_or_default(),
279            ghostscript: task.text("gs")?.unwrap_or_default(),
280        })
281    } else {
282        None
283    };
284    let polynomial = plan.polynomial;
285    let mut diagnostics = if completed_pages.is_none() && task.flag("debug")? {
286        vec![
287            format!("caltable = {}", plan.caltable.display()),
288            format!(
289                "msName = {}.",
290                plan.measurement_set
291                    .as_deref()
292                    .map_or_else(String::new, |ms| ms.display().to_string())
293            ),
294            format!(
295                "Preparing {} bandpass panels for {} versus {}, overlay={:?}",
296                plan.panels.len(),
297                request.y_axis,
298                task.text("xaxis")?.unwrap_or_default(),
299                task.text("overlay")?.unwrap_or_default(),
300            ),
301        ]
302    } else {
303        Vec::new()
304    };
305    diagnostics.extend(plan.selection_notice.iter().cloned());
306    let reply = Reply {
307        pages: if completed_pages.is_none() {
308            pages(plan, display).map_err(TaskFailure::execution)?
309        } else {
310            Vec::new()
311        },
312        pdf,
313        empty_tuple,
314        sequential: display.sequential_names,
315        polynomial,
316        resample: display.resample,
317        diagnostics,
318    };
319    TaskSuccess::new(value, &reply)
320        .map(|success| success.with_note("Use casars.tasks.plotbandpass to render the figures."))
321}

323fn plain(value: CasaValue, empty_tuple: bool, message: &str) -> Result<TaskSuccess, TaskFailure> {
324    TaskSuccess::new(
325        value,
326        &Reply {
327            pages: Vec::new(),
328            pdf: None,
329            empty_tuple,
330            sequential: false,
331            polynomial: false,
332            resample: 1,
333            diagnostics: vec![message.to_string()],
334        },
335    )
336}

338/// A valid optional oscillator, or no result when CASA declines its spelling.
339fn local_oscillator(task: &TaskInvocation<'_>) -> Result<Option<Option<f64>>, TaskFailure> {
340    match task.value("lo1") {
341        Some(ParameterValue::String(value)) if value.is_empty() => Ok(Some(None)),
342        Some(ParameterValue::String(value)) => {
343            let decimal = value.split_once('.').is_some_and(|(whole, part)| {
344                !whole.is_empty()
345                    && !part.is_empty()
346                    && whole
347                        .bytes()
348                        .chain(part.bytes())
349                        .all(|byte| byte.is_ascii_digit())
350            });
351            let value = decimal.then(|| value.parse::<f64>().ok()).flatten();
352            Ok(value.map(|hz| Some(if hz > 1e6 { hz * 1e-9 } else { hz })))
353        }
354        Some(ParameterValue::Float(_)) | Some(ParameterValue::Integer(_)) => Err(
355            TaskFailure::execution("expected string or bytes-like object, got 'float'"),
356        ),
357        _ => Ok(Some(None)),
358    }
359}

361fn selector_with_names_in_vector(task: &TaskInvocation<'_>) -> Option<&'static str> {
362    ["antenna", "field", "spw"].into_iter().find(|parameter| {
363        matches!(task.value(parameter), Some(ParameterValue::Array(values))
364        if values.iter().any(|value| {
365            matches!(value, ParameterValue::String(text) if text.parse::<i32>().is_err())
366        }))
367    })
368}

370fn selector(task: &TaskInvocation<'_>, name: &str) -> Result<Vec<String>, TaskFailure> {
371    let convert = |value: &ParameterValue| match value {
372        ParameterValue::String(value) => Ok(value.clone()),
373        ParameterValue::Integer(value) => Ok(value.to_string()),
374        _ => Err(TaskFailure::validation("expected a name or an index").at(name)),
375    };
376    match task.value(name) {
377        Some(ParameterValue::Array(values)) => values.iter().map(convert).collect(),
378        Some(value) => Ok(vec![convert(value)?]),
379        None => Ok(Vec::new()),
380    }
381}
382
383fn selector_numeric_text(task: &TaskInvocation<'_>, name: &str) -> Result<String, TaskFailure> {
384    match task.value(name) {
385        Some(ParameterValue::Float(value)) => Ok(value.to_string()),
386        _ => Ok(selector(task, name)?.join("")),
387    }
388}

390fn overlay(text: &str) -> Result<Overlay, TaskFailure> {
391    let mut mode = Overlay::default();
392    for word in text.split(',').map(str::trim) {
393        match word {
394            "" => (),
395            "antenna" => mode.antenna = true,
396            "time" => mode.time = true,
397            "spw" => mode.window = true,
398            "baseband" => mode.baseband = true,
399            _ => return Err(TaskFailure::validation("unknown bandpass overlay").at("overlay")),
400        }
401    }
402    Ok(mode)
403}

405fn numeric(task: &TaskInvocation<'_>, name: &str) -> Result<f64, TaskFailure> {
406    match task.value(name) {
407        Some(ParameterValue::Bool(value)) => Ok(f64::from(u8::from(*value))),
408        _ => task.number(name),

417fn automatic_flag(task: &TaskInvocation<'_>, name: &str) -> Result<Option<bool>, TaskFailure> {
418    match task.value(name) {
419        Some(ParameterValue::Bool(value)) => Ok(Some(*value)),
420        Some(ParameterValue::String(value)) if value == "auto" => Ok(None),
421        Some(ParameterValue::String(_)) => Ok(Some(name == "showlines")),
422        _ => Err(TaskFailure::validation("expected a Boolean or auto").at(name)),
423    }
424}

426fn subplot(task: &TaskInvocation<'_>) -> Result<Option<[u32; 2]>, TaskFailure> {
427    let value = selector(task, "subplot")?.join("");
428    let mut size = value.trim().parse::<i32>().map_err(|_| {
429        TaskFailure::execution(format!("invalid literal for int() with base 10: '{value}'"))
430    })?;
431    if size > 100 {
432        if matches!(task.value("subplot"), Some(ParameterValue::String(_))) {
433            return Err(TaskFailure::execution(
434                "unsupported operand type(s) for //=: 'str' and 'int'",
435            ));
436        }
437        size /= 10;
438    }
439    Ok(match (size / 10, size % 10) {
440        (rows @ 1..=8, 1) => Some([rows as u32, 1]),
441        (rows @ 2..=4, 2) => Some([rows as u32, 2]),
442        _ => None,
443    })
444}

446fn plot_limits(task: &TaskInvocation<'_>) -> Result<Option<[f64; 4]>, TaskFailure> {
447    let values = task.numbers("plotrange")?;
448    if values.is_empty() {
449        return Ok(Some([0.0; 4]));
450    }
451    Ok(values.get(..4).and_then(|limits| limits.try_into().ok()))
452}
453
454fn phase_limits(task: &TaskInvocation<'_>) -> Result<Option<Option<[f64; 2]>>, TaskFailure> {
455    let values = selector(task, "phase")?;
456    Ok(match values.as_slice() {
457        [value] if value.is_empty() => Some(None),
458        [lower, upper] => lower
459            .parse()
460            .ok()
461            .zip(upper.parse().ok())
462            .map(|(lo, hi)| (lo != 0.0 || hi != 0.0).then_some([lo, hi])),
463        _ => None,
464    })
465}

467fn channel_range(
468    task: &TaskInvocation<'_>,
469    frequency_axis: bool,
470) -> Result<Option<ChannelRange>, TaskFailure> {
471    let values = selector(task, "chanrange")?;
472    let error = || {
473        TaskFailure::validation("chanrange must be a channel pair or a percentage").at("chanrange")
474    };
475    match values.as_slice() {
476        [value] if value.is_empty() => Ok(None),
477        [value] if frequency_axis => {
478            if let Some(percent) = value.strip_suffix('%') {
479                let percent = percent.parse::<f64>().map_err(|_| error())?;
480                return Ok((0.0 < percent && percent < 100.0)
481                    .then_some(ChannelRange::CentralPercent(percent)));
482            }
483            let (start, stop) = value.split_once('~').ok_or_else(error)?;
484            let start = start.trim().parse().map_err(|_| error())?;
485            let stop = stop.trim().parse().map_err(|_| error())?;
486            Ok(([start, stop] != [0, 0]).then_some(ChannelRange::Indices([start, stop])))
487        }
488        [start, stop] => {
489            let start = start.parse().map_err(|_| error())?;
490            let stop = stop.parse().map_err(|_| error())?;
491            Ok(([start, stop] != [0, 0]).then_some(ChannelRange::Indices([start, stop])))
492        }
493        _ => Err(error()),
494    }
