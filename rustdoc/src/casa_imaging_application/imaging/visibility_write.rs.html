2//! Visibilities written back in the final pass: the final model into
3//! `MODEL_DATA` and the continuum-subtracted data into `CORRECTED_DATA`,
4//! through the batch route's update session.
5
6use casa_imaging_model::{ContinuumFitRule, SequentialContinuumTransform};
7use casa_imaging_runtime::pass::SourceError;
8
9use super::observation_source::BatchRoute;
10pub(crate) use super::observation_source::BatchWriter as VisibilityWriter;
11
12/// Where the visibilities go: the update session of the batch route the
13/// run reads.
14#[derive(Clone, Copy)]
15pub(crate) struct VisibilityWriteTarget<'a>(pub(crate) &'a BatchRoute);
16
17impl<'a> VisibilityWriteTarget<'a> {
18    /// Heap bytes the write holds at most, admitted with the pass that
19    /// writes ([`BatchRoute::write_bytes`]).
20    pub(crate) fn retained_bytes(&self) -> Result<u64, SourceError> {
21        self.0.write_bytes()
22    }
23
24    /// Begin the write; `transform` is the continuum transform whose output
25    /// `CORRECTED_DATA` holds.
26    pub(crate) fn begin(
27        &self,
28        transform: Option<&'a SequentialContinuumTransform>,
29    ) -> Result<VisibilityWriter<'a>, SourceError> {
30        VisibilityWriter::begin(
31            self.0
32                .write_back()
33                .ok_or("the batch route was opened to read only")?,
34            transform,
35        )
36    }
37}
38
39/// Whether a channel of a row under the continuum `rule` reaches the line
40/// output, and so is written to either column: every channel without a
41/// continuum transform; fit-only channels keep their contents.
42pub(super) fn writes_channel(rule: Option<&ContinuumFitRule>, channel: u32) -> bool {
43    rule.is_none_or(|rule| {
44        rule.channel_use(channel)
45            .is_some_and(|role| role.contributes_to_output())
46    })
47}
