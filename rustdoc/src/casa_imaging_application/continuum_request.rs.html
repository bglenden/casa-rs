355/// Prepare, execute, and publish one native continuum request.
356pub fn execute_continuum(
357    request: ContinuumImagingRequest,
358) -> Result<ContinuumImagingResult, ApplicationDispatchError> {
359    let prepared = prepare(request).map_err(ApplicationDispatchError::Preparation)?;
360    let outcome = crate::execute(prepared)?;
361    let minor_cycles = outcome.output.minor_cycles.clone();
362    let minor = minor_cycles.last();
363    let minor_iterations = outcome.output.total_minor_iterations;
364    let actual_minor_iterations = outcome.output.total_actual_minor_iterations;
365    let product_names = outcome
366        .output
367        .planned_products
368        .members()
369        .iter()
370        .map(|member| member.name().to_string())
371        .collect();
372    Ok(ContinuumImagingResult {
373        minor_iterations,
374        actual_minor_iterations,
375        minor_stop_reason: minor.map(|value| match value.stop_reason {
376            crate::NativeMinorCycleStopReason::ThresholdReached => {
377                ContinuumStopReason::ThresholdReached
378            }
379            crate::NativeMinorCycleStopReason::IterationBound => {
380                ContinuumStopReason::IterationBound
381            }
382            crate::NativeMinorCycleStopReason::StalenessBound => {
383                ContinuumStopReason::StalenessBound
384            }
385            crate::NativeMinorCycleStopReason::MultiscaleDivergence => {
386                ContinuumStopReason::MultiscaleDivergence
387            }
388        }),
389        minor_cycles,
390        product_names,
391        outcome,
392    })
393}

427#[allow(clippy::too_many_arguments)]
428fn prepare_spectral_axis(
429    request: &ContinuumImagingRequest,
430    measurement_set: &MeasurementSet,
431    row_selection: &SelectedObservationRowSelection,
432    selection_io: MsSelectionIoBudget,
433    moving_phase_centre: Option<&SelectedObservationEphemeris>,
434    spectral_windows: &[SourceSpectralWindow],
435    source_frequency_reference: FrequencyRef,
436    anchor_time_mjd_seconds: f64,
437    time_bounds_mjd_seconds: [f64; 2],
438    field_id: usize,
439    phase: MDirection,
440    direction: DirectionCoordinateSpec,
441    frame_engine: &casa_ms::derived::engine::MsCalEngine,
442    moving_rest_frame: Option<&MeasFrame>,
443    source_rest_frequency_hz: Option<f64>,
444    continuum_selected_source_channels: Option<BTreeMap<usize, Vec<usize>>>,
445    continuum_spectral_envelope: Option<SelectedObservationSpectralEnvelope>,
446) -> Result<PreparedSpectralAxis, crate::ApplicationError> {
447    let source_frame = imaging_frequency_frame(source_frequency_reference)?;
448    match &request.spectral_mode {
449        SpectralImagingMode::Continuum => {
450            let selected_source_channels = continuum_selected_source_channels.ok_or_else(|| {
451                boxed("continuum imaging requires a prepared selected-channel map")
452            })?;
453            let spectral_envelope = continuum_spectral_envelope
454                .ok_or_else(|| boxed("continuum imaging requires a selected spectral envelope"))?;
455            let [lower_hz, upper_hz] = spectral_envelope.edges_hz();
456            let output_frequency_reference = FrequencyRef::LSRK;
457            let output_frame = imaging_frequency_frame(output_frequency_reference)?;
458            let reference_frequency_hz = spectral_envelope.midpoint_hz();
459            let increment_hz = upper_hz - lower_hz;
460            Ok(PreparedSpectralAxis {
461                selected_source_channels,
462                source_frame,
463                output_frequency_reference,
464                output_frame,
465                anchor: spectral_frame_anchor(
466                    source_frame,
467                    output_frame,
468                    anchor_time_mjd_seconds,
469                    direction,
470                    frame_engine,
471                )?,
472                wcs: SpectralWcs::Linear {
473                    channels: 1,
474                    reference_pixel: 0.0,
475                    reference_frequency_hz,
476                    increment_hz,
477                },
478                rest_frequency: RestFrequency::NotApplicable,
479                doppler: DopplerConvention::NotApplicable,
480                sampling: SpectralSamplingLaw::IDENTITY,
481                basis: ReconstructionBasis::Constant,
482                output_channels: 1,
483                reference_frequency_hz,
484                increment_hz,
485            })
486        }
487        SpectralImagingMode::JointContinuumLine => {
488            let [window] = spectral_windows else {
489                return Err(boxed(
490                    "joint continuum-line imaging requires exactly one selected spectral window",
491                ));
492            };
493            let selected = selected_channels(request, window.spw_id, &window.frequencies_hz)?;
494            let frequencies = selected
495                .iter()
496                .map(|channel| window.frequencies_hz[*channel])
497                .collect::<Vec<_>>();
498            let reference_frequency_hz = frequencies[0];
499            let increment_hz = if frequencies.len() > 1 {
500                frequencies[1] - frequencies[0]
501            } else {
502                window.channel_widths_hz[selected[0]]
503            };
504            if !increment_hz.is_finite()
505                || increment_hz == 0.0
506                || frequencies.iter().enumerate().any(|(index, frequency)| {
507                    let expected = reference_frequency_hz + index as f64 * increment_hz;
508                    (*frequency - expected).abs() > expected.abs().max(1.0) * 1.0e-12
509                })
510            {
511                return Err(boxed(
512                    "joint continuum-line output requires a finite linear selected source axis",
513                ));
514            }
515            Ok(PreparedSpectralAxis {
516                selected_source_channels: BTreeMap::from([(window.spw_id, selected)]),
517                source_frame,
518                output_frequency_reference: source_frequency_reference,
519                output_frame: source_frame,
520                anchor: SpectralFrameAnchor::NotApplicable,
521                wcs: SpectralWcs::Linear {
522                    channels: frequencies.len(),
523                    reference_pixel: 0.0,
524                    reference_frequency_hz,
525                    increment_hz,
526                },
527                rest_frequency: RestFrequency::NotApplicable,
528                doppler: DopplerConvention::NotApplicable,
529                sampling: SpectralSamplingLaw::IDENTITY,
530                basis: ReconstructionBasis::ChannelLocal {
531                    channels: frequencies.len(),
532                },
533                output_channels: frequencies.len(),
534                reference_frequency_hz,
535                increment_hz,
536            })
537        }
538        SpectralImagingMode::Cube {
539            axis,
540            output_channels,
541        } => {
542            let [window] = spectral_windows else {
543                return Err(boxed(
544                    "native cube imaging currently requires exactly one selected spectral window",
545                ));
546            };
547            let frequencies_hz = &window.frequencies_hz;
548            let channel_widths_hz = &window.channel_widths_hz;
549            let output_channels = output_channels.unwrap_or(frequencies_hz.len());
550            let (setup, support) = CubeSpectralSetup::for_casa_cube_axis(
551                source_frequency_reference,
552                frequencies_hz,
553                channel_widths_hz,
554                output_channels,
555                axis,
556                anchor_time_mjd_seconds,
557                field_id,
558                Some(phase),
559                time_bounds_mjd_seconds,
560                frame_engine,
561            )?;
562            let mut selected_source_channels = support.indices;
563            if let Some(explicit) = explicit_spw_channels(request, window.spw_id, frequencies_hz)? {
564                let explicit = explicit.into_iter().collect::<BTreeSet<_>>();
565                selected_source_channels.retain(|channel| explicit.contains(channel));
566            }
567            if selected_source_channels.is_empty() {
568                return Err(boxed(
569                    "cube axis and SPW selector have no common source channels",
570                ));
571            }
572            let reference_frequency_hz = setup.output_channel_frequencies_hz[0];
573            let increment_hz = if output_channels > 1 {
574                setup.output_channel_frequencies_hz[1] - reference_frequency_hz
575            } else {
576                setup.output_channel_widths_hz[0]
577            };
578            if !increment_hz.is_finite() || increment_hz == 0.0 {
579                return Err(boxed(
580                    "cube output frequency increment must be finite and non-zero",
581                ));
582            }
583            let output_frequency_reference = setup.output_freq_ref;
584            let output_frame = imaging_frequency_frame(output_frequency_reference)?;
585            let (rest_frequency, doppler) = match axis.rest_frequency_hz {
586                None => (
587                    RestFrequency::NotApplicable,
588                    DopplerConvention::NotApplicable,
589                ),
590                Some(hertz) => (
591                    RestFrequency::Line { hertz },
592                    match axis.veltype {
593                        DopplerRef::RADIO => DopplerConvention::Radio,
594                        DopplerRef::Z => DopplerConvention::Optical,
595                        DopplerRef::BETA => DopplerConvention::Relativistic,
596                        DopplerRef::RATIO | DopplerRef::GAMMA => {
597                            return Err(boxed("cube Doppler convention is not supported"));
598                        }
599                    },
600                ),
601            };
602            let sampling = match setup.interpolation {
603                CubeInterpolation::Nearest => SpectralSamplingLaw::NEAREST,
604                CubeInterpolation::Linear => SpectralSamplingLaw::LINEAR,
605                CubeInterpolation::Cubic => SpectralSamplingLaw::CUBIC,
606            };
607            Ok(PreparedSpectralAxis {
608                selected_source_channels: BTreeMap::from([(
609                    window.spw_id,
610                    selected_source_channels,
611                )]),
612                source_frame,
613                output_frequency_reference,
614                output_frame,
615                anchor: spectral_frame_anchor(
616                    source_frame,
617                    output_frame,
618                    anchor_time_mjd_seconds,
619                    direction,
620                    frame_engine,
621                )?,
622                wcs: SpectralWcs::Linear {
623                    channels: output_channels,
624                    reference_pixel: 0.0,
625                    reference_frequency_hz,
626                    increment_hz,
627                },
628                rest_frequency,
629                doppler,
630                sampling,
631                basis: ReconstructionBasis::ChannelLocal {
632                    channels: output_channels,
633                },
634                output_channels,
635                reference_frequency_hz,
636                increment_hz,
637            })
638        }
639        SpectralImagingMode::MtmfsViaCube {
640            axis,
641            output_channels,
642        } => prepare_mvc_spectral_axis(
643            request,
644            measurement_set,
645            row_selection,
646            selection_io,
647            moving_phase_centre,
648            spectral_windows,
649            source_frequency_reference,
650            source_frame,
651            anchor_time_mjd_seconds,
652            time_bounds_mjd_seconds,
653            field_id,
654            phase,
655            direction,
656            frame_engine,
657            axis,
658            *output_channels,
659        ),
660        SpectralImagingMode::CubeSource {
661            axis,
662            output_channels,
663        } => {
664            let [window] = spectral_windows else {
665                return Err(boxed(
666                    "source-frame cube imaging requires exactly one selected spectral window",
667                ));
668            };
669            let frame = moving_rest_frame.ok_or_else(|| {
670                boxed("source-frame cube imaging requires an ephemeris radial velocity")
671            })?;
672            let mut native_axis = axis.clone();
673            native_axis.specmode = casa_ms::spectral_selection::CubeSpecMode::Cubedata;
674            native_axis.outframe = source_frequency_reference;
675            let output_channels = output_channels.unwrap_or(window.frequencies_hz.len());
676            let (setup, support) = CubeSpectralSetup::for_casa_cube_axis(
677                source_frequency_reference,
678                &window.frequencies_hz,
679                &window.channel_widths_hz,
680                output_channels,
681                &native_axis,
682                anchor_time_mjd_seconds,
683                field_id,
684                Some(phase),
685                time_bounds_mjd_seconds,
686                frame_engine,
687            )?;
688            let factor = casa_ms::convert_frequency_to_frame_with_frame(
689                source_frequency_reference,
690                FrequencyRef::REST,
691                1.0,
692                Some(frame),
693            )?;
694            let reference_frequency_hz = setup.output_channel_frequencies_hz[0] * factor;
695            let increment_hz = if output_channels > 1 {
696                (setup.output_channel_frequencies_hz[1] - setup.output_channel_frequencies_hz[0])
697                    * factor
698            } else {
699                setup.output_channel_widths_hz[0] * factor
700            };
701            let rest_frequency_hz = axis
702                .rest_frequency_hz
703                .or(source_rest_frequency_hz)
704                .ok_or_else(|| {
705                    boxed("source-frame cube imaging requires REST_FREQUENCY metadata")
706                })?;
707            Ok(PreparedSpectralAxis {
708                selected_source_channels: BTreeMap::from([(window.spw_id, support.indices)]),
709                source_frame,
710                output_frequency_reference: FrequencyRef::REST,
711                output_frame: FrequencyFrame::Rest,
712                anchor: spectral_frame_anchor(
713                    source_frame,
714                    FrequencyFrame::Rest,
715                    anchor_time_mjd_seconds,
716                    direction,
717                    frame_engine,
718                )?,
719                wcs: SpectralWcs::Linear {
720                    channels: output_channels,
721                    reference_pixel: 0.0,
722                    reference_frequency_hz,
723                    increment_hz,
724                },
725                rest_frequency: RestFrequency::Line {
726                    hertz: rest_frequency_hz,
727                },
728                doppler: match axis.veltype {
729                    DopplerRef::RADIO => DopplerConvention::Radio,
730                    DopplerRef::Z => DopplerConvention::Optical,
731                    DopplerRef::BETA => DopplerConvention::Relativistic,
732                    DopplerRef::RATIO | DopplerRef::GAMMA => {
733                        return Err(boxed(
734                            "source-frame cube Doppler convention is not supported",
735                        ));
736                    }
737                },
738                sampling: match axis.interpolation {
739                    CubeInterpolation::Nearest => SpectralSamplingLaw::NEAREST,
740                    CubeInterpolation::Linear => SpectralSamplingLaw::LINEAR,
741                    CubeInterpolation::Cubic => SpectralSamplingLaw::CUBIC,
742                },
743                basis: ReconstructionBasis::ChannelLocal {
744                    channels: output_channels,
745                },
746                output_channels,
747                reference_frequency_hz,
748                increment_hz,
749            })
750        }
751    }

754#[allow(clippy::too_many_arguments)]
755fn prepare_mvc_spectral_axis(
756    request: &ContinuumImagingRequest,
757    measurement_set: &MeasurementSet,
758    row_selection: &SelectedObservationRowSelection,
759    selection_io: MsSelectionIoBudget,
760    moving_phase_centre: Option<&SelectedObservationEphemeris>,
761    spectral_windows: &[SourceSpectralWindow],
762    source_frequency_reference: FrequencyRef,
763    source_frame: FrequencyFrame,
764    anchor_time_mjd_seconds: f64,
765    time_bounds_mjd_seconds: [f64; 2],
766    field_id: usize,
767    phase: MDirection,
768    direction: DirectionCoordinateSpec,
769    frame_engine: &casa_ms::derived::engine::MsCalEngine,
770    axis: &CubeAxisConfig,
771    requested_output_channels: Option<usize>,
772) -> Result<PreparedSpectralAxis, crate::ApplicationError> {
773    let mut selected_by_spw = BTreeMap::new();
774    let mut spectral_selection = Vec::with_capacity(spectral_windows.len());
775    let mut selected_channel_count = 0_usize;
776    let mut output_frequency_reference = None;
777    // CASA MVC derives the major-cycle cube from the complete selected band,
778    // then replaces task start/width with that global linear axis.
779    let mut range_axis = axis.clone();
780    range_axis.start = None;
781    range_axis.width = None;
782
783    for window in spectral_windows {
784        let selected = selected_channels(request, window.spw_id, &window.frequencies_hz)?;
785        selected_channel_count = selected_channel_count
786            .checked_add(selected.len())
787            .ok_or_else(|| boxed("selected MVC channel count overflows usize"))?;
788        let (probe, _) = CubeSpectralSetup::for_casa_cube_axis(
789            source_frequency_reference,
790            &window.frequencies_hz,
791            &window.channel_widths_hz,
792            window.frequencies_hz.len(),
793            &range_axis,
794            anchor_time_mjd_seconds,
795            field_id,
796            Some(phase.clone()),
797            time_bounds_mjd_seconds,
798            frame_engine,
799        )?;
800        if output_frequency_reference.is_some_and(|reference| reference != probe.output_freq_ref) {
801            return Err(boxed(
802                "selected MVC spectral windows resolve to different output frequency frames",
803            ));
804        }
805        output_frequency_reference.get_or_insert(probe.output_freq_ref);
806
807        spectral_selection.push(SpectralWindowSelection::new(
808            u32::try_from(window.spw_id).map_err(|_| boxed("SPW id exceeds u32"))?,
809            selected
810                .iter()
811                .copied()
812                .map(|channel| {
813                    u32::try_from(channel).map_err(|_| boxed("channel index exceeds u32"))
814                })
815                .collect::<Result<Vec<_>, _>>()?,
816        ));
817        selected_by_spw.insert(window.spw_id, selected);
818    }
819
820    let output_channels = requested_output_channels.unwrap_or(selected_channel_count);
821    if output_channels == 0 {
822        return Err(boxed(
823            "MVC requires a positive output channel count and finite selected frequency range",
824        ));
825    }
826    let output_frequency_reference =
827        output_frequency_reference.expect("nonempty selected spectral windows");
828    let range = measurement_set.selected_observation_spectral_range(
829        row_selection,
830        &spectral_selection,
831        source_frequency_reference,
832        output_frequency_reference,
833        field_id,
834        anchor_time_mjd_seconds,
835        phase.clone(),
836        moving_phase_centre,
837        frame_engine,
838        selection_io,
839    )?;
840    let [global_low_hz, global_high_hz] = range.selected_edges_hz();
841    let [reference_low_hz, _] = range.reference_edges_hz();
842    let expansion_reference_frequency_hz = (global_low_hz + global_high_hz) / 2.0;
843    let increment_hz = (global_high_hz - global_low_hz) / output_channels as f64;
844    let first_channel_centre_hz = global_low_hz.max(reference_low_hz) + increment_hz / 2.0;
845    let public_reference_frequency_hz =
846        first_channel_centre_hz + (output_channels as f64 - 1.0) * increment_hz / 2.0;
847    let expansion_reference_pixel =
848        (expansion_reference_frequency_hz - first_channel_centre_hz) / increment_hz;
849    let mut global_axis = axis.clone();
850    global_axis.start = Some(casa_ms::CubeAxisValue::FrequencyHz {
851        hz: first_channel_centre_hz,
852        frame: Some(output_frequency_reference),
853    });
854    global_axis.width = Some(casa_ms::CubeAxisValue::FrequencyHz {
855        hz: increment_hz,
856        frame: Some(output_frequency_reference),
857    });
858
859    let mut setup = None;
860    let mut selected_source_channels = BTreeMap::new();
861    for window in spectral_windows {
862        let (window_setup, support) = CubeSpectralSetup::for_casa_cube_axis(
863            source_frequency_reference,
864            &window.frequencies_hz,
865            &window.channel_widths_hz,
866            output_channels,
867            &global_axis,
868            anchor_time_mjd_seconds,
869            field_id,
870            Some(phase.clone()),
871            time_bounds_mjd_seconds,
872            frame_engine,
873        )?;
874        if setup.as_ref().is_some_and(|shared: &CubeSpectralSetup| {
875            shared.output_channel_frequencies_hz != window_setup.output_channel_frequencies_hz
876                || shared.output_channel_widths_hz != window_setup.output_channel_widths_hz
877        }) {
878            return Err(boxed(
879                "selected MVC spectral windows did not resolve to one shared output axis",
880            ));
881        }
882        setup.get_or_insert(window_setup);
883        let selected = selected_by_spw
884            .get(&window.spw_id)
885            .expect("selected MVC SPW");
886        let support = support.indices.into_iter().collect::<BTreeSet<_>>();
887        let selected = selected
888            .iter()
889            .copied()
890            .filter(|channel| support.contains(channel))
891            .collect::<Vec<_>>();
892        if selected.is_empty() {
893            return Err(boxed(format!(
894                "MVC output axis has no supporting source channels in SPW {}",
895                window.spw_id
896            )));
897        }
898        selected_source_channels.insert(window.spw_id, selected);
899    }
900
901    let output_frame = imaging_frequency_frame(output_frequency_reference)?;
902    let (rest_frequency, doppler) = match axis.rest_frequency_hz {
903        None => (
904            RestFrequency::NotApplicable,
905            DopplerConvention::NotApplicable,
906        ),
907        Some(hertz) => (
908            RestFrequency::Line { hertz },
909            match axis.veltype {
910                DopplerRef::RADIO => DopplerConvention::Radio,
911                DopplerRef::Z => DopplerConvention::Optical,
912                DopplerRef::BETA => DopplerConvention::Relativistic,
913                DopplerRef::RATIO | DopplerRef::GAMMA => {
914                    return Err(boxed("MVC Doppler convention is not supported"));
915                }
916            },
917        ),
918    };
919    // CASA's MVC helper forces the cube-major GridFT interpolation to nearest
920    // after deriving the shared frequency range; task interpolation is not
921    // carried into this internal cube (`imager_mtmfs_via_cube.py`).
922    let sampling = SpectralSamplingLaw::NEAREST;
923    Ok(PreparedSpectralAxis {
924        selected_source_channels,
925        source_frame,
926        output_frequency_reference,
927        output_frame,
928        anchor: spectral_frame_anchor(
929            source_frame,
930            output_frame,
931            anchor_time_mjd_seconds,
932            direction,
933            frame_engine,
934        )?,
935        wcs: SpectralWcs::Linear {
936            channels: output_channels,
937            reference_pixel: expansion_reference_pixel,
938            reference_frequency_hz: expansion_reference_frequency_hz,
939            increment_hz,
940        },
941        rest_frequency,
942        doppler,
943        sampling,
944        basis: ReconstructionBasis::ChannelLocal {
945            channels: output_channels,
946        },
947        output_channels,
948        reference_frequency_hz: public_reference_frequency_hz,
949        increment_hz: global_high_hz - global_low_hz,
950    })
951}

953fn spectral_frame_anchor(
954    source_frame: FrequencyFrame,
955    output_frame: FrequencyFrame,
956    anchor_time_mjd_seconds: f64,
957    direction: DirectionCoordinateSpec,
958    frame_engine: &casa_ms::derived::engine::MsCalEngine,
959) -> Result<SpectralFrameAnchor, crate::ApplicationError> {
960    if source_frame == output_frame {
961        return Ok(SpectralFrameAnchor::NotApplicable);
962    }
963    let [x_metres, y_metres, z_metres] = frame_engine.observatory_position().as_itrf();
964    Ok(SpectralFrameAnchor::Conversion {
965        epoch: Epoch::new(
966            anchor_time_mjd_seconds / 86_400.0,
967            imaging_time_scale(frame_engine.time_reference())?,
968        ),
969        direction: direction.reference_direction(),
970        observatory_position: ItrfPosition::new(x_metres, y_metres, z_metres),
971    })
972}

974fn prepare(
975    mut request: ContinuumImagingRequest,
976) -> Result<ApplicationRequest<CasaImageProductSink>, crate::ApplicationError> {
977    canonicalize_polarizations(&mut request.polarizations);
978    validate_request(&request)?;
979    let ms = MeasurementSet::open(&request.measurement_set)?;
980    let data_description = ms.data_description()?;
981    let spectral_window = ms.spectral_window()?;
982    let polarization = ms.polarization()?;
983    let ddids = selected_data_descriptions(&request, &data_description)?;
984    let row_selection = ms.selected_observation_row_selection(
985        &ddids,
986        request.field_ids.as_deref(),
987        request.uv_range.as_deref(),
988        request.intent.as_deref(),
989    )?;
990    let content_budget = SelectedObservationContentBudget::new(64 << 20, 2, 4);
991    let candidate_bindings = ddids
992        .iter()
993        .copied()
994        .map(|ddid| {
995            usize::try_from(ddid)
996                .map_err(|_| boxed("selected DATA_DESC_ID is negative"))
997                .and_then(|ddid| data_description_binding(&data_description, ddid))
998        })
999        .collect::<Result<Vec<_>, _>>()?;
1000    let candidate_spw_ids = candidate_bindings
1001        .iter()
1002        .map(|(spw_id, _)| *spw_id)
1003        .collect::<BTreeSet<_>>();
1004    let mut spectral_windows = Vec::with_capacity(candidate_spw_ids.len());
1005    for spw_id in candidate_spw_ids {
1006        let frequency_reference =
1007            FrequencyRef::from_casacore_code(spectral_window.meas_freq_ref(spw_id)?)
1008                .ok_or_else(|| boxed("selected SPW has an unsupported frequency frame"))?;
1009        spectral_windows.push(SourceSpectralWindow {
1010            spw_id,
1011            frequency_reference,
1012            frequencies_hz: spectral_window.chan_freq(spw_id)?,
1013            channel_widths_hz: spectral_window.chan_width(spw_id)?,
1014        });
1015    }
1016    let mut continuum_selected_source_channels =
1017        if matches!(request.spectral_mode, SpectralImagingMode::Continuum) {
1018            Some(
1019                spectral_windows
1020                    .iter()
1021                    .map(|window| {
1022                        Ok((
1023                            window.spw_id,
1024                            selected_channels(&request, window.spw_id, &window.frequencies_hz)?,
1025                        ))
1026                    })
1027                    .collect::<Result<BTreeMap<_, _>, crate::ApplicationError>>()?,
1028            )
1029        } else {
1030            None
1031        };
1032    let frame_engine = casa_ms::derived::engine::MsCalEngine::new(&ms)?;
1033    let mut continuum_spectral_reducer = continuum_selected_source_channels
1034        .as_ref()
1035        .map(|selected| {
1036            ms.selected_observation_spectral_envelope_reducer(
1037                &row_selection,
1038                spectral_windows.iter().map(|window| {
1039                    SelectedObservationSpectralWindow::borrow_selected(
1040                        u32::try_from(window.spw_id).expect("nonnegative i32 SPW fits u32"),
1041                        window.frequency_reference,
1042                        &window.frequencies_hz,
1043                        &window.channel_widths_hz,
1044                        selected
1045                            .get(&window.spw_id)
1046                            .expect("selected every candidate spectral window"),
1047                    )
1048                }),
1049                FrequencyRef::LSRK,
1050                &frame_engine,
1051                content_budget.available_bytes(),
1052            )
1053        })
1054        .transpose()?;
1055    let spectral_reducer_bytes = continuum_spectral_reducer.as_ref().map_or(
1056        0,
1057        SelectedObservationSpectralEnvelopeReducer::retained_bytes,
1058    );
1059    let row_available_bytes = content_budget
1060        .available_bytes()
1061        .checked_sub(spectral_reducer_bytes)
1062        .ok_or_else(|| boxed("selected spectral envelope exhausts the row traversal budget"))?;
1063    let mut selected_rows = SelectedRowsBuilder::with_data_description_capacity(
1064        u64::try_from(ms.row_count()).map_err(|_| boxed("MS row count exceeds u64"))?,
1065        ddids.len(),
1066    );
1067    let mut selected_rows_error = None;
1068    let mut selected_ddids = BTreeSet::new();
1069    let mut selected_fields = BTreeSet::new();
1070    let mut first_selected_time_mjd_seconds = None;
1071    let mut selected_time_bounds_mjd_seconds = [f64::INFINITY, f64::NEG_INFINITY];
1072    let main_table = ms.main_table();
1073    let mut weight_spectrum_complete = main_table.column_accessor("WEIGHT_SPECTRUM").is_ok();
1074    let mut weight_spectrum_error = None;
1075    let mut spectral_envelope_error = None;
1076    ms.visit_selected_observation_rows(
1077        &row_selection,
1078        MsSelectionIoBudget {
1079            available_bytes: row_available_bytes,
1080            maximum_live_blocks: content_budget.maximum_live_blocks(),
1081            requested_bytes_per_row: SelectedObservationRow::STORAGE_BYTES_PER_ROW,
1082            storage_alignment_rows: None,
1083        },
1084        |row| {
1085            if spectral_envelope_error.is_none()
1086                && let Some(reducer) = continuum_spectral_reducer.as_mut()
1087            {
1088                spectral_envelope_error = reducer.observe(row).err();
1089            }
1090            if weight_spectrum_complete {
1091                match main_table
1092                    .column_accessor("WEIGHT_SPECTRUM")
1093                    .and_then(|column| column.array_cell_is_defined_uncached(row.physical_row()))
1094                {
1095                    Ok(defined) => weight_spectrum_complete = defined,
1096                    Err(error) => weight_spectrum_error = Some(error),
1097                }
1098            }
1099            selected_ddids.insert(row.data_description_id());
1100            selected_fields.insert(row.field_id());
1101            first_selected_time_mjd_seconds.get_or_insert(row.time_mjd_seconds());
1102            selected_time_bounds_mjd_seconds[0] =
1103                selected_time_bounds_mjd_seconds[0].min(row.time_mjd_seconds());
1104            selected_time_bounds_mjd_seconds[1] =
1105                selected_time_bounds_mjd_seconds[1].max(row.time_mjd_seconds());
1106            if selected_rows_error.is_none() {
1107                selected_rows_error = selected_rows
1108                    .push(SelectedMainRow::new(
1109                        u64::try_from(row.physical_row()).expect("row bounded by MS row count"),
1110                        u32::try_from(row.data_description_id())
1111                            .expect("validated nonnegative DDID"),
1112                    ))
1113                    .err();
1114            }
1115        },
1116    )?;
1117    if let Some(error) = selected_rows_error {
1118        return Err(Box::new(error));
1119    }
1120    if let Some(error) = weight_spectrum_error {
1121        return Err(Box::new(error));
1122    }
1123    if let Some(error) = spectral_envelope_error {
1124        return Err(Box::new(error));
1125    }
1126    let continuum_spectral_envelope = continuum_spectral_reducer
1127        .map(SelectedObservationSpectralEnvelopeReducer::finish)
1128        .transpose()?;
1129    let rows = selected_rows.finish();
1130    if rows.selected_row_count() == 0 {
1131        return Err(boxed("selection resolved to no rows"));
1132    }
1133    let selected_field = request
1134        .phase_center_field
1135        .unwrap_or_else(|| *selected_fields.first().expect("nonempty selection"));
1136    if !selected_fields.contains(&selected_field) {
1137        return Err(boxed(format!(
1138            "phase-center FIELD_ID {selected_field} is not part of selected fields {selected_fields:?}"
1139        )));
1140    }
1141    let field_id =
1142        usize::try_from(selected_field).map_err(|_| boxed("selected FIELD_ID is negative"))?;
1143    let bindings = selected_ddids
1144        .into_iter()
1145        .map(|ddid| {
1146            usize::try_from(ddid)
1147                .map_err(|_| boxed("selected DATA_DESC_ID is negative"))
1148                .and_then(|ddid| data_description_binding(&data_description, ddid))
1149        })
1150        .collect::<Result<Vec<_>, _>>()?;
1151    let spw_ids = bindings
1152        .iter()
1153        .map(|(spw_id, _)| *spw_id)
1154        .collect::<BTreeSet<_>>();
1155    spectral_windows.retain(|window| spw_ids.contains(&window.spw_id));
1156    if let Some(selected) = continuum_selected_source_channels.as_mut() {
1157        selected.retain(|spw_id, _| spw_ids.contains(spw_id));
1158    }
1159    let mut source_frequency_reference = None;
1160    for window in &spectral_windows {
1161        let reference = window.frequency_reference;
1162        if source_frequency_reference.is_some_and(|selected| selected != reference) {
1163            return Err(boxed(
1164                "selected spectral windows use different source frequency frames",
1165            ));
1166        }
1167        source_frequency_reference.get_or_insert(reference);
1168    }
1169    let phase = casa_ms::derived::engine::resolve_field_phase_direction_j2000(&ms, field_id)?;
1170    let default_main_direction = SkyDirection::new(
1171        DirectionFrame::J2000,
1172        phase.as_angles().0,
1173        phase.as_angles().1,
1174    );
1175    let anchor_time_mjd_seconds =
1176        first_selected_time_mjd_seconds.expect("nonempty selected row traversal");
1177    let measures_provider = casa_ms::open_measures_runtime()?;
1178    let measures_identity =
1179        SelectedObservationMeasures::new(Arc::clone(&measures_provider))?.identity();
1180    let (phase_centre_law, ephemeris, main_direction) = match request.phase_center.as_deref() {
1181        None => (
1182            PhaseCentreLaw::Fixed(default_main_direction),
1183            None,
1184            default_main_direction,
1185        ),
1186        Some(text) if text.split_ascii_whitespace().next() == Some("J2000") => {
1187            let direction = parse_phase_center_direction(text)?;
1188            (PhaseCentreLaw::Fixed(direction), None, direction)
1189        }
1190        Some("TRACKFIELD") => {
1191            let ephemeris = SelectedObservationEphemeris::tracked_fields(
1192                &ms,
1193                selected_fields
1194                    .iter()
1195                    .map(|field| usize::try_from(*field).expect("validated FIELD_ID")),
1196                content_budget.reference_data_budget(),
1197            )?;
1198            let direction = frame_engine.ephemeris_direction_j2000(
1199                anchor_time_mjd_seconds,
1200                field_id,
1201                "TRACKFIELD",
1202                &ephemeris,
1203            )?;
1204            let (longitude, latitude) = direction.as_angles();
1205            (
1206                PhaseCentreLaw::Ephemeris("TRACKFIELD".to_string()),
1207                Some(ephemeris),
1208                SkyDirection::new(DirectionFrame::J2000, longitude, latitude),
1209            )
1210        }

1617fn source_rest_frequency(
1618    measurement_set: &MeasurementSet,
1619    field_id: usize,
1620    spectral_windows: &[SourceSpectralWindow],
1621) -> Result<Option<f64>, crate::ApplicationError> {
1622    let source_id = measurement_set.field()?.source_id(field_id)?;
1623    if source_id < 0 {
1624        return Ok(None);
1625    }
1626    let source = measurement_set.source()?;
1627    let selected = spectral_windows
1628        .iter()
1629        .map(|window| i32::try_from(window.spw_id))
1630        .collect::<Result<BTreeSet<_>, _>>()?;
1631    let mut resolved = None;
1632    for row in 0..source.row_count() {
1633        if source.i32(row, "SOURCE_ID")? != source_id {
1634            continue;
1635        }
1636        let spectral_window_id = source.i32(row, "SPECTRAL_WINDOW_ID")?;
1637        if spectral_window_id >= 0 && !selected.contains(&spectral_window_id) {
1638            continue;
1639        }
1640        let Some(ArrayValue::Float64(values)) = source.optional_array(row, "REST_FREQUENCY")?
1641        else {
1642            continue;
1643        };
1644        let Some(value) = values.first().copied().filter(|value| *value > 0.0) else {
1645            continue;
1646        };
1647        if resolved.is_some_and(|prior: f64| prior.to_bits() != value.to_bits()) {
1648            return Err(boxed(
1649                "selected SOURCE rows disagree on REST_FREQUENCY metadata",
1650            ));
1651        }
1652        resolved = Some(value);
1653    }
1654    Ok(resolved)
1655}

1657fn attached_field_ephemerides(
1658    measurement_set: &MeasurementSet,
1659    selected_fields: &BTreeSet<i32>,
1660    budget: casa_ms::SelectedObservationReferenceDataBudget,
1661) -> Result<Option<SelectedObservationEphemeris>, crate::ApplicationError> {
1662    let field = measurement_set.field()?;
1663    let mut attached = Vec::new();
1664    for field_id in selected_fields {
1665        let field_id = usize::try_from(*field_id)?;
1666        if field
1667            .ephemeris_id(field_id)?
1668            .is_some_and(|value| value >= 0)
1669        {
1670            attached.push(field_id);
1671        }
1672    }
1673    if attached.is_empty() {
1674        return Ok(None);
1675    }
1676    if attached.len() != selected_fields.len() {
1677        return Err(boxed(
1678            "moving-source selection mixes FIELD rows with and without attached ephemerides",
1679        ));
1680    }
1681    SelectedObservationEphemeris::tracked_fields(measurement_set, attached, budget)
1682        .map(Some)
1683        .map_err(Into::into)
1684}

1700fn scientific_instrument_model(
1701    request: &ContinuumImagingRequest,
1702    ms: &MeasurementSet,
1703) -> Result<Option<(InstrumentModel, LogicalIdentity)>, crate::ApplicationError> {
1704    let mosaic = request
1705        .task_requirements
1706        .contains(&TaskRequirement::MosaicGridder);
1707    if !mosaic
1708        && !matches!(
1709            request.spectral_mode,
1710            SpectralImagingMode::MtmfsViaCube { .. }
1711        )
1712    {
1713        return Ok(None);
1714    }
1715    let observation = ms.observation()?;
1716    let telescopes = (0..observation.row_count())
1717        .map(|row| {
1718            observation
1719                .string(row, "TELESCOPE_NAME")
1720                .map(|name| name.trim().to_string())
1721        })
1722        .collect::<Result<BTreeSet<_>, _>>()?;
1723    let telescope_names = telescopes.iter().map(String::as_str).collect::<Vec<_>>();
1724    let supported_telescope = if mosaic {
1725        !telescope_names.is_empty()
1726            && telescope_names
1727                .iter()
1728                .all(|name| matches!(*name, "ALMA" | "ACA"))
1729    } else {
1730        telescope_names == ["ALMA"]
1731    };
1732    if !supported_telescope {
1733        return Err(boxed(format!(
1734            "primary-beam response requires ALMA/ACA observation metadata; found {telescopes:?}"
1735        )));
1736    }
1737    let antenna = ms.antenna()?;
1738    if antenna.row_count() == 0 {
1739        return Err(boxed(
1740            "channel-major primary-beam response requires ANTENNA dish metadata",
1741        ));
1742    }
1743    let mut hasher = Sha256::new();
1744    let instrument_model = if mosaic {
1745        hasher.update(b"casa-rs-instrument-reference/casa-alma-aca-heterogeneous-response-v1");
1746        InstrumentModel::CasaAlmaAcaHeterogeneousInterferometricResponseV1
1747    } else {
1748        hasher.update(b"casa-rs-instrument-reference/casa-aca7m-direct-pb-v1");
1749        InstrumentModel::CasaAca7mInterferometricDirectPbV1
1750    };
1751    hasher.update((telescopes.len() as u64).to_le_bytes());
1752    for telescope in &telescopes {
1753        hasher.update((telescope.len() as u64).to_le_bytes());
1754        hasher.update(telescope.as_bytes());
1755    }
1756    hasher.update((antenna.row_count() as u64).to_le_bytes());
1757    for row in 0..antenna.row_count() {
1758        let diameter = antenna.dish_diameter(row)?;
1759        let supported_diameter = instrument_model_supports_diameter(mosaic, diameter);
1760        if !supported_diameter {
1761            let expected = if mosaic {
1762                "CASA 12 m or 7 m antenna classes"
1763            } else {
1764                "one homogeneous ACA 7 m antenna class"
1765            };
1766            return Err(boxed(format!(
1767                "ALMA/ACA response requires {expected}; row {row} has diameter {diameter} m"
1768            )));
1769        }
1770        hasher.update(diameter.to_bits().to_le_bytes());
1771    }
1772    Ok(Some((
1773        instrument_model,
1774        LogicalIdentity::from_sha256(hasher.finalize().into()),
1775    )))
1776}
1777
1778fn instrument_model_supports_diameter(mosaic: bool, diameter_m: f64) -> bool {
1779    diameter_m.is_finite()
1780        && if mosaic {
1781            (diameter_m - 12.0).abs() < 0.5 || (diameter_m - 7.0).abs() < 1.0
1782        } else {
1783            (6.0..8.0).contains(&diameter_m)
1784        }
1785}

1789) -> Result<casa_imaging_products::AnalyticPrimaryBeamModel, crate::ApplicationError> {
1790    match telescopes

1792        .map(String::as_str)
1793        .collect::<Vec<_>>()
1794        .as_slice()

1797        [] => Err(boxed(
1798            "standard primary-beam publication requires OBSERVATION telescope metadata",

1800        names => Err(boxed(format!(
1801            "standard primary-beam publication has no installed analytic model for telescope set {names:?}"

1806fn validate_request(request: &ContinuumImagingRequest) -> Result<(), crate::ApplicationError> {
1807    if request.phase_center_field.is_some() && request.phase_center.is_some() {
1808        return Err(boxed(
1809            "phase_center and phase_center_field are mutually exclusive",
1810        ));
1811    }
1812    if request.image_size == 0
1813        || !request.cell_arcsec.is_finite()
1814        || request.cell_arcsec <= 0.0
1815        || !request.gain.is_finite()
1816        || !request.threshold_jy.is_finite()
1817        || !request.psf_cutoff.is_finite()
1818        || request.psf_cutoff <= 0.0
1819        || !request.primary_beam_cutoff.is_finite()
1820        || !(0.0..1.0).contains(&request.primary_beam_cutoff)
1821        || (request.algorithm != ContinuumAlgorithm::Dirty
1822            && (request.cycle_iterations == 0 || request.maximum_major_cycles == Some(0)))
1823        || request
1824            .noise_sigma
1825            .is_some_and(|sigma| !sigma.is_finite() || sigma < 0.0)
1826    {
1827        return Err(boxed(
1828            "native continuum geometry and controls must be finite and positive",
1829        ));
1830    }
1831    if request.save_model_column && request.algorithm == ContinuumAlgorithm::Dirty {
1832        return Err(boxed(
1833            "MODEL_DATA persistence requires a solved final model, not a dirty-only request",
1834        ));
1835    }
1836    if request.save_continuum_residual && request.continuum_subtraction.is_none() {
1837        return Err(boxed(
1838            "CORRECTED_DATA residual persistence requires continuum subtraction",
1839        ));
1840    }
1841    if request.continuum_subtraction.is_some()
1842        && !matches!(
1843            request.spectral_mode,
1844            SpectralImagingMode::Cube { .. } | SpectralImagingMode::CubeSource { .. }
1845        )
1846    {
1847        return Err(boxed(
1848            "visibility-domain continuum subtraction requires channel-local cube imaging",
1849        ));
1850    }
1851    if request.outlier_file.is_some()
1852        && (!matches!(request.spectral_mode, SpectralImagingMode::Continuum)
1853            || !matches!(
1854                request.algorithm,
1855                ContinuumAlgorithm::Dirty | ContinuumAlgorithm::Hogbom
1856            )
1857            || request.weighting != ContinuumWeighting::Natural)
1858    {
1859        return Err(boxed(
1860            "the installed multi-domain slice requires MFS, natural weighting, and dirty or Hogbom reconstruction",
1861        ));
1862    }
1863    Ok(())
1864}

1866fn prepare_continuum_transform(
1867    controls: Option<&VisibilityContinuumSubtraction>,
1868    field_id: i32,
1869    spw_id: usize,
1870    frequencies_hz: &[f64],
1871    output_channels: &[usize],
1872) -> Result<(Option<SequentialContinuumTransform>, Vec<usize>), crate::ApplicationError> {
1873    let Some(controls) = controls else {
1874        return Ok((None, output_channels.to_vec()));
1875    };
1876    let order = u8::try_from(controls.fit_order)
1877        .map_err(|_| boxed("continuum fit order exceeds the supported contract range"))?;
1878    let selectors = parse_spw_selector(&controls.fit_spw)?;
1879    if selectors.len() != 1 || usize::try_from(selectors[0].spw_id).ok() != Some(spw_id) {
1880        return Err(boxed(
1881            "continuum fit selector must name exactly the selected spectral window",
1882        ));
1883    }
1884    let fit_channels = match selectors.into_iter().next().expect("one selector").channels {
1885        Some(selector) => resolve_channel_selector_selection(frequencies_hz, &selector)?.indices,
1886        None => (0..frequencies_hz.len()).collect(),
1887    };
1888    let fit = fit_channels.into_iter().collect::<BTreeSet<_>>();
1889    let output = output_channels.iter().copied().collect::<BTreeSet<_>>();
1890    let selected = fit.union(&output).copied().collect::<Vec<_>>();
1891    let roles = selected
1892        .iter()
1893        .copied()
1894        .map(|channel| {
1895            let use_role = match (fit.contains(&channel), output.contains(&channel)) {
1896                (true, false) => ContinuumChannelUse::FitOnly,
1897                (false, true) => ContinuumChannelUse::ApplyOnly,
1898                (true, true) => ContinuumChannelUse::FitAndApply,
1899                (false, false) => unreachable!("set union member"),
1900            };
1901            Ok(ContinuumChannelRole::new(
1902                u32::try_from(channel).map_err(|_| boxed("channel index exceeds u32"))?,
1903                use_role,
1904            ))
1905        })
1906        .collect::<Result<Vec<_>, crate::ApplicationError>>()?;
1907    let rule = ContinuumFitRule::new(
1908        field_id,
1909        u32::try_from(spw_id).map_err(|_| boxed("SPW id exceeds u32"))?,
1910        order,
1911        roles,
1912    )?;
1913    Ok((
1914        Some(SequentialContinuumTransform::new(vec![rule])?),
1915        selected,
1916    ))
1917}

2204fn parse_sexagesimal(text: &str, hours: bool) -> Option<f64> {
2205    let trimmed = text.trim();
2206    let sign = if trimmed.starts_with('-') { -1.0 } else { 1.0 };
2207    let body = trimmed.trim_start_matches(['+', '-']);
2208    let fields = if body.contains(':') {
2209        body.split(':').map(str::to_owned).collect::<Vec<_>>()
2210    } else if body.contains('h') || body.contains('d') || body.contains('m') || body.contains('s') {
2211        body.replace(['h', 'd', 'm', 's'], " ")
2212            .split_whitespace()
2213            .map(str::to_owned)
2214            .collect()
2215    } else if !hours && body.matches('.').count() >= 2 {
2216        let mut split = body.split('.');
2217        let major = split.next()?.to_owned();
2218        let minutes = split.next()?.to_owned();
2219        let seconds = split.collect::<Vec<_>>().join(".");
2220        vec![major, minutes, seconds]
2221    } else {
2222        return None;
2223    };
2224    let [major, minutes, seconds] = fields.as_slice() else {
2225        return None;
2226    };
2227    let major = major.parse::<f64>().ok()?;
2228    let minutes = minutes.parse::<f64>().ok()?;
2229    let seconds = seconds.parse::<f64>().ok()?;
2230    if !(major.is_finite()
2231        && minutes.is_finite()
2232        && seconds.is_finite()
2233        && (0.0..60.0).contains(&minutes)
2234        && (0.0..60.0).contains(&seconds))
2235    {
2236        return None;
2237    }
2238    Some(sign * (major.abs() + minutes / 60.0 + seconds / 3600.0))
2239}

2417fn specification(
2418    request: &ContinuumImagingRequest,
2419    spectral: &PreparedSpectralAxis,
2420    instrument_model: Option<InstrumentModel>,
2421) -> Result<ProblemSpecification, crate::ApplicationError> {
2422    let mosaic = request
2423        .task_requirements
2424        .contains(&TaskRequirement::MosaicGridder);
2425    let algorithm = reconstruction_algorithm(&request.algorithm);
2426    let basis = match (&request.spectral_mode, &request.algorithm) {
2427        (SpectralImagingMode::MtmfsViaCube { .. }, ContinuumAlgorithm::Mtmfs { terms, .. }) => {
2428            ReconstructionBasis::TaylorViaChannelMajor {
2429                terms: *terms,
2430                channels: spectral.output_channels,
2431            }
2432        }
2433        (_, ContinuumAlgorithm::Mtmfs { terms, .. }) => {
2434            ReconstructionBasis::Taylor { terms: *terms }
2435        }
2436        (
2437            _,
2438            ContinuumAlgorithm::JointContinuumLine {
2439                continuum_terms,
2440                line_channels,
2441                ..
2442            },
2443        ) => ReconstructionBasis::JointContinuumLine {
2444            continuum_terms: *continuum_terms,
2445            line_terms: line_channels.len(),
2446        },
2447        _ => spectral.basis,
2448    };
2449    let (weighting, density) = match request.weighting {
2450        ContinuumWeighting::Natural => {
2451            (WeightingScheme::Natural, WeightDensityScope::NotApplicable)
2452        }
2453        ContinuumWeighting::Uniform => (
2454            WeightingScheme::Uniform,
2455            WeightDensityScope::GlobalSelection,
2456        ),
2457        ContinuumWeighting::Briggs(robust) => (
2458            WeightingScheme::Briggs { robust },
2459            WeightDensityScope::GlobalSelection,
2460        ),
2461        ContinuumWeighting::BriggsBandwidthTaper(robust) => (
2462            WeightingScheme::BriggsBandwidthTaper { robust },
2463            WeightDensityScope::GlobalSelection,
2464        ),
2465    };
2466    let mut reconstruction = ReconstructionContract::new(
2467        basis,
2468        algorithm.clone(),
2469        if algorithm == ReconstructionAlgorithm::Dirty {
2470            ReconstructionControls::new(0, 1.0, 0.0)
2471        } else {

2473                ReconstructionControls::new(request.iterations, request.gain, request.threshold_jy)
2474                    .with_cycle_limits(request.cycle_iterations, request.maximum_major_cycles)
2475                    .with_hogbom_iteration_accounting(request.hogbom_iteration_accounting)
2476                    .with_cycle_threshold(
2477                        request.cycle_factor,
2478                        request.minimum_psf_fraction,
2479                        request.maximum_psf_fraction,
2480                    );
2481            request
2482                .noise_sigma
2483                .map_or(controls, |sigma| controls.with_noise_sigma(sigma))
2484        },
2485        PolarizationContract::new(request.polarizations.clone()),
2486    );
2487    if let ContinuumAlgorithm::JointContinuumLine {
2488        continuum_anchor_channels,
2489        line_channels,
2490        maximum_condition_number,
2491        ..
2492    } = &request.algorithm

2494        reconstruction = reconstruction.with_joint_continuum_line(
2495            casa_imaging_model::JointContinuumLineContract::new(
2496                continuum_anchor_channels.clone(),
2497                line_channels.clone(),
2498                *maximum_condition_number,
2499            ),
2500        );
2501    }
2502    let mut science = ScientificContract::new(
2503        SpectralContract::new(
2504            spectral.sampling,
2505            match (&request.algorithm, request.beam_policy) {
2506                (ContinuumAlgorithm::Mtmfs { .. }, _)
2507                | (ContinuumAlgorithm::JointContinuumLine { .. }, _) => {
2508                    SpectralCoupling::CommonRestoringBeam
2509                }
2510                (_, ContinuumBeamPolicy::PerPlane) => SpectralCoupling::Independent,
2511                (_, ContinuumBeamPolicy::Common) => SpectralCoupling::CommonRestoringBeam,
2512            },
2513        ),
2514        MeasurementEquationContract::new(
2515            if instrument_model.is_some() {
2516                InstrumentResponse::PrimaryBeam
2517            } else {
2518                InstrumentResponse::Scalar
2519            },
2520            DeclaredInnerProducts::new(
2521                ModelInnerProduct::HermitianEuclidean,
2522                VisibilityInnerProduct::HermitianEuclidean,
2523            ),
2524        ),
2525    );

2531        reconstruction,

2536                request.normalization,
2537                mosaic,

2543                (ContinuumAlgorithm::Mtmfs { .. }, _)
2544                | (ContinuumAlgorithm::JointContinuumLine { .. }, _) => RestoringBeamPolicy::Common,
2545                (_, ContinuumBeamPolicy::PerPlane) => RestoringBeamPolicy::PerPlane,
2546                (_, ContinuumBeamPolicy::Common) => RestoringBeamPolicy::Common,
2547            },
2548            ProductValidityPolicies::new(
2549                PrimaryBeamValidityPolicy::new(
2550                    request.primary_beam_cutoff,
2551                    ProductSupportComparison::StrictlyGreater,
2552                    ProductBlankingPolicy::ZeroAndFalseMask,
2553                )?,
2554                TaylorValidityPolicy::new(
2555                    TaylorSupportReference::PrincipalResidualTaylor0PositiveMaximum,
2556                    0.1,
2557                    ProductSupportComparison::StrictlyGreater,
2558                    ProductBlankingPolicy::ZeroAndFalseMask,
2559                )?,
2560            ),
2561        ),
2562        ObservationTransactionRequirements::new(if request.save_model_column {
2563            ModelColumnWrite::SelectedRows

2881    let statistics = unsafe { statistics.assume_init() };

2883    Ok((
2884        block_count_to_u64(statistics.f_blocks).saturating_mul(block_size),
2885        block_count_to_u64(statistics.f_bavail).saturating_mul(block_size),

2905fn request_digest(request: &ContinuumImagingRequest, domain: &[u8]) -> [u8; 32] {
2906    let mut hasher = Sha256::new();
2907    hasher.update(domain);
2908    hasher.update(request.measurement_set.as_os_str().as_encoded_bytes());
2909    hasher.update(request.image_name.as_os_str().as_encoded_bytes());
2910    hasher.update(request.image_size.to_le_bytes());
2911    hasher.update(request.cell_arcsec.to_bits().to_le_bytes());
2912    if let Some(field) = request.phase_center_field {
2913        hasher.update([1]);
2914        hasher.update(field.to_le_bytes());
2915    } else {
2916        hasher.update([0]);
2917    }
2918    if let Some(phase_center) = request.phase_center.as_deref() {
2919        hasher.update([1]);
2920        hasher.update(phase_center.as_bytes());
2921    } else {
2922        hasher.update([0]);
2923    }
2924    if let Some(outlier_file) = request.outlier_file.as_deref() {
2925        hasher.update([1]);
2926        hasher.update(outlier_file.as_os_str().as_encoded_bytes());
2927        if let Ok(contents) = std::fs::read(outlier_file) {
2928            hasher.update(Sha256::digest(contents));
2929        }
2930    } else {
2931        hasher.update([0]);
2932    }
2933    hasher.finalize().into()
2934}
