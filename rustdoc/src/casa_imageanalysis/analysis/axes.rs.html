2//! Where the image tasks find an image's axes, and moving the reference
3//! pixels of a coordinate-system record with a subimage.
4//!
5//! casa-rs reads a CASA image whose coordinates own transposed axes with
6//! the coordinates in axis order, and drops a linear coordinate's removed
7//! axes (#1001), so its coordinates can differ from the record's: a
8//! record's coordinates are found where the record places them.
9

19pub(super) struct StridedSection<'a> {
20    pub(super) start: &'a [usize],
21    /// Counts of selected pixels on each axis.
22    pub(super) shape: &'a [usize],
23    pub(super) stride: &'a [usize],
24}
25
26impl StridedSection<'_> {
27    pub(super) fn cursors(&self, dense_cursor: Vec<usize>) -> TraversalCursorIter {
28        // Reading a strided tile may fetch its dense enclosing box. Keep that
29        // box bounded by the chosen source tile on every axis.
30        let packed = dense_cursor
31            .into_iter()
32            .zip(self.stride)
33            .map(|(size, &step)| size.div_ceil(step).max(1))
34            .collect::<Vec<_>>();
35        TraversalCursorIter::new(self.shape.to_vec(), packed, TraversalSpec::tiles())
36    }
37
38    pub(super) fn source_start(&self, position: &[usize]) -> Vec<usize> {
39        position
40            .iter()
41            .zip(self.start)
42            .zip(self.stride)
43            .map(|((&offset, &start), &step)| start + offset * step)
44            .collect()
45    }
46
47    pub(super) fn dense_shape(&self, shape: &[usize]) -> Vec<usize> {
48        shape
49            .iter()
50            .zip(self.stride)
51            .map(|(&count, &step)| (count - 1) * step + 1)
52            .collect()
53    }
54}

56/// The pixel axes of `image`'s direction plane, (longitude, latitude):
57/// its direction coordinate's wherever they are, or the first two axes of an
58/// image without one.
59///
60/// The direction axes can be swapped or separated by other coordinates.
61///
62/// - Regions take their plane here, as CASA does.
63/// - `feather` does too, where CASA 6.7.6's `feather` fails on such an
64///   image (`docs/CASA (C++) bugs.md`).
65/// - `impv` takes it only on axes 0 and 1 until its output has a PV image's
66///   coordinates (#1024).
67pub(super) fn direction_plane_axes<T: ImagePixel>(
68    image: &(impl ImageInterface<T> + ?Sized),
69) -> Result<[usize; 2], ImageError> {
70    direction_plane_axes_for_coordinates(image.coordinates())
71}
72
73pub(super) fn direction_plane_axes_for_coordinates(
74    coordinates: &CoordinateSystem,
75) -> Result<[usize; 2], ImageError> {
76    let Some(index) = coordinates.find_coordinate(CoordinateType::Direction) else {
77        return Ok([0, 1]);
78    };
79    let axes = coordinates.coordinate_pixel_axes(index).unwrap_or_default();
80    let [Some(longitude), Some(latitude)] = axes.as_slice() else {
81        return Err(ImageError::InvalidMetadata(
82            "selecting a direction plane requires both direction axes".into(),
83        ));
84    };
85    Ok([*longitude, *latitude])
86}
87
88pub(super) fn world_axes_for_plane(
89    coordinates: &CoordinateSystem,
90    plane: [usize; 2],
91) -> Result<[usize; 2], ImageError> {
92    let [Some(first), Some(second)] = plane.map(|axis| coordinates.pixel_axis_to_world_axis(axis))
93    else {
94        return Err(ImageError::InvalidMetadata(
95            "a world region requires both world axes of the image plane".into(),
96        ));
97    };
98    Ok([first, second])
99}

101/// `coordinates` for a subimage starting at pixel `start`.
102pub(super) fn subimage_coordinates(
103    coordinates: &CoordinateSystem,
104    start: &[usize],
105    shape: &[usize],
106) -> Result<CoordinateSystem, ImageError> {
107    let mut record = coordinates.to_record();
108    shift_record_reference_pixels(&mut record, start, shape)?;
109    coordinates
110        .restore_record_with_layout(&record)
111        .map_err(|error| {
112            ImageError::InvalidMetadata(format!(
113                "failed to build subimage coordinate system: {error}"
114            ))
115        })
116}

118/// `coordinates` for a subimage starting at pixel `start`, optionally replacing
119/// the Stokes coordinate with `stokes`.
120pub(super) fn subimage_coordinates_with_stokes(
121    coordinates: &CoordinateSystem,
122    start: &[usize],
123    shape: &[usize],
124    stokes: Option<&[StokesType]>,
125) -> Result<CoordinateSystem, ImageError> {
126    let mut coords = subimage_coordinates(coordinates, start, shape)?;
127    if let Some(stokes_list) = stokes {
128        replace_stokes_coordinate(&mut coords, stokes_list);
129    }
130    Ok(coords)
131}

133/// Stokes list owned by pixel `axis` of `coords`.
134pub(super) fn stokes_for_axis(
135    coords: &CoordinateSystem,
136    axis: usize,
137) -> Result<Vec<StokesType>, ImageError> {
138    if let Some((index, _)) = coords.pixel_axis_owner(axis)
139        && let CoordinateModel::Stokes(stokes) = coords.coordinate(index)
140    {
141        return Ok(stokes.stokes().to_vec());
142    }
143    Err(ImageError::InvalidMetadata(
144        "Stokes selection requires a Stokes coordinate".into(),
145    ))
146}

148/// Replaces any Stokes coordinate in `coords` with `stokes`.
149pub(super) fn replace_stokes_coordinate(coords: &mut CoordinateSystem, stokes: &[StokesType]) {
150    if let Some(idx) = coords.find_coordinate(CoordinateType::Stokes) {
151        coords.replace_coordinate(idx, StokesCoordinate::new(stokes.to_vec()));
152    }
153}

155/// `image`'s own `coords` record for a subimage starting at pixel `start`,
156/// optionally replacing its Stokes list.
157pub(super) fn legacy_subimage_coordinates_with_stokes<T: ImagePixel>(
158    image: &PagedImage<T>,
159    start: &[usize],
160    shape: &[usize],
161    stokes: Option<&[StokesType]>,
162) -> Option<RecordValue> {

165    };
166    subimage_coordinate_record(coords.clone(), start, shape, stokes)

169pub(super) fn subimage_coordinate_record(
170    mut coords: RecordValue,
171    start: &[usize],
172    shape: &[usize],
173    stokes: Option<&[StokesType]>,
174) -> Option<RecordValue> {
175    // The image's coordinates were read from this record, so its fields
176    // resolve; if not, the output keeps the coordinates casa-rs writes.
177    shift_record_reference_pixels(&mut coords, start, shape).ok()?;
178    if let Some(stokes_list) = stokes {
179        replace_stokes_in_record(&mut coords, stokes_list).ok()?;
180    }
181    Some(coords)
182}

184/// Updates the `stokes*` sub-record inside a stored `coords` record to hold `stokes`.
185pub(super) fn replace_stokes_in_record(
186    record: &mut RecordValue,
187    stokes: &[StokesType],
188) -> Result<(), ImageError> {
189    let placed = CoordinateSystem::record_coordinates(record).map_err(|error| {
190        ImageError::InvalidMetadata(format!("stokes record coordinates: {error}"))
191    })?;
192    for coordinate in placed {
193        if coordinate.field.starts_with("stokes")
194            && matches!(record.get(&coordinate.field), Some(Value::Record(_)))
195        {
196            replace_stokes_axis_in_record(record, &coordinate.field, stokes)?;
197        }
198    }
199    Ok(())
200}

202/// Replace a single stored Stokes axis and synchronize its reference fields.
203///
204/// # Errors
205///
206/// Returns an error if the named coordinate record is missing.
207pub fn replace_stokes_axis_in_record(
208    record: &mut RecordValue,
209    field: &str,
210    stokes: &[StokesType],
211) -> Result<(), ImageError> {
212    let Some(Value::Record(stokes_rec)) = record.get_mut(field) else {
213        return Err(ImageError::InvalidMetadata(format!(
214            "Missing Stokes coordinate {field}"
215        )));
216    };
217    let use_codes = matches!(
218        stokes_rec.get("stokes"),
219        Some(Value::Array(ArrayValue::Int32(_)))
220    );
221    let labels = if use_codes {
222        ArrayValue::from_i32_vec(stokes.iter().map(|s| s.code()).collect())
223    } else {
224        ArrayValue::from_string_vec(stokes.iter().map(|s| s.name().to_string()).collect())
225    };
226    stokes_rec.upsert("stokes", Value::Array(labels));
227    if let Some(&first) = stokes.first() {
228        if matches!(stokes_rec.get("crval"), Some(Value::Array(_))) {
229            stokes_rec.upsert(
230                "crval",
231                Value::Array(ArrayValue::from_f64_vec(vec![f64::from(first.code())])),
232            );
233        }
234        if matches!(stokes_rec.get("crpix"), Some(Value::Array(_))) {
235            stokes_rec.upsert("crpix", Value::Array(ArrayValue::from_f64_vec(vec![0.0])));
236        }
237        if let Some(suffix) = field.strip_prefix("stokes") {
238            let wr_key = format!("worldreplace{suffix}");
239            if matches!(record.get(&wr_key), Some(Value::Array(_))) {
240                record.upsert(
241                    wr_key,
242                    Value::Array(ArrayValue::from_f64_vec(vec![f64::from(first.code())])),
243                );
244            }
245        }
246    }
247    Ok(())
248}

250/// Moves each coordinate in `record` back by `start`, the subimage's first
251/// pixel, along the coordinate's own axes.
252///
253/// Each coordinate is found as the record places it
254/// ([`CoordinateSystem::record_coordinates`]): a transposed casacore image
255/// numbers its coordinates in creation order, and a linear coordinate keeps
256/// its removed axes in the record, where they do not move. A lookup table
257/// (a tabular spectral coordinate's, or a tabular coordinate's) is adjusted
258/// as well. Spectral tables are rebuilt over the selected channels, matching
259/// casacore's `CoordinateSystem::subImageInSitu` for nonlinear frequencies.
260pub(super) fn shift_record_reference_pixels(
261    record: &mut RecordValue,
262    start: &[usize],
263    shape: &[usize],
264) -> Result<(), ImageError> {
265    section_record_reference_pixels(record, start, shape, None)
266}

268pub(super) fn section_record_reference_pixels(
269    record: &mut RecordValue,
270    start: &[usize],
271    shape: &[usize],
272    stride: Option<&[usize]>,
273) -> Result<(), ImageError> {
274    let placed = CoordinateSystem::record_coordinates(record).map_err(|error| {
275        ImageError::InvalidMetadata(format!("subimage coordinate system: {error}"))
276    })?;
277    for coordinate in placed {
278        let shifts = coordinate
279            .pixel_axes
280            .iter()
281            .map(|axis| axis.map_or(0.0, |axis| start[axis] as f64))
282            .collect::<Vec<_>>();
283        let steps = stride.map(|stride| {
284            coordinate
285                .pixel_axes
286                .iter()
287                .map(|axis| axis.map_or(1.0, |axis| stride[axis] as f64))
288                .collect::<Vec<_>>()
289        });
290        let steps = steps.as_deref().unwrap_or_default();
291        let Some(Value::Record(coordinate_record)) = record.get_mut(&coordinate.field) else {
292            continue;
293        };
294        if coordinate_record.get("quality").is_some() {
295            if let [Some(axis)] = coordinate.pixel_axes.as_slice() {
296                section_quality_labels(
297                    coordinate_record,
298                    start[*axis],
299                    shape[*axis],
300                    stride.map_or(1, |stride| stride[*axis]),
301                )?;
302            }
303            continue;
304        }
305        if coordinate.field.starts_with("spectral")
306            && let Some(Some(axis)) = coordinate.pixel_axes.first()
307            && casa_coordinates::SpectralCoordinate::subimage_tabular_record_with_stride(
308                coordinate_record,
309                start[*axis],
310                shape[*axis],
311                stride.map_or(1, |stride| stride[*axis]),
312            )
313            .map_err(|error| {
314                ImageError::InvalidMetadata(format!("subimage spectral coordinate: {error}"))
315            })?
316        {
317            continue;
318        }
319        section_pixels(coordinate_record, &shifts, steps);
320        for parameters in ["wcs", "tabular"] {
321            if let Some(Value::Record(inner)) = coordinate_record.get_mut(parameters) {
322                section_pixels(inner, &shifts, steps);
323            }
324        }
325    }
326    Ok(())
327}

329/// Keep the planes selected by the image section in a native Quality subrecord.
330///
331/// # Errors
332/// The stride must be positive and every selected plane must have a label.
333pub fn section_quality_labels(
334    record: &mut RecordValue,
335    start: usize,
336    count: usize,
337    stride: usize,
338) -> Result<(), ImageError> {
339    if stride == 0 {
340        return Err(ImageError::InvalidMetadata(
341            "Quality image section requires a positive stride".into(),
342        ));
343    }
344    let quality = QualityCoordinate::from_record(record)?;
345    let labels: Vec<_> = quality
346        .values()
347        .iter()
348        .copied()
349        .skip(start)
350        .step_by(stride)
351        .take(count)
352        .collect();
353    if labels.len() != count {
354        return Err(ImageError::InvalidMetadata(
355            "Quality axis has fewer planes than the image section".into(),
356        ));
357    }
358    write_quality_labels(record, &labels)
359}

361/// Replace the plane labels and reference values of a native Quality subrecord.
362///
363/// # Errors
364/// An empty label list cannot describe a Quality coordinate.
365pub fn write_quality_labels(
366    record: &mut RecordValue,
367    labels: &[QualityType],
368) -> Result<(), ImageError> {
369    let first = labels.first().ok_or_else(|| {
370        ImageError::InvalidMetadata("A Quality axis requires at least one plane".into())
371    })?;
372    record.upsert(
373        "quality",
374        ArrayValue::from_string_vec(labels.iter().map(|kind| kind.name().to_owned()).collect())
375            .into(),
376    );
377    record.upsert(
378        "crval",
379        ArrayValue::from_f64_vec(vec![f64::from(first.code())]).into(),
380    );
381    record.upsert("crpix", ArrayValue::from_f64_vec(vec![0.0]).into());
382    Ok(())
383}

385/// Adjust the reference pixels and scale the columns of the coordinate transform.
386fn section_pixels(record: &mut RecordValue, shifts: &[f64], steps: &[f64]) {
387    let step = |axis| steps.get(axis).copied().unwrap_or(1.0);
388    let position = |axis, value| (value - shifts.get(axis).copied().unwrap_or(0.0)) / step(axis);
389    match record.get_mut("crpix") {
390        Some(Value::Array(array)) => map_numeric_values(array, position),
391        Some(Value::Scalar(ScalarValue::Float64(crpix))) => *crpix = position(0, *crpix),
392        _ => {}
393    }
394    if let Some(Value::Array(array)) = record.get_mut("pixelvalues") {
395        map_numeric_values(array, |_, value| position(0, value));
396    }
397    if steps.iter().any(|&step| step != 1.0) {
398        match record.get_mut("cdelt") {
399            Some(Value::Array(array)) => {
400                map_numeric_values(array, |axis, value| value * step(axis))
401            }
402            Some(Value::Scalar(ScalarValue::Float64(cdelt))) => *cdelt *= step(0),
403            _ => {}
404        }
405        // CDELT scales world rows. The stored PC transposes the matrix
406        // used by the transform, so its first index is the input axis.
407        let shape = [steps.len(), steps.len()];
408        match record.get_mut("pc") {
409            Some(Value::Array(ArrayValue::Float64(matrix))) if matrix.shape() == shape => {
410                for (index, coefficient) in matrix.indexed_iter_mut() {
411                    *coefficient *= steps[index[0]] / steps[index[1]];
412                }
413            }
414            Some(Value::Array(ArrayValue::Float32(matrix))) if matrix.shape() == shape => {
415                for (index, coefficient) in matrix.indexed_iter_mut() {
416                    *coefficient *= (steps[index[0]] / steps[index[1]]) as f32;
417                }
418            }
419            _ => {}
420        }
421    }
422}

424/// Transforms the elements of a stored floating-point coordinate array.
425fn map_numeric_values(array: &mut ArrayValue, transform: impl Fn(usize, f64) -> f64) {
426    let values = match array {
427        ArrayValue::Float64(values) => values
428            .iter()
429            .enumerate()
430            .map(|(i, value)| transform(i, *value))
431            .collect::<Vec<_>>(),
432        ArrayValue::Float32(values) => values
433            .iter()
434            .enumerate()
435            .map(|(i, value)| transform(i, f64::from(*value)))
436            .collect::<Vec<_>>(),
437        _ => return,
438    };
439    *array = ArrayValue::from_f64_vec(values);
440}

543    #[test]
544    fn sparse_source_windows_never_exceed_a_dense_cursor() {
545        let section = StridedSection {
546            start: &[1, 3],
547            shape: &[5, 8],
548            stride: &[100_000, 3],

550        let mut read_count = 0;

554            let extent = section.dense_shape(&cursor.shape);
