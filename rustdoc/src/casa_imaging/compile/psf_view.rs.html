26pub(crate) struct PsfView {
27    specification: ProblemSpecification,
28    geometry: GeometryInput,
29    windows: Option<Vec<Vec<SpectralWindowSelection>>>,
30    reservation: Reservation,
31}
32
33impl PsfView {
34    pub(crate) fn compile(
35        self,
36        main: &CompiledProblem,
37        lifecycle: ModelLifecycleRequirements,
38    ) -> Result<Admitted<CompiledProblem>, crate::ApplicationDispatchError> {
39        let observation = match self.windows {
40            Some(windows) => alternate_observation(main.observation(), windows)
41                .map_err(|error| crate::ApplicationDispatchError::Preparation(error.into()))?,
42            None => main.observation().clone(),
43        };
44        Ok(Admitted {
45            value: casa_imaging_model::compile(ProblemInput::new(
46                self.specification,
47                self.geometry,
48                observation,
49                lifecycle,
50            ))
51            .map_err(crate::ApplicationDispatchError::Compile)?,
52            reservation: self.reservation,
53        })
54    }
55}

57/// CASA forms the alternate PSF as a new mosaic view of the same selection.
58/// The alternate phase law participates in row packing and convolution, not
59/// just the image header. The other products retain the main view.
60pub(super) fn prepare(
61    request: &ImagingRequest,
62    main: &PreparedImageDomain,
63    tracks_source: bool,
64    surveyed: &Surveyed<'_>,
65    context: &RunContext,
66    specification: &ProblemSpecification,
67    writer: &mut CasaImageWriter,
68) -> Result<Option<PsfView>, PrepareError> {
69    let Some(text) = request.psfphasecenter.as_deref() else {
70        return Ok(None);
71    };
72    if !request.calcpsf || !matches!(request.gridder, Gridder::Mosaic { .. }) {
73        return Ok(None);
74    }
75    // A single selected channel leaves only the tiny frame-conversion
76    // changes to determine the Taylor slope of the tracked continuum.
77    if tracks_source
78        && request.deconvolver == Deconvolver::Mtmfs
79        && surveyed
80            .spectral
81            .selected_source_channels
82            .iter()
83            .flat_map(|windows| windows.values())
84            .map(Vec::len)
85            .sum::<usize>()
86            == 1
87    {
88        return Err(PrepareError::MovingTaylorPsfSingleChannel);
89    }
90    let phase = direction::parse_phase_center_direction(text)?;
91    let (longitude, latitude) = (phase.longitude_rad(), phase.latitude_rad());
92    direction::refuse_ncp_at_equator(request.projection, latitude)?;
93    let direction = direction::direction_spec(
94        main.image_size,
95        request.cell,
96        request.projection,
97        phase.frame(),
98        [longitude, latitude],
99    );
100    // Cover both compiled and uncompiled single-domain metadata during
101    // conversion. The alternate view shares the observation manifest
102    // when frame conversion leaves the row and channel reads unchanged.
103    let variable = (request.restfreq.len() as u64 + request.scales.len() as u64) * 32
104        + (main.coordinates.obs_info().telescope.len() as u64
105            + main.coordinates.obs_info().observer.len() as u64)
106            * 2;
107    let cube = request.specmode == SpecMode::Cube;
108    let mut reservation = admit(
109        &context.host,
110        &context.policy,
111        &Demand {
112            phase: "separate mosaic PSF view",
113            memory: 64 * 1024 + variable,
114        },
115    )?;
116    let mut axis = None;
117    let mut windows = None;
118    if cube {
119        let mut cube_charge = admit_cube_axis(surveyed, context)?;
120        let alternate = alternate_axis(request, surveyed, phase, direction)?;
121        let changed_selection = alternate.selected_source_channels
122            != surveyed.spectral.selected_source_channels
123            || alternate.window_reads != surveyed.spectral.window_reads;
124        if changed_selection {
125            windows = Some(alternate_windows(surveyed, &alternate)?);
126            cube_charge.retain(alternate_selection_bytes(surveyed));
127            reservation.join(cube_charge);
128        }
129        axis = Some(alternate);
130    }
131    let spectral = axis.as_ref().unwrap_or(surveyed.spectral);
132    let coordinates = coordinates(request, main, direction, spectral);
133    let geometry = GeometryInput::new(
134        vec![main.spec().with_direction(direction)],
135        CentreLaws::new(
136            PhaseCentreLaw::Fixed(phase),
137            instrument::pointing_centre_law(&request.gridder),
138        ),
139        UvwCoordinateLaw::MosaicPhaseTrackingCentre,
140        spectral.coordinate(),
141    );
142    writer.set_main_psf_coordinates(coordinates);
143    Ok(Some(PsfView {
144        specification: specification.clone(),
145        geometry,
146        windows,
147        reservation,
148    }))
149}

151fn alternate_axis(
152    request: &ImagingRequest,
153    surveyed: &Surveyed<'_>,
154    phase: SkyDirection,
155    direction: DirectionCoordinateSpec,
156) -> Result<PreparedSpectralAxis<'static>, PrepareError> {
157    let frame = FrameContext {
158        anchor_time_mjd_seconds: surveyed.survey.image_time_mjd_seconds,
159        field_id: *surveyed.survey.fields.first().expect("selected field") as usize,
160        phase: MDirection::from_angles(
161            phase.longitude_rad(),
162            phase.latitude_rad(),
163            DirectionRef::J2000,
164        ),
165        direction,
166        engine: surveyed.engine,
167    };
168    let spectral = spectral::prepare_axis(
169        request,
170        (surveyed.stores, surveyed.reference),
171        None,
172        &frame,
173        None,
174        request.reffreq,
175    )?;
176    if spectral.output_channels != surveyed.spectral.output_channels {
177        return Err(PrepareError::AlternatePsfChannelCount {
178            main: surveyed.spectral.output_channels,
179            psf: spectral.output_channels,
180        });
181    }
182    Ok(spectral)
183}

200/// Validate the restart against the independently calculated alternate grid
201/// at the saved direction. Other products still use the main grid.
202pub(super) fn saved_cube_grid(
203    path: &std::path::Path,
204    request: &ImagingRequest,
205    main: &PreparedImageDomain,
206    surveyed: &Surveyed<'_>,
207    context: &RunContext,
208) -> Result<Option<RegridTemplate>, PrepareError> {
209    let image = PagedImage::<f32>::open(path).map_err(|error| PrepareError::Restart {
210        path: path.to_path_buf(),
211        reason: error.to_string(),
212    })?;
213    let actual = image.coordinates();
214    if actual.n_coordinates() == 0 {
215        return Ok(None);
216    }
217    let CoordinateModel::Direction(saved) = actual.coordinate(0) else {
218        return Ok(None);
219    };
220    if saved.direction_ref() != DirectionRef::J2000 {
221        return Ok(None);
222    }
223    let values = saved.reference_value();
224    let main_values = main.coordinates.coordinate(0).reference_value();
225    if main.direction.reference_direction().frame() == DirectionFrame::J2000
226        && values
227            .iter()
228            .zip(main_values)
229            .all(|(a, b)| (a - b).abs() <= 1e-13)
230    {
231        return Ok(None);
232    }
233    let phase = SkyDirection::new(DirectionFrame::J2000, values[0], values[1]);
234    direction::refuse_ncp_at_equator(request.projection, values[1])?;
235    let direction = direction::direction_spec(
236        main.image_size,
237        request.cell,
238        request.projection,
239        phase.frame(),
240        [values[0], values[1]],
241    );
242    let _axis = admit_cube_axis(surveyed, context)?;
243    let spectral = alternate_axis(request, surveyed, phase, direction)?;
244    let [nx, ny] = main.image_size;
245    Ok(Some(RegridTemplate {
246        coordinates: coordinates(request, main, direction, &spectral),
247        shape: vec![nx, ny, request.stokes.len(), spectral.output_channels],
248    }))
249}

251fn alternate_windows(
252    surveyed: &Surveyed<'_>,
253    spectral: &PreparedSpectralAxis<'_>,
254) -> Result<Vec<Vec<SpectralWindowSelection>>, PrepareError> {
255    surveyed
256        .stores
257        .iter()
258        .zip(spectral.selected_source_channels.iter())
259        .zip(&spectral.window_reads)
260        .map(|((store, channels), reads)| {
261            let factors = envelope::first_buffer_beam_factors(
262                (store.catalogue(), &store.engine),
263                &store.survey,
264                None,
265            )?;
266            selection::window_selections(&store.survey, channels, &factors, reads)
267        })
268        .collect()
269}

271fn alternate_observation(
272    main: &ObservationSnapshot,
273    windows: Vec<Vec<SpectralWindowSelection>>,
274) -> Result<ObservationSnapshot, casa_imaging_model::CompileObservationError> {
275    let sources = main
276        .sources()
277        .iter()
278        .zip(windows)
279        .map(|(source, windows)| {
280            let rows = source.selection();
281            ObservationSourceInput::new(
282                source.provenance().clone(),
283                ObservationSelection::new(
284                    rows.samples(),
285                    rows.rows_filter().clone(),
286                    rows.partitions().to_vec(),
287                    windows,
288                ),
289                source.columns(),
290                source.corrected_data_present(),
291            )
292        })
293        .collect();
294    compile_observation(ObservationSnapshotInput::new(sources))
295}

297fn alternate_selection_bytes(surveyed: &Surveyed<'_>) -> u64 {
298    surveyed
299        .stores
300        .iter()
301        .map(|store| {
302            let windows = &store.survey.spectral_windows;
303            let channels: usize = windows
304                .iter()
305                .map(|window| window.frequencies_hz.len())
306                .sum();
307            // Each window holds the catalog, three indexes and up to three
308            // interval lists, each bounded by the number of physical channels.
309            let channel_bytes = size_of::<f64>() + 3 * (size_of::<u32>() + size_of::<[f64; 2]>());
310            let partitions =
311                store.survey.partitions.len() * size_of::<casa_imaging_model::PartitionSelection>();
312            (windows.len() * size_of::<SpectralWindowSelection>()
313                + channels * channel_bytes
314                + partitions) as u64
315        })
316        .sum()
317}

319fn admit_cube_axis(
320    surveyed: &Surveyed<'_>,
321    context: &RunContext,
322) -> Result<Reservation, PrepareError> {
323    // The alternate axis builder uses the same surveyed inputs and channel
324    // scratch as the main cube, while the main specification is live.
325    let workspace: u64 = surveyed
326        .stores
327        .iter()
328        .map(|store| store.survey_charge.memory())
329        .sum();
330    Ok(admit(
331        &context.host,
332        &context.policy,
333        &Demand {
334            phase: "separate mosaic cube spectral axis",
335            memory: workspace + alternate_selection_bytes(surveyed),
336        },
337    )?)
