2//! What CASA 6.7.6.14's calibration tasks return (#931), built from
3//! casa-rs's reports. `tools/casa-compat/record_casa_returns.py` records the
4//! structure of CASA's own returns, which the task tests compare these with.
5//!
6//! - `gaincal` and `bandpass` return the tables they applied and solved, the
7//!   data they selected and how far their solution intervals got
8//!   (`solvestats`, [`SolveStats`]).
9//! - `fluxscale` returns its flux densities per field and spectral window,
10//!   with the fit across windows.
11//! - `applycal` and `gencal` return `None`.
12

22/// `gaincal`'s and `bandpass`'s return: the tables applied while solving,
23/// sorted as CASA lists them, the data selected, the table solved, and the
24/// solve's interval counts.
25pub fn solve_record(
26    applied: &[ApplyCalibrationTableSpec],
27    selected: &SelectedData,
28    solved: &Path,
29    stats: &SolveStats,
30) -> CasaValue {
31    let mut apply_tables = applied
32        .iter()
33        .map(|spec| spec.path.display().to_string())
34        .collect::<Vec<_>>();
35    apply_tables.sort();
36    apply_tables.dedup();
37    let ids = |ids: &[i32]| CasaValue::int64_vector(ids.iter().map(|&id| i64::from(id)));
38    CasaValue::dict([
39        ("apply_tables", CasaValue::str_vector(apply_tables)),
40        (
41            "selectvis",
42            CasaValue::dict([
43                ("antennas", ids(&selected.antennas)),
44                ("field", ids(&selected.fields)),
45                ("intents", CasaValue::str_vector(selected.intents.clone())),
46                ("observation", ids(&selected.observations)),
47                ("scan", ids(&selected.scans)),
48                ("spw", ids(&selected.spectral_windows)),
49            ]),
50        ),
51        (
52            "solve_tables",
53            CasaValue::str_vector([solved.display().to_string()]),
54        ),
55        ("solvestats", solvestats(stats)),
56    ])
57}

59/// CASA's `solvestats`: the counts summed over the windows, then each
60/// window's as `spw<id>`, each with its antennas' as `ant<id>`. CASA sorts
61/// every dictionary's keys as text, so `ant10` comes before `ant2`.
62fn solvestats(stats: &SolveStats) -> CasaValue {
63    let windows = stats.windows.iter().map(|window| {
64        let antennas = window.antennas.iter().map(|antenna| {
65            let mut entries = counts(&antenna.counts);
66            entries.push((
67                "used_as_refant".to_string(),
68                CasaValue::int64_vector([antenna.used_as_refant as i64]),
69            ));
70            (format!("ant{}", antenna.antenna_id), sorted(entries))
71        });
72        let mut entries = counts(&window.counts);
73        entries.extend(antennas);
74        (format!("spw{}", window.spw_id), sorted(entries))
75    });
76    let mut entries = counts(&stats.total().unwrap_or_default());
77    entries.extend(windows);
78    sorted(entries)
79}

81/// One stage's counts, as CASA names them.
82fn counts(counts: &SolveCounts) -> Vec<(String, CasaValue)> {
83    let vector = |values: &[u64]| CasaValue::int64_vector(values.iter().map(|&v| v as i64));
84    vec![
85        (
86            "above_minblperant".to_string(),
87            vector(&counts.above_minblperant),
88        ),
89        ("above_minsnr".to_string(), vector(&counts.above_minsnr)),
90        ("data_unflagged".to_string(), vector(&counts.data_unflagged)),
91        ("expected".to_string(), vector(&counts.expected)),
92    ]
93}
94
95/// `entries` as a dictionary sorted by key.
96fn sorted(mut entries: Vec<(String, CasaValue)>) -> CasaValue {
97    entries.sort_by(|(left, _), (right, _)| left.cmp(right));
98    CasaValue::dict(entries)
99}

101/// `fluxscale`'s return: each transfer field's flux densities, keyed by its
102/// id, then the windows' frequencies, ids and names.
103pub fn fluxscale_record(report: &FluxScaleReport) -> CasaValue {
104    let fields = report
105        .fields
106        .iter()
107        .map(|(field_id, field)| (field_id.to_string(), field_record(field)));
108    let windows = [
109        (
110            "freq".to_string(),
111            CasaValue::float64_vector(report.frequencies_hz.iter().copied()),
112        ),
113        (
114            "spwID".to_string(),
115            CasaValue::int64_vector(report.spw_ids.iter().map(|&id| i64::from(id))),
116        ),
117        (
118            "spwName".to_string(),
119            CasaValue::str_vector(report.spw_names.clone()),
120        ),
121    ];
122    CasaValue::dict(fields.chain(windows))
123}

125/// One field's flux densities per window, keyed by the window's id, and
126/// its fit across windows.
127fn field_record(field: &FluxScaleFieldResult) -> CasaValue {
128    let windows = field.spw_results.iter().map(|(spw_id, result)| {
129        (
130            spw_id.to_string(),
131            CasaValue::dict([
132                ("fluxd", CasaValue::float64_vector(result.fluxd)),
133                ("fluxdErr", CasaValue::float64_vector(result.fluxd_err)),
134                ("numSol", CasaValue::float64_vector(result.num_sol)),
135            ]),
136        )
137    });
138    let fit = [
139        ("covarMat", matrix(&field.covar_mat)),
140        ("fieldName", CasaValue::string(field.field_name.clone())),
141        ("fitFluxd", CasaValue::float(field.fit_fluxd)),
142        ("fitFluxdErr", CasaValue::float(field.fit_fluxd_err)),
143        ("fitRefFreq", CasaValue::float(field.fit_ref_frequency_hz)),
144        (
145            "spidx",
146            CasaValue::float64_vector(field.spidx.iter().copied()),
147        ),
148        (
149            "spidxerr",
150            CasaValue::float64_vector(field.spidx_err.iter().copied()),
151        ),
152    ]
153    .map(|(key, value)| (key.to_string(), value));
154    CasaValue::dict(windows.chain(fit))
155}

157/// `rows` as a two-dimensional `float64` array; none is CASA's `(0, 0)`.
158fn matrix(rows: &[Vec<f64>]) -> CasaValue {
159    let columns = rows.first().map_or(0, Vec::len);
160    CasaValue::FloatArray {
161        dtype: FloatDtype::Float64,
162        shape: vec![rows.len(), columns],
163        data: rows.iter().flatten().copied().map(CasaFloat).collect(),
164    }
165}
