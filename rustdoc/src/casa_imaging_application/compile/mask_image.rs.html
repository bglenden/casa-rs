3//! A CASA mask image, read onto an image domain's direction grid and
4//! matched to its channels and Stokes planes as tclean matches it
5//! (`SDMaskHandler::fillMask`, `expandMask`, `copyMask`).
6//!
7//! tclean expands a mask when it has one channel and the image several, or
8//! else one Stokes plane and the image several (`expandMask`):
9//!
10//! * a continuum Stokes I mask to every channel and Stokes plane;
11//! * a continuum mask of the image's Stokes planes to every channel, plane
12//!   by plane, and a continuum mask of another number of Stokes planes is
13//!   refused;
14//! * a cube Stokes I mask to every Stokes plane, channel by channel, when
15//!   its channels are the image's (their number and spectral coordinate);
16//!   otherwise it is refused.
17//!
18//! Any other mask is regridded onto the image's direction and spectral
19//! axes (`copyMask`): casa-rs reads one whose Stokes planes are the
20//! image's and whose channels are the image's, or one continuum channel,
21//! and does not regrid a mask along frequency or Stokes yet (#980).
22//!
23//! The planes share one direction map. Copy an aligned plane directly.
24//! On a shifted grid, sample signed values linearly, treating invalid
25//! input pixels as zero, then take finite positive support as CASA does.
26

39/// The bound on the tiles a mask image's read caches, as a restart's CLEAN
40/// mask is read.
41const CACHE_BYTES: usize = 64 * 1024;
42
43/// The planes of the image a mask is read onto: its channels and Stokes
44/// planes, and its spectral coordinate.
45#[derive(Clone, Copy)]
46pub(super) struct TargetPlanes<'a> {
47    pub(super) channels: usize,
48    pub(super) polarizations: usize,
49    pub(super) spectral: Option<&'a SpectralCoordinate>,
50}

52/// Where a mask image's axes are, by pixel axis.
53struct MaskAxes {
54    shape: Vec<usize>,
55    direction: [usize; 2],
56    stokes: Option<usize>,
57    spectral: Option<usize>,
58}
59
60impl MaskAxes {
61    fn extent(&self, axis: Option<usize>) -> usize {
62        axis.map_or(1, |axis| self.shape[axis])
63    }
64
65    /// The start and shape of the direction plane of `channel` and
66    /// `stokes`: one pixel along every other axis.
67    fn plane(&self, channel: usize, stokes: usize) -> (Vec<usize>, Vec<usize>) {
68        let mut start = vec![0; self.shape.len()];
69        let mut shape = vec![1; self.shape.len()];
70        for axis in self.direction {
71            shape[axis] = self.shape[axis];
72        }
73        if let Some(axis) = self.spectral {
74            start[axis] = channel;
75        }
76        if let Some(axis) = self.stokes {
77            start[axis] = stokes;
78        }
79        (start, shape)
80    }
81
82    /// Pixels of one direction plane.
83    fn plane_cells(&self) -> u64 {
84        self.direction.iter().fold(1_u64, |cells, &axis| {
85            cells.saturating_mul(self.shape[axis] as u64)
86        })
87    }
88}

90/// A mask image of real pixels.
91enum RealImage {
92    Float32(PagedImage<f32>),
93    Float64(PagedImage<f64>),
94}

96/// The sampling mode reused across all planes of an input mask.
97enum MaskSampling {
98    Aligned,
99    Affine(DirectionAffine),
100    Mapped(Box<[[f64; 2]]>),
101}
102
103impl MaskSampling {
104    fn source_pixel(&self, [x, y]: [usize; 2], stride: usize) -> [f64; 2] {
105        match self {
106            Self::Aligned => unreachable!("aligned masks are copied directly"),
107            Self::Affine(map) => map.source_pixel([x as f64, y as f64]),
108            Self::Mapped(map) => map[x * stride + y],
109        }
110    }
111}

113/// A real mask image opened onto an image domain, read plane by plane once
114/// its read is admitted ([`Self::reading_bytes`]), as a restart's CLEAN
115/// mask is.
116pub(crate) struct MaskImage {
117    image: RealImage,
118    axes: MaskAxes,
119    /// The planes tclean gives it on the image (`SDMaskHandler::fillMask`).
120    held: MaskPlanes,
121    source: DirectionCoordinateSpec,
122    target: DirectionCoordinateSpec,
123    target_size: [usize; 2],
124    /// Bytes of one pixel of the image.
125    pixel_bytes: u64,
126    tile_bytes: u64,
127}
128
129impl MaskImage {
130    /// Open the mask image at `path` for `target`'s direction grid of
131    /// `target_size` pixels on an image of `planes`, reading only its
132    /// description: CASA's refusal, or the planes it gives the mask.
133    pub(super) fn open(
134        path: &Path,
135        target: DirectionCoordinateSpec,
136        target_size: [usize; 2],
137        planes: TargetPlanes<'_>,
138    ) -> Result<Self, PrepareError> {
139        let (image, pixel_bytes) = match AnyPagedImage::open_with_cache(path, CACHE_BYTES)? {
140            AnyPagedImage::Float32(image) => (RealImage::Float32(image), 4_u64),
141            AnyPagedImage::Float64(image) => (RealImage::Float64(image), 8),
142            AnyPagedImage::Complex32(_) | AnyPagedImage::Complex64(_) => {
143                return Err(PrepareError::ComplexMaskImage);
144            }
145        };
146        let (shape, coordinates, tile) = match &image {
147            RealImage::Float32(image) => (image.shape(), image.coordinates(), image.tile_shape()),
148            RealImage::Float64(image) => (image.shape(), image.coordinates(), image.tile_shape()),
149        };
150        let axes = mask_axes(shape, coordinates)?;
151        let held = expansion(path, coordinates, &axes, planes)?;
152        let direction = coordinates
153            .find_coordinate(CoordinateType::Direction)
154            .ok_or(PrepareError::MaskImageDirection)?;
155        let source = super::domains::direction_model_spec(coordinates.coordinate(direction))?;
156        let tile_bytes = tile.iter().fold(pixel_bytes, |bytes, &side| {
157            bytes.saturating_mul(side as u64)
158        });
159        Ok(Self {
160            image,
161            axes,
162            held,
163            source,
164            target,
165            target_size,
166            pixel_bytes,
167            tile_bytes,
168        })
169    }
170
171    /// Change the output coordinate for the allocator-counted read law.
172    pub(crate) fn with_target_direction(mut self, target: DirectionCoordinateSpec) -> Self {
173        self.target = target;
174        self
175    }

177    /// Whether the saved mask and output have the same direction grid.
178    pub(super) fn matches_direction_grid(&self) -> bool {
179        if self.axes.direction.map(|axis| self.axes.shape[axis]) != self.target_size
180            || self.source.projection() != self.target.projection()
181            || self.source.reference_direction().frame()
182                != self.target.reference_direction().frame()
183        {
184            return false;
185        }
186        if self.source == self.target {
187            return true;
188        }
189        // Reading the CASA header can round the same WCS differently. Use
190        // the displacement in pixels so this allowance cannot admit a shift.
191        let [nx, ny] = self.target_size.map(|size| size.saturating_sub(1) as f64);
192        let source = self.source.coordinate();
193        let target = self.target.coordinate();
194        [
195            [0.0, 0.0],
196            [nx, 0.0],
197            [0.0, ny],
198            [nx, ny],
199            [nx / 2.0, ny / 2.0],
200        ]
201        .into_iter()
202        .all(|pixel| {
203            target
204                .pixel_to_world(pixel)
205                .and_then(|world| source.world_to_pixel(world))
206                .is_ok_and(|mapped| {
207                    mapped
208                        .into_iter()
209                        .zip(pixel)
210                        .all(|(a, b)| (a - b).abs() <= 1e-7)
211                })
212        })
213    }
214
215    fn exact_affine(&self) -> Option<DirectionAffine> {
216        direction_affine(
217            &self.source.coordinate(),
218            &self.target.coordinate(),
219            self.target_size,
220        )
221    }
222
223    /// MEM's measured image-mask shapes have an aligned or affine grid.
224    pub(super) fn has_shared_tangent_grid(&self) -> bool {
225        self.matches_direction_grid() || self.exact_affine().is_some()
226    }
227
228    fn sampling(&self) -> Result<MaskSampling, PrepareError> {
229        if self.matches_direction_grid() {
230            return Ok(MaskSampling::Aligned);
231        }
232        if let Some(affine) = self.exact_affine() {
233            return Ok(MaskSampling::Affine(affine));
234        }
235        let [nx, ny] = self.target_size;
236        let cells = nx.checked_mul(ny).ok_or(MaskError::InvalidShape)?;
237        let mut positions = Vec::with_capacity(cells);
238        let source = self.source.coordinate();
239        let target = self.target.coordinate();
240        for x in 0..nx {
241            for y in 0..ny {
242                let position = target
243                    .pixel_to_world([x as f64, y as f64])
244                    .and_then(|world| source.world_to_pixel(world))
245                    .map_err(|_| MaskError::UnsupportedReprojection)?;
246                if !position.into_iter().all(f64::is_finite) {
247                    return Err(MaskError::UnsupportedReprojection.into());
248                }
249                positions.push(position);
250            }
251        }
252        Ok(MaskSampling::Mapped(positions.into_boxed_slice()))
253    }
254
255    /// Target pixels of one plane.
256    fn target_plane(&self) -> u64 {
257        (self.target_size[0] as u64).saturating_mul(self.target_size[1] as u64)
258    }
259
260    /// Heap bytes [`Self::read`] holds at most: one source plane and its
261    /// default pixel mask, the map for a general direction transform,
262    /// the collected and shared support, the tile cache and I/O buffers.
263    pub(crate) fn reading_bytes(&self) -> u64 {
264        let map = if self.has_shared_tangent_grid() {
265            0
266        } else {
267            self.target_plane()
268                .saturating_mul(size_of::<[f64; 2]>() as u64)
269        };
270        self.axes
271            .plane_cells()
272            .saturating_mul(self.pixel_bytes + 1)
273            .saturating_add(map)
274            .saturating_add(self.retained_bytes().saturating_mul(2))
275            .saturating_add(self.tile_bytes.saturating_mul(2))
276            .saturating_add(CACHE_BYTES as u64)
277    }
278
279    /// Heap bytes the plan keeps of [`Self::read`]: its shared support, a
280    /// plane for each plane it holds.
281    pub(crate) fn retained_bytes(&self) -> u64 {
282        self.target_plane().saturating_mul(self.held.count() as u64)
283    }
284
285    /// The plan of the mask on the target grid: each source plane read and
286    /// its support reprojected, one plane at a time.
287    pub(crate) fn read(self) -> Result<ReconstructionMaskPlan, PrepareError> {
288        let held = self.held;
289        let cells = self.target_size[0] * self.target_size[1];
290        let mut support = Vec::with_capacity(cells * held.count());
291        let sampling = self.sampling()?;
292        for channel in 0..held.channels() {
293            for stokes in 0..held.polarizations() {
294                let (start, shape) = self.axes.plane(channel, stokes);
295                let target = (&sampling, self.target_size, &mut support);
296                match &self.image {
297                    RealImage::Float32(image) => {
298                        append_plane_support(image, &start, &shape, self.axes.direction, target)?;
299                    }
300                    RealImage::Float64(image) => {
301                        append_plane_support(image, &start, &shape, self.axes.direction, target)?;
302                    }
303                }
304            }
305        }
306        let support: Arc<[bool]> = support.into();
307        Ok(ReconstructionMaskPlan::Reprojected {
308            coordinate: self.target,
309            planes: held,
310            support,
311        })
312    }
313}

315/// The mask image at `path` opened onto its own direction grid, for an
316/// image of `channels` channels on its own spectral coordinate and
317/// `polarizations` Stokes planes: the counting law's view of the reader
318/// compile uses ([`crate::mask_image_law`]).
319pub(crate) fn open_onto_itself(
320    path: &Path,
321    channels: usize,
322    polarizations: usize,
323) -> Result<MaskImage, PrepareError> {
324    let (coordinates, shape) = match AnyPagedImage::open(path)? {
325        AnyPagedImage::Float32(image) => (image.coordinates().clone(), image.shape().to_vec()),
326        AnyPagedImage::Float64(image) => (image.coordinates().clone(), image.shape().to_vec()),
327        AnyPagedImage::Complex32(_) | AnyPagedImage::Complex64(_) => {
328            return Err(PrepareError::ComplexMaskImage);
329        }
330    };
331    let axes = mask_axes(&shape, &coordinates)?;
332    let direction = coordinates
333        .find_coordinate(CoordinateType::Direction)
334        .ok_or(PrepareError::MaskImageDirection)?;
335    let target = super::domains::direction_model_spec(coordinates.coordinate(direction))?;
336    let spectral = match coordinates
337        .find_coordinate(CoordinateType::Spectral)
338        .map(|index| coordinates.coordinate(index))
339    {
340        Some(CoordinateModel::Spectral(spectral)) => Some(&**spectral),
341        _ => None,
342    };
343    MaskImage::open(
344        path,
345        target,
346        axes.direction.map(|axis| shape[axis]),
347        TargetPlanes {
348            channels,
349            polarizations,
350            spectral,
351        },
352    )
353}

368/// Where the direction, Stokes and spectral axes of an image of `shape`
369/// are: each coordinate's pixel axes follow those of the coordinates
370/// before it. Every other axis has one pixel.
371fn mask_axes(shape: &[usize], coordinates: &CoordinateSystem) -> Result<MaskAxes, PrepareError> {
372    let first_axis = |index: usize| {
373        (0..index)
374            .map(|k| coordinates.coordinate(k).n_pixel_axes())
375            .sum::<usize>()
376    };
377    let refuse = || PrepareError::MaskImageShape {
378        shape: shape.to_vec(),
379    };
380    let direction = coordinates
381        .find_coordinate(CoordinateType::Direction)
382        .ok_or(PrepareError::MaskImageDirection)?;
383    let longitude = first_axis(direction);
384    let axes = MaskAxes {
385        shape: shape.to_vec(),
386        direction: [longitude, longitude + 1],
387        stokes: coordinates
388            .find_coordinate(CoordinateType::Stokes)
389            .map(first_axis),
390        spectral: coordinates
391            .find_coordinate(CoordinateType::Spectral)
392            .map(first_axis),
393    };
394    let named = [
395        Some(axes.direction[0]),
396        Some(axes.direction[1]),
397        axes.stokes,
398        axes.spectral,
399    ];
400    if shape.len() < longitude + 2
401        || named.iter().flatten().any(|axis| *axis >= shape.len())
402        || (0..shape.len()).any(|axis| !named.contains(&Some(axis)) && shape[axis] != 1)
403    {
404        return Err(refuse());
405    }
406    Ok(axes)
407}

409/// Extract the two direction axes of a slice that is singleton elsewhere.
410fn direction_plane<T>(mut plane: ArrayViewD<'_, T>, axes: [usize; 2]) -> ArrayView2<'_, T> {
411    for axis in (0..plane.ndim()).rev() {
412        if !axes.contains(&axis) {
413            plane = plane.index_axis_move(Axis(axis), 0);
414        }
415    }
416    plane
417        .into_dimensionality::<Ix2>()
418        .expect("two direction axes")
419}

421/// Write one plane's support, x-major. In CASA's copyMask the default
422/// validity mask zeros source values before interpolation.
423fn append_plane_support<T>(
424    image: &PagedImage<T>,
425    start: &[usize],
426    shape: &[usize],
427    direction: [usize; 2],
428    (sampling, target, support): (&MaskSampling, [usize; 2], &mut Vec<bool>),
429) -> Result<(), PrepareError>
430where
431    T: ImagePixel + Into<f64> + Copy,
432{
433    let values = image.get_slice(start, shape)?;
434    let mask = image.get_mask_slice(start, shape, &vec![1; shape.len()])?;
435    let values = direction_plane(values.view(), direction);
436    let mask = mask
437        .as_ref()
438        .map(|mask| direction_plane(mask.view(), direction));
439    let sample = |x: usize, y: usize| {
440        if mask.as_ref().is_none_or(|mask| mask[[x, y]]) {
441            values[[x, y]].into()
442        } else {
443            0.0
444        }
445    };
446    let positive = |value: f64| value.is_finite() && value > 0.0;
447    let source_shape = [values.nrows(), values.ncols()];
448    if matches!(sampling, MaskSampling::Aligned) {
449        for x in 0..target[0] {
450            for y in 0..target[1] {
451                support.push(positive(sample(x, y)));
452            }
453        }
454    } else {
455        for x in 0..target[0] {
456            for y in 0..target[1] {
457                let at = sampling.source_pixel([x, y], target[1]);
458                let value = linear_sample(source_shape, at, |i, j| Some(sample(i, j)));
459                support.push(value.is_some_and(|value| {
460                    // CASA's mask lattice stores Float before fillMask
461                    // thresholds it. Native Float64 masks keep their precision.
462                    if size_of::<T>() == size_of::<f32>() {
463                        let value = value as f32;
464                        value.is_finite() && value > 0.0
465                    } else {
466                        positive(value)
467                    }
468                }));
469            }
470        }
471    }
472    Ok(())
473}

475/// The planes tclean gives a mask of `axes` on an image of `target`
476/// (`SDMaskHandler::fillMask`), or its refusal.
477fn expansion(
478    path: &Path,
479    coordinates: &CoordinateSystem,
480    axes: &MaskAxes,
481    target: TargetPlanes<'_>,
482) -> Result<MaskPlanes, PrepareError> {
483    let channels = axes.extent(axes.spectral);
484    let stokes = axes.extent(axes.stokes);
485    let refuse = |reason: String| PrepareError::MaskImageExpansion {
486        path: path.into(),
487        reason,
488    };
489    let stokes_refusal = || {
490        refuse(format!(
491            "Cannot extend the input mask of {stokes} Stokes planes to the output mask of {} \
492             Stokes planes. Please modify the input mask to make it a Stokes I mask or a mask \
493             with the same number of Stokes planes as the output.",
494            target.polarizations
495        ))
496    };
497    let expands =
498        (channels == 1 && target.channels > 1) || (stokes == 1 && target.polarizations > 1);
499    if expands {
500        // `expandMask`.
501        return if channels == 1 {
502            match stokes {
503                1 => Ok(MaskPlanes::SHARED),
504                stokes if stokes == target.polarizations => Ok(MaskPlanes::new(1, stokes)?),
505                _ => Err(stokes_refusal()),
506            }
507        } else if stokes == 1 {
508            same_channels(coordinates, axes, target).map_err(refuse)?;
509            Ok(MaskPlanes::new(channels, 1)?)
510        } else {
511            Err(refuse(format!(
512                "Input mask,{} does not conform with the number of channels in output mask",
513                path.display()
514            )))
515        };
516    }
517    // `copyMask` regrids the mask's direction and spectral axes onto the
518    // image's: a plane for each Stokes plane and channel the image has.
519    let copies = stokes == target.polarizations
520        && (channels == 1
521            || (channels == target.channels && same_channels(coordinates, axes, target).is_ok()));
522    if copies {
523        Ok(MaskPlanes::new(channels, stokes)?)
524    } else {
525        Err(PrepareError::MaskImageRegrid {
526            path: path.into(),
527            shape: axes.shape.clone(),
528        })
529    }
530}

532/// Whether the mask's channels are the image's: as many, on a spectral
533/// coordinate near the image's (`SDMaskHandler::compareSpectralCoordinate`,
534/// casacore `SpectralCoordinate::near` at its tolerance of 1e-6 on the
535/// frequency system, rest frequency and linear mapping); CASA's refusal
536/// otherwise.
537fn same_channels(
538    coordinates: &CoordinateSystem,
539    axes: &MaskAxes,
540    target: TargetPlanes<'_>,
541) -> Result<(), String> {
542    let channels = axes.extent(axes.spectral);
543    let spectral = coordinates
544        .find_coordinate(CoordinateType::Spectral)
545        .map(|index| coordinates.coordinate(index));
546    let near = match (spectral, target.spectral) {
547        (Some(CoordinateModel::Spectral(mask)), Some(image)) => spectral_near(mask, image),
548        _ => false,
549    };
550    if channels != target.channels {
551        return Err(if near {
552            "Cannot extend the input mask. The number of the channels in Input mask does not \
553             match with that of the output mask. Use a single channel mask or a mask that \
554             matches the spectral coordiante of the output."
555        } else {
556            "Cannot extend the input mask. Spectral coordiante and the number of channels of \
557             the input mask does not match with those of the output mask. Use a single channel \
558             mask or a mask that matches the spectral coordiante of the output."
559        }
560        .to_string());
561    }
562    if !near {
563        return Err(
564            "Cannot extend the input mask. Spectral coordiante of Input mask does not \
565                    match with that of the output mask. Use a single channel mask or a mask \
566                    that matches the spectral coordiante of the output."
567                .to_string(),
568        );
569    }
570    Ok(())
571}

573/// casacore `SpectralCoordinate::near` at its default tolerance, on what
574/// places the channels: the frequency system, the rest frequency and the
575/// linear mapping.
576fn spectral_near(mask: &SpectralCoordinate, image: &SpectralCoordinate) -> bool {
577    let near = |left: f64, right: f64| {
578        left == right || (left - right).abs() <= 1.0e-6 * left.abs().max(right.abs())
579    };
580    let pairs = |left: Vec<f64>, right: Vec<f64>| {
581        left.len() == right.len()
582            && left
583                .iter()
584                .zip(&right)
585                .all(|(left, right)| near(*left, *right))
586    };
587    let (mask_model, image_model) = (
588        CoordinateModel::Spectral(Box::new(mask.clone())),
589        CoordinateModel::Spectral(Box::new(image.clone())),
590    );
591    mask.frequency_ref() == image.frequency_ref()
592        && mask_model.axis_units() == image_model.axis_units()
593        && near(mask.rest_frequency(), image.rest_frequency())
594        && pairs(mask_model.reference_value(), image_model.reference_value())
595        && pairs(mask_model.increment(), image_model.increment())
596        && pairs(mask_model.reference_pixel(), image_model.reference_pixel())
597}

623        coordinates.add_coordinate(StokesCoordinate::new(types[..stokes].to_vec()));
624        coordinates.add_coordinate(spectral(increment_hz));
625        let axes = mask_axes(&[8, 8, stokes, channels], &coordinates).unwrap();
626        (coordinates, axes)
627    }

629    /// tclean's planes for a mask of (Stokes, channels) on an image of
630    /// (polarizations, channels), whose channels are 1 MHz wide.
631    fn expand(
632        (stokes, channels): (usize, usize),
633        (polarizations, image_channels): (usize, usize),
634        increment_hz: f64,
635    ) -> Result<MaskPlanes, PrepareError> {
636        let (coordinates, axes) = mask(stokes, channels, increment_hz);
637        let image = spectral(1.0e6);
638        expansion(
639            Path::new("in.mask"),
640            &coordinates,
641            &axes,
642            TargetPlanes {
643                channels: image_channels,
644                polarizations,
645                spectral: Some(&image),
646            },
647        )
648    }

650    fn refused(result: Result<MaskPlanes, PrepareError>) -> String {
651        match result {
652            Err(PrepareError::MaskImageExpansion { reason, .. }) => reason,
653            other => panic!("expected CASA's refusal, not {other:?}"),
654        }
655    }

657    /// `SDMaskHandler::fillMask` and `expandMask`: what each mask becomes on
658    /// each image, and what CASA refuses.
659    #[test]
660    fn masks_expand_to_the_image_planes_as_casa_expands_them() {
661        let planes = |channels, stokes| MaskPlanes::new(channels, stokes).unwrap();
662        let near = 1.0e6;
663        // A continuum Stokes I mask to every plane.
664        for image in [(1, 4), (4, 1), (4, 4), (1, 1)] {
665            assert_eq!(expand((1, 1), image, near).unwrap(), MaskPlanes::SHARED);
666        }
667        // A continuum mask of the image's Stokes planes to every channel.
668        assert_eq!(expand((4, 1), (4, 4), near).unwrap(), planes(1, 4));
669        assert_eq!(expand((4, 1), (4, 1), near).unwrap(), planes(1, 4));
670        // A cube Stokes I mask to every Stokes plane, on the image's
671        // channels; a cube of the image's planes as it is.
672        assert_eq!(expand((1, 3), (4, 3), near).unwrap(), planes(3, 1));
673        assert_eq!(expand((1, 3), (1, 3), near).unwrap(), planes(3, 1));
674        assert_eq!(expand((4, 3), (4, 3), near).unwrap(), planes(3, 4));
675        // CASA refuses a continuum mask of another number of Stokes planes
676        // on a cube, and a cube Stokes I mask of other channels.
677        assert!(
678            refused(expand((2, 1), (4, 4), near))
679                .starts_with("Cannot extend the input mask of 2 Stokes planes")
680        );
681        assert!(refused(expand((1, 3), (4, 4), near)).contains("number of the channels"));
682        assert!(refused(expand((1, 3), (4, 3), 2.0e6)).contains("Spectral coordiante of Input"));
683        assert!(refused(expand((1, 3), (4, 4), 2.0e6)).contains("and the number of channels"));
684        // tclean regrids any other mask along frequency or Stokes, which
685        // casa-rs does not yet.
686        for (mask, image, increment) in [
687            ((2, 1), (4, 1), near),
688            ((1, 3), (1, 3), 2.0e6),
689            ((1, 3), (1, 1), near),
690            ((2, 3), (4, 3), near),
691        ] {
692            assert!(
693                matches!(
694                    expand(mask, image, increment),
695                    Err(PrepareError::MaskImageRegrid { .. })
696                ),
697                "{mask:?} on {image:?}"
698            );
699        }
700    }
