2//! Project MS discovery and simanalyze's diameter-squared concat weights.
3use super::{Parameters, failure};
4use crate::simobserve::SimobserveError;
5use crate::simutil::analysis;
6use casa_core::ScalarValue;
7use casa_ms::MeasurementSet;
8use std::path::{Path, PathBuf};
9
10pub(super) struct Inputs {
11    pub synthesis: Vec<PathBuf>,
12    pub total_power: Option<PathBuf>,
13}
14
15pub(super) fn discover(
16    p: &Parameters<'_>,
17    directory: &Path,
18    candidates: &[PathBuf],
19) -> Result<Inputs, SimobserveError> {
20    let requested = p.strings("vis");
21    let default = requested.is_empty() || requested == ["default"];
22    let sets = if default {
23        candidates
24            .iter()
25            .filter(|path| path.to_string_lossy().ends_with(".ms"))
26            .filter(|path| {
27                let name = path.file_name().unwrap().to_string_lossy();
28                let noisy = if name.ends_with(".sd.ms") {
29                    name.replace(".sd.ms", ".noisy.sd.ms")
30                } else {
31                    name.replace(".ms", ".noisy.ms")
32                };
33                name.contains(".noisy.") || !directory.join(noisy).exists()
34            })
35            .cloned()
36            .collect::<Vec<_>>()
37    } else {
38        requested
39            .iter()
40            .flat_map(|value| value.split(','))
41            .map(|value| p.locate(directory, value.trim()))
42            .filter(|path| path.exists())
43            .collect()
44    };
45    let mut result = Inputs {
46        synthesis: Vec::new(),
47        total_power: None,
48    };
49    for path in sets {
50        if analysis::is_total_power(&path).map_err(failure)? {
51            if result.total_power.replace(path).is_some() {
52                return Err(failure(
53                    "CASA silently discards extra total-power MS inputs; select one with vis (#841)",
54                ));
55            }
56        } else {
57            result.synthesis.push(path);
58        }
59    }
60    // A previous concatenation is an output, not another independent input.
61    let combined = directory.join(format!(
62        "{}.concat.ms",
63        directory.file_name().unwrap().to_string_lossy()
64    ));
65    if default && result.synthesis.len() > 1 {
66        result.synthesis.retain(|path| path != &combined);
67    }
68    if result.synthesis.is_empty() && result.total_power.is_none() && p.flag("image") {
69        return Err(failure("No MS found to image"));
70    }
71    Ok(result)
72}

74pub(super) fn concat_weights(inputs: &[PathBuf]) -> Result<Vec<f64>, SimobserveError> {
75    let mut weights = Vec::with_capacity(inputs.len());
76    let mut reference = None;
77    for path in inputs {
78        let ms = MeasurementSet::open(path).map_err(failure)?;
79        let antennas = ms.antenna().map_err(failure)?;
80        let diameter = antennas.dish_diameter(0).map_err(failure)?;
81        if ms.spectral_window().map_err(failure)?.row_count() != 1 {
82            return Err(failure(
83                "concat weights require one spectral window per MS; concatenate manually",
84            ));
85        }
86        for row in 0..antennas.row_count() {
87            if antennas.dish_diameter(row).map_err(failure)? != diameter {
88                return Err(failure(
89                    "concat weights require one antenna diameter per MS; concatenate manually",
90                ));
91            }
92        }
93        let interval = ms
94            .main_table()
95            .column_accessor("INTERVAL")
96            .map_err(failure)?;
97        let ScalarValue::Float64(integration) = interval.scalar_cell(0).map_err(failure)? else {
98            return Err(failure("invalid MS integration interval"));
99        };
100        let integration = *integration;
101        for row in 0..ms.main_table().row_count() {
102            if interval.scalar_cell(row).map_err(failure)? != &ScalarValue::Float64(integration) {
103                return Err(failure(
104                    "concat weights require constant integration time; concatenate manually",
105                ));
106            }
107        }
108        let (first_diameter, first_integration) = *reference.get_or_insert((diameter, integration));
109        if integration != first_integration {
110            return Err(failure(
111                "concat weights require matching integration times; concatenate manually",
112            ));
113        }
114        weights.push((diameter / first_diameter).powi(2));
115    }
116    Ok(weights)
117}
