10const CONTROLS: [&str; 4] = ["xframe", "xinterp", "yframe", "yinterp"];
11
12pub(super) fn parse_pointing_axes(
13    task: &TaskInvocation<'_>,
14    [x, y]: [Option<MsAxis>; 2],
15) -> Result<Option<PointingAxes>, TaskFailure> {
16    let parse_axis = |axis: Option<MsAxis>, frame: &str, interp: &str| {
17        parse_pure_axis(axis, frame, interp).map_err(TaskFailure::validation)
18    };
19    let y_is_pointing = y.is_some_and(MsAxis::is_pointing);
20    if y_is_pointing {
21        let frame = control_list(task, "yframe")?;
22        let interp = control_list(task, "yinterp")?;
23        let sides = control_list(task, "yaxislocation")?;
24        if let Control::List(interpolations) = interp {
25            let mut pairs = Vec::with_capacity(interpolations.len());
26            let mut right_ordinates = Vec::with_capacity(interpolations.len());
27            for (index, interp) in interpolations.iter().enumerate() {
28                let frame = frame.parallel("yframe", index, interpolations.len())?;
29                let side = sides.parallel("yaxislocation", index, interpolations.len())?;
30                // The public task applies the ordinate's frame and interpolation to both axes.
31                pairs.push([parse_axis(x, frame, interp)?, parse_axis(y, frame, interp)?]);
32                right_ordinates.push(side == "right");
33            }
34            return Ok(Some(PointingAxes::new(pairs, right_ordinates)));
35        }
36        if matches!(frame, Control::List(_)) {
37            return Err(TaskFailure::validation("not yet implemented: yframe=list").at("yframe"));
38        }
39    }
40    let parse = |axis: Option<MsAxis>, frame, interp| {
41        if !axis.is_some_and(MsAxis::is_pointing) {
42            refuse_controls(task, [frame, interp])?;
43            return Ok(None);
44        }
45        parse_axis(
46            axis,
47            scalar_control(task, frame)?,
48            scalar_control(task, interp)?,
49        )
50    };
51    let pair = [
52        parse(x, CONTROLS[0], CONTROLS[1])?,
53        parse(y, CONTROLS[2], CONTROLS[3])?,
54    ];
55    if pair.iter().all(Option::is_none) {
56        return Ok(None);
57    }
58    let right = match control_list(task, "yaxislocation")? {
59        Control::Scalar(side) => side == "right",
60        Control::List(sides) => y_is_pointing && sides.first() == Some(&"right"),
61    };
62    Ok(Some(PointingAxes::new(vec![pair], vec![right])))
63}
64
65enum Control<'a> {
66    Scalar(&'a str),
67    List(Vec<&'a str>),
68}
69
70impl<'a> Control<'a> {
71    fn parallel(&self, name: &str, index: usize, len: usize) -> Result<&'a str, TaskFailure> {
72        match self {
73            Self::Scalar(value) => Ok(value),
74            Self::List(values) if values.len() == len => Ok(values[index]),
75            Self::List(values) => Err(TaskFailure::validation(format!(
76                "Length mismatch: {name} has {} entries and yinterp has {len}",
77                values.len()
78            ))
79            .at(name)),
80        }
81    }
82}
83
84fn control_list<'a>(task: &'a TaskInvocation<'_>, name: &str) -> Result<Control<'a>, TaskFailure> {
85    if let Some(ParameterValue::Array(values)) = task.value(name)
86        && !(name != "yaxislocation" && values.is_empty())
87    {
88        let values =
89            super::calibration::axis_names(task.value(name)).map_err(TaskFailure::validation)?;
90        if name != "yaxislocation" && values.contains(&"") {
91            return Err(TaskFailure::validation(format!("invalid {name} value ''")).at(name));
92        }
93        return Ok(Control::List(values));
94    }
95    scalar_control(task, name).map(Control::Scalar)
96}
97
98pub(super) fn refuse_nonpointing_controls(task: &TaskInvocation<'_>) -> Result<(), TaskFailure> {
99    refuse_controls(task, CONTROLS)
100}
101
102fn refuse_controls<const N: usize>(
103    task: &TaskInvocation<'_>,
104    names: [&str; N],
105) -> Result<(), TaskFailure> {
106    for name in names {
107        if !scalar_control(task, name)?.is_empty() {
108            return Err(TaskFailure::validation(format!(
109                "{name} requires a MeasurementSet Ant-Ra or Ant-Dec axis (#1032)"
110            ))
111            .at(name));
112        }
113    }
114    Ok(())
115}
116
117fn scalar_control<'a>(task: &'a TaskInvocation<'_>, name: &str) -> Result<&'a str, TaskFailure> {
118    match task.value(name) {
119        None => Ok(""),
120        Some(ParameterValue::String(text)) => Ok(text),
121        Some(ParameterValue::Array(values)) if values.is_empty() => Ok(""),
122        _ => Err(TaskFailure::validation(format!(
123            "list-valued PlotMS pointing control {name} is pending #1032"
124        ))
125        .at(name)),
126    }
127}
