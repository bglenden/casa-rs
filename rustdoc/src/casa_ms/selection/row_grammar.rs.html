20pub(super) fn apply(
21    ms: &MeasurementSet,
22    mut compiled: CompiledMsSelection,
23    options: &ListObsOptions,
24    flagdata: bool,
25) -> MsResult<CompiledMsSelection> {
26    if let Some(text) = options.spw.as_deref().filter(|text| !text.is_empty()) {
27        let spw = if flagdata {
28            ms.resolve_flagdata_spw_selector(text)?
29        } else {
30            ms.resolve_spw_selector(text)?
31        };
32        compiled.spw_data_desc_ids = spw
33            .row_data_descriptions()
34            .map(|ids| ids.iter().map(|&id| id as i32).collect());
35        compiled.spw_channel_selections = if flagdata {
36            for warning in spw.warnings() {
37                tracing::warn!(%warning, "SPW selection");
38            }
39            flagdata_channel_ranges(ms, &spw, false)?
40        } else {
41            selected_channel_ranges(ms, &spw)?
42        };
43    }
44    if let Some(text) = options.antenna.as_deref().filter(|text| !text.is_empty()) {
45        let baseline = if flagdata {
46            ms.resolve_antenna_selector_with_warnings(text)?
47        } else {
48            ms.resolve_antenna_selector(text)?
49        };
50        compiled.casa_taql_exprs.push(baseline_predicate(&baseline));
51        compiled.casa_baseline = Some(baseline);
52    }
53    for (column, label, text, target) in [
54        (
55            "SCAN_NUMBER",
56            "scan",
57            options.scan.as_deref(),
58            &mut compiled.casa_scan_ranges,
59        ),
60        (
61            "OBSERVATION_ID",
62            "observation",
63            options.observation.as_deref(),
64            &mut compiled.casa_observation_ranges,
65        ),
66    ] {
67        let text = if flagdata {
68            text.map(flagdata_c_string)
69        } else {
70            text
71        };
72        if let Some(text) = text.filter(|text| !text.is_empty()) {
73            let text = if flagdata {
74                flagdata_lexer_prefix(text)
75            } else {
76                text
77            };
78            let ranges = parse_numeric_row_id_ranges(text, label)?;
79            compiled
80                .casa_taql_exprs
81                .push(row_ids_predicate(column, &ranges));
82            *target = Some(ranges);
83        }
84    }
85    Ok(compiled)
86}

88pub(crate) fn selected_channel_ranges(
89    ms: &MeasurementSet,
90    spw: &SpwSelection,
91) -> MsResult<BTreeMap<i32, ChannelSelection>> {
92    channel_ranges(ms, spw, false)
93}
94
95fn channel_ranges(
96    ms: &MeasurementSet,
97    spw: &SpwSelection,
98    retain_full_segments: bool,
99) -> MsResult<BTreeMap<i32, ChannelSelection>> {
100    let windows = ms.spectral_window()?;
101    let mut full_windows = BTreeSet::new();
102    let mut selected = BTreeMap::<_, ChannelSelection>::new();
103    for range in spw.channel_list() {
104        let id = range.spw_id as i32;
105        if !retain_full_segments
106            && range.start == 0
107            && range.step <= 1
108            && range.stop == windows.num_chan(range.spw_id)? - 1
109        {
110            full_windows.insert(id);
111            selected.remove(&id);
112        } else if !full_windows.contains(&id) {
113            selected
114                .entry(id)
115                .or_insert_with(|| ChannelSelection {
116                    segments: Vec::new(),
117                })
118                .segments
119                .push(ChannelSelectionSegment {
120                    start: range.start,
121                    end: range.stop,
122                    stride: range.step.max(1) as usize,
123                });
124        }
125    }
126    Ok(selected)
127}

129/// Build the flag agent’s per-window channel ranges, retaining empty entries for excluded windows.
130pub fn flagdata_channel_ranges(
131    ms: &MeasurementSet,
132    spw: &SpwSelection,
133    retain_full_segments: bool,
134) -> MsResult<BTreeMap<i32, ChannelSelection>> {
135    let mut ranges = channel_ranges(ms, spw, retain_full_segments)?;
136    let named = spw
137        .channel_list()
138        .iter()
139        .map(|range| range.spw_id)
140        .collect::<BTreeSet<_>>();
141    // Keep windows with no matching channels distinct from named windows
142    // whose full channel axes were selected.
143    for window in 0..ms.spectral_window()?.row_count() {
144        if !named.contains(&window) {
145            ranges.insert(
146                window as i32,
147                ChannelSelection {
148                    segments: Vec::new(),
149                },
150            );
151        }
152    }
153    Ok(ranges)
154}

156/// Return the expression prefix available to CASA’s NUL-terminated flag-agent parser.
157pub fn flagdata_c_string(text: &str) -> &str {
158    text.split_once('\0').map_or(text, |(prefix, _)| prefix)
159}

161/// CASA's flagdata row lexers stop at the first non-ASCII byte.
162pub(super) fn flagdata_lexer_prefix(text: &str) -> &str {
163    text.split_once(|character: char| !character.is_ascii())
164        .map_or(text, |(prefix, _)| prefix)
165}

167pub(super) fn apply_array_feed_uv(
168    ms: &MeasurementSet,
169    mut compiled: CompiledMsSelection,
170    options: &ListObsOptions,
171    flagdata: bool,
172) -> MsResult<CompiledMsSelection> {
173    let array = options.array.as_deref();
174    let array = if flagdata {
175        array.map(flagdata_c_string)
176    } else {
177        array
178    };
179    if let Some(text) = array.filter(|text| !text.is_empty()) {
180        let ranges = if flagdata {
181            // Flagging preserves the stored ARRAY_IDs and needs only their row predicate.
182            parse_numeric_row_id_ranges(flagdata_lexer_prefix(text), "array")?
183        } else {
184            ArraySelection::parse(text)?.ranges
185        };
186        compiled
187            .casa_taql_exprs
188            .push(row_ids_predicate("ARRAY_ID", &ranges));
189        compiled.casa_array_ranges = Some(ranges);
190    }
191    apply_feed(ms, &mut compiled, options.feed.as_deref())?;
192    let uvrange = options.uvrange.as_deref();
193    let uvrange = if flagdata {
194        uvrange.map(flagdata_c_string)
195    } else {
196        uvrange
197    };
198    if let Some(text) = uvrange.filter(|text| !text.is_empty()) {
199        let text = if flagdata {
200            flagdata_lexer_prefix(text)
201        } else {
202            text
203        };
204        let selection = UvDistanceSelection::parse(ms, text)?;
205        compiled.casa_taql_exprs.push(selection.to_taql());
206        compiled.casa_uv = Some(selection);
207    }
208    Ok(compiled)
209}

211pub(super) fn apply_feed(
212    ms: &MeasurementSet,
213    selection: &mut CompiledMsSelection,
214    feed: Option<&str>,
215) -> MsResult<()> {
216    if let Some(text) = feed.filter(|text| !text.is_empty()) {
217        let feeds = FeedSelection::parse(ms, text)?;
218        selection.casa_taql_exprs.push(feeds.to_taql());
219        selection.casa_feed = Some(feeds);
220    }
221    Ok(())
222}

224fn row_ids_predicate(column: &str, ranges: &[(i32, i32)]) -> String {
225    disjunction(
226        ranges
227            .iter()
228            .map(|&(start, end)| {
229                if start == end {
230                    format!("{column}=={start}")
231                } else {
232                    format!("({column}>={start} AND {column}<={end})")
233                }
234            })
235            .collect(),
236    )
237}

239pub(super) fn with_time(
240    ms: &MeasurementSet,
241    selection: CompiledMsSelection,
242    text: &str,
243    rows_per_block: Option<usize>,
244    agent_time_default_row: Option<usize>,
245) -> MsResult<CompiledMsSelection> {
246    if text.is_empty() {
247        return Ok(selection);
248    }
249    let first = if let Some(original_first) = agent_time_default_row {
250        flagdata_agent_time_row(ms, &selection, rows_per_block, original_first)?
251    } else if selection.where_clauses().is_empty() {
252        first_unflagged_time_row(ms)?
253    } else {
254        selection.first_match(ms, rows_per_block)?.unwrap_or(0)
255    };
256    let mut time = resolve_time_selector(text, time_defaults(ms.main_table(), first)?)?;
257    // AgentFlagger tests the closed endpoints of getTimeList().
258    if agent_time_default_row.is_some() {
259        time.use_agent_intervals();
260    }
261    let terms = time.terms().iter().map(|term| match term.condition {
262        TimeCondition::Near { time_s, tolerance_s } =>
263            format!("ABS(TIME - {time_s}) <= {tolerance_s}"),
264        TimeCondition::AtOrAfter { time_s } => format!("TIME >= {time_s}"),
265        TimeCondition::AtOrBefore { time_s } => format!("TIME <= {time_s}"),
266        TimeCondition::Within { lower_s, upper_s } =>
267            format!("(TIME >= {lower_s} AND TIME <= {upper_s})"),
268        TimeCondition::EdgeInclusive { lower_s, upper_s, edge_s } => {
269            let edge = f64::from(edge_s);
270            format!("((TIME > {lower_s} OR ABS(TIME - {lower_s}) < {edge}) AND (TIME < {upper_s} OR ABS(TIME - {upper_s}) < {edge}))")
271        }
272    }).collect();
273    let mut selection = selection;
274    selection.casa_taql_exprs.push(disjunction(terms));
275    selection.casa_time = Some(time);
276    Ok(selection)
277}

279fn time_defaults(table: &Table, first: usize) -> MsResult<TimeSelectorDefaults> {
280    let rows = [first, 0];
281    let rows = if first == 0 { &rows[..1] } else { &rows[..] };
282    let float_column = |column: &str| {
283        table
284            .column_accessor(column)?
285            .scalar_cells_owned_for_rows(rows)?
286            .into_iter()
287            .map(|value| match value {
288                Some(ScalarValue::Float64(value)) => Ok(value),
289                _ => Err(MsError::InvalidInput(format!(
290                    "MAIN {column} must contain double-precision scalars"
291                ))),
292            })
293            .collect::<MsResult<Vec<_>>>()
294    };
295    let times = float_column("TIME")?;
296    let exposures = if has_column(table, "EXPOSURE") {
297        Some(float_column("EXPOSURE")?)
298    } else {
299        None
300    };
301    let at = |index| TimeDefaults {
302        time_mjd_s: times[index],
303        exposure_s: exposures.as_ref().map_or(0.1, |values| values[index]),
304    };
305    Ok(TimeSelectorDefaults {
306        first_term: at(0),
307        later_terms: at(rows.len() - 1),
308    })
309}

311/// Find the first MAIN row available for the flag agent’s default epoch.
312pub fn first_unflagged_time_row(ms: &MeasurementSet) -> MsResult<usize> {
313    let table = ms.main_table();
314    if !has_column(table, "FLAG_ROW") {
315        return Ok(0);
316    }
317    let budget = default_selection_budget()?;
318    let limit = rows_per_block_for_budget(budget)
319        .map_err(|error| MsError::InvalidInput(error.to_string()))?;
320    let flags = table.column_accessor("FLAG_ROW")?;
321    let mut start = 0_usize;
322    let mut block = 1_usize;
323    while start < table.row_count() {
324        let end = start.saturating_add(block).min(table.row_count());
325        let rows = (start..end).collect::<Vec<_>>();
326        for (row, flag) in rows
327            .iter()
328            .copied()
329            .zip(flags.scalar_cells_owned_for_rows(&rows)?)
330        {
331            match flag {
332                Some(ScalarValue::Bool(false)) => return Ok(row),
333                Some(ScalarValue::Bool(true)) => {}
334                _ => {
335                    return Err(MsError::InvalidInput(format!(
336                        "MAIN FLAG_ROW must contain a Boolean at row {row}"
337                    )));
338                }
339            }
340        }
341        start = end;
342        block = block.saturating_mul(2).min(limit);
343    }
344    Ok(0)
345}

347fn flagdata_agent_time_row(
348    ms: &MeasurementSet,
349    selection: &CompiledMsSelection,
350    rows_per_block: Option<usize>,
351    original_first: usize,
352) -> MsResult<usize> {
353    // FlagAgentBase adds array, field, and scan to MSSelection before parsing
354    // timerange. The remaining agent selectors are added afterwards.
355    if selection.field_ids.is_none()
356        && selection.casa_array_ranges.is_none()
357        && selection.casa_scan_ranges.is_none()
358    {
359        return Ok(original_first);
360    }
361    let prefix = CompiledMsSelection {
362        field_ids: selection.field_ids.clone(),
363        casa_array_ranges: selection.casa_array_ranges.clone(),
364        casa_scan_ranges: selection.casa_scan_ranges.clone(),
365        ..CompiledMsSelection::default()
366    };
367    Ok(prefix.first_match(ms, rows_per_block)?.unwrap_or(0))
368}

370fn disjunction(terms: Vec<String>) -> String {
371    if terms.is_empty() {
372        "FALSE".to_string()
373    } else {
374        terms.join(" OR ")
375    }
376}

378/// One compact condition per first antenna. The full grammar distinguishes
379/// equal and unequal out-of-range IDs, so keep those cases in the predicate.
380pub(super) fn baseline_predicate(selection: &BaselineSelection) -> String {
381    let count = selection.antenna_count();
382    let outside = |column| format!("({column}<0 OR {column}>={count})");
383    let mut terms = Vec::new();
384    for first in 0..=count {
385        let mut accepted = Vec::new();
386        let mut rejected = Vec::new();
387        for second in 0..count {
388            let ids = if selection.holds(first, second) {
389                &mut accepted
390            } else {
391                &mut rejected
392            };
393            ids.push(second as i32);
394        }
395        let mut second_terms = Vec::new();
396        if accepted.len() > rejected.len() {
397            let existing = format!("(ANTENNA2>=0 AND ANTENNA2<{count})");
398            second_terms.push(if rejected.is_empty() {
399                existing
400            } else {
401                format!(
402                    "({existing} AND NOT (ANTENNA2 IN [{}]))",
403                    int_list(&rejected)
404                )
405            });
406        } else if !accepted.is_empty() {
407            second_terms.push(format!("ANTENNA2 IN [{}]", int_list(&accepted)));
408        }
409
410        let same = first == count && selection.holds(first, count);
411        let different = selection.holds(first, count + 1);
412        match (same, different, first == count) {
413            (true, true, _) | (false, true, false) => second_terms.push(outside("ANTENNA2")),
414            (true, false, _) => second_terms.push("ANTENNA2==ANTENNA1".into()),
415            (false, true, true) => {
416                second_terms.push(format!("({} AND ANTENNA2!=ANTENNA1)", outside("ANTENNA2")));
417            }
418            _ => {}
419        }
420        if !second_terms.is_empty() {
421            let first_term = if first == count {
422                outside("ANTENNA1")
423            } else {
424                format!("ANTENNA1=={first}")
425            };
426            terms.push(format!(
427                "({first_term} AND ({}))",
428                disjunction(second_terms)
429            ));
430        }
431    }
432    disjunction(terms)
