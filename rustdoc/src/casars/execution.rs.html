13/// How long a cancelled child has to exit after SIGINT before it is
14/// killed: a cancelled imaging run stops within one block of a pass, or
15/// before its next phase (#677).
16const CANCEL_GRACE: Duration = Duration::from_secs(5);
17
18#[derive(Debug, Clone)]
19pub(crate) struct ExecutionPlan {
20    pub command: ResolvedCommand,
21    pub arguments: Vec<OsString>,
22    pub stdin: Option<String>,
23    pub working_directory: PathBuf,
24    pub renderer: Option<String>,
25    pub file_output_path: Option<String>,
26}

28#[derive(Debug)]
29pub(crate) enum ExecutionEvent {
30    Stdout(String),
31    Stderr(String),
32    Exited(ExecutionExit),
33}
34
35#[derive(Debug, Clone, Copy)]
36pub(crate) struct ExecutionExit {
37    pub code: Option<i32>,
38    pub success: bool,
39}

41#[derive(Debug)]
42pub(crate) struct RunningProcess {
43    receiver: Receiver<ExecutionEvent>,
44    child: Arc<Mutex<Child>>,
45}

47#[derive(Debug)]
48pub(crate) struct BlockingExecution {

54impl RunningProcess {
55    pub(crate) fn try_recv(&self) -> Result<ExecutionEvent, mpsc::TryRecvError> {
56        self.receiver.try_recv()
57    }
58
59    /// Stop the child: ask it with SIGINT, and kill it if it has not exited
60    /// within [`CANCEL_GRACE`]. `casars-imager` handles SIGINT by stopping
61    /// the run, removing its product staging and paged cube state and
62    /// exiting with status 130, none of which a kill allows (#677).
63    pub(crate) fn cancel(&self) -> Result<(), String> {
64        self.cancel_within(CANCEL_GRACE)
65    }
66
67    fn cancel_within(&self, grace: Duration) -> Result<(), String> {
68        let mut child = lock(&self.child)?;
69        if child
70            .try_wait()
71            .map_err(|error| format!("wait for child: {error}"))?
72            .is_some()
73        {
74            return Ok(());
75        }
76        if !interrupt(&child) {
77            return child
78                .kill()
79                .map_err(|error| format!("terminate child process: {error}"));
80        }
81        drop(child);
82        let child = Arc::clone(&self.child);
83        thread::spawn(move || {
84            let deadline = Instant::now() + grace;
85            loop {
86                let Ok(mut child) = lock(&child) else {
87                    return;
88                };
89                if !matches!(child.try_wait(), Ok(None)) {
90                    return;
91                }
92                if Instant::now() >= deadline {
93                    let _ = child.kill();
94                    return;
95                }
96                drop(child);
97                thread::sleep(Duration::from_millis(25));
98            }
99        });
100        Ok(())
101    }
102}

104fn lock(child: &Mutex<Child>) -> Result<std::sync::MutexGuard<'_, Child>, String> {
105    child
106        .lock()
107        .map_err(|_| "failed to acquire child-process lock".to_string())
108}
109
110/// Send SIGINT to `child`, which the caller holds unreaped, so its process
111/// id is still its own; whether the signal was sent.
112#[cfg(unix)]
113fn interrupt(child: &Child) -> bool {
114    let Ok(pid) = libc::pid_t::try_from(child.id()) else {
115        return false;
116    };
117    // SAFETY: `kill` only sends a signal; the process id names our own
118    // child, which cannot be reaped while the caller holds its lock.
119    unsafe { libc::kill(pid, libc::SIGINT) == 0 }
120}
121
122#[cfg(not(unix))]
123fn interrupt(_child: &Child) -> bool {
124    false
125}

127pub(crate) fn spawn_process(plan: &ExecutionPlan) -> Result<RunningProcess, String> {
128    let mut command = plan.command.command();
129    command
130        .args(&plan.arguments)
131        .current_dir(&plan.working_directory)
132        .stdin(if plan.stdin.is_some() {
133            Stdio::piped()
134        } else {
135            Stdio::null()
136        })
137        .stdout(Stdio::piped())
138        .stderr(Stdio::piped());
139
140    let mut child = command
141        .spawn()
142        .map_err(|error| format!("spawn subprocess: {error}"))?;
143    if let Some(payload) = plan.stdin.as_deref() {
144        child
145            .stdin
146            .take()
147            .ok_or_else(|| "subprocess stdin was not captured".to_string())?
148            .write_all(payload.as_bytes())
149            .map_err(|error| format!("write subprocess stdin: {error}"))?;
150    }
151    let stdout = child
152        .stdout
153        .take()
154        .ok_or_else(|| "subprocess stdout was not captured".to_string())?;
155    let stderr = child
156        .stderr
157        .take()
158        .ok_or_else(|| "subprocess stderr was not captured".to_string())?;
159
160    let child = Arc::new(Mutex::new(child));
161    let (tx, rx) = mpsc::channel();
162
163    let stdout_tx = tx.clone();
164    let stdout_handle = thread::spawn(move || {
165        read_stream(stdout, move |chunk| {
166            let _ = stdout_tx.send(ExecutionEvent::Stdout(chunk));
167        })
168    });
169
170    let stderr_tx = tx.clone();
171    let stderr_handle = thread::spawn(move || {
172        read_stream(stderr, move |chunk| {
173            let _ = stderr_tx.send(ExecutionEvent::Stderr(chunk));
174        })
175    });
176
177    let child_for_wait = Arc::clone(&child);
178    thread::spawn(move || {
179        let exit_status = wait_for_child_exit(&child_for_wait);
180        let _ = stdout_handle.join();
181        let _ = stderr_handle.join();
182        let exit = match exit_status {
183            Ok(status) => exit_from_status(status),
184            Err(_error) => ExecutionExit {
185                code: None,
186                success: false,
187            },
188        };
189        let _ = tx.send(ExecutionEvent::Exited(exit));
190    });
191
192    Ok(RunningProcess {
193        receiver: rx,
194        child,
195    })
196}

198pub(crate) fn run_process_blocking(plan: &ExecutionPlan) -> Result<BlockingExecution, String> {
199    let process = spawn_process(plan)?;
200    let mut stdout = String::new();
201    let mut stderr = String::new();
202    loop {
203        match process
204            .receiver
205            .recv()
206            .map_err(|error| format!("receive subprocess output: {error}"))?
207        {
208            ExecutionEvent::Stdout(chunk) => {
209                print!("{chunk}");
210                std::io::stdout()
211                    .flush()
212                    .map_err(|error| format!("flush subprocess stdout: {error}"))?;
213                stdout.push_str(&chunk);
214            }
215            ExecutionEvent::Stderr(chunk) => {
216                eprint!("{chunk}");
217                std::io::stderr()
218                    .flush()
219                    .map_err(|error| format!("flush subprocess stderr: {error}"))?;
220                stderr.push_str(&chunk);
221            }
222            ExecutionEvent::Exited(exit) => {
223                return Ok(BlockingExecution {
224                    exit,
225                    stdout,
226                    stderr,
227                });
228            }
229        }
230    }
231}

233fn wait_for_child_exit(child: &Arc<Mutex<Child>>) -> Result<ExitStatus, String> {
234    loop {
235        let status = lock(child)?
236            .try_wait()
237            .map_err(|error| format!("wait for child: {error}"))?;
238        if let Some(status) = status {
239            return Ok(status);
240        }
241        thread::sleep(Duration::from_millis(25));
242    }
243}
244
245fn read_stream<R>(stream: R, mut send: impl FnMut(String))
246where
247    R: Read,
248{
249    let mut reader = BufReader::new(stream);
250    let mut line = String::new();
251    loop {
252        line.clear();
253        match reader.read_line(&mut line) {
254            Ok(0) => break,
255            Ok(_) => send(line.clone()),
256            Err(_) => break,
257        }
258    }
259}
260
261fn exit_from_status(status: ExitStatus) -> ExecutionExit {
262    ExecutionExit {
263        code: status.code(),
264        success: status.success(),
265    }
266}

344    /// Spawn `script` under `sh` and wait for it to print `ready`, which it
345    /// does once its signal trap is set.
346    fn ready(script: &str) -> RunningProcess {
347        let plan = ExecutionPlan {
348            command: ResolvedCommand::direct("sh"),
349            arguments: vec!["-c".into(), script.into()],
350            stdin: None,
351            working_directory: PathBuf::from("."),
352            renderer: None,
353            file_output_path: None,
354        };
355        let process = spawn_process(&plan).expect("spawn process");
356        let deadline = Instant::now() + Duration::from_secs(10);
357        while Instant::now() < deadline {
358            match process.try_recv() {
359                Ok(ExecutionEvent::Stdout(chunk)) if chunk.contains("ready") => return process,
360                Ok(_) | Err(mpsc::TryRecvError::Empty) => thread::sleep(Duration::from_millis(10)),
361                Err(error) => panic!("unexpected channel error: {error}"),
362            }
363        }
364        panic!("the child never got ready");
365    }

367    /// The child's output after `ready` and its exit, within `timeout`.
368    fn finish(process: &RunningProcess, timeout: Duration) -> (String, ExecutionExit) {
369        let mut stdout = String::new();
370        let deadline = Instant::now() + timeout;
371        while Instant::now() < deadline {
372            match process.try_recv() {
373                Ok(ExecutionEvent::Stdout(chunk)) => stdout.push_str(&chunk),
374                Ok(ExecutionEvent::Exited(exit)) => return (stdout, exit),
375                Ok(ExecutionEvent::Stderr(_)) | Err(mpsc::TryRecvError::Empty) => {
376                    thread::sleep(Duration::from_millis(10));
377                }
378                Err(error) => panic!("unexpected channel error: {error}"),
379            }
380        }
381        panic!("the child did not exit within {timeout:?}");
382    }

384    /// #677: cancelling interrupts the child, which exits through its own
385    /// SIGINT handler (as `casars-imager` does, with status 130) rather
386    /// than being killed.
387    #[cfg(unix)]
388    #[test]
389    fn cancel_interrupts_the_child_before_killing_it() {
390        let process =
391            ready("trap 'echo stopped; exit 130' INT; echo ready; while :; do sleep 0.05; done");
392        let started = Instant::now();
393        process.cancel().expect("cancel process");
394        let (stdout, exit) = finish(&process, Duration::from_secs(10));
395        assert!(stdout.contains("stopped"), "the handler ran: {stdout:?}");
396        assert_eq!(exit.code, Some(130));
397        assert!(
398            started.elapsed() < CANCEL_GRACE,
399            "it stopped itself before the kill"
400        );
401    }
