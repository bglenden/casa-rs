2//! A start model's seeding outside a run, for the counting law in
3//! `tests/start_model_memory.rs` (#841 Phase 6b): what the initial pass
4//! admits for it, and the samples it then seeds the model with.
5
6use std::path::Path;
7
8use casa_images::RegridTemplate;
9
10use crate::compile::StartModel;
11
12/// The bytes the initial pass admits to seed a model of `template` from the
13/// start model at `path`.
14///
15/// # Errors
16///
17/// A start model the run would refuse.
18pub fn seeding_bytes(path: &Path, template: RegridTemplate) -> Result<u64, String> {
19    StartModel::onto(path, template)
20        .map(|start| start.seeding_bytes())
21        .map_err(|error| error.to_string())
22}
23
24/// The main domain's samples the initial pass seeds the model of `template`
25/// with from the start model at `path`, in the model's canonical order.
26///
27/// # Errors
28///
29/// A start model the run would refuse or cannot read.
30pub fn main_samples(path: &Path, template: RegridTemplate) -> Result<Vec<f32>, String> {
31    StartModel::onto(path, template)
32        .map_err(|error| error.to_string())?
33        .main_samples()
34        .map_err(|error| error.to_string())
35}
