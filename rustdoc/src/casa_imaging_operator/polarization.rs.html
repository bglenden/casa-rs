12/// Feed basis of the selected correlations.
13#[derive(Clone, Copy, Debug, PartialEq, Eq)]
14pub enum FeedBasis {
15    /// Values are already Stokes parameters.
16    Stokes,
17    /// Orthogonal X/Y receptors.
18    Linear,
19    /// Orthogonal R/L receptors.
20    Circular,
21}
22
23/// One grid polarization plane in CASA's complex-image representation.
24#[derive(Clone, Copy, Debug, PartialEq, Eq)]
25pub enum GridPolarization {
26    /// One correlation (or one Stokes parameter when the data are Stokes).
27    Correlation(Correlation),
28    /// Stokes I gridded directly: both parallel hands feed one plane and
29    /// their weights both count in its `sumwt` (CASA `isIOnly`).
30    StokesI,
31}
32
33/// Correlation routing in both directions of the operator.
34///
35/// The adjoint grids each selected correlation into the grid plane
36/// [`PolarizationRouting::pol_map`] names and converts the complex image
37/// planes to the requested coordinates with
38/// [`PolarizationRouting::to_requested`] (CASA `StokesImageUtil::To`,
39/// real part taken). The forward transform expands requested model planes
40/// to grid planes with [`PolarizationRouting::from_requested`]
41/// (`StokesImageUtil::From`). A requested plane's `sumwt` follows
42/// `StokesImageUtil::ToStokesSumWt` ([`SumwtSource`]). A grid plane no
43/// selected correlation feeds stays empty, as CASA images Stokes Q and U of
44/// parallel-hand data: zero planes with zero `sumwt`.
45#[derive(Clone, Debug, PartialEq)]
46pub struct PolarizationRouting {
47    correlations: Vec<Correlation>,
48    requested: Vec<PolarizationCoordinate>,
49    grid: Vec<GridPolarization>,
50    pol_map: Vec<Option<u8>>,
51    feed: FeedBasis,
52    to_requested: Vec<Complex64>,
53    to_requested_psf: Vec<Complex64>,
54    from_requested: Vec<Complex64>,
55    sumwt_source: Vec<SumwtSource>,
56}
57
58/// The grid planes whose weight sums give a requested plane's `sumwt`, as
59/// `StokesImageUtil::ToStokesSumWt` (with coordinates) forms it.
60#[derive(Clone, Copy, Debug, PartialEq, Eq)]
61pub enum SumwtSource {
62    /// One grid plane's.
63    Plane(usize),
64    /// The smaller of two grid planes'.
65    Min(usize, usize),
66}
67
68impl SumwtSource {
69    /// This source's `sumwt` from each grid plane's, `plane`.
70    pub fn sumwt(self, plane: impl Fn(usize) -> f64) -> f64 {
71        match self {
72            Self::Plane(gpol) => plane(gpol),
73            Self::Min(first, second) => plane(first).min(plane(second)),
74        }
75    }
76}
77
78impl PolarizationRouting {
79    /// Route the selected `correlations` (block order) to the `requested`
80    /// image coordinates.
81    ///
82    /// Requested coordinates are all Stokes or all correlations, the
83    /// selected correlations share one feed basis, every grid plane CASA's
84    /// representation needs receives at least one selected correlation, and
85    /// requested correlations must be selected.
86    pub fn compile(
87        correlations: &[Correlation],
88        requested: &[PolarizationCoordinate],
89    ) -> Result<Self, OperatorError> {
90        let feed = feed_basis(correlations)?;
91        if requested.is_empty() {
92            return Err(OperatorError::Polarization {
93                reason: "no polarization coordinate requested",
94            });
95        }
96        if requested
97            .iter()
98            .enumerate()
99            .any(|(index, coordinate)| requested[..index].contains(coordinate))
100        {
101            return Err(OperatorError::Polarization {
102                reason: "requested polarization coordinates repeat",
103            });
104        }
105        let stokes_requested = requested.iter().all(|coordinate| is_stokes(*coordinate));
106        if !stokes_requested && requested.iter().any(|coordinate| is_stokes(*coordinate)) {
107            return Err(OperatorError::Polarization {
108                reason: "requested coordinates mix Stokes parameters and correlations",
109            });
110        }
111        let grid = if !stokes_requested {
112            requested
113                .iter()
114                .map(|coordinate| {
115                    let correlation = correlation_of(*coordinate);
116                    correlations
117                        .contains(&correlation)
118                        .then_some(GridPolarization::Correlation(correlation))
119                        .ok_or(OperatorError::Polarization {
120                            reason: "requested correlation is not selected",
121                        })
122                })
123                .collect::<Result<Vec<_>, _>>()?
124        } else if feed == FeedBasis::Stokes {
125            requested
126                .iter()
127                .map(|coordinate| {
128                    let correlation = correlation_of(*coordinate);
129                    correlations
130                        .contains(&correlation)
131                        .then_some(GridPolarization::Correlation(correlation))
132                        .ok_or(OperatorError::Polarization {
133                            reason: "requested Stokes parameter is not selected",
134                        })
135                })
136                .collect::<Result<Vec<_>, _>>()?
137        } else {
138            casa_correlation_representation(feed, requested)
139        };
140        let pol_map = correlations
141            .iter()
142            .map(|correlation| {
143                grid.iter()
144                    .position(|plane| match plane {
145                        GridPolarization::Correlation(target) => target == correlation,
146                        GridPolarization::StokesI => correlation.is_parallel_hand(),
147                    })
148                    .map(|index| index as u8)
149            })
150            .collect::<Vec<_>>();
151        if !(0..grid.len()).any(|gpol| pol_map.contains(&Some(gpol as u8))) {
152            return Err(OperatorError::Polarization {
153                reason: "selected correlations feed no grid polarization",
154            });
155        }
156        let to_requested = requested
157            .iter()
158            .flat_map(|coordinate| {
159                grid.iter()
160                    .map(|plane| to_coefficient(feed, *coordinate, *plane))
161            })
162            .collect();
163        let to_requested_psf = requested
164            .iter()
165            .flat_map(|coordinate| {
166                grid.iter()
167                    .map(|plane| psf_coefficient(feed, requested, *coordinate, *plane))
168            })
169            .collect();
170        let from_requested = grid
171            .iter()
172            .flat_map(|plane| {
173                requested
174                    .iter()
175                    .map(|coordinate| from_coefficient(feed, *plane, *coordinate))
176            })
177            .collect();
178        let sumwt_source = requested
179            .iter()
180            .enumerate()
181            .map(|(index, coordinate)| {
182                if stokes_requested {
183                    stokes_sumwt_source(feed, requested, index, grid.len())
184                } else {
185                    SumwtSource::Plane(
186                        grid.iter()
187                            .position(|plane| {
188                                *plane == GridPolarization::Correlation(correlation_of(*coordinate))
189                            })
190                            .expect("requested correlations are grid planes"),
191                    )
192                }
193            })
194            .collect();
195        Ok(Self {
196            correlations: correlations.to_vec(),
197            requested: requested.to_vec(),
198            grid,
199            pol_map,
200            feed,
201            to_requested,
202            to_requested_psf,
203            from_requested,
204            sumwt_source,
205        })
206    }

208    /// Selected correlations in block order.
209    #[must_use]
210    pub fn correlations(&self) -> &[Correlation] {
211        &self.correlations
212    }
213
214    /// Requested image coordinates in output order.
215    #[must_use]
216    pub fn requested(&self) -> &[PolarizationCoordinate] {
217        &self.requested
218    }
219
220    /// Grid polarization planes.
221    #[must_use]
222    pub fn grid(&self) -> &[GridPolarization] {
223        &self.grid
224    }
225
226    /// Number of grid polarization planes.
227    #[must_use]
228    pub fn grid_pols(&self) -> usize {
229        self.grid.len()
230    }
231
232    /// Grid plane fed by each selected correlation (`None`: not gridded).
233    #[must_use]
234    pub fn pol_map(&self) -> &[Option<u8>] {
235        &self.pol_map
236    }
237
238    /// Feed basis of the selected correlations.
239    #[must_use]
240    pub const fn feed(&self) -> FeedBasis {
241        self.feed
242    }
243
244    /// Coefficient of grid plane `gpol` in requested plane `requested`; the
245    /// requested image is the real part of the sum.
246    #[must_use]
247    pub fn to_requested(&self, requested: usize, gpol: usize) -> Complex64 {
248        self.to_requested[requested * self.grid.len() + gpol]
249    }
250
251    /// Coefficient of grid plane `gpol` in the PSF image of requested plane
252    /// `requested` (CASA `StokesImageUtil::ToStokesPSF`): with one or two
253    /// requested Stokes parameters every plane takes the parallel-hand sum,
254    /// or the cross-hand sum for U and V on linear feeds and Q and U on
255    /// circular feeds; with three or four, every plane takes the first
256    /// parameter's conversion; a single-correlation grid, a Stokes grid or a
257    /// request with a non-Stokes coordinate takes the data conversion.
258    ///
259    /// The operator forms its weight image with these coefficients too,
260    /// whereas CASA applies `ToStokesPSF` to the PSF only; no kernel set
261    /// with a weight image is installed before the AW and mosaic operators.
262    #[must_use]
263    pub fn to_requested_psf(&self, requested: usize, gpol: usize) -> Complex64 {
264        self.to_requested_psf[requested * self.grid.len() + gpol]
265    }
266
267    /// Coefficient of requested model plane `requested` in grid plane `gpol`.
268    #[must_use]
269    pub fn from_requested(&self, gpol: usize, requested: usize) -> Complex64 {
270        self.from_requested[gpol * self.requested.len() + requested]
271    }
272
273    /// The grid planes whose weight sums give requested plane `requested`'s
274    /// `sumwt`.
275    #[must_use]
276    pub fn sumwt_source(&self, requested: usize) -> SumwtSource {
277        self.sumwt_source[requested]
278    }
279}

281/// `StokesImageUtil::ToStokesSumWt` for requested Stokes plane `index` of
282/// `requested` over `grid_planes` grid planes, on physical feeds: one Stokes
283/// parameter from two planes takes their smaller weight sum; two take
284/// `min(c0, c1)` and `c1` (CASA's own pairing); four take
285/// `min(RR, LL)` for I and V and `min(RL, LR)` for Q and U on circular feeds,
286/// `min(XX, YY)` for I and Q and `min(XY, YX)` for U and V on linear ones.
287/// One grid plane, data already in Stokes, and the combinations CASA's
288/// function leaves unset take grid plane 0's.
289fn stokes_sumwt_source(
290    feed: FeedBasis,
291    requested: &[PolarizationCoordinate],
292    index: usize,
293    grid_planes: usize,
294) -> SumwtSource {
295    use PolarizationCoordinate::{StokesI, StokesQ, StokesV};
296    if feed == FeedBasis::Stokes {
297        return SumwtSource::Plane(0);
298    }
299    match (requested.len(), grid_planes) {
300        (1, 2) => SumwtSource::Min(0, 1),
301        (2, 2) if index == 0 => SumwtSource::Min(0, 1),
302        (2, 2) => SumwtSource::Plane(1),
303        (4, 4) => match (feed, requested[index]) {
304            (FeedBasis::Circular, StokesI | StokesV) | (FeedBasis::Linear, StokesI | StokesQ) => {
305                SumwtSource::Min(0, 3)
306            }
307            _ => SumwtSource::Min(1, 2),
308        },
309        _ => SumwtSource::Plane(0),
310    }
311}

313fn feed_basis(correlations: &[Correlation]) -> Result<FeedBasis, OperatorError> {
314    let Some(first) = correlations.first() else {
315        return Err(OperatorError::Polarization {
316            reason: "no correlation selected",
317        });
318    };
319    let feed = basis_of(*first)?;
320    for (index, correlation) in correlations.iter().enumerate() {
321        if basis_of(*correlation)? != feed {
322            return Err(OperatorError::Polarization {
323                reason: "selected correlations mix feed bases",
324            });
325        }
326        if correlations[..index].contains(correlation) {
327            return Err(OperatorError::Polarization {
328                reason: "selected correlations repeat",
329            });
330        }
331    }
332    Ok(feed)
333}
334
335fn basis_of(correlation: Correlation) -> Result<FeedBasis, OperatorError> {
336    match correlation {
337        Correlation::I | Correlation::Q | Correlation::U | Correlation::V => Ok(FeedBasis::Stokes),
338        Correlation::XX | Correlation::XY | Correlation::YX | Correlation::YY => {
339            Ok(FeedBasis::Linear)
340        }
341        Correlation::RR | Correlation::RL | Correlation::LR | Correlation::LL => {
342            Ok(FeedBasis::Circular)
343        }
344        _ => Err(OperatorError::Polarization {
345            reason: "selected correlation is not a Stokes, linear or circular product",
346        }),
347    }
348}

350const fn is_stokes(coordinate: PolarizationCoordinate) -> bool {
351    matches!(
352        coordinate,
353        PolarizationCoordinate::StokesI
354            | PolarizationCoordinate::StokesQ
355            | PolarizationCoordinate::StokesU
356            | PolarizationCoordinate::StokesV
357    )
358}
359
360const fn correlation_of(coordinate: PolarizationCoordinate) -> Correlation {
361    match coordinate {
362        PolarizationCoordinate::StokesI => Correlation::I,
363        PolarizationCoordinate::StokesQ => Correlation::Q,
364        PolarizationCoordinate::StokesU => Correlation::U,
365        PolarizationCoordinate::StokesV => Correlation::V,
366        PolarizationCoordinate::LinearXx => Correlation::XX,
367        PolarizationCoordinate::LinearXy => Correlation::XY,
368        PolarizationCoordinate::LinearYx => Correlation::YX,
369        PolarizationCoordinate::LinearYy => Correlation::YY,
370        PolarizationCoordinate::CircularRr => Correlation::RR,
371        PolarizationCoordinate::CircularRl => Correlation::RL,
372        PolarizationCoordinate::CircularLr => Correlation::LR,
373        PolarizationCoordinate::CircularLl => Correlation::LL,
374    }
375}

377/// CASA `StokesImageUtil::changeLabelsStokesToCorrStokes`: the complex
378/// grid planes that serve a Stokes request on linear or circular feeds.
379fn casa_correlation_representation(
380    feed: FeedBasis,
381    requested: &[PolarizationCoordinate],
382) -> Vec<GridPolarization> {
383    use Correlation::{LL, LR, RL, RR, XX, XY, YX, YY};
384    use PolarizationCoordinate::{StokesI, StokesQ, StokesU, StokesV};
385    let planes = |list: &[Correlation]| {
386        list.iter()
387            .map(|correlation| GridPolarization::Correlation(*correlation))
388            .collect::<Vec<_>>()
389    };
390    let all = match feed {
391        FeedBasis::Linear => planes(&[XX, XY, YX, YY]),
392        FeedBasis::Circular | FeedBasis::Stokes => planes(&[RR, RL, LR, LL]),
393    };
394    match (feed, requested) {
395        (_, [StokesI]) => vec![GridPolarization::StokesI],
396        (FeedBasis::Circular, [StokesQ] | [StokesU] | [StokesQ, StokesU]) => planes(&[RL, LR]),
397        (FeedBasis::Circular, [StokesV] | [StokesI, StokesV]) => planes(&[RR, LL]),
398        (FeedBasis::Linear, [StokesQ] | [StokesI, StokesQ]) => planes(&[XX, YY]),
399        (FeedBasis::Linear, [StokesU] | [StokesV] | [StokesU, StokesV]) => planes(&[XY, YX]),
400        _ => all,
401    }
402}

404/// `StokesImageUtil::To` (`CStokesVector::applySlinInv` / `applyScircInv`):
405/// the coefficient of one grid plane in one requested plane.
406fn to_coefficient(
407    feed: FeedBasis,
408    requested: PolarizationCoordinate,
409    plane: GridPolarization,
410) -> Complex64 {
411    use Correlation::{LL, LR, RL, RR, XX, XY, YX, YY};
412    use PolarizationCoordinate::{StokesI, StokesQ, StokesU, StokesV};
413    let one = Complex64::new(1.0, 0.0);
414    let half = Complex64::new(0.5, 0.0);
415    let half_i = Complex64::new(0.0, 0.5);
416    let correlation = match plane {
417        GridPolarization::StokesI => {
418            return if requested == StokesI {
419                one
420            } else {
421                Complex64::default()
422            };
423        }
424        GridPolarization::Correlation(correlation) => correlation,
425    };
426    if !is_stokes(requested) || feed == FeedBasis::Stokes {
427        return if correlation_of(requested) == correlation {
428            one
429        } else {
430            Complex64::default()
431        };
432    }
433    match (feed, requested, correlation) {
434        (FeedBasis::Linear, StokesI, XX | YY)
435        | (FeedBasis::Linear, StokesQ, XX)
436        | (FeedBasis::Linear, StokesU, XY | YX)
437        | (FeedBasis::Circular, StokesI, RR | LL)
438        | (FeedBasis::Circular, StokesQ, RL | LR)
439        | (FeedBasis::Circular, StokesV, RR) => half,
440        (FeedBasis::Linear, StokesQ, YY) | (FeedBasis::Circular, StokesV, LL) => -half,
441        (FeedBasis::Linear, StokesV, YX) | (FeedBasis::Circular, StokesU, LR) => half_i,
442        (FeedBasis::Linear, StokesV, XY) | (FeedBasis::Circular, StokesU, RL) => -half_i,
443        _ => Complex64::default(),
444    }
445}

447/// `StokesImageUtil::ToStokesPSF` on physical feeds: with one or two
448/// requested Stokes parameters, every PSF plane comes from one Stokes row
449/// of the data conversion (CASA applies the cross-hand rule to both planes
450/// whenever either requested parameter is a cross-hand one); with more, every
451/// plane is the first requested parameter's PSF (`sv(outMap(0))`), since the
452/// grid holds more than one correlation.
453fn psf_coefficient(
454    feed: FeedBasis,
455    requested_all: &[PolarizationCoordinate],
456    requested: PolarizationCoordinate,
457    plane: GridPolarization,
458) -> Complex64 {
459    use PolarizationCoordinate::{StokesI, StokesQ, StokesU, StokesV};
460    if feed == FeedBasis::Stokes
461        || plane == GridPolarization::StokesI
462        || !requested_all
463            .iter()
464            .all(|coordinate| is_stokes(*coordinate))
465    {
466        return to_coefficient(feed, requested, plane);
467    }
468    if requested_all.len() > 2 {
469        return to_coefficient(feed, requested_all[0], plane);
470    }
471    let cross_hand = |coordinate: &PolarizationCoordinate| match feed {
472        FeedBasis::Linear => matches!(coordinate, StokesU | StokesV),
473        FeedBasis::Circular | FeedBasis::Stokes => matches!(coordinate, StokesQ | StokesU),
474    };
475    let row = if requested_all.iter().any(cross_hand) {
476        match feed {
477            FeedBasis::Linear => StokesU,
478            FeedBasis::Circular | FeedBasis::Stokes => StokesQ,
479        }
480    } else {
481        StokesI
482    };
483    to_coefficient(feed, row, plane)
484}

486/// `StokesImageUtil::From` (`CStokesVector::applySlin` / `applyScirc`): the
487/// coefficient of one requested model plane in one grid plane.
488fn from_coefficient(
489    feed: FeedBasis,
490    plane: GridPolarization,
491    requested: PolarizationCoordinate,
492) -> Complex64 {
493    use Correlation::{LL, LR, RL, RR, XX, XY, YX, YY};
494    use PolarizationCoordinate::{StokesI, StokesQ, StokesU, StokesV};
495    let one = Complex64::new(1.0, 0.0);
496    let i = Complex64::new(0.0, 1.0);
497    let correlation = match plane {
498        GridPolarization::StokesI => {
499            return if requested == StokesI {
500                one
501            } else {
502                Complex64::default()
503            };
504        }
505        GridPolarization::Correlation(correlation) => correlation,
506    };
507    if !is_stokes(requested) || feed == FeedBasis::Stokes {
508        return if correlation_of(requested) == correlation {
509            one
510        } else {
511            Complex64::default()
512        };
513    }
514    match (feed, correlation, requested) {
515        (FeedBasis::Linear, XX, StokesI | StokesQ)
516        | (FeedBasis::Linear, YY, StokesI)
517        | (FeedBasis::Linear, XY | YX, StokesU)
518        | (FeedBasis::Circular, RR, StokesI | StokesV)
519        | (FeedBasis::Circular, LL, StokesI)
520        | (FeedBasis::Circular, RL | LR, StokesQ) => one,
521        (FeedBasis::Linear, YY, StokesQ) | (FeedBasis::Circular, LL, StokesV) => -one,
522        (FeedBasis::Linear, XY, StokesV) | (FeedBasis::Circular, RL, StokesU) => i,
523        (FeedBasis::Linear, YX, StokesV) | (FeedBasis::Circular, LR, StokesU) => -i,
524        _ => Complex64::default(),
525    }
