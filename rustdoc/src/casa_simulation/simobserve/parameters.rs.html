6/// CASA's `simobserve` parameters, as `casatasks.simobserve` takes them.
7///
8/// The defaults are CASA 6.7.6's. `compwidth`'s is CASA's literal
9/// `'"8GHz"'`, quotes included, which is not a quantity: a component-only
10/// simulation with the default therefore observes 2 GHz, as in CASA.
11#[derive(Debug, Clone, PartialEq)]
12pub struct SimobserveParameters {
13    /// The project directory, and the first part of every file's name
14    /// (`project`). CASA takes a name in the working directory; casa-rs
15    /// also takes a path, and names the files by its last component.
16    pub project: String,
17    /// The model image, a CASA image or FITS file (`skymodel`); `$project`
18    /// is replaced by `<project>.<configuration>`.
19    pub skymodel: String,
20    /// The peak brightness the model is scaled to (`inbright`).
21    pub inbright: String,
22    /// The direction of the model's centre (`indirection`).
23    pub indirection: String,
24    /// The model's cell (`incell`).
25    pub incell: String,
26    /// The frequency of the model's centre channel (`incenter`).
27    pub incenter: String,
28    /// The model's channel width (`inwidth`).
29    pub inwidth: String,
30    /// A component list, observed with the model or alone (`complist`).
31    pub complist: String,
32    /// The bandwidth of a component-only simulation (`compwidth`).
33    pub compwidth: String,
34    /// The channels of a component-only simulation (`comp_nchan`).
35    pub comp_nchan: i64,
36    /// Lay out the pointings, rather than read them from `ptgfile`
37    /// (`setpointings`).
38    pub setpointings: bool,
39    /// The pointing file read when `setpointings` is false (`ptgfile`).
40    pub ptgfile: String,
41    /// The integration time (`integration`).
42    pub integration: String,
43    /// The map's centre, or the pointings themselves when there are several
44    /// (`direction`); the model's centre when empty. One direction lays a
45    /// map out ([`Directions::One`]); several are the pointings
46    /// ([`Directions::List`]). A list of one, which CASA centres at its
47    /// mean rounded at precision 5 ([`Directions::split`]), is taken as its
48    /// direction: the task contract resolves a string to a list of one
49    /// (`docs/CASA (C++) bugs.md`).
50    pub direction: Directions,
51    /// The map's size (`mapsize`); the model's when empty.
52    pub mapsize: Vec<String>,
53    /// The map's layout: `hexagonal`, `square`, `ALMA` or `ALMA2012`
54    /// (`maptype`).
55    pub maptype: String,
56    /// The pointings' spacing, an angle, `<n>PB` or `nyquist`
57    /// (`pointingspacing`); λ/D/√3 for interferometers when empty.
58    pub pointingspacing: String,
59    /// A phase calibrator's direction (`caldirection`).
60    pub caldirection: String,
61    /// The calibrator's flux (`calflux`).
62    pub calflux: String,
63    /// `int` or `sd` to observe, `''` to corrupt an existing MeasurementSet
64    /// (`obsmode`).
65    pub obsmode: String,
66    /// The date of the observation (`refdate`).
67    pub refdate: String,
68    /// The hour angle at the middle of the observation, or `transit`
69    /// (`hourangle`).
70    pub hourangle: String,
71    /// The observation's length, or a count of passes over the map with no
72    /// unit (`totaltime`).
73    pub totaltime: String,
74    /// The interferometer's antenna configuration (`antennalist`).
75    pub antennalist: String,
76    /// The single dish's antenna configuration (`sdantlist`).
77    pub sdantlist: String,
78    /// The antenna of `sdantlist` that observes (`sdant`).
79    pub sdant: i64,
80    /// The spectral window's frequency frame (`outframe`).
81    pub outframe: String,
82    /// `tsys-atm`, `tsys-manual` or `''` (`thermalnoise`).
83    pub thermalnoise: String,
84    /// The precipitable water vapour, mm (`user_pwv`).
85    pub user_pwv: f64,
86    /// The ground's temperature, K (`t_ground`).
87    pub t_ground: f64,
88    /// The atmosphere's temperature, K (`t_sky`), for `tsys-manual`.
89    pub t_sky: f64,
90    /// The zenith opacity (`tau0`), for `tsys-manual`.
91    pub tau0: f64,
92    /// The noise's random seed (`seed`).
93    pub seed: i64,
94    /// The polarization leakage's amplitude (`leakage`).
95    pub leakage: f64,
96    /// ALMA's correlator, for the correlator efficiency (`correlator`).
97    pub correlator: String,
98    /// `both`, `screen`, `file` or `none` (`graphics`).
99    pub graphics: String,
100    /// Log more, and apply the shadow and elevation limits (`verbose`).
101    pub verbose: bool,
102    /// Replace the project's files (`overwrite`). CASA's default, true,
103    /// replaces them without asking: the task writes only inside the
104    /// project directory.
105    pub overwrite: bool,
106}
