2//! The source's counting law (#932, step 4): the observation's batches, as
3//! the passes read them, hold at most what their admission charges them. A
4//! whole run's counting law (`tests/admission_memory.rs`) cannot see the
5//! source's charge beside the rest of the run's, so this reads the source
6//! alone, through the traversals a run makes, on a route opened and
7//! admitted as a run admits it. A request that writes visibilities back
8//! also drives its write (step 5): the source and the sink of a writing
9//! pass hold at most the route's charge and the write's.
10

19/// What [`drive_batch_source`] admitted and read.
20#[derive(Clone, Copy, Debug, PartialEq, Eq)]
21pub struct SourceDrive {
22    /// Bytes admitted for the opened observation and the source: the
23    /// route's charge without the passes' blocks.
24    pub charged: u64,
25    /// Bytes of the one block the drive fills
26    /// ([`NativeBlock::capacity_bytes`]), which the passes' share holds.
27    pub block: u64,
28    /// Blocks filled, over every traversal.
29    pub blocks: usize,
30}
31
32/// Compile `request`, admit its batch route as a run does, read its source
33/// once, then call `mark` and read it again with
34/// a fresh source: flags and weights over every plane, visibilities over
35/// every plane and, for a cube, visibilities in restricted waves of
36/// `planes_per_wave` planes. The first reading leaves the opened store's
37/// own state (its storage managers' column caches, which the run's law
38/// bounds by the MeasurementSet's size) and the process caches as a run's
39/// later passes find them, so what a caller counts from `mark` on is the
40/// source's: its batches, survey, pointing, geometry and packing, and the
41/// block it fills. The admission is held until this returns. Kernel sets do
42/// not ramp to the pointing here.
43///
44/// # Errors
45///
46/// When the request does not compile, the route is refused or not
47/// admitted, or a traversal fails.
48pub fn drive_batch_source(
49    request: &ImagingRequest,
50    context: &RunContext,
51    planes_per_wave: u32,
52    mark: &mut dyn FnMut(),
53) -> Result<SourceDrive, ApplicationDispatchError> {
54    let resolved = std::cell::Cell::new(None);
55    let (problem, input, native, _resident) = crate::compile_run(request, context, &resolved)?;
56    let admitted =
57        crate::native::batch_route(&problem, input.observation, (false, false), &native.runtime)?;
58    let planes = planes(&problem);
59    let source = || {
60        ObservationSource::new(
61            &problem,
62            &admitted.value,
63            planes,
64            imaging::plane_bounds(&problem),
65            false,
66        )
67    };
68    read(source(), planes, planes_per_wave)?;
69    mark();
70    let (blocks, block) = read(source(), planes, planes_per_wave)?;
71    // The passes' share of the charge: what they hold for blocks of the
72    // source's largest.
73    let passes = source_block_bytes(source().maximum_block());
74    Ok(SourceDrive {
75        charged: admitted.reservation.memory().saturating_sub(passes),
76        block,
77        blocks,
78    })
79}

81/// Compile `request`, which writes visibilities back, admit its batch
82/// route as a run does, read its source
83/// once, then call `mark` and drive a writing pass: a sink begun for the
84/// next traversal, every block of a fresh source's traversal of every
85/// plane handed to it with a zero prediction for each sample, and the
86/// write completed. What a caller counts from `mark` on is the source's and
87/// the write's: the updaters, the patches and the provenance; `charged`
88/// adds the write's charge (the batch route's `write_bytes`) to the
89/// route's. The predictions handed to the sink are allocated before
90/// `mark`: the passes hold them.
91///
92/// # Errors
93///
94/// As [`drive_batch_source`], and when the write fails.
95pub fn drive_batch_write(
96    request: &ImagingRequest,
97    context: &RunContext,
98    mark: &mut dyn FnMut(),
99) -> Result<SourceDrive, ApplicationDispatchError> {
100    let resolved = std::cell::Cell::new(None);
101    let (problem, input, native, _resident) = crate::compile_run(request, context, &resolved)?;
102    let admitted = crate::native::batch_route(
103        &problem,
104        input.observation,
105        (input.write_model_column, input.write_corrected_data),
106        &native.runtime,
107    )?;
108    let failed = |error: casa_imaging_runtime::pass::SourceError| {
109        ApplicationDispatchError::Native(error.to_string().into())
110    };
111    let planes = planes(&problem);
112    let source = || {
113        ObservationSource::new(
114            &problem,
115            &admitted.value,
116            planes,
117            imaging::plane_bounds(&problem),
118            false,
119        )
120    };
121    read(source(), planes, planes)?;
122    let target = VisibilityWriteTarget(&admitted.value);
123    let write = target.retained_bytes().map_err(failed)?;
124    let mut source = source();
125    let maximum = source.maximum_block();
126    let mut block = NativeBlock::with_capacity(maximum);
127    let predictions =
128        vec![Complex32::default(); maximum.rows * maximum.channels() * maximum.correlations];
129    mark();
130    let mut writer = target
131        .begin(problem.visibility_transform())
132        .map_err(failed)?;
133    source
134        .begin(
135            PlaneRange::new(0, planes),
136            false,
137            SampleContent::Visibilities,
138        )
139        .map_err(failed)?;
140    let mut blocks = 0;
141    while source.fill(&mut block).map_err(failed)? {
142        let cells = block.len() * block.channels() * block.correlations();
143        writer
144            .write(&block, &predictions[..cells])
145            .map_err(failed)?;
146        blocks += 1;
147    }
148    writer.complete().map_err(failed)?;
149    let passes = source_block_bytes(maximum);
150    Ok(SourceDrive {
151        charged: admitted.reservation.memory().saturating_sub(passes) + write,
152        block: block.capacity_bytes(),
153        blocks,
154    })
155}

157/// Read `source` through a run's traversals of `planes` planes, waves of
158/// `planes_per_wave` included, into one block: the blocks filled, and the
159/// block's bytes.
160fn read(
161    mut source: ObservationSource<'_>,
162    planes: u32,
163    planes_per_wave: u32,
164) -> Result<(usize, u64), ApplicationDispatchError> {
165    let mut block = NativeBlock::with_capacity(source.maximum_block());
166    let mut blocks = 0;
167    let mut traverse = |range: PlaneRange, restrict: bool, content: SampleContent| {
168        source.begin(range, restrict, content)?;
169        while source.fill(&mut block)? {
170            blocks += 1;
171        }
172        Ok::<_, casa_imaging_runtime::pass::SourceError>(())
173    };
174    let all = PlaneRange::new(0, planes);
175    let step = planes_per_wave.clamp(1, planes);
176    traverse(all, false, SampleContent::FlagsAndWeights)
177        .and_then(|()| traverse(all, false, SampleContent::Visibilities))
178        .and_then(|()| {
179            (0..planes)
180                .step_by(step as usize)
181                .filter(|_| step < planes)
182                .try_for_each(|start| {
183                    traverse(
184                        PlaneRange::new(start, (start + step).min(planes)),
185                        true,
186                        SampleContent::Visibilities,
187                    )
188                })
189        })
190        .map_err(|error| ApplicationDispatchError::Native(error.to_string().into()))?;
191    Ok((blocks, block.capacity_bytes()))
192}

194/// The planes of `problem`'s basis a wave divides: a cube's channels, or
195/// one.
196fn planes(problem: &CompiledProblem) -> u32 {
197    match problem.reconstruction().basis() {
198        ReconstructionBasis::ChannelLocal { channels } => {
199            u32::try_from(channels).expect("a cube's channels fit a plane index")
200        }
201        _ => 1,
202    }
203}
