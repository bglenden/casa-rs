4use std::collections::HashMap;
5use std::sync::Arc;
6
7use casa_core::measures::frequency::MFrequency;
8
9use super::ChannelBin;
10use super::plotms_ordinates::ScalarSpanPlan;
11use super::plotms_time_baselines::TimeBaselinePlan;
12use casa_ms::columns::frequency_columns::ChanFreqColumn;

14#[derive(Debug, Clone, PartialEq)]
15pub(super) struct SpectralChannel {
16    pub(super) number: usize,
17    pub(super) frequency_hz: f64,
18}
19
20/// The displayed channels and frequencies for an averaged visibility buffer.
21/// PlotMS takes the window ID from the first buffer and the grid from the
22/// buffer still attached to the iterator when it caches the average.
23#[derive(Debug, Clone, PartialEq)]
24pub struct MsScatterPlotMsAverageSpw {
25    /// Spectral-window ID of the first input visibility buffer.
26    pub spectral_window: i32,
27    pub scan_field: Option<(i32, i32)>,
28    channels: Arc<[SpectralChannel]>,
29}
30
31impl MsScatterPlotMsAverageSpw {
32    pub(super) fn channel(&self, ordinal: usize) -> &SpectralChannel {
33        &self.channels[ordinal]
34    }
35
36    pub fn frequency_hz(&self, display_channel: usize) -> Option<f64> {
37        let index = self
38            .channels
39            .binary_search_by_key(&display_channel, |channel| channel.number)
40            .ok()?;
41        Some(self.channels[index].frequency_hz)
42    }
43}
44
45#[derive(Default)]
46pub(super) struct SpectralAverageGroups {
47    averages: Vec<Option<MsScatterPlotMsAverageSpw>>,
48    windows: HashMap<i32, SpectralWindowChannels>,
49}
50
51struct SpectralWindowChannels {
52    input_channels: usize,
53    display_channels: Arc<[SpectralChannel]>,
54}
55
56impl SpectralAverageGroups {
57    pub(super) fn output_window(&self, group: usize) -> Option<i32> {
58        self.averages
59            .get(group)?
60            .as_ref()
61            .map(|average| average.spectral_window)
62    }
63
64    pub(super) fn observe(
65        &mut self,
66        group: usize,
67        spw: i32,
68        input_channels: usize,
69        bins: &[ChannelBin],
70        frequencies: &ChanFreqColumn<'_>,
71        starts_buffer: bool,
72    ) -> Result<(), String> {
73        let window = match self.windows.entry(spw) {
74            std::collections::hash_map::Entry::Occupied(entry) => entry.into_mut(),
75            std::collections::hash_map::Entry::Vacant(entry) => {
76                let frequencies = frequencies
77                    .get_frequencies(spw as usize)
78                    .map_err(|error| error.to_string())?;
79                check_input_channels(spw, frequencies.len(), input_channels)?;
80                let display_channels = bins
81                    .iter()
82                    .map(|bin| {
83                        // The iterator averages visibility samples from the
84                        // compressed selection but advertises physical bins.
85                        let physical_bin = &frequencies[bin.start..bin.end];
86                        SpectralChannel {
87                            number: bin.display_channel(),
88                            frequency_hz: physical_bin.iter().map(MFrequency::hz).sum::<f64>()
89                                / physical_bin.len() as f64,
90                        }
91                    })
92                    .collect();
93                entry.insert(SpectralWindowChannels {
94                    input_channels,
95                    display_channels,
96                })
97            }
98        };
99        check_input_channels(spw, window.input_channels, input_channels)?;
100        if group >= self.averages.len() {
101            self.averages.resize_with(group + 1, || None);
102        }
103        let average = self.averages[group].get_or_insert_with(|| MsScatterPlotMsAverageSpw {
104            spectral_window: spw,
105            scan_field: None,
106            channels: window.display_channels.clone(),
107        });
108        if starts_buffer {
109            average.channels.clone_from(&window.display_channels);
110        }
111        Ok(())
112    }
113
114    pub(super) fn into_averages(
115        mut self,
116        spans: Option<&ScalarSpanPlan>,
117        time_plan: Option<&TimeBaselinePlan>,
118    ) -> Arc<[Option<MsScatterPlotMsAverageSpw>]> {
119        if let Some(plan) = time_plan {
120            return plan
121                .spectral_window_buffers()
122                .map(|(first, last, scan_field)| {
123                    let mut average = self.averages.get(first)?.as_ref()?.clone();
124                    let last = self.averages.get(last)?.as_ref()?;
125                    average.channels.clone_from(&last.channels);
126                    average.scan_field = Some(scan_field);
127                    Some(average)
128                })
129                .collect();
130        }
131        if let Some(spans) = spans {
132            for (group, average) in self.averages.iter_mut().enumerate() {
133                if let Some(average) = average {
134                    let span = spans.metadata(group);
135                    average.scan_field = Some((span.scan, span.field));
136                }
137            }
138        }
139        self.averages.into()
140    }
141}
142
143fn check_input_channels(spw: i32, frequencies: usize, input_channels: usize) -> Result<(), String> {
144    if frequencies != input_channels {
145        return Err(format!(
146            "SPECTRAL_WINDOW row {spw} reported {frequencies} channel frequencies for {input_channels} data channels"
147        ));
148    }
149    Ok(())
150}
