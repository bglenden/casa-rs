2//! Small symmetric covariance eigensystem for the WVR proposal distribution.
3
4const ORDER: usize = 4;
5type Matrix = [[f64; ORDER]; ORDER];
6
7pub(super) fn principal_axes(mut covariance: Matrix, dimensions: usize) -> Matrix {
8    let mut basis =
9        std::array::from_fn(|row| std::array::from_fn(|column| f64::from(row == column)));
10    tridiagonalize(&mut covariance, &mut basis, dimensions);
11    diagonalize(&mut covariance, &mut basis, dimensions);
12    let mut order = [0, 1, 2, 3];
13    order[..dimensions].sort_by(|&left, &right| {
14        covariance[left][left]
15            .abs()
16            .total_cmp(&covariance[right][right].abs())
17    });
18    std::array::from_fn(|axis| {
19        let column = order[axis];
20        let step = covariance[column][column].max(0.0).sqrt() * 0.1;
21        std::array::from_fn(|row| basis[row][column] * step)
22    })
23}

25fn tridiagonalize(matrix: &mut Matrix, basis: &mut Matrix, size: usize) {
26    for column in 0..size - 2 {
27        let first = column + 1;
28        let head = matrix[first][column];
29        let norm = (first + 1..size).fold(0.0_f64, |norm, row| norm.hypot(matrix[row][column]));
30        if norm == 0.0 {
31            continue;
32        }
33        let subdiagonal = -head.hypot(norm).copysign(head);
34        let factor = (subdiagonal - head) / subdiagonal;
35        let mut reflector = [0.0; ORDER];
36        reflector[first] = 1.0;
37        for row in first + 1..size {
38            reflector[row] = matrix[row][column] / (head - subdiagonal);
39            matrix[row][column] = 0.0;
40            matrix[column][row] = 0.0;
41        }
42        matrix[column][first] = subdiagonal;
43        matrix[first][column] = subdiagonal;
44        let mut update = [0.0; ORDER];
45        for row in first..size {
46            update[row] = factor * dot(&matrix[row], &reflector, first, size);
47        }
48        let correction = -0.5 * factor * dot(&reflector, &update, first, size);
49        for row in first..size {
50            update[row] += correction * reflector[row];
51        }
52        for row in first..size {
53            for col in first..=row {
54                matrix[row][col] -= reflector[row] * update[col] + update[row] * reflector[col];
55                matrix[col][row] = matrix[row][col];
56            }
57        }
58        for row in &mut basis[..size] {
59            let projection = factor * dot(row, &reflector, first, size);
60            for col in first..size {
61                row[col] -= projection * reflector[col];
62            }
63        }
64    }
65}

67fn dot(left: &[f64; ORDER], right: &[f64; ORDER], start: usize, end: usize) -> f64 {
68    (start..end).map(|index| left[index] * right[index]).sum()
69}

71fn diagonalize(matrix: &mut Matrix, basis: &mut Matrix, size: usize) {
72    // Householder reduction followed by implicit QR with a trailing Wilkinson
73    // shift. The rotation convention fixes eigenvector signs for seeded draws.
74    for _ in 0..128 {
75        for index in 1..size {
76            let scale = matrix[index - 1][index - 1].abs() + matrix[index][index].abs();
77            if matrix[index][index - 1].abs() < f64::EPSILON * scale {
78                matrix[index][index - 1] = 0.0;
79                matrix[index - 1][index] = 0.0;
80            }
81        }
82        let Some(last) = (1..size).rfind(|&index| matrix[index][index - 1] != 0.0) else {
83            break;
84        };
85        let first = (1..last)
86            .rfind(|&index| matrix[index][index - 1] == 0.0)
87            .unwrap_or(0);
88        qr_sweep(matrix, basis, first, last);
89    }
90}

92fn qr_sweep(matrix: &mut Matrix, basis: &mut Matrix, first: usize, last: usize) {
93    let gap = (matrix[last - 1][last - 1] - matrix[last][last]) * 0.5;
94    let edge = matrix[last][last - 1];
95    let denominator = (gap.abs() + gap.hypot(edge)).copysign(gap);
96    let shift = matrix[last][last] - edge * (edge / denominator);
97    let pivot = matrix[first][first];
98    let mut x = if f64::EPSILON * shift.abs() > pivot.abs() + matrix[first + 1][first].abs() {
99        pivot
100    } else {
101        pivot - shift
102    };
103    let mut y = matrix[first + 1][first];
104    for left in first..last {
105        let right = left + 1;
106        let (cosine, sine) = annihilate(x, y);
107        for row in &mut matrix[first..=last] {
108            rotate(row, left, right, cosine, sine);
109        }
110        let (upper, lower) = matrix.split_at_mut(right);
111        for (a, b) in upper[left][first..=last]
112            .iter_mut()
113            .zip(&mut lower[0][first..=last])
114        {
115            (*a, *b) = (cosine * *a - sine * *b, sine * *a + cosine * *b);
116        }
117        for row in &mut *basis {
118            rotate(row, left, right, cosine, sine);
119        }
120        if left > first {
121            matrix[right][left - 1] = 0.0;
122            matrix[left - 1][right] = 0.0;
123        }
124        if right < last {
125            x = matrix[right][left];
126            y = matrix[right + 1][left];
127        }
128    }
129    for row in first + 1..=last {
130        matrix[row - 1][row] = matrix[row][row - 1];
131    }
132}

134fn rotate(row: &mut [f64; ORDER], left: usize, right: usize, c: f64, s: f64) {
135    let (a, b) = (row[left], row[right]);
136    row[left] = c * a - s * b;
137    row[right] = s * a + c * b;
138}
139
140fn annihilate(x: f64, y: f64) -> (f64, f64) {
141    if y == 0.0 {
142        return (1.0, 0.0);
143    }
144    let length = x.hypot(y);
145    let sign = if y.abs() > x.abs() { -y } else { x }.signum();
146    (sign * x / length, -sign * y / length)
147}
