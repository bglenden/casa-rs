20#[derive(Debug, Clone, Copy, PartialEq)]
21enum SpectralBasisPlan {
22    ChannelLocal,
23    Polynomial(BlockNormalPlan),
24}
25
26impl SpectralBasisPlan {
27    const fn coefficient_terms(self, slab: SpectralSlabPlan) -> usize {
28        match self {
29            Self::ChannelLocal => slab.core_depth(),
30            Self::Polynomial(plan) => plan.coefficient_term_count(),
31        }
32    }
33
34    const fn normal_moments(self, slab: SpectralSlabPlan) -> usize {
35        match self {
36            Self::ChannelLocal => slab.core_depth(),
37            Self::Polynomial(plan) => plan.normal_moment_count(),
38        }
39    }
40
41    fn normal_moment_index(self, row: usize, column: usize) -> Option<usize> {
42        match self {
43            Self::Polynomial(plan) => plan.normal_moment_index(row, column),
44            Self::ChannelLocal => None,
45        }
46    }
47}

49/// The output channels one normal state covers.
50///
51/// A constant or Taylor state covers its one channel; a channel-local state
52/// covers its core range of the output axis.
53#[derive(Debug, Clone, Copy, PartialEq, Eq)]
54pub struct SpectralSlabPlan {
55    total_channels: usize,
56    core_start: usize,
57    core_end: usize,
58}
59
60impl SpectralSlabPlan {
61    /// Return the total compiled output-channel count.
62    #[must_use]
63    pub const fn total_channels(self) -> usize {
64        self.total_channels
65    }
66
67    /// Return the half-open output-channel range owned by this slab.
68    #[must_use]
69    pub const fn core_range(self) -> std::ops::Range<usize> {
70        self.core_start..self.core_end
71    }
72
73    /// Return the number of output channels owned by this slab.
74    #[must_use]
75    pub const fn core_depth(self) -> usize {
76        self.core_end - self.core_start
77    }
78}

100/// Unnormalized spectral primitives; these are not Product Graph artifacts.
101#[derive(Debug)]
102pub struct SpectralOperatorPrimitives {
103    shape: [usize; 2],
104    slab: SpectralSlabPlan,
105    basis: SpectralBasisPlan,
106    polarizations: usize,
107    dirty: Box<[Complex64]>,
108    pub(crate) cube_real: Option<CubeRealFields>,
109    psf: Box<[Complex64]>,
110    sensitivity: Box<[f64]>,
111    sum_weights: Box<[f64]>,
112    published_sum_weights: Box<[f64]>,
113    validity: Box<[SpectralChannelValidity]>,
114}
115
116impl SpectralOperatorPrimitives {
117    /// Return `[width, height]` shared by every plane.
118    #[must_use]
119    pub const fn shape(&self) -> [usize; 2] {
120        self.shape
121    }
122
123    /// Return the exact output-channel core represented by the flattened planes.
124    #[must_use]
125    pub const fn slab(&self) -> SpectralSlabPlan {
126        self.slab
127    }
128
129    /// Return the number of model-coefficient residual terms.
130    #[must_use]
131    pub const fn coefficient_term_count(&self) -> usize {
132        self.basis.coefficient_terms(self.slab)
133    }
134
135    /// Return the number of reconstruction polarization planes.
136    #[must_use]
137    pub const fn polarization_count(&self) -> usize {
138        self.polarizations
139    }
140
141    /// Return the number of retained normal moments.
142    #[must_use]
143    pub const fn normal_moment_count(&self) -> usize {
144        self.basis.normal_moments(self.slab)
145    }
146
147    /// Return the Taylor reference frequency, or `None` for channel-local state.
148    #[must_use]
149    pub const fn reference_frequency_hz(&self) -> Option<f64> {
150        match self.basis {
151            SpectralBasisPlan::Polynomial(plan) => Some(plan.reference_frequency_hz()),
152            SpectralBasisPlan::ChannelLocal => None,
153        }
154    }
155
156    /// Map one coefficient-block pair to its retained normal moment.
157    #[must_use]
158    pub fn normal_moment_index(&self, row: usize, column: usize) -> Option<usize> {
159        self.basis.normal_moment_index(row, column)
160    }
161
162    /// Return the unnormalized dirty normal-state plane.
163    #[must_use]
164    pub fn dirty(&self) -> crate::NormalValues<'_> {
165        self.cube_real
166            .as_ref()
167            .map_or(crate::NormalValues::Complex(&self.dirty), |real| {
168                crate::NormalValues::Real(&real.dirty)
169            })
170    }
171
172    /// Return the unnormalized point-spread-function plane.
173    #[must_use]
174    pub fn psf(&self) -> crate::NormalValues<'_> {
175        self.cube_real
176            .as_ref()
177            .map_or(crate::NormalValues::Complex(&self.psf), |real| {
178                crate::NormalValues::Real(&real.psf)
179            })
180    }
181
182    /// Borrow compact real fields before the channel-local normal fold consumes them.
183    #[doc(hidden)]
184    pub fn cube_real_fields(&self) -> Option<&CubeRealFields> {
185        self.cube_real.as_ref()
186    }
187
188    /// Return scalar-response sensitivity in normal-state units.
189    #[must_use]
190    pub fn sensitivity(&self) -> crate::SensitivityValues<'_> {
191        if self.cube_real.is_some() {
192            crate::SensitivityValues::PerPlane {
193                weights: &self.sum_weights,
194                cells: self.shape[0] * self.shape[1],
195            }
196        } else {
197            crate::SensitivityValues::Dense(&self.sensitivity)
198        }
199    }
200
201    /// Return normal-moment-major, polarization-minor exact sum weights.
202    ///
203    /// The length is `normal_moment_count() * polarization_count()`.
204    #[must_use]
205    pub const fn sum_weights(&self) -> &[f64] {
206        &self.sum_weights
207    }
208
209    /// Return CASA publication-statistic numerators in normal-moment-major,
210    /// polarization-minor order.
211    ///
212    /// The length is `normal_moment_count() * polarization_count()`. Channel-
213    /// major Taylor state retains `sum(W² x^t)` here so products can apply the
214    /// single complete-family `sum(W)` denominator. Publication does not change
215    /// scientific normalization.
216    #[must_use]
217    pub const fn published_sum_weights(&self) -> &[f64] {
218        &self.published_sum_weights
219    }
220
221    /// Return support-entry-major, polarization-minor validity.
222    ///
223    /// The support-entry count is core-channel depth for channel-local state
224    /// or one for Taylor state.
225    #[must_use]
226    pub const fn channel_validity(&self) -> &[SpectralChannelValidity] {
227        &self.validity
228    }
229
230    /// Return the exact scalar sum weight for the one-plane continuum case.
231    ///
232    /// Cube consumers use [`Self::sum_weights`] instead.
233    #[must_use]
234    pub fn sum_weight(&self) -> f64 {
235        assert_eq!(
236            self.sum_weights.len(),
237            1,
238            "cube normal state has per-channel weights"
239        );
240        self.sum_weights[0]
241    }
242}

328#[derive(Debug)]
329pub(crate) struct SpectralPrimitiveDomains {
330    domains: Box<[SpectralDomainPrimitives]>,
331}
332
333impl SpectralPrimitiveDomains {
334    pub(crate) fn owned_bytes(&self) -> usize {
335        self.iter()
336            .map(|domain| {
337                let p = domain.primitives();
338                size_of::<SpectralDomainPrimitives>()
339                    + std::mem::size_of_val(p.dirty.as_ref())
340                    + p.cube_real.as_ref().map_or(0, |real| {
341                        std::mem::size_of_val(real.dirty.as_ref())
342                            + std::mem::size_of_val(real.psf.as_ref())
343                    })
344                    + std::mem::size_of_val(p.psf.as_ref())
345                    + std::mem::size_of_val(p.sensitivity.as_ref())
346                    + std::mem::size_of_val(p.sum_weights())
347                    + std::mem::size_of_val(p.published_sum_weights())
348                    + std::mem::size_of_val(p.channel_validity())
349                    + match &domain.domain_role {
350                        ImageDomainRole::Main => 0,
351                        ImageDomainRole::Outlier(name) => name.capacity(),
352                    }
353            })
354            .sum()
355    }
356
357    pub(crate) fn new(
358        domains: Box<[SpectralDomainPrimitives]>,
359    ) -> Result<Self, SpectralOperatorError> {
360        if domains.is_empty()
361            || domains
362                .iter()
363                .enumerate()
364                .any(|(ordinal, domain)| domain.domain_ordinal != ordinal)
365        {
366            return Err(SpectralOperatorError::DomainProjectionMismatch);
367        }
368        Ok(Self { domains })
369    }
370
371    pub(crate) fn len(&self) -> usize {
372        self.domains.len()
373    }
374
375    pub(crate) fn iter(&self) -> std::slice::Iter<'_, SpectralDomainPrimitives> {
376        self.domains.iter()
377    }
378
379    pub(crate) fn primary(&self) -> &SpectralOperatorPrimitives {
380        &self.domains[0].primitives
381    }
382
383    pub(crate) fn get(&self, ordinal: usize) -> Option<&SpectralDomainPrimitives> {
384        self.domains.get(ordinal)
385    }
386}
387
388impl std::ops::Deref for SpectralPrimitiveDomains {
389    type Target = SpectralOperatorPrimitives;
390
391    fn deref(&self) -> &Self::Target {
392        &self.domains[0].primitives
393    }
394}

396fn checked_cells(shape: [usize; 2]) -> Result<usize, SpectralOperatorError> {
397    shape[0]
398        .checked_mul(shape[1])
399        .ok_or(SpectralOperatorError::ResidencyOverflow)
400}

402/// Exact reason a normal state or its operators rejected their input.
403#[derive(Debug, Clone, PartialEq, Eq, Error)]
404pub enum SpectralOperatorError {
405    /// A spatial transform (FFT) failed; the run cannot continue.
406    #[error("spatial execution: {0}")]
407    SpatialExecution(String),
408    /// A bounded Normal State backing operation failed.
409    #[error("normal-state storage: {0}")]
410    NormalStorage(String),
411    /// An authoritative model window could not be loaded.
412    #[error(transparent)]
413    ModelAccess(#[from] crate::ModelLifecycleError),
414    /// Runtime supplied science or weighting state for another compiled problem.
415    #[error("spectral operator science and weighting state do not match the compiled problem")]
416    ProblemMismatch,
417    /// The problem is outside the supported scalar-response Stokes-I bases.
418    #[error("spectral operator requires a supported scalar-response Stokes-I reconstruction basis")]
419    UnsupportedProblem,
420    /// A requested output slab is empty or outside the compiled spectral axis.
421    #[error("spectral operator slab is empty or outside the output spectral axis")]
422    InvalidSlab,
423    /// The current serial operator supports only centered identity-PC SIN geometry.
424    #[error("spectral operator does not support this direction-coordinate geometry")]
425    UnsupportedGeometry,
426    /// A resident-byte calculation overflowed.
427    #[error("spectral operator residency cannot be represented")]
428    ResidencyOverflow,
429    /// A weighted contribution contains an invalid numerical value.
430    #[error("spectral operator sample is non-finite or outside its numerical domain")]
431    InvalidSample,
432    /// Selected row geometry did not provide one canonical projection per image domain.
433    #[error("selected row image-domain projections do not match the compiled geometry")]
434    DomainProjectionMismatch,
435    /// The operator generated a non-finite value under a rejecting numerics contract.
436    #[error("spectral operator generated a non-finite value")]
437    GeneratedNonfinite,
438    /// A pass's images do not cover the state's channels exactly once.
439    #[error("spectral operator pass coverage does not match the normal state")]
440    IncompleteCoverage,
441    /// A later major pass did not carry the exact prior invariant normal state.
442    #[error("spectral operator reusable normal state does not match the residual refresh")]
443    ReusableNormalStateMismatch,
