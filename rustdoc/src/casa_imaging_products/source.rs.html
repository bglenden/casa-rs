3//! Typed source catalog for continuum products.
4//!
5//! Every source role keeps its own typed owner record. The catalog is minted
6//! only from a whole released Major-Cycle join, so one atomic reconciliation
7//! result carries its multiple distinct completions together and no role's
8//! evidence can be substituted for another.
9

13    ImageDomainReconstructionMasks, MajorCycleCompletion, MajorCycleCompletionId, ModelGeneration,
14    ModelGenerationId, NormalStateCatalog, ReconstructionMask, ReconstructionMaskGenerationId,
15};

20#[derive(Debug, Clone, Copy)]
21enum ContinuumNormalStateCatalog {
22    PlaneV1,
23    ChannelSlabV1,
24    TaylorBlockV1,
25    JointBlockV1,
26}

28/// Identity catalog of every source role behind one continuum generation.
29///
30/// Constructed only by [`ContinuumSourceCatalog::from_major_cycle`]; there is
31/// no per-field public constructor, so a commitment set always describes one
32/// inseparable reconciliation lineage.
33#[derive(Debug, Clone)]
34pub struct ContinuumSourceCatalog {
35    problem: CompiledProblem,
36    graph_id: ProductGraphId,
37    major_cycle_completion: MajorCycleCompletionId,
38    normal_state_completion: FinalNormalStateCompletionId,
39    normal_state_content: LogicalIdentity,
40    normal_state_catalog: ContinuumNormalStateCatalog,
41    input_model_generation: ModelGenerationId,
42    final_model_generation: ModelGenerationId,
43    weighting_generation: casa_imaging_reconstruction::WeightingGenerationId,
44    replay: casa_imaging_reconstruction::WeightingReplayId,
45    coverage: casa_imaging_reconstruction::WeightingReplayCoverageId,
46    selected_generation: casa_imaging_model::SelectedObservationGenerationId,
47    continuum_transform_generation: Option<casa_imaging_model::ContinuumTransformGenerationId>,
48    sample_count: u64,
49    block_count: u64,
50    reconstruction_mask: Option<ReconstructionMaskGenerationId>,
51    line_reconstruction_mask: Option<ReconstructionMaskGenerationId>,
52}
53
54impl ContinuumSourceCatalog {
55    /// Mint the catalog from one whole released Major-Cycle join.
56    ///
57    /// The problem must be the exact compiled problem the join reconciled.
58    ///
59    /// # Errors
60    ///
61    /// Rejects joins from another compiled problem and unsupported catalogs.
62    pub fn from_major_cycle(
63        problem: &CompiledProblem,
64        join: &MajorCycleCompletion,
65    ) -> Result<Self, ProductsError> {
66        Self::from_major_cycle_with_mask(problem, join, None)
67    }
68
69    /// Mint the catalog with the exact reconstruction mask used by the final
70    /// bounded solve.
71    ///
72    /// The mask is a distinct scientific source from Final Normal State
73    /// sensitivity/validity. Its immutable generation is therefore bound into
74    /// the product commitment rather than inferred by the product owner.
75    pub fn from_major_cycle_with_mask(
76        problem: &CompiledProblem,
77        join: &MajorCycleCompletion,
78        mask: Option<&ReconstructionMask>,
79    ) -> Result<Self, ProductsError> {
80        let normal_state = join.normal_state();
81        if normal_state.problem_id() != problem.problem_id() {
82            return Err(ProductsError::SourceLineageMismatch);
83        }
84        if join.model_completion().problem() != problem.problem_id() {
85            return Err(ProductsError::SourceLineageMismatch);
86        }
87        let normal_state_catalog = match normal_state.catalog() {
88            NormalStateCatalog::UnnormalizedPlaneV1 => ContinuumNormalStateCatalog::PlaneV1,
89            NormalStateCatalog::UnnormalizedChannelSlabV1 => {
90                ContinuumNormalStateCatalog::ChannelSlabV1
91            }
92            NormalStateCatalog::UnnormalizedTaylorBlockV1 => {
93                ContinuumNormalStateCatalog::TaylorBlockV1
94            }
95            NormalStateCatalog::UnnormalizedJointBlockV1 => {
96                ContinuumNormalStateCatalog::JointBlockV1
97            }
98        };
99        if mask.is_some_and(|mask| {
100            mask.problem_id() != problem.problem_id() || mask.shape() != normal_state.shape()
101        }) {
102            return Err(ProductsError::SourceLineageMismatch);
103        }
104        Ok(Self {
105            graph_id: problem.product_graph().graph_id(),
106            major_cycle_completion: join.completion_id(),
107            normal_state_completion: normal_state.completion_id(),
108            normal_state_content: normal_state.content_identity(),
109            normal_state_catalog,
110            input_model_generation: normal_state.input_model_generation(),
111            final_model_generation: normal_state.final_model_generation(),
112            weighting_generation: normal_state.weighting_generation(),
113            replay: normal_state.replay_id(),
114            coverage: normal_state.coverage(),
115            selected_generation: normal_state.selected_generation(),
116            continuum_transform_generation: normal_state.continuum_transform_generation(),
117            sample_count: normal_state.sample_count(),
118            block_count: normal_state.block_count(),
119            reconstruction_mask: mask.map(ReconstructionMask::generation_id),
120            line_reconstruction_mask: None,
121            problem: problem.clone(),
122        })
123    }
124
125    /// Mint the catalog with both exact spatial supports of a joint solve.
126    pub fn from_major_cycle_with_coupled_masks(
127        problem: &CompiledProblem,
128        join: &MajorCycleCompletion,
129        masks: &CoupledReconstructionMask,
130    ) -> Result<Self, ProductsError> {
131        let mut catalog = Self::from_major_cycle_with_mask(problem, join, Some(masks.continuum()))?;
132        if masks.line().problem_id() != problem.problem_id()
133            || masks.line().shape() != join.normal_state().shape()
134            || join.normal_state().coupled_mask_generation() != Some(masks.generation_id())
135        {
136            return Err(ProductsError::SourceLineageMismatch);
137        }
138        catalog.line_reconstruction_mask = Some(masks.line().generation_id());
139        Ok(catalog)
140    }
141
142    /// Mint the catalog with one exact reconstruction support per image domain.
143    pub fn from_major_cycle_with_domain_masks(
144        problem: &CompiledProblem,
145        join: &MajorCycleCompletion,
146        masks: &ImageDomainReconstructionMasks,
147    ) -> Result<Self, ProductsError> {
148        if masks.len() != join.normal_state().domain_count()
149            || masks.len() != problem.geometry().domains().len()
150            || join.normal_state().image_domain_mask_generation() != Some(masks.generation_id())
151            || masks
152                .iter()
153                .zip(problem.geometry().domains())
154                .any(|(mask, domain)| {
155                    mask.problem_id() != problem.problem_id()
156                        || mask.shape() != domain.shape().pixels()
157                        || mask.coordinate() != domain.direction()
158                })
159        {
160            return Err(ProductsError::SourceLineageMismatch);
161        }
162        let mut catalog = Self::from_major_cycle(problem, join)?;
163        catalog.reconstruction_mask = Some(masks.generation_id());
164        Ok(catalog)
165    }

191    /// Return the schema-versioned digest binding every role's evidence.
192    #[must_use]
193    pub fn commitment_id(&self) -> [u8; 32] {
194        let mut encoder = Encoder::new(COMMITMENT_DOMAIN, COMMITMENT_VERSION);
195        encoder.identity(self.problem.problem_id().as_bytes());
196        encoder.identity(self.graph_id.as_bytes());
197        encoder.identity(self.major_cycle_completion.as_bytes());
198        encoder.identity(self.normal_state_completion.as_bytes());
199        encoder.identity(self.normal_state_content.as_bytes());
200        encoder.u8(match self.normal_state_catalog {
201            ContinuumNormalStateCatalog::PlaneV1 => 0,
202            ContinuumNormalStateCatalog::ChannelSlabV1 => 1,
203            ContinuumNormalStateCatalog::TaylorBlockV1 => 2,
204            ContinuumNormalStateCatalog::JointBlockV1 => 3,
205        });
206        encoder.identity(self.input_model_generation.as_bytes());
207        encoder.identity(self.final_model_generation.as_bytes());
208        encoder.identity(self.weighting_generation.as_bytes());
209        encoder.identity(self.replay.as_bytes());
210        encoder.identity(self.coverage.as_bytes());
211        encoder.identity(self.selected_generation.as_bytes());
212        match self.continuum_transform_generation {
213            Some(generation) => {
214                encoder.u8(1);
215                encoder.identity(generation.as_bytes());
216            }
217            None => encoder.u8(0),
218        }
219        encoder.u64(self.sample_count);
220        encoder.u64(self.block_count);
221        match self.reconstruction_mask {
222            Some(generation) => {
223                encoder.u8(1);
224                encoder.identity(generation.as_bytes());
225            }
226            None => encoder.u8(0),
227        }
228        match self.line_reconstruction_mask {
229            Some(generation) => {
230                encoder.u8(1);
231                encoder.identity(generation.as_bytes());
232            }
233            None => encoder.u8(0),
234        }
235        encoder.finish()
236    }

253/// Borrowed scientific payloads for producing planned members.
254///
255/// Like the catalog, this value is minted only from a whole join plus its
256/// exact compiled problem, so production always consumes the same atomic
257/// reconciliation result the commitments name.
258#[derive(Debug)]
259pub struct ContinuumProductInputs<'a> {
260    problem: &'a CompiledProblem,
261    normal_state: &'a FinalNormalState,
262    final_model: &'a ModelGeneration,
263    reconstruction_mask: Option<&'a ReconstructionMask>,
264    domain_reconstruction_masks: Option<&'a ImageDomainReconstructionMasks>,
265    coupled_masks: Option<&'a CoupledReconstructionMask>,
266}
267
268impl<'a> ContinuumProductInputs<'a> {
269    /// Bind the borrowed payloads of one released join.
270    ///
271    /// # Errors
272    ///
273    /// Rejects joins whose normal state belongs to another compiled problem.
274    pub fn from_major_cycle(
275        problem: &'a CompiledProblem,
276        join: &'a MajorCycleCompletion,
277    ) -> Result<Self, ProductsError> {
278        if join.normal_state().problem_id() != problem.problem_id() {
279            return Err(ProductsError::SourceLineageMismatch);
280        }
281        Ok(Self {
282            problem,
283            normal_state: join.normal_state(),
284            final_model: join.final_model(),
285            reconstruction_mask: None,
286            domain_reconstruction_masks: None,
287            coupled_masks: None,
288        })
289    }
290
291    /// Bind the exact reconstruction mask used by the final bounded solve.
292    ///
293    /// The mask must belong to this problem and model-grid shape.
294    pub fn with_reconstruction_mask(
295        mut self,
296        mask: &'a ReconstructionMask,
297    ) -> Result<Self, ProductsError> {
298        if mask.problem_id() != self.problem.problem_id()
299            || mask.shape() != self.normal_state.shape()
300        {
301            return Err(ProductsError::SourceLineageMismatch);
302        }
303        self.reconstruction_mask = Some(mask);
304        self.domain_reconstruction_masks = None;
305        self.coupled_masks = None;
306        Ok(self)
307    }
308
309    /// Bind both exact spatial supports used by a joint solve.
310    pub fn with_coupled_reconstruction_masks(
311        mut self,
312        masks: &'a CoupledReconstructionMask,
313    ) -> Result<Self, ProductsError> {
314        for mask in [masks.continuum(), masks.line()] {
315            if mask.problem_id() != self.problem.problem_id()
316                || mask.shape() != self.normal_state.shape()
317            {
318                return Err(ProductsError::SourceLineageMismatch);
319            }
320        }
321        self.reconstruction_mask = Some(masks.continuum());
322        self.domain_reconstruction_masks = None;
323        self.coupled_masks = Some(masks);
324        Ok(self)
325    }
326
327    /// Bind the exact spatial support for every canonical image domain.
328    pub fn with_domain_reconstruction_masks(
329        mut self,
330        masks: &'a ImageDomainReconstructionMasks,
331    ) -> Result<Self, ProductsError> {
332        if masks.len() != self.normal_state.domain_count()
333            || masks.len() != self.problem.geometry().domains().len()
334            || self.normal_state.image_domain_mask_generation() != Some(masks.generation_id())
335            || masks
336                .iter()
337                .zip(self.problem.geometry().domains())
338                .any(|(mask, domain)| {
339                    mask.problem_id() != self.problem.problem_id()
340                        || mask.shape() != domain.shape().pixels()
341                        || mask.coordinate() != domain.direction()
342                })
343        {
344            return Err(ProductsError::SourceLineageMismatch);
345        }
346        self.reconstruction_mask = None;
347        self.coupled_masks = None;
348        self.domain_reconstruction_masks = Some(masks);
349        Ok(self)
350    }

358    /// Return radians-per-pixel on each direction axis of the main domain.
359    #[must_use]
360    pub fn cell_size_rad(&self) -> [f64; 2] {
361        self.cell_size_rad_for_domain(&ImageDomainRole::Main)
362            .expect("compiled geometry contains exactly one main domain")
363    }
364
365    /// Return radians-per-pixel on each direction axis of one image domain.
366    pub(crate) fn cell_size_rad_for_domain(
367        &self,
368        role: &ImageDomainRole,
369    ) -> Result<[f64; 2], ProductsError> {
370        let domain = self
371            .problem
372            .geometry()
373            .domains()
374            .iter()
375            .find(|domain| domain.role() == role)
376            .ok_or(ProductsError::SourceLineageMismatch)?;
377        let increment = domain.direction().increment_rad();
378        Ok([increment[0].abs(), increment[1].abs()])
379    }
380
381    /// Resolve the canonical model-domain ordinal from a compiler-owned role.
382    pub(crate) fn model_domain_ordinal(
383        &self,
384        role: &ImageDomainRole,
385    ) -> Result<usize, ProductsError> {
386        let mut matches = self
387            .final_model
388            .shape()
389            .domain_roles()
390            .iter()
391            .enumerate()
392            .filter(|(_, candidate)| *candidate == role);
393        let (ordinal, _) = matches.next().ok_or(ProductsError::SourceLineageMismatch)?;
394        if matches.next().is_some() {
395            return Err(ProductsError::SourceLineageMismatch);
396        }
397        Ok(ordinal)
398    }

426    pub fn reconstruction_mask_generation(&self) -> Option<ReconstructionMaskGenerationId> {
427        self.domain_reconstruction_masks
428            .map(ImageDomainReconstructionMasks::generation_id)
429            .or_else(|| {

433    }
