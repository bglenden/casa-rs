27pub(super) struct ModelInputs {
28    read_column: bool,
29    virtual_models: BTreeMap<i32, Arc<[Entry]>>,
30    frequencies: BTreeMap<i32, (Vec<f64>, FrequencyRef)>,
31    geometry: Option<MsCalEngine>,
32    dynamic_fields: BTreeSet<i32>,
33    pub(super) phase_times: BTreeMap<usize, f64>,
34    finite_fields: BTreeSet<i32>,
35    model_buffers: BTreeMap<usize, Arc<ModelBuffer>>,

38struct ModelBuffer {
39    selection: [i32; 4],
40    focus_rows: Vec<FocusSample>,
41}
42
43struct FocusSample {
44    index: usize,
45    antennas: [i32; 2],
46}
47
48struct Entry {
49    selection: EntrySelection,
50    prediction: Prediction,
51    scratch_channels: Vec<[i32; 4]>,
52}
53
54enum EntrySelection {
55    Rows(Vec<[i32; 4]>),
56    FieldAndWindow { fields: Vec<i32>, windows: Vec<i32> },
57}
58
59impl EntrySelection {
60    fn matches(&self, combination: [i32; 4]) -> bool {
61        match self {
62            Self::Rows(rows) => rows.contains(&combination),
63            Self::FieldAndWindow { fields, windows } => {
64                fields.contains(&combination[0]) && windows.contains(&combination[1])
65            }
66        }
67    }
68}
69
70enum Prediction {
71    Components(ComponentList),
72    FourierGrid(Box<FourierModel>),
73}

75impl ModelInputs {
76    pub(super) fn new(
77        ms: &MeasurementSet,
78        rows: &[SelectedSolveRow],
79        source: GainSolveModelSource,
80    ) -> Result<Self, GainSolveError> {
81        let read_column = source == GainSolveModelSource::ModelColumn
82            || (source == GainSolveModelSource::FromMeasurementSet
83                && ms
84                    .main_table()
85                    .schema()
86                    .is_some_and(|schema| schema.contains_column("MODEL_DATA")));
87        let mut inputs = Self::empty(read_column);
88        if source != GainSolveModelSource::FromMeasurementSet {
89            return Ok(inputs);
90        }
91        let fields: BTreeSet<_> = rows.iter().map(|row| row.field_id).collect();
92        let mut records = BTreeMap::<_, Arc<[Entry]>>::new();
93        for field in fields {
94            if let Some(record) =
95                virtual_model(ms, field).map_err(|source| read_error(ms, source))?
96            {
97                let entries = match records.entry(record.key) {
98                    std::collections::btree_map::Entry::Occupied(entry) => entry.into_mut(),
99                    std::collections::btree_map::Entry::Vacant(entry) => {
100                        entry.insert(decode_entries(&record.record)?.into())
101                    }
102                };
103                inputs.virtual_models.insert(field, Arc::clone(entries));
104            }
105        }
106        inputs.load_geometry(ms, rows)?;
107        Ok(inputs)
108    }
109
110    pub(super) fn from_entries(
111        ms: &MeasurementSet,
112        rows: &[SelectedSolveRow],
113        entries: &[VirtualModelEntry],
114    ) -> Result<Self, GainSolveError> {
115        let mut inputs = Self::empty(false);
116        let entries: Arc<[Entry]> = entries
117            .iter()
118            .map(|entry| {
119                decode_entry(
120                    entry.kind,
121                    EntrySelection::Rows(entry.index_combination.clone()),
122                    &entry.container,
123                )
124                .map(|mut model| {
125                    model.scratch_channels = entry.channel_selection.clone();
126                    model
127                })
128            })
129            .collect::<Result<Vec<_>, _>>()?
130            .into();
131        for row in rows {
132            inputs
133                .virtual_models
134                .entry(row.field_id)
135                .or_insert_with(|| Arc::clone(&entries));
136        }
137        inputs.load_geometry(ms, rows)?;
138        Ok(inputs)
139    }

141    fn empty(read_column: bool) -> Self {
142        Self {
143            read_column,
144            virtual_models: BTreeMap::new(),
145            frequencies: BTreeMap::new(),
146            geometry: None,
147            dynamic_fields: BTreeSet::new(),
148            phase_times: BTreeMap::new(),
149            finite_fields: BTreeSet::new(),
150            model_buffers: BTreeMap::new(),
151        }
152    }
153
154    fn load_geometry(
155        &mut self,
156        ms: &MeasurementSet,
157        rows: &[SelectedSolveRow],
158    ) -> Result<(), GainSolveError> {
159        self.finite_fields = self.virtual_models.iter().filter_map(|(&field, models)| {
160            models.iter().any(|entry| matches!(&entry.prediction, Prediction::FourierGrid(grid) if grid.needs_refocus())).then_some(field)
161        }).collect();
162        if self
163            .virtual_models
164            .values()
165            .any(|entries| !entries.is_empty())
166        {
167            let spectral_window = ms
168                .spectral_window()
169                .map_err(|source| read_error(ms, source))?;
170            let windows: BTreeSet<_> = rows
171                .iter()
172                .filter(|row| self.virtual_models.contains_key(&row.field_id))
173                .map(|row| row.data_spw_id)
174                .collect();
175            for window in windows {
176                let frequencies = spectral_window
177                    .chan_freq(window as usize)
178                    .map_err(|source| read_error(ms, source))?;
179                let reference = spectral_window
180                    .frequency_reference(window as usize)
181                    .map_err(|source| read_error(ms, source))?;
182                self.frequencies.insert(window, (frequencies, reference));
183            }
184            let engine = MsCalEngine::new(ms)
185                .and_then(|engine| engine.with_field_ephemerides(ms))
186                .map_err(|source| read_error(ms, source))?;
187            let field = ms.field().map_err(|source| read_error(ms, source))?;
188            for &id in self.virtual_models.keys() {
189                let index = id as usize;
190                if engine.field_moves(index)
191                    || field
192                        .num_poly(index)
193                        .map_err(|source| read_error(ms, source))?
194                        > 0
195                {
196                    self.dynamic_fields.insert(id);
197                }
198            }
199            self.geometry = Some(engine);
200        }
201        Ok(())
202    }

204    pub(super) fn reads_column(&self, field: i32) -> bool {
205        self.read_column && !self.virtual_models.contains_key(&field)
206    }

208    pub(super) fn dynamic_fields(&self) -> &BTreeSet<i32> {

210    }

212    pub(super) fn uses_fourier(&self, field: i32) -> bool {
213        self.virtual_models.get(&field).is_some_and(|entries| {
214            entries
215                .iter()
216                .any(|entry| matches!(entry.prediction, Prediction::FourierGrid(_)))
217        })
218    }

220    pub(super) fn needs_buffers(&self) -> bool {
221        !self.virtual_models.is_empty()
222    }
223
224    pub(super) fn bind_buffers(
225        &mut self,
226        ms: &MeasurementSet,
227        rows: &[BucketedRow<'_>],
228        combine: GainSolveCombine,
229    ) -> Result<(), GainSolveError> {
230        if !self.needs_buffers() {
231            return Ok(());
232        }
233        let mut buffers = BTreeMap::<VisibilityBuffer, Vec<&SelectedSolveRow>>::new();
234        for row in rows
235            .iter()
236            .filter(|(row, _, _)| self.virtual_models.contains_key(&row.field_id))
237        {
238            buffers
239                .entry(VisibilityBuffer::new(row, combine))
240                .or_default()
241                .push(row.0);
242        }
243        for rows in buffers.into_values() {
244            let first = rows[0];
245            let state = get_i32(ms.main_table(), first.row_index, "STATE_ID")?;
246            let selection = [first.field_id, first.data_spw_id, first.scan_number, state];
247            let focus_rows = if self.finite_fields.contains(&first.field_id) {
248                rows.iter()
249                    .map(|row| FocusSample {
250                        index: row.row_index,
251                        antennas: [row.antenna1, row.antenna2],
252                    })
253                    .collect()
254            } else {
255                Vec::new()
256            };
257            let buffer = Arc::new(ModelBuffer {
258                selection,
259                focus_rows,
260            });
261            for row in rows {
262                self.model_buffers
263                    .insert(row.row_index, Arc::clone(&buffer));
264            }
265        }
266        Ok(())
267    }

269    /// A defined virtual model overrides MODEL_DATA. CASA uses a unit point
270    /// model if none of the field's entries matches this visibility buffer.
271    pub(super) fn predict(
272        &self,
273        ms: &MeasurementSet,
274        row: &SelectedSolveRow,
275        correlations: &[i32],
276        raw_flags: &ArrayD<bool>,
277    ) -> Result<Option<ArrayD<Complex32>>, GainSolveError> {
278        self.predict_model(ms, row, correlations, (raw_flags, None), false)
279    }
280
281    pub(super) fn predict_model(
282        &self,
283        ms: &MeasurementSet,
284        row: &SelectedSolveRow,
285        correlations: &[i32],
286        (raw_flags, previous): (&ArrayD<bool>, Option<ArrayD<Complex32>>),
287        scratch_model: bool,
288    ) -> Result<Option<ArrayD<Complex32>>, GainSolveError> {
289        let Some(entries) = self.virtual_models.get(&row.field_id) else {
290            return Ok(None);
291        };
292        if entries.is_empty() {
293            return Ok(None);
294        }
295        let buffer = &self.model_buffers[&row.row_index];
296        let mut matching = entries
297            .iter()
298            .filter(|entry| entry.selection.matches(buffer.selection))
299            .peekable();
300        if matching.peek().is_none() {
301            return Ok(None);
302        }
303        // CASA restores the last component recipe registered for a visibility
304        // buffer. A MeasurementSet can retain older overlapping recipes, for
305        // example after replacing a setjy model with imager.ft.
306        let virtual_components = (!scratch_model)
307            .then(|| {
308                entries.iter().rev().find(|entry| {
309                    matches!(&entry.prediction, Prediction::Components(_))
310                        && entry.selection.matches(buffer.selection)
311                })
312            })
313            .flatten();
314        let engine = self
315            .geometry
316            .as_ref()
317            .expect("virtual models have geometry");
318        let phase_time = self
319            .phase_times
320            .get(&row.row_index)
321            .copied()
322            .unwrap_or(row.time_seconds);
323        let field = field_direction(ms, engine, row.field_id as usize, phase_time)
324            .map_err(|source| read_error(ms, source))?;
325        let frame = MeasFrame::new()
326            .with_epoch(MEpoch::from_mjd(
327                row.time_seconds / 86400.0,
328                engine.time_reference(),
329            ))
330            .with_position(engine.observatory_position().clone())
331            .with_direction(field.clone());
332        let uvw = uvw(ms, row.row_index)?;
333        let (frequencies, frequency_ref) = &self.frequencies[&row.data_spw_id];
334        let (basis, offset) = match correlations.first().copied() {
335            Some(5..=8) => (Polarisation::Circular, 5),
336            Some(9..=12) => (Polarisation::Linear, 9),
337            _ => {
338                return Err(model_error(
339                    "virtual component prediction requires linear or circular correlations",
340                ));
341            }
342        };
343        if correlations
344            .iter()
345            .any(|&code| !(offset..offset + 4).contains(&code))
346        {
347            return Err(model_error(
348                "virtual component prediction requires one correlation basis",
349            ));
350        }
351        let shape = [correlations.len(), frequencies.len()];
352        let mut model = match previous {
353            Some(model) if model.shape() != shape => {
354                return Err(model_error("MODEL_DATA does not match its layout"));
355            }
356            Some(model) if model.t().is_standard_layout() => model,
357            Some(model) => ArrayD::from_shape_fn(IxDyn(&shape).f(), |index| model[index]),
358            None => ArrayD::<Complex32>::zeros(IxDyn(&shape).f()),
359        };
360        let mut components_written = false;
361        for entry in matching {
362            let components = match &entry.prediction {
363                Prediction::Components(components) => {
364                    if virtual_components.is_some_and(|active| !std::ptr::eq(active, entry)) {
365                        continue;
366                    }
367                    components_written = true;
368                    components
369                }
370                Prediction::FourierGrid(grid) => {
371                    grid.predict(
372                        FourierRow {
373                            ms,
374                            row,
375                            correlations,
376                            frequencies,
377                            raw_flags,
378                            frequency_ref: *frequency_ref,
379                            engine,
380                            field: &field,
381                            phase_time,
382                            frame: &frame,
383                            uvw,
384                            focus_rows: &buffer.focus_rows,
385                            scratch_channels: entry
386                                .scratch_channels
387                                .iter()
388                                .find(|channels| channels[0] == row.data_spw_id)
389                                .copied(),
390                        },
391                        &mut model,
392                        if !scratch_model {
393                            FourierMerge::VirtualSum
394                        } else if components_written {
395                            FourierMerge::ScratchReplaceComponents
396                        } else {
397                            FourierMerge::ScratchSum
398                        },
399                    )?;
400                    components_written = false;
401                    continue;
402                }
403            };
404            for component in components.iter() {
405                let projection =
406                    UvwMachine::with_frame(component.shape().reference_direction(), &field, &frame)
407                        .map_err(model_error)?;
408                let (rotated, phase) = projection.convert_projected([-uvw[0], -uvw[1], uvw[2]]);
409                let baseline = [-rotated[0], -rotated[1], rotated[2]];
410                for (channel, &frequency) in frequencies.iter().enumerate() {
411                    let visibility = component
412                        .visibility(baseline, frequency, basis)
413                        .map_err(model_error)?;
414                    let rotation = Complex64::from_polar(
415                        1.0,
416                        std::f64::consts::TAU * phase * frequency / 299_792_458.0,
417                    );
418                    for (correlation, &code) in correlations.iter().enumerate() {
419                        let value = visibility[(code - offset) as usize] * rotation;
420                        model[[correlation, channel]] +=
421                            Complex32::new(value.re as f32, value.im as f32);
422                    }
423                }
424            }
425        }
426        Ok(Some(model))
427    }

430fn decode_entries(record: &RecordValue) -> Result<Vec<Entry>, GainSolveError> {
431    if record.get("version").is_some() {
432        return VirtualModelEntry::entries(record)
433            .map_err(model_error)?
434            .into_iter()
435            .map(|entry| {
436                decode_entry(
437                    entry.kind,
438                    EntrySelection::Rows(entry.index_combination),
439                    &entry.container,
440                )
441            })
442            .collect();
443    }
444    let mut entries = Vec::new();
445    for (kind, count, prefix) in [
446        (VirtualModelKind::FtMachine, "numft", "ft"),
447        (VirtualModelKind::ComponentList, "numcl", "cl"),
448    ] {
449        let Some(Value::Scalar(ScalarValue::Int32(count))) = record.get(count) else {
450            continue;
451        };
452        for index in 0..*count {
453            let name = format!("{prefix}_{index}");
454            let Some(Value::Record(entry)) = record.get(&name) else {
455                return Err(model_error(format!(
456                    "legacy virtual model has no {name} record"
457                )));
458            };
459            let Some(Value::Record(container)) = entry.get("container") else {
460                return Err(model_error(format!("legacy model {name} has no container")));
461            };
462            let selection = EntrySelection::FieldAndWindow {
463                fields: legacy_ids(entry, "fields")?,
464                windows: legacy_ids(entry, "spws")?,
465            };
466            entries.push(decode_entry(kind, selection, container)?);
467        }
468    }
469    Ok(entries)
470}
471
472fn decode_entry(
473    kind: VirtualModelKind,
474    selection: EntrySelection,
475    container: &RecordValue,
476) -> Result<Entry, GainSolveError> {
477    let prediction = match kind {
478        VirtualModelKind::ComponentList => {
479            Prediction::Components(ComponentList::from_record(container).map_err(model_error)?)
480        }
481        VirtualModelKind::FtMachine => {
482            Prediction::FourierGrid(Box::new(FourierModel::new(container.clone())))
483        }
484    };
485    Ok(Entry {
486        selection,
487        prediction,
488        scratch_channels: Vec::new(),
489    })
490}
491
492fn legacy_ids(record: &RecordValue, field: &str) -> Result<Vec<i32>, GainSolveError> {
493    match record.get(field) {
494        Some(Value::Array(ArrayValue::Int32(values))) if values.ndim() == 1 => {
495            Ok(values.iter().copied().collect())
496        }
497        _ => Err(model_error(format!(
498            "legacy virtual model has no {field} vector"
499        ))),
500    }
501}

503fn field_direction(
504    ms: &MeasurementSet,
505    engine: &MsCalEngine,
506    field_id: usize,
507    time: f64,
508) -> Result<MDirection, MsError> {
509    if engine.field_moves(field_id) {
510        return engine.observation_direction_j2000(time, field_id);
511    }
512    let field = ms.field()?;
513    let direction = raw_field_phase_direction(ms, field_id)?;
514    if field.num_poly(field_id)? == 0 {
515        return Ok(direction);
516    }
517    let dt = time - field.time(field_id)?;
518    let polynomial = DirectionColumn::phase_dir(field.table()).get_direction_at(field_id, dt)?;
519    let (longitude, latitude) = polynomial.as_angles();
520    Ok(MDirection::from_angles(
521        longitude,
522        latitude,
523        direction.refer(),
524    ))
525}

527fn uvw(ms: &MeasurementSet, row: usize) -> Result<[f64; 3], GainSolveError> {
528    let values = ms
529        .main_table()
530        .cell_accessor(row, "UVW")
531        .and_then(|cell| cell.array())
532        .map_err(|source| read_error(ms, source.into()))?;
533    if let ArrayValue::Float64(values) = values
534        && let Some(values) = values.as_slice()
535        && let Ok(values) = values.try_into()
536    {
537        return Ok(values);
538    }
539    Err(model_error("UVW must contain three real coordinates"))
540}

542fn read_error(ms: &MeasurementSet, source: MsError) -> GainSolveError {
543    GainSolveError::OpenMeasurementSet {
544        path: ms
545            .path()
546            .unwrap_or(Path::new("<in-memory>"))
547            .display()
548            .to_string(),
549        source,
550    }
551}

554    GainSolveError::VirtualModel(error.to_string())
