3//! A static mask cut to the primary beam: tclean's `usemask='pb'`, every
4//! pixel whose `.pb` exceeds `pbmask`, and a named user mask under a
5//! `pbmask` above zero, kept where the `.pb` exceeds it
6//! (`SynthesisDeconvolver::setupMask`, `SDMaskHandler::makePBMask`).
7//!
8//! tclean forms the mask once, at the run's first boundary, from the `.pb`
9//! of every channel and polarization, so it holds a plane for each; the
10//! next boundary keeps it ([`ReconstructionMaskPlan::next_cycle`]). The
11//! owner of the minor cycle reads each plane's beam ([`MaskBeamRead`]): the
12//! weight image's under a response, the telescope's analytic beam
13//! otherwise.
14//!
15//! [`ReconstructionMaskPlan::next_cycle`]: super::ReconstructionMaskPlan::next_cycle
16
17use std::sync::Arc;
18
19use super::auto_planes::MaskPlaneKey;
20use super::{MaskError, MaskPlanes, ReconstructionMask};
21
22/// The pixels of one plane, x-major, whose `.pb` exceeds a level (CASA's
23/// Float `pb > pbmask`); `None` for a run without a `.pb`. The plane's
24/// channel counts from the normal state's first, and a Taylor family is
25/// one plane, its principal beam's.
26pub type MaskBeamRead<'a, E> =
27    dyn Fn(MaskPlaneKey, f32) -> Result<Option<Vec<bool>>, E> + Sync + 'a;
28
29/// Whether `level` is a `.pb` level tclean takes for `pbmask`: in `[0, 1)`
30/// (`SynthesisParamsDeconv::verify`).
31pub(super) fn valid_level(level: f64) -> bool {
32    (0.0..1.0).contains(&level)
33}
34
35/// `mask`, a static mask of domain `domain`, cut to the pixels whose `.pb`
36/// exceeds `level` in each of `planes`' channels and polarizations, as
37/// `beam` reads them (`iif(pb > pbmask, mask, 0)`). The new support is
38/// written in place, plane after plane; it holds one plane of the beam at
39/// a time beside it.
40///
41/// # Errors
42///
43/// [`MaskError::NoPrimaryBeam`] for a run without a `.pb`, as tclean's
44/// `makePBMask` refuses one; [`MaskError::ShapeMismatch`] when `mask` or a
45/// beam plane does not fit the planes; or what `beam` returns.
46pub(super) fn cut_to_beam<E>(
47    mask: &ReconstructionMask,
48    (domain, planes): (usize, MaskPlanes),
49    level: f64,
50    beam: Option<&MaskBeamRead<'_, E>>,
51) -> Result<ReconstructionMask, E>
52where
53    E: From<MaskError>,
54{
55    let beam = beam.ok_or(MaskError::NoPrimaryBeam)?;
56    let cells = mask.shape[0] * mask.shape[1];
57    // One allocation, written in place and shared as it is.
58    let mut support = (0..planes.count() * cells)
59        .map(|_| false)
60        .collect::<Arc<[bool]>>();
61    let out = Arc::get_mut(&mut support).expect("a support no one else holds yet");
62    let mut planes_out = out.chunks_mut(cells);
63    for channel in 0..planes.channels() {
64        for polarization in 0..planes.polarizations() {
65            let within = mask
66                .plane(channel, polarization)
67                .ok_or(MaskError::ShapeMismatch)?;
68            let kept = beam(
69                MaskPlaneKey {
70                    domain,
71                    channel,
72                    polarization,
73                },
74                level as f32,
75            )?
76            .ok_or(MaskError::NoPrimaryBeam)?;
77            if kept.len() != cells {
78                return Err(MaskError::ShapeMismatch.into());
79            }
80            let plane = planes_out.next().expect("a support plane per plane");
81            for ((set, within), kept) in plane.iter_mut().zip(within).zip(&kept) {
82                *set = *within && *kept;
83            }
84        }
85    }
86    Ok(ReconstructionMask::from_plane_support(
87        mask.coordinate,
88        mask.shape,
89        planes,
90        support,
91    )?)
92}

94#[cfg(test)]
95mod tests {
96    use super::*;
97    use casa_imaging_model::{DirectionCoordinateSpec, DirectionFrame, Projection, SkyDirection};
98
99    fn grid() -> DirectionCoordinateSpec {
100        DirectionCoordinateSpec::new(
101            Projection::Sin,
102            SkyDirection::new(DirectionFrame::J2000, 1.0, 0.5),
103            [1.0, 1.0],
104            [-1.0e-4, 1.0e-4],
105            [[1.0, 0.0], [0.0, 1.0]],
106            [180.0, 90.0],
107        )
108    }
109
110    /// `makePBMask`: each plane of the mask is the static mask where its
111    /// own plane's `.pb` exceeds the level, and the level reaches the beam.
112    #[test]
113    fn each_plane_keeps_the_static_mask_where_its_beam_exceeds_the_level() {
114        let shared = ReconstructionMask::from_reprojected_support(
115            grid(),
116            [2, 2],
117            &[true, true, false, true],
118        )
119        .unwrap();
120        let planes = MaskPlanes::new(2, 2).unwrap();
121        // Plane (c, p)'s beam: its pixel i is kept when c + p ≤ i, and
122        // only at the level 0.5.
123        let beam = |key: MaskPlaneKey, level: f32| {
124            assert_eq!((key.domain, level), (3, 0.5));
125            Ok::<_, MaskError>(Some(
126                (0..4)
127                    .map(|index| key.channel + key.polarization <= index)
128                    .collect(),
129            ))
130        };
131        let cut = cut_to_beam(&shared, (3, planes), 0.5, Some(&beam)).unwrap();
132        assert_eq!(cut.planes(), planes);
133        assert_eq!(cut.plane(0, 0).unwrap(), [true, true, false, true]);
134        assert_eq!(cut.plane(0, 1).unwrap(), [false, true, false, true]);
135        assert_eq!(cut.plane(1, 0).unwrap(), [false, true, false, true]);
136        assert_eq!(cut.plane(1, 1).unwrap(), [false, false, false, true]);
137    }
138
139    /// A run without a `.pb` cannot make one, as `makePBMask` refuses it.
140    #[test]
141    fn a_run_without_a_beam_is_refused() {
142        let full = ReconstructionMask::full_plane(grid(), [2, 2]).unwrap();
143        let none = |_: MaskPlaneKey, _: f32| Ok::<_, MaskError>(None);
144        let planes = MaskPlanes::SHARED;
145        assert_eq!(
146            cut_to_beam(&full, (0, planes), 0.0, Some(&none)).unwrap_err(),
147            MaskError::NoPrimaryBeam
148        );
149        assert_eq!(
150            cut_to_beam::<MaskError>(&full, (0, planes), 0.0, None).unwrap_err(),
151            MaskError::NoPrimaryBeam
152        );
153    }
