24enum Axes {
25    Missing,
26    Numbers(Vec<i64>),
27    Names(Vec<String>),
28    Unsupported,
29}
30
31impl Axes {
32    fn capture(value: Option<&Bound<'_, PyAny>>) -> PyResult<Self> {
33        let Some(value) = value.filter(|value| !value.is_none()) else {
34            return Ok(Self::Missing);
35        };
36        if let Ok(value) = value.cast::<PyString>() {
37            return Ok(Self::Names(vec![value.to_str()?.to_owned()]));
38        }
39        if let Some(index) = integer(value) {
40            return Ok(Self::Numbers(vec![index]));
41        }
42        let numpy_kind = dtype_kind(value);
43        if value.is_instance_of::<PyList>()
44            || value.is_instance_of::<PyTuple>()
45            || numpy_kind.is_some()
46        {
47            let Ok(items) = value.try_iter() else {
48                return Ok(Self::Unsupported);
49            };
50            let items = items.collect::<PyResult<Vec<_>>>()?;
51            if items.is_empty() {
52                return Ok(if matches!(numpy_kind.as_deref(), Some("i" | "u")) {
53                    Self::Numbers(Vec::new())
54                } else {
55                    Self::Missing
56                });
57            }
58            if items.iter().all(is_bool) {
59                return Ok(Self::Missing);
60            }
61            if let Some(values) = items.iter().map(integer).collect() {
62                return Ok(Self::Numbers(values));
63            }
64            if let Ok(values) = items
65                .iter()
66                .map(|item| item.extract::<String>())
67                .collect::<PyResult<Vec<_>>>()
68            {
69                return Ok(Self::Names(values));
70            }
71        }
72        Ok(Self::Unsupported)
73    }
74
75    fn history(&self, input: Option<&Bound<'_, PyAny>>) -> String {
76        match self {
77            Self::Numbers(values) if input.is_some_and(|value| integer(value).is_some()) => {
78                values[0].to_string()
79            }
80            Self::Numbers(values) => format!(
81                "[{}]",
82                values
83                    .iter()
84                    .map(i64::to_string)
85                    .collect::<Vec<_>>()
86                    .join(",")
87            ),
88            Self::Names(values)
89                if input.is_some_and(|value| value.is_instance_of::<PyString>()) =>
90            {
91                format!("'{}'", values[0])
92            }
93            Self::Names(values) => format!(
94                "[{}]",
95                values
96                    .iter()
97                    .map(|value| format!("{value:?}"))
98                    .collect::<Vec<_>>()
99                    .join(",")
100            ),
101            _ => "[]".into(),
102        }
103    }
104
105    fn resolve(&self, coordinates: &CoordinateSystem) -> PyResult<Vec<usize>> {
106        match self {
107            Self::Missing => Err(raised("axes must be specified")),
108            Self::Unsupported => Err(raised("Unsupported type for parameter axes")),
109            Self::Numbers(indices) => indices
110                .iter()
111                .map(|&axis| {
112                    usize::try_from(axis).map_err(|_| raised("Collapse axes must be nonnegative"))
113                })
114                .collect(),
115            Self::Names(names) => {
116                let mut available = vec![String::new(); coordinates.n_world_axes()];
117                for index in 0..coordinates.n_coordinates() {
118                    let model = coordinates.coordinate(index);
119                    for (axis, name) in coordinates
120                        .coordinate_world_axes(index)
121                        .expect("existing coordinate")
122                        .into_iter()
123                        .zip(model.axis_names())
124                    {
125                        if let Some(axis) = axis {
126                            available[axis] = name.to_ascii_lowercase();
127                        }
128                    }
129                }
130                names
131                    .iter()
132                    .map(|name| {
133                        let name = name.to_ascii_lowercase();
134                        let matches: Vec<_> = available
135                            .iter()
136                            .enumerate()
137                            .filter(|(_, axis)| axis.starts_with(&name))
138                            .collect();
139                        match matches.as_slice() {
140                            [(axis, _)] => {
141                                coordinates.world_axis_to_pixel_axis(*axis).ok_or_else(|| {
142                                    raised(format!("World axis {name} has no image pixel axis"))
143                                })
144                            }
145                            [] => Err(raised(format!(
146                                "No axis matches requested axis {name}. Image axis names are [{}]",
147                                available.join(", ")
148                            ))),
149                            _ => Err(raised(format!(
150                                "Multiple axes [{}] match requested axis {name}",
151                                matches
152                                    .iter()
153                                    .map(|(_, name)| name.as_str())
154                                    .collect::<Vec<_>>()
155                                    .join(",")
156                            ))),
157                        }
158                    })
159                    .collect()
160            }
161        }
162    }
163}

165fn dtype_kind(value: &Bound<'_, PyAny>) -> Option<String> {
166    value
167        .getattr("dtype")
168        .ok()?
169        .getattr("kind")
170        .ok()?
171        .extract()
172        .ok()
173}
174
175fn is_bool(value: &Bound<'_, PyAny>) -> bool {
176    value.is_instance_of::<PyBool>() || dtype_kind(value).as_deref() == Some("b")
177}
178
179fn integer(value: &Bound<'_, PyAny>) -> Option<i64> {
180    if is_bool(value) {
181        return None;
182    }
183    value.extract().ok()
184}

186struct Request<'a> {
187    function: &'a str,
188    axes: Axes,
189    axes_history: String,
190    outfile: &'a str,
191    region: Option<RegionArgument>,
192    box_selection: &'a str,
193    chans: &'a str,
194    stokes: &'a str,
195    mask: &'a str,
196    overwrite: bool,
197    stretch: bool,
198}
199
200impl Request<'_> {
201    fn copy<T: CollapsePixel>(
202        &self,
203        source: &dyn ImageInterface<T>,
204        record: RecordValue,
205        beams: &RecordValue,
206        path: Option<&Path>,
207    ) -> PyResult<(Storage<T>, Option<RecordValue>)> {
208        let axes = self.axes.resolve(source.coordinates())?;
209        let options = ImageCollapseOptions {
210            function: CollapseFunction::parse(self.function).map_err(raised)?,
211            axes: &axes,
212            region: self.region.as_ref().and_then(RegionArgument::selection),
213            spatial: (!self.box_selection.is_empty()).then_some(self.box_selection),
214            chans: (!self.chans.is_empty()).then_some(self.chans),
215            stokes: (!self.stokes.is_empty()).then_some(self.stokes),
216            mask: (!self.mask.is_empty()).then_some(self.mask),
217            stretch: self.stretch,
218            destination: (!self.outfile.trim().is_empty()).then(|| Path::new(self.outfile.trim())),
219        };
220        let mut collapse = ImageCollapse::new(source, record, options, beams).map_err(raised)?;
221        let mut storage = Storage::create(
222            collapse.section.shape.clone(),
223            collapse.section.coordinates.clone(),
224            path,
225            None,
226        )
227        .map_err(raised)?;
228        with_storage!(&mut storage, image => collapse.copy_to(source, image.as_mut()))
229            .map_err(raised)?;
230        Ok((storage, collapse.section.coordinate_record.take()))
231    }
232
233    fn build(
234        &self,
235        source: &ImageHandle,
236        beams: &RecordValue,
237        path: Option<&Path>,
238        history: bool,
239    ) -> PyResult<ImageHandle> {
240        if let Some(region) = &self.region {
241            region.validate_selectors(self.box_selection, self.chans, self.stokes)?;
242        }
243        let record = source.coordinate_record();
244        let (mut output, coords) = match source {
245            ImageHandle::Float(Storage::Temp {
246                image,
247                fits: Some(fits),
248                ..
249            }) => {
250                let (storage, coords) = self.copy(&fits.image_view(image), record, beams, path)?;
251                (ImageHandle::Float(storage), coords)
252            }
253            ImageHandle::Float(image) => {
254                let (storage, coords) =
255                    with_storage!(image, image => self.copy(image.as_ref(), record, beams, path))?;
256                (ImageHandle::Float(storage), coords)
257            }
258            ImageHandle::Complex(image) => {
259                let (storage, coords) =
260                    with_storage!(image, image => self.copy(image.as_ref(), record, beams, path))?;
261                (ImageHandle::Complex(storage), coords)
262            }
263            _ => unreachable!("checked single-precision image"),
264        };
265        if let Some(coords) = coords {
266            output.set_coordinate_record(coords)?;
267        }
268        let entries: Vec<ImageHistoryRow> = with_pixels!(source, storage => match storage {
269            Storage::Paged(image) => image_history_rows(image.name().expect("native source")).map_err(raised)?,
270            Storage::Temp { image, .. } => image.history_rows().map_err(raised)?.into_owned(),
271        });
272        with_pixels!(&mut output, storage => with_storage!(storage, image => image.append_history_entries(&entries))).map_err(raised)?;
273        if history {
274            let Self {
275                function,
276                axes_history,
277                outfile,
278                region,
279                box_selection,
280                chans,
281                stokes,
282                mask,
283                overwrite,
284                stretch,
285                ..
286            } = self;
287            let region = match region.as_ref().and_then(RegionArgument::selection) {
288                Some(SubimageRegion::Text(text)) => format!("'{text}'"),
289                Some(SubimageRegion::Record(_)) => "{...}".into(),
290                None => region.as_ref().map_or_else(
291                    || "''".into(),
292                    |region| {
293                        let history = region.history();
294                        if history == "\"\"" {
295                            "''".into()
296                        } else {
297                            history
298                        }
299                    },
300                ),
301            };
302            let name = source.path().map_or_else(
303                || "Temporary_Image".into(),
304                |path| path.display().to_string(),
305            );
306            output.append_history_at("::image::collapse", &[
307                format!("Ran ia.collapse on {name}"),
308                format!("ia.collapse(function='{function}', axes={axes_history}, outfile='{outfile}', region={region}, box='{box_selection}', chans='{chans}', stokes='{stokes}', mask='{mask}', overwrite={overwrite}, stretch={stretch})"),
309            ])?;
310        }
311        output.save().map_err(raised)?;
312        Ok(output)
313    }
314}

316#[pymethods]
317impl PyImageTool {
318    /// Reduce each selected image section to a statistic along the specified axes.
319    #[allow(clippy::too_many_arguments)]
320    #[pyo3(signature = (function="", axes=None, outfile="", region=None, r#box="", chans="", stokes="", mask="", overwrite=false, stretch=false), text_signature = "($self, function='', axes=[], outfile='', region='', box='', chans='', stokes='', mask='', overwrite=False, stretch=False)")]
321    fn collapse(
322        &self,
323        py: Python<'_>,
324        function: &str,
325        axes: Option<&Bound<'_, PyAny>>,
326        outfile: &str,
327        region: Option<RegionArgument>,
328        r#box: &str,
329        chans: &str,
330        stokes: &str,
331        mask: &str,
332        #[pyo3(from_py_with = require_bool)] overwrite: bool,
333        #[pyo3(from_py_with = require_bool)] stretch: bool,
334    ) -> PyResult<Py<Self>> {
335        let captured = Axes::capture(axes)?;
336        let axes_history = captured.history(axes);
337        let request = Request {
338            function,
339            axes: captured,
340            axes_history,
341            outfile,
342            region,
343            box_selection: r#box,
344            chans,
345            stokes,
346            mask,
347            overwrite,
348            stretch,
349        };
350        let product = py.detach(|| {
351            let state = self.state();
352            let Some(attached) = state.image.as_ref() else {
353                return Ok(None);
354            };
355            let mut live_coordinates = None;
356            let outfile = outfile.trim();
357            let mut product = publish_image(outfile, overwrite, |path| {
358                let mut image = attached.lock();
359                image.check_single_precision("collapse")?;
360                if !outfile.is_empty() {
361                    for source in image
362                        .path()
363                        .into_iter()
364                        .chain(attached.operand_paths().iter().map(|path| path.as_path()))
365                    {
366                        ensure_outfile_does_not_alias_input(Path::new(outfile), source)
367                            .map_err(raised)?;
368                    }
369                }
370                attached.require_pixel_read_lock(&mut image)?;
371                let beams = attached.beam_view().clone();
372                let staged = request.build(&image, beams.record(), path, state.history);
373                let released = attached.release_requested_lock(&mut image);
374                let staged = staged?;
375                released?;
376                live_coordinates = AttachedImage::capture_live_coordinates(&staged)?;
377                Ok(staged)
378            })?;
379            if let Some((record, system)) = live_coordinates {
380                product.keep_live_coordinates(record, system);
381            }
382            Ok::<_, PyErr>(Some(product))
383        })?;
384        Py::new(py, product.map_or_else(Self::new, Self::attached))
385    }
