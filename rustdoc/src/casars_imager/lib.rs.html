25/// Spectral imaging family requested at the task surface.
26#[derive(Debug, Clone, Copy, PartialEq, Eq)]
27pub enum SpectralMode {
28    /// Continuum multi-frequency synthesis.
29    Mfs,
30    /// Frame-aware spectral cube.
31    Cube,
32    /// Spectral cube in the native data frame.
33    Cubedata,
34}
35
36impl SpectralMode {
37    fn parse(value: &str) -> Result<Self, String> {
38        match value.to_ascii_lowercase().as_str() {
39            "mfs" => Ok(Self::Mfs),
40            "cube" => Ok(Self::Cube),
41            "cubedata" => Ok(Self::Cubedata),
42            _ => Err(format!("unsupported spectral mode {value:?}")),
43        }
44    }
45
46    pub(crate) const fn cube_specmode(self) -> casa_ms::spectral_selection::CubeSpecMode {
47        match self {
48            Self::Mfs | Self::Cube => casa_ms::spectral_selection::CubeSpecMode::Cube,
49            Self::Cubedata => casa_ms::spectral_selection::CubeSpecMode::Cubedata,
50        }
51    }
52}

397impl CliConfig {
398    /// Parse the direct CLI surface. Machine JSON requests are handled before
399    /// this method by `TaskCliHost`.
400    pub fn parse(args: impl IntoIterator<Item = OsString>) -> Result<Self, String> {
401        let mut config = Self::defaults();
402        let args = args.into_iter().collect::<Vec<_>>();
403        let mut index = 0;
404        while index < args.len() {
405            let flag = args[index]
406                .to_str()
407                .ok_or_else(|| "CLI argument is not UTF-8".to_string())?;
408            let value = |offset: usize| -> Result<&str, String> {
409                args.get(index + offset)
410                    .and_then(|value| value.to_str())
411                    .ok_or_else(|| format!("{flag} requires a value"))
412            };
413            let mut consumed = 2;
414            match flag {
415                "--ms" => config.ms = PathBuf::from(value(1)?),
416                "--imagename" => config.imagename = PathBuf::from(value(1)?),
417                "--imsize" => config.imsize = parse(value(1)?, flag)?,
418                "--cell-arcsec" => config.cell_arcsec = parse(value(1)?, flag)?,
419                "--projection" if value(1)?.eq_ignore_ascii_case("sin") => {}
420                "--projection" => return Err("only SIN projection is supported".to_string()),
421                "--field" => config.field_ids = Some(parse_ids(value(1)?, flag)?),
422                "--uvrange" => config.uvrange = Some(value(1)?.to_string()),
423                "--intent" => config.intent = Some(value(1)?.to_string()),
424                "--phasecenter-field" => config.phasecenter_field = Some(parse(value(1)?, flag)?),
425                "--phasecenter" => config.phasecenter = Some(value(1)?.to_string()),
426                "--ddid" => config.ddid = Some(parse(value(1)?, flag)?),
427                "--spw" => {
428                    let text = value(1)?.to_string();
429                    config.spw = text.parse().ok();
430                    config.spw_selector = Some(text);
431                }
432                "--channel-start" => config.channel_start = Some(parse(value(1)?, flag)?),
433                "--channel-count" | "--nchan" => {
434                    config.channel_count = Some(parse(value(1)?, flag)?);
435                }
436                "--fitspw" => {
437                    let selector = value(1)?;
438                    config.continuum_fit_spw = (!selector.is_empty()
439                        && !selector.eq_ignore_ascii_case("none"))
440                    .then(|| selector.to_string());
441                }
442                "--fitorder" => config.continuum_fit_order = parse(value(1)?, flag)?,
443                "--datacolumn" => config.datacolumn = Some(value(1)?.to_string()),
444                "--savemodel" => config.save_model = parse_save_model(value(1)?)?,
445                "--save-continuum-residual" => {
446                    config.save_continuum_residual = true;
447                    consumed = 1;
448                }
449                "--startmodel" => config.start_model = Some(PathBuf::from(value(1)?)),
450                "--outlierfile" => config.outlier_file = Some(PathBuf::from(value(1)?)),
451                "--corr" | "--stokes" => config.correlation = Some(value(1)?.to_string()),
452                "--specmode" => config.spectral_mode = SpectralMode::parse(value(1)?)?,
453                "--start" => {
454                    config.cube_axis.start = Some(
455                        CubeAxisValue::parse(value(1)?, config.cube_axis.veltype)
456                            .map_err(|error| error.to_string())?,
457                    )
458                }
459                "--width" => {
460                    config.cube_axis.width = Some(
461                        CubeAxisValue::parse(value(1)?, config.cube_axis.veltype)
462                            .map_err(|error| error.to_string())?,
463                    )
464                }
465                "--outframe" => {
466                    config.cube_axis.outframe = value(1)?
467                        .parse()
468                        .map_err(|error| format!("parse --outframe: {error}"))?
469                }
470                "--veltype" => {
471                    config.cube_axis.veltype = value(1)?
472                        .parse()
473                        .map_err(|error| format!("parse --veltype: {error}"))?
474                }
475                "--interpolation" => {
476                    config.cube_axis.interpolation = parse_cube_interpolation(value(1)?)?
477                }
478                "--restfreq" => {
479                    config.cube_axis.rest_frequency_hz = Some(
480                        casa_ms::parse_rest_frequency_hz(value(1)?)
481                            .map_err(|error| error.to_string())?,
482                    )
483                }
484                "--weighting" => config.weighting = parse_weighting(value(1)?, None)?,
485                "--robust" => {
486                    let robust = parse(value(1)?, flag)?;
487                    config.weighting = parse_weighting("briggs", Some(robust))?;
488                }
489                "--deconvolver" => config.deconvolver = parse_deconvolver(value(1)?)?,
490                "--nterms" => config.nterms = parse(value(1)?, flag)?,
491                "--scales" => config.multiscale_scales = parse_csv(value(1)?, flag)?,
492                "--smallscalebias" => config.small_scale_bias = parse(value(1)?, flag)?,
493                "--niter" => config.niter = parse(value(1)?, flag)?,
494                "--nmajor" => {
495                    let limit = parse::<i64>(value(1)?, flag)?;
496                    config.nmajor = match limit {
497                        -1 => None,
498                        0.. => Some(limit as usize),
499                        _ => return Err("--nmajor expects -1 or a non-negative value".to_string()),
500                    };
501                }
502                "--maximum-model-update-jy" => {
503                    config.maximum_model_update_jy = Some(parse(value(1)?, flag)?)
504                }
505                "--fullsummary" => {
506                    config.fullsummary = true;
507                    consumed = 1;
508                }
509                "--gain" => config.gain = parse(value(1)?, flag)?,
510                "--threshold-jy" | "--threshold" => config.threshold_jy = parse(value(1)?, flag)?,
511                "--nsigma" => config.nsigma = parse(value(1)?, flag)?,
512                "--psf-cutoff" | "--psfcutoff" => config.psf_cutoff = parse(value(1)?, flag)?,
513                "--pblimit" => config.mosaic_pb_limit = parse(value(1)?, flag)?,
514                "--pbcor" => {
515                    config.pbcor = true;
516                    consumed = 1;
517                }
518                "--write-pb" => {
519                    config.write_pb = true;
520                    consumed = 1;

558                "--mask-box" => config.mask_boxes.push(parse_box(value(1)?)?),

819/// Run the machine or direct CLI surface.
820pub fn run_with_cli_args(args: impl IntoIterator<Item = OsString>) -> Result<(), String> {
821    let raw_args = args.into_iter().collect::<Vec<_>>();
822    let mut managed_output = false;
823    let mut args = Vec::with_capacity(raw_args.len());
824    let mut index = 0;
825    while index < raw_args.len() {
826        if raw_args[index].to_str() == Some("--managed-output") {
827            managed_output = raw_args
828                .get(index + 1)
829                .and_then(|value| value.to_str())
830                .map(|value| parse_bool(value, "--managed-output"))
831                .transpose()?
832                .unwrap_or(true);
833            index += usize::from(
834                raw_args
835                    .get(index + 1)
836                    .and_then(|value| value.to_str())
837                    .is_some_and(|value| matches!(value, "true" | "false")),
838            ) + 1;
839            continue;
840        }
841        args.push(raw_args[index].clone());
842        index += 1;
843    }
844    let host = casa_task_runtime::TaskCliHost::new(
845        imager_task_schema_bundle(),
846        |request: ImagerTaskRequest| request.execute(),
847    );
848    if let Some(output) = host.dispatch(&args).map_err(|error| error.to_string())? {
849        println!("{output}");
850        return Ok(());
851    }
852    if args
853        .iter()
854        .any(|arg| matches!(arg.to_str(), Some("-h" | "--help")))
855    {
856        println!("{}", command_schema("casars-imager").render_help());
857        return Ok(());
858    }
859    let config = CliConfig::parse(args)?;
860    let request = ImagerRunTaskRequest::from_cli_config(&config);
861    let summary = run_from_request(&request)?;
862    let result = ImagerRunTaskResult::from_run(request, &summary);
863    if managed_output {
864        println!(
865            "{}",
866            serde_json::to_string_pretty(&ManagedImagingOutput::from_task_result(&result))
867                .map_err(|error| format!("serialize managed imaging output: {error}"))?
868        );
869    } else {
870        for warning in &result.run.warnings {
871            eprintln!("warning: {warning}");
872        }
873        println!(
874            "Wrote CASA-compatible products at prefix {} ({} gridded samples, {} major cycles, {} minor iterations, stop={:?})",
875            result.request.image_name.display(),
876            result.run.gridded_samples,
877            result.run.major_cycles,
878            result.run.minor_iterations,
879            result.run.clean_stop_reason,
880        );
881    }
882    Ok(())
883}

1019fn parse_uv_taper(value: &str) -> Result<GaussianUvTaper, String> {
1020    use casa_types::quanta::{Quantity, Unit};
1021    let values = value.split(',').collect::<Vec<_>>();
1022    if values.len() != 3 {
1023        return Err("--uvtaper expects major,minor,position-angle".to_string());
1024    }
1025    let radians = Unit::new("rad").expect("rad unit");
1026    let quantity = |text: &str| -> Result<f64, String> {
1027        text.parse::<Quantity>()
1028            .map_err(|error| format!("parse UV taper {text:?}: {error}"))?
1029            .get_value_in(&radians)
1030            .map_err(|error| format!("convert UV taper {text:?}: {error}"))
1031    };
1032    Ok(GaussianUvTaper {
1033        major: UvTaperSize::ImageFwhmRad(quantity(values[0])?),
1034        minor: UvTaperSize::ImageFwhmRad(quantity(values[1])?),
1035        position_angle_rad: quantity(values[2])?,
1036    })
1037}
