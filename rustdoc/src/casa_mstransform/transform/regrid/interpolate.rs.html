2//! Sample and flag interpolation for MeasurementSet frequency regridding.
3
4use std::ops::{Add, Div, Mul, Sub};

6use casa_core::ArrayValue;

10#[derive(Clone, Copy, PartialEq, Eq)]
11pub(super) enum Method {
12    Nearest,
13    Linear,
14    Cubic,
15    Spline,
16    FftShift,
17}
18
19impl Method {
20    pub(super) fn parse(value: &str) -> Option<Self> {
21        Some(match value {
22            "nearest" => Self::Nearest,
23            "linear" => Self::Linear,
24            "cubic" => Self::Cubic,
25            "spline" => Self::Spline,
26            "fftshift" => Self::FftShift,
27            _ => return None,
28        })
29    }
30}

32struct Sample {
33    // Indices in monotonic order; the actual MS frequency axis may descend.
34    bracket: [usize; 2],
35    fraction: f64,
36    outside: bool,
37    cubic: Option<[(usize, f32); 4]>,
38}
39
40pub(super) struct Interpolation {
41    samples: Vec<Sample>,
42    centres: Vec<f64>,
43    descending: bool,
44    method: Method,
45}
46
47impl Interpolation {
48    pub(super) fn plan(input: &[f64], output: &[f64], method: Method) -> Result<Self, TableError> {
49        let descending = input.first() > input.last();
50        let mut centres = input.to_vec();
51        if descending {
52            centres.reverse();
53        }
54        if centres.is_empty() || centres.windows(2).any(|pair| pair[0] >= pair[1]) {
55            return Err(TableError::Schema(
56                "Regridding requires ordered, distinct frequencies.".into(),
57            ));
58        }
59        let n = centres.len();
60        let last = n - 1;
61        let method = if method == Method::Cubic && n < 4 {
62            Method::Linear
63        } else {
64            method
65        };
66        let samples = output
67            .iter()
68            .map(|&frequency| {
69                let include_equal = descending && matches!(method, Method::Cubic | Method::Spline);
70                let at = centres.partition_point(|value| {
71                    *value < frequency || (include_equal && *value == frequency)
72                });
73                if n == 1 {
74                    return Sample {
75                        bracket: [0, 0],
76                        fraction: 0.,
77                        outside: true,
78                        cubic: None,
79                    };
80                }
81                let right = at.clamp(1, last);
82                let left = right - 1;
83                let fraction = (frequency - centres[left]) / (centres[right] - centres[left]);
84                let outside = frequency < centres[0] || frequency > centres[last];
85                let (bracket, cubic) = match method {
86                    Method::Nearest => {
87                        let index = if fraction < 0.5 || (fraction == 0.5 && !descending) {
88                            left
89                        } else {
90                            right
91                        };
92                        ([index; 2], None)
93                    }
94                    Method::Linear | Method::FftShift => {
95                        let indices = if fraction <= f64::EPSILON {
96                            [left; 2]
97                        } else if fraction >= 1. - f64::EPSILON {
98                            [right; 2]
99                        } else {
100                            [left, right]
101                        };
102                        (indices, None)
103                    }
104                    Method::Cubic => {
105                        let start = left.saturating_sub(1).min(n - 4);
106                        let coefficients = std::array::from_fn(|offset| {
107                            let index = start + offset;
108                            let x = centres[index];
109                            let weight = (start..start + 4)
110                                .filter(|&other| other != index)
111                                .map(|other| (frequency - centres[other]) / (x - centres[other]))
112                                .product::<f64>();
113                            (index, weight as f32)
114                        });
115                        ([left, right], Some(coefficients))
116                    }
117                    Method::Spline => ([left, right], None),
118                };
119                Sample {
120                    bracket,
121                    fraction,
122                    outside,
123                    cubic,
124                }
125            })
126            .collect();
127        Ok(Self {
128            samples,
129            centres,
130            descending,
131            method,
132        })
133    }
134
135    fn physical(&self, channel: usize) -> usize {
136        if self.descending {
137            self.centres.len() - 1 - channel
138        } else {
139            channel
140        }
141    }
142
143    fn values<T: SampleValue>(&self, input: ArrayView2<'_, T>) -> Array2<T> {
144        // MeasurementSet cells store all correlations of a channel together.
145        let mut output = Array2::default((input.nrows(), self.samples.len()).f());
146        let mut coefficients = Vec::new();
147        let mut curvature = Vec::new();
148        for (corr, mut row) in output.rows_mut().into_iter().enumerate() {
149            let get = |channel| input[[corr, self.physical(channel)]];
150            if self.method == Method::Spline {
151                second_derivatives(
152                    self.centres.len(),
153                    |channel| self.centres[self.physical(channel)],
154                    |channel| input[[corr, channel]],
155                    &mut coefficients,
156                    &mut curvature,
157                );
158            }
159            for (output, sample) in row.iter_mut().zip(&self.samples) {
160                let [left, right] = sample.bracket;
161                *output = if let Some(cubic) = sample.cubic {
162                    cubic
163                        .into_iter()
164                        .fold(T::default(), |sum, (channel, coefficient)| {
165                            sum + get(channel) * coefficient
166                        })
167                } else if left == right {
168                    get(left)
169                } else if self.method == Method::Spline {
170                    let distance = (self.centres[right] - self.centres[left]) as f32;
171                    let b = sample.fraction as f32;
172                    let a = 1. - b;
173                    get(left) * a
174                        + get(right) * b
175                        + (curvature[self.physical(left)] * (a * a * a - a)
176                            + curvature[self.physical(right)] * (b * b * b - b))
177                            * (distance * distance / 6.)
178                } else {
179                    let a = get(left);
180                    a + (get(right) - a) * sample.fraction as f32
181                };
182            }
183        }
184        output
185    }

187    pub(super) fn apply(&self, value: ArrayValue) -> Result<ArrayValue, TableError> {
188        let error =
189            || TableError::Schema("Regridding needs matching correlation and channel axes.".into());
190        if value.shape().get(1) != Some(&self.centres.len()) {
191            return Err(error());
192        }
193        match value {
194            ArrayValue::Float32(data) => {
195                let input = data
196                    .view()
197                    .into_dimensionality::<Ix2>()
198                    .map_err(|_| error())?;
199                Ok(ArrayValue::Float32(self.values(input).into_dyn()))
200            }
201            ArrayValue::Complex32(data) => {
202                let input = data
203                    .view()
204                    .into_dimensionality::<Ix2>()
205                    .map_err(|_| error())?;
206                Ok(ArrayValue::Complex32(self.values(input).into_dyn()))
207            }
208            ArrayValue::Bool(data) => {
209                let input = data
210                    .view()
211                    .into_dimensionality::<Ix2>()
212                    .map_err(|_| error())?;
213                let flags = Array2::from_shape_fn(
214                    (input.nrows(), self.samples.len()).f(),
215                    |(corr, chan)| {
216                        let sample = &self.samples[chan];
217                        let [left, right] = sample.bracket.map(|index| self.physical(index));
218                        sample.outside || input[[corr, left]] || input[[corr, right]]
219                    },
220                );
221                Ok(ArrayValue::Bool(flags.into_dyn()))
222            }
223            _ => Err(error()),
224        }
225    }
226
227    pub(super) fn apply_categories(&self, input: ArrayView3<'_, bool>) -> ArrayValue {
228        let (corr, channels, categories) = input.dim();
229        let output = ndarray::Array3::from_shape_fn(
230            (corr, self.samples.len(), categories).f(),
231            |(corr, chan, category)| {
232                let [left, right] = self.samples[chan].bracket.map(|index| self.physical(index));
233                input[[corr, left.min(channels - 1), category]]
234                    || input[[corr, right.min(channels - 1), category]]
235            },
236        );
237        ArrayValue::Bool(output.into_dyn())
238    }
239}

241trait SampleValue:
242    Copy
243    + Default
244    + Add<Output = Self>
245    + Sub<Output = Self>
246    + Mul<f32, Output = Self>
247    + Div<f32, Output = Self>
248{
249}
250impl SampleValue for f32 {}
251impl SampleValue for num_complex::Complex32 {}
252
253fn second_derivatives<T: SampleValue>(
254    n: usize,
255    coordinate: impl Fn(usize) -> f64,
256    get: impl Fn(usize) -> T,
257    diagonal: &mut Vec<f32>,
258    curvature: &mut Vec<T>,
259) {
260    diagonal.resize(n, 0.);
261    curvature.resize(n, T::default());
262    diagonal[0] = 0.;
263    curvature[0] = T::default();
264    curvature[n - 1] = T::default();
265    for i in 1..n - 1 {
266        let a = (coordinate(i) - coordinate(i - 1)) as f32;
267        let c = (coordinate(i + 1) - coordinate(i)) as f32;
268        let b = 2. * (a + c) - a * diagonal[i - 1];
269        let gradient_diff = (get(i + 1) - get(i)) / c - (get(i) - get(i - 1)) / a;
270        diagonal[i] = c / b;
271        curvature[i] = (gradient_diff * 6. - curvature[i - 1] * a) / b;
272    }
273    for i in (2..n - 1).rev() {
274        curvature[i] = curvature[i] - curvature[i + 1] * diagonal[i];
275    }
