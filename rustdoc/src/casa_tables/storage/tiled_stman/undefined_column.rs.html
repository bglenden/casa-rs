2//! A `TiledShapeStMan` column whose every cell is undefined.
3//!
4//! casacore gives a `TiledShapeStMan` column a hypercube only when a row is
5//! given a shape. A column no row has written, such as the FLAG_CATEGORY of a
6//! MeasurementSet from CASA's simulator, has only the dummy hypercube 0, no
7//! data file and an empty row map, and every cell reads as undefined.
8
9use std::path::Path;
10
11use casa_types::RecordValue;
12
13use super::{
14    StreamedTiledPrimitiveType, TiledStManHeader, TiledVariant, TsmCubeInfo,
15    invalidate_shared_tile_cache_for_table, write_tiled_header,
16};
17use crate::storage::StorageError;
18
19/// Install `dm_name` as a `TiledShapeStMan` column of `row_count` undefined
20/// cells of rank `cell_rank`, laid out as casacore leaves a column no row
21/// has written.
22///
23/// `default_tile_shape` (cell axes then rows) is the tile shape casacore
24/// would give a hypercube created later, for example when a row is given a
25/// shape.
26///
27/// # Errors
28///
29/// A tile shape whose rank is not `cell_rank + 1`, or a failure to write the
30/// header.
31pub fn install_undefined_tiled_shape_column(
32    table_path: &Path,
33    dm_seq_nr: u32,
34    dm_name: &str,
35    row_count: usize,
36    value_type: StreamedTiledPrimitiveType,
37    cell_rank: usize,
38    default_tile_shape: &[usize],
39) -> Result<(), StorageError> {
40    if default_tile_shape.len() != cell_rank + 1 {
41        return Err(StorageError::FormatMismatch(format!(
42            "default tile shape {default_tile_shape:?} does not fit cells of rank {cell_rank}"
43        )));
44    }
45    std::fs::create_dir_all(table_path)?;
46    let header = TiledStManHeader {
47        big_endian: cfg!(target_endian = "big"),
48        seq_nr: dm_seq_nr,
49        nrrow: row_count as u64,
50        col_data_types: vec![value_type.casacore_type()],
51        hypercolumn_name: dm_name.to_string(),
52        max_cache_size: 0,
53        nrdim: (cell_rank + 1) as u32,
54        files: Vec::new(),
55        cubes: vec![TsmCubeInfo {
56            values: RecordValue::default(),
57            extensible: false,
58            cube_shape: Vec::new(),
59            tile_shape: Vec::new(),
60            file_seq_nr: -1,
61            file_offset: 0,
62        }],
63    };
64    let variant = TiledVariant::Shape {
65        default_tile_shape: default_tile_shape
66            .iter()
67            .map(|&extent| extent as i32)
68            .collect(),
69        nr_used_row_map: 0,
70        row_map: Vec::new(),
71        cube_map: Vec::new(),
72        pos_map: Vec::new(),
73    };
74    write_tiled_header(
75        &table_path.join(format!("table.f{dm_seq_nr}")),
76        &variant,
77        &header,
78    )?;
79    invalidate_shared_tile_cache_for_table(table_path);
80    Ok(())
81}

83#[cfg(test)]
84mod tests {
85    use super::super::read_tiled_header;
86    use super::*;
87
88    #[test]
89    fn an_undefined_column_has_only_the_dummy_cube_and_no_row_map() {
90        let directory = tempfile::tempdir().expect("temporary directory");
91        let table_path = directory.path().join("main.table");
92        install_undefined_tiled_shape_column(
93            &table_path,
94            3,
95            "FLAG_CATEGORY",
96            7,
97            StreamedTiledPrimitiveType::Bool,
98            3,
99            &[2, 64, 1, 1024],
100        )
101        .expect("install");
102        let (variant, header) =
103            read_tiled_header(&table_path.join("table.f3")).expect("read header");
104        assert_eq!(header.nrrow, 7);
105        assert_eq!(header.nrdim, 4);
106        assert!(header.files.is_empty());
107        assert_eq!(header.cubes.len(), 1);
108        assert!(header.cubes[0].cube_shape.is_empty());
109        let TiledVariant::Shape {
110            default_tile_shape,
111            nr_used_row_map,
112            row_map,
113            ..
114        } = variant
115        else {
116            panic!("expected a TiledShapeStMan header");
117        };
118        assert_eq!(default_tile_shape, vec![2, 64, 1, 1024]);
119        assert_eq!(nr_used_row_map, 0);
120        assert!(row_map.is_empty());
121        assert!(
122            !table_path.join("table.f3_TSM0").exists(),
123            "an undefined column has no data file"
124        );
125    }
126
127    #[test]
128    fn a_tile_shape_of_the_wrong_rank_is_refused() {
129        let directory = tempfile::tempdir().expect("temporary directory");
130        let error = install_undefined_tiled_shape_column(
131            directory.path(),
132            0,
133            "FLAG_CATEGORY",
134            1,
135            StreamedTiledPrimitiveType::Bool,
136            3,
137            &[2, 64, 1024],
138        )
139        .expect_err("rank mismatch");
140        assert!(error.to_string().contains("rank 3"), "{error}");
141    }
