8/// Keep the legacy imager's unpadded diagnostic out of unverified execution
9/// modes, particularly virtual-model publication whose demand uses 1.2.
10pub(super) fn validate_ft_padding(
11    request: &super::ImagingRequest,
12) -> Result<(), super::InvalidRequest> {
13    use super::{BackendRequest, DataColumn, Gridder, SaveModel, SpecMode, Weighting};
14    let Some(padding) = request.ft_padding else {
15        return Ok(());
16    };
17    if padding.get() == 1.0
18        && matches!(request.gridder, Gridder::Standard { facets } if facets.get() == 1)
19        && request.specmode == SpecMode::Mfs
20        && request.vis.len() == 1
21        && request.stokes == [super::PolarizationCoordinate::StokesI]
22        && request.weighting == Weighting::Natural
23        && request.datacolumn == Some(DataColumn::Data)
24        && request.niter == 0
25        && request.calcpsf
26        && !request.calcres
27        && !request.restoration
28        && request.savemodel == SaveModel::None
29        && !request.tclean_restart
30        && !request.has_start_model()
31        && request.outlierfile.is_none()
32        && request.backend == BackendRequest::Cpu
33        && request.gridprecision == Some(GridPrecision::F64)
34    {
35        Ok(())
36    } else {
37        Err(super::InvalidRequest {
38            parameters: "ft_padding",
39            reason: "only padding 1.0 for a CPU/f64 natural-weighted standard Stokes-I MFS PSF-only run, without model saving or continuation, is verified",
40        })
41    }
42}

44pub(crate) fn validate_awp2_scope(
45    request: &super::ImagingRequest,
46) -> Result<(), super::InvalidRequest> {
47    use super::{
48        Deconvolver, FrequencyRef, Gridder, InvalidRequest, PolarizationCoordinate,
49        ProductNormalization, SaveModel, SpecMode, Weighting,
50    };
51    let Gridder::Awp2 {
52        wprojplanes,
53        computepastep,
54        normtype,
55        ..
56    } = request.gridder
57    else {
58        return Ok(());
59    };
60    let mvc = request.mvc_output_channels.is_some() || request.mvc_auto_channels;
61    let mvc_squint = mvc
62        && computepastep < 180.0
63        && wprojplanes.is_some_and(|count| count.get() > 1)
64        && request.stokes == [PolarizationCoordinate::StokesI]
65        && request.weighting == Weighting::Uniform;
66    if !matches!(request.specmode, SpecMode::Mfs | SpecMode::Cube)
67        || (request.specmode == SpecMode::Cube
68            && (request.savemodel != SaveModel::None
69                || (!mvc_squint
70                    && (computepastep < 180.0
71                        || wprojplanes.is_some_and(|count| count.get() != 1)
72                        || request.weighting != Weighting::Natural))))
73        || (mvc && normtype != ProductNormalization::FlatNoise)
74        || (request.deconvolver != Deconvolver::Hogbom
75            && !(request.deconvolver == Deconvolver::Mtmfs
76                && request.specmode == SpecMode::Mfs
77                && request.nterms == 2
78                && request.stokes == [PolarizationCoordinate::StokesI]
79                && ((computepastep >= 180.0
80                    && wprojplanes.is_none_or(|count| count.get() == 1)
81                    && request.weighting == Weighting::Natural)
82                    || mvc_squint)
83                && request.savemodel == SaveModel::None))
84        || !matches!(request.weighting, Weighting::Natural | Weighting::Uniform)
85        || wprojplanes.is_some_and(|count| count.get() > usize::from(u16::MAX))
86        || !computepastep.is_finite()
87        || computepastep <= 0.0
88        || if computepastep < 180.0 && !mvc_squint {
89            request.stokes
90                != [
91                    PolarizationCoordinate::StokesI,
92                    PolarizationCoordinate::StokesV,
93                ]
94                || wprojplanes.is_none_or(|count| count.get() < 2)
95        } else {
96            use PolarizationCoordinate::{StokesI, StokesQ, StokesU, StokesV};
97            match request.stokes.as_slice() {
98                [StokesI] => false,
99                [StokesQ]
100                | [StokesU]
101                | [StokesV]
102                | [StokesI, StokesV]
103                | [StokesQ, StokesU]
104                | [StokesI, StokesQ, StokesU, StokesV] => {
105                    wprojplanes.is_some_and(|count| count.get() != 1)
106                        || request.savemodel != SaveModel::None
107                }
108                _ => true,
109            }
110        }
111        || request.vis.len() != 1
112        || request.outframe != FrequencyRef::LSRK
113        || request.outlierfile.is_some()
114        || request.has_start_model()
115        || request.tclean_restart
116        || request.savemodel == SaveModel::Virtual
117        || !request.calcpsf
118        || !request.calcres
119        || request.gridprecision == Some(GridPrecision::F32)
120    {
121        return Err(InvalidRequest {
122            parameters: "gridder",
123            reason: "AWP2 currently supports fresh double-precision Högbom MFS with natural or uniform weighting, and natural-weighted cubes with one W plane, no squint and no saved model. Unsquinted polarized imaging is available for I, Q, U, V, IV, QU and IQUV with one W plane and no saved model. Natural-weighted two-term Stokes I MT-MFS also runs without squint or a saved model and with one W plane. AWP2 MVC requires flatnoise normalization and supports natural weighting without squint or W projection, or uniform weighting with Stokes I, squint and multiple W planes (#1071)",
124        });
125    }
126    Ok(())
127}

129pub(super) fn major_cycle_limit<'de, D: Deserializer<'de>>(
130    deserializer: D,
131) -> Result<Option<usize>, D::Error> {
132    let value = Value::deserialize(deserializer)?;
133    match value.as_i64() {
134        Some(-1) => Ok(None),
135        Some(limit) => usize::try_from(limit)
136            .map(Some)
137            .map_err(|_| unexpected(&value, "-1 or a major-cycle count")),
138        None => Err(unexpected(&value, "-1 or a major-cycle count")),
139    }
140}

142pub(super) fn hogbom_iteration_mode<'de, D: Deserializer<'de>>(
143    deserializer: D,
144) -> Result<HogbomIterationAccounting, D::Error> {
145    match choice(deserializer)?.as_str() {
146        "strict" => Ok(HogbomIterationAccounting::Strict),
147        "casa-inclusive" => Ok(HogbomIterationAccounting::CasaInclusive),
148        other => Err(D::Error::custom(format!("hogbom_iteration_mode {other:?}"))),
149    }
150}
151
152pub(super) fn normalization<'de, D: Deserializer<'de>>(
153    deserializer: D,
154) -> Result<ProductNormalization, D::Error> {
155    match choice(deserializer)?.as_str() {
156        "flatnoise" => Ok(ProductNormalization::FlatNoise),
157        "flatsky" => Ok(ProductNormalization::FlatSky),
158        "pbsquare" => Ok(ProductNormalization::PbSquare),
159        other => Err(D::Error::custom(format!("normtype {other:?}"))),
160    }
161}
162
163pub(super) fn grid_precision<'de, D: Deserializer<'de>>(
164    deserializer: D,
165) -> Result<Option<GridPrecision>, D::Error> {
166    match choice(deserializer)?.as_str() {
167        "auto" => Ok(None),
168        "f32" => Ok(Some(GridPrecision::F32)),
169        "f64" => Ok(Some(GridPrecision::F64)),
170        other => Err(D::Error::custom(format!("gridprecision {other:?}"))),
171    }
172}

174#[cfg(test)]
175mod tests {
176    use super::super::{BackendRequest, GridPrecision, tests::request};
177    use casa_imaging_operator::PaddingFactor;
178    use serde_json::json;
179
180    #[test]
181    fn legacy_padding_is_only_available_for_the_verified_quick_psf_workflow() {
182        let mut virtual_model = request(json!({"savemodel": "virtual", "niter": 1}));
183        virtual_model.ft_padding = Some(PaddingFactor::new(2.0).expect("padding"));
184        assert_eq!(
185            virtual_model
186                .validate()
187                .expect_err("unadmitted virtual grid")
188                .parameters,
189            "ft_padding"
190        );
191        let mut valid = request(json!({
192            "niter": 0, "calcpsf": true, "calcres": false, "restoration": false,
193            "weighting": "natural", "datacolumn": "data",
194        }));
195        valid.backend = BackendRequest::Cpu;
196        valid.gridprecision = Some(GridPrecision::F64);
197        valid.ft_padding = Some(PaddingFactor::new(1.0).expect("padding"));
198        valid.validate().expect("verified PSF");
199        for overrides in [
200            json!({"savemodel": "virtual"}),
201            json!({"savemodel": "modelcolumn"}),
202            json!({"niter": 1}),
203            json!({"calcres": true}),
204            json!({"calcpsf": false}),
205            json!({"restoration": true}),
206            json!({"gridder": "wproject", "wprojplanes": 16}),
207            json!({"specmode": "cube"}),
208            json!({"startmodel": "model.image"}),
209            json!({"stokes": "IQUV"}),
210        ] {
211            let mut values = json!({
212                "niter": 0, "calcpsf": true, "calcres": false, "restoration": false,
213                "weighting": "natural", "datacolumn": "data",
214            });
215            values
216                .as_object_mut()
217                .expect("object")
218                .extend(overrides.as_object().expect("object").clone());
219            let mut invalid = request(values);
220            invalid.backend = valid.backend;
221            invalid.gridprecision = valid.gridprecision;
222            invalid.ft_padding = valid.ft_padding;
223            assert_eq!(
224                invalid
225                    .validate()
226                    .expect_err(&format!("unsupported padding scope: {overrides}"))
227                    .parameters,
228                "ft_padding",
229                "{overrides}"
230            );
231        }
232        for factor in [1.2, 2.0] {
233            let mut invalid = valid.clone();
234            invalid.ft_padding = Some(PaddingFactor::new(factor).expect("padding"));
235            assert_eq!(
236                invalid
237                    .validate()
238                    .expect_err("unverified factor")
239                    .parameters,
240                "ft_padding"
241            );
242        }
243        let mut gpu = valid.clone();
244        gpu.backend = BackendRequest::Gpu;
245        assert_eq!(
246            gpu.validate().expect_err("unverified backend").parameters,
247            "ft_padding"
248        );
249        valid.gridprecision = Some(GridPrecision::F32);
250        assert_eq!(
251            valid
252                .validate()
253                .expect_err("unverified precision")
254                .parameters,
255            "ft_padding"
256        );
257    }
