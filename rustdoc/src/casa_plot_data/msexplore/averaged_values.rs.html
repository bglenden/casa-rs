5pub(super) fn finalize_averaged_axis_value(
6    axis: MsAxis,
7    accumulator: &AveragedPointAccumulator,
8    scalar_average: bool,
9    avgtime_enabled: bool,
10    use_x_samples: bool,
11    reference_time_average: bool,
12) -> Option<f64> {
13    if axis.is_visibility_math() {
14        let samples = if use_x_samples {
15            &accumulator.x_visibility
16        } else {
17            &accumulator.y_visibility
18        };
19        return samples.project(axis, scalar_average, reference_time_average);
20    }
21    match axis {
22        MsAxis::Time => {
23            let samples = accumulator
24                .numeric_axis_samples
25                .iter()
26                .find(|(candidate, _)| *candidate == axis)
27                .map(|(_, samples)| samples)?;
28            if avgtime_enabled {
29                if samples.len() == 1 {
30                    return Some(
31                        samples[0]
32                            - average_float_samples(&accumulator.time_interval_samples)? / 2.0,
33                    );
34                }
35                let min_time = samples.iter().copied().min_by(f64::total_cmp)?;
36                let max_time = samples.iter().copied().max_by(f64::total_cmp)?;
37                return Some(min_time + (max_time - min_time) / 2.0);
38            }
39            average_float_samples(samples)
40        }
41        MsAxis::Flag => accumulator
42            .has_flag_samples
43            .then_some(if accumulator.any_flag_sample {
44                1.0
45            } else {
46                0.0
47            }),
48        MsAxis::FlagRow => accumulator
49            .has_flag_row_samples
50            .then_some(if accumulator.any_flag_row { 1.0 } else { 0.0 }),
51        _ => accumulator
52            .numeric_axis_samples
53            .iter()
54            .find(|(candidate, _)| *candidate == axis)
55            .and_then(|(_, samples)| average_float_samples(samples)),
56    }
57}
