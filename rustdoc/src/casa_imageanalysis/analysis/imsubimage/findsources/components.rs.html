26pub(in crate::analysis::imsubimage) struct SkyPlane<'a> {

28    pub(in crate::analysis::imsubimage) sky_order: [usize; 2],

33impl SkyPlane<'_> {
34    pub(in crate::analysis::imsubimage) fn world(
35        &self,
36        pixel: [f64; 2],
37    ) -> Result<[f64; 2], ImageError> {
38        Ok(self
39            .direction
40            .pixel_to_world(self.sky_order.map(|axis| pixel[axis]))?)
41    }
42
43    pub(super) fn component(
44        &self,
45        peak: Peak,
46        ellipse: Option<PixelEllipse>,
47    ) -> Result<SkyComponent, ImageError> {
48        let center = self.world(peak.position)?;
49        let direction = MDirection::from_angles(
50            center[0],
51            center[1],
52            self.direction.conversion_direction_ref(),
53        );
54        let (shape, value, unit) = if let Some(ellipse) = ellipse {
55            let axes = self.world_ellipse(peak.position, center, ellipse, [1.; 2])?;
56            let area = gaussian_area(axes.major_axis(), axes.minor_axis());
57            let (value, unit) = self.integrated_flux(peak.height, area);
58            let shape = ComponentShape::new(
59                direction,
60                ShapeKind::Gaussian {
61                    axes,
62                    errors: Default::default(),
63                },
64            )
65            .map_err(|err| invalid(err.to_string()))?;
66            (shape, value, unit)
67        } else {
68            (
69                ComponentShape::point(direction),
70                peak.height * point_gain(self.brightness),
71                "Jy",
72            )
73        };
74        let flux = Flux::new(
75            [
76                Complex64::new(value, 0.),
77                Complex64::ZERO,
78                Complex64::ZERO,
79                Complex64::ZERO,
80            ],
81            unit,
82            Polarisation::Stokes,
83        )
84        .map_err(|err| invalid(err.to_string()))?;
85        let mut spectrum = SpectralModel::constant(MFrequency::new(1e9, FrequencyRef::LSRK));
86        spectrum
87            .set_frequency_unit(unit_named("GHz"))
88            .map_err(|err| invalid(err.to_string()))?;
89        Ok(SkyComponent::new(flux, shape, spectrum))
90    }

92    pub(in crate::analysis::imsubimage) fn world_ellipse(
93        &self,
94        center_pixel: [f64; 2],
95        center_world: [f64; 2],
96        ellipse: PixelEllipse,
97        pixel_stride: [f64; 2],
98    ) -> Result<Ellipse, ImageError> {
99        let endpoints = [
100            (ellipse.major, ellipse.x_to_y),
101            (ellipse.minor, ellipse.x_to_y + FRAC_PI_2),
102        ];
103        let axis = |(width, angle): (f64, f64)| {
104            let (sin, cos) = angle.sin_cos();
105            let tip = self.world([
106                center_pixel[0] + 0.5 * width * cos * pixel_stride[0],
107                center_pixel[1] + 0.5 * width * sin * pixel_stride[1],
108            ])?;
109            Ok::<_, ImageError>(sky_axis(center_world, tip))
110        };
111        let first = axis(endpoints[0])?;
112        let second = axis(endpoints[1])?;
113        let (major, minor) = if first.0 >= second.0 {
114            (first, second)
115        } else {
116            (second, first)
117        };
118        Ellipse::new(major.0, minor.0, major.1.rem_euclid(PI))
119            .map_err(|err| invalid(err.to_string()))
120    }

122    pub(in crate::analysis::imsubimage) fn integrated_flux(
123        &self,
124        peak: f64,
125        area: f64,
126    ) -> (f64, &'static str) {
127        let increments = self.direction.increment();
128        let pixel_area = (increments[0] * increments[1]).abs();
129        let beam_area = self.beam.map_or_else(
130            || gaussian_area(increments[0].abs(), increments[1].abs()),
131            |beam| beam.area(),
132        );
133        let rad2 = unit_named("rad2");
134        let local = [
135            (
136                "pixel",
137                UnitVal {
138                    factor: pixel_area,
139                    dim: rad2.val().dim,
140                },
141            ),
142            (
143                "beam",
144                UnitVal {
145                    factor: beam_area,
146                    dim: rad2.val().dim,
147                },
148            ),
149        ];
150        // CASA's integrated-temperature and velocity exceptions are keyed
151        // by the original brightness spelling. Other non-Jy integrals fall
152        // back to Jy/pixel, even when their physical dimensions conform.
153        let name = match self.brightness {
154            "K" => "K.rad.rad",
155            "Jy/beam.km/s" => "Jy.km/s",
156            _ => "Jy",
157        };
158        if let Ok(unit) = Unit::with_local_definitions(self.brightness, &local) {
159            let integrated = *unit.val() * *rad2.val();
160            let output = unit_named(name);
161            if integrated.conformant(output.val()) {
162                return (peak * area * integrated.factor / output.val().factor, name);
163            }
164        }
165        (peak * area / pixel_area, "Jy")
166    }
167}

169pub(in crate::analysis::imsubimage) fn gaussian_area(major: f64, minor: f64) -> f64 {
170    PI / (4. * LN_2) * major * minor
171}
172
173fn unit_named(name: &str) -> Unit {
174    Unit::new(name).expect("standard CASA unit")
175}
176
177fn point_gain(brightness: &str) -> f64 {
178    let dimensionless = [("pixel", UnitVal::NODIM), ("beam", UnitVal::NODIM)];
179    let Ok(unit) = Unit::with_local_definitions(brightness, &dimensionless) else {
180        return 1.;
181    };
182    ["Jy", "Jy.km/s", "K"]
183        .into_iter()
184        .map(unit_named)
185        .find(|target| target.conformant(&unit))
186        .map_or(1., |target| unit.val().factor / target.val().factor)
187}

189fn sky_axis(
190    [longitude, latitude]: [f64; 2],
191    [tip_longitude, tip_latitude]: [f64; 2],
192) -> (f64, f64) {
193    let delta_lon = tip_longitude - longitude;
194    let (sin_lat, cos_lat) = latitude.sin_cos();
195    let (sin_tip, cos_tip) = tip_latitude.sin_cos();
196    let east = cos_tip * delta_lon.sin();
197    let north = cos_lat * sin_tip - sin_lat * cos_tip * delta_lon.cos();
198    let cosine = sin_lat * sin_tip + cos_lat * cos_tip * delta_lon.cos();
199    (2. * east.hypot(north).atan2(cosine), east.atan2(north))
200}

202pub(super) fn format_shape_record(component: &mut RecordValue) {
203    let Some(Value::Record(shape)) = component.get_mut("shape") else {
204        return;
205    };
206    // The list's normal serializer uses the units of a table-read shape.
207    // Findsources instead exposes the units assigned by pixel-to-sky fitting.
208    for (name, unit, multiplier) in [
209        ("majoraxis", "arcsec", 60.),
210        ("minoraxis", "arcsec", 60.),
211        ("majoraxiserror", "arcmin", 1.),
212        ("minoraxiserror", "arcmin", 1.),
213        ("positionangleerror", "deg", 1.),
214    ] {
215        if let Some(Value::Record(quantity)) = shape.get_mut(name) {
216            if let Some(Value::Scalar(ScalarValue::Float64(value))) = quantity.get_mut("value") {
217                *value *= multiplier;
218            }
219            quantity.upsert("unit", Value::Scalar(ScalarValue::String(unit.into())));
220        }
221    }
