16#[derive(Clone, Copy)]
17pub(crate) enum PriorSolveBoundary {
18    Accor,
19    Baseline,
20    Bandpass,
21    Delay,
22    Gain,
23    GainSpline,
24    ScalarGain,
25    Leakage { parang_terms: usize },
26    CrossHand { parang_terms: usize },
27    PositionAngle { parang_terms: usize },
28}
29
30impl PriorSolveBoundary {
31    fn viscal_order(self) -> u8 {
32        viscal_order(match self {
33            Self::Baseline => "M Mueller",
34            Self::Bandpass => "B Jones",
35            Self::Delay => "K Jones",
36            Self::Accor | Self::Gain | Self::GainSpline => "G Jones",
37            Self::ScalarGain => "T Jones",
38            Self::Leakage { .. } => "D Jones",
39            Self::CrossHand { .. } => "X Jones",
40            Self::PositionAngle { .. } => "PosAng Jones",
41        })
42    }
43}

63/// Apply priors as the solve reads each selected row.
64pub struct SolvePriorApply {
65    plan: ApplyPlan,
66    chain: ApplyChain,
67    rows: HashMap<usize, usize>,
68    geometry: RefCell<RowGeometryCache>,
69    forward: bool,
70    calwt: bool,
71    share_input_flags: bool,
72}
73
74impl SolvePriorApply {
75    fn new(
76        ms: &MeasurementSet,
77        plan: ApplyPlan,
78        forward: bool,
79        share_input_flags: bool,
80        parang_terms: usize,
81    ) -> Result<Self, ApplyExecutionError> {
82        let rows = plan
83            .selected_rows
84            .iter()
85            .enumerate()
86            .map(|(position, row)| (row.row_index, position))
87            .collect();
88        let mut chain = ApplyChain::load(ms, &plan, forward)?;
89        chain.repeat_parang(parang_terms);
90        let calwt = !forward && applies_preview_weights(&plan, &chain);
91        Ok(Self {
92            plan,
93            chain,
94            rows,
95            geometry: RefCell::default(),
96            forward,
97            calwt,
98            share_input_flags,
99        })
100    }
101
102    pub fn single_dish(ms: &MeasurementSet, plan: ApplyPlan) -> Result<Self, ApplyExecutionError> {
103        let mut priors = Self::new(ms, plan, false, false, 1)?;
104        priors.chain.single_dish_solve = true;
105        Ok(priors)
106    }
107
108    pub(crate) fn correct_single_dish(
109        &self,
110        row: usize,
111        correlations: &[i32],
112        data: &ArrayValue,
113        flags: ArrayValue,
114    ) -> Result<Option<EvaluatedApplyRow>, ApplyExecutionError> {
115        let Some(&position) = self.rows.get(&row) else {
116            return Ok(None);
117        };
118        let converted;
119        let data = if let ArrayValue::Float32(samples) = data {
120            converted = complex_visibility_from_float(samples);
121            &converted
122        } else {
123            data
124        };
125        self.apply(position, correlations, data, flags).map(Some)
126    }
127
128    pub(crate) fn correct(
129        &self,
130        row: usize,
131        correlations: &[i32],
132        data: &ArrayValue,
133        flags: &ArrayValue,
134    ) -> Result<Option<EvaluatedApplyRow>, ApplyExecutionError> {
135        let Some(&position) = self.rows.get(&row) else {
136            return Ok(None);
137        };
138        let flags = if self.share_input_flags {
139            shared_channel_flags(flags).unwrap_or_else(|| flags.clone())
140        } else {
141            flags.clone()
142        };
143        self.apply(position, correlations, data, flags).map(Some)
144    }
145
146    pub(crate) fn corrupt(
147        &self,
148        row: usize,
149        correlations: &[i32],
150        model: ArrayD<Complex32>,
151        flags: &ArrayD<bool>,
152    ) -> Result<(ArrayD<Complex32>, Option<ArrayD<bool>>), ApplyExecutionError> {
153        let Some(&position) = self.rows.get(&row) else {
154            return Ok((model, None));
155        };
156        let applied = self.apply(
157            position,
158            correlations,
159            &ArrayValue::Complex32(model),
160            ArrayValue::Bool(flags.clone()),
161        )?;
162        let ArrayValue::Complex32(model) = applied.corrected_data else {
163            unreachable!("forward application returns complex visibility data")
164        };
165        let ArrayValue::Bool(flags) = applied.flags else {
166            unreachable!("apply flags are boolean")
167        };
168        Ok((model, Some(flags)))
169    }
170
171    fn apply(
172        &self,
173        position: usize,
174        correlations: &[i32],
175        data: &ArrayValue,
176        flags: ArrayValue,
177    ) -> Result<EvaluatedApplyRow, ApplyExecutionError> {
178        let row = &self.plan.selected_rows[position];
179        if !super::availability::available(row, &self.plan) {
180            let shape = IxDyn(data.shape());
181            return Ok(EvaluatedApplyRow {
182                corrected_data: data.clone(),
183                flags: ArrayValue::Bool(ArrayD::from_elem(shape.clone(), true)),
184                model_gains: None,
185                weight_factors: (!self.forward).then(|| ArrayD::zeros(shape)),
186                plot_weights: None,
187            });
188        }
189        let unit_weights = self
190            .calwt
191            .then(|| ArrayValue::Float32(ArrayD::from_elem(IxDyn(&[correlations.len()]), 1.0)));
192        let result = apply_row(
193            row,
194            ExecutionRowInputs {
195                correlation_types: correlations,
196                data,
197                original_flags: Some(&flags),
198                original_weight: unit_weights.as_ref(),
199                original_weight_spectrum: None,
200                has_weight_spectrum: self.calwt,
201                preserved_corrected_data: None,
202                preserved_weight_spectrum: None,
203                channel_grid: None,
204                forward_model: self.forward,
205                skip_unavailable_terms: false,
206                trial: false,
207                parang_calwt: false,
208            },
209            &self.plan,
210            &self.chain,
211            Some(&mut self.geometry.borrow_mut()),
212            None,
213            None,
214        )?;
215        Ok(EvaluatedApplyRow {
216            corrected_data: result.corrected_data,
217            flags: result.updated_flags.unwrap_or(flags),
218            model_gains: None,
219            weight_factors: match result.updated_weight_spectrum {
220                Some(ArrayValue::Float32(factors)) => Some(factors),
221                _ => None,
222            },
223            plot_weights: None,
224        })
225    }
226}

228pub(crate) fn evaluate_solve_priors(
229    ms: &MeasurementSet,
230    mut plan: ApplyPlan,
231    boundary: PriorSolveBoundary,
232    share_input_flags: bool,
233) -> Result<PreappliedRows, ApplyExecutionError> {
234    let deferred_data = matches!(
235        boundary,
236        PriorSolveBoundary::Leakage { .. }
237            | PriorSolveBoundary::CrossHand { .. }
238            | PriorSolveBoundary::PositionAngle { .. }
239    );
240    let parang_terms = match boundary {
241        PriorSolveBoundary::Leakage { parang_terms }
242        | PriorSolveBoundary::CrossHand { parang_terms }
243        | PriorSolveBoundary::PositionAngle { parang_terms } => parang_terms,
244        _ => usize::from(plan.parang),
245    };
246    let polarized = deferred_data || has_polarization_matrix(&plan);
247    let baseline = plan
248        .calibration_tables
249        .iter()
250        .any(|table| super::baseline::is_subtype(&table.summary.table_subtype));
251    let spectral_opacity = plan
252        .calibration_tables
253        .iter()
254        .any(|table| table.summary.table_subtype == "TfOpac");
255    let gain_boundary = matches!(
256        boundary,
257        PriorSolveBoundary::Gain | PriorSolveBoundary::ScalarGain
258    );
259    let solve_order = boundary.viscal_order();
260    // VisEquation::collapse2 corrupts MODEL_DATA with any prior at or after
261    // the solved term. Its weights then come from division by that model,
262    // regardless of those priors' calwt settings.
263    let gain_model_prior = gain_boundary
264        && plan
265            .calibration_tables
266            .iter()
267            .any(|table| viscal_order(&table.summary.table_subtype) >= solve_order);
268    let type_ordered = polarized
269        || baseline
270        || spectral_opacity
271        || gain_model_prior
272        || matches!(
273            boundary,
274            PriorSolveBoundary::Accor
275                | PriorSolveBoundary::Baseline
276                | PriorSolveBoundary::GainSpline
277        );
278    if !type_ordered && gain_boundary {
279        return evaluate_apply_rows(ms, &plan, share_input_flags).map(Into::into);
280    }
281    // The measurement equation corrects DATA with earlier terms and corrupts
282    // MODEL_DATA with the remaining terms. The channel-dependent opacity
283    // then enters the solve weights through the model normalization.
284    let (data_tables, mut model_tables): (Vec<_>, Vec<_>) =
285        std::mem::take(&mut plan.calibration_tables)
286            .into_iter()
287            .partition(|table| {
288                let subtype = table.summary.table_subtype.as_str();
289                if type_ordered {
290                    return viscal_order(subtype) < solve_order;
291                }
292                match subtype {
293                    "KAntPos Jones" => true,
294                    "B Jones" | "BPOLY" => matches!(boundary, PriorSolveBoundary::Delay),
295                    _ => false,
296                }
297            });
298    let forward_parang = type_ordered && plan.parang && viscal_order("P Jones") >= solve_order;
299    plan.parang &= !forward_parang;
300    plan.calibration_tables = data_tables;
301    let data_prior = (deferred_data && (!plan.calibration_tables.is_empty() || plan.parang))
302        .then(|| SolvePriorApply::new(ms, plan.clone(), false, share_input_flags, parang_terms))
303        .transpose()?;
304    let mut rows = if deferred_data {
305        HashMap::new()
306    } else {
307        evaluate_apply_rows(ms, &plan, share_input_flags)?
308    };
309    if model_tables.is_empty() && !forward_parang {
310        return Ok(PreappliedRows {
311            data: rows,
312            data_prior,
313            model: None,
314            polarized,
315        });
316    }
317    for table in &mut model_tables {
318        table.calwt = false;
319    }
320    plan.calibration_tables = model_tables;
321    plan.parang = forward_parang;
322    // Additive offsets must operate on the actual model, just as matrix terms
323    // must. Applying them to unit visibilities cannot represent a gain factor.
324    if polarized || forward_parang || super::additive::in_plan(&plan) {
325        return Ok(PreappliedRows {
326            data: rows,
327            data_prior,
328            model: Some(SolvePriorApply::new(
329                ms,
330                plan,
331                true,
332                share_input_flags && !gain_boundary,
333                parang_terms,
334            )?),
335            polarized,
336        });
337    }
338    for_each_apply_row(
339        ms,
340        &plan,
341        "DATA",
342        true,
343        share_input_flags,
344        PreviewWeightSource::UnitFactors,
345        None,
346        |index, row| {
347            let data = rows.get_mut(&index).expect("same selection for both sides");
348            let ArrayValue::Bool(data_flags) = &mut data.flags else {
349                unreachable!("apply flags are boolean")
350            };
351            let ArrayValue::Bool(model_flags) = row.flags else {
352                unreachable!("model calibration flags are boolean")
353            };
354            data_flags.zip_mut_with(&model_flags, |data, &model| *data |= model);
355            let ArrayValue::Complex32(gains) = row.corrected_data else {
356                unreachable!("unit visibilities are complex")
357            };
358            data.model_gains = Some(gains);
359        },
360    )?;
361    Ok(rows.into())
362}
