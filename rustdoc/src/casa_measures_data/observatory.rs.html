9/// A single observatory row from casacore `geodetic/Observatories`.
10#[derive(Debug, Clone, PartialEq)]
11pub struct ObservatoryEntry {
12    /// Table `MJD` column.
13    pub mjd: f64,
14    /// Table `Name` column.
15    pub name: String,
16    /// Table `Type` column.
17    pub observatory_type: String,
18    /// Table `Long` column, in degrees.
19    pub longitude_deg: f64,
20    /// Table `Lat` column, in degrees.
21    pub latitude_deg: f64,
22    /// Table `Height` column, in metres.
23    pub height_m: f64,
24    /// Table `X` column, in metres.
25    pub x_m: f64,
26    /// Table `Y` column, in metres.
27    pub y_m: f64,
28    /// Table `Z` column, in metres.
29    pub z_m: f64,
30    /// Table `Source` column.
31    pub source: String,
32    /// Table `Comment` column.
33    pub comment: String,
34    /// Table `AntennaResponses` column.
35    pub antenna_responses: String,
36}
37
38impl ObservatoryEntry {
39    /// The observatory's position, as casacore's `MeasTable` reads it: the
40    /// `Long`, `Lat` and `Height` columns for either `Type`.
41    ///
42    /// For a `WGS84` entry they are geodetic longitude, latitude and height
43    /// above the ellipsoid. For an `ITRF` entry casacore takes them as a
44    /// spherical vector, `Height` its length from the Earth's centre, and
45    /// does not read `X`, `Y` and `Z`; so does this. Where an entry's two
46    /// descriptions disagree, its position is CASA's, recorded in
47    /// `docs/CASA (C++) bugs.md`. `None` for any other `Type`.
48    pub fn position(&self) -> Option<ObservatoryPosition> {
49        let observatory_type = self.observatory_type.trim();
50        let longitude_rad = self.longitude_deg.to_radians();
51        let latitude_rad = self.latitude_deg.to_radians();
52        if observatory_type.eq_ignore_ascii_case("ITRF") {
53            let (sin_lon, cos_lon) = longitude_rad.sin_cos();
54            let (sin_lat, cos_lat) = latitude_rad.sin_cos();
55            // casacore's MVPosition order: the unit vector, then its length.
56            Some(ObservatoryPosition::Itrf {
57                x_m: cos_lon * cos_lat * self.height_m,
58                y_m: sin_lon * cos_lat * self.height_m,
59                z_m: sin_lat * self.height_m,
60            })
61        } else if observatory_type.eq_ignore_ascii_case("WGS84") {
62            Some(ObservatoryPosition::Wgs84 {
63                longitude_rad,
64                latitude_rad,
65                height_m: self.height_m,
66            })
67        } else {
68            None
69        }
70    }
71}

73/// In-memory catalog of observatory metadata.
74#[derive(Debug, Clone, PartialEq)]
75pub struct ObservatoryCatalog {
76    entries: Vec<ObservatoryEntry>,
77}
78
79impl ObservatoryCatalog {
80    /// Build a catalog from explicitly provided entries.
81    pub fn from_entries(entries: Vec<ObservatoryEntry>) -> Self {
82        Self { entries }
83    }
84
85    /// Iterate over the catalog entries in source order.
86    pub fn iter(&self) -> impl Iterator<Item = &ObservatoryEntry> {
87        self.entries.iter()
88    }
89
90    /// Borrow all entries.
91    pub fn entries(&self) -> &[ObservatoryEntry] {
92        &self.entries
93    }
94
95    /// The observatory `name` finds, as casacore's `MeasTable` finds it: the
96    /// entry of that name ignoring case, else the one entry the name agrees
97    /// with over the shorter of their lengths ([`unique_prefix_position`]).
98    pub fn get(&self, name: &str) -> Option<&ObservatoryEntry> {
99        unique_prefix_position(name, &self.entries, |entry| entry.name.as_str())
100            .map(|at| &self.entries[at])
101    }
102
103    pub(crate) fn retained_heap_bytes(&self) -> Option<usize> {
104        let mut bytes = self
105            .entries
106            .capacity()
107            .checked_mul(size_of::<ObservatoryEntry>())?;
108        for entry in &self.entries {
109            bytes = bytes
110                .checked_add(entry.name.capacity())?
111                .checked_add(entry.observatory_type.capacity())?
112                .checked_add(entry.source.capacity())?
113                .checked_add(entry.comment.capacity())?
114                .checked_add(entry.antenna_responses.capacity())?;
115        }
116        Some(bytes)
117    }
118}

120#[cfg(test)]
121mod tests {
122    use super::*;
123
124    fn entry(observatory_type: &str, long_lat_height: [f64; 3], xyz: [f64; 3]) -> ObservatoryEntry {
125        ObservatoryEntry {
126            mjd: 0.0,
127            name: "SMA".to_owned(),
128            observatory_type: observatory_type.to_owned(),
129            longitude_deg: long_lat_height[0],
130            latitude_deg: long_lat_height[1],
131            height_m: long_lat_height[2],
132            x_m: xyz[0],
133            y_m: xyz[1],
134            z_m: xyz[2],
135            source: String::new(),
136            comment: String::new(),
137            antenna_responses: String::new(),
138        }
139    }
140
141    /// The SMA row of the measures data's `Observatories` table, whose `X`,
142    /// `Y`, `Z` lie 13.8 km from its `Long`, `Lat`, `Height`, and the
143    /// Cartesian position CASA 6.7.6.14 gives SMA
144    /// (`crates/casa-simulation/tests/data/readantenna_casa.json`).
145    #[test]
146    fn an_itrf_entry_is_where_casacore_places_it() {
147        let sma = entry(
148            "ITRF",
149            [-155.477_522_222_222_2, 19.824_205_555_555_555, 6_382_248.0],
150            [-5_464_555.493, -2_492_927.989, 2_150_797.176],
151        );
152        let Some(ObservatoryPosition::Itrf { x_m, y_m, z_m }) = sma.position() else {
153            panic!("an ITRF position");
154        };
155        let casa = [
156            -5_462_448.949_394_94,
157            -2_491_969.916_598_48,
158            2_164_446.105_479_709,
159        ];
160        for (found, casa) in [x_m, y_m, z_m].into_iter().zip(casa) {
161            assert!((found - casa).abs() < 1e-6, "{found} {casa}");
162        }
163    }
164
165    #[test]
166    fn a_wgs84_entry_is_geodetic_and_another_type_has_no_position() {
167        let alma = entry("WGS84 ", [-67.754_929, -23.022_886, 5056.8], [0.0; 3]);
168        assert_eq!(
169            alma.position(),
170            Some(ObservatoryPosition::Wgs84 {
171                longitude_rad: -1.182_546_595_504_989_2,
172                latitude_rad: -0.401_825_164_011_307_2,
173                height_m: 5056.8,
174            })
175        );
176        assert_eq!(entry("J2000", [0.0; 3], [0.0; 3]).position(), None);
177    }
