10pub(super) fn bin_widths(
11    ms: &MeasurementSet,
12    rows: &[SelectedSolveRow],
13    interval: Option<BandpassFrequencyInterval>,
14) -> Result<BTreeMap<i32, usize>, BandpassSolveError> {
15    match interval {
16        Some(BandpassFrequencyInterval::Channels(count)) if count > 1.0 => Ok(rows
17            .iter()
18            .map(|row| (row.data_spw_id, count as usize))
19            .collect()),
20        Some(BandpassFrequencyInterval::Hertz(hertz)) => bandwidth_bin_widths(ms, hertz),
21        _ => Ok(BTreeMap::new()),
22    }
23}
24
25fn bandwidth_bin_widths(
26    ms: &MeasurementSet,
27    hertz: f64,
28) -> Result<BTreeMap<i32, usize>, BandpassSolveError> {
29    let fail = |reason: String| BandpassSolveError::UnsupportedConfiguration { reason };
30    let window = ms
31        .spectral_window()
32        .map_err(|error| fail(error.to_string()))?;
33    let mut widths = BTreeMap::new();
34    for spw in 0..window.row_count() {
35        let channels = window
36            .chan_width(spw)
37            .map_err(|error| fail(error.to_string()))?;
38        let width = channels
39            .first()
40            .filter(|width| width.is_finite() && **width != 0.0)
41            .ok_or_else(|| fail(format!("SPW {spw} has no valid first channel width")))?;
42        widths.insert(spw as i32, (hertz / width.abs()).floor().max(1.0) as usize);
43    }
44    // Once the layer is installed, even windows with a width of one use
45    // the frequency averager's scalar weights for the time centroid.
46    if widths.values().all(|&width| width == 1) {
47        widths.clear();
48    }
49    Ok(widths)
50}

52pub(super) fn write(
53    path: &Path,
54    rows: &[BandpassSolutionRow],
55    channel_bin_widths: &BTreeMap<i32, usize>,
56    selected_rows: &[SelectedSolveRow],
57    combine_windows: bool,
58) -> Result<(), BandpassSolveError> {
59    let mut table = Table::open(TableOptions::new(path)).map_err(metadata_error)?;
60    let selections = selected_rows
61        .iter()
62        .map(|row| (row.data_spw_id, row.channel_selection.as_ref()))
63        .collect::<BTreeMap<_, _>>();
64    let grid = |table: &Table, spw| {
65        select_grid(
66            table,
67            spw as usize,
68            selections.get(&spw).copied().flatten(),
69            channel_bin_widths.get(&spw).copied().unwrap_or(1),
70        )
71    };
72    let aggregate = if combine_windows && selections.len() > 1 {
73        let mut selected = selections.keys().copied();
74        let reference = selected.next().expect("nonempty windows");
75        let mut combined = grid(&table, reference)?;
76        let frequency_offset = combined.frequencies[0];
77        let width_scale = combined.widths[0];
78        for index in 0..combined.frequencies.len() {
79            let width = combined.widths[index] / width_scale;
80            combined.frequencies[index] = (combined.frequencies[index] - frequency_offset) * width;
81            combined.effective_bandwidths[index] = width;
82            combined.widths[index] = width * width;
83        }
84        for spw in selected {
85            combined.accumulate(&grid(&table, spw)?, frequency_offset, width_scale)?;
86        }
87        for index in 0..combined.frequencies.len() {
88            let bandwidth = combined.effective_bandwidths[index];
89            if !bandwidth.is_finite() || bandwidth <= 0.0 {
90                return Err(metadata_error(
91                    "Cannot combine windows with invalid channel widths",
92                ));
93            }
94            combined.frequencies[index] =
95                combined.frequencies[index] / bandwidth + frequency_offset;
96            combined.widths[index] = combined.widths[index] / bandwidth * width_scale;
97            combined.effective_bandwidths[index] *= width_scale;
98        }
99        Some((reference, combined))
100    } else {
101        None
102    };
103    for row in 0..table.row_count() {
104        let spw = row as i32;
105        let present = rows.iter().any(|r| r.spw_id == spw);
106        table
107            .cell_accessor_mut(row, "FLAG_ROW")
108            .and_then(|mut c| c.set(Value::Scalar(ScalarValue::Bool(!present))))
109            .map_err(metadata_error)?;
110        if let Some((reference, combined)) = &aggregate {
111            if *reference == spw {
112                combined.write(&mut table, row)?;
113            }
114        } else if present {
115            grid(&table, spw)?.write(&mut table, row)?;
116        }
117    }
118    table.save(TableOptions::new(path)).map_err(metadata_error)
119}

121struct ChannelGrid {
122    frequencies: Vec<f64>,
123    widths: Vec<f64>,
124    effective_bandwidths: Vec<f64>,
125}
126
127fn select_grid(
128    table: &Table,
129    row: usize,
130    selection: Option<&ChannelSelection>,
131    bin_width: usize,
132) -> Result<ChannelGrid, BandpassSolveError> {
133    let array = |column| -> Result<Vec<f64>, BandpassSolveError> {
134        let ArrayValue::Float64(values) = table
135            .cell_accessor(row, column)
136            .and_then(|c| c.array())
137            .map_err(metadata_error)?
138        else {
139            return Err(metadata_error(format!("SPW {row} {column} is not Float64")));
140        };
141        Ok(values.iter().copied().collect())
142    };
143    let frequencies = array("CHAN_FREQ")?;
144    let widths = array("CHAN_WIDTH")?;
145    if frequencies.len() != widths.len() {
146        return Err(metadata_error("SPW channel width shape mismatch"));
147    }
148    let channels = crate::solve::grouping::selected_channels(selection, frequencies.len())?;
149    let bins = channels.chunks(bin_width);
150    let frequencies = bins
151        .clone()
152        .map(|bin| bin.iter().map(|&channel| frequencies[channel]).sum::<f64>() / bin.len() as f64)
153        .collect();
154    let widths = bins
155        .map(|bin| bin.iter().map(|&channel| widths[channel].abs()).sum())
156        .collect::<Vec<_>>();
157    Ok(ChannelGrid {
158        frequencies,
159        effective_bandwidths: widths.clone(),
160        widths,
161    })
162}
163
164impl ChannelGrid {
165    fn accumulate(
166        &mut self,
167        other: &Self,
168        frequency_offset: f64,
169        width_scale: f64,
170    ) -> Result<(), BandpassSolveError> {
171        if self.frequencies.len() != other.frequencies.len() {
172            return Err(metadata_error(
173                "Cannot combine spectral windows with different channel counts",
174            ));
175        }
176        for index in 0..self.frequencies.len() {
177            let width = other.widths[index] / width_scale;
178            self.frequencies[index] += (other.frequencies[index] - frequency_offset) * width;
179            self.widths[index] += width * width;
180            self.effective_bandwidths[index] += width;
181        }
182        Ok(())
183    }
184
185    fn write(&self, table: &mut Table, row: usize) -> Result<(), BandpassSolveError> {
186        let count = self.frequencies.len();
187        let total: f64 = self.frequencies.iter().sum();
188        // SolvableVisCal::setCTFrequencies aliases resolution to frequency
189        // and sums frequencies as bandwidth. CASA requires exact agreement
190        // when appending; these fields are not used to apply the solutions.
191        for (column, values) in [
192            ("CHAN_FREQ", &self.frequencies),
193            ("RESOLUTION", &self.frequencies),
194            ("CHAN_WIDTH", &self.widths),
195            ("EFFECTIVE_BW", &self.effective_bandwidths),
196        ] {
197            table
198                .cell_accessor_mut(row, column)
199                .and_then(|mut c| c.set(Value::Array(ArrayValue::from_f64_vec(values.clone()))))
200                .map_err(metadata_error)?;
201        }
202        for (column, value) in [
203            ("NUM_CHAN", Value::Scalar(ScalarValue::Int32(count as i32))),
204            (
205                "TOTAL_BANDWIDTH",
206                Value::Scalar(ScalarValue::Float64(total)),
207            ),
208        ] {
209            table
210                .cell_accessor_mut(row, column)
211                .and_then(|mut c| c.set(value))
212                .map_err(metadata_error)?;
213        }
214        Ok(())
215    }
216}
217
218fn metadata_error(error: impl std::fmt::Display) -> BandpassSolveError {
219    BandpassSolveError::UnsupportedConfiguration {
220        reason: error.to_string(),
221    }
