33    #[pyo3(signature = (), text_signature = "($self)")]
34    fn activityrec(&self, py: Python<'_>) -> PyResult<Py<PyAny>> {
35        let record = self.with_lifecycle(py, |state| Ok(state.activity_record()))?;
36        let value =
37            casacore_value_to_casa(&Value::Record(record)).map_err(PyRuntimeError::new_err)?;
38        reply_value(py, &value)
39    }

41    #[pyo3(signature = (r#type = None, t = None, table = None, field = None, interp = None, select = None, calwt = None, spwmap = None, opacity = None), text_signature = "($self, type='', t=0.0, table='', field=[], interp='linear', select='', calwt=False, spwmap=[-1], opacity=[0.0])")]
42    #[allow(clippy::too_many_arguments)]
43    fn setapply(
44        &self,
45        py: Python<'_>,
46        #[pyo3(from_py_with = supplied)] r#type: Option<Bound<'_, PyAny>>,
47        #[pyo3(from_py_with = supplied)] t: Option<Bound<'_, PyAny>>,
48        #[pyo3(from_py_with = supplied)] table: Option<Bound<'_, PyAny>>,
49        #[pyo3(from_py_with = supplied)] field: Option<Bound<'_, PyAny>>,
50        #[pyo3(from_py_with = supplied)] interp: Option<Bound<'_, PyAny>>,
51        #[pyo3(from_py_with = supplied)] select: Option<Bound<'_, PyAny>>,
52        #[pyo3(from_py_with = supplied)] calwt: Option<Bound<'_, PyAny>>,
53        #[pyo3(from_py_with = supplied)] spwmap: Option<Bound<'_, PyAny>>,
54        #[pyo3(from_py_with = supplied)] opacity: Option<Bound<'_, PyAny>>,
55    ) -> PyResult<bool> {
56        let kind = text(r#type, "type", "")?;
57        let pending_time = number(t, "setapply", 3, 0.0).is_err();
58        let mut spec = ApplyCalibrationTableSpec::new(text(table, "table", "")?);
59        let field = listing_selector(field.as_ref())?;
60        let interpolation = text(interp, "interp", "linear")?;
61        // The tool wrapper forwards an empty SPW selector for traditional apply.
62        let _selection = text(select, "select", "")?;
63        if let Some(value) = calwt {
64            spec.calwt = strict_bool(&value, "setapply", 8)?;
65        }
66        let (windows, pending_windows) = window_map_with_status(spwmap, &[-1])?;
67        spec.spwmap = windows;
68        let tableless_opacity = spec.path.as_os_str().is_empty()
69            && (kind.eq_ignore_ascii_case("TOpac") || kind.eq_ignore_ascii_case("TfOpac"));
70        let mut opacity_values = Vec::new();
71        let pending_opacity = opacity.is_some_and(|value| {
72            if tableless_opacity && let Ok(list) = value.cast::<PyList>() {
73                for item in list.iter() {
74                    let start = opacity_values.len();
75                    if append_double_vector(&item, &mut opacity_values, 1).is_err() {
76                        // CASA treats an unsupported list element as zero.
77                        opacity_values.truncate(start);
78                        opacity_values.push(0.0);
79                    }
80                }
81                return false;
82            }
83            value.is_instance_of::<PyTuple>()
84                || append_double_vector(&value, &mut opacity_values, 0).is_err()
85        });
86        let mut warn_about_opacity = false;
87        let warning = &mut warn_about_opacity;
88        let configured = self.attached(py, move |state| {
89            if tableless_opacity {
90                let result = state.set_tableless_opacity(&kind, opacity_values);
91                *warning = !matches!(result, Err(CalibraterLifecycleError::Unattached));
92                result
93            } else {
94                state.set_traditional_apply(spec, &kind, &field, &interpolation)
95            }
96        });
97        if warn_about_opacity {
98            emit(
99                py,
100                LOGGER,
101                &[
102                    LogLine::warning("Invoking opacity application without a caltable (e.g.,"),
103                    LogLine::warning(
104                        " via opacity=[...] in calibration tasks) will soon be disabled.",
105                    ),
106                    LogLine::warning(" Please begin using gencal to generate an opacity caltable,"),
107                    LogLine::warning(" and then supply that table in the standard manner."),
108                ],
109            )?;
110        }
111        let configured = configured?;
112        // CASA's SWIG numerics can set an exception and still call C++ with a
113        // fallback value. Registration is visible when Python reports it.
114        if pending_time || pending_windows || pending_opacity {
115            return Err(setapply_conversion_error());
116        }
117        Ok(configured)
118    }

120    #[pyo3(signature = (applymode = None), text_signature = "($self, applymode='')")]
121    fn correct(
122        &self,
123        py: Python<'_>,
124        #[pyo3(from_py_with = supplied)] applymode: Option<Bound<'_, PyAny>>,

126        let mode = text(applymode, "applymode", "")?;

135    #[pyo3(signature = (filein), text_signature = "($self, filein)")]
136    fn parsecallibfile(&self, py: Python<'_>, filein: &Bound<'_, PyAny>) -> PyResult<Py<PyAny>> {
137        let mut filename = text(Some(filein.clone()), "filein", "")?;
138        if let Some(end) = filename.find('\0') {
139            filename.truncate(end);
140        }
141        if filename.is_empty() {
142            return Err(PyRuntimeError::new_err(
143                "Cal Library parser failed to open file",
144            ));
145        }
146        let record = py
147            .detach(|| load_callibrary_record(&filename))
148            .map_err(|error| {
149                PyRuntimeError::new_err(match error {
150                    CallibError::ReadFile { source, .. }
151                        if source.kind() == ErrorKind::NotFound =>
152                    {
153                        format!("Specified cal library file ('{filename}') does not exist!")
154                    }
155                    CallibError::ReadFile { .. } => "Cal Library parser failed to open file".into(),
156                    CallibError::ParseLine { line_number, .. } => {
157                        format!("Cal Library file: Parse error on line {line_number}: syntax error")
158                    }
159                    error => error.to_string(),
160                })
161            })?;
162        let value =
163            casacore_value_to_casa(&Value::Record(record)).map_err(PyRuntimeError::new_err)?;
164        reply_value(py, &value)
165    }
166    #[pyo3(signature = (), text_signature = "($self)")]
167    fn returndict(&self, py: Python<'_>) -> PyResult<Py<PyAny>> {
168        let record = self.with_lifecycle(py, |state| match state.configuration_record() {
169            Ok(record) => Ok(Some(record)),
170            Err(CalibraterLifecycleError::Unattached) => Ok(None),
171            Err(error) => Err(runtime_error(error)),
172        })?;
173        if let Some(record) = record {
174            let value =
175                casacore_value_to_casa(&Value::Record(record)).map_err(PyRuntimeError::new_err)?;
176            return reply_value(py, &value);
177        }
178        emit(
179            py,
180            LOGGER,
181            &[LogLine::error("Must first open a MeasurementSet.")],
182        )?;
183        Ok(PyDict::new(py).into_any().unbind())
184    }

186    #[pyo3(signature = (callib = None), text_signature = "($self, callib={})")]
187    fn setcallib(
188        &self,
189        py: Python<'_>,
190        #[pyo3(from_py_with = supplied)] callib: Option<Bound<'_, PyAny>>,
191    ) -> PyResult<bool> {
192        let record = library_argument(callib.as_ref())?;
193        self.attached(py, move |state| state.set_callibrary(&record))
194    }
195
196    #[pyo3(signature = (callib = None), text_signature = "($self, callib={})")]
197    fn validatecallib(
198        &self,
199        py: Python<'_>,
200        #[pyo3(from_py_with = supplied)] callib: Option<Bound<'_, PyAny>>,
201    ) -> PyResult<bool> {
202        let record = library_argument(callib.as_ref())?;
203        let diagnostics = py
204            .detach(|| validate_callibrary_record(&record))
205            .map_err(PyRuntimeError::new_err)?;
206        let lines = diagnostics
207            .into_iter()
208            .map(LogLine::error)
209            .collect::<Vec<_>>();
210        emit(py, LOGGER, &lines)?;
211        Ok(true)
212    }

215fn setapply_conversion_error() -> PyErr {
216    PySystemError::new_err(
217        "<built-in function calibrater_setapply> returned a result with an exception set",
218    )
219}
220
221fn library_argument(value: Option<&Bound<'_, PyAny>>) -> PyResult<RecordValue> {
222    let Some(value) = value else {
223        return Ok(RecordValue::default());
224    };
225    let record = value
226        .cast::<PyDict>()
227        .map_err(|_| PyTypeError::new_err("_callib is not a dictionary"))?;
228    let converted = parameter_value_to_casacore(&library_parameters(record, 0)?)
229        .map_err(PyRuntimeError::new_err)?;
230    let Value::Record(record) = converted else {
231        unreachable!("library argument was converted as a record");
232    };
233    Ok(record)
234}
235
236fn library_parameters(record: &Bound<'_, PyDict>, depth: usize) -> PyResult<ParameterValue> {
237    if depth > 32 {
238        return Err(PyRuntimeError::new_err("record nesting is too deep"));
239    }
240    let fields = record
241        .iter()
242        .filter(|(_, value)| !value.is_none())
243        .map(|(key, value)| {
244            let value = match value.cast::<PyDict>() {
245                Ok(nested) => library_parameters(nested, depth + 1)?,
246                Err(_) => parameter_value(&value)?,
247            };
248            Ok((key.extract::<String>()?, value))
249        })
250        .collect::<PyResult<_>>()?;
251    Ok(ParameterValue::Table(fields))
