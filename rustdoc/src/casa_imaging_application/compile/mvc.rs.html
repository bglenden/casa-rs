9/// Construct the cube over the complete selected band CASA's
10/// `PyMtmfsViaCubeSynthesisImager::determineFreqRange` surveyed. Its output
11/// depth is independent of the input selection, and its nearest-neighbor
12/// cube forms the dirty normals that the Taylor imager subsequently uses.
13pub(crate) fn cube_request(
14    request: &ImagingRequest,
15    [lower, upper]: [f64; 2],
16    context: &crate::RunContext,
17) -> Result<casa_imaging_runtime::Admitted<ImagingRequest>, super::PrepareError> {
18    let channels = resolve_cube_depth(request.mvc_output_channels, [lower, upper], request.nterms)?;
19    let reservation = casa_imaging_runtime::admit(
20        &context.host,
21        &context.policy,
22        &casa_imaging_runtime::Demand {
23            phase: "MVC cube request",
24            memory: super::mvc_request::clone_demand(request)?,
25        },
26    )?;
27    let mut cube = request.clone();
28    cube.mvc_output_channels = Some(channels);
29    cube.mvc_auto_channels = false;
30    let channels = channels.get();
31    cube.specmode = SpecMode::Cube;
32    cube.deconvolver = Deconvolver::Hogbom;
33    cube.nterms = 1;
34    cube.niter = 0;
35    cube.start = Some(format!("{lower}Hz"));
36    cube.width = Some(format!("{}Hz", (upper - lower) / channels as f64));
37    cube.interpolation = CubeInterpolation::Nearest;
38    cube.restoration = false;
39    cube.write_pb = true;
40    Ok(casa_imaging_runtime::Admitted {
41        value: cube,
42        reservation,
43    })
44}

46/// CASA chooses roughly ten planes per octave, but at least `nterms + 1`
47/// when its bandwidth-derived count is below five. A supplied count is used
48/// as-is; every count must cover the requested Taylor terms.
49fn resolve_cube_depth(
50    requested: Option<std::num::NonZeroUsize>,
51    [lower, upper]: [f64; 2],
52    terms: usize,
53) -> Result<std::num::NonZeroUsize, super::PrepareError> {
54    let invalid = |reason| super::PrepareError::MvcCubeDepth { reason };
55    if !lower.is_finite() || !upper.is_finite() || lower <= 0.0 || upper < lower {
56        return Err(invalid("the selected frequency band is invalid"));
57    }
58    let depth = if let Some(count) = requested {
59        count.get()
60    } else {
61        let estimate = ((upper - lower) / (0.1 * lower)).floor();
62        if !estimate.is_finite() || estimate >= usize::MAX as f64 {
63            return Err(invalid("the bandwidth-derived channel count is too large"));
64        }
65        let estimate = estimate as usize;
66        if estimate < 5 {
67            terms
68                .checked_add(1)
69                .ok_or_else(|| invalid("too many Taylor terms"))?
70        } else {
71            estimate
72        }
73    };
74    if depth < terms {
75        return Err(invalid("nchan must be at least nterms"));
76    }
77    std::num::NonZeroUsize::new(depth).ok_or_else(|| invalid("nchan must be positive"))
78}

114impl TaylorView {
115    /// Compile this view against the already compiled cube observation. Its
116    /// selection is shared; no MeasurementSet is opened or surveyed again.
117    #[cfg_attr(
118        not(test),
119        allow(dead_code, reason = "MVC execution is being wired in #1069")
120    )]
121    pub(crate) fn compile_from_cube(
122        self,
123        cube: &casa_imaging_model::CompiledProblem,
124    ) -> Result<CompiledTaylorView, casa_imaging_model::CompileProblemError> {
125        let problem = casa_imaging_model::compile(casa_imaging_model::ProblemInput::new(
126            self.specification,
127            self.geometry,
128            cube.observation().clone(),
129            self.model_lifecycle,
130        ))?;
131        Ok(CompiledTaylorView {
132            problem,
133            coordinates: self.coordinates,
134            expansion_reference_hz: self.expansion_reference_hz,
135            reservation: self.reservation,
136        })
137    }
138}

154/// Build the Taylor image view without opening or surveying the observation
155/// again. CASA keeps its Taylor image WCS at the band centre while the
156/// supplied `reffreq` governs its cube-to-Taylor expansion.
157pub(super) fn prepare_taylor(inputs: TaylorInputs<'_>) -> Result<TaylorView, super::PrepareError> {
158    use casa_imaging_model::{ModelBounds, ModelLifecycleRequirements, NumericPrecision};
159    let TaylorInputs {
160        parent,
161        context,
162        band,
163        surveyed,
164        frame,
165        centre,
166        observation_info,
167        object_name,
168        instrument,
169        has_primary_beam,
170    } = inputs;
171    // This initial MVC implementation owns exactly one fixed-centre image
172    // domain and one Stokes-I, two-term standard-gridder problem. In
173    // particular, an outlier file or facets would make the small metadata
174    // envelope below scale with data it does not yet account for.
175    check_view_scope(parent, surveyed.stores.len(), centre.ephemeris.is_some())?;
176    let reservation = casa_imaging_runtime::admit(
177        &context.host,
178        &context.policy,
179        &casa_imaging_runtime::Demand {
180            phase: "MVC Taylor view",
181            memory: view_demand(parent, observation_info, object_name)?,
182        },
183    )?;
184    let expansion_reference_hz = parent.reffreq.unwrap_or_else(|| band.gridded.midpoint_hz());
185    let request = parent;
186    let mut spectral = super::spectral::prepare_axis(
187        request,
188        (surveyed.stores, surveyed.reference),
189        Some(band),
190        frame,
191        None,
192        None,
193    )?;
194    // CASA's `cube2tt` copies the cube image coordinates and changes only
195    // the spectral reference pixel, value and increment. The rest-frequency
196    // label therefore comes from the cube, not the independently prepared
197    // continuum axis (`PyMtmfsViaCubeSynthesisImager::copy_image`).
198    spectral.image.rest_frequency_hz = surveyed.spectral.image.rest_frequency_hz;
199    let taylor_surveyed = super::Surveyed {
200        stores: surveyed.stores,
201        reference: surveyed.reference,
202        catalogue: surveyed.catalogue,
203        msv2: surveyed.msv2,
204        engine: surveyed.engine,
205        survey: surveyed.survey,
206        spectral: &spectral,
207    };
208    let domains = super::domains::prepare_domains(
209        request,
210        centre,
211        (&spectral.image, spectral.output_channels),
212        (observation_info, object_name),
213        super::domains::MaskPreparation::GeometryOnly,
214    )?;
215    let mut specification_inputs = super::specification_inputs(
216        request,
217        &taylor_surveyed,
218        instrument,
219        has_primary_beam,
220        None,
221    )?;
222    specification_inputs.force_primary_beam = true;
223    let specification =
224        super::specification::specification(request, &spectral, specification_inputs)?;
225    let geometry = super::geometry(request, &domains, centre, &spectral);
226    let samples = super::domains::model_samples(&domains, request.nterms, &request.stokes)?;
227    Ok(TaylorView {
228        specification,
229        geometry,
230        model_lifecycle: ModelLifecycleRequirements::new(
231            ModelBounds::new(samples, samples, f64::MAX, f64::MAX)?,
232            NumericPrecision::F64,
233        ),
234        coordinates: domains
235            .into_iter()
236            .map(|domain| (domain.role, domain.coordinates))
237            .collect(),
238        expansion_reference_hz,
239        reservation,
240    })
241}

243/// The first native MVC slice forms a fresh cube and its two-term Taylor
244/// products. Cleaning currently needs a negative pblimit, no nsigma, and
245/// spatial-only user boxes; restart and side effects remain refused (#1069).
246pub(super) fn check_execution_scope(
247    request: &ImagingRequest,
248    selected_fields: usize,
249) -> Result<(), super::PrepareError> {
250    use casa_imaging_model::RestoringBeamPolicy;
251    use casa_types::measures::frequency::FrequencyRef;
252    if request.vis.len() != 1
253        || selected_fields != 1
254        || (request.niter != 0
255            && (request.nsigma != 0.0
256                || request.mask_box.iter().any(|region| region.range.is_some())))
257        || !request.calcpsf
258        || !request.calcres
259        || request.tclean_restart
260        || request.has_start_model()
261        || request.savemodel != crate::SaveModel::None
262        || request.outlierfile.is_some()
263        || request.fitspw.is_some()
264        || request.pbcor
265        || request.usemask != crate::UseMask::User
266        || request.mask_image.is_some()
267        || request.pbmask != 0.0
268        || request.outframe != FrequencyRef::LSRK
269        || !matches!(
270            request.restoringbeam,
271            RestoringBeamPolicy::PerPlane | RestoringBeamPolicy::None
272        )
273    {
274        return Err(super::PrepareError::MvcExecutionUnsupported);
275    }
276    Ok(())
277}

279/// The metadata envelope below applies only to this bounded view.
280fn check_view_scope(
281    parent: &ImagingRequest,
282    store_count: usize,
283    moving_centre: bool,
284) -> Result<(), super::PrepareError> {
285    if store_count != 1
286        || parent.outlierfile.is_some()
287        || moving_centre
288        || !matches!(parent.gridder, crate::Gridder::Standard { facets } if facets.get() == 1)
289        || parent.stokes != [casa_imaging_model::PolarizationCoordinate::StokesI]
290        || parent.nterms != 2
291        || parent.weighting != crate::Weighting::Natural
292    {
293        return Err(super::PrepareError::MvcViewUnsupported);
294    }
295    Ok(())
296}

298/// Bound the additional view while it is prepared and compiled. The two
299/// rest-frequency arrays overlap while the WCS is built; the scales remain
300/// in the reconstruction contract. Metadata names are cloned for the
301/// temporary domain and coordinate system. The fixed envelope covers the
302/// one-domain, one-facet, two-term, one-Stokes problem's bounded coordinate,
303/// product-list, transaction and compile bookkeeping, including overlap of
304/// the prepared and compiled forms. Larger modes are refused above.
305fn view_demand(
306    request: &ImagingRequest,
307    observation: &casa_coordinates::ObsInfo,
308    object_name: &str,
309) -> Result<u64, super::PrepareError> {
310    const FIXED_VIEW_BYTES: usize = 64 * 1024;
311    let overflow = || super::PrepareError::MvcCubeDepth {
312        reason: "Taylor-view memory exceeds addressable size",
313    };
314    let frequencies = request
315        .restfreq
316        .len()
317        .max(1)
318        .checked_mul(2 * size_of::<f64>())
319        .ok_or_else(overflow)?;
320    let scales = request
321        .scales
322        .len()
323        .max(1)
324        .checked_mul(size_of::<f64>())
325        .ok_or_else(overflow)?;
326    let telescope = observation
327        .telescope
328        .len()
329        .checked_mul(2)
330        .ok_or_else(overflow)?;
331    let bytes = [
332        frequencies,
333        scales,
334        telescope,
335        observation.observer.len(),
336        object_name.len(),
337        request.imagename.as_os_str().len(),
338    ]
339    .into_iter()
340    .try_fold(FIXED_VIEW_BYTES, |sum, value| {
341        sum.checked_add(value).ok_or_else(overflow)
342    })?;
343    u64::try_from(bytes).map_err(|_| overflow())
344}
