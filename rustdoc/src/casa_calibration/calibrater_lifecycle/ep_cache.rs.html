18const AUX: &str = "aux.dat";
19const BEAM: &str = "avgPB";
20const STAGE_PREFIX: &str = ".ep-cache-";
21const JOURNAL: &str = "pending-slot";

23pub(super) struct EpCache {

25}

27pub(super) struct EpCacheProducts {
28    pub average_beam: ArrayD<f32>,
29    pub convolution: ArrayD<Complex32>,
30    pub weights: ArrayD<Complex32>,
31    pub support: usize,
32    pub sampling: usize,

36struct Slot {

42impl EpCache {
43    pub(super) fn open(path: &Path) -> MsResult<Self> {
44        if path.as_os_str().is_empty() {
45            return Err(invalid(
46                "Got null string for disk cache dir. in ConvFuncDiskCache::initCache()",
47            ));
48        }
49        std::fs::create_dir_all(path).map_err(invalid)?;
50        let cache = Self {
51            root: path.canonicalize().map_err(invalid)?,
52        };
53        cache.slots()?;
54        Ok(cache)
55    }
56
57    pub(super) fn path(&self) -> &Path {
58        &self.root
59    }
60
61    fn slots(&self) -> MsResult<Vec<Slot>> {
62        let path = self.root.join(AUX);
63        match std::fs::read_to_string(&path) {
64            Ok(document) => decode_slots(&document),
65            Err(error) if error.kind() == std::io::ErrorKind::NotFound => Ok(Vec::new()),
66            Err(error) => Err(invalid(format!("cannot read {}: {error}", path.display()))),
67        }
68    }
69
70    pub(super) fn read(
71        &self,
72        host: &HostResources,
73        angle_rad: f64,
74        maximum_gap_deg: f64,
75        shape: &[usize],
76        correlations: &[StokesType],
77    ) -> MsResult<Option<EpCacheProducts>> {
78        let slots = self.slots()?;
79        let angle = angle_rad.to_degrees();
80        let matching = slots
81            .iter()
82            .enumerate()
83            .map(|(index, slot)| ((slot.angle_deg - angle).abs(), index, slot))
84            .filter(|(gap, _, _)| *gap <= maximum_gap_deg)
85            .min_by(|left, right| left.0.total_cmp(&right.0));
86        let Some((_, index, slot)) = matching else {
87            return Ok(None);
88        };
89        let convolution = self.kernel("CF", index, slot, correlations.len())?;
90        let weights = self.kernel("CFWT", index, slot, correlations.len())?;
91        let bytes = |image: &PagedImage<Complex32>| {
92            image
93                .shape()
94                .iter()
95                .fold(size_of::<Complex32>() as u64, |size, &length| {
96                    size.saturating_mul(length as u64)
97                })
98        };
99        let memory = admit(
100            host,
101            &ResourcePolicy::Balanced,
102            &Demand {
103                phase: "legacy EP cached convolutions",
104                memory: bytes(&convolution).saturating_add(bytes(&weights)),
105            },
106        )
107        .map_err(invalid)?;
108        let convolution = read_kernel(&convolution)?;
109        let weights = read_kernel(&weights)?;
110        let beam = PagedImage::<f32>::open(self.root.join(BEAM)).map_err(invalid)?;
111        if shape != beam.shape() {
112            return Err(invalid(
113                "legacy EP cache average beam has a different image shape",
114            ));
115        }
116        let average_beam = beam.get().map_err(invalid)?;
117        if average_beam.iter().any(|value| !value.is_finite())
118            || !average_beam.iter().any(|value| *value >= 1e-4)
119        {
120            return Err(invalid(
121                "legacy EP cache average beam contains invalid values",
122            ));
123        }
124        Ok(Some(EpCacheProducts {
125            average_beam,
126            convolution,
127            weights,
128            support: slot.support,
129            sampling: slot.sampling,
130            _memory: memory,
131        }))
132    }
133
134    fn kernel(
135        &self,
136        prefix: &str,
137        index: usize,
138        slot: &Slot,
139        pols: usize,
140    ) -> MsResult<PagedImage<Complex32>> {
141        let path = self.root.join(format!("{prefix}0_{index}"));
142        let image = PagedImage::<Complex32>::open(&path).map_err(invalid)?;
143        let span = slot
144            .support
145            .checked_mul(slot.sampling)
146            .and_then(|span| span.checked_mul(2));
147        let [nx, ny, count, 1] = *image.shape() else {
148            return Err(invalid(format!(
149                "invalid legacy EP cache kernel {}",
150                path.display()
151            )));
152        };
153        if nx != ny || count != pols || span.is_none_or(|span| span >= nx) {
154            return Err(invalid(format!(
155                "invalid legacy EP cache kernel {}",
156                path.display()
157            )));
158        }
159        Ok(image)
160    }

162    pub(super) fn publish(
163        &self,
164        angle_rad: f64,
165        cell_rad: [f64; 2],
166        beam_coordinates: &CoordinateSystem,
167        shape: [usize; 4],
168        correlations: &[StokesType],
169        products: &LegacyEpConvolution,
170    ) -> MsResult<()> {
171        let mut slots = self.slots()?;
172        let index = slots.len();
173        let names = product_names(index);
174        let staged = tempfile::Builder::new()
175            .prefix(STAGE_PREFIX)
176            .tempdir_in(&self.root)
177            .map_err(invalid)?;
178        let mut kernel_coordinates = beam_coordinates.clone();
179        let linear =
180            LinearCoordinate::new(2, vec!["UU".into(), "VV".into()], vec!["lambda".into(); 2])
181                .with_reference_pixel(vec![1024.; 2])
182                .with_increment(
183                    cell_rad
184                        .map(|cell| 1. / (cell * (products.sampling * shape[0]) as f64))
185                        .to_vec(),
186                );
187        let axis = kernel_coordinates
188            .find_coordinate(CoordinateType::Direction)
189            .ok_or_else(|| invalid("legacy EP model is missing the direction axes"))?;
190        kernel_coordinates.replace_coordinate(axis, linear);
191        let pol = kernel_coordinates
192            .find_coordinate(CoordinateType::Stokes)
193            .ok_or_else(|| invalid("legacy EP model is missing the polarization axis"))?;
194        kernel_coordinates.replace_coordinate(pol, StokesCoordinate::new(correlations.to_vec()));
195        for (name, kernel) in names[..2].iter().zip(&products.kernels) {
196            let side = products.kernel_side;
197            let shape = [side, side, correlations.len(), 1];
198            let pixels = ArrayD::from_shape_fn(IxDyn(&shape).f(), |i| kernel[i[0] + side * i[1]]);
199            save_image(staged.path().join(name), kernel_coordinates.clone(), pixels)?;
200        }
201        if index == 0 {
202            let pixels = ArrayD::from_shape_fn(IxDyn(&shape).f(), |i| {
203                products.average_beam[i[0] + shape[0] * i[1]]
204            });
205            save_image(staged.path().join(BEAM), beam_coordinates.clone(), pixels)?;
206        } else {
207            let beam = PagedImage::<f32>::open(self.root.join(BEAM)).map_err(invalid)?;
208            if beam.shape() != shape {
209                return Err(invalid(
210                    "legacy EP cache average beam has a different image shape",
211                ));
212            }
213        }
214        let angle = format!("{:.4}", angle_rad.to_degrees());
215        slots.push(Slot {
216            angle_deg: angle.parse().map_err(invalid)?,
217            support: products.support,
218            sampling: products.sampling,
219        });
220        let mut manifest = format!("{} 1\n", slots.len());
221        for slot in slots {
222            writeln!(
223                manifest,
224                "{} {} {}  {}",
225                slot.angle_deg, slot.support, slot.support, slot.sampling
226            )
227            .expect("writing to a String cannot fail");
228        }
229        std::fs::write(staged.path().join(AUX), manifest).map_err(invalid)?;
230        self.publish_files(staged.path(), index, &names)
231    }

233    /// Clean up a previously abandoned publication while holding the cache's
234    /// exclusive lock. Only a stage with a journal can have changed the root.
235    pub(super) fn recover(&self) -> MsResult<()> {
236        let committed = self.slots()?.len();
237        for entry in std::fs::read_dir(&self.root).map_err(invalid)? {
238            let entry = entry.map_err(invalid)?;
239            if !entry
240                .file_name()
241                .to_string_lossy()
242                .starts_with(STAGE_PREFIX)
243                || !entry.file_type().map_err(invalid)?.is_dir()
244            {
245                continue;
246            }
247            let stage = entry.path();
248            let marker = stage.join(JOURNAL);
249            let document = match std::fs::read_to_string(marker) {
250                Ok(document) => document,
251                Err(error) if error.kind() == std::io::ErrorKind::NotFound => continue,
252                Err(error) => return Err(invalid(error)),
253            };
254            let index: usize = document.parse().map_err(invalid)?;
255            if committed == index {
256                self.rollback(&stage, &product_names(index))?;
257            } else if committed < index {
258                return Err(invalid(
259                    "legacy EP cache manifest predates its publication journal",
260                ));
261            }
262            std::fs::remove_dir_all(stage).map_err(invalid)?;
263        }
264        Ok(())
265    }
266
267    fn rollback(&self, stage: &Path, names: &[String]) -> MsResult<()> {
268        for name in names {
269            let target = self.root.join(name);
270            // A product still present in the stage was never promoted.
271            if !stage.join(name).exists() && target.is_dir() {
272                std::fs::remove_dir_all(target).map_err(invalid)?;
273            }
274        }
275        Ok(())
276    }
277
278    fn publish_files(&self, staged: &Path, index: usize, names: &[String]) -> MsResult<()> {
279        for name in names {
280            let target = self.root.join(name);
281            if target.exists() {
282                return Err(invalid(format!(
283                    "legacy EP cache already contains {}",
284                    target.display()
285                )));
286            }
287        }
288        std::fs::write(staged.join(JOURNAL), index.to_string()).map_err(invalid)?;
289        let published = (|| -> std::io::Result<()> {
290            for name in names {
291                std::fs::rename(staged.join(name), self.root.join(name))?;
292            }
293            // Readers see the previous or complete new slot list.
294            std::fs::rename(staged.join(AUX), self.root.join(AUX))
295        })();
296        if let Err(error) = published {
297            self.rollback(staged, names)?;
298            return Err(invalid(error));
299        }
300        Ok(())
301    }
302}

304fn read_kernel(image: &PagedImage<Complex32>) -> MsResult<ArrayD<Complex32>> {
305    let kernels = image.get().map_err(invalid)?;
306    if kernels
307        .iter()
308        .any(|v| !v.re.is_finite() || !v.im.is_finite())
309        || !kernels.iter().any(|v| v.re != 0. || v.im != 0.)
310    {
311        return Err(invalid("invalid sample in legacy EP cache kernel"));
312    }
313    Ok(kernels)
314}
315
316fn product_names(index: usize) -> Vec<String> {
317    let mut names = vec![format!("CF0_{index}"), format!("CFWT0_{index}")];
318    if index == 0 {
319        names.push(BEAM.to_owned());
320    }
321    names
322}

324fn save_image<T: ImagePixel>(
325    path: PathBuf,
326    coordinates: CoordinateSystem,
327    pixels: ArrayD<T>,
328) -> MsResult<()> {
329    let mut image =
330        PagedImage::create(pixels.shape().to_vec(), coordinates, path).map_err(invalid)?;
331    image.put_slice(&pixels, &[0; 4]).map_err(invalid)?;
332    image.save().map_err(invalid)
333}

335fn decode_slots(document: &str) -> MsResult<Vec<Slot>> {
336    let mut fields = document.split_whitespace();
337    let slots = parse::<usize>(&mut fields)?;
338    let planes = parse::<usize>(&mut fields)?;
339    if planes != 1 {
340        return Err(invalid("legacy EP requires a one-plane convolution cache"));
341    }
342    let mut decoded = Vec::new();
343    for _ in 0..slots {
344        let angle_deg = parse::<f64>(&mut fields)?;
345        let support = parse::<usize>(&mut fields)?;
346        let _ = parse::<usize>(&mut fields)?; // CASA reload uses X support for both axes.
347        let sampling = parse::<f64>(&mut fields)?;
348        if !angle_deg.is_finite()
349            || support == 0
350            || !(1. ..usize::MAX as f64).contains(&sampling)
351            || sampling.fract() != 0.
352        {
353            return Err(invalid(
354                "invalid support or angle in legacy EP cache aux.dat",
355            ));
356        }
357        decoded.push(Slot {
358            angle_deg,
359            support,
360            sampling: sampling as usize,
361        });
362    }
363    if fields.next().is_some() {
364        return Err(invalid("extra fields in legacy EP cache aux.dat"));
365    }
366    Ok(decoded)
367}
368
369fn parse<T: std::str::FromStr>(fields: &mut SplitWhitespace<'_>) -> Result<T, MsError> {
370    fields
371        .next()
372        .and_then(|value| value.parse().ok())
373        .ok_or_else(|| invalid("invalid legacy EP cache aux.dat"))
374}
