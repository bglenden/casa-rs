31/// Produce every planned member and write it to `output`'s staging.
32///
33/// Runs restoring-beam fitting, restoration, residual scaling, normalization,
34/// validity, and metadata exactly once per member, in the planned publication
35/// order. [`CasaImageWriter::publish`] then replaces the targets.
36///
37/// # Errors
38///
39/// Unsupported roles, failed beam fits, generated non-finite payloads, and
40/// image writes that fail. A failed generation discards every member it
41/// staged, so a later [`CasaImageWriter::publish`] publishes none of them.
42pub fn produce_continuum_members(
43    planned: &PlannedContinuumGeneration,
44    inputs: &ContinuumProductInputs<'_>,
45    storage_plan: ProductStoragePlan,
46    execution: &impl ProductWindowExecutor,
47    output: &CasaImageWriter,
48) -> Result<PublishedContinuumGeneration, ProductsError> {
49    if inputs.cube_taylor_primary_beam().is_some() && planned.primary_beam_model.is_none() {
50        return Err(ProductsError::MissingPrimaryBeamModel);
51    }
52    let kept = output.staged_count();
53    let produced = if inputs.normal_state().basis() == NormalBasis::Taylor {
54        produce_taylor_members(planned, inputs, storage_plan, output)
55    } else {
56        produce_plane_members(planned, inputs, storage_plan, execution, output)
57    };
58    if produced.is_err() {
59        output.discard_staged_after(kept);
60    }
61    produced
62}

64fn field_residual_mask(
65    planned: &PlannedContinuumGeneration,
66    member: &PlannedMember,
67    output: &CasaImageWriter,
68    policy: crate::plan::FieldResidualMask,
69) -> Result<Option<Option<SavedPixelMask>>, ProductsError> {
70    use crate::plan::FieldResidualMask;
71    let domain = member.axes().domain();
72    let restored = member.role == ProductRole::RestoredImage(ProductTerm::Single);
73    if restored && !planned.rebuilds_field_pb && !output.has_saved_pb(domain) {
74        return Ok(Some(output.existing_mask(domain, ".image", |_| true)?));
75    }
76
77    let mut has_default = false;
78    let mut mask = output.existing_mask(domain, ".residual", |name| {
79        has_default = true;
80        if name == "mask0" {
81            policy != FieldResidualMask::CustomOnly
82        } else {
83            !restored
84        }
85    })?;
86    if !has_default && policy == FieldResidualMask::SavedOrPrimaryBeam {
87        mask = output.existing_mask(domain, ".pb", |name| name == "mask0")?;
88        if mask.is_none() && planned.rebuilds_field_pb {
89            // A newly created PB supplies its generated mask.
90            return Ok(None);
91        }
92    }
93    if restored && mask.is_none() {
94        // With a PB, restore removes an old mask0. It leaves a custom
95        // output mask unless copyMask replaces it with a residual mask0.
96        mask = output.existing_mask(domain, ".image", |name| name != "mask0")?;
97    }
98    Ok(Some(mask))
99}

101fn produce_plane_members(
102    planned: &PlannedContinuumGeneration,
103    inputs: &ContinuumProductInputs<'_>,
104    storage_plan: ProductStoragePlan,
105    execution: &impl ProductWindowExecutor,
106    output: &CasaImageWriter,
107) -> Result<PublishedContinuumGeneration, ProductsError> {
108    let fitted_beams = if planned.requires_beam() {
109        fit_beams(planned, inputs, storage_plan, execution)?
110    } else {
111        Box::new([])
112    };
113    let restoring_beams = restoring_beams(inputs, &fitted_beams)?;
114    for member in &planned.members {
115        if matches!(member.role, ProductRole::SumWeights(_))
116            && planned.rebuilds_field_psfs
117            && output.has_positive_saved_scalar_sum(member.axes().domain())?
118        {
119            continue;
120        }
121        let saved_mask = single_term_member_mask(planned, member, inputs, output)?;
122        let windows = MemberWindows::new(
123            planned,
124            member,
125            inputs,
126            storage_plan,
127            &fitted_beams,
128            &restoring_beams,
129        )?;
130        let image = output.begin(
131            member,
132            windows.layout,
133            &planned.beams_for_member(member, &fitted_beams, &restoring_beams),
134            planned.uses_weight_image,
135            saved_mask
136                .as_ref()
137                .map(|mask| mask.as_ref().map(|mask| mask.name.as_str())),
138        )?;
139        windows.write(
140            ProductMemberWriter::new(windows.layout, image),
141            execution,
142            (
143                saved_mask
144                    .as_ref()
145                    .map(|mask| mask.as_ref().map(|mask| mask.pixels.as_slice())),
146                output,
147            ),
148        )?;
149    }
150    Ok(planned.published(fitted_beams, restoring_beams))
151}

153fn single_term_member_mask(
154    planned: &PlannedContinuumGeneration,
155    member: &PlannedMember,
156    inputs: &ContinuumProductInputs<'_>,
157    output: &CasaImageWriter,
158) -> Result<Option<Option<SavedPixelMask>>, ProductsError> {
159    let generated = match member.role {
160        ProductRole::RestoredImage(ProductTerm::Single)
161        | ProductRole::Residual(ProductTerm::Single)
162            if planned.saved_residual_mask_read_bytes.is_some() =>
163        {
164            if let Some(policy) = planned.field_residual_mask {
165                return field_residual_mask(planned, member, output, policy);
166            }
167            Some(output.saved_residual_mask(
168                member.axes().domain(),
169                ProductTerm::Single,
170                matches!(member.role, ProductRole::RestoredImage(_)),
171                false,
172            )?)
173        }
174        ProductRole::PrimaryBeam(ProductTerm::Single) if planned.rebuilds_field_pb => {
175            output.existing_pb_mask(member.axes().domain())?.map(Some)
176        }
177        ProductRole::PbCorrectedImage(ProductTerm::Single) => match planned.saved_cube_pbcor_mask {
178            Some(crate::plan::SavedPbcorMask::Existing(_)) => {
179                Some(output.existing_image_mask(member.axes().domain(), ".image.pbcor")?)
180            }
181            Some(crate::plan::SavedPbcorMask::PrimaryBeam)
182                if inputs
183                    .saved_mosaic_beam()
184                    .is_some_and(|beam| !beam.has_generated_mask()) =>
185            {
186                Some(None)
187            }
188            _ => None,
189        },
190        _ => None,
191    };
192    Ok(generated.map(|mask| {
193        mask.map(|pixels| SavedPixelMask {
194            name: "mask0".to_owned(),
195            pixels,
196        })
197    }))
198}

200/// Fit a restoring beam to every valid PSF plane, domain by domain, then
201/// channel by channel, then polarization, a wave of the team's workers at a
202/// time.
203fn fit_beams(
204    planned: &PlannedContinuumGeneration,
205    inputs: &ContinuumProductInputs<'_>,
206    storage_plan: ProductStoragePlan,
207    execution: &impl ProductWindowExecutor,
208) -> Result<Box<[Option<RestoringBeam>]>, ProductsError> {
209    let normal_state = inputs.normal_state();
210    let channels = normal_state.channel_count();
211    let polarizations = normal_state.polarization_count();
212    let beam_count = normal_state
213        .domain_count()
214        .checked_mul(channels)
215        .and_then(|count| count.checked_mul(polarizations))
216        .ok_or(ProductsError::ResourceDemandOverflow(
217            "generation beam count",
218        ))?;
219    let mut fitted = Vec::with_capacity(beam_count);
220    for start in (0..beam_count).step_by(storage_plan.maximum_workers()) {
221        let count = storage_plan.maximum_workers().min(beam_count - start);
222        let mut slots = vec![None; count];
223        execution.prepare(&mut slots, &|index| {
224            let ordinal = start + index;
225            let domain =
226                &inputs.problem().geometry().domains()[ordinal / (channels * polarizations)];
227            let domain_ordinal = inputs.model_domain_ordinal(domain.role());
228            if let Some(saved) = inputs.saved_psf_beams(domain_ordinal) {
229                return Ok(PlaneBeam::Fitted(
230                    saved[ordinal % (channels * polarizations)],
231                ));
232            }
233            let plane = normal_state.read_plane(
234                inputs.model_domain_ordinal(domain.role()),
235                (ordinal / polarizations) % channels,
236                ordinal % polarizations,
237            )?;
238            // A freshly formed PSF is normalized before CASA fits the beam,
239            // so normPSF zeros a sub-threshold peak. A reused PSF is already
240            // in image units: CASA attempts the fit for every positive peak,
241            // and FitGaussianPSF refuses positive peaks below 1e-7.
242            let psf = psf_real_plane(&plane)?;
243            let peak = psf_peak(psf.iter().copied())?;
244            if !psf_normalizes(peak) {
245                // The alternate PSF can be blank where the main cube still
246                // has data. CASA's makeImageBeamSet supplies the largest
247                // fitted beam for that channel without blanking the residual.
248                let sum = plane.sum_weight();
249                let supported = plane.validity()
250                    == casa_imaging_deconvolution::SpectralChannelValidity::Valid
251                    && sum.is_finite()
252                    && sum > 0.0;
253                return Ok(if supported || (planned.reuses_single_psf && peak > 0.0) {
254                    PlaneBeam::Refused
255                } else {
256                    PlaneBeam::Blank
257                });
258            }
259            fit_plane_beam(
260                &psf,
261                plane.shape(),
262                inputs.cell_size_rad_for_domain(domain.role()),
263                planned.psf_cutoff(),
264            )
265        })?;
266        fitted.extend(
267            slots
268                .into_iter()
269                .map(|slot| slot.expect("the executor fills every slot")),
270        );
271    }
272    Ok(settle_beams(&fitted, channels * polarizations))
273}

275/// The beams each plane is restored with, by the problem's restoring-beam
276/// policy: none, each plane's fitted beam, one beam enclosing a domain's
277/// fitted beams, or the beam the request gives. A plane with no fitted beam
278/// (a blank PSF) restores with its domain's largest restoring beam, as
279/// `SIImageStore::makeImageBeamSet` replaces a null beam by the beam set's
280/// largest (`getMaxAreaBeam`) before `restore` restores every plane: a
281/// start model regridded onto a channel no sample reaches is restored
282/// there. A domain with no fitted beam restores none. A Taylor family's planes
283/// are its polarizations.
284pub(crate) fn restoring_beams(
285    inputs: &ContinuumProductInputs<'_>,
286    fitted_beams: &[Option<RestoringBeam>],
287) -> Result<Box<[Option<RestoringBeam>]>, ProductsError> {
288    let mut beams = policy_restoring_beams(inputs, fitted_beams)?;
289    let normal_state = inputs.normal_state();
290    let planes = normal_state.channel_count() * normal_state.polarization_count();
291    if planes > 0 {
292        for domain_beams in beams.chunks_mut(planes) {
293            let largest = domain_beams
294                .iter()
295                .flatten()
296                .copied()
297                .max_by(|left, right| {
298                    (left.major_fwhm_rad() * left.minor_fwhm_rad())
299                        .total_cmp(&(right.major_fwhm_rad() * right.minor_fwhm_rad()))
300                });
301            for beam in domain_beams.iter_mut().filter(|beam| beam.is_none()) {
302                *beam = largest;
303            }
304        }
305    }
306    Ok(beams)
307}

309/// [`restoring_beams`] before blank planes take their domain's largest.
310fn policy_restoring_beams(
311    inputs: &ContinuumProductInputs<'_>,
312    fitted_beams: &[Option<RestoringBeam>],
313) -> Result<Box<[Option<RestoringBeam>]>, ProductsError> {
314    Ok(match inputs.problem().products().restoring_beam() {
315        RestoringBeamPolicy::None => Box::new([]),
316        RestoringBeamPolicy::PerPlane => fitted_beams.into(),
317        RestoringBeamPolicy::Fixed(beam) => {
318            let beam = fixed_restoring_beam(beam)?;
319            fitted_beams
320                .iter()
321                .map(|fitted| fitted.map(|_| beam))
322                .collect()
323        }
324        RestoringBeamPolicy::Common => {
325            let normal_state = inputs.normal_state();
326            let planes = normal_state.channel_count() * normal_state.polarization_count();
327            let mut selected = Vec::with_capacity(fitted_beams.len());
328            for domain_beams in fitted_beams.chunks(planes) {
329                let valid = domain_beams.iter().flatten().copied().collect::<Vec<_>>();
330                if valid.is_empty() {
331                    selected.extend_from_slice(domain_beams);
332                } else {
333                    let common = RestoringBeam::common_enclosing(&valid)
334                        .map_err(|error| ProductsError::BeamFitFailed(error.to_string()))?;
335                    selected.extend(domain_beams.iter().map(|fitted| fitted.map(|_| common)));
336                }
337            }
338            selected.into_boxed_slice()
339        }
340    })
341}

343/// What generating one member's windows needs.
344struct MemberWindows<'a, 'inputs> {
345    planned: &'a PlannedContinuumGeneration,
346    member: &'a PlannedMember,
347    inputs: &'a ContinuumProductInputs<'inputs>,
348    layout: ProductWindowLayout,
349    domain_ordinal: usize,
350    plane_shape: [usize; 2],
351    fitted_beams: &'a [Option<RestoringBeam>],
352    restoring_beams: &'a [Option<RestoringBeam>],
353    workers: usize,
354    fft_threads: usize,
355}
356
357impl<'a, 'inputs> MemberWindows<'a, 'inputs> {
358    fn new(
359        planned: &'a PlannedContinuumGeneration,
360        member: &'a PlannedMember,
361        inputs: &'a ContinuumProductInputs<'inputs>,
362        storage_plan: ProductStoragePlan,
363        fitted_beams: &'a [Option<RestoringBeam>],
364        restoring_beams: &'a [Option<RestoringBeam>],
365    ) -> Result<Self, ProductsError> {
366        let domain_ordinal = inputs.model_domain_ordinal(member.axes().domain());
367        let layout = storage_plan.layout(member.axes())?;
368        let windows = inputs
369            .normal_state()
370            .channel_count()
371            .div_ceil(layout.maximum_channels());
372        Ok(Self {
373            planned,
374            member,
375            inputs,
376            layout,
377            domain_ordinal,
378            plane_shape: inputs.final_model().shape().domains()[domain_ordinal].pixels(),
379            fitted_beams,
380            restoring_beams,
381            workers: storage_plan.maximum_workers().min(windows),
382            fft_threads: storage_plan.fft_threads(windows),
383        })
384    }
385
386    /// Generate the member's windows a wave of workers at a time and write
387    /// them in channel order.
388    fn write(
389        &self,
390        mut writer: ProductMemberWriter<'_>,
391        execution: &impl ProductWindowExecutor,
392        (saved_mask, output): (Option<Option<&[bool]>>, &CasaImageWriter),
393    ) -> Result<(), ProductsError> {
394        let channels = self.inputs.normal_state().channel_count();
395        let window_channels = self.layout.maximum_channels();
396        let windows = channels.div_ceil(window_channels);
397        for wave_start in (0..windows).step_by(self.workers) {
398            let mut slots: Vec<Option<ProductWindow>> = (0..self.workers.min(windows - wave_start))
399                .map(|_| None)
400                .collect();
401            execution.prepare(&mut slots, &|index| {
402                let start = (wave_start + index) * window_channels;
403                self.window(
404                    start..(start + window_channels).min(channels),
405                    saved_mask,
406                    output,
407                )
408            })?;
409            for slot in slots {
410                writer.write(slot.expect("the executor fills every slot"))?;
411            }
412        }
413        writer.finish()
414    }
415
416    /// One window: every polarization plane of its channels.
417    fn window(
418        &self,
419        channels: Range<usize>,
420        saved_mask: Option<Option<&[bool]>>,
421        output: &CasaImageWriter,
422    ) -> Result<ProductWindow, ProductsError> {
423        let mut window = self.layout.window(channels.clone())?;
424        let reconstruction_only = matches!(
425            self.member.role,
426            ProductRole::Model(ProductTerm::Single | ProductTerm::Taylor(0))
427                | ProductRole::CleanMask
428        );
429        for channel in channels.clone() {
430            for polarization in 0..self.inputs.normal_state().polarization_count() {
431                let local = channel - channels.start;
432                let saved_model = (self.planned.saved_cube_model_read_bytes > 0
433                    && matches!(
434                        self.member.role,
435                        ProductRole::Model(_)
436                            | ProductRole::RestoredImage(_)
437                            | ProductRole::PbCorrectedImage(_)
438                    ))
439                .then(|| {
440                    output.saved_model_plane(self.member.axes().domain(), channel, polarization)
441                })
442                .transpose()?;
443                if reconstruction_only {
444                    let plane =
445                        self.reconstruction_plane(channel, polarization, saved_model.as_ref())?;
446                    self.scatter(
447                        &mut window.payload,
448                        &window.shape,
449                        polarization,
450                        local,
451                        &plane,
452                    );
453                } else {
454                    self.normal_plane(
455                        &mut window,
456                        (channel, polarization, local),
457                        saved_mask,
458                        saved_model.as_ref(),
459                    )?;
460                }
461            }
462        }
463        Ok(window)
464    }

466    fn scatter<T: Copy>(
467        &self,
468        target: &mut [T],
469        shape: &[usize; 4],
470        polarization: usize,
471        local_channel: usize,
472        plane: &[T],
473    ) {
474        scatter_image_polarization_plane(
475            target,
476            self.member.axes().order(),
477            *shape,
478            polarization,
479            local_channel,
480            self.plane_shape,
481            plane,

483    }

485    /// A member formed from the final model or the CLEAN mask alone.
486    fn reconstruction_plane(
487        &self,
488        channel: usize,
489        polarization: usize,
490        saved_model: Option<&ndarray::ArrayD<f32>>,
491    ) -> Result<Vec<f32>, ProductsError> {
492        let [width, height] = self.plane_shape;
493        if self.member.role == ProductRole::CleanMask {
494            return Ok(reconstruction_support_plane(
495                self.inputs,
496                self.member.axes().domain(),
497                (channel, polarization),
498                width * height,
499            ));
500        }
501        let model = model_real_plane(
502            self.inputs.final_model(),
503            self.domain_ordinal,
504            channel,
505            polarization,
506            self.plane_shape,
507        )?;
508        if self.planned.primary_beam_model != Some(AnalyticPrimaryBeamModel::MosaicSensitivity)
509            || self.planned.saved_weighted_residual
510        {
511            return Ok(model);
512        }
513        let plane =
514            self.inputs
515                .normal_state()
516                .read_plane(self.domain_ordinal, channel, polarization)?;
517        let products = self.inputs.problem().products();
518        apparent_model(
519            model,
520            &plane,
521            products.normalization(),
522            products.validity().primary_beam(),
523            self.planned.primary_beam_model,
524            saved_model,
525        )
526    }

528    /// A member formed from a normal-state plane: its values, and its pixel
529    /// mask where it stores one.
530    fn normal_plane(
531        &self,
532        window: &mut ProductWindow,
533        (channel, polarization, local_channel): (usize, usize, usize),
534        saved_mask: Option<Option<&[bool]>>,
535        saved_model: Option<&ndarray::ArrayD<f32>>,
536    ) -> Result<(), ProductsError> {
537        let plane =
538            self.inputs
539                .normal_state()
540                .read_plane(self.domain_ordinal, channel, polarization)?;
541        if matches!(self.member.role, ProductRole::SumWeights(_)) {
542            // CASA's `.sumwt` is the PSF gridding's sum:
543            // `FTMachine::finalizeToSkyNew` seeds the image store's sumwt
544            // once, from `makePSF` (`refim_mawproject`: 8987.97 is the weight
545            // cell's norms, 0.24 % above the data gridding's). The flat-noise
546            // products divide by the raw sensitivity, where every such sum
547            // cancels, as CASA's `divideResidualByWeight` reduces to.
548            // A faceted domain publishes each facet store's own sum.
549            let facets = self.inputs.facet_sum_weights(self.domain_ordinal);
550            let index = channel * self.inputs.normal_state().polarization_count() + polarization;
551            scatter_polarization_plane_state(
552                &mut window.payload,
553                self.member.axes(),
554                window.shape,
555                polarization,
556                local_channel,
557                |facet| match facets {
558                    Some(facets) => facets[facet][index] as f32,
559                    None => plane.sum_weight() as f32,
560                },
561            );
562            return Ok(());
563        }
564        let planes = self.inputs.normal_state().polarization_count();
565        let beam_index =
566            (self.domain_ordinal * self.inputs.normal_state().channel_count() + channel) * planes
567                + polarization;
568        let mut values = produce_plane_member(&PlaneMemberRequest {
569            member: self.member,
570            inputs: self.inputs,
571            plane: &plane,
572            domain_ordinal: self.domain_ordinal,
573            polarization,
574            reuses_single_psf: self.planned.reuses_single_psf,
575            saved_weighted_residual: self.planned.saved_weighted_residual,
576            saved_cube_model_plane: saved_model,
577            fitted_beam: self.fitted_beams.get(beam_index).copied().flatten(),
578            restoring_beam: self.restoring_beams.get(beam_index).copied().flatten(),
579            primary_beam_model: self.planned.primary_beam_model,
580            fft_threads: self.fft_threads,
581        })?;
582        if matches!(self.member.role, ProductRole::Residual(ProductTerm::Single))
583            && let Some(mean_pb) = self.inputs.cube_taylor_primary_beam()
584        {
585            let channel_pb = primary_beam_plane(
586                self.planned.primary_beam_model,
587                self.inputs,
588                self.member.axes().domain(),
589                &plane,
590                polarization,
591            )?;
592            let cutoff = self
593                .inputs
594                .problem()
595                .products()
596                .validity()
597                .primary_beam()
598                .cutoff();
599            adjust_cube_residual(&mut values, mean_pb, &channel_pb, cutoff)?;
600        }
601        let validity = |rule| {
602            product_plane_validity(
603                rule,
604                &plane,
605                self.planned.primary_beam_model,
606                self.inputs,
607                (self.member.axes().domain(), polarization),
608            )
609        };
610        if self.member.validity != ProductValidityRule::All {
611            zero_invalid_plane_values(&mut values, &validity(self.member.validity)?);
612        }
613        let support = match (saved_mask, self.member.storage.pixel_mask()) {
614            (Some(mask), _) => mask.map(|mask| {
615                let cells = self.plane_shape[0] * self.plane_shape[1];
616                let offset = (channel * planes + polarization) * cells;
617                std::borrow::Cow::Borrowed(&mask[offset..offset + cells])
618            }),
619            (None, ProductPixelMask::Explicit(rule)) => {
620                Some(std::borrow::Cow::Owned(validity(rule)?))
621            }
622            (None, ProductPixelMask::Absent) => None,
623        };
624        if let Some(support) = support {
625            self.scatter(
626                &mut window.validity,
627                &window.shape,
628                polarization,
629                local_channel,
630                &support,
631            );
632        }
633        self.scatter(
634            &mut window.payload,
635            &window.shape,
636            polarization,
637            local_channel,
638            &values,
639        );
640        Ok(())
641    }
642}

644/// CASA leaves an entire cube residual plane untouched when its published PB
645/// is blank. Otherwise its common/channel PB ratio applies only above the
646/// channel PB cutoff (`SynthesisUtilMethods::removeFreqDepPB`).
647pub(crate) fn adjust_cube_residual(
648    values: &mut [f32],
649    mean_pb: &[f32],
650    channel_pb: &[f32],
651    cutoff: f32,
652) -> Result<(), ProductsError> {
653    if channel_pb.iter().any(|&value| value > cutoff) {
654        for ((value, &common), &channel) in values.iter_mut().zip(mean_pb).zip(channel_pb) {
655            *value = if channel > cutoff {
656                common * *value / channel
657            } else {
658                0.0
659            };
660            if !value.is_finite() {
661                return Err(ProductsError::GeneratedNonfinite);
662            }
663        }
664    }
665    Ok(())
666}

668fn produce_taylor_members(
669    planned: &PlannedContinuumGeneration,
670    inputs: &ContinuumProductInputs<'_>,
671    storage_plan: ProductStoragePlan,
672    output: &CasaImageWriter,
673) -> Result<PublishedContinuumGeneration, ProductsError> {
674    let products = TaylorProducts::build(inputs, planned, output)?;
675    let fitted_beams = if planned.requires_beam() {
676        products.fitted_beams()
677    } else {
678        Box::new([])
679    };
680    let restoring_beams = match inputs.problem().products().restoring_beam() {
681        RestoringBeamPolicy::None => Box::new([]),
682        RestoringBeamPolicy::PerPlane => products.restoring_beams(),
683        // One beam restores every Stokes plane, and the image records it
684        // alone, as `SDAlgorithmMSMFS::restore` sets one restoring beam.
685        RestoringBeamPolicy::Common | RestoringBeamPolicy::Fixed(_) => products
686            .restoring_beams()
687            .iter()
688            .copied()
689            .find(Option::is_some)
690            .into_iter()
691            .collect(),
692    };
693    let refresh_mvc_mask =
694        inputs.taylor_primary_beam().is_some() && inputs.saved_mosaic_beam().is_some();
695    for member in &planned.members {
696        let beams = planned.beams_for_member(member, &fitted_beams, &restoring_beams);
697        if planned.refits_saved_psf_beams && member.role == ProductRole::Psf(ProductTerm::Taylor(0))
698        {
699            output.stage_saved_psf_beam(member, &beams)?;
700            continue;
701        }
702        let mut payload =
703            if let (ProductRole::SumWeights(ProductTerm::Taylor(moment)), Some(facets)) =
704                (member.role, inputs.facet_sum_weights(0))
705            {
706                facet_moment_weights(
707                    member,
708                    inputs.normal_state().polarization_count(),
709                    moment,
710                    facets,
711                )
712            } else {
713                products.payload(member.role)?
714            };
715        // An ordinary Taylor PB is formed at one frequency and zeroed by
716        // `makePBImage`. MVC's supplied PB is the weighted cube PB: CASA
717        // preserves its values below pblimit while marking those pixels
718        // invalid. The other products still blank outside their support.
719        let supplied_pb = inputs.taylor_primary_beam().is_some()
720            && member.role == ProductRole::PrimaryBeam(ProductTerm::Taylor(0));
721        if member.validity != ProductValidityRule::All && !supplied_pb {
722            zero_invalid_plane_values(
723                &mut payload,
724                &products.validity(member.role, member.validity, refresh_mvc_mask)?,
725            );
726        }
727        let saved_residual_mask = match member.role {
728            ProductRole::RestoredImage(ProductTerm::Taylor(term))
729                if planned.saved_residual_mask_read_bytes.is_some() =>
730            {
731                Some(output.saved_residual_mask(
732                    member.axes().domain(),
733                    ProductTerm::Taylor(term),
734                    true,
735                    false,
736                )?)
737            }
738            _ => None,
739        };
740        let mask_override = saved_residual_mask
741            .as_ref()
742            .map(|mask| mask.as_ref().map(|_| "mask0"));
743        let validity = match (saved_residual_mask, member.storage.pixel_mask()) {
744            (Some(Some(mask)), _) => mask,
745            (Some(None), _)
746            | (None, ProductPixelMask::Absent)
747            | (
748                None,
749                ProductPixelMask::Explicit(
750                    ProductValidityRule::All | ProductValidityRule::FinalNormalState,
751                ),
752            ) => vec![true; member.payload_values],
753            (None, ProductPixelMask::Explicit(rule)) => {
754                products.validity(member.role, rule, refresh_mvc_mask)?
755            }
756        };
757        if payload.len() != member.payload_values || validity.len() != member.payload_values {
758            return Err(ProductsError::PayloadLengthMismatch {
759                expected: member.payload_values,
760                actual: payload.len().max(validity.len()),
761            });
762        }
763        let layout = storage_plan.layout(member.axes())?;
764        let image = output.begin(
765            member,
766            layout,
767            &beams,
768            planned.uses_weight_image,
769            mask_override,
770        )?;
771        let mut writer = ProductMemberWriter::new(layout, image);
772        writer.write_coupled(&payload, &validity)?;
773        writer.finish()?;
774    }
775    Ok(planned.published(fitted_beams, restoring_beams))
776}

778/// CASA's Taylor sum-weight image has one pixel per facet for this moment.
779fn facet_moment_weights(
780    member: &PlannedMember,
781    polarizations: usize,
782    moment: usize,
783    facets: &[Vec<f64>],
784) -> Vec<f32> {
785    let axes = member.axes();
786    let mut payload = vec![0.0; member.payload_values];
787    for polarization in 0..polarizations {
788        let index = moment * polarizations + polarization;
789        scatter_polarization_plane_state(
790            &mut payload,
791            axes,
792            axes.shape(),
793            polarization,
794            0,
795            |facet| facets[facet][index] as f32,
796        );
797    }
798    payload
799}

801#[cfg(test)]
802mod mvc_adjustment_tests {
803    use super::adjust_cube_residual;
804
805    #[test]
806    fn residual_correction_uses_strict_channel_support_and_skips_a_blank_plane() {
807        let mut residual = [1.0, 2.0, 3.0, 4.0];
808        let common = [0.5, 0.6, 0.7, 0.8];
809        adjust_cube_residual(&mut residual, &common, &[0.1, 0.2, 0.4, 1.0], 0.2).unwrap();
810        assert_eq!(residual, [0.0, 0.0, 0.7 * 3.0 / 0.4, 0.8 * 4.0]);
811        let mut blank = [1.0, 2.0, 3.0, 4.0];
812        adjust_cube_residual(&mut blank, &common, &[0.0, 0.1, 0.2, 0.0], 0.2).unwrap();
813        assert_eq!(blank, [1.0, 2.0, 3.0, 4.0]);
814        let mut overflow = [f32::MAX];
815        assert!(adjust_cube_residual(&mut overflow, &[f32::MAX], &[1.0], 0.2).is_err());
816    }
