3//! tclean's continuation on an earlier run's products (#944, #980).
4//!
5//! CASA's `SIImageStore` opens every product that exists at `imagename`
6//! and uses its pixels: the model is the start of the solve, a CLEAN mask is
7//! reused when no new mask is given (`SynthesisDeconvolver::setupMask`), and
8//! the PSF, sums of weights and residual stand unless `calcpsf` or `calcres`
9//! forms them again (`task_tclean.py`). casa-rs reads them on the same grid
10//! ([`check_grid`]) for a run of the main image of every spectral mode,
11//! Taylor family and Stokes set it images. What it does not continue is
12//! refused here, before a run summary or any product is written, and the
13//! earlier products are preserved.

31/// The images of a run: one image per product, or a Taylor family of
32/// `terms` coefficients (`mtmfs`), whose PSF, sums of weights and weight
33/// images have `2·terms − 1` moments and whose primary beam is term 0's.
34#[derive(Clone, Copy, Debug, PartialEq, Eq)]
35pub(crate) enum Family {
36    /// `.model`, `.psf`, ...
37    Single,
38    /// `.model.tt0`, `.psf.tt0`, ...
39    Taylor {
40        /// The Taylor coefficients.
41        terms: usize,
42    },
43}
44
45impl fmt::Display for Family {
46    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
47        match self {
48            Self::Single => formatter.write_str("single-term"),
49            Self::Taylor { terms } => write!(formatter, "{terms}-term Taylor"),
50        }
51    }
52}
53
54impl Family {
55    pub(crate) fn of(request: &ImagingRequest) -> Self {
56        match request.deconvolver {
57            Deconvolver::Mtmfs => Self::Taylor {
58                terms: request.nterms,
59            },
60            _ => Self::Single,
61        }
62    }
63
64    /// The suffixes of `base`'s images (`model`, `psf`, ...), in term order.
65    pub(crate) fn names(self, base: &str) -> Vec<String> {
66        match self {
67            Self::Single => vec![base.to_string()],
68            Self::Taylor { terms } => {
69                let count = match base {
70                    "psf" | "sumwt" | "weight" => 2 * terms - 1,
71                    "pb" => 1,
72                    _ => terms,
73                };
74                (0..count).map(|term| format!("{base}.tt{term}")).collect()
75            }
76        }
77    }
78
79    /// Whether `suffix` names a product of a run of this family: the
80    /// images, their primary-beam corrections, the CLEAN mask and a Taylor
81    /// family's spectral index.
82    fn admits(self, suffix: &str) -> bool {
83        const BASES: [&str; 7] = ["model", "image", "residual", "psf", "sumwt", "weight", "pb"];
84        if suffix == "mask" {
85            return true;
86        }
87        let image = suffix.strip_suffix(".pbcor").unwrap_or(suffix);
88        let pbcor = image.len() != suffix.len();
89        match self {
90            Self::Single => {
91                BASES.contains(&image) && (!pbcor || image == "image")
92                    || matches!(suffix, "image.pbcor")
93            }
94            Self::Taylor { terms } => {
95                let is_primary_beam = image.starts_with("pb.tt")
96                    && (0..terms).any(|term| image == format!("pb.tt{term}"));
97                (!pbcor && is_primary_beam)
98                    || matches!(
99                        suffix,
100                        "alpha" | "alpha.error" | "alpha.pbcor" | "beta" | "beta.error"
101                    )
102                    || BASES
103                        .iter()
104                        .any(|base| self.names(base).iter().any(|name| name == image))
105                        && (!pbcor || image.starts_with("image."))
106            }
107        }
108    }
109}

113    matches!(suffix, "model" | "pb" | "psf" | "residual" | "sumwt")

116/// Check that `request` can continue on what exists at its `imagename`.
117/// Task entry points call this before creating a run summary.
118pub(crate) fn check(request: &ImagingRequest) -> Result<(bool, bool), PrepareError> {
119    let existing = continuation_products(request, &request.imagename)?;
120    let outliers = request
121        .outlierfile
122        .as_deref()
123        .map(|path| {
124            crate::compile::outlier_output_prefixes(path, request.task_workspace.as_deref())
125        })
126        .transpose()?
127        .unwrap_or_default();
128    let mut all = vec![(request.imagename.as_path(), existing)];
129    for output in &outliers {
130        all.push((output, products(output)?));
131    }
132    let continuing = all.iter().any(|(_, paths)| !paths.is_empty());
133    if !outliers.is_empty()
134        && !supports_outlier_restart(request)
135        && let Some(path) = all.iter().find_map(|(_, paths)| paths.first())
136    {
137        return Err(PrepareError::ExistingImageProduct { path: path.clone() });
138    }
139    let family = Family::of(request);
140    let mut model = false;
141    for (prefix, paths) in all {
142        model |= check_products(request, family, prefix, &paths)?;
143    }
144    check_inputs(request, continuing)?;
145    Ok((continuing, model))
146}

148fn supports_outlier_restart(request: &ImagingRequest) -> bool {
149    matches!(request.gridder, Gridder::Standard { facets } if facets.get() == 1)
150        && request.specmode == crate::SpecMode::Mfs
151        && request.deconvolver == Deconvolver::Hogbom
152        && request.weighting == crate::Weighting::Natural
153        && request.stokes == [crate::PolarizationCoordinate::StokesI]
154        && request.usemask == UseMask::User
155        && (request.calcpsf && request.calcres
156            || request.savemodel == crate::SaveModel::None && !request.pbcor)
157}

159pub(crate) fn supports_saved_automask(request: &ImagingRequest) -> bool {
160    matches!(request.gridder, Gridder::Standard { facets } if facets.get() == 1)
161        && request.specmode == crate::SpecMode::Mfs
162        && request.deconvolver == Deconvolver::Hogbom
163        && request.stokes == [crate::PolarizationCoordinate::StokesI]
164        && request.outlierfile.is_none()
165        && request.calcpsf
166        && request.calcres

169fn check_products(
170    request: &ImagingRequest,
171    family: Family,
172    prefix: &Path,
173    paths: &[PathBuf],
174) -> Result<bool, PrepareError> {
175    let mvc = request.mvc_output_channels.is_some() || request.mvc_auto_channels;
176    let mut has_model = false;
177    for path in paths {
178        let suffix = suffix(prefix, path);
179        let refuse = |reason: String| PrepareError::Restart {
180            path: path.clone(),
181            reason,
182        };
183        let cube_product = mvc && mvc_cube_product(&suffix);
184        if !family.admits(&suffix) && !cube_product {
185            return Err(refuse(format!(
186                "it is not a product of this run's {family} images; choose a new imagename (#944)"
187            )));
188        }
189        if suffix.starts_with("model") && !(mvc && suffix == "model") {
190            has_model |= path.exists();
191            if request.has_start_model() {
192                return Err(refuse(
193                    "startmodel cannot be supplied when the output model already exists".into(),
194                ));
195            }
196        }
197        if suffix == "mask" && request.niter > 0 {
198            if request.usemask == UseMask::AutoMultithresh && !supports_saved_automask(request) {
199                return Err(refuse(
200                    "continuing auto-multithresh on an existing CLEAN mask is not implemented for these controls (#944)"
201                        .into(),
202                ));
203            }
204            if prefix == request.imagename
205                && (!request.mask_images.is_empty()
206                    || request.mask_polygon.is_some()
207                    || !request.mask_box.is_empty()
208                    || request.explicit_empty_mask)
209            {
210                return Err(refuse(
211                    "merging a new mask into an existing CLEAN mask is not implemented yet".into(),
212                ));
213            }
214        }
215    }
216    Ok(has_model)
217}

219/// Whether a direction-dependent run reuses its PSF, weight and PB
220/// products for a residual pass or saved-product restoration.
221pub(crate) fn reuses_direction_dependent_psf(request: &ImagingRequest) -> bool {
222    if request.calcpsf {
223        return false;
224    }
225    match (&request.gridder, Family::of(request)) {
226        (Gridder::Mosaic { .. }, Family::Single) => {
227            if request.calcres {
228                matches!(
229                    request.specmode,
230                    crate::SpecMode::Mfs | crate::SpecMode::Cube
231                )
232            } else {
233                supports_saved_mosaic_residual(request)
234            }
235        }
236        (Gridder::Mosaic { .. }, Family::Taylor { .. }) => {
237            request.calcres && !request.pbcor && request.supports_separate_psf()
238        }
239        (Gridder::Awproject(_), Family::Taylor { terms }) => {
240            request.specmode == crate::SpecMode::Mfs
241                && if request.calcres {
242                    !request.pbcor || terms == 2
243                } else {
244                    terms == 2
245                        && request.restoration
246                        && request.savemodel == crate::SaveModel::None
247                        && !request.pbcor
248                }
249        }
250        _ => false,
251    }
252}

254/// Ordinary single-term CLEAN or restoration from a saved mosaic residual.
255fn supports_saved_mosaic_residual(request: &ImagingRequest) -> bool {
256    matches!(
257        request.gridder,
258        Gridder::Mosaic {
259            normtype: casa_imaging_model::ProductNormalization::FlatNoise,
260            ..
261        }
262    ) && matches!(
263        request.specmode,
264        crate::SpecMode::Mfs | crate::SpecMode::Cube
265    ) && request.deconvolver == Deconvolver::Hogbom
266        && request.weighting == crate::Weighting::Natural
267        && request.stokes == [crate::PolarizationCoordinate::StokesI]
268        && request.usemask == UseMask::User
269        && request.nsigma == 0.0
270        && request.pbmask == 0.0
271        && request.savemodel == crate::SaveModel::None
272}

274/// Whether the ordinary mosaic cube reforms its PSF before using its saved residual.
275pub(crate) fn rebuilds_mosaic_cube_psf(request: &ImagingRequest) -> bool {
276    request.calcpsf
277        && !request.calcres
278        && request.specmode == crate::SpecMode::Cube
279        && !request.uses_separate_psf()
280        && supports_saved_mosaic_residual(request)
281}

283/// Check that what `calcpsf=False` and `calcres=False` read exists and that
284/// casa-rs reads it for the request's gridder; `continuing` is whether the
285/// run continues on existing products at all.
286pub(crate) fn check_inputs(request: &ImagingRequest, continuing: bool) -> Result<(), PrepareError> {
287    if request.calcpsf && request.calcres {
288        return Ok(());
289    }
290    let refuse = |path: PathBuf, reason: String| Err(PrepareError::Restart { path, reason });
291    let flags = match (request.calcpsf, request.calcres) {
292        (false, false) => "calcpsf=False and calcres=False",
293        (false, true) => "calcpsf=False",
294        _ => "calcres=False",
295    };
296    if !request.calcpsf
297        && !request.calcres
298        && request.niter > 0
299        && request.pblimit > 0.0
300        && matches!(request.gridder, Gridder::Awproject(_))
301        && reuses_direction_dependent_psf(request)
302    {
303        return refuse(
304            request.imagename.clone(),
305            "AW CLEAN from a saved Taylor residual with a positive pblimit requires saved residual-mask support (#944)".into(),
306        );
307    }
308    let unsupported = match &request.gridder {
309        _ if request.outlierfile.is_some() && !supports_outlier_restart(request) => {
310            Some("with outlier fields")
311        }
312        Gridder::Standard { facets } | Gridder::Wproject { facets, .. } if facets.get() > 1 => {
313            Some("with facets")
314        }
315        Gridder::Standard { .. } | Gridder::Wproject { .. } => None,
316        Gridder::Mosaic { .. } | Gridder::Awproject(_)
317            if reuses_direction_dependent_psf(request) =>
318        {
319            None
320        }
321        Gridder::Mosaic { .. } if rebuilds_mosaic_cube_psf(request) => None,
322        Gridder::Mosaic { .. } => Some("with the mosaic gridder"),
323        Gridder::Awproject(_) => Some("with A-projection"),
324        Gridder::Awp2 { .. } => Some("with AWP2 (#1071)"),
325    };
326    if let Some(unsupported) = unsupported {
327        return refuse(
328            request.imagename.clone(),
329            format!("{flags} {unsupported} is not implemented yet (#980)"),
330        );
331    }
332    let family = Family::of(request);
333    let missing = |base: &str| {
334        family
335            .names(base)
336            .into_iter()
337            .map(|name| product_path(&request.imagename, &name))
338            .find(|path| !continuing || !path.exists())
339    };
340    if !request.calcpsf {
341        check_saved_psf(&request.imagename, family, continuing)?;
342    }
343    if reuses_direction_dependent_psf(request) {
344        for base in ["weight", "pb"] {
345            if let Some(path) = missing(base) {
346                let gridder = match request.gridder {
347                    Gridder::Mosaic { .. } => "the mosaic gridder",
348                    _ => "A-projection",
349                };
350                return refuse(
351                    path,
352                    format!("calcpsf=False with {gridder} requires the saved weight and PB images"),
353                );
354            }
355        }
356    }
357    if !request.calcres
358        && (request.niter > 0
359            || request.outlierfile.is_some()
360            || reuses_direction_dependent_psf(request)
361            || rebuilds_mosaic_cube_psf(request))
362        && let Some(path) = missing("residual")
363    {
364        return refuse(
365            path,
366            if request.niter > 0 {
367                "calcres=False with niter > 0 cleans the residual an earlier run left at imagename, \
368                 and this one does not exist"
369            } else {
370                "calcres=False requires the saved residual at imagename for this restart"
371            }
372            .into(),
373        );
374    }
375    Ok(())
376}

378fn check_saved_psf(prefix: &Path, family: Family, continuing: bool) -> Result<(), PrepareError> {
379    for base in ["psf", "sumwt"] {
380        if let Some(path) = family
381            .names(base)
382            .into_iter()
383            .map(|name| product_path(prefix, &name))
384            .find(|path| !continuing || !path.exists())
385        {
386            return Err(PrepareError::Restart {
387                path,
388                reason: concat!(
389                    "calcpsf=False reads the PSF and sums of weights an earlier run left at ",
390                    "imagename, and this one does not exist"
391                )
392                .into(),
393            });
394        }
395    }
396    Ok(())
397}

399/// Whether a `calcres=False` run reads a residual: it continues on an
400/// earlier run whose residual exists. A fresh PSF-only run can publish
401/// its start model independently of the residual.
402pub(crate) fn reads_residual(request: &ImagingRequest) -> bool {
403    !request.calcres
404        && request.tclean_restart
405        && Family::of(request)
406            .names("residual")
407            .iter()
408            .all(|name| product_path(&request.imagename, name).exists())
409}

411/// Whether the run forms or reads a residual for the residual product
412/// and, with `restoration`, the restored image.
413pub(crate) fn has_residual(request: &ImagingRequest) -> bool {
414    request.calcres || reads_residual(request)

417/// The product `prefix` names with `suffix`, appended to the prefix's
418/// bytes as publication appends it.
419pub(crate) fn product_path(prefix: &Path, suffix: &str) -> PathBuf {
420    let mut path = prefix.as_os_str().to_os_string();
421    path.push(".");
422    path.push(suffix);
423    path.into()
424}

426/// The suffix of `path`, a product of `prefix`, after its dot.
427pub(crate) fn suffix(prefix: &Path, path: &Path) -> String {
428    let path = path.as_os_str().to_string_lossy();
429    let prefix = prefix.as_os_str().to_string_lossy();
430    path.strip_prefix(prefix.as_ref())
431        .and_then(|suffix| suffix.strip_prefix('.'))
432        .unwrap_or("")
433        .to_string()
434}

436pub(crate) fn check_prefix(prefix: &Path) -> Result<(), PrepareError> {
437    if let Some(path) = products(prefix)?.into_iter().next() {
438        return Err(PrepareError::ExistingImageProduct { path });
439    }
440    Ok(())
441}

443/// Products to reuse on a continuation. A user may deliberately name a
444/// separate mask image such as `image.mask.dim.source` as an input. CASA
445/// opens `image.mask` as the output mask and reads the named mask independently.
446pub(crate) fn continuation_products(
447    request: &ImagingRequest,
448    prefix: &Path,
449) -> Result<Vec<PathBuf>, PrepareError> {
450    let mut existing = products(prefix)?;
451    let inputs: Vec<_> = request
452        .mask_images
453        .iter()
454        .filter_map(|path| fs::canonicalize(path).ok())
455        .collect();
456    if !inputs.is_empty() {
457        existing.retain(|path| {
458            !suffix(prefix, path).starts_with("mask.")
459                || fs::canonicalize(path).map_or(true, |path| !inputs.contains(&path))
460        });
461    }
462    Ok(existing)
463}

465pub(crate) fn products(prefix: &Path) -> Result<Vec<PathBuf>, PrepareError> {
466    let mut products = Vec::new();
467    // Publication appends to the original path bytes, including a trailing
468    // slash: `out/` writes `out/.model`, not the sibling `out.model`.
469    let model = product_path(prefix, "model");
470    let directory = model
471        .parent()
472        .filter(|p| !p.as_os_str().is_empty())
473        .unwrap_or(Path::new("."));
474    let entries = match fs::read_dir(directory) {
475        Ok(entries) => entries,
476        Err(error)
477            if matches!(
478                error.kind(),
479                io::ErrorKind::NotFound | io::ErrorKind::NotADirectory
480            ) =>
481        {
482            return Ok(products);
483        }
484        Err(source) => {
485            return Err(PrepareError::InspectOutputDirectory {
486                path: directory.into(),
487                source,
488            });
489        }
490    };
491    let Some(stem) = model.file_name() else {
492        return Ok(products);
493    };
494    let stem = stem.as_encoded_bytes();
495    let stem = stem.strip_suffix(b".model").expect("appended suffix");
496    for entry in entries {
497        let entry = entry.map_err(|source| PrepareError::InspectOutputDirectory {
498            path: directory.into(),
499            source,
500        })?;
501        let name = entry.file_name();
502        let Some(suffix) = name
503            .as_encoded_bytes()
504            .strip_prefix(stem)
505            .and_then(|name| name.strip_prefix(b"."))
506            .and_then(|bytes| std::str::from_utf8(bytes).ok())
507        else {
508            continue;
509        };
510        // Includes Taylor terms and derived products (.image.tt0.pbcor),
511        // orphan masks/PBs and dangling symlinks. A failed run's summary
512        // alone is not a scientific product and must permit a retry.
513        if matches!(
514            suffix.split('.').next(),
515            Some(
516                "image"
517                    | "model"
518                    | "residual"
519                    | "psf"
520                    | "sumwt"
521                    | "weight"
522                    | "pb"
523                    | "mask"
524                    | "alpha"
525                    | "beta"
526                    | "gridwt"
527            )
528        ) {
529            products.push(product_path(prefix, suffix));
530        }
531    }
532    products.sort();
533    Ok(products)
534}
