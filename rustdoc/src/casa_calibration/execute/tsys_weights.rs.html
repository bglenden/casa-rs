8pub(crate) struct TsysPlan {
9    plan: ApplyPlan,
10    table: LoadedCalibrationTable,
11    correlations: HashMap<i32, Vec<i32>>,
12    bandwidths: HashMap<i32, f64>,
13    exposures: Option<Vec<f64>>,
14}
15
16impl TsysPlan {
17    pub(crate) fn read(
18        ms: &MeasurementSet,
19        spec: ApplyCalibrationTableSpec,
20        by_exposure: bool,
21    ) -> Result<Self, ApplyExecutionError> {
22        let (plan, _) = plan_with_visibility_metadata(
23            ms,
24            &ApplyPlanRequest {
25                selection: casa_ms::MsSelection::default(),
26                apply_mode: ApplyMode::Trial,
27                parang: false,
28                calibration_tables: vec![spec],
29            },
30            None,
31        )?;
32        let planned_table = &plan.calibration_tables[0];
33        if planned_table.summary.table_subtype != "B TSYS" {
34            return Err(ApplyExecutionError::UnsupportedCalibrationTable {
35                path: planned_table.spec.path.display().to_string(),
36                reason: "weight initialization requires a B TSYS table".into(),
37            });
38        }
39        let ms_error = |source| ApplyExecutionError::MutateMeasurementSet {
40            path: display_ms_path(ms),
41            source,
42        };
43        let spw = ms.spectral_window().map_err(ms_error)?;
44        let bandwidths = plan
45            .measurement_set_spectral_windows
46            .iter()
47            .map(|window| {
48                let widths = spw.effective_bw(window.spw_id as usize)?;
49                Ok((
50                    window.spw_id,
51                    widths.iter().sum::<f64>() / widths.len() as f64,
52                ))
53            })
54            .collect::<MsResult<_>>()
55            .map_err(ms_error)?;
56        let exposures = by_exposure
57            .then(|| exposures(ms))
58            .transpose()
59            .map_err(ms_error)?;
60        let correlations = load_correlation_types_by_ddid(ms).map_err(ms_error)?;
61        let table = fields::load(planned_table, None, ms)?;
62        Ok(Self {
63            plan,
64            table,
65            correlations,
66            bandwidths,
67            exposures,
68        })
69    }

71    pub(crate) fn max_cell_shape(&self) -> (usize, usize) {
72        (
73            self.correlations.values().map(Vec::len).max().unwrap_or(1),
74            self.plan
75                .measurement_set_spectral_windows
76                .iter()
77                .map(|spw| spw.num_chan as usize)
78                .max()
79                .unwrap_or(1),
80        )
81    }

83    pub(crate) fn row_shapes(&self) -> impl Iterator<Item = [usize; 2]> + '_ {
84        (0..self.plan.selected_rows.len()).map(|index| self.row_shape(index))
85    }

87    pub(crate) fn row_shape(&self, index: usize) -> [usize; 2] {
88        let row = &self.plan.selected_rows[index];
89        [
90            self.correlations[&row.data_desc_id].len(),
91            window(&self.plan.measurement_set_spectral_windows, row.data_spw_id).num_chan as usize,
92        ]
93    }

95    pub(crate) fn values(
96        &self,
97        index: usize,
98        spectrum: bool,
99    ) -> Result<Option<Vec<ArrayValue>>, ApplyExecutionError> {
100        let row = &self.plan.selected_rows[index];
101        let plan = &self.plan.calibration_tables[0];
102        let data_spw = window(&self.plan.measurement_set_spectral_windows, row.data_spw_id);
103        let cal_spw_id = plan
104            .spw_mapping
105            .iter()
106            .find(|mapping| mapping.data_spw_id == row.data_spw_id)
107            .expect("planned data spectral window")
108            .calibration_spw_id;
109        let Some(cal_spw) = plan
110            .calibration_spectral_windows
111            .iter()
112            .find(|window| window.spw_id == cal_spw_id)
113        else {
114            return Ok(None);
115        };
116        let field = plan
117            .resolved_nearest_gainfields
118            .iter()
119            .find(|mapping| mapping.measurement_set_field_id == row.field_id)
120            .map_or(-1, |mapping| mapping.calibration_field_id);
121        let left = self.temperatures(row, field, row.antenna1, cal_spw, data_spw)?;
122        let right = self.temperatures(row, field, row.antenna2, cal_spw, data_spw)?;
123        let correlations = &self.correlations[&row.data_desc_id];
124        let factor = if row.antenna1 == row.antenna2 {
125            1.0
126        } else {
127            2.0
128        };
129        let exposure = self
130            .exposures
131            .as_ref()
132            .map_or(1.0, |values| values[row.row_index]);
133        let bandwidth = (factor * self.bandwidths[&row.data_spw_id] * exposure) as f32;
134        temperature_weights(
135            left.as_ref(),
136            right.as_ref(),
137            correlations,
138            data_spw.num_chan as usize,
139            bandwidth,
140            spectrum,
141        )
142        .map(Some)
143    }

145    fn temperatures(
146        &self,
147        row: &ApplyRowPlan,
148        field: i32,
149        antenna: i32,
150        input: &SpectralWindowPlan,
151        output: &SpectralWindowPlan,
152    ) -> Result<Option<ComplexSolution>, ApplyExecutionError> {
153        let grid = fields::lookup(
154            &self.table,
155            field,
156            input.spw_id,
157            antenna,
158            (row, row.scan_number),
159            output,
160            0.,
161        )?;
162        let Some(grid) = grid else { return Ok(None) };
163        let CalibrationGrid::Complex(grid) = grid.grid.as_ref() else {
164            return Ok(None);
165        };
166        let (input, output) = (
167            &input.channel_frequencies_hz,
168            &output.channel_frequencies_hz,
169        );
170        let (relative_input, relative_output);
171        let (input, output) = if matches!(
172            self.table.interp,
173            ApplyInterpolationMode::Casa {
174                relative_frequency: true,
175                ..
176            }
177        ) {
178            relative_input = crate::lookup_frequency::relative_axis(input);
179            relative_output = crate::lookup_frequency::relative_axis(output);
180            (relative_input.as_slice(), relative_output.as_slice())
181        } else {
182            (input.as_slice(), output.as_slice())
183        };
184        ComplexSolution::of(grid)
185            .resample_real_array_frequency(input, output, self.table.interp.frequency_mode())
186            .map(Some)
187    }
188}

197fn exposures(ms: &MeasurementSet) -> MsResult<Vec<f64>> {
198    use casa_tables::RequiredScalarColumnValues;
199
200    let mut columns = ms
201        .main_table()
202        .required_scalar_columns_owned(&["EXPOSURE"])?;
203    match columns.remove("EXPOSURE") {
204        Some(RequiredScalarColumnValues::Float64(values)) => Ok(values),
205        _ => Err(MsError::InvalidInput("EXPOSURE must be float64".into())),
206    }
207}

209fn temperature_weights(
210    left: Option<&ComplexSolution>,
211    right: Option<&ComplexSolution>,
212    correlations: &[i32],
213    channels: usize,
214    bandwidth: f32,
215    spectrum: bool,
216) -> Result<Vec<ArrayValue>, ApplyExecutionError> {
217    let mut weights = Vec::with_capacity(correlations.len());
218    let mut sigmas = Vec::with_capacity(correlations.len());
219    let shape = IxDyn(&[correlations.len(), if spectrum { channels } else { 0 }]).f();
220    let mut weight_spectrum = ArrayD::<f32>::zeros(shape.clone());
221    let mut sigma_spectrum = ArrayD::<f32>::zeros(shape);
222    for index in 0..correlations.len() {
223        let pair = if correlations.len() == 1 {
224            Some((0, 0))
225        } else {
226            bandpass::receptors(correlations, index)
227        };
228        let (p, q) = pair.ok_or_else(|| ApplyExecutionError::UnsupportedCorrelationLayout {
229            data_desc_id: -1,
230            correlation_types: correlations
231                .iter()
232                .map(|code| stokes_name(*code).into())
233                .collect(),
234        })?;
235        if spectrum {
236            let mut row_weights = Vec::with_capacity(channels);
237            let mut row_sigmas = Vec::with_capacity(channels);
238            for channel in 0..channels {
239                let u = valid_temperature(left, p, channel).map_or(0.0, f32::recip);
240                let v = valid_temperature(right, q, channel).map_or(0.0, f32::recip);
241                let weight = bandwidth * (u * v);
242                let sigma = weight.sqrt().recip();
243                weight_spectrum[[index, channel]] = weight;
244                sigma_spectrum[[index, channel]] = sigma;
245                row_weights.push(weight);
246                row_sigmas.push(sigma);
247            }
248            weights.push(weights::lower_median(row_weights));
249            sigmas.push(weights::lower_median(row_sigmas));
250        } else {
251            let weight =
252                bandwidth * (inverse_mean(left, p, channels) * inverse_mean(right, q, channels));
253            weights.push(weight);
254            sigmas.push(weight.sqrt().recip());
255        }
256    }
257    let mut output = vec![
258        ArrayValue::Float32(ndarray::Array1::from_vec(weights).into_dyn()),
259        ArrayValue::Float32(ndarray::Array1::from_vec(sigmas).into_dyn()),
260    ];
261    if spectrum {
262        output.extend([
263            ArrayValue::Float32(weight_spectrum),
264            ArrayValue::Float32(sigma_spectrum),
265        ]);
266    }
267    Ok(output)
268}

270fn valid_temperature(
271    solution: Option<&ComplexSolution>,
272    receptor: usize,
273    channel: usize,
274) -> Option<f32> {
275    let solution = solution?;
276    let receptor = receptor.min(solution.receptors.saturating_sub(1));
277    let value = solution.value(receptor, channel).re;
278    (!solution.flags[receptor * solution.channels + channel] && value >= 0.0).then_some(value)
279}
280
281fn inverse_mean(solution: Option<&ComplexSolution>, receptor: usize, channels: usize) -> f32 {
282    let (sum, count) = (0..channels)
283        .filter_map(|channel| valid_temperature(solution, receptor, channel))
284        .fold((0.0_f64, 0), |(sum, count), value| {
285            (sum + f64::from(value), count + 1)
286        });
287    // A missing solution or an all-zero temperature leaves the scalar
288    // weight factor at unity. The spectrum calculation uses each channel,
289    // so a valid zero channel there has infinite inverse temperature.
290    if sum == 0.0 {
291        1.0
292    } else {
293        (f64::from(count) / sum) as f32
294    }
