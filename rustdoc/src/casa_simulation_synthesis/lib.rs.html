15const GRIDDER_SUPPORT: usize = 3;
16const GRIDDER_TAP_COUNT: usize = GRIDDER_SUPPORT * 2 + 1;
17const GRIDDER_PRODUCT_TAP_COUNT: usize = GRIDDER_TAP_COUNT * GRIDDER_TAP_COUNT;

19/// Failure to construct a simulation synthesis operator.
20#[derive(Debug, Error, PartialEq, Eq)]
21pub enum SynthesisError {
22    /// The requested image geometry is not usable by the standard gridder.
23    #[error("invalid simulation synthesis request: {0}")]
24    InvalidRequest(String),
25}
26
27/// Two-dimensional image geometry for the simulation model plane.
28#[derive(Debug, Clone, Copy, PartialEq)]
29pub struct ImageGeometry {
30    /// `[nx, ny]` image shape in pixels.
31    pub image_shape: [usize; 2],
32    /// `[dx, dy]` positive pixel size in radians.
33    pub cell_size_rad: [f64; 2],
34}
35
36impl ImageGeometry {
37    fn nx(self) -> usize {
38        self.image_shape[0]
39    }
40
41    fn ny(self) -> usize {
42        self.image_shape[1]
43    }
44
45    fn validate(self) -> Result<(), SynthesisError> {
46        if self.nx() < 8 || self.ny() < 8 {
47            return Err(SynthesisError::InvalidRequest(
48                "image shape must be at least 8x8".to_string(),
49            ));
50        }
51        if !(self.cell_size_rad[0].is_finite()
52            && self.cell_size_rad[0] > 0.0
53            && self.cell_size_rad[1].is_finite()
54            && self.cell_size_rad[1] > 0.0)
55        {
56            return Err(SynthesisError::InvalidRequest(
57                "cell sizes must be finite positive radians".to_string(),
58            ));
59        }
60        Ok(())
61    }
62}

73/// Cached CASA-compatible Airy voltage-pattern evaluator.
74#[derive(Clone)]
75pub struct AiryVoltagePattern {
76    table: AnnularApertureVoltageTable,
77}
78
79impl AiryVoltagePattern {
80    /// Build a cached evaluator for one dish and blockage geometry.
81    pub fn new(model: AiryPrimaryBeam) -> Self {
82        let maximum_radius_arcmin_ghz =
83            casa_airy_max_radius_arcmin_ghz(model.dish_diameter_m, model.blockage_diameter_m);
84        Self {
85            table: AnnularApertureVoltageTable::new(
86                model.dish_diameter_m,
87                model.blockage_diameter_m,
88                maximum_radius_arcmin_ghz,
89            ),
90        }
91    }
92
93    /// Evaluate the voltage pattern at direction-cosine offsets in radians.
94    pub fn evaluate_offsets(&self, l_rad: f64, m_rad: f64, frequency_hz: f64) -> f32 {
95        if !(l_rad.is_finite()
96            && m_rad.is_finite()
97            && frequency_hz.is_finite()
98            && frequency_hz > 0.0)
99        {
100            return 0.0;
101        }
102        // Preserve CASA PBMath1D's Float rounding at the radius boundaries.
103        let l_deg = l_rad.to_degrees() as f32;
104        let m_deg = m_rad.to_degrees() as f32;
105        let radius_deg = (l_deg * l_deg + m_deg * m_deg).sqrt();
106        let radius_arcmin_ghz =
107            (f64::from(radius_deg) * 60.0 * (frequency_hz / 1.0e9)) as f32 as f64;
108        self.table.evaluate(radius_arcmin_ghz)
109    }
110}

112/// Serial standard-MFS predictor for one simulation model plane.
113pub struct StandardMfsModelPredictor {
114    gridder: StandardGridder,
115    model_grid: Option<Array2<Complex32>>,
116}
117
118impl StandardMfsModelPredictor {
119    /// Build a predictor for one image geometry and model plane.
120    pub fn new(geometry: ImageGeometry, model: &Array2<f32>) -> Result<Self, SynthesisError> {
121        let gridder = StandardGridder::new_with_casa_composite_padding(geometry)?;
122        let model_has_components = model.iter().any(|value| value.abs() > 0.0);
123        let model_grid = model_has_components.then(|| centered_fft2(&gridder.apodize_model(model)));
124        Ok(Self {
125            gridder,
126            model_grid,
127        })
128    }
129
130    /// Predict the model visibility at one `(u, v)` coordinate in wavelengths.
131    pub fn predict(&self, u_lambda: f64, v_lambda: f64) -> Complex32 {
132        let Some(model_grid) = self.model_grid.as_ref() else {
133            return Complex32::new(0.0, 0.0);
134        };
135        if let Some(predicted) = self
136            .gridder
137            .degrid_sample_product_planned_sectdgrid(model_grid, u_lambda, v_lambda)
138        {
139            return predicted;
140        }
141        let Some(plan) = self.gridder.plan_sample(u_lambda, v_lambda) else {
142            return Complex32::new(0.0, 0.0);
143        };
144        self.gridder
145            .degrid_sample_product_planned(model_grid, &plan)
146    }
147}

149#[derive(Clone, Copy)]
150struct TapSet {
151    indices: [usize; GRIDDER_TAP_COUNT],
152    weights: [f32; GRIDDER_TAP_COUNT],
153}
154
155#[derive(Clone, Copy)]
156struct ProductTapSet {
157    flat_indices: [usize; GRIDDER_PRODUCT_TAP_COUNT],
158    weights: [f32; GRIDDER_PRODUCT_TAP_COUNT],
159}
160
161struct StandardGridder {
162    geometry: ImageGeometry,
163    grid_shape: [usize; 2],
164    image_blc: [usize; 2],
165    oversampling: usize,
166    kernel_table: Vec<f32>,
167    normalized_tap_weights: Vec<[f32; GRIDDER_TAP_COUNT]>,
168    correction_x: Vec<f32>,
169    correction_y: Vec<f32>,
170    du_lambda: f64,
171    dv_lambda: f64,
172}
173
174impl StandardGridder {
175    fn new_with_casa_composite_padding(geometry: ImageGeometry) -> Result<Self, SynthesisError> {
176        geometry.validate()?;
177        let grid_shape = [
178            casa_composite_padded_len(geometry.nx(), 1.3),
179            casa_composite_padded_len(geometry.ny(), 1.3),
180        ];
181        let image_blc = [
182            (grid_shape[0] - geometry.nx() + usize::from(grid_shape[0].is_multiple_of(2))) / 2,
183            (grid_shape[1] - geometry.ny() + usize::from(grid_shape[1].is_multiple_of(2))) / 2,
184        ];
185        let oversampling = 100usize;
186        let mut kernel_table = vec![0.0f32; oversampling * (GRIDDER_SUPPORT + 1)];
187        for (index, kernel) in kernel_table
188            .iter_mut()
189            .enumerate()
190            .take(oversampling * GRIDDER_SUPPORT)
191        {
192            let distance = index as f64 / (GRIDDER_SUPPORT as f64 * oversampling as f64);
193            *kernel = spheroidal_kernel(distance * GRIDDER_SUPPORT as f64, GRIDDER_SUPPORT as f64);
194        }
195        let normalized_tap_weights = build_normalized_tap_weights(&kernel_table, oversampling);
196        let correction_x = build_correction_axis(grid_shape[0]);
197        let correction_y = build_correction_axis(grid_shape[1]);
198        Ok(Self {
199            geometry,
200            grid_shape,
201            image_blc,
202            oversampling,
203            kernel_table,
204            normalized_tap_weights,
205            correction_x,
206            correction_y,
207            du_lambda: 1.0 / (grid_shape[0] as f64 * geometry.cell_size_rad[0]),
208            dv_lambda: 1.0 / (grid_shape[1] as f64 * geometry.cell_size_rad[1]),
209        })
210    }
211
212    fn plan_sample(&self, u_lambda: f64, v_lambda: f64) -> Option<ProductTapSet> {
213        let x = self.sample_taps(self.grid_coordinate_x(u_lambda), self.grid_shape[0])?;
214        let y = self.sample_taps(self.grid_coordinate_y(v_lambda), self.grid_shape[1])?;
215        Some(flatten_tap_products(&x, &y, self.grid_shape[1]))
216    }
217
218    fn degrid_sample_product_planned(
219        &self,
220        grid: &Array2<Complex32>,
221        taps: &ProductTapSet,
222    ) -> Complex32 {
223        if let Some(storage) = grid.as_slice_memory_order() {
224            let mut value = Complex32::new(0.0, 0.0);
225            for tap in 0..GRIDDER_PRODUCT_TAP_COUNT {
226                value += storage[taps.flat_indices[tap]] * taps.weights[tap];
227            }
228            return value;
229        }
230        let mut value = Complex32::new(0.0, 0.0);
231        for tap in 0..GRIDDER_PRODUCT_TAP_COUNT {
232            let flat_index = taps.flat_indices[tap];
233            value += grid[(
234                flat_index / self.grid_shape[1],
235                flat_index % self.grid_shape[1],
236            )] * taps.weights[tap];
237        }
238        value
239    }
240
241    fn degrid_sample_product_planned_sectdgrid(
242        &self,
243        grid: &Array2<Complex32>,
244        u_lambda: f64,
245        v_lambda: f64,
246    ) -> Option<Complex32> {
247        let x_taps =
248            self.sample_taps_unnormalized(self.grid_coordinate_x(u_lambda), self.grid_shape[0])?;
249        let y_taps =
250            self.sample_taps_unnormalized(self.grid_coordinate_y(v_lambda), self.grid_shape[1])?;
251        let mut value = Complex32::new(0.0, 0.0);
252        let mut norm = 0.0f32;
253        if let Some(storage) = grid.as_slice_memory_order() {
254            let grid_stride = self.grid_shape[1];
255            for x_tap in 0..GRIDDER_TAP_COUNT {
256                let x_index = x_taps.indices[x_tap];
257                let x_weight = x_taps.weights[x_tap];
258                for y_tap in 0..GRIDDER_TAP_COUNT {
259                    let weight = x_weight * y_taps.weights[y_tap];
260                    value += storage[x_index * grid_stride + y_taps.indices[y_tap]] * weight;
261                    norm += weight;
262                }
263            }
264        } else {
265            for x_tap in 0..GRIDDER_TAP_COUNT {
266                let x_index = x_taps.indices[x_tap];
267                let x_weight = x_taps.weights[x_tap];
268                for y_tap in 0..GRIDDER_TAP_COUNT {
269                    let weight = x_weight * y_taps.weights[y_tap];
270                    value += grid[(x_index, y_taps.indices[y_tap])] * weight;
271                    norm += weight;
272                }
273            }
274        }
275        (norm > 0.0 && norm.is_finite()).then(|| value / norm)
276    }
277
278    fn apodize_model(&self, model: &Array2<f32>) -> Array2<Complex32> {
279        let mut apodized = Array2::<Complex32>::zeros((self.grid_shape[0], self.grid_shape[1]));
280        for x in 0..self.geometry.nx() {
281            for y in 0..self.geometry.ny() {
282                let grid_x = self.image_blc[0] + x;
283                let grid_y = self.image_blc[1] + y;
284                let correction = self.correction_x[grid_x] * self.correction_y[grid_y];
285                apodized[(grid_x, grid_y)] = Complex32::new(model[(x, y)] * correction, 0.0);
286            }
287        }
288        apodized
289    }
290
291    fn grid_coordinate_x(&self, u_lambda: f64) -> f64 {
292        u_lambda / self.du_lambda + self.grid_shape[0] as f64 / 2.0
293    }
294
295    fn grid_coordinate_y(&self, v_lambda: f64) -> f64 {
296        -v_lambda / self.dv_lambda + self.grid_shape[1] as f64 / 2.0
297    }
298
299    fn sample_taps(&self, coordinate: f64, size: usize) -> Option<TapSet> {
300        if !coordinate.is_finite() {
301            return None;
302        }
303        let anchor = coordinate.round() as isize;
304        let offset = ((anchor as f64 - coordinate) * self.oversampling as f64).round() as isize;
305        let start = anchor - GRIDDER_SUPPORT as isize;
306        let end = anchor + GRIDDER_SUPPORT as isize;
307        if start < 0 || end >= size as isize {
308            return None;
309        }
310        let half = self.oversampling as isize / 2;
311        let weight_index = offset + half;
312        if weight_index < 0 || weight_index >= self.normalized_tap_weights.len() as isize {
313            return None;
314        }
315        let mut indices = [0usize; GRIDDER_TAP_COUNT];
316        for (tap, index) in (start..=end).enumerate() {
317            indices[tap] = index as usize;
318        }
319        Some(TapSet {
320            indices,
321            weights: self.normalized_tap_weights[weight_index as usize],
322        })
323    }
324
325    fn sample_taps_unnormalized(&self, coordinate: f64, size: usize) -> Option<TapSet> {
326        if !coordinate.is_finite() {
327            return None;
328        }
329        let anchor = coordinate.round() as isize;
330        let offset = ((anchor as f64 - coordinate) * self.oversampling as f64).round() as isize;
331        let mut indices = [0usize; GRIDDER_TAP_COUNT];
332        let mut weights = [0.0f32; GRIDDER_TAP_COUNT];
333        let mut norm = 0.0f32;
334        for (tap, index) in
335            ((anchor - GRIDDER_SUPPORT as isize)..=(anchor + GRIDDER_SUPPORT as isize)).enumerate()
336        {
337            if index < 0 || index >= size as isize {
338                return None;
339            }
340            let delta = index - anchor;
341            let lookup = (delta * self.oversampling as isize + offset).unsigned_abs();
342            let weight = self.kernel_table.get(lookup).copied().unwrap_or(0.0);
343            indices[tap] = index as usize;
344            weights[tap] = weight;
345            norm += weight;
346        }
347        (norm > 0.0).then_some(TapSet { indices, weights })
348    }
349}

351fn centered_fft2(input: &Array2<Complex32>) -> Array2<Complex32> {
352    let mut transformed = shift2(input, true);
353    let mut fft = Fft2::<f32>::new([transformed.shape()[0], transformed.shape()[1]])
354        .expect("valid synthesis FFT shape");
355    fft.transform(
356        transformed
357            .as_slice_mut()
358            .expect("contiguous synthesis plane"),
359        false,
360    )
361    .expect("valid synthesis FFT plan");
362    shift2(&transformed, false)
363}
364
365fn shift2(input: &Array2<Complex32>, inverse: bool) -> Array2<Complex32> {
366    let nx = input.shape()[0];
367    let ny = input.shape()[1];
368    let mut output = Array2::<Complex32>::zeros((nx, ny));
369    let x_shift = if inverse { nx.div_ceil(2) } else { nx / 2 };
370    let y_shift = if inverse { ny.div_ceil(2) } else { ny / 2 };
371    for x in 0..nx {
372        for y in 0..ny {
373            output[(x, y)] = input[((x + x_shift) % nx, (y + y_shift) % ny)];
374        }
375    }
376    output
377}

379fn build_normalized_tap_weights(
380    kernel_table: &[f32],
381    oversampling: usize,
382) -> Vec<[f32; GRIDDER_TAP_COUNT]> {
383    let half = oversampling as isize / 2;
384    (-half..=half)
385        .map(|offset| {
386            let mut weights = [0.0f32; GRIDDER_TAP_COUNT];
387            let mut norm = 0.0f32;
388            for (tap, delta) in
389                (-(GRIDDER_SUPPORT as isize)..=(GRIDDER_SUPPORT as isize)).enumerate()
390            {
391                let lookup = (delta * oversampling as isize + offset).unsigned_abs();
392                let weight = kernel_table.get(lookup).copied().unwrap_or(0.0);
393                weights[tap] = weight;
394                norm += weight;
395            }
396            if norm > 0.0 {
397                for weight in &mut weights {
398                    *weight /= norm;
399                }
400            }
401            weights
402        })
403        .collect()
404}

406fn flatten_tap_products(x_taps: &TapSet, y_taps: &TapSet, stride: usize) -> ProductTapSet {
407    let mut flat_indices = [0usize; GRIDDER_PRODUCT_TAP_COUNT];
408    let mut weights = [0.0f32; GRIDDER_PRODUCT_TAP_COUNT];
409    let mut slot = 0usize;
410    for x_tap in 0..GRIDDER_TAP_COUNT {
411        for y_tap in 0..GRIDDER_TAP_COUNT {
412            flat_indices[slot] = x_taps.indices[x_tap] * stride + y_taps.indices[y_tap];
413            weights[slot] = x_taps.weights[x_tap] * y_taps.weights[y_tap];
414            slot += 1;
415        }
416    }
417    ProductTapSet {
418        flat_indices,
419        weights,
420    }
421}

423fn padded_len(image_len: usize, padding_factor: f64) -> usize {
424    let padded = (padding_factor * image_len as f64 - 0.5)
425        .floor()
426        .max(image_len as f64) as usize;
427    if padded.is_multiple_of(2) {
428        padded
429    } else {
430        padded + 1
431    }
432}
433
434fn casa_composite_padded_len(image_len: usize, padding_factor: f64) -> usize {
435    let mut padded = padded_len(image_len, padding_factor);
436    while !is_casa_composite_len(padded) {
437        padded += 2;
438    }
439    padded
440}
441
442fn is_casa_composite_len(mut value: usize) -> bool {
443    for factor in [2, 3, 5] {
444        while value > 1 && value.is_multiple_of(factor) {
445            value /= factor;
446        }
447    }
448    value == 1
449}

451fn build_correction_axis(size: usize) -> Vec<f32> {
452    let center = size as f64 / 2.0;
453    (0..size)
454        .map(|index| {
455            let nu = ((index as f64 - center).abs() / center).clamp(0.0, 1.0);
456            let value = grdsf(nu);
457            if value > 1.0e-6 {
458                (1.0 / value) as f32
459            } else {
460                0.0
461            }
462        })
463        .collect()
464}

466fn spheroidal_kernel(distance: f64, support: f64) -> f32 {
467    if !(distance.is_finite() && distance <= support) {
468        return 0.0;
469    }
470    let nu = distance / support;
471    if nu > 1.0 {
472        return 0.0;
473    }
474    ((1.0 - nu * nu) * grdsf(nu)) as f32
475}
476
477fn grdsf(nu: f64) -> f64 {
478    const P0: [f64; 5] = [
479        8.203_343e-2,
480        -3.644_705e-1,
481        6.278_660e-1,
482        -5.335_581e-1,
483        2.312_756e-1,
484    ];
485    const P1: [f64; 5] = [
486        4.028_559e-3,
487        -3.697_768e-2,
488        1.021_332e-1,
489        -1.201_436e-1,
490        6.412_774e-2,
491    ];
492    const Q0: [f64; 3] = [1.0, 8.212_018e-1, 2.078_043e-1];
493    const Q1: [f64; 3] = [1.0, 9.599_102e-1, 2.918_724e-1];
494    if !(0.0..=1.0).contains(&nu) {
495        return 0.0;
496    }
497    let (p, q, nu_end) = if nu < 0.75 {
498        (&P0, &Q0, 0.75)
499    } else {
500        (&P1, &Q1, 1.0)
501    };
502    let delta_nu_sq = nu * nu - nu_end * nu_end;
503    let numerator = p.iter().enumerate().fold(0.0, |sum, (order, coefficient)| {
504        sum + coefficient * delta_nu_sq.powi(order as i32)
505    });
506    let denominator = q.iter().enumerate().fold(0.0, |sum, (order, coefficient)| {
507        sum + coefficient * delta_nu_sq.powi(order as i32)
508    });
509    if denominator == 0.0 {
510        0.0
511    } else {
512        numerator / denominator
513    }
514}

516fn casa_airy_max_radius_arcmin_ghz(dish_diameter_m: f64, blockage_diameter_m: f64) -> f64 {
517    if approx_eq(dish_diameter_m, 10.7, 1.0e-6) && approx_eq(blockage_diameter_m, 0.75, 1.0e-6) {
518        250.0
519    } else if approx_eq(dish_diameter_m, 6.25, 1.0e-6)
520        && approx_eq(blockage_diameter_m, 0.75, 1.0e-6)
521    {
522        500.0
523    } else {
524        1.784 * 60.0
525    }
526}
527
528fn approx_eq(left: f64, right: f64, tolerance: f64) -> bool {
529    (left - right).abs() <= tolerance
530}
