2//! Imaging port step 3b's equality law (#932): the row geometry today's
3//! selected-observation route gives one imaging request, beside the batch
4//! geometry ([`GeometryPlan`]) of the same MeasurementSet read through the
5//! observation boundary, compared sample for sample. Evidence for the port
6//! only: it goes with today's route at the cutover.
7//!
8//! Rows are matched by MAIN row number: a MeasurementSet batch sample's
9//! through its grid ([`Msv2Observation::main_row`]), so padding matches
10//! nothing. A sample of another store is matched through its data
11//! description, its antennas by name and its time to the millisecond, a
12//! key two of the MeasurementSet's rows may share; such rows, and rows two
13//! samples claim, are counted as duplicates and fail the law.
14

35/// The law's tolerance: every quantity relative to its scale.
36pub const TOLERANCE: f64 = 1.0e-12;
37
38/// One request's two geometries compared.
39#[derive(Clone, Debug, Default)]
40pub struct GeometryComparison {
41    /// Rows today's route delivered.
42    pub rows: usize,
43    /// Of them, rows no batch sample matched.
44    pub unmatched: usize,
45    /// Rows today's route delivered twice, rows two batch samples matched,
46    /// and MeasurementSet rows that share another store's key.
47    pub duplicates: usize,
48    /// Per quantity, the largest difference relative to its scale and the
49    /// physical row where it occurred.
50    pub differences: BTreeMap<&'static str, (f64, u64)>,
51    /// Per quantity, the values today's route gave and the batch geometry
52    /// was compared on.
53    pub compared: BTreeMap<&'static str, usize>,
54}
55
56impl GeometryComparison {
57    /// Whether the law holds: there were rows, every one matched exactly
58    /// once, and no quantity differs by more than [`TOLERANCE`].
59    #[must_use]
60    pub fn holds(&self) -> bool {
61        self.rows > 0 && self.unmatched == 0 && self.duplicates == 0 && self.largest() <= TOLERANCE
62    }
63
64    /// The largest relative difference of any quantity.
65    #[must_use]
66    pub fn largest(&self) -> f64 {
67        self.differences
68            .values()
69            .map(|(difference, _)| *difference)
70            .fold(0.0, worst)
71    }
72
73    /// Record one difference; a NaN one (either side absent or not finite)
74    /// counts as infinite.
75    fn record(&mut self, quantity: &'static str, difference: f64, row: u64) {
76        *self.compared.entry(quantity).or_default() += 1;
77        let difference = worst(difference, 0.0);
78        let entry = self.differences.entry(quantity).or_insert((0.0, row));
79        if difference > entry.0 {
80            *entry = (difference, row);
81        }
82    }
83}
84
85/// The larger of two differences, a NaN infinite: `f64::max` would drop it.
86fn worst(a: f64, b: f64) -> f64 {
87    if a.is_nan() || b.is_nan() {
88        f64::INFINITY
89    } else {
90        a.max(b)
91    }
92}

94/// Rows by MAIN row number.

97/// A row's identity in another store: data description, antennas, and
98/// time in milliseconds (today's route delivers its time as days).
99type Key = (u32, u32, u32, i64);
100
101fn key(data_description: u32, antennas: [u32; 2], time_s: f64) -> Key {
102    (
103        data_description,
104        antennas[0],
105        antennas[1],
106        (time_s * 1.0e3).round() as i64,
107    )
108}

125/// Compare the geometry of `request`'s rows on today's route with the
126/// batch geometry of its MeasurementSet.
127///
128/// # Errors
129///
130/// When either route cannot be prepared or read.
131pub fn compare_geometry(
132    request: &ImagingRequest,
133    context: &RunContext,
134) -> Result<GeometryComparison, ApplicationError> {
135    let Prepared {
136        problem,
137        path,
138        measures,
139        today,
140    } = prepare(request, context)?;
141    let observation = Msv2Observation::open_with(&path, &OpenOptions::default())?;
142    let batches = msv2_rows(&problem, &observation, measures, &today.rows)?;
143    let mut comparison = compare(&today.rows, &batches);
144    comparison.duplicates += today.repeated;
145    Ok(comparison)
146}

148/// Compare the batch geometry of `request`'s rows in its MeasurementSet
149/// with their batch geometry in `converted`, the same observation in
150/// another store (an MSv4 conversion): what the store alone changes. A
151/// converted partition is matched to the data description of the
152/// MeasurementSet's partitions of the same content
153/// ([`PartitionEntry::same_content`](casa_observation::PartitionEntry::same_content)),
154/// and a converted antenna to the MeasurementSet's of its name.
155///
156/// # Errors
157///
158/// When either store cannot be opened or read.
159pub fn compare_stores(
160    request: &ImagingRequest,
161    context: &RunContext,
162    converted: &Path,
163) -> Result<GeometryComparison, ApplicationError> {
164    let Prepared {
165        problem,
166        path,
167        measures,
168        today,
169    } = prepare(request, context)?;
170    let observation = Msv2Observation::open_with(&path, &OpenOptions::default())?;
171    let reference = msv2_rows(&problem, &observation, Arc::clone(&measures), &today.rows)?;
172    let other = casa_ms::observation_io::open(converted, &OpenOptions::default())?;
173    let bridge = Bridge::new(&observation, other.catalogue());
174    let rows = batch_rows(
175        &problem,
176        other.as_ref(),
177        &bridge.partitions(),
178        |batch, sample| bridge.row(batch, sample, &today.keys),
179        measures,
180        &reference.rows,
181    )?;
182    let mut comparison = compare(&reference.rows, &rows);
183    comparison.rows = today.rows.len();
184    comparison.unmatched += today.rows.len() - reference.rows.len();
185    comparison.duplicates += today.repeated + reference.duplicates + today.shared;
186    Ok(comparison)
187}

189/// What both comparisons start from: the compiled problem of a request,

194    path: std::path::PathBuf,
195    measures: Arc<dyn MeasuresProvider>,

199/// Today's route's rows.
200struct Today {
201    rows: Rows,
202    /// Each row's [`Key`].
203    keys: HashMap<Key, u64>,
204    /// Rows delivered a second time.
205    repeated: usize,
206    /// Rows whose key another, earlier row has.
207    shared: usize,

210/// The rows a store's batch samples matched.
211struct Found {
212    rows: Rows,
213    /// Samples that matched a row an earlier sample had.
214    duplicates: usize,

217fn prepare(request: &ImagingRequest, context: &RunContext) -> Result<Prepared, ApplicationError> {
218    let resolved = std::cell::Cell::new(None);
219    let (problem, input, _native, _resident) = compile_run(request, context, &resolved)?;
220    let measures = input.observation.measures_provider();
221    let path = std::path::PathBuf::from(input.observation.locator());
222    let Admitted { value: access, .. } = finalize_source_access(
223        &problem,
224        input.initial_access,
225        imaging::row_samples(&problem)?,
226        &context.host,
227        &context.policy,
228    )?;
229    let today = today_rows(&problem, access)?;
230    Ok(Prepared {
231        problem,
232        path,
233        measures,
234        today,
235    })
236}

238/// `actual`'s rows against `expected`'s.
239fn compare(expected: &Rows, actual: &Found) -> GeometryComparison {
240    let mut comparison = GeometryComparison {
241        rows: expected.len(),
242        duplicates: actual.duplicates,
243        ..GeometryComparison::default()
244    };
245    for (row, want) in expected {
246        match actual.rows.get(row) {
247            Some(got) => compare_row(want, got, &mut comparison),
248            None => comparison.unmatched += 1,
249        }
250    }
251    comparison
252}

254/// Today's route's rows of `problem`.
255fn today_rows(
256    problem: &CompiledProblem,
257    access: casa_ms::ResolvedSelectedObservationAccess,
258) -> Result<Today, ApplicationError> {
259    let selected = access.into_deferred().open(problem)?;
260    let mut stream = selected.into_block_stream(problem);
261    let mut block = stream.create_storage();
262    let mut geometry = SelectedObservationNumericGeometry::new(
263        stream.maximum_rows_per_block(),
264        row_layout(problem).channels(),
265    )?;
266    let mut today = Today {
267        rows: HashMap::new(),
268        keys: HashMap::new(),
269        repeated: 0,
270        shared: 0,
271    };
272    while let Some(filled) = stream.fill_next(&mut block)? {
273        let projected = filled.project_numeric_geometry(problem, &mut geometry)?;
274        for index in 0..projected.row_count() {
275            let numeric = projected.numeric_row(index);
276            let row = numeric.row;
277            let channels = numeric.channels.len();
278            let frequencies = &projected.frequencies_hz()[index * channels..(index + 1) * channels];
279            let antennas = [row.metadata.antenna1, row.metadata.antenna2].map(|id| id as u32);
280            let time_s = row.coordinates.time.mjd_days() * 86_400.0;
281            let identity = key(row.data_description_id as u32, antennas, time_s);
282            let earlier = today.keys.insert(identity, row.physical_row);
283            if earlier.is_some_and(|earlier| earlier != row.physical_row) {
284                today.shared += 1;
285            }
286            let repeated = today.rows.insert(
287                row.physical_row,
288                Row {
289                    physical_row: row.physical_row,
290                    field: row.metadata.field_id as u32,
291                    raw_uvw_m: row.coordinates.raw_uvw_m,
292                    projections: row
293                        .domain_projections()
294                        .iter()
295                        .map(|projection| {
296                            let model = projection.model();
297                            (model.transformed_uvw_m(), model.phase_shift_m())
298                        })
299                        .collect(),
300                    aw_pixels: row
301                        .domain_projections()
302                        .iter()
303                        .map(|projection| projection.aw_pointing_pixel())
304                        .collect(),
305                    pointing: row.coordinates.pointing_directions,
306                    parallactic: row.coordinates.parallactic_angles_rad,
307                    responses: row.metadata.antenna_responses,
308                    frequencies: numeric
309                        .channels
310                        .iter()
311                        .zip(frequencies)
312                        .map(|(channel, frequency)| (channel.channel_index, *frequency))
313                        .collect(),
314                },
315            );
316            if repeated.is_some() {
317                today.repeated += 1;
318            }
319        }
320    }
321    Ok(today)
322}

324/// The batch geometry of `expected`'s rows in a MeasurementSet, each
325/// sample matched to its MAIN row through the partition's grid.
326fn msv2_rows(
327    problem: &CompiledProblem,
328    observation: &Msv2Observation,
329    measures: Arc<dyn MeasuresProvider>,
330    expected: &Rows,
331) -> Result<Found, ApplicationError> {
332    batch_rows(
333        problem,
334        observation,
335        &observation.catalogue().partition_ids(),
336        |batch, sample| {
337            let per_time = batch.samples_per_time();
338            observation
339                .main_row(
340                    batch.partition,
341                    batch.time.offsets_s()[sample / per_time],
342                    batch.baselines[sample % per_time],
343                )
344                .map(u64::from)
345        },
346        measures,
347        expected,
348    )
349}

351/// The batch geometry of `expected`'s rows in `partitions` of
352/// `observation`; `row_of` gives a sample's row, `None` for one that is
353/// no row of `expected`'s.
354fn batch_rows(
355    problem: &CompiledProblem,
356    observation: &dyn Observation,
357    partitions: &[PartitionId],
358    row_of: impl Fn(&VisibilityBatch<'_>, usize) -> Option<u64>,
359    measures: Arc<dyn MeasuresProvider>,
360    expected: &Rows,
361) -> Result<Found, ApplicationError> {
362    let catalogue = Arc::clone(observation.catalogue());
363    let plan = GeometryPlan::compile(problem, &catalogue, measures)?;
364    let mut traversal = GeometryTraversal::default();
365    let mut geometry = BatchGeometry::default();
366    let mut found = Found {
367        rows: HashMap::new(),
368        duplicates: 0,
369    };
370    let centroids = matches!(
371        problem.geometry().centres().pointing(),
372        PointingCentreLaw::Observation(law)
373            if law.time_sampling() == casa_imaging_model::PointingTimeSampling::VisibilityTimeCentroid
374    );
375    for &partition in partitions {
376        let entry = catalogue.partition(partition);
377        let data_group = entry
378            .data_groups
379            .first()
380            .ok_or_else(|| format!("partition {} has no data group", partition.0))?;
381        let request = AccessRequest {
382            partitions: vec![partition],
383            data_group: data_group.name.clone(),
384            roles: Roles {
385                values: false,
386                weights: false,
387                time_centroid: centroids,
388                ..Roles::PAYLOAD
389            },
390            channels: 0..entry.frequencies_hz.len(),
391            times: TimeSelection::All,
392            fields: FieldSelection::All,
393            weights: WeightRequest::Stored,
394            correlations: CorrelationRequest::ALL,
395            grouping: Grouping::None,
396            batch_bytes: 1 << 26,
397        };
398        let mut reader = observation.bind_read(&request)?;
399        let mut buffers = BatchBuffers::default();
400        while let Some(event) = reader.next(&mut buffers)? {
401            let ReadEvent::Batch(batch) = event else {
402                continue;
403            };
404            if batch.samples() == 0 {
405                continue;
406            }
407            let samples = (0..batch.samples())
408                .map(|sample| row_of(&batch, sample).filter(|row| expected.contains_key(row)))
409                .collect::<Vec<_>>();
410            let rows = samples.iter().map(Option::is_some).collect::<Vec<_>>();
411            let pointing = plan
412                .reads_pointing()
413                .then(|| observation.pointing(&pointing_request(problem, &batch, partition)))
414                .transpose()?;
415            plan.apply(
416                &batch,
417                &rows,
418                pointing.as_ref(),
419                &mut traversal,
420                &mut geometry,
421            )?;
422            for (sample, row) in samples.iter().enumerate() {
423                let Some(row) = row else {
424                    continue;
425                };
426                let got = batch_row(&batch, &geometry, sample, plan.charts(), &expected[row]);
427                if found.rows.insert(*row, got).is_some() {
428                    found.duplicates += 1;
429                }
430            }
431        }
432    }
433    Ok(found)
434}

436/// How another store's samples find the MeasurementSet's rows.
437struct Bridge {
438    /// Each matched converted partition's data description.
439    data_descriptions: BTreeMap<PartitionId, u32>,
440    /// Each converted antenna's MeasurementSet antenna.
441    antennas: Vec<Option<u32>>,
442}
443
444impl Bridge {
445    fn new(observation: &Msv2Observation, converted: &Catalogue) -> Self {
446        let reference = observation.catalogue();
447        let data_descriptions = converted
448            .partition_ids()
449            .into_iter()
450            .filter_map(|partition| {
451                let entry = converted.partition(partition);
452                let mut matching = reference
453                    .partition_ids()
454                    .into_iter()
455                    .filter(|candidate| reference.partition(*candidate).same_content(entry))
456                    .map(|candidate| observation.data_description_id(candidate));
457                let first = matching.next()?;
458                matching
459                    .all(|other| other == first)
460                    .then_some((partition, first))
461            })
462            .collect();
463        let antennas = converted
464            .antennas
465            .iter()
466            .map(|antenna| {
467                let mut matching = reference
468                    .antennas
469                    .iter()
470                    .enumerate()
471                    .filter(|(_, candidate)| same_antenna(candidate, antenna))
472                    .map(|(index, _)| index as u32);
473                let first = matching.next()?;
474                matching.next().is_none().then_some(first)
475            })
476            .collect();
477        Self {
478            data_descriptions,
479            antennas,
480        }
481    }
482
483    /// The converted partitions matched to a data description.
484    fn partitions(&self) -> Vec<PartitionId> {
485        self.data_descriptions.keys().copied().collect()
486    }
487
488    /// The MeasurementSet row `keys` gives `sample` of `batch`.
489    fn row(
490        &self,
491        batch: &VisibilityBatch<'_>,
492        sample: usize,
493        keys: &HashMap<Key, u64>,
494    ) -> Option<u64> {
495        let data_description = *self.data_descriptions.get(&batch.partition)?;
496        let per_time = batch.samples_per_time();
497        let baseline = batch.baselines[sample % per_time];
498        let antenna = |id: AntennaId| self.antennas.get(id.0 as usize).copied().flatten();
499        let time_s = batch
500            .time
501            .seconds_since(sample / per_time, MjdHighPrec::new(0.0, 0.0));
502        let antennas = [antenna(baseline.antenna1)?, antenna(baseline.antenna2)?];
503        keys.get(&key(data_description, antennas, time_s)).copied()
504    }
505}

507/// Whether `converted` names `reference`: as stored, or as XRADIO names an
508/// MSv4 antenna (`NAME_STATION`).
509fn same_antenna(reference: &AntennaEntry, converted: &AntennaEntry) -> bool {
510    converted.name == reference.name
511        || converted.name == format!("{}_{}", reference.name, reference.station)
512}

514/// The pointing request covering `batch`'s times for its partition's
515/// antennas.
516fn pointing_request(
517    problem: &CompiledProblem,
518    batch: &VisibilityBatch<'_>,
519    partition: PartitionId,
520) -> PointingRequest {
521    let mut antennas: Vec<AntennaId> = batch
522        .baselines
523        .iter()
524        .flat_map(|baseline| [baseline.antenna1, baseline.antenna2])
525        .collect();
526    antennas.sort_unstable();
527    antennas.dedup();
528    let times = batch.time.offsets_s();
529    let mut span = times[0]..=times[times.len() - 1];
530    if let Some(centroid) = batch.timing.centroid {
531        let offsets = centroid.offsets_s();
532        let low = offsets.iter().copied().fold(*span.start(), f64::min);
533        let high = offsets.iter().copied().fold(*span.end(), f64::max);
534        span = low..=high;
535    }
536    let direction = crate::imaging::pointing_direction(problem);
537    PointingRequest {
538        partition,
539        direction,
540        antennas,
541        times: span,
542        budget_bytes: 1 << 34,
543    }
544}

546/// Sample `sample`'s geometry as a row; `expected` gives the native
547/// channel indices of today's row.
548fn batch_row(
549    batch: &VisibilityBatch<'_>,
550    geometry: &BatchGeometry,
551    sample: usize,
552    charts: usize,
553    expected: &Row,
554) -> Row {
555    let time = sample / batch.samples_per_time();
556    let frequencies = geometry.frequencies_hz(time);
557    Row {
558        physical_row: expected.physical_row,
559        field: batch.field[time].0,
560        raw_uvw_m: batch
561            .group
562            .uvw
563            .map_or([f64::NAN; 3], |uvw| uvw.metres_at(sample)),
564        projections: (0..charts)
565            .map(|chart| {
566                let projection = geometry.projection(sample, chart);
567                (projection.uvw_m, projection.phase_shift_m)
568            })
569            .collect(),
570        aw_pixels: (0..charts)
571            .map(|chart| geometry.aw_pointing_pixel(sample, chart))
572            .collect(),
573        pointing: geometry.pointing(sample),
574        parallactic: geometry.parallactic_angles(sample),
575        responses: geometry.antenna_responses(sample),
576        frequencies: expected
577            .frequencies
578            .iter()
579            .map(|(channel, _)| {
580                let index = *channel as usize - batch.first_channel;
581                (
582                    *channel,
583                    frequencies.get(index).copied().unwrap_or(f64::NAN),
584                )
585            })
586            .collect(),
587    }
588}

590/// Record the differences of one row.
591fn compare_row(expected: &Row, actual: &Row, comparison: &mut GeometryComparison) {
592    let row = expected.physical_row;
593    let length = norm(expected.raw_uvw_m).max(1.0);
594    let exact = |same: bool| if same { 0.0 } else { f64::INFINITY };
595    comparison.record("field", exact(expected.field == actual.field), row);
596    comparison.record(
597        "raw uvw",
598        norm(difference(expected.raw_uvw_m, actual.raw_uvw_m)) / length,
599        row,
600    );
601    comparison.record(
602        "charts",
603        exact(expected.projections.len() == actual.projections.len()),
604        row,
605    );
606    for (want, got) in expected.projections.iter().zip(&actual.projections) {
607        comparison.record("uvw", norm(difference(want.0, got.0)) / length, row);
608        comparison.record("phase shift", (want.1 - got.1).abs() / length, row);
609    }
610    for (want, got) in expected.aw_pixels.iter().zip(&actual.aw_pixels) {
611        let difference = match (want, got) {
612            (None, None) => continue,
613            (Some(want), Some(got)) => {
614                ((want[0] - got[0]).hypot(want[1] - got[1])) / want[0].hypot(want[1]).max(1.0)
615            }
616            _ => f64::INFINITY,
617        };
618        comparison.record("AW pointing pixel", difference, row);
619    }
620    let angle = |want: SkyDirection, got: SkyDirection| {
621        worst(
622            exact(want.frame() == got.frame()),
623            worst(
624                (want.longitude_rad() - got.longitude_rad()).abs(),
625                (want.latitude_rad() - got.latitude_rad()).abs(),
626            ),
627        )
628    };
629    comparison.record(
630        "pointing",
631        worst(
632            angle(expected.pointing.antenna1, actual.pointing.antenna1),
633            angle(expected.pointing.antenna2, actual.pointing.antenna2),
634        ),
635        row,
636    );
637    match (expected.parallactic, actual.parallactic) {
638        (None, None) => {}
639        (Some(want), Some(got)) => comparison.record(
640            "parallactic angle",
641            worst((want[0] - got[0]).abs(), (want[1] - got[1]).abs()),
642            row,
643        ),
644        _ => comparison.record("parallactic angle", f64::INFINITY, row),
645    }
646    if expected.responses.is_some() || actual.responses.is_some() {
647        comparison.record(
648            "aperture classes",
649            exact(expected.responses == actual.responses),
650            row,
651        );
652    }
653    for ((_, want), (_, got)) in expected.frequencies.iter().zip(&actual.frequencies) {
654        comparison.record("frequency", (want - got).abs() / want.abs(), row);
655    }
656}

658fn difference(a: [f64; 3], b: [f64; 3]) -> [f64; 3] {
659    [a[0] - b[0], a[1] - b[1], a[2] - b[2]]
660}
