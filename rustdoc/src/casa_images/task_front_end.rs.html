2//! The image task binaries on the shared task front end (#931).
3//!
4//! Each binary's `main` builds its front end here, passing the surfaces it
5//! hosts and a callback that reads the resolved run into its typed request.
6//! The front end owns the command line, `--json-run`, Last and logging.
7
8use casa_provider_contracts::CasaValue;
9use casa_task_runtime::{TaskFailure, TaskFrontEnd, TaskInvocation, TaskSuccess};
10
11use crate::{ImageAnalysisTaskResult, ImageError, image_analysis_task_schema_bundle};
12
13/// The front end of the image binary `program`, which hosts `surfaces`.
14pub fn image_task_front_end<F>(program: &str, surfaces: &[&str], execute: F) -> TaskFrontEnd<F>
15where
16    F: Fn(&TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure>,
17{
18    TaskFrontEnd::new(
19        program,
20        image_analysis_task_schema_bundle(),
21        surfaces,
22        execute,
23    )
24    .expect("the image-analysis contract carries every image surface")
25}
26
27/// A product-writing image task's reply: CASA's return value `None`, and
28/// its summary as the result, which a command line prints as JSON.
29pub fn image_task_success(
30    result: Result<ImageAnalysisTaskResult, ImageError>,
31) -> Result<TaskSuccess, TaskFailure> {
32    image_task_returning(CasaValue::None, result)
33}
34
35/// A product-writing image task's reply that returns CASA's `value`, as
36/// `immath` returns `True`.
37pub fn image_task_returning(
38    value: CasaValue,
39    result: Result<ImageAnalysisTaskResult, ImageError>,
40) -> Result<TaskSuccess, TaskFailure> {
41    let result = result.map_err(|error| TaskFailure::execution(error.to_string()))?;
42    let text = serde_json::to_string_pretty(&result)
43        .map_err(|error| TaskFailure::execution(format!("serialize the result: {error}")))?;
44    Ok(TaskSuccess::new(value, &result)?.with_text(text + "\n"))
45}
