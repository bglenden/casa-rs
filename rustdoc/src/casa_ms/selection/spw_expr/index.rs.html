12/// The most iterations a stepped frequency search may take before it is
13/// refused: CASA's loop over single-precision frequencies either ends, stalls
14/// (and never returns), or would run for longer than any selection should.
15const STEP_LIMIT: u64 = 1 << 27;
16
17/// The most window indices a list mixing indices and frequencies may expand
18/// to; such a list is positional (see [`SpwIndex::window_list`]).
19const MIXED_LIMIT: u64 = 1 << 24;
20
21/// `MSSpwIndex` over the SPECTRAL_WINDOW rows.
22pub(super) struct SpwIndex<'r, 'a> {
23    windows: &'r [SpectralWindowRow<'a>],
24    /// Windows whose NAME is the decimal text of an integer at or past the
25    /// row count, by that integer: the first such row, as
26    /// `matchNameAsIntID` keeps (`MSSpwIndex.cc:82-98`).
27    numeric_names: BTreeMap<i64, usize>,
28}
29
30/// The windows a list of window indices names, after `matchNameAsIntID` and
31/// `matchId`.
32pub(super) struct WindowList {
33    pub(super) ids: Vec<usize>,
34    /// Some of the indices that name no window, and how many there are.
35    pub(super) missing: Vec<i64>,
36    pub(super) missing_count: u64,
37}
38
39impl WindowList {
40    fn miss(&mut self, id: i64, count: u64) {
41        if self.missing.len() < 16 {
42            self.missing.push(id);
43        }
44        self.missing_count += count;
45    }
46}

48impl<'r, 'a> SpwIndex<'r, 'a> {
49    pub(super) fn new(windows: &'r [SpectralWindowRow<'a>]) -> Self {
50        let count = windows.len() as i64;
51        let mut numeric_names = BTreeMap::new();
52        for (id, window) in windows.iter().enumerate() {
53            if let Ok(value) = window.name.parse::<i32>() {
54                let value = i64::from(value);
55                if value >= count && value.to_string() == window.name {
56                    numeric_names.entry(value).or_insert(id);
57                }
58            }
59        }
60        Self {
61            windows,
62            numeric_names,
63        }
64    }
65
66    pub(super) fn window(&self, id: usize) -> &SpectralWindowRow<'a> {
67        &self.windows[id]
68    }

70    /// `matchName` (`MSSpwIndex.cc:102-109`): every row whose NAME equals
71    /// `name`, flagged rows too.
72    pub(super) fn match_name(&self, name: &str) -> Vec<usize> {
73        self.ids_where(|id| self.windows[id].name == name)
74    }

76    /// `matchRegexOrPattern(pattern, False)` (`MSSpwIndex.cc:55-75`): every
77    /// row whose whole NAME the shell-style pattern matches.
78    pub(super) fn match_pattern(&self, pattern: &str) -> Result<Vec<usize>, String> {
79        let matcher = FullMatch::compile_pattern(pattern)?;
80        Ok(self.ids_where(|id| matcher.matches(self.windows[id].name)))
81    }

83    /// The rows `keep` accepts, in row order: `matchGT`, `matchLT` and
84    /// `matchGTAndLT` on identifiers (`MSSpwIndex.cc:344-371`).
85    pub(super) fn ids_where(&self, keep: impl Fn(usize) -> bool) -> Vec<usize> {
86        (0..self.windows.len()).filter(|&id| keep(id)).collect()
87    }

89    /// The rows whose REF_FREQUENCY `keep` accepts: the frequency forms of
90    /// `matchGT`, `matchLT` and `matchGTAndLT` (`MSSpwIndex.cc:313-340`).
91    pub(super) fn references_where(&self, keep: impl Fn(f64) -> bool) -> Vec<usize> {
92        self.ids_where(|id| keep(self.windows[id].ref_frequency_hz))
93    }

95    /// `matchFrequencyRange(f0, f1, approx, f3)` (`MSSpwIndex.cc:210-309`),
96    /// in single precision: a window's frequency is the `Float` midpoint of
97    /// its first and last channel. One frequency (`f0 == f1`, or `f1 < 0`)
98    /// selects the first window within its largest signed CHAN_WIDTH, `~f`
99    /// every window within its TOTAL_BANDWIDTH, a range every window whose
100    /// midpoint it holds, and a stepped range every window one of its steps
101    /// falls within the largest signed CHAN_WIDTH of.
102    pub(super) fn match_frequency_range(
103        &self,
104        (f0, f1): (f32, f32),
105        approx: bool,
106        f3: f32,
107    ) -> Result<Vec<usize>, String> {
108        let exact = !approx && (f1 < 0.0 || f0 == f1);
109        let mut terminations = Vec::new();
110        let mut ids = Vec::new();
111        for (id, window) in self.windows.iter().enumerate() {
112            let (widths, frequencies) = (window.chan_width_hz, window.chan_freq_hz);
113            let (Some(first), Some(last)) = (frequencies.first(), frequencies.last()) else {
114                return Err(no_channels(id));
115            };
116            if widths.is_empty() {
117                return Err(no_channels(id));
118            }
119            let max_width = widths.iter().copied().fold(f64::NEG_INFINITY, f64::max);
120            #[expect(clippy::cast_possible_truncation, reason = "C++ Float variables")]
121            let (step, reference) = (
122                if f3 < 0.0 {
123                    widths.iter().copied().fold(f64::INFINITY, f64::min) as f32
124                } else {
125                    f3
126                },
127                f64::midpoint(*last, *first) as f32,
128            );
129            let found = if approx {
130                #[expect(clippy::cast_possible_truncation, reason = "a C++ Float")]
131                let total = window.total_bandwidth_hz as f32;
132                (reference - f0).abs() <= total
133            } else if exact {
134                f64::from((reference - f0).abs()) < max_width
135            } else if f3 == 0.0 {
136                reference >= f0 && reference <= f1
137            } else {
138                stepped_match((f0, f1), step, (reference, max_width), &mut terminations)?
139            };
140            if found {
141                ids.push(id);
142                if exact {
143                    break;
144                }
145            }
146        }
147        if ids.is_empty() {
148            let range = if f0 == f1 {
149                format!("frequency {f0}")
150            } else {
151                format!("frequency range {f0}~{f1}")
152            };
153            return Err(format!("No matching SPW found for {range} Hz."));
154        }
155        Ok(ids)
156    }

158    /// The windows a list of window elements names: `convertToSpwIndex`
159    /// (`MSSpwIndex.cc:641-688`), then `matchNameAsIntID`, which replaces an
160    /// index past the last row by the first window NAMEd with its decimal
161    /// text, then `matchId` (`MSSpwIndex.cc:113-136`), which keeps the
162    /// indices of existing rows in order and reports the others.
163    ///
164    /// A list of indices alone is resolved without expanding the indices
165    /// past the last row, so `0~2000000000` costs no more than its windows.
166    /// A frequency element replaces the whole list built so far and leaves
167    /// the write position where it was, so a later index element overwrites
168    /// it from there and grows it with zeros; such a list is simulated
169    /// exactly, and refused where CASA would write past its end.
170    pub(super) fn window_list(&self, elements: &[Element]) -> Result<WindowList, String> {
171        if elements.iter().any(|element| element.unit == Unit::Hertz) {
172            return self.mixed_window_list(elements);
173        }
174        let mut list = WindowList {
175            ids: Vec::new(),
176            missing: Vec::new(),
177            missing_count: 0,
178        };
179        let count = self.windows.len() as i64;
180        for element in elements {
181            let Some(progression) = Progression::of(element)? else {
182                continue;
183            };
184            let negatives = progression.count_below(0);
185            if negatives > 0 {
186                list.miss(progression.start, negatives);
187            }
188            let mut value = progression.first_at_or_above(0);
189            while value < count && value <= progression.last {
190                list.ids
191                    .push(usize::try_from(value).expect("a nonnegative row index"));
192                value += progression.step;
193            }
194            let beyond = progression.count_from(count);
195            if beyond == 0 {
196                continue;
197            }
198            let mut named = 0;
199            for (&value, &id) in self.numeric_names.range(count..=progression.last) {
200                if progression.holds(value) {
201                    list.ids.push(id);
202                    named += 1;
203                }
204            }
205            if beyond > named {
206                let first_unnamed = (0..)
207                    .map(|k| progression.first_at_or_above(count) + k * progression.step)
208                    .find(|value| !self.numeric_names.contains_key(value))
209                    .expect("an unnamed index exists");
210                list.miss(first_unnamed, beyond - named);
211            }
212        }
213        Ok(list)
214    }

216    /// [`window_list`](Self::window_list) of a list holding a frequency.
217    fn mixed_window_list(&self, elements: &[Element]) -> Result<WindowList, String> {
218        let mut values: Vec<i64> = Vec::new();
219        let mut position = 0usize;
220        for element in elements {
221            if element.unit == Unit::Hertz {
222                values = self
223                    .match_frequency_range((element.start, element.stop), false, element.step)?
224                    .into_iter()
225                    .map(|id| id as i64)
226                    .collect();
227                continue;
228            }
229            let Some(progression) = Progression::of(element)? else {
230                continue;
231            };
232            let added = progression.count_from(progression.start);
233            if (values.len() as u64).saturating_add(added) > MIXED_LIMIT {
234                return Err(format!(
235                    "Spw expression: a window list mixing frequencies and more than \
236                     {MIXED_LIMIT} indices is not supported"
237                ));
238            }
239            values.resize(values.len() + added as usize, 0);
240            let mut value = progression.start;
241            while value <= progression.last {
242                let Some(slot) = values.get_mut(position) else {
243                    return Err("Spw expression: CASA writes this window list past its end \
244                                (an index after a frequency that matched fewer windows than \
245                                the indices before it)"
246                        .to_string());
247                };
248                *slot = value;
249                position += 1;
250                value += progression.step;
251            }
252        }
253        let count = self.windows.len() as i64;
254        let mut list = WindowList {
255            ids: Vec::new(),
256            missing: Vec::new(),
257            missing_count: 0,
258        };
259        for value in values {
260            let value = if value >= count {
261                self.numeric_names
262                    .get(&value)
263                    .map_or(value, |&id| id as i64)
264            } else {
265                value
266            };
267            match usize::try_from(value) {
268                Ok(id) if value < count => list.ids.push(id),
269                _ => list.miss(value, 1),
270            }
271        }
272        Ok(list)
273    }

275    /// `convertToChannelIndex` (`MSSpwIndex.cc:394-589`): a `[start, stop,
276    /// step]` per window and element, the whole window when there are no
277    /// elements. The messages it logs are appended to `warnings`.
278    pub(super) fn channel_triples(
279        &self,
280        spws: &[usize],
281        elements: Option<&[Element]>,
282        warnings: &mut Vec<String>,
283    ) -> Result<Vec<[i32; 3]>, String> {
284        let mut triples = Vec::with_capacity(spws.len() * elements.map_or(1, <[_]>::len));
285        let mut log = ChannelLog::default();
286        let mut some_failed = false;
287        let mut any_found = false;
288        for &spw in spws {
289            let window = &self.windows[spw];
290            let channels = i32::try_from(window.chan_freq_hz.len())
291                .ok()
292                .filter(|&channels| channels > 0)
293                .ok_or_else(|| no_channels(spw))?;
294            let Some(elements) = elements else {
295                triples.push([0, channels - 1, 1]);
296                continue;
297            };
298            for element in elements {
299                let triple = match element.unit {
300                    Unit::Index => {
301                        let (triple, clamped) =
302                            index_triple(element, spw, channels, (&mut log, &mut *warnings))?;
303                        some_failed |= clamped;
304                        any_found = true;
305                        triple
306                    }
307                    Unit::Hertz => {
308                        let triple = frequency_triple(element, spw, window, &mut log)?;
309                        let found = triple[0] != -1 && triple[1] != -1;
310                        some_failed |= !found;
311                        any_found |= found;
312                        triple
313                    }
314                };
315                triples.push(triple);
316            }
317        }
318        if some_failed && !any_found {
319            let ids = spws.iter().map(ToString::to_string).collect::<Vec<_>>();
320            log.post(
321                &format!("Found no matching SPW(s) [{}]", ids.join(", ")),
322                warnings,
323            );
324        }
325        log.finish(warnings);
326        Ok(triples)
327    }
328}

330/// `convertToChannelIndex`'s two message buffers: the `ostringstream`
331/// `Mesg`, which every clamp appends to and logs whole, and the `LogIO`
332/// `log_l`, which holds text written without `LogIO::POST` (the
333/// non-constant width notice) until its next post or the end of the call.
334#[derive(Default)]
335struct ChannelLog {
336    clamps: String,
337    pending: String,
338}
339
340impl ChannelLog {
341    /// A clamp: `Mesg << ...; log_l << Mesg.str() << LogIO::POST`.
342    fn clamp(&mut self, message: &str, warnings: &mut Vec<String>) {
343        self.clamps.push_str(message);
344        let posted = std::mem::take(&mut self.pending) + &self.clamps;
345        warnings.push(posted);
346    }
347
348    /// `log_l << message << LogIO::POST`.
349    fn post(&mut self, message: &str, warnings: &mut Vec<String>) {
350        warnings.push(std::mem::take(&mut self.pending) + message);
351    }
352
353    /// `log_l << message`, posted later.
354    fn hold(&mut self, message: &str) {
355        self.pending.push_str(message);
356    }
357
358    /// `log_l`'s destructor posts what is left.
359    fn finish(self, warnings: &mut Vec<String>) {
360        if !self.pending.is_empty() {
361            warnings.push(self.pending);
362        }
363    }
364}

366/// An index element's `[start, stop, step]` (`MSSpwIndex.cc:421-463`): `-1`
367/// is the window's first or last channel, a single channel clamps to the
368/// window, a range clamps each end, and a step below 1 is 1. Returns
369/// whether CASA logged a clamp.
370fn index_triple(
371    element: &Element,
372    spw: usize,
373    channels: i32,
374    (log, warnings): (&mut ChannelLog, &mut Vec<String>),
375) -> Result<([i32; 3], bool), String> {
376    let as_int = |value: f32| integer(value).map(|value| value as i32);
377    let (mut start, mut stop, step) = (
378        as_int(element.start)?,
379        as_int(element.stop)?,
380        as_int(element.step)?,
381    );
382    let last = channels - 1;
383    if start == -1 {
384        start = 0;
385    }
386    if stop == -1 {
387        stop = last;
388    }
389    let mut clamped = false;
390    if stop == start {
391        start = start.max(0);
392        stop = start;
393        if stop > last {
394            log.clamp(
395                &format!(
396                    "Spot-channel {stop} out of range for SPW {spw} (valid range 0~{last}). \
397                     Limiting it to be within the available range."
398                ),
399                warnings,
400            );
401            start = last;
402            stop = last;
403            clamped = true;
404        }
405    } else {
406        if stop > last {
407            log.clamp(
408                &format!(
409                    "Channel {stop} out of range for SPW {spw} (valid range 0~{last}). \
410                     Limiting it to be within the available range."
411                ),
412                warnings,
413            );
414            clamped = true;
415        }
416        start = start.min(last).max(0);
417        stop = stop.max(0).min(last);
418    }
419    Ok(([start, stop, if step <= 0 { 1 } else { step }], clamped))
420}

422/// A frequency element's `[start, stop, step]` (`MSSpwIndex.cc:464-536`):
423/// the first channel at or above the start and the last at or below the
424/// stop, by channel centres, read in the order the first CHAN_WIDTH's sign
425/// gives (not the frequencies' own order), -1 where none is; swapped when
426/// reversed; the step in hertz divided by the largest signed width.
427fn frequency_triple(
428    element: &Element,
429    spw: usize,
430    window: &SpectralWindowRow<'_>,
431    log: &mut ChannelLog,
432) -> Result<[i32; 3], String> {
433    let (frequencies, widths) = (window.chan_freq_hz, window.chan_width_hz);
434    let Some(&first_width) = widths.first() else {
435        return Err(no_channels(spw));
436    };
437    if first_width.abs() == 0.0 {
438        return Err("Error in the MS SPECTRAL_WINDOW sub-table (channel width==0).".to_string());
439    }
440    let ascending = first_width / first_width.abs() > 0.0;
441    let mut start = find_channel(element.start, frequencies, true, ascending);
442    let mut stop = find_channel(element.stop, frequencies, false, ascending);
443    let max_width = widths.iter().copied().fold(f64::NEG_INFINITY, f64::max);
444    let min_width = widths.iter().copied().fold(f64::INFINITY, f64::min);
445    if min_width != max_width {
446        log.hold(
447            "Channel width across the band is not constant.  Using the maximum of the channel \
448             width range.",
449        );
450    }
451    #[expect(clippy::cast_possible_truncation, reason = "a C++ Float")]
452    let step = (f64::from(element.step) / max_width).abs() as f32;
453    let step = integer(step)? as i32;
454    if start > stop {
455        std::mem::swap(&mut start, &mut stop);
456    }
457    Ok([start, stop, if step <= 0 { 1 } else { step }])
458}

460/// `findChanIndex_p` (`MSSpwIndex.cc:593-637`). The frequency is a C
461/// `Float`, compared with the `Double` channel centres.
462fn find_channel(frequency: f32, centres: &[f64], at_or_above: bool, ascending: bool) -> i32 {
463    let frequency = f64::from(frequency);
464    let last = centres.len() - 1;
465    let found = match (ascending, at_or_above) {
466        (true, true) if frequency <= centres[0] => Some(0),
467        (true, true) => centres.iter().position(|&c| c >= frequency),
468        (true, false) if frequency >= centres[last] => Some(last),
469        (true, false) => centres.iter().rposition(|&c| c <= frequency),
470        (false, true) if frequency <= centres[last] => Some(last),
471        (false, true) => centres.iter().rposition(|&c| c >= frequency),
472        (false, false) if frequency >= centres[0] => Some(0),
473        (false, false) => centres.iter().position(|&c| c <= frequency),
474    };
475    found.map_or(-1, |index| index as i32)
476}

482/// One stepped frequency search of [`SpwIndex::match_frequency_range`]:
483/// `for (Float freq = f0; freq <= f1; freq += step)`, until a step falls
484/// within `max_width` of `reference`. A search CASA never leaves (a step
485/// that no longer moves the single-precision frequency, or one that never
486/// reaches `f1`) is refused; `terminations` caches, per step, whether the
487/// whole loop ends, which does not depend on the window.
488fn stepped_match(
489    (f0, f1): (f32, f32),
490    step: f32,
491    (reference, max_width): (f32, f64),
492    terminations: &mut Vec<(u32, Result<(), String>)>,
493) -> Result<bool, String> {
494    let near = |frequency: f32| f64::from((frequency - reference).abs()) < max_width;
495    if step > 0.0 {
496        if max_width > 0.0 {
497            let mut frequency = f0;
498            let mut count = 0;
499            while frequency <= f1 {
500                if near(frequency) {
501                    return Ok(true);
502                }
503                // The differences only grow from here.
504                if f64::from(frequency - reference) >= max_width {
505                    break;
506                }
507                let next = frequency + step;
508                if next <= frequency {
509                    return Err(never_returns());
510                }
511                frequency = next;
512                count += 1;
513                if count > STEP_LIMIT {
514                    return Err(too_long());
515                }
516            }
517        }
518        let termination = match terminations
519            .iter()
520            .find(|(bits, _)| *bits == step.to_bits())
521        {
522            Some((_, termination)) => termination.clone(),
523            None => {
524                let termination = terminates((f0, f1), step);
525                terminations.push((step.to_bits(), termination.clone()));
526                termination
527            }
528        };
529        return termination.map(|()| false);
530    }
531    if step.is_nan() {
532        // One pass: the frequency becomes NaN and leaves the loop.
533        return Ok(f0 <= f1 && near(f0));
534    }
535    // A step of zero or below: the frequency never passes `f1`.
536    let mut frequency = f0;
537    let mut count = 0;
538    while frequency <= f1 {
539        if near(frequency) {
540            return Ok(true);
541        }
542        let unreachable = max_width.is_nan() || max_width <= 0.0;
543        if unreachable || f64::from(reference - frequency) >= max_width {
544            return Err(never_returns());
545        }
546        let next = frequency + step;
547        if next == frequency {
548            return Err(never_returns());
549        }
550        frequency = next;
551        count += 1;
552        if count > STEP_LIMIT {
553            return Err(too_long());
554        }
555    }
556    // `f0` above `f1`: the loop does not run.
557    Ok(false)
558}
559
560/// Whether `for (Float freq = f0; freq <= f1; freq += step)` ends.
561fn terminates((f0, f1): (f32, f32), step: f32) -> Result<(), String> {
562    let mut frequency = f0;
563    let mut count = 0;
564    while frequency <= f1 {
565        let next = frequency + step;
566        if next <= frequency {
567            return Err(never_returns());
568        }
569        frequency = next;
570        count += 1;
571        if count > STEP_LIMIT {
572            return Err(too_long());
573        }
574    }
575    Ok(())
576}

578fn never_returns() -> String {
579    "Spw expression: CASA's stepped frequency search for this range never ends (its \
580     single-precision frequency stops advancing or moves away from the range end)"
581        .to_string()
582}

591/// An index element's values: `for (Int ii = start; ii <= stop; ii += step)`
592/// with a step below 1 taken as 1 (`MSSpwIndex.cc:658-667`).
593struct Progression {
594    start: i64,
595    last: i64,
596    step: i64,
597}
598
599impl Progression {
600    /// The values of `element`, or `None` when there are none.
601    fn of(element: &Element) -> Result<Option<Self>, String> {
602        let start = i64::from(integer(element.start)? as i32);
603        let stop = i64::from(integer(element.stop)? as i32);
604        let step = i64::from(integer(element.step)? as i32).max(1);
605        if start > stop {
606            return Ok(None);
607        }
608        let last = start + (stop - start) / step * step;
609        if last + step > i64::from(i32::MAX) {
610            return Err(format!(
611                "Spw expression: the window range {start}~{stop}^{step} overflows CASA's C Int \
612                 loop"
613            ));
614        }
615        Ok(Some(Self { start, last, step }))
616    }
617
618    fn holds(&self, value: i64) -> bool {
619        value >= self.start && value <= self.last && (value - self.start) % self.step == 0
620    }
621
622    /// The first value at or above `bound`, past [`last`](Self::last) if
623    /// none.
624    fn first_at_or_above(&self, bound: i64) -> i64 {
625        if bound <= self.start {
626            self.start
627        } else {
628            self.start + (bound - self.start + self.step - 1) / self.step * self.step
629        }
630    }
631
632    /// How many values lie below `bound`.
633    fn count_below(&self, bound: i64) -> u64 {
634        self.count_from(self.start) - self.count_from(bound)
635    }
636
637    /// How many values lie at or above `bound`.
638    fn count_from(&self, bound: i64) -> u64 {
639        let first = self.first_at_or_above(bound);
640        if first > self.last {
641            0
642        } else {
643            ((self.last - first) / self.step + 1) as u64
644        }
645    }
