10pub(super) type Temperatures = [f64; 4];

12pub(super) struct RadiometerIntegration {
13    pub time_seconds: f64,
14    pub field_id: i32,
15    pub source_id: i32,
16    pub state_id: i32,
17    pub temperatures: Vec<Option<Temperatures>>,
18}

20#[derive(Clone, Copy)]
21pub(super) struct Observation {

23    pub field_id: i32,
24    pub source_id: i32,

29pub(super) struct RadiometerSeries {
30    pub integrations: Vec<RadiometerIntegration>,
31    pub sample_counts: Vec<usize>,
32    pub valid_counts: Vec<usize>,
33    pub fitted_antennas: Vec<bool>,
34    pub observations: Vec<Observation>,
35    pub statistic_window: Option<i32>,
36}
37
38impl RadiometerSeries {
39    pub fn read(
40        ms: &MeasurementSet,
41        wvr_windows: &[i32],
42        offsets_table: Option<&Table>,
43    ) -> Result<Self, MsError> {
44        let main = ms.main_table();
45        let antenna_count = ms.antenna()?.row_count();
46        let windows = ms.spectral_window()?;
47        let data_descriptions = ms.data_description()?;
48        // The loader follows the selected four-channel windows through DATA_DESCRIPTION.
49        let mut descriptions = vec![None; windows.row_count()];
50        let mut window_of_description = Vec::with_capacity(data_descriptions.row_count());
51        for desc in 0..data_descriptions.row_count() {
52            let window = data_descriptions.spectral_window_id(desc)?;
53            window_of_description.push(window);
54            let index = usize::try_from(window)
55                .ok()
56                .filter(|&index| index < windows.row_count())
57                .ok_or_else(|| invalid("WVR input has an invalid spectral-window reference"))?;
58            if windows.num_chan(index)? == 4
59                && (wvr_windows.is_empty() || wvr_windows.contains(&window))
60            {
61                descriptions[index].get_or_insert(desc as i32);
62            }
63        }
64        let statistic_window = wvr_windows.first().copied().or_else(|| {
65            descriptions
66                .iter()
67                .position(Option::is_some)
68                .map(|id| id as i32)
69        });
70        let descriptions: Vec<_> = descriptions.into_iter().flatten().collect();
71        let primary = descriptions
72            .iter()
73            .min()
74            .copied()
75            .ok_or_else(|| invalid("No four-channel WVR spectral windows selected"))?;
76        let metadata = main.required_scalar_columns_owned(&[
77            COL_TIME,
78            COL_FIELD_ID,
79            COL_ANTENNA1,
80            COL_ANTENNA2,
81            "DATA_DESC_ID",
82            "STATE_ID",
83        ])?;
84        let RequiredScalarColumnValues::Float64(times) = &metadata[COL_TIME] else {
85            return Err(invalid("MeasurementSet TIME is not Float64"));
86        };
87        let ints = |name| match &metadata[name] {
88            RequiredScalarColumnValues::Int32(values) => Ok(values.as_slice()),
89            _ => Err(invalid(format!("MeasurementSet {name} is not Int32"))),
90        };
91        let fields = ints(COL_FIELD_ID)?;
92        let antennas = ints(COL_ANTENNA1)?;
93        let antenna2 = ints(COL_ANTENNA2)?;
94        let descs = ints("DATA_DESC_ID")?;
95        let states = ints("STATE_ID")?;
96        let mut sorted: Vec<_> = (0..main.row_count())
97            .filter(|&row| times[row].is_finite())
98            .collect();
99        sorted.sort_by(|&left, &right| times[left].total_cmp(&times[right]));
100        let (observations, rows) = read_observations(
101            ms,
102            &sorted,
103            times,
104            fields,
105            descs,
106            &descriptions,
107            &window_of_description,
108        )?;
109        let fitted_antennas = wvr_antennas(ms, &rows, wvr_windows, antennas)?;
110
111        let mut offsets = vec![None; rows.len()];
112        if let Some(table) = offsets_table {
113            let samples = read_offsets(table)?;
114            let selected = rows
115                .iter()
116                .enumerate()
117                .filter(|&(_, &row)| antennas[row] == antenna2[row]);
118            if !samples.is_empty() && selected.clone().count() != samples.len() {
119                return Err(invalid(
120                    "Provided Offsets Table number of rows does not match the number of WVR data rows in MS",
121                ));
122            }
123            for ((index, &row), (time, antenna, values)) in selected.zip(samples) {
124                if time == times[row] && antenna == antennas[row] {
125                    offsets[index] = Some(values);
126                }
127            }
128        }
129        let field_table = ms.field()?;
130        let mut series = Self {
131            integrations: Vec::new(),
132            sample_counts: vec![0; antenna_count],
133            valid_counts: vec![0; antenna_count],
134            fitted_antennas,
135            observations,
136            statistic_window,
137        };
138        let rows: Vec<_> = rows.into_iter().zip(offsets).collect();
139        for group in rows.chunk_by(|left, right| times[left.0] == times[right.0]) {
140            let Some(&(first, _)) = group.iter().find(|entry| descs[entry.0] == primary) else {
141                continue;
142            };
143            let mut unflagged = false;
144            for &(row, _) in group.iter().filter(|entry| descs[entry.0] == primary) {
145                unflagged |= all_unflagged(main, row)?;
146                if unflagged {
147                    break;
148                }
149            }
150            if !unflagged {
151                continue;
152            }
153            let mut integration = RadiometerIntegration {
154                time_seconds: times[first],
155                field_id: fields[first],
156                source_id: field_table.source_id(fields[first] as usize)?,
157                state_id: states[first],
158                temperatures: vec![None; antenna_count],
159            };
160            for &(row, offset) in group
161                .iter()
162                .filter(|entry| antennas[entry.0] == antenna2[entry.0])
163            {
164                let antenna = usize::try_from(antennas[row])
165                    .ok()
166                    .filter(|&id| id < antenna_count)
167                    .ok_or_else(|| invalid("WVR data refer to an unknown antenna"))?;
168                series.sample_counts[antenna] += 1;
169                let reading = temperatures(main, row, offset.unwrap_or_default())?;
170                series.valid_counts[antenna] += usize::from(reading.is_some());
171                integration.temperatures[antenna] = reading;
172            }
173            series.integrations.push(integration);
174        }
175        Ok(series)
176    }
177}

179fn read_observations(
180    ms: &MeasurementSet,
181    sorted: &[usize],
182    times: &[f64],
183    fields: &[i32],
184    descriptions: &[i32],
185    selected_descriptions: &[i32],
186    window_of_description: &[i32],
187) -> Result<(Vec<Observation>, Vec<usize>), MsError> {
188    let field_table = ms.field()?;
189    let sources = (0..field_table.row_count())
190        .map(|field| field_table.source_id(field))
191        .collect::<Result<Vec<_>, _>>()?;
192    let mut observations = Vec::<Observation>::new();
193    let mut selected = Vec::new();
194    for &row in sorted {
195        let field_id = fields[row];
196        let source_id = *sources
197            .get(field_id as usize)
198            .ok_or_else(|| invalid("WVR observation has an unknown FIELD_ID"))?;
199        let description_id = descriptions[row];
200        let window_id = *window_of_description
201            .get(description_id as usize)
202            .ok_or_else(|| invalid("WVR observation has an unknown DATA_DESC_ID"))?;
203        if selected_descriptions.contains(&description_id) {
204            selected.push(row);
205        }
206        let time_seconds = times[row];
207        if let Some(previous) = observations.last_mut().filter(|previous| {
208            previous.time_seconds == time_seconds
209                && previous.field_id == field_id
210                && previous.window_id == window_id
211        }) {
212            previous.repeated = true;
213        } else {
214            observations.push(Observation {
215                time_seconds,
216                field_id,
217                source_id,
218                window_id,
219                repeated: false,
220            });
221        }
222    }
223    Ok((observations, selected))
224}

226fn wvr_antennas(
227    ms: &MeasurementSet,
228    selected: &[usize],
229    windows: &[i32],
230    antennas: &[i32],
231) -> Result<Vec<bool>, MsError> {
232    let mut present = vec![false; ms.antenna()?.row_count()];
233    let feed = ms.feed()?;
234    let spw = ms.spectral_window()?;
235    for row in 0..feed.row_count() {
236        let id = feed.i32(row, "SPECTRAL_WINDOW_ID")?;
237        let selected = if windows.is_empty() {
238            id >= 0 && (id as usize) < spw.row_count() && spw.num_chan(id as usize)? == 4
239        } else {
240            windows.contains(&id)
241        };
242        if selected && let Some(value) = present.get_mut(feed.i32(row, "ANTENNA_ID")? as usize) {
243            *value = true;
244        }
245    }
246    if !present.iter().any(|&value| value) {
247        for &row in selected {
248            if let Some(value) = present.get_mut(antennas[row] as usize) {
249                *value = true;
250            }
251        }
252    }
253    Ok(present)
254}

256fn all_unflagged(main: &Table, row: usize) -> Result<bool, MsError> {
257    let ArrayValue::Bool(flags) = main.cell_accessor(row, COL_FLAG)?.array()? else {
258        return Err(invalid("MeasurementSet FLAG is not a Boolean array"));
259    };
260    Ok(!flags.iter().any(|&flag| flag))
261}

263fn temperatures(
264    main: &Table,
265    row: usize,
266    offset: Temperatures,
267) -> Result<Option<Temperatures>, MsError> {
268    if !all_unflagged(main, row)? {
269        return Ok(None);
270    }
271    let ArrayValue::Complex32(data) = main.cell_accessor(row, "DATA")?.array()? else {
272        return Err(invalid("WVR DATA is not a Complex32 array"));
273    };
274    let [correlations, channels] = data.shape() else {
275        return Err(invalid("WVR DATA must have correlation and channel axes"));
276    };
277    if *correlations == 0 || correlations * channels < 4 {
278        return Err(invalid("WVR DATA must contain four temperatures"));
279    }
280    // The first four casacore cells are read in storage order.
281    let values = std::array::from_fn(|slot| {
282        f64::from(data[[slot % correlations, slot / correlations]].re) - offset[slot]
283    });
284    Ok(values
285        .iter()
286        .all(|&value| 2.7 < value && value < 300.0)
287        .then_some(values))
288}

290fn read_offsets(table: &Table) -> Result<Vec<(f64, i32, Temperatures)>, MsError> {
291    let mut offsets = Vec::with_capacity(table.row_count());
292    for row in 0..table.row_count() {
293        let ScalarValue::Float64(time) = table.cell_accessor(row, "TIME")?.scalar()? else {
294            return Err(invalid("Offsets Table TIME must be Float64"));
295        };
296        let ScalarValue::Int32(antenna) = table.cell_accessor(row, "ANTENNA")?.scalar()? else {
297            return Err(invalid("Offsets Table ANTENNA must be Int32"));
298        };
299        let ArrayValue::Float64(values) = table.cell_accessor(row, "OFFSETS")?.array()? else {
300            return Err(invalid("Offsets Table OFFSETS must be Float64"));
301        };
302        if values.shape() != [4] {
303            return Err(invalid("Offsets Table requires four offsets per row"));
304        }
305        let values = std::array::from_fn(|channel| values[[channel]]);
306        offsets.push((*time, *antenna, values));
307    }
308    offsets.sort_by(|left, right| left.0.total_cmp(&right.0));
309    Ok(offsets)
310}
