235/// One LSRK spectral window of an idealized synthetic observation.
236///
237/// Channel frequencies are used directly for prediction and recorded as LSRK
238/// in the MeasurementSet. The simulator does not model time-dependent TOPO
239/// tuning or Doppler resampling of an observed spectrum.
240#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
241pub struct SpectralSetup {
242    /// Spectral-window name.
243    pub name: String,
244    /// First channel center frequency in Hz, in LSRK.
245    pub start_frequency_hz: f64,
246    /// Channel width in Hz.
247    pub channel_width_hz: f64,
248    /// Number of channels.
249    pub channel_count: usize,
250}
251
252impl SpectralSetup {
253    /// Return the channel-center frequencies in Hz.
254    pub fn channel_frequencies_hz(&self) -> Vec<f64> {
255        (0..self.channel_count)
256            .map(|index| self.start_frequency_hz + index as f64 * self.channel_width_hz)
257            .collect()
258    }
259
260    /// Return the central reference frequency in Hz.
261    pub fn reference_frequency_hz(&self) -> f64 {
262        self.start_frequency_hz + (self.channel_count / 2) as f64 * self.channel_width_hz
263    }
264}

266/// Polarization/correlation layout for a synthetic observation.
267#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
268pub struct PolarizationSetup {
269    /// Receptor basis used in FEED and POLARIZATION metadata.
270    #[serde(default)]
271    pub basis: PolarizationBasis,
272    /// Number of correlations written per visibility row. Supported values are
273    /// 1, 2, and 4.
274    pub correlation_count: usize,
275    /// Fractional source Stokes `[Q/I, U/I, V/I]` applied to the scalar sky
276    /// model during synthetic visibility prediction. All zero, the model is
277    /// unpolarized, predicted as CASA predicts it: the parallel hands hold
278    /// Stokes I and the cross hands of four correlations are zero.
279    #[serde(default)]
280    pub source_stokes_fractional_quv: [f64; 3],
281}
282
283impl Default for PolarizationSetup {
284    fn default() -> Self {
285        Self {
286            basis: PolarizationBasis::Circular,
287            correlation_count: 2,
288            source_stokes_fractional_quv: [0.0; 3],
289        }
290    }
291}
292
293impl PolarizationSetup {
294    /// Build a setup from the dialog-facing correlation count.
295    pub fn new(basis: PolarizationBasis, correlation_count: usize) -> super::Result<Self> {
296        let setup = Self {
297            basis,
298            correlation_count,
299            source_stokes_fractional_quv: [0.0; 3],
300        };
301        setup.validate()?;
302        Ok(setup)
303    }
304
305    /// Validate that the correlation layout is supported by the native writer.
306    pub fn validate(&self) -> super::Result<()> {
307        match self.correlation_count {
308            1 | 2 | 4 => {}
309            other => Err(super::error(format!(
310                "polarization correlation_count {other} is unsupported; expected 1, 2, or 4"
311            )))?,
312        }
313        if self
314            .source_stokes_fractional_quv
315            .iter()
316            .any(|value| !value.is_finite())
317        {
318            return Err(super::error(
319                "source Stokes fractions must be finite".to_string(),
320            ));
321        }
322        let polarized_fraction = self
323            .source_stokes_fractional_quv
324            .iter()
325            .map(|value| value * value)
326            .sum::<f64>()
327            .sqrt();
328        if polarized_fraction > 1.0 + 1.0e-12 {
329            return Err(super::error(
330                "source fractional polarization must not exceed unity".to_string(),
331            ));
332        }
333        if polarized_fraction > 0.0
334            && (self.basis != PolarizationBasis::Linear || self.correlation_count != 4)
335        {
336            return Err(super::error(
337                "polarized synthetic prediction currently requires four linear correlations"
338                    .to_string(),
339            ));
340        }
341        Ok(())
342    }
343}

482/// Per-component spectral model for analytic components.
483#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema, PartialEq)]
484pub struct AnalyticSpectrum {
485    /// Continuum flux density in Jy at `reference_frequency_hz`.
486    pub flux_jy: f64,
487    /// Continuum spectral index.
488    #[serde(default)]
489    pub spectral_index: f64,
490    /// Optional spectral-index reference frequency in Hz.
491    #[serde(default)]
492    pub reference_frequency_hz: Option<f64>,
493    /// Additive Gaussian emission-line peak flux density in Jy.
494    #[serde(default)]
495    pub line_peak_jy: f64,
496    /// Emission-line center as a fraction of the channel range.
497    #[serde(default = "default_line_center_fraction")]
498    pub line_center_fraction: f64,
499    /// Emission-line sigma as a fraction of the channel range.
500    #[serde(default = "default_line_sigma_fraction")]
501    pub line_sigma_fraction: f64,
502    /// Subtractive Gaussian absorption-line peak flux density in Jy.
503    #[serde(default)]
504    pub absorption_peak_jy: f64,
505    /// Absorption-line center as a fraction of the channel range.
506    #[serde(default = "default_line_center_fraction")]
507    pub absorption_center_fraction: f64,
508    /// Absorption-line sigma as a fraction of the channel range.
509    #[serde(default = "default_line_sigma_fraction")]
510    pub absorption_sigma_fraction: f64,
511}
512
513fn default_line_center_fraction() -> f64 {
514    0.5
515}
516
517fn default_line_sigma_fraction() -> f64 {
518    0.1
519}
520
521impl AnalyticSpectrum {
522    /// Evaluate the component flux density for one channel of `window`.
523    ///
524    /// The continuum is a single power law across every window of the
525    /// observation, referenced to `reference_frequency_hz` or, when that is
526    /// absent, to `observation_reference_frequency_hz` (the first window's
527    /// reference frequency for a generated observation). Line and absorption
528    /// profiles are placed by fraction of `window`'s own channel range.
529    pub fn flux_for_channel(
530        &self,
531        window: &SpectralSetup,
532        channel: usize,
533        observation_reference_frequency_hz: f64,
534    ) -> f64 {
535        let reference_frequency_hz = self
536            .reference_frequency_hz
537            .unwrap_or(observation_reference_frequency_hz);
538        let frequency_hz = window.start_frequency_hz + channel as f64 * window.channel_width_hz;
539        let continuum = if reference_frequency_hz > 0.0 && frequency_hz > 0.0 {
540            self.flux_jy * (frequency_hz / reference_frequency_hz).powf(self.spectral_index)
541        } else {
542            self.flux_jy
543        };
544        continuum
545            + gaussian_channel_profile(
546                self.line_peak_jy,
547                self.line_center_fraction,
548                self.line_sigma_fraction,
549                window.channel_count,
550                channel,
551            )
552            - gaussian_channel_profile(
553                self.absorption_peak_jy,
554                self.absorption_center_fraction,
555                self.absorption_sigma_fraction,
556                window.channel_count,
557                channel,
558            )
559    }
560}
561
562fn gaussian_channel_profile(
563    peak_jy: f64,
564    center_fraction: f64,
565    sigma_fraction: f64,
566    channel_count: usize,
567    channel: usize,
568) -> f64 {
569    if peak_jy == 0.0 || sigma_fraction <= 0.0 || channel_count == 0 {
570        return 0.0;
571    }
572    let channel_fraction = if channel_count == 1 {
573        0.5
574    } else {
575        channel as f64 / (channel_count - 1) as f64
576    };
577    let offset = (channel_fraction - center_fraction) / sigma_fraction;
578    peak_jy * (-0.5 * offset * offset).exp()
579}

602    pub observer: String,

605    /// J2000 phase center `[right_ascension, declination]` in radians.
606    pub phase_center_rad: [f64; 2],

608    /// `phase_center_rad` define a single-field observation.
609    #[serde(default)]
610    pub fields: Vec<Field>,
611    /// Observation start time in MJD seconds UTC.
612    pub start_time_mjd_seconds: f64,
613    /// Requested on-source duration in seconds.
614    pub duration_seconds: f64,

616    pub integration_seconds: f64,

618    #[serde(default = "default_simobserve_elevation_limit_rad")]
619    pub elevation_limit_rad: f64,
620    /// Permit continuous tracks that include samples below the elevation limit.
621    ///
622    /// When false, the native simulator flags below-limit samples. Requested
623    /// observing times are retained; this developer helper does not reschedule.
624    #[serde(default)]
625    pub allow_below_elevation_limit: bool,
626    /// Antenna configuration.
627    pub antennas: Vec<Antenna>,
628    /// Spectral windows, observed simultaneously.
629    ///
630    /// Window `i` writes `SPECTRAL_WINDOW` row `i` and `DATA_DESCRIPTION` row
631    /// `i`, and every time sample writes one MAIN row per baseline per window
632    /// with `DATA_DESC_ID = i`. A FITS model's spectral planes and bandpass
633    /// corruption gains index the windows' channels concatenated in order.
634    pub spectral_windows: Vec<SpectralSetup>,
635    /// Polarization/correlation setup.

703/// Deterministic corruption controls for tutorial-grade synthetic observations.
704#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
705#[serde(deny_unknown_fields)]
706pub struct CorruptionConfig {
707    /// Seed used for deterministic random draws.
708    pub seed: u64,
709    /// Additive visibility noise, parameterized like CASA `simulator.setnoise`.
710    #[serde(default)]
711    pub noise: Option<NoiseCorruption>,
712    /// Per-antenna complex gain corruption, parameterized like CASA `simulator.setgain`.
713    #[serde(default)]
714    pub gain: Option<GainCorruption>,
715    /// Polarization leakage, parameterized like CASA `setleakage`: CASA's
716    /// linearised leakage of the cross hands with four correlations, applied
717    /// before the gains; with two, an approximation on the parallel hands,
718    /// applied after the gains and the noise. One correlation has no other
719    /// hand to leak into, so a leakage that changes the data is refused.
720    #[serde(default)]
721    pub leakage: Option<PolarizationLeakageCorruption>,
722}
