10/// Gaussian uv taper in baseline half-widths at half maximum.

13/// Which CASA cell arithmetic a density grid uses.
14#[derive(Clone, Copy, Debug, PartialEq, Eq)]
15pub enum DensityCellRule {
16    /// `VisImagingWeight`: single-precision coordinates truncated toward
17    /// zero, `v` not mirrored.
18    Standard,
19    /// `BriggsCubeWeightor`: rounded coordinates of the negated `u` and `v`,
20    /// built in double and looked up in single precision.
21    Cube,
22}
23
24/// A sample's density-cell coordinates in wavelengths, rounded the way CASA
25/// forms them before it picks a density cell (`VisImagingWeight::weightUniform`
26/// and its accumulation loop, `BriggsCubeWeightor::getWeightUniform`):
27/// `Float f = ν/c`, then `Float u = uvw_u·f` and `Float v = uvw_v·f`. At
28/// large image sizes one single-precision ulp is a visible fraction of a
29/// cell, so the rounding order decides which cell a boundary sample uses.
30/// Gridding and the uv taper keep the double coordinates of the
31/// [`Placement`].
32#[derive(Clone, Copy, Debug, PartialEq)]
33pub struct DensityUv {
34    /// u in wavelengths.
35    pub u: f32,
36    /// v in wavelengths.
37    pub v: f32,
38    /// CASA's `Float f = ν/c`, in wavelengths per metre.
39    pub wavelengths_per_m: f32,
40    /// The baseline's u and v in metres.
41    pub uv_m: [f64; 2],
42}
43
44impl DensityUv {
45    /// CASA's single-precision coordinates of baseline `uvw_m` at
46    /// `frequency_hz`.
47    #[must_use]
48    pub fn casa(uvw_m: [f64; 3], frequency_hz: f64) -> Self {
49        let per_m = (frequency_hz / SPEED_OF_LIGHT_M_PER_S) as f32;
50        let wavelengths_per_m = f64::from(per_m);
51        Self {
52            u: (uvw_m[0] * wavelengths_per_m) as f32,
53            v: (uvw_m[1] * wavelengths_per_m) as f32,
54            wavelengths_per_m: per_m,
55            uv_m: [uvw_m[0], uvw_m[1]],
56        }
57    }
58
59    /// The uv distance in wavelengths that CASA's radial weighting scales a
60    /// weight by (`VisImagingWeight::weightRadial`):
61    /// `f·sqrt(square(u) + square(v))`, the metre coordinates in double and
62    /// `f` the single-precision `ν/c`.
63    #[must_use]
64    pub fn radial_distance(self) -> f64 {
65        let [u, v] = self.uv_m;
66        f64::from(self.wavelengths_per_m) * (u * u + v * v).sqrt()
67    }
68}
69
70/// CASA `C::c`.
71pub(crate) const SPEED_OF_LIGHT_M_PER_S: f64 = 299_792_458.0;

73/// Shape of a density grid: image-sized cells per plane.
74#[derive(Clone, Copy, Debug, PartialEq)]
75pub struct DensityGridShape {
76    /// Cells along u (the image width).
77    pub width: usize,
78    /// Cells along v (the image height).
79    pub height: usize,
80    /// Density planes: 1 for a global generation; for a per-channel
81    /// generation the output channel count plus `padding` on each side; for
82    /// a per-field generation the FIELD_IDs from `first_field` on.
83    pub planes: usize,
84    /// Per-channel generation: density planes on each side of the output
85    /// axis (CASA `BriggsCubeWeightor::estimateSwingChanPad`), so native
86    /// channels just beyond the image still find their own plane. 0 for a
87    /// global generation.
88    pub padding: u32,
89    /// Half-width `b` of the box of `(2b + 1)²` cells about its own that
90    /// each sample adds to: CASA `superuniform`'s `uBox = vBox`
91    /// (`VisImagingWeight`) and a cube density's `superUniformBox`
92    /// (`BriggsCubeWeightor`), both from `npixels`. 0 adds to one cell.
93    pub box_half_width: usize,
94    /// Signed cell increments `[Δx, Δy]` in radians: the image's, or for
95    /// a square `npixels` grid CASA's `(n·Δ)/n` (`SynthesisImagerVi2::weight`).
96    pub increment_rad: [f64; 2],
97    /// Cell arithmetic.
98    pub rule: DensityCellRule,
99    /// A per-field generation (tclean's `mosweight`, `VisImagingWeight`'s
100    /// `multiField`), on [`DensityCellRule::Standard`] cells: plane `p` is
101    /// the density of FIELD_ID `first_field + p`, gridded from that field's
102    /// samples alone, and a sample reads its own field's plane. `None` for a
103    /// global or per-channel generation.
104    pub first_field: Option<u32>,
105}
106
107impl DensityGridShape {
108    /// Cells over every plane.
109    #[must_use]
110    pub const fn cells(&self) -> usize {
111        self.width * self.height * self.planes
112    }
113
114    /// Bytes a [`DensityGrid`] of this shape holds: an `f64` per cell and a
115    /// weight sum per plane. A weighting generation keeps its grid for the
116    /// whole run.
117    #[must_use]
118    pub const fn bytes(&self) -> u64 {
119        ((self.cells() + self.planes) * size_of::<f64>()) as u64
120    }
121
122    /// The cell increment CASA gives `VisImagingWeight` for a uniform or
123    /// Briggs density of `cells` cells along an image axis of increment
124    /// `increment_rad`: `SynthesisImagerVi2::weight` forms the field of view
125    /// `cells·Δ` and divides it by `cells` again, in double, whether `cells`
126    /// is `npixels` or the image's own size.
127    #[must_use]
128    pub fn visimagingweight_increment(cells: usize, increment_rad: f64) -> f64 {
129        (cells as f64 * increment_rad) / cells as f64
130    }
131
132    /// The density plane a sample on output plane `plane` from FIELD_ID
133    /// `field` grids into and reads: its field's for a per-field generation
134    /// (`None` outside its span), else `plane`, which `plane_of`
135    /// resolves.
136    #[must_use]
137    pub fn density_plane(&self, plane: u32, field: u32) -> Option<u32> {
138        match self.first_field {
139            Some(first) => field
140                .checked_sub(first)
141                .filter(|plane| (*plane as usize) < self.planes),
142            None => Some(plane),
143        }
144    }
145
146    /// Density plane of a placement: plane 0 for a global generation.
147    fn plane_of(&self, placement_plane: u32) -> Option<usize> {
148        if self.planes == 1 {
149            Some(0)
150        } else {
151            let plane = placement_plane as usize;
152            (plane < self.planes).then_some(plane)
153        }
154    }
155
156    /// CASA's integer cell origin `[nx/2, ny/2]` (`uorigin_p`, `vorigin_p`;
157    /// `GridFT`'s `offset_p` likewise): integer division, so an odd size
158    /// puts the origin on the lower of the two middle cells.
159    fn origin(&self) -> [usize; 2] {
160        [self.width / 2, self.height / 2]
161    }
162
163    /// CASA's `Float` cell scales `uscale_p = nx·Δx`, `vscale_p = ny·Δy`,
164    /// each rounded once from the double product
165    /// (`VisImagingWeight::VisImagingWeight`, `BriggsCubeWeightor::init`).
166    fn scale(&self) -> [f32; 2] {
167        [
168            (self.width as f64 * self.increment_rad[0]) as f32,
169            (self.height as f64 * self.increment_rad[1]) as f32,
170        ]
171    }
172
173    /// The first cell of the box a density placement adds to, the
174    /// `(2b + 1)²` cells about its own cell, when the whole box is accepted
175    /// (`VisImagingWeight`: `(ucell − uBox) > 0 && (ucell + uBox) < nx`, and
176    /// likewise in v; `GridFT`'s `onmygrid` likewise keeps a sample only
177    /// when its whole support lies on the grid). Standard placements carry
178    /// [`DensityUv`] coordinates; cube placements carry the double
179    /// coordinates CASA grids its weight density with (`GridFT::put`'s
180    /// `locuvw`: `nint(scale·u + offset + 1)` in Fortran indexing, of the
181    /// `u` and `v` that `FTMachine::negateUV` negated).
182    fn build_box(&self, plane: usize, u: f64, v: f64) -> Option<usize> {
183        let [x, y] = match self.rule {
184            DensityCellRule::Standard => self.standard_xy(u as f32, v as f32),
185            DensityCellRule::Cube => {
186                let [origin_x, origin_y] = self.origin().map(|origin| origin as f64);
187                let x = (-u * self.width as f64 * self.increment_rad[0] + origin_x + 1.0).round()
188                    as isize
189                    - 1;
190                let y = (-v * self.height as f64 * self.increment_rad[1] + origin_y + 1.0).round()
191                    as isize
192                    - 1;
193                [x, y]
194            }
195        };
196        let half = isize::try_from(self.box_half_width).unwrap_or(isize::MAX);
197        self.cell_index(plane, x.saturating_add(half), y.saturating_add(half))?;
198        self.cell_index(plane, x.saturating_sub(half), y.saturating_sub(half))
199    }
200
201    /// The share of a sample's weight each cell of its box takes.
202    /// `VisImagingWeight` adds the whole weight to every cell of the box;
203    /// `BriggsCubeWeightor` grids through `GridFT` with a convolution
204    /// function of ones over the box, which `sectggridd` divides by its sum
205    /// `(2b + 1)²`.
206    fn box_share(&self) -> f64 {
207        match self.rule {
208            DensityCellRule::Standard => 1.0,
209            DensityCellRule::Cube => {
210                let side = (2 * self.box_half_width + 1) as f64;
211                1.0 / (side * side)
212            }
213        }
214    }
215
216    /// The cell a lookup reads: `BriggsCubeWeightor::getWeightUniform`
217    /// rounds `uscale·u + uorigin` in `Float` with `u = −uvw_u·f` and
218    /// `v = −uvw_v·f`, as its grid was built. Evaluated in a mirrored
219    /// orientation, an interior cell differs only at an exact `Float` tie
220    /// (`k + ½`), where rounding half away from zero is not mirror-symmetric;
221    /// and on an odd width the accepted cells `0 < x < width` are not
222    /// mirror-symmetric about the origin `width / 2`, so the outermost
223    /// column would be read where CASA reads none, or the reverse.
224    fn lookup_cell(&self, plane: usize, uv: DensityUv) -> Option<usize> {
225        match self.rule {
226            DensityCellRule::Standard => self.standard_cell(plane, uv.u, uv.v),
227            DensityCellRule::Cube => {
228                let [scale_x, scale_y] = self.scale();
229                let [origin_x, origin_y] = self.origin().map(|origin| origin as f32);
230                let x = (scale_x * -uv.u + origin_x).round();
231                let y = (scale_y * -uv.v + origin_y).round();
232                self.cell_index(plane, x as isize, y as isize)
233            }
234        }
235    }
236
237    /// `VisImagingWeight`'s cell.
238    fn standard_cell(&self, plane: usize, u: f32, v: f32) -> Option<usize> {
239        let [x, y] = self.standard_xy(u, v);
240        self.cell_index(plane, x, y)
241    }
242
243    /// `VisImagingWeight`: `Int(uscale·u + uorigin)` in `Float`, truncated
244    /// toward zero.
245    fn standard_xy(&self, u: f32, v: f32) -> [isize; 2] {
246        let [scale_x, scale_y] = self.scale();
247        let [origin_x, origin_y] = self.origin().map(|origin| origin as f32);
248        [
249            (scale_x * u + origin_x) as isize,
250            (scale_y * v + origin_y) as isize,
251        ]
252    }
253
254    /// CASA accepts `0 < cell < extent` only.
255    fn cell_index(&self, plane: usize, x: isize, y: isize) -> Option<usize> {
256        if x <= 0 || y <= 0 || x >= self.width as isize || y >= self.height as isize {
257            return None;
258        }
259        Some((plane * self.height + y as usize) * self.width + x as usize)
260    }
261}

263/// Gridded weight density per plane, the input to uniform and Briggs
264/// weights.
265#[derive(Clone, Debug, PartialEq)]
266pub struct DensityGrid {
267    shape: DensityGridShape,
268    cells: Vec<f64>,
269    sum_weights: Vec<f64>,
270}
271
272impl DensityGrid {
273    /// An empty grid of `shape`, ready to [`DensityGrid::accumulate`] blocks.
274    #[must_use]
275    pub fn new(shape: DensityGridShape) -> Self {
276        Self {
277            cells: vec![0.0; shape.cells()],
278            sum_weights: vec![0.0; shape.planes],
279            shape,
280        }
281    }
282
283    /// Add one block of density samples.
284    ///
285    /// Blocks carry one polarization whose weight is CASA's unpolarized input
286    /// weight; each sample adds its weight's box share
287    /// ([`DensityGridShape::box_half_width`]) to every cell of the box about
288    /// its cell and of the box about the conjugate cell, each when that whole
289    /// box lies inside the grid (CASA tests the two independently), and its
290    /// weight once to the plane's weight sum when its own box does.
291    pub fn accumulate(&mut self, block: &SampleBlock<'_>) {
292        assert_eq!(block.npol, 1, "density blocks carry one unpolarized weight");
293        let share = self.shape.box_share();
294        let side = 2 * self.shape.box_half_width + 1;
295        for (placement, weight) in block.placements.iter().zip(block.weights) {
296            if *weight <= 0.0 {
297                continue;
298            }
299            let Some(plane) = self.shape.plane_of(placement.plane) else {
300                continue;
301            };
302            let weight = f64::from(*weight);
303            let primary = self.shape.build_box(plane, placement.u, placement.v);
304            let conjugate = self.shape.build_box(plane, -placement.u, -placement.v);
305            for first in primary.into_iter().chain(conjugate) {
306                // CASA's loops: v outer, u inner.
307                for row in 0..side {
308                    let start = first + row * self.shape.width;
309                    for cell in &mut self.cells[start..start + side] {
310                        *cell += weight * share;
311                    }
312                }
313            }
314            if primary.is_some() {
315                self.sum_weights[plane] += weight;
316            }
317        }
318    }
319
320    /// The shape.
321    #[must_use]
322    pub const fn shape(&self) -> &DensityGridShape {
323        &self.shape
324    }
325
326    /// Every cell, plane-major then `[y][x]`.
327    #[must_use]
328    pub fn cells(&self) -> &[f64] {
329        &self.cells
330    }
331
332    /// Cells of one plane.
333    #[must_use]
334    pub fn plane(&self, plane: usize) -> &[f64] {
335        let cells = self.shape.width * self.shape.height;
336        &self.cells[plane * cells..(plane + 1) * cells]
337    }
338
339    /// Input weight summed once per sample whose own box lies inside the
340    /// grid, per plane.
341    #[must_use]
342    pub fn sum_weights(&self) -> &[f64] {
343        &self.sum_weights
344    }
345
346    /// Density at the lookup cell of `uv` on `plane`; `None` outside the
347    /// grid.
348    #[must_use]
349    pub fn lookup(&self, plane: usize, uv: DensityUv) -> Option<f64> {
350        self.shape
351            .lookup_cell(plane, uv)
352            .map(|cell| self.cells[cell])
353    }
354
355    /// Briggs factors `f² = (5·10^{−robust})² / (Σd² / Σd)` per plane, with
356    /// `Σd` the density sum (standard) or twice the input weight sum (cube),
357    /// and `d² = 1` (`rmode = "norm"`).
358    #[must_use]
359    pub fn robust_factors(&self, robust: f64) -> RobustFactors {
360        let cells = self.shape.width * self.shape.height;
361        let f2 = (0..self.shape.planes)
362            .map(|plane| {
363                let density = &self.cells[plane * cells..(plane + 1) * cells];
364                let square_sum = density.iter().map(|value| value * value).sum::<f64>();
365                let sum = match self.shape.rule {
366                    DensityCellRule::Standard => density.iter().sum::<f64>(),
367                    DensityCellRule::Cube => 2.0 * self.sum_weights[plane],
368                };
369                if sum > 0.0 && square_sum > 0.0 {
370                    (5.0 * 10_f64.powf(-robust)).powi(2) / (square_sum / sum)
371                } else {
372                    0.0
373                }
374            })
375            .collect();
376        RobustFactors {
377            robust,
378            f2,
379            d2: 1.0,
380        }
381    }
382
383    /// Absolute Briggs factors (CASA `briggsabs`, `rmode = "abs"`):
384    /// `f² = robust²` and `d² = 2·noise²` on every plane, each held in
385    /// `Float` (`VisImagingWeight`'s `f2_p` and `d2_p`; `BriggsCubeWeightor`
386    /// likewise).
387    #[must_use]
388    pub fn absolute_factors(&self, robust: f64, noise_jy: f64) -> RobustFactors {
389        let f2 = f64::from((robust * robust) as f32);
390        RobustFactors {
391            robust,
392            f2: vec![f2; self.shape.planes],
393            d2: f64::from((2.0 * (noise_jy * noise_jy)) as f32),
394        }
395    }
396}

398/// CASA's Briggs factors for one robustness: `f²` per density plane and the
399/// offset `d²`, so that a sample whose cell has density `D` weighs
400/// `w / (D·f² + d²)`.
401#[derive(Clone, Debug, PartialEq)]
402pub struct RobustFactors {
403    robust: f64,
404    f2: Vec<f64>,
405    d2: f64,
406}
407
408impl RobustFactors {
409    /// The robustness parameter.
410    #[must_use]
411    pub const fn robust(&self) -> f64 {
412        self.robust
413    }
414
415    /// `f²` of `plane`.
416    #[must_use]
417    pub fn factor(&self, plane: usize) -> f64 {
418        self.f2[plane]
419    }
420
421    /// `d²`: 1 for normalized Briggs, `2·noise²` for absolute Briggs.
422    #[must_use]
423    pub const fn offset(&self) -> f64 {
424        self.d2
425    }
426}

428/// The density grid of every block from `source` (see
429/// [`DensityGrid::accumulate`]).
430pub fn build_density_grid<'a>(
431    source: impl Iterator<Item = SampleBlock<'a>>,
432    shape: DensityGridShape,
433) -> DensityGrid {
434    let mut grid = DensityGrid::new(shape);
435    for block in source {
436        grid.accumulate(&block);
437    }
438    grid
439}

441/// CASA `briggsbwtaper` (CAS-13021): the fractional bandwidth
442/// `2(ν_last − ν_first)/(ν_last + ν_first)` of the image's spectral axis,
443/// which turns each sample's uv distance in cells into a divisor of the
444/// Briggs density term so every cube channel is weighted as if its uv
445/// coverage had the continuum's radial smearing.
446#[derive(Clone, Copy, Debug, PartialEq)]
447pub struct BandwidthTaper {
448    fractional_bandwidth: f64,
449}
450
451impl BandwidthTaper {
452    /// From the first and last channel centres of the image's spectral axis
453    /// (`SynthesisImagerVi2::weight`); the two must be finite, positive and
454    /// distinct.
455    pub fn from_frequency_range(first_hz: f64, last_hz: f64) -> Result<Self, OperatorError> {
456        let low = first_hz.min(last_hz);
457        let high = first_hz.max(last_hz);
458        if !(low.is_finite() && high.is_finite()) || low <= 0.0 || high <= low {
459            return Err(OperatorError::Weighting {
460                reason: "the bandwidth taper needs two distinct positive frequencies",
461            });
462        }
463        Ok(Self {
464            fractional_bandwidth: 2.0 * (high - low) / (high + low),
465        })
466    }
467
468    /// `2(ν_last − ν_first)/(ν_last + ν_first)`.
469    #[must_use]
470    pub const fn fractional_bandwidth(self) -> f64 {
471        self.fractional_bandwidth
472    }
473
474    /// CASA `uvDistanceFactor`: `n + 0.5`, or `(4 − n)/(4 − 2n)` when that is
475    /// below 1.5, with `n` the fractional bandwidth times the sample's
476    /// distance from the uv origin in grid cells.
477    fn uv_distance_factor(self, shape: &DensityGridShape, u: f64, v: f64) -> f64 {
478        let u_cells = u * shape.width as f64 * shape.increment_rad[0];
479        let v_cells = v * shape.height as f64 * shape.increment_rad[1];
480        let cells = self.fractional_bandwidth * u_cells.hypot(v_cells);
481        let factor = cells + 0.5;
482        if factor < 1.5 {
483            (4.0 - cells) / (4.0 - 2.0 * cells)
484        } else {
485            factor
486        }
487    }
488}

490/// The imaging-weight rule of one run.
491#[derive(Clone, Debug, PartialEq)]
492pub enum WeightingGeneration {
493    /// Input weight, optionally tapered.
494    Natural {
495        /// Gaussian uv taper.
496        taper: Option<Taper>,
497    },
498    /// Input weight times the sample's uv distance in wavelengths
499    /// ([`DensityUv::radial_distance`], CASA `radial`), optionally tapered.
500    Radial {
501        /// Gaussian uv taper.
502        taper: Option<Taper>,
503    },
504    /// Uniform (`robust` absent), Briggs, absolute Briggs or Briggs
505    /// bandwidth-taper density weighting.
506    Density {
507        /// Gridded weight density.
508        grid: DensityGrid,
509        /// Briggs factors; `None` for uniform weighting.
510        robust: Option<RobustFactors>,
511        /// CASA `briggsbwtaper`; needs `robust`.
512        bandwidth: Option<BandwidthTaper>,
513        /// Gaussian uv taper.
514        taper: Option<Taper>,
515    },
516}
517
518impl WeightingGeneration {
519    /// Uniform, Briggs or Briggs bandwidth-taper weighting over `grid`.
520    pub fn density(
521        grid: DensityGrid,
522        robust: Option<f64>,
523        bandwidth: Option<BandwidthTaper>,
524        taper: Option<Taper>,
525    ) -> Result<Self, OperatorError> {
526        let robust = match robust {
527            Some(robust) if !robust.is_finite() => {
528                return Err(OperatorError::Weighting {
529                    reason: "robustness is not finite",
530                });
531            }
532            Some(robust) => Some(grid.robust_factors(robust)),
533            None => None,
534        };
535        if bandwidth.is_some() && robust.is_none() {
536            return Err(OperatorError::Weighting {
537                reason: "the bandwidth taper is a Briggs mode and needs a robustness",
538            });
539        }
540        Ok(Self::Density {
541            grid,
542            robust,
543            bandwidth,
544            taper,
545        })
546    }
547
548    /// CASA `briggsabs` over `grid`: `w / (D·robust² + 2·noise²)`
549    /// ([`DensityGrid::absolute_factors`]). Refused when both terms vanish,
550    /// as `SynthesisImagerVi2::weight` refuses `robust = 0` with `noise = 0`.
551    pub fn absolute_density(
552        grid: DensityGrid,
553        robust: f64,
554        noise_jy: f64,
555        taper: Option<Taper>,
556    ) -> Result<Self, OperatorError> {
557        if !(robust.is_finite() && noise_jy.is_finite()) {
558            return Err(OperatorError::Weighting {
559                reason: "absolute Briggs robustness and noise must be finite",
560            });
561        }
562        let factors = grid.absolute_factors(robust, noise_jy);
563        if factors.d2 == 0.0 && factors.f2.iter().all(|f2| *f2 == 0.0) {
564            return Err(OperatorError::Weighting {
565                reason: "absolute Briggs with robust and noise both zero divides by zero",
566            });
567        }
568        Ok(Self::Density {
569            grid,
570            robust: Some(factors),
571            bandwidth: None,
572            taper,
573        })
574    }
575
576    /// The padding of a per-channel density generation (CASA
577    /// `BriggsCubeWeightor`), whose output samples take the nearest native
578    /// channel's weight; `None` when native weights are interpolated
579    /// linearly (`VisImagingWeight`).
580    pub(crate) fn cube_padding(&self) -> Option<u32> {
581        match self {
582            Self::Density { grid, .. } if grid.shape.rule == DensityCellRule::Cube => {
583                Some(grid.shape.padding)
584            }
585            Self::Natural { .. } | Self::Radial { .. } | Self::Density { .. } => None,
586        }
587    }
588
589    /// The density plane a sample on output plane `plane` from FIELD_ID
590    /// `field` reads ([`DensityGridShape::density_plane`]); `plane` for a
591    /// weighting without a density.
592    #[must_use]
593    pub fn density_plane(&self, plane: u32, field: u32) -> Option<u32> {
594        match self {
595            Self::Density { grid, .. } => grid.shape.density_plane(plane, field),
596            Self::Natural { .. } | Self::Radial { .. } => Some(plane),
597        }
598    }
599
600    /// Imaging weight of one placement from its unpolarized input weight;
601    /// `density_uv` picks the density cell on the density plane
602    /// `placement.plane` (ignored by a global generation; a field's plane
603    /// for a per-field one, [`Self::density_plane`]) and gives radial
604    /// weighting its uv distance. Pure; CASA's cell rules live in
605    /// [`DensityGrid::lookup`].
606    #[must_use]
607    pub fn imaging_weight(
608        &self,
609        placement: &Placement,
610        density_uv: DensityUv,
611        input_weight: f32,
612    ) -> f32 {
613        if input_weight <= 0.0 {
614            return 0.0;
615        }
616        let input = f64::from(input_weight);
617        let (weighted, taper) = match self {
618            Self::Natural { taper } => (input, *taper),
619            Self::Radial { taper } => (density_uv.radial_distance() * input, *taper),
620            Self::Density {
621                grid,
622                robust,
623                bandwidth,
624                taper,
625            } => {
626                let Some(plane) = grid.shape.plane_of(placement.plane) else {
627                    return 0.0;
628                };
629                // CASA weighs a sample whose density cell is empty or off
630                // the grid at zero, under uniform and Briggs alike
631                // (`VisImagingWeight::weightUniform` requires `a_gwt > 0`;
632                // `BriggsCubeWeightor::getWeightUniform` likewise).
633                let Some(density) = grid
634                    .lookup(plane, density_uv)
635                    .filter(|density| *density > 0.0)
636                else {
637                    return 0.0;
638                };
639                let weighted = match robust {
640                    None => input / density,
641                    Some(factors) => {
642                        let distance = bandwidth.map_or(1.0, |bandwidth| {
643                            bandwidth.uv_distance_factor(&grid.shape, placement.u, placement.v)
644                        });
645                        input / (density * factors.factor(plane) / distance + factors.offset())
646                    }
647                };
648                (weighted, *taper)
649            }
650        };
651        (weighted * gaussian_taper(taper, placement.u, placement.v)) as f32
652    }
653}
654
655/// CASA `VisImagingWeight::filter`: an elliptical Gaussian in the uv plane.
656fn gaussian_taper(taper: Option<Taper>, u: f64, v: f64) -> f64 {
657    let Some(taper) = taper else {
658        return 1.0;
659    };
660    let sine = taper.position_angle_rad().sin();
661    let cosine = taper.position_angle_rad().cos();
662    let rotated_u = sine * u + cosine * v;
663    let rotated_v = -cosine * u + sine * v;
664    let major = std::f64::consts::LN_2 / taper.major_lambda().powi(2);
665    let minor = std::f64::consts::LN_2 / taper.minor_lambda().powi(2);
666    (-major * rotated_u.powi(2) - minor * rotated_v.powi(2)).exp()
667}
