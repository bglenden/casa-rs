17/// What the run takes from its antenna configuration.
18pub(super) struct Array {
19    /// The configuration, reduced to `sdant`'s antenna for a single dish.
20    pub(super) config: AntennaConfigFile,
21    /// `pb`: the primary beam, `1.13 λ/D` at the model's frequency for the
22    /// mean dish, arcsec.
23    pub(super) primary_beam_arcsec: f64,
24    /// `psfsize`: the resolution, arcsec: `approxBeam` of the whole
25    /// configuration for an interferometer, the primary beam for a single
26    /// dish.
27    pub(super) resolution_arcsec: f64,
28}

30/// `project + "." + <configuration>`: the antenna list's file name without
31/// `.cfg`, with `;` made `_` in ALMA resolution strings.
32pub(super) fn stem(project: &str, antennalist: &str) -> Result<String, SimobserveError> {
33    let name = if upper_prefix(antennalist, 4) == "ALMA" {
34        antennalist.replace(';', "_")
35    } else if !antennalist.is_empty() {
36        antennalist.to_string()
37    } else {
38        return Err(SimobserveError::raised(
39            "The name of antenna list (antennalist/sdantlist) is not specified",
40        ));
41    };
42    let name = name.replace(".cfg", "");
43    let name = name.split('/').next_back().unwrap_or_default();
44    Ok(format!("{project}.{name}"))
45}

47/// The first `count` characters of `text` in upper case, as Python's
48/// `str.upper(text[0:count])`.
49fn upper_prefix(text: &str, count: usize) -> String {
50    text.chars().take(count).collect::<String>().to_uppercase()
51}

53/// `ctsys.resolve("alma/simmos")`: the first data root holding it, or the
54/// unresolved name.
55fn repository(data_roots: &[PathBuf]) -> PathBuf {
56    data_roots
57        .iter()
58        .map(|root| root.join("alma").join("simmos"))
59        .find(|path| path.is_dir())
60        .unwrap_or_else(|| PathBuf::from("alma/simmos"))
61}

63/// Resolve the antenna list, read it, and form the primary beam and the
64/// resolution; for a single dish, keep `sdant`'s antenna.
65pub(super) fn resolve(run: &mut Run<'_>, input: &SkyInput) -> Result<Array, SimobserveError> {
66    let repository = repository(&run.request.data_roots);
67    let frequency_ghz = qa::value_in(&input.model.specrefval, "GHz")?;
68    if upper_prefix(&run.antennalist, 4) == "ALMA" {
69        run.antennalist = alma_configuration(&run.antennalist, frequency_ghz, &repository)?;
70    }
71    let path = search(run, &repository)?;
72    let full = read_antenna_config(&path, &*run.request.options.measures)?;
73    let mut config = full.clone();
74    let antennas = config.x.len();
75    if antennas == 1 {
76        if run.predict && run.uvmode {
77            return Err(SimobserveError::raised(
78                "antennalist contains only 1 antenna",
79            ));
80        }
81        run.uvmode = false;
82    }
83    if !run.uvmode {
84        config = single_dish(config, run.parameters.sdant);
85    }
86    let mean_diameter = numpy_mean(&config.dish_diameter);
87    let primary_beam_arcsec = PBCOEFF * 0.29979 / frequency_ghz / mean_diameter * 3600. * 180. / PI;
88    let resolution_arcsec = if run.uvmode {
89        approximate_beam(&full, frequency_ghz)?
90    } else {
91        if !input.components_only {
92            let size = &input.model.size;
93            let smallest = qa::value_in(&size[0], "arcsec")?.min(qa::value_in(&size[1], "arcsec")?);
94            if smallest < 2.5 * primary_beam_arcsec {
95                return Err(SimobserveError::raised(format!(
96                    "skymodel should be larger than 2.5*primary beam. Your skymodel: {smallest:.3} \
97                     arcsec < {:.3} arcsec: 2.5*primary beam",
98                    2.5 * primary_beam_arcsec
99                )));
100            }
101        }
102        primary_beam_arcsec
103    };
104    check_correlator(&run.parameters.correlator, &config.telescope)?;
105    Ok(Array {
106        config,
107        primary_beam_arcsec,
108        resolution_arcsec,
109    })
110}

112/// The configuration an ALMA resolution string names, in `repository`:
113/// Cycle 1's or Cycle 2's (`alma.cycle1;0.5arcsec`), else the full array's
114/// (`alma;0.5arcsec`), at `frequency_ghz`; any other string unchanged.
115fn alma_configuration(
116    antennalist: &str,
117    frequency_ghz: f64,
118    repository: &Path,
119) -> Result<String, SimobserveError> {
120    let upper = antennalist.to_uppercase();
121    let resolution = |tail: &str| -> Result<Option<f64>, SimobserveError> {
122        if !qa::is_quantity(tail) {
123            return Ok(None);
124        }
125        if !repository.exists() {
126            return Err(SimobserveError::raised(format!(
127                "failed to find antenna configuration repository at {}",
128                repository.display()
129            )));
130        }
131        Ok(Some(qa::value_in(&qa::read(tail)?, "arcsec")?))
132    };
133    let resolved = |name: String| -> Result<String, SimobserveError> {
134        let path = repository.join(name).display().to_string();
135        tracing::info!("converted resolution to antennalist {path}");
136        Ok(path)
137    };
138    if let Some(tail) = cycle_tail(&upper, '1')
139        && let Some(arcsec) = resolution(&tail.to_lowercase())?
140    {
141        let number = 1.044 - 6.733 * (arcsec * frequency_ghz / 345.).log10();
142        return resolved(format!("alma.cycle1.{}.cfg", configuration(number, 6.0)));
143    }
144    if let Some(tail) = cycle_tail(&upper, '2')
145        && let Some(arcsec) = resolution(&tail.to_lowercase())?
146    {
147        let number = 10_f64.powf(0.91 - 0.74 * (arcsec * frequency_ghz / 345.));
148        return resolved(format!("alma.cycle2.{}.cfg", configuration(number, 7.0)));
149    }
150    let tail: String = antennalist.chars().skip(5).collect();
151    if let Some(arcsec) = resolution(&tail)? {
152        let number = (2.867 - (arcsec * 1000. * frequency_ghz / 672.).log10()) / 0.0721;
153        return resolved(format!("alma.out{:02}.cfg", configuration(number, 28.0)));
154    }
155    Ok(antennalist.to_string())
156}

158/// `str(int(round(max(1, min(largest, number)))))`, Python rounding half to
159/// even.
160fn configuration(number: f64, largest: f64) -> i64 {
161    number.min(largest).max(1.0).round_ties_even() as i64
162}
163
164/// The tail Python's `re.match('.*CYCLE.?<cycle>.?;(.*)', upper)` captures:
165/// the greedy `.*` takes the last `CYCLE` that matches, and each `.?` tries
166/// one character before none. `.` matches anything but a newline.
167fn cycle_tail(upper: &str, cycle: char) -> Option<String> {
168    let chars: Vec<char> = upper.chars().collect();
169    let line_end = chars.iter().position(|&c| c == '\n').unwrap_or(chars.len());
170    let any = |index: usize| index < line_end;
171    let word: Vec<char> = "CYCLE".chars().collect();
172    if line_end < word.len() {
173        return None;
174    }
175    for start in (0..=line_end - word.len()).rev() {
176        if chars[start..start + word.len()] != word[..] {
177            continue;
178        }
179        let after = start + word.len();
180        for first in [1, 0] {
181            let digit = after + first;
182            if first == 1 && !any(after) || chars.get(digit) != Some(&cycle) || !any(digit) {
183                continue;
184            }
185            for second in [1, 0] {
186                let colon = digit + 1 + second;
187                if second == 1 && !any(digit + 1) || chars.get(colon) != Some(&';') {
188                    continue;
189                }
190                return Some(chars[colon + 1..line_end].iter().collect());
191            }
192        }
193    }
194    None
195}

197/// The antenna list's path, searched as CASA searches it: in the project
198/// directory, then as given, then in the repository.
199fn search(run: &Run<'_>, repository: &Path) -> Result<PathBuf, SimobserveError> {
200    let name = &run.antennalist;
201    let in_project = PathBuf::from(format!("{}/{name}", run.parameters.project));
202    let path = if in_project.exists() {
203        in_project
204    } else if !Path::new(name).exists() && repository.join(name).exists() {
205        repository.join(name)
206    } else {
207        PathBuf::from(name)
208    };
209    if !path.exists() {
210        return Err(SimobserveError::raised(format!(
211            "Couldn't find antennalist: {}",
212            path.display()
213        )));
214    }
215    Ok(path)
216}

218/// `config` reduced to antenna `sdant` (counted from the end when
219/// negative), or to the first when `sdant` is out of range.
220fn single_dish(mut config: AntennaConfigFile, sdant: i64) -> AntennaConfigFile {
221    let count = config.x.len() as i64;
222    let sdant = if sdant > count - 1 || sdant < -count {
223        tracing::warn!("antenna index {sdant} is out of range. setting sdant=0");
224        0
225    } else {
226        sdant
227    };
228    let index = (if sdant < 0 { count + sdant } else { sdant }) as usize;
229    let keep = |values: &mut Vec<f64>| *values = vec![values[index]];
230    keep(&mut config.x);
231    keep(&mut config.y);
232    keep(&mut config.z);
233    keep(&mut config.dish_diameter);
234    config.pad_names = vec![config.pad_names[index].clone()];
235    config.antenna_names = vec![config.antenna_names[index].clone()];
236    config
237}

239/// NumPy's mean of `values`: their pairwise sum (`add.reduce`) over their
240/// count.
241pub(super) fn numpy_mean(values: &[f64]) -> f64 {
242    pairwise_sum(values) / values.len() as f64
243}
244
245/// NumPy's pairwise summation: plain below 8 values, eight running sums up
246/// to 128, and halves (kept to multiples of 8) above.
247fn pairwise_sum(values: &[f64]) -> f64 {
248    let count = values.len();
249    if count < 8 {
250        return values.iter().fold(0.0, |sum, value| sum + value);
251    }
252    if count <= 128 {
253        let mut sums: [f64; 8] = std::array::from_fn(|index| values[index]);
254        let whole = count - count % 8;
255        for block in values[8..whole].as_chunks::<8>().0 {
256            for (sum, value) in sums.iter_mut().zip(block) {
257                *sum += value;
258            }
259        }
260        let sum = ((sums[0] + sums[1]) + (sums[2] + sums[3]))
261            + ((sums[4] + sums[5]) + (sums[6] + sums[7]));
262        return values[whole..].iter().fold(sum, |sum, value| sum + value);
263    }
264    let half = count / 2;
265    let half = half - half % 8;
266    pairwise_sum(&values[..half]) + pairwise_sum(&values[half..])
267}

269/// CASA's refusal of a correlator the telescope does not have.
270fn check_correlator(correlator: &str, telescope: &str) -> Result<(), SimobserveError> {
271    if correlator.is_empty() {
272        return Ok(());
273    }
274    let allowed: &[&str] = match telescope {
275        "ALMASD" => &["ACASpec", "ACA"],
276        "ALMA" => &["BLC", "BLC_4x4bit"],
277        "ACA" => &["BLC_4x4bit", "BLC", "ACA"],
278        _ => return Ok(()),
279    };
280    if allowed.contains(&correlator) {
281        return Ok(());
282    }
283    Err(SimobserveError::raised(format!(
284        "correlator = {correlator} is not allowed for telescope = {telescope}. Allowed values: {}",
285        allowed.join(" ")
286    )))
287}

293    #[test]
294    fn the_stem_names_the_configuration() {
295        let stem = |list: &str| stem("p", list).expect("a stem");
296        assert_eq!(stem("alma.out10.cfg"), "p.alma.out10");
297        assert_eq!(stem("alma;0.5arcsec"), "p.alma_0.5arcsec");
298        assert_eq!(stem("/data/vla.a.cfg"), "p.vla.a");
299        assert_eq!(stem("configs/"), "p.");
300        assert!(super::stem("p", "").is_err());
301    }
302
303    #[test]
304    fn cycle_tails_match_as_python_matches_them() {
305        let tail = |text: &str, cycle| cycle_tail(text, cycle);
306        assert_eq!(
307            tail("ALMA.CYCLE1;0.5ARCSEC", '1').as_deref(),
308            Some("0.5ARCSEC")
309        );
310        assert_eq!(
311            tail("ALMA.CYCLE.1.;1ARCSEC", '1').as_deref(),
312            Some("1ARCSEC")
313        );
314        assert_eq!(tail("ALMA_CYCLE2;0.1ARCSEC", '1'), None);
315        assert_eq!(tail("CYCLE1;A;B", '1').as_deref(), Some("A;B"));
316        // The greedy `.*` takes the last CYCLE that matches.
317        assert_eq!(tail("CYCLE1;X CYCLE1;Y", '1').as_deref(), Some("Y"));
318        assert_eq!(tail("ALMA;0.5ARCSEC", '1'), None);
319        assert_eq!(tail("AB", '1'), None);
320    }

330    #[test]
331    fn numpy_means_sum_pairwise() {
332        // Below 8 values, NumPy sums in order.
333        assert_eq!(numpy_mean(&[1.0, 2.0, 4.0]), 7.0 / 3.0);
334        // From 8 on, eight running sums: summed in order, 1e16 swallows the
335        // seven ones before -1e16 cancels it (7); the running sums keep
336        // them (14).
337        let mut values = vec![1.0; 16];
338        values[0] = 1e16;
339        values[8] = -1e16;
340        assert_eq!(pairwise_sum(&values), 14.0);
341        let large: Vec<f64> = (0..300).map(f64::from).collect();
342        assert_eq!(pairwise_sum(&large), 44_850.0);
343    }
