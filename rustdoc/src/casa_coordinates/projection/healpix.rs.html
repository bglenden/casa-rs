3use std::f64::consts::{FRAC_PI_2, PI, TAU};
4
5use crate::CoordinateError;
6
7/// The equatorial band and polar facets in Calabretta & Roukema (2007).
8pub(super) struct Healpix {
9    facets: f64,
10    layers: f64,
11    half_width: f64,
12}
13
14impl Healpix {
15    pub(super) fn new(parameters: &[f64]) -> Result<Self, CoordinateError> {
16        let facets = parameters.first().copied().unwrap_or(4.0);
17        let layers = parameters.get(1).copied().unwrap_or(3.0);
18        if !(facets.is_finite() && layers.is_finite() && facets > 0.0 && layers > 0.0) {
19            return Err(CoordinateError::ConversionFailed(
20                "HPX needs positive, finite longitude and latitude facet counts".into(),
21            ));
22        }
23        Ok(Self {
24            facets,
25            layers,
26            half_width: PI / facets,
27        })
28    }
29
30    fn facet_offset(&self, longitude: f64, southern: bool) -> f64 {
31        let facet = ((longitude + PI) * self.facets / TAU).floor();
32        let center = -PI + (2.0 * facet + 1.0) * self.half_width;
33        let offset = longitude - center;
34        if southern && self.layers.round() % 2.0 == 0.0 {
35            let index = (longitude / self.half_width).floor() + self.facets.round() % 2.0;
36            if index % 2.0 == 0.0 {
37                offset + self.half_width
38            } else {
39                offset - self.half_width
40            }
41        } else {
42            offset
43        }
44    }
45
46    pub(super) fn project(&self, phi: f64, theta: f64) -> Result<(f64, f64), CoordinateError> {
47        let sin_theta = theta.sin();
48        let z = sin_theta.abs();
49        if z <= (self.layers - 1.0) / self.layers {
50            return Ok((phi, self.half_width * self.layers * sin_theta / 2.0));
51        }
52        let sigma = (self.layers * (1.0 - z)).sqrt();
53        let offset = self.facet_offset(phi, theta < 0.0);
54        let x = phi + offset * (sigma - 1.0);
55        let y = self.half_width * ((self.layers + 1.0) / 2.0 - sigma);
56        Ok((if x > PI { TAU - x } else { x }, y.copysign(theta)))
57    }
58
59    pub(super) fn deproject(&self, x: f64, y: f64) -> Result<(f64, f64), CoordinateError> {
60        const TOLERANCE: f64 = 2.0e-14;
61        let invalid =
62            || CoordinateError::ConversionFailed("Pixel is outside the HPX projection".into());
63        let outer = self.half_width * (self.layers + 1.0) / 2.0;
64        if !x.is_finite() || !y.is_finite() || y.abs() > outer {
65            return Err(invalid());
66        }
67        let (phi, theta) = if y.abs() <= self.half_width * (self.layers - 1.0) / 2.0 {
68            (x, (2.0 * y / (self.half_width * self.layers)).asin())
69        } else {
70            let sigma = (self.layers + 1.0) / 2.0 - y.abs() / self.half_width;
71            let offset = self.facet_offset(x, y < 0.0);
72            let difference = if sigma == 0.0 {
73                1.0e9 * offset
74            } else {
75                offset / sigma
76            };
77            let sin_theta = 1.0 - sigma * sigma / self.layers;
78            if difference.abs() > self.half_width + TOLERANCE || sin_theta < -1.0 {
79                return Err(invalid());
80            }
81            (x + difference - offset, sin_theta.asin().copysign(y))
82        };
83        if phi.abs() > PI + TOLERANCE || theta.abs() > FRAC_PI_2 + TOLERANCE {
84            return Err(invalid());
85        }
86        Ok((phi.clamp(-PI, PI), theta))
87    }
88}
