9pub(super) fn read_ssm_string(

22pub(super) fn read_fixed_ssm_string(bytes: &[u8]) -> Result<String, StorageError> {
23    let len = bytes
24        .iter()
25        .position(|&byte| byte == 0)
26        .unwrap_or(bytes.len());
27    String::from_utf8(bytes[..len].to_vec()).map_err(|e| {
28        StorageError::FormatMismatch(format!("SSM fixed string is not valid UTF-8: {e}"))
29    })
30}

32/// Read `length` bytes from the SSM string bucket chain starting at
33/// `offset` in the data area of bucket `bucket_nr`: a string, or a string
34/// array's elements. Data longer than a bucket's rest continues at the
35/// start of the next bucket its header links (casacore's
36/// `SSMStringHandler::getData`).
37pub(super) fn read_ssm_string_bytes(
38    file: &mut impl DataRead,
39    header: &SsmHeader,
40    buckets: &mut LastBucket,
41    bucket_nr: i32,
42    offset: i32,
43    length: i32,
44) -> Result<Vec<u8>, StorageError> {
45    if length == 0 {
46        return Ok(Vec::new());
47    }
48    if bucket_nr < 0 || offset < 0 || length < 0 {
49        return Err(StorageError::FormatMismatch(format!(
50            "invalid SSM string locator bucket={} offset={} length={}",
51            bucket_nr, offset, length
52        )));
53    }
54
55    let mut result = Vec::with_capacity(length as usize);
56    let mut remaining = length as usize;
57    let mut cur_bucket = bucket_nr;
58    let mut cur_offset = offset as usize;
59
60    while remaining > 0 {
61        let bucket = buckets.read(file, header, cur_bucket as u32)?;
62        let data_start = STRING_BUCKET_HEADER + cur_offset;
63        let Some(available) = bucket.len().checked_sub(data_start) else {
64            return Err(StorageError::FormatMismatch(format!(
65                "SSM string offset {cur_offset} is beyond bucket {cur_bucket}"
66            )));
67        };
68        let chunk = remaining.min(available);
69        result.extend_from_slice(&bucket[data_start..data_start + chunk]);
70        remaining -= chunk;
71
72        if remaining > 0 {
73            // String bucket metadata (nextBucket) is always canonical (BE)
74            // per C++ CanonicalConversion in SSMStringHandler.
75            let next_bucket = read_i32_be(&bucket[12..16]);
76            if next_bucket < 0 {
77                return Err(StorageError::FormatMismatch(
78                    "SSM string chain ended prematurely".to_string(),
79                ));
80            }
81            cur_bucket = next_bucket;
82            cur_offset = 0;
83        }
84    }
85
86    Ok(result)
87}

89pub(super) fn read_ssm_string_array_value(
90    file: &mut impl DataRead,
91    header: &SsmHeader,
92    buckets: &mut LastBucket,
93    bucket_nr: i32,
94    offset: i32,
95    length: i32,
96    fixed_shape: Option<&[i32]>,
97) -> Result<Option<Value>, StorageError> {
98    if length == 0 {
99        return Ok(None);
100    }
101    let block = read_ssm_string_bytes(file, header, buckets, bucket_nr, offset, length)?;
102    string_arrays::decode(&block, fixed_shape).map(Some)
