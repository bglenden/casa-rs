14/// The old CASA iterator selects a continuous channel range. It replaces
15/// strides with one and flags holes between the selected segments.
16pub(crate) fn channel_selection(
17    selection: Option<&ChannelSelection>,
18    count: usize,
19) -> Result<(Vec<usize>, Vec<bool>), GainSolveError> {
20    let Some(selection) = selection else {
21        return Ok(((0..count).collect(), Vec::new()));
22    };
23    let mut selection = selection.clone();
24    for segment in &mut selection.segments {
25        segment.stride = 1;
26    }
27    let mut selected = super::selected_channels(Some(&selection), count)?;
28    selected.sort_unstable();
29    let mut excluded = vec![true; count];
30    for &channel in &selected {
31        excluded[channel] = false;
32    }
33    let envelope = selected
34        .first()
35        .zip(selected.last())
36        .map_or_else(Vec::new, |(&first, &last)| (first..=last).collect());
37    Ok((envelope, excluded))
38}

40#[derive(Debug, Clone, Default)]
41pub(crate) struct PolynomialAccumulation {
42    spectra: BTreeMap<(i32, i32, [i32; 2], usize), BaselineSpectrum>,
43    descriptions: BTreeSet<i32>,
44    time: WeightedTime,
45}
46
47#[derive(Debug, Clone)]
48pub(crate) struct BaselineSpectrum {
49    pub(crate) weighted_visibilities: Vec<Complex32>,
50    pub(crate) weight: f32,
51}
52
53impl PolynomialAccumulation {
54    pub(crate) fn descriptions(&self) -> &BTreeSet<i32> {
55        &self.descriptions
56    }
57
58    pub(crate) fn spectra(
59        &self,
60    ) -> impl Iterator<Item = (i32, [i32; 2], usize, &BaselineSpectrum)> {
61        self.spectra
62            .iter()
63            .map(|(&(description, _, antennas, hand), spectrum)| {
64                (description, antennas, hand, spectrum)
65            })
66    }
67
68    pub(crate) fn time(&self) -> Option<f64> {
69        self.time.mean()
70    }
71
72    pub(super) fn observe(
73        &mut self,
74        row: &SolveSampleRow,
75        samples: &SolveSamples<'_>,
76        correlations: &[i32],
77        channels: SelectedChannels<'_>,
78        point_stokes: [f32; 4],
79    ) {
80        let width = channels.bin_width.unwrap_or(1);
81        let channel_flags = samples::channel_flags(samples);
82        let bin_count = channels.indices.len().div_ceil(width);
83        let excluded = |channel: usize| channels.excluded.get(channel).copied().unwrap_or(false);
84        let weight_at = |correlation, channel| {
85            channels
86                .bin_weights
87                .map_or(1.0, |weights| weights[[correlation, channel]])
88        };
89        let usable = |correlation, channel| {
90            !excluded(channel)
91                && !samples.flags[[correlation, channel]]
92                && weight_at(correlation, channel) > 0.0
93        };
94        // VisBuffer::channelAve averages each correlation separately.
95        // Its output channel is shared-flagged if one hand has no sample.
96        let averaged_flags = (width > 1).then(|| {
97            channels
98                .indices
99                .chunks(width)
100                .map(|bin| {
101                    (0..correlations.len())
102                        .any(|correlation| bin.iter().all(|&channel| !usable(correlation, channel)))
103                })
104                .collect::<Vec<_>>()
105        });
106        let mut antennas = [row.antenna1, row.antenna2];
107        let reverse_baseline = antennas[0] > antennas[1];
108        antennas.sort_unstable();
109        let mut time_weight = 0.0;
110        let last = correlations.len() - 1;
111        for (hand, correlation) in [0, last]
112            .into_iter()
113            .take(correlations.len().min(2))
114            .enumerate()
115        {
116            let point = samples.point_model.map_or_else(
117                || {
118                    samples::point_visibility(
119                        point_stokes,
120                        correlations[correlation],
121                        correlations[0],
122                    )
123                },
124                |model| model[correlation],
125            );
126            let model = |channel| {
127                samples
128                    .model
129                    .map_or(point, |model| model[[correlation, channel]])
130            };
131            let flag = |channel: usize| channel_flags[channel] || excluded(channel);
132            let mut amplitude_sum = 0.0;
133            let mut count = 0;
134            for &channel in channels.indices {
135                if !flag(channel) {
136                    let amplitude = model(channel).norm();
137                    if amplitude > 0.0 {
138                        amplitude_sum += amplitude;
139                        count += 1;
140                    }
141                }
142            }
143            if count == 0
144                || averaged_flags
145                    .as_ref()
146                    .is_some_and(|flags| flags.iter().all(|flag| *flag))
147            {
148                continue;
149            }
150
151            // VisBuffer::normalize scales the scalar weight by the squared
152            // mean model amplitude. VisBuffAccumulator then assigns that
153            // same row weight to all channels, including flagged ones in the
154            // denominator of a time average.
155            let weight = samples.weights[[correlation, 0]]
156                * (amplitude_sum / count as f32).powi(2)
157                * width as f32;
158            if weight <= 0.0 {
159                continue;
160            }
161            let spectrum = self
162                .spectra
163                .entry((row.data_desc_id, row.field_id, antennas, hand))
164                .or_insert_with(|| BaselineSpectrum {
165                    weighted_visibilities: vec![Complex32::default(); bin_count],
166                    weight: 0.0,
167                });
168            spectrum.weight += weight;
169            for (index, (sum, bin)) in spectrum
170                .weighted_visibilities
171                .iter_mut()
172                .zip(channels.indices.chunks(width))
173                .enumerate()
174            {
175                if averaged_flags.as_ref().is_some_and(|flags| flags[index]) {
176                    continue;
177                }
178                let mut visibility = Complex32::default();
179                let mut weight_sum = 0.0_f64;
180                for &channel in bin {
181                    if if width == 1 {
182                        !flag(channel)
183                    } else {
184                        usable(correlation, channel)
185                    } {
186                        let data = samples.data[[correlation, channel]];
187                        let model = model(channel);
188                        let channel_weight = if width == 1 {
189                            1.0
190                        } else {
191                            weight_at(correlation, channel)
192                        };
193                        // Normalization precedes binning. It only touches a
194                        // channel when the input's shared flag is clear.
195                        if flag(channel) {
196                            visibility += data * channel_weight;
197                        } else if model.norm_sqr() > 0.0 {
198                            let divided = Complex64::new(data.re.into(), data.im.into())
199                                / Complex64::new(model.re.into(), model.im.into());
200                            visibility += Complex32::new(divided.re as f32, divided.im as f32)
201                                * channel_weight;
202                        }
203                        weight_sum += f64::from(channel_weight);
204                    }
205                }
206                if weight_sum > 0.0 {
207                    let visibility = if reverse_baseline {
208                        visibility.conj()
209                    } else {
210                        visibility
211                    };
212                    *sum += visibility * (weight / weight_sum as f32);
213                }
214            }
215            time_weight += f64::from(weight);
216        }
217        if time_weight > 0.0 {
218            self.descriptions.insert(row.data_desc_id);
219            self.time.observe(row.time_seconds, time_weight);
220        }
221    }
222}
