2//! `setoptions(ftmachine='sd')`: total power, as CASA's simulator predicts
3//! it (`SDGrid`, built by `Simulator::createSkyEquation`), by sampling the
4//! model in the image plane.
5//!
6//! For a row pointed at pixel `(x, y)` of the model, `SDGrid::get` calls
7//! `dgridsd` (`fgridsd.f`), the model summed through the kernel around the
8//! pointing's nearest pixel; the visibility is the sum's conjugate.
9//! `casa-imaging-singledish` owns the kernels and that sum, which the
10//! single-dish gridder shares ([`KernelTable::degrid`]):
11//! - `BOX`: 1 out to a pixel, reaching the nearest pixel only;
12//! - `SF`: CASA's spheroid `(1 − ν²) grdsf(ν)` over 3 pixels;
13//! - `PB`: the telescope's power pattern along the x axis, at the frequency
14//!   of model plane 0 for every channel ([`KernelTable::primary_beam`]).
15//!
16//! The model is a Stokes I image: the parallel hands take its value and the
17//! cross hands none. A channel takes the model plane `matchChannel` gives
18//! it, and none off the model's spectral axis. Every row of a scan is
19//! pointed at the scan's field, so a field's samples serve all its rows.
20//!
21//! Only autocorrelations are predicted, and flagged rows are not. CASA
22//! skips a row with a positive uvw component, which leaves a cross
23//! correlation with no positive component predicted as if it were one dish;
24//! casa-rs predicts autocorrelations only (`docs/CASA (C++) bugs.md`).
25

33/// The `SDGrid` kernel `setoptions(gridfunction=…)` names (`SDGrid::init`).
34#[derive(Debug, Clone, Copy, PartialEq, Eq)]
35pub enum SingleDishKernel {
36    /// `PB`: the telescope's primary beam.
37    Pb,
38    /// `BOX`: the nearest pixel.
39    Box,
40    /// `SF`: CASA's prolate spheroid over 3 pixels.
41    Spheroidal,
42}
43
44impl SingleDishKernel {
45    /// The kernel `gridfunction` names, in any case; `None` for the others
46    /// (`GAUSS` and `GJINC`, which CASA also has, among them).
47    #[must_use]
48    pub fn from_grid_function(grid_function: &str) -> Option<Self> {
49        match grid_function.to_lowercase().as_str() {
50            "pb" => Some(Self::Pb),
51            "box" => Some(Self::Box),
52            "sf" => Some(Self::Spheroidal),
53            _ => None,
54        }
55    }
56
57    /// The kernel with the simulator's support, CASA's default.
58    const fn grid_function(self) -> GridFunction {
59        match self {
60            Self::Pb => GridFunction::PrimaryBeam,
61            Self::Box => GridFunction::BOX,
62            Self::Spheroidal => GridFunction::SPHEROIDAL,
63        }
64    }
65
66    /// The kernel's table on `model`'s pixels; `beam` is the `PB` kernel's
67    /// pattern.
68    fn table(self, model: &SkyModel, beam: Option<&BeamPattern>) -> KernelTable {
69        match (self.grid_function(), beam) {
70            (GridFunction::PrimaryBeam, Some(beam)) => {
71                KernelTable::primary_beam(beam, model.increment_rad[0], model.plane_frequency_hz(0))
72            }
73            (GridFunction::PrimaryBeam, None) => unreachable!("a PB kernel has a beam"),
74            (function, _) => KernelTable::new(function).expect("a kernel of fixed support"),
75        }
76    }
77}

79/// A window the model is predicted into: its channels and correlations.
80#[derive(Debug, Clone)]
81pub(crate) struct SingleDishWindow {
82    /// Channel frequencies, Hz.
83    pub(crate) frequencies_hz: Vec<f64>,
84    /// casacore `Stokes::StokesTypes` codes, one per correlation.
85    pub(crate) correlations: Vec<i32>,
86}

88/// Where a total-power prediction samples the model: each distinct pointing
89/// direction of the scans it covers, and each scan's pointing. A scan
90/// points at its phase centre, which `observemany` can move off its field
91/// (the POINTING table casa-rs writes follows it).
92#[derive(Debug)]
93pub(crate) struct SingleDishPointings {
94    /// Each distinct pointing's J2000 direction, rad.
95    pub(crate) directions_rad: Vec<[f64; 2]>,
96    /// `[scan]`: the scan's pointing, an index into `directions_rad`.
97    pub(crate) of_scan: Vec<usize>,
98}
99
100impl SingleDishPointings {
101    /// The pointings of scans pointed at `phase_centres_rad`, in scan order.
102    pub(crate) fn of_scans(phase_centres_rad: impl ExactSizeIterator<Item = [f64; 2]>) -> Self {
103        let mut directions_rad: Vec<[f64; 2]> = Vec::new();
104        let of_scan = phase_centres_rad
105            .map(|centre| {
106                let same = |known: &[f64; 2]| known.map(f64::to_bits) == centre.map(f64::to_bits);
107                directions_rad.iter().position(same).unwrap_or_else(|| {
108                    directions_rad.push(centre);
109                    directions_rad.len() - 1
110                })
111            })
112            .collect();
113        Self {
114            directions_rad,
115            of_scan,
116        }
117    }
118}

120/// A model prepared for total-power prediction: each pointing's samples in
121/// each window.
122#[derive(Debug)]
123pub(crate) struct SingleDishPrediction {
124    /// `[pointing][window]`: `[channel][correlation]`.
125    samples: Vec<Vec<Vec<Complex32>>>,
126    /// `[scan]`: the scan's pointing.
127    of_scan: Vec<usize>,
128    /// The bytes [`Self::bytes`] counts.
129    _reservation: Reservation,
130}
131
132impl SingleDishPrediction {
133    /// Bytes a prediction of `windows` at `pointings` holds, known before it
134    /// is formed: the samples and the scans' pointings.
135    pub(crate) fn bytes(windows: &[SingleDishWindow], pointings: &SingleDishPointings) -> u64 {
136        let per_pointing: usize = size_of::<Vec<Vec<Complex32>>>()
137            + windows
138                .iter()
139                .map(|window| {
140                    size_of::<Vec<Complex32>>()
141                        + window.frequencies_hz.len()
142                            * window.correlations.len()
143                            * size_of::<Complex32>()
144                })
145                .sum::<usize>();
146        (pointings.directions_rad.len() * per_pointing
147            + pointings.of_scan.len() * size_of::<usize>()) as u64
148    }
149
150    /// Bytes preparing the prediction holds beside it:
151    /// - the kernel, as tabulated and as kept (`beam` is the `PB` kernel's
152    ///   pattern, `None` for the others);
153    /// - a model plane read, with the beam's power plane and their product
154    ///   when `setvp(dovp=True)` applies it first, and the image's tile
155    ///   cache ([`SkyModel::cache_bytes`]);
156    /// - each window channel's plane, and each pointing's direction, pixel
157    ///   and sample of the plane read.
158    pub(crate) fn preparation_bytes(
159        model: &SkyModel,
160        beam: Option<&BeamPattern>,
161        (windows, pointings): (&[SingleDishWindow], &SingleDishPointings),
162    ) -> u64 {
163        // The PB kernel's, or the larger of BOX's and SF's.
164        let kernel = match beam {
165            Some(beam) => GridFunction::PrimaryBeam.forming_bytes(
166                beam.support_pixels(model.increment_rad[0], model.plane_frequency_hz(0)),
167            ),
168            None => GridFunction::SPHEROIDAL.forming_bytes(0),
169        };
170        let channels: usize = windows.iter().map(|w| w.frequencies_hz.len()).sum();
171        let planes_of =
172            windows.len() * size_of::<Vec<Option<usize>>>() + channels * size_of::<Option<usize>>();
173        let per_pointing = 2 * size_of::<[f64; 2]>() + size_of::<f32>();
174        kernel
175            + (planes_of + pointings.directions_rad.len() * per_pointing) as u64
176            + 2 * model.plane_bytes()
177            + model.cache_bytes()
178    }
179
180    /// Prepare `model` with `kernel` for `windows` at `pointings`, holding
181    /// `reservation`, its admission of [`Self::bytes`]. `beams` are the
182    /// telescope's pattern, which the `PB` kernel needs, and the pattern
183    /// `setvp(dovp=True)` applies to the model first, centred on each
184    /// pointing, as CASA's `SkyEquation` does whatever the kernel (so a
185    /// `PB` kernel then weights the model by the power pattern twice).
186    ///
187    /// # Errors
188    ///
189    /// The model's pixels could not be read, or a pointing could not be
190    /// placed on them.
191    pub(crate) fn new(
192        model: &SkyModel,
193        kernel: SingleDishKernel,
194        (beam, voltage_pattern): (Option<&BeamPattern>, Option<&BeamPattern>),
195        (windows, pointings): (&[SingleDishWindow], SingleDishPointings),
196        reservation: Reservation,
197    ) -> Result<Self, PredictError> {
198        let SingleDishPointings {
199            directions_rad,
200            of_scan,
201        } = pointings;
202        let kernel = kernel.table(model, beam);
203        let axis = SpectralAxis::new(
204            model.first_frequency_hz,
205            model.frequency_increment_hz,
206            u32::try_from(model.planes())
207                .map_err(|_| PredictError::UnsupportedModel("too many model planes".into()))?,
208        )?;
209        // Each window channel's model plane (`matchChannel`), or none.
210        let planes_of: Vec<Vec<Option<usize>>> = windows
211            .iter()
212            .map(|window| {
213                window
214                    .frequencies_hz
215                    .iter()
216                    .map(|&frequency| axis.nearest_channel(frequency).map(|plane| plane as usize))
217                    .collect()
218            })
219            .collect();
220        let pixels: Vec<[f64; 2]> = directions_rad
221            .iter()
222            .map(|direction| model.pixel_of(*direction))
223            .collect::<Result<_, _>>()?;
224        // The samples, zero until a plane fills them. Sized exactly: they
225        // are kept, and charged by their length.
226        let mut samples: Vec<Vec<Vec<Complex32>>> = directions_rad
227            .iter()
228            .map(|_| {
229                windows
230                    .iter()
231                    .map(|window| {
232                        vec![
233                            Complex32::new(0.0, 0.0);
234                            window.frequencies_hz.len() * window.correlations.len()
235                        ]
236                    })
237                    .collect()
238            })
239            .collect();
240        // One plane read at a time, each pointing's sample of it written
241        // into the channels it feeds, so no plane × pointing matrix is held
242        // (#874 review).
243        let mut by_pointing = vec![0.0_f32; directions_rad.len()];
244        for plane in 0..model.planes() {
245            if !planes_of
246                .iter()
247                .flatten()
248                .any(|&mapped| mapped == Some(plane))
249            {
250                continue;
251            }
252            let values = model.read_plane(plane)?;
253            for (value, pixel) in by_pointing.iter_mut().zip(&pixels) {
254                *value = match voltage_pattern {
255                    Some(pattern) => {
256                        let apparent =
257                            &values * &super::beam::power_plane(pattern, model, plane, *pixel);
258                        kernel.degrid(apparent.view(), *pixel)
259                    }
260                    None => kernel.degrid(values.view(), *pixel),
261                };
262            }
263            for (pointing_windows, &value) in samples.iter_mut().zip(&by_pointing) {
264                for ((cells, window), planes) in
265                    pointing_windows.iter_mut().zip(windows).zip(&planes_of)
266                {
267                    let correlations = window.correlations.len();
268                    for (channel, _) in planes
269                        .iter()
270                        .enumerate()
271                        .filter(|(_, mapped)| **mapped == Some(plane))
272                    {
273                        let channel_cells = &mut cells[channel * correlations..][..correlations];
274                        for (cell, &code) in channel_cells.iter_mut().zip(&window.correlations) {
275                            // The parallel hands carry Stokes I.
276                            if matches!(code, 1 | 5 | 8 | 9 | 12) {
277                                *cell = Complex32::new(value, 0.0);
278                            }
279                        }
280                    }
281                }
282            }
283        }
284        tracing::debug!(
285            support = kernel.support(),
286            bytes = kernel.bytes(),
287            "single-dish kernel"
288        );
289        Ok(Self {
290            samples,
291            of_scan,
292            _reservation: reservation,
293        })
294    }
295
296    /// The samples of scan `scan`'s pointing in window `window`,
297    /// `[channel][correlation]`; empty for a scan not prepared.
298    pub(crate) fn samples(&self, scan: usize, window: usize) -> &[Complex32] {
299        self.of_scan
300            .get(scan)
301            .map_or(&[], |&pointing| self.samples[pointing][window].as_slice())
302    }
303}
