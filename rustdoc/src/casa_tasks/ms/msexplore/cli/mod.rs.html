202/// Use the existing PlotMS front end with caller-provided calibration and model sources.
203pub fn task_front_end_with_plot_sources(
204    payload: Option<PathBuf>,
205    load_calibration: Option<PlotMsCalibrationLoader>,
206    load_models: Option<PlotMsModelLoader>,
207) -> TaskFrontEnd<impl Fn(&TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure>> {
208    TaskFrontEnd::new_with_deferred_schema(
209        "msexplore",
210        msexplore_task_schema_bundle(),
211        &["msexplore", "plotms"],
212        move |task| run_task(task, payload.as_deref(), load_calibration, load_models),
213    )
214    .expect("the msexplore contract carries msexplore and plotms")
215}

217/// Run one resolved `msexplore` or `plotms` request. `msexplore` returns its
218/// summary, as CASA's `listobs` returns its record; `plotms` returns None.
219pub(crate) fn run_task(
220    task: &TaskInvocation<'_>,
221    payload: Option<&Path>,
222    load_calibration: Option<PlotMsCalibrationLoader>,
223    load_models: Option<PlotMsModelLoader>,
224) -> Result<TaskSuccess, TaskFailure> {
225    if task.surface() == "plotms"
226        && let Some(path) = super::calibration::input(task)?
227    {
228        if super::plotms_phasecenter::requested(task)?.is_some() {
229            return Err(TaskFailure::validation(
230                "PlotMS phasecenter for calibration tables is pending (#1032)",
231            )
232            .at("phasecenter"));
233        }
234        super::plotms_pointing::refuse_nonpointing_controls(task)?;
235        if super::plotms_figure::requested_position(task)? != [0, 0] {
236            return Err(TaskFailure::validation(
237                "nondefault grid placement for PlotMS calibration tables is pending #1032",
238            ));
239        }
240        return super::calibration::run(task, &path, payload);
241    }
242    let options = if task.surface() == "plotms" {
243        let library = task.texts("callib")?;
244        let calibration = if library.is_empty() {
245            None
246        } else {
247            let load = load_calibration.ok_or_else(|| {
248                TaskFailure::validation("use casars.tasks.plotms for on-the-fly calibration")
249                    .at("callib")
250            })?;
251            Some((load, library))
252        };
253        let atmosphere = super::calibration::atmosphere::Options::from_task(task)?;
254        let visibility_document = calibration.is_some() || atmosphere.is_some();
255        let (mut options, visibility_document) =
256            plotms_options::options_from_plotms(task, visibility_document)?;
257        if options.avgbaseline && options.iteraxis == Some(MsIterationAxis::Antenna) {
258            // CASA disables antenna iteration when all baselines are combined.
259            options.iteraxis = None;
260        }
261        if options.iteraxis.is_some_and(MsIterationAxis::groups_by_row)
262            && (calibration.is_some() || atmosphere.is_some())
263        {
264            return Err(TaskFailure::validation(
265                "PlotMS antenna, baseline or time iteration with a callibrary or overlays is pending (#1032)",
266            )
267            .at("iteraxis"));
268        }
269        let phasecenter = super::plotms_phasecenter::requested(task)?
270            .map(|value| {
271                options.plotms_phasecenter(&value, calibration.is_none() && atmosphere.is_none())
272            })
273            .transpose()?;
274        let iterated_averaging = options.supports_averaged_ordinate_iteration();
275        let calibrated_ordinates = calibration.is_some()
276            && atmosphere.is_none()
277            && options.supports_calibrated_ordinates();
278        if options.iteraxis.is_some()
279            && options.y_axis2.is_some()
280            && ((options.plotms_averaging_active() && !iterated_averaging)
281                || options.has_mixed_ordinate_dimensions()
282                || (calibration.is_some() && !calibrated_ordinates)
283                || atmosphere.is_some())
284        {
285            return Err(TaskFailure::validation(
286                "PlotMS iterated ordinate lists with averaging, mixed axes, a callibrary or overlays are pending #1032",
287            ));
288        }
289        if !options.additional_ordinates.is_empty() {
290            if calibration.is_some()
291                || (options.plotms_averaging_active()
292                    && !options.supports_visibility_ordinate_averaging(atmosphere.is_some()))
293            {
294                return Err(TaskFailure::validation(
295                    "more than two MeasurementSet ordinates with this averaging or a callibrary are pending #1032",
296                ));
297            }
298            if options.has_mixed_ordinate_dimensions() && !options.uses_channel_ordinate_masks() {
299                return Err(TaskFailure::validation(
300                    "mixed MeasurementSet visibility and metadata ordinates or geometry abscissae are pending #1032",
301                ));
302            }
303        }
304        if calibration.is_some()
305            && options.secondary_y_data_column.is_some()
306            && !calibrated_ordinates
307        {
308            return Err(TaskFailure::validation(
309                "independent ordinate columns with a MeasurementSet callibrary are pending #1032",
310            ));
311        }
312        if calibration.is_some()
313            && options
314                .plotms_axis_columns()
315                .is_some_and(|columns| columns.contains(&MsDataColumn::Float))
316        {
317            return Err(TaskFailure::validation(
318                "MeasurementSet FLOAT_DATA with a callibrary is pending #1032",
319            ));
320        }
321        options.validate_plotms_columns(
322            task,
323            calibration.is_some(),
324            calibrated_ordinates,
325            atmosphere.is_some(),
326            iterated_averaging,
327        )?;
328        if visibility_document {
329            return run_visibility_plot(
330                task,
331                options,
332                payload,
333                calibration,
334                atmosphere,
335                load_models,
336                phasecenter,
337            );
338        }
339        super::plotms_pointing::refuse_nonpointing_controls(task)?;
340        if super::plotms_figure::requested_position(task)? != [0, 0] {
341            return Err(TaskFailure::validation(
342                "grid placement for this PlotMS axis pair is pending #1032",
343            ));
344        }
345        options
346    } else {
347        options_from(task)?
348    };
349    let outcome = run(options)?;
350    let success = if task.surface() == "plotms" {
351        TaskSuccess::new(CasaValue::None, &outcome.result)?
352    } else {
353        let summary = serde_json::to_value(&outcome.result.summary)
354            .map_err(|error| TaskFailure::execution(format!("serialize the summary: {error}")))?;
355        TaskSuccess::new(CasaValue::from_plain_json(&summary), &outcome.result)?
356    };
357    let success = success.with_text(outcome.text);
358    Ok(match outcome.note {
359        Some(note) => success.with_note(note),
360        None => success,
361    })
362}

364fn run_visibility_plot(
365    task: &TaskInvocation<'_>,
366    mut options: CliOptions,
367    array_payload: Option<&Path>,
368    calibration: Option<(PlotMsCalibrationLoader, Vec<String>)>,
369    atmosphere: Option<super::calibration::atmosphere::Options>,
370    load_models: Option<PlotMsModelLoader>,
371    phasecenter: Option<casa_core::measures::direction::MDirection>,
372) -> Result<TaskSuccess, TaskFailure> {
373    let is_text = super::plotms_text::requests_text(task, options.plot_output.as_deref())?;
374    let offset_grid = calibration.is_none() && atmosphere.is_none();
375    let static_grid =
376        super::plotms_figure::static_grid(task, options.iteraxis.is_some(), offset_grid)?;
377    if static_grid.is_some() {
378        // The scatter computation receives a single plot. The grid places it on the page.
379        options.gridrows = 1;
380        options.gridcols = 1;
381    }
382    let pointing =
383        super::plotms_pointing::parse_pointing_axes(task, [options.x_axis, options.y_axis])?;
384    if let Some(axes) = &pointing {
385        options.prepare_pointing_axes(task, axes, calibration.is_some() || atmosphere.is_some())?;
386    }
387    super::task_contract::refuse_unwritable_outputs(
388        &[("plotfile", options.plot_output.as_deref())],
389        options.overwrite,
390        "set overwrite=True to replace it",
391    )
392    .map_err(TaskFailure::validation)?;
393    let ms = MeasurementSet::open(&options.ms_path).map_err(|error| {
394        TaskFailure::execution(format!(
395            "open MeasurementSet {}: {error}",
396            options.ms_path.display()
397        ))
398    })?;
399    let mut axis_columns = options.plotms_axis_columns();
400    let extended_derived = calibration.is_none()
401        && axis_columns.is_some_and(|columns| options.supports_extended_derived(columns));
402    let implicit_derived_model = atmosphere.is_none()
403        && (!options.avgfield
404            || (options.avgtime.is_some() && !options.avgbaseline && !options.avgantenna));
405    let raw_symbols = options.plotms_symbol_visibility(
406        task,
407        axis_columns,
408        calibration.is_some(),
409        pointing.is_some(),
410    );
411    let prefer_float_data = if calibration.is_none()
412        && let Some(columns) = &mut axis_columns
413    {
414        options.resolve_visibility_storage(
415            &ms,
416            columns,
417            implicit_derived_model,
418            phasecenter.is_some(),
419        )?
420    } else {
421        false
422    };
423    let averaging = options.plotms_averaging_active();
424    for (axis, column) in &mut options.additional_ordinates {
425        let ([resolved], _) = plotms_columns::resolve_storage_columns(
426            &ms,
427            [*column],
428            [Some(*axis)],
429            averaging,
430            false,
431        )?;
432        *column = resolved;
433    }
434    if !options.selection.selectdata {
435        options.selection = MsSelection::default();
436    }
437    let library = calibration
438        .as_ref()
439        .map_or(&[][..], |(_, entries)| entries.as_slice());
440    let calibrated = calibration
441        .as_ref()
442        .map(|(load, _)| {
443            let channel_weighting = options.secondary_y_data_column.is_some()
444                && options.avgchannel.is_some_and(|width| width > 1);
445            let mut input = load(
446                &ms,
447                &options.selection,
448                library,
449                channel_weighting,
450                task.workspace(),
451            )?;
452            input.x_data_column = options.x_data_column;
453            Ok::<_, TaskFailure>(input)
454        })
455        .transpose()?;
456    let show = task.flag("showgui")?;
457    let output = options.plot_output.as_deref();
458    let text_file = output.filter(|_| is_text);
459    let figure_file = output.filter(|_| !is_text);
460    let mut plots = Vec::new();
461    if output.is_some() || show {
462        let spec = build_explore_spec(&options, pointing.is_some(), true)
463            .map_err(TaskFailure::validation)?;
464        let plot = &spec.plots[0];
465        let clock_labels = (show || figure_file.is_some())
466            && (plot.x_axis == MsAxis::Time || plot.y_axes.contains(&MsAxis::Time));
467        let (payload, clock_reference) = match &calibrated {
468            Some(input) => {
469                build_calibrated_plotms_payload(&ms, &spec, input, options.secondary_y_data_column)
470                    .map(|payload| (payload, None))
471            }
472            None => build_uncalibrated_plotms_payload(
473                &ms,
474                &spec,
475                PlotMsColumnSelection {
476                    x: options.x_data_column,
477                    secondary_y: options.secondary_y_data_column,
478                    additional_y: &options.additional_ordinates,
479                    weights: options.weight_axis_columns,
480                    raw_symbols,
481                    pointing: pointing.as_ref(),
482                    phasecenter,
483                    extended_derived,
484                    implicit_derived_model,
485                },
486                prefer_float_data,
487                load_models,
488                clock_labels,
489            ),
490        }
491        .map_err(TaskFailure::execution)?;
492        let overlays = if let Some(atmosphere) = atmosphere {
493            super::plotms_atmosphere::build(
494                &ms,
495                &options.selection,
496                &payload,
497                atmosphere,
498                &spec.plots[0],
499            )?
500        } else {
501            Vec::new()
502        };
503        if show || figure_file.is_some() {
504            let arrays = array_payload.ok_or_else(|| {
505                TaskFailure::validation("use casars.tasks.plotms for rendered figures")
506            })?;
507            plots = super::plotms_figure::prepare(
508                task,
509                &payload,
510                arrays,
511                figure_file,
512                show,
513                &overlays,
514                super::plotms_figure::AxisPresentation {
515                    columns: options.figure_axis_columns(
516                        axis_columns,
517                        calibrated.is_some(),
518                        pointing.as_ref(),
519                    ),
520                    spec: plot,
521                    clock_reference,
522                    frequency_frame: super::plotms_figure::reporting_frame(&ms, plot, &payload),
523                },
524            )?;
525            if let Some(layout) = static_grid {
526                plots[0].document.layout = Some(layout);
527            }
528        }
529        if let Some(path) = text_file {
530            super::plotms_text::write_export(
531                task,
532                &ms,
533                &payload,
534                path,
535                library,
536                &overlays,
537                super::plotms_text::ExportAxes {
538                    ordinates: &spec.plots[0].y_axes,
539                    channel_major: calibrated.is_none()
540                        || options.secondary_y_data_column.is_some(),
541                },
542            )?;
543        }
544    }
545    TaskSuccess::new(
546        CasaValue::None,
547        &serde_json::json!({"text_file": text_file, "plots": plots}),
548    )
549}

551/// Run one command line. Everything a refusal can find is checked before the
552/// first write: the outputs, the flag arguments and the plot (#931). The flag
553/// edit is then made in memory, so a refusal that needs the MeasurementSet
554/// also comes first, and is saved after the summary is written.
555fn run(options: CliOptions) -> Result<RunOutcome, TaskFailure> {
556    let refused = TaskFailure::validation;
557    let failed = TaskFailure::execution;
558    super::task_contract::refuse_unwritable_outputs(
559        &[
560            ("summary output", options.summary_output.as_deref()),
561            ("flag output", options.flag_output.as_deref()),
562            ("plot output", options.plot_output.as_deref()),
563        ],
564        options.overwrite,
565        "pass --overwrite to replace it",
566    )
567    .map_err(refused)?;
568    let flag_edit = build_flag_edit_spec(&options).map_err(refused)?;
569    let explore_spec = if options.plot_output.is_some() || flag_edit.is_some() {
570        Some(build_explore_spec(&options, false, false).map_err(refused)?)
571    } else {
572        None
573    };
574    let mut ms = MeasurementSet::open(&options.ms_path).map_err(|error| {
575        failed(if options.ms_path.is_dir() {
576            format!(
577                "msexplore currently supports MeasurementSets only; failed to open {} as an MS: {error}",
578                options.ms_path.display()
579            )
580        } else {
581            format!("open MeasurementSet {}: {error}", options.ms_path.display())
582        })
583    })?;
584
585    let summary =
586        MeasurementSetSummary::from_ms_with_options(&ms, &options.selection.to_summary_options())
587            .map_err(|error| failed(error.to_string()))?;
588    let summary_text = summary
589        .render(options.summary_format)
590        .map_err(|error| failed(error.to_string()))?;
591    // Refused here, before anything is written: the edit is in memory.
592    let flag_preview =
593        edit_flags_in_memory(&mut ms, &options, flag_edit.as_ref(), explore_spec.as_ref())
594            .map_err(refused)?;
595    let mut text = String::new();
596    match options.summary_output.as_deref() {
597        Some(path) => write_output(path, options.overwrite, &summary_text).map_err(failed)?,
598        None => text = summary_text,
599    }
600    let mut note = None;
601    if let Some(preview) = &flag_preview {
602        if options.flag_apply {
603            ms.save_main_table_only()
604                .map_err(|error| failed(error.to_string()))?;
605        }
606        note = write_or_describe_flag_preview(&options, preview).map_err(failed)?;
607    }
608
609    let plot_export = match &options.plot_output {
610        Some(plot_output) => {
611            let explore_spec = explore_spec.as_ref().ok_or_else(|| {
612                failed("msexplore plot export lost its prepared explore spec".to_string())
613            })?;
614            let payload = build_msexplore_payload(&ms, explore_spec).map_err(failed)?;
615            match options.plot_format {
616                MsExportFormat::Txt => {
617                    export_msexplore_plot(&payload, plot_output).map_err(failed)?;
618                }
619            }
620            Some(MsExplorePlotArtifact {
621                output_path: plot_output.clone(),
622                format: options.plot_format,
623            })
624        }
625        None => None,
626    };
627
628    Ok(RunOutcome {
629        result: MsExploreRunTaskResult {
630            summary,
631            summary_output_path: options.summary_output,
632            flag_edit_preview: flag_preview,
633            flag_edit_output_path: options.flag_output,
634            plot_export,
635        },
636        text,
637        note,
638    })
639}

641/// What one run produced: the provider's result, the summary for stdout
642/// when no `output` file was named, and the flag-edit line for stderr when
643/// no `flag_output` file was.
644struct RunOutcome {

646    text: String,
647    note: Option<String>,

815/// Make the requested flag edit in memory, with `--flag-apply`, and return
816/// its preview; the caller saves it.
817fn edit_flags_in_memory(
818    ms: &mut MeasurementSet,
819    options: &CliOptions,
820    flag_edit: Option<&MsFlagEditSpec>,
821    explore_spec: Option<&MsExploreSpec>,
822) -> Result<Option<MsFlagEditPreview>, String> {
823    let Some(action) = options.flag_action else {
824        return Ok(None);
825    };
826    if options.flag_selected {
827        if options.flag_apply && options.flag_output.is_none() {
828            return apply_selection_flag_edit_direct(ms, &options.selection, action).map(Some);
829        }
830        let preview = preview_selection_flag_edit(ms, &options.selection, action)?;
831        if options.flag_apply {
832            apply_selection_flag_edit_preview(ms, &preview)?;
833        }
834        return Ok(Some(preview));
835    }
836    let (Some(explore_spec), Some(flag_edit)) = (explore_spec, flag_edit) else {
837        return Err("msexplore lost its prepared flag edit".to_string());
838    };
839    if options.flag_apply {
840        apply_msexplore_flag_edit_for_request(ms, explore_spec, flag_edit).map(Some)
841    } else {
842        preview_msexplore_flag_edit_for_request(ms, explore_spec, flag_edit).map(Some)
843    }
844}

863fn preview_selection_flag_edit(
864    ms: &MeasurementSet,
865    selection: &MsSelection,
866    action: MsFlagAction,
867) -> Result<MsFlagEditPreview, String> {
868    let listobs_options = selection.to_summary_options();
869    let row_numbers = resolve_selected_rows_with_msselect(ms, selection, &listobs_options)?;
870    let mut sample_edits = Vec::new();
871    let mut row_edits = Vec::new();
872
873    for row in row_numbers {
874        let old_flag_row = ms
875            .flag_row_column()
876            .get(row)
877            .map_err(|error| error.to_string())?;
878        let old_matrix = clone_flag_matrix(ms, row)?;
879        let mut new_matrix = old_matrix.clone();
880        let (corr_count, chan_count) = old_matrix.dim();
881        let mut changed_samples = 0usize;
882        for corr in 0..corr_count {
883            for chan in 0..chan_count {
884                let old_flag = old_matrix[(corr, chan)];
885                let new_flag = match action {
886                    MsFlagAction::Flag => true,
887                    MsFlagAction::Unflag => false,
888                };
889                if old_flag != new_flag {
890                    new_matrix[(corr, chan)] = new_flag;
891                    sample_edits.push(MsFlagSampleEdit {
892                        row,
893                        corr,
894                        chan,
895                        old_flag,
896                        new_flag,
897                    });
898                    changed_samples += 1;
899                }
900            }
901        }
902        let new_flag_row = new_matrix.iter().all(|value| *value);
903        if changed_samples > 0 || old_flag_row != new_flag_row {
904            row_edits.push(MsFlagRowEdit {
905                row,
906                old_flag_row,
907                new_flag_row,
908                changed_samples,
909            });
910        }
911    }
912
913    Ok(MsFlagEditPreview {
914        plot_title: "selection flag edit".to_string(),
915        plot_index: None,
916        panel_key: None,
917        panel_label: None,
918        x_axis: MsAxis::Time,
919        y_axis: MsAxis::Flag,
920        region: MsFlagRegion {
921            x_min: 0.0,
922            x_max: 0.0,
923            y_min: 0.0,
924            y_max: 0.0,
925        },
926        action,
927        extcorr: true,
928        extchannel: true,
929        matched_points: sample_edits.len(),
930        affected_rows: row_edits.len(),
931        affected_samples: sample_edits.len(),
932        sample_edits,
933        row_edits,
934    })
935}

937fn apply_selection_flag_edit_preview(
938    ms: &mut MeasurementSet,
939    preview: &MsFlagEditPreview,
940) -> Result<(), String> {
941    let mut row_updates = BTreeMap::<usize, (ndarray::Array2<bool>, bool)>::new();
942    for row_edit in &preview.row_edits {
943        row_updates.insert(
944            row_edit.row,
945            (clone_flag_matrix(ms, row_edit.row)?, row_edit.new_flag_row),
946        );
947    }
948    for sample in &preview.sample_edits {
949        let (matrix, _) = row_updates
950            .get_mut(&sample.row)
951            .ok_or_else(|| format!("flag edit lost planned row {}", sample.row))?;
952        matrix[(sample.corr, sample.chan)] = sample.new_flag;
953    }
954    for (row, (matrix, flag_row)) in row_updates {
955        ms.main_table_mut()
956            .cell_accessor_mut(row, "FLAG")
957            .map_err(|error| error.to_string())?
958            .set(Value::Array(ArrayValue::Bool(matrix.into_dyn())))
959            .map_err(|error| error.to_string())?;
960        ms.main_table_mut()
961            .cell_accessor_mut(row, "FLAG_ROW")
962            .map_err(|error| error.to_string())?
963            .set(Value::Scalar(ScalarValue::Bool(flag_row)))
964            .map_err(|error| error.to_string())?;
965    }
966    Ok(())
967}

969fn apply_selection_flag_edit_direct(
970    ms: &mut MeasurementSet,
971    selection: &MsSelection,
972    action: MsFlagAction,
973) -> Result<MsFlagEditPreview, String> {
974    let listobs_options = selection.to_summary_options();
975    let row_numbers = resolve_selected_rows_with_msselect(ms, selection, &listobs_options)?;
976    let mut affected_rows = 0usize;
977    let mut affected_samples = 0usize;
978
979    for row in row_numbers {
980        let mut matrix = clone_flag_matrix(ms, row)?;
981        let mut changed_samples = 0usize;
982        let new_flag = match action {
983            MsFlagAction::Flag => true,
984            MsFlagAction::Unflag => false,
985        };
986        for value in &mut matrix {
987            if *value != new_flag {
988                *value = new_flag;
989                changed_samples += 1;
990            }
991        }
992        if changed_samples == 0 {
993            continue;
994        }
995        let new_flag_row = matrix.iter().all(|value| *value);
996        ms.main_table_mut()
997            .cell_accessor_mut(row, "FLAG")
998            .map_err(|error| error.to_string())?
999            .set(Value::Array(ArrayValue::Bool(matrix.into_dyn())))
1000            .map_err(|error| error.to_string())?;
1001        ms.main_table_mut()
1002            .cell_accessor_mut(row, "FLAG_ROW")
1003            .map_err(|error| error.to_string())?
1004            .set(Value::Scalar(ScalarValue::Bool(new_flag_row)))
1005            .map_err(|error| error.to_string())?;
1006        affected_rows += 1;
1007        affected_samples += changed_samples;
1008    }
1009
1010    Ok(MsFlagEditPreview {
1011        plot_title: "selection flag edit".to_string(),
1012        plot_index: None,
1013        panel_key: None,
1014        panel_label: None,
1015        x_axis: MsAxis::Time,
1016        y_axis: MsAxis::Flag,
1017        region: MsFlagRegion {
1018            x_min: 0.0,
1019            x_max: 0.0,
1020            y_min: 0.0,
1021            y_max: 0.0,
1022        },
1023        action,
1024        extcorr: true,
1025        extchannel: true,
1026        matched_points: affected_samples,
1027        affected_rows,
1028        affected_samples,
1029        sample_edits: Vec::new(),
1030        row_edits: Vec::new(),
1031    })
1032}

1154fn parse_header_items(value: Option<&str>) -> Result<Vec<MsPageHeaderItem>, String> {
1155    let Some(value) = value else {
1156        return Ok(Vec::new());
1157    };
1158    let mut items = Vec::new();
1159    for raw in value.split(',') {
1160        let token = raw.trim();
1161        if token.is_empty() {
1162            continue;
1163        }
1164        let item = MsPageHeaderItem::parse(token).ok_or_else(|| format!(
1165            "unsupported --headeritems value {token:?}; expected a comma-separated subset of: filename, ycolumn, obsdate, obstime, observer, projid, telescope, targname, targdir"
1166        ))?;
1167        if !items.contains(&item) {
1168            items.push(item);
1169        }
1170    }
1171    Ok(items)
1172}
1173
1174fn merge_header_items(target: &mut Vec<MsPageHeaderItem>, extra: Vec<MsPageHeaderItem>) {
1175    for item in extra {
1176        if !target.contains(&item) {
1177            target.push(item);
1178        }
1179    }
1180}

1283        xlabel: text("xlabel")?,
1284        ylabel: text("ylabel")?,
1285        symbol_size: parsed(task, "symbol_size", number::<u32>)?,
1286        showlegend: flag("showlegend")?,
1287        legendposition: required(task, "legendposition", legend_position)?,
1288        showmajorgrid: flag("showmajorgrid")?,
1289        showminorgrid: flag("showminorgrid")?,
1290        headeritems: text("headeritems")?,
1291        max_points: count(task, "max_points")?,
1292        flag_action: parsed(task, "flag_action", flag_action)?,
1293        flag_xmin: parsed(task, "flag_xmin", number::<f64>)?,
1294        flag_xmax: parsed(task, "flag_xmax", number::<f64>)?,
1295        flag_ymin: parsed(task, "flag_ymin", number::<f64>)?,
1296        flag_ymax: parsed(task, "flag_ymax", number::<f64>)?,
1297        flag_plotindex: parsed(task, "flag_plotindex", number::<usize>)?,
1298        flag_panel: text("flag_panel")?,
1299        flag_extcorr: flag("flag_extcorr")?,
1300        flag_extchannel: flag("flag_extchannel")?,
1301        flag_selected: flag("flag_selected")?,
1302        flag_apply: flag("flag_apply")?,
1303        flag_output: path("flag_output")?,
1304        plot_output: path("plot_output")?,
1305        plot_format: required(task, "plot_format", MsExportFormat::parse)?,
1306    };
1307    let plot_control_used = PLOT_CONTROLS.iter().any(|name| task.is_set(name));
1308    check_options(&options, plot_control_used).map_err(TaskFailure::validation)?;
1309    Ok(options)
1310}

1312/// Refuse the combinations a plot or flag edit cannot take.
1313fn check_options(options: &CliOptions, plot_control_used: bool) -> Result<(), String> {
1314    let single_plot =
1315        options.preset.is_some() || (options.x_axis.is_some() && options.y_axis.is_some());
1316    if options.page_spec.is_some() && plot_control_used {
1317        return Err(
1318            "--page-spec cannot be combined with the single-plot preset/axis/layout flags; put those settings in the JSON page spec instead".to_string(),
1319        );
1320    }
1321    if options.plot_output.is_some() && options.page_spec.is_none() && !single_plot {
1322        return Err(
1323            "--plot-output requires either --preset or both --xaxis and --yaxis".to_string(),
1324        );
1325    }
1326    if options.flag_action.is_some()
1327        && !options.flag_selected
1328        && options.page_spec.is_none()
1329        && !single_plot
1330    {
1331        return Err(
1332            "--flag-action requires either --preset or both --xaxis and --yaxis".to_string(),
1333        );
1334    }
1335    if options.flag_plotindex.is_some() && options.flag_panel.is_some() {
1336        return Err(
1337            "msexplore staged flag editing accepts either --flag-plotindex or --flag-panel, not both".to_string(),
1338        );
1339    }
1340    let region_controls = options.flag_xmin.is_some()
1341        || options.flag_xmax.is_some()
1342        || options.flag_ymin.is_some()
1343        || options.flag_ymax.is_some()
1344        || options.flag_plotindex.is_some()
1345        || options.flag_panel.is_some()
1346        || options.flag_extcorr
1347        || options.flag_extchannel;
1348    if options.flag_selected && region_controls {
1349        return Err(
1350            "msexplore --flag-selected cannot be combined with plot-region flag-edit controls"
1351                .to_string(),
1352        );
1353    }
1354    if options.flag_action.is_none()
1355        && (region_controls
1356            || options.flag_selected
1357            || options.flag_apply
1358            || options.flag_output.is_some())
1359    {
1360        return Err(
1361            "msexplore flag-edit region, extension, apply, and output flags require --flag-action"
1362                .to_string(),
1363        );
1364    }
1365    Ok(())
1366}

1368/// `name` parsed by `parse`, or `None` when unset; a refusal names it. A
1369/// number or `none` resolves as a number when it is one.
1370fn parsed<T>(
1371    task: &TaskInvocation<'_>,
1372    name: &str,
1373    parse: impl Fn(&str) -> Result<T, String>,
1374) -> Result<Option<T>, TaskFailure> {
1375    let text = match task.value(name) {
1376        Some(ParameterValue::Integer(value)) => Some(value.to_string()),
1377        Some(ParameterValue::Float(value)) => Some(value.to_string()),
1378        _ => task.text(name)?,
1379    };
1380    text.map(|text| parse(&text).map_err(|message| TaskFailure::validation(message).at(name)))
1381        .transpose()
1382}
1383
1384/// `name` parsed by `parse`; it must be set.
1385fn required<T>(
1386    task: &TaskInvocation<'_>,
1387    name: &str,
1388    parse: impl Fn(&str) -> Result<T, String>,
1389) -> Result<T, TaskFailure> {
1390    parsed(task, name, parse)?
1391        .ok_or_else(|| TaskFailure::validation(format!("parameter {name:?} is required")).at(name))
1392}
1393
1394fn number<T: std::str::FromStr>(text: &str) -> Result<T, String> {
1395    text.trim()
1396        .parse()
1397        .map_err(|_| format!("invalid numeric value {text:?}"))
1398}

1400/// `name`, a count, which cannot be negative.
1401fn count(task: &TaskInvocation<'_>, name: &str) -> Result<usize, TaskFailure> {
1402    let value = task.integer(name)?;
1403    usize::try_from(value).map_err(|_| {
1404        TaskFailure::validation(format!("{name} cannot be negative, found {value}")).at(name)
1405    })
1406}

1408fn legend_position(text: &str) -> Result<MsLegendPosition, String> {
1409    serde_json::from_value(serde_json::Value::String(text.to_string())).map_err(|_| {
1410        "unsupported value for --legendposition; expected one of: upperRight, upperLeft, lowerRight, lowerLeft, exteriorRight, exteriorLeft, exteriorTop, exteriorBottom".to_string()
1411    })
1412}
1413
1414fn flag_action(text: &str) -> Result<MsFlagAction, String> {
1415    match text.to_ascii_lowercase().as_str() {
1416        "flag" => Ok(MsFlagAction::Flag),
1417        "unflag" => Ok(MsFlagAction::Unflag),
1418        other => Err(format!(
1419            "unsupported value for --flag-action {other:?}; expected flag or unflag"
1420        )),
1421    }
1422}

1424fn write_output(path: &std::path::Path, overwrite: bool, text: &str) -> Result<(), String> {
1425    if path.exists() && !overwrite {
1426        return Err(format!(
1427            "refusing to overwrite existing output {}; pass --overwrite to replace it",
1428            path.display()
1429        ));
1430    }
1431    fs::write(path, text).map_err(|error| error.to_string())
1432}
