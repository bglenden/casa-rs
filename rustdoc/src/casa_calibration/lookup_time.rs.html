20/// Amplitude/phase interpolation in time for calibration tables.
21#[derive(
22    Debug, Clone, Copy, PartialEq, Eq, serde::Serialize, serde::Deserialize, schemars::JsonSchema,
23)]
24pub enum TimeInterpolation {
25    /// Nearest time, clamped to the endpoint outside the table's time range.
26    Nearest,
27    /// Linear amplitude and shortest phase between adjacent solutions.
28    Linear,
29    /// Cubic polynomial through four neighboring time samples.
30    Cubic,
31    /// Natural cubic spline through all samples, with continuous phase.
32    Spline,
33}

35impl CalibrationTableLayout {
36    /// Additional retained heap for the selected time interpolator.
37    /// Spline coefficients are built once, then shared by all lookups.
38    #[must_use]
39    pub fn time_kept_bytes(self, mode: TimeInterpolation) -> u64 {
40        if mode == TimeInterpolation::Spline {
41            (self.rows as u64)
42                .saturating_mul(128)
43                .saturating_add((self.values as u64).saturating_mul(24))
44        } else {
45            0
46        }
47    }
48
49    /// Temporary heap while forming time coefficients, beside retained heap.
50    /// One parameter's curve is solved at a time; its length cannot exceed
51    /// the physical row count. Cubic queries use a four-element stack stencil.
52    #[must_use]
53    pub fn time_loading_bytes(self, mode: TimeInterpolation) -> u64 {
54        if mode == TimeInterpolation::Spline {
55            (self.rows as u64).saturating_mul(48)
56        } else {
57            0
58        }
59    }
60}

62impl CalibrationTableLookup {
63    /// Load a simulator table, treating all fields as one time series per
64    /// spectral window and antenna. Cubic follows CASA's four-time local
65    /// stencils, including flagged samples; only the adjacent time flags are
66    /// combined. Spline corrects CASA's uninitialized-interpolator crash and
67    /// incomplete natural-spline back-substitution (#1053).
68    ///
69    /// Callers admit the table layout's loading/kept bytes and its additional
70    /// time loading/kept bytes before calling, retaining the kept charges.
71    ///
72    /// # Errors
73    /// The parameter table cannot be read. Spline time axes must be
74    /// strictly increasing after CASA's relative Float time conversion, and
75    /// their parameter shapes must match. Interior cubic queries require at
76    /// least four times; exact and endpoint queries do not.
77    pub fn open_across_fields(
78        path: impl AsRef<Path>,
79        mode: TimeInterpolation,
80    ) -> Result<Self, ApplyExecutionError> {
81        Self::open_for_simulation(path, TimeInterpolationOptions::new(mode))
82    }
83}

106fn grid<'a>(
107    path: &Path,
108    solution: &'a CalibrationSolution,
109) -> Result<&'a GainGrid, ApplyExecutionError> {
110    match solution.grid.as_ref() {
111        CalibrationGrid::Complex(grid) => Ok(grid),
112        _ => Err(invalid(
113            path,
114            "time interpolation requires complex parameters",
115        )),
116    }

119fn value(grid: &GainGrid, index: usize) -> (Complex32, bool) {
120    let receptor = index / grid.channel_count;
121    let channel = index % grid.channel_count;
122    if grid.values.ndim() == 1 {
123        (grid.values[[receptor]], grid.flags[[receptor]])
124    } else {
125        (
126            grid.values[[receptor, channel]],
127            grid.flags[[receptor, channel]],
128        )
129    }
130}

132fn relative(time: f64, reference: f64) -> f64 {
133    f64::from((time - reference) as f32)
134}

136struct Registration {
137    reference: f64,
138    time: f64,
139    upper: usize,
140    exact: bool,
141}
142
143impl Registration {
144    fn new(rows: &[CalibrationSolution], time: f64) -> Self {
145        let reference = (rows[0].time_seconds - 1.).floor();
146        let time = relative(time, reference);
147        let upper = rows.partition_point(|row| relative(row.time_seconds, reference) < time);
148        let exact = upper == 0
149            || upper == rows.len()
150            || relative(rows[upper].time_seconds, reference) == time;
151        Self {
152            reference,
153            time,
154            upper,
155            exact,
156        }
157    }
158    fn nearest(&self, rows: &[CalibrationSolution]) -> usize {
159        if self.exact {
160            self.upper.min(rows.len() - 1)
161        } else {
162            // CTTimeInterp1 subtracts the registered Float epochs before
163            // comparing the distances. A rounded tie selects the earlier row.
164            let later =
165                (relative(rows[self.upper].time_seconds, self.reference) - self.time) as f32;
166            let earlier =
167                (self.time - relative(rows[self.upper - 1].time_seconds, self.reference)) as f32;
168            if later < earlier {
169                self.upper
170            } else {
171                self.upper - 1
172            }
173        }
174    }
175}

177impl TimePlan {
178    /// Whether every response in this window retains its registered parameters.
179    pub(crate) fn same_window_solution(&self, first: TimeQuery, second: TimeQuery) -> bool {
180        let scope = self.scope.coordinates(first.observation, first.scan);
181        if first.window != second.window
182            || scope != self.scope.coordinates(second.observation, second.scan)
183        {
184            return false;
185        }
186        self.series.iter().all(|(&(obs, scan, window, _), rows)| {
187            if (obs, scan) != scope || window != first.window || rows.is_empty() {
188                return true;
189            }
190            let start = Registration::new(rows, first.time);
191            let end = Registration::new(rows, second.time);
192            if self.mode == TimeInterpolation::Nearest || (start.exact && end.exact) {
193                start.nearest(rows) == end.nearest(rows)
194            } else {
195                start.time == end.time
196            }
197        })
198    }
199
200    fn candidates(&self, query: &TimeQuery) -> Option<SelectedSeries<'_>> {
201        let (observation, scan) = self.scope.coordinates(query.observation, query.scan);
202        self.series
203            .get_key_value(&(observation, scan, query.window, query.antenna))
204            .or_else(|| {
205                self.series
206                    .get_key_value(&(observation, scan, query.window, -1))
207            })
208            .filter(|(_, rows)| !rows.is_empty())
209            .map(|(key, rows)| (key, rows.as_slice()))
210    }
211
212    /// Exact, nearest and endpoint queries can share an immutable payload.
213    /// Real families can share it when the caller can use the raw parameters.
214    pub(crate) fn registered_grid(
215        &self,
216        query: &TimeQuery,
217        allow_raw_real: bool,
218    ) -> Option<std::sync::Arc<CalibrationGrid>> {
219        if (self.real && !allow_raw_real) || query.frequency_hz.is_some() {
220            return None;
221        }
222        let (_, rows) = self.candidates(query)?;
223        let registration = Registration::new(rows, query.time);
224        (registration.exact || self.mode == TimeInterpolation::Nearest)
225            .then(|| std::sync::Arc::clone(&rows[registration.nearest(rows)].grid))
226    }
227
228    pub(crate) fn solution(
229        &self,
230        path: &Path,
231        query: TimeQuery,
232    ) -> Result<Option<ComplexSolution>, ApplyExecutionError> {
233        let Some((key, candidates)) = self.candidates(&query) else {
234            return Ok(None);
235        };
236        let spw = query.window;
237        let frequency_hz = query.frequency_hz;
238        let registration = Registration::new(candidates, query.time);
239        let Registration {
240            reference,
241            time,
242            upper,
243            exact,
244        } = registration;
245        if exact || self.mode == TimeInterpolation::Nearest {
246            let index = registration.nearest(candidates);
247            let mut output = ComplexSolution::of(grid(path, &candidates[index])?);
248            if let Some(frequency) = frequency_hz.filter(|_| self.cycles.contains_key(key)) {
249                let scale = self.phase_scale(path, spw, frequency)?;
250                for (parameter, value) in output.values.iter_mut().enumerate() {
251                    let phase = value.arg()
252                        + self.cycles[key][index * output.receptors * output.channels + parameter];
253                    *value = crate::lookup::complex_from_casa_polar(value.norm(), phase * scale);
254                }
255            }
256            return Ok(Some(output));
257        }
258        let lower = upper - 1;
259        let first = if self.mode == TimeInterpolation::Cubic {
260            if candidates.len() < 4 {
261                return Err(invalid(
262                    path,
263                    "cubic time interpolation requires four samples for an interior output",
264                ));
265            }
266            lower.saturating_sub(1).min(candidates.len() - 4)
267        } else {
268            lower
269        };
270        let stencil = &candidates[first
271            ..first
272                + if self.mode == TimeInterpolation::Cubic {
273                    4
274                } else {
275                    2
276                }];
277        let base = grid(path, &candidates[lower])?;
278        for row in stencil {
279            if grid(path, row)?.values.shape() != base.values.shape() {
280                return Err(invalid(
281                    path,
282                    "time interpolation requires matching parameter shapes",
283                ));
284            }
285        }
286        if !stencil.windows(2).all(|pair| {
287            relative(pair[0].time_seconds, reference) < relative(pair[1].time_seconds, reference)
288        }) {
289            return Err(invalid(path, "time samples must be strictly increasing"));
290        }
291        let high = grid(path, &candidates[upper])?;
292        let (receptors, channels) = (base.receptor_count, base.channel_count);
293        let spline = self.splines.get(key);
294        let phase_scale = frequency_hz
295            .filter(|_| self.cycles.contains_key(key))
296            .map(|frequency| self.phase_scale(path, spw, frequency))
297            .transpose()?;
298        let mut values = Vec::with_capacity(receptors * channels);
299        let mut flags = Vec::with_capacity(receptors * channels);
300        for index in 0..receptors * channels {
301            let mut points = [(0., 0., 0.); 4];
302            for (i, row) in stencil.iter().enumerate() {
303                let z = value(grid(path, row)?, index).0;
304                let phase = spline.map_or_else(
305                    || if self.real { 0. } else { z.arg() },
306                    |s| s.phases[(first + i) * receptors * channels + index],
307                );
308                points[i] = (
309                    relative(row.time_seconds, reference),
310                    if self.real { z.re } else { z.norm() },
311                    phase,
312                );
313                if i > 0 && spline.is_none() {
314                    points[i].2 = unwrap_phase(points[i - 1].2, points[i].2);
315                }
316            }
317            let result = if self.mode == TimeInterpolation::Cubic {
318                sample_polar(&points, time, Curve::Cubic, false, &[])
319            } else if self.mode == TimeInterpolation::Linear {
320                // CTTimeInterp1 uses Interpolate1D<Float, Array<Float>>.
321                // Round the fraction and the array arithmetic to Float.
322                let [
323                    (low_time, low_amp, low_phase),
324                    (high_time, high_amp, high_phase),
325                    ..,
326                ] = points;
327                let low_time = low_time as f32;
328                let fraction = (time as f32 - low_time) / (high_time as f32 - low_time);
329                let blend = |low, high| low + fraction * (high - low);
330                Some((blend(low_amp, high_amp), blend(low_phase, high_phase)))
331            } else {
332                let spline = spline.expect("spline coefficients");
333                let left = lower * receptors * channels + index;
334                let right = left + receptors * channels;
335                sample_polar(
336                    &points[..2],
337                    time,
338                    Curve::Spline,
339                    false,
340                    &[spline.second[left], spline.second[right]],
341                )
342            };
343            let (amplitude, mut phase) = result.expect("nonempty time stencil");
344            if let Some(scale) = phase_scale {
345                // A spline already retains the globally unwrapped phase. Local
346                // linear/cubic stencils need CASA's lower-bracket cycle offset.
347                if spline.is_none() {
348                    phase += self.cycles[key][lower * receptors * channels + index];
349                }
350                phase *= scale;
351            }
352            values.push(crate::lookup::complex_from_casa_polar(amplitude, phase));
353            flags.push(value(base, index).1 || value(high, index).1);
354        }
355        Ok(Some(ComplexSolution {
356            receptors,
357            channels,
358            values,
359            flags,
360        }))
361    }

364struct Spline {
365    phases: Vec<f32>,
366    second: Vec<[f64; 2]>,
367}
368
369impl Spline {
370    fn new(
371        path: &Path,
372        series: &[CalibrationSolution],
373        real: bool,
374    ) -> Result<Self, ApplyExecutionError> {
375        let base = grid(path, &series[0])?;
376        let count = base.receptor_count * base.channel_count;
377        let size = series
378            .len()
379            .checked_mul(count)
380            .ok_or_else(|| invalid(path, "time coefficient count overflows usize"))?;
381        let reference = (series[0].time_seconds - 1.).floor();
382        for pair in series.windows(2) {
383            if relative(pair[0].time_seconds, reference)
384                >= relative(pair[1].time_seconds, reference)
385            {
386                return Err(invalid(
387                    path,
388                    "spline time samples must be strictly increasing",
389                ));
390            }
391        }
392        for row in series {
393            if grid(path, row)?.values.shape() != base.values.shape() {
394                return Err(invalid(
395                    path,
396                    "time interpolation requires matching parameter shapes",
397                ));
398            }
399        }
400        let mut result = Self {
401            phases: vec![0.; size],
402            second: vec![[0.; 2]; size],
403        };
404        let mut points = Vec::with_capacity(series.len());
405        for index in 0..count {
406            points.clear();
407            let mut phase = 0.;
408            for (i, row) in series.iter().enumerate() {
409                let z = value(grid(path, row)?, index).0;
410                phase = if real {
411                    0.
412                } else if i == 0 {
413                    z.arg()
414                } else {
415                    unwrap_phase(phase, z.arg())
416                };
417                points.push((
418                    relative(row.time_seconds, reference),
419                    if real { z.re } else { z.norm() },
420                    phase,
421                ));
422                result.phases[i * count + index] = phase;
423            }
424            for (i, second) in spline_derivatives(&points).into_iter().enumerate() {
425                result.second[i * count + index] = second;
426            }
427        }
428        Ok(result)
429    }
