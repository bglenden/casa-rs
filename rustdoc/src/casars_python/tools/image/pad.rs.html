21fn copy(
22    image: &dyn ImageInterface<f32>,
23    coordinates: RecordValue,
24    options: ImagePadOptions<'_>,
25    beams: &BeamView,
26    path: Option<&Path>,
27) -> PyResult<ImageHandle> {
28    let mut pad = ImagePad::new(image, coordinates, options, beams.record()).map_err(raised)?;
29    let info = {
30        let source = beams.beam_set().map_err(raised)?;
31        let mut info = pad.section.image_info().to_record();
32        if source.is_multi() {
33            let channels = pad.beam_planes(
34                image.find_axis(CoordinateType::Spectral),
35                source.n_channels(),
36            );
37            let polarizations =
38                pad.beam_planes(image.find_axis(CoordinateType::Stokes), source.n_stokes());
39            subset_beam_record(
40                &mut info,
41                Some(beams.record()),
42                &source,
43                &channels,
44                &polarizations,
45            )
46            .map_err(raised)?;
47        } else {
48            beams.apply_to(&mut info);
49        }
50        info
51    };
52    let record = pad
53        .section
54        .coordinate_record
55        .take()
56        .expect("padded coordinate record");
57    let mut output = Storage::create(
58        pad.section.shape.clone(),
59        pad.section.coordinates.clone(),
60        path,
61        None,
62    )
63    .map_err(raised)?;
64    with_storage!(&mut output, dest => pad.copy_to(image, dest.as_mut())).map_err(raised)?;
65    with_storage!(&mut output, dest => dest.set_image_info_record(info)).map_err(raised)?;
66    let mut output = ImageHandle::Float(output);
67    output.set_coordinate_record(record)?;
68    Ok(output)
69}

71#[pymethods]
72impl PyImageTool {
73    /// Select an image region and pad the two direction axes.
74    #[allow(clippy::too_many_arguments)]
75    #[pyo3(signature = (outfile="", npixels=1, value=0.0, padmask=false, overwrite=false, region=RegionArgument::default(), r#box="", chans="", stokes="", mask="", stretch=false, wantreturn=true), text_signature = "($self, outfile='', npixels=1, value=0.0, padmask=False, overwrite=False, region={}, box='', chans='', stokes='', mask='', stretch=False, wantreturn=True)")]
76    fn pad(
77        &self,
78        py: Python<'_>,
79        outfile: &str,
80        npixels: i64,
81        value: f64,
82        #[pyo3(from_py_with = require_bool)] padmask: bool,
83        #[pyo3(from_py_with = require_bool)] overwrite: bool,
84        region: RegionArgument,
85        r#box: &str,
86        chans: &str,
87        stokes: &str,
88        mask: &str,
89        #[pyo3(from_py_with = require_bool)] stretch: bool,
90        #[pyo3(from_py_with = require_bool)] wantreturn: bool,
91    ) -> PyResult<Py<Self>> {
92        let output = py.detach(|| {
93            let state = self.state();
94            let Some(attachment) = state.image.as_ref() else { return Ok(None) };
95            if !matches!(&*attachment.lock(), ImageHandle::Float(_)) {
96                return Err(raised("This method only supports Float valued images"));
97            }
98            let pixels = usize::try_from(npixels).ok().filter(|&count| count > 0)
99                .ok_or_else(|| raised("Value of npixels must be greater than zero"))?;
100            region.validate_selectors(r#box, chans, stokes)?;
101            if !outfile.is_empty() && !overwrite && Path::new(outfile).symlink_metadata().is_ok() {
102                return Err(raised(format!(
103                    "Replaceable flag is false and there is already a file or directory named {outfile} so the output image will not be written"
104                )));
105            }
106            let output = publish_image(outfile, overwrite, |path| {
107                let mut input = attachment.lock();
108                if !outfile.is_empty() {
109                    for source in input.path().into_iter().chain(attachment.operand_paths().iter().map(|path| path.as_path())) {
110                        ensure_outfile_does_not_alias_input(Path::new(outfile), source).map_err(raised)?;
111                    }
112                }
113                attachment.require_pixel_read_lock(&mut input)?;
114                let staged: PyResult<ImageHandle> = (|| {
115                    let ImageHandle::Float(source) = &*input else {
116                        unreachable!("checked float pixels")
117                    };
118                    let options = ImagePadOptions {
119                        pixels,
120                        value: value as f32,
121                        padmask,
122                        region: region.selection(),
123                        box_selection: (!r#box.is_empty()).then_some(r#box),
124                        channels: (!chans.is_empty()).then_some(chans),
125                        stokes: (!stokes.is_empty()).then_some(stokes),
126                        mask: (!mask.is_empty()).then_some(mask),
127                        stretch,
128                        destination: (!outfile.is_empty()).then(|| Path::new(outfile)),
129                    };
130                    let coordinates = input.coordinate_record();
131                    let beams = attachment.beam_view().clone();
132                    let history = match source {
133                        Storage::Paged(image) => image_history_rows(image.name().expect("native image path")).map_err(raised)?,
134                        Storage::Temp { image, .. } => image.history_rows().map_err(raised)?.into_owned(),
135                    };
136                    let mut output = match source {
137                        Storage::Temp { image, fits: Some(fits), .. } => {
138                            copy(&fits.image_view(image), coordinates, options, &beams, path)
139                        }
140                        _ => with_storage!(source, image => copy(image.as_ref(), coordinates, options, &beams, path)),
141                    }?;
142                    let ImageHandle::Float(dest) = &mut output else {
143                        unreachable!("padding produces float pixels")
144                    };
145                    with_storage!(dest, image => image.append_history_entries(&history)).map_err(raised)?;
146                    if state.history {
147                        let selected = region.history();
148                        let value = if value.is_nan() { "nan".into() } else { format!("{value:.6}") };
149                        output.record_history("pad", format!("ia.pad(outfile={outfile:?}, npixels={npixels}, value={value}, padmask={padmask}, overwrite={overwrite}, region={selected}, box={box:?}, chans={chans:?}, stokes={stokes:?}, mask={mask:?}, stretch={stretch}, wantreturn={wantreturn})", box = r#box))?;
150                    }
151                    let source_name = input.path().map_or_else(
152                        || "Temporary_Image".into(),
153                        |path| path.display().to_string(),
154                    );
155                    output.append_history_at(
156                        "::ImagePadder::_reportOldNewImageShapes",
157                        &[
158                            format!("Original {source_name} size => {:?}", input.shape()),
159                            format!("New {outfile} size => {:?}", output.shape()),
160                        ],
161                    )?;
162                    output.save().map_err(raised)?;
163                    Ok(output)
164                })();
165                let released = attachment.release_requested_lock(&mut input);
166                let result = staged?;
167                released?;
168                Ok(result)
169            })?;
170            Ok::<_, PyErr>(wantreturn.then_some(output))
171        })?;
172        Py::new(py, output.map_or_else(Self::new, Self::attached))
173    }
174}
