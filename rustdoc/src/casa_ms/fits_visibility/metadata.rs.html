62pub(super) struct Header {
63    pub(super) telescope: String,
64    pub(super) observer: String,
65    pub(super) object: String,
66    pub(super) epoch: f64,
67    pub(super) date: String,
68    pub(super) release_date: String,
69    pub(super) frame: Option<String>,
70    pub(super) rest_frequency: f64,
71    pub(super) alternate_value: Option<f64>,
72    pub(super) alternate_pixel: Option<f64>,
73    pub(super) velocity_reference: Option<i32>,
74    pub(super) history: Vec<String>,

77impl Header {
78    pub(super) fn read(fits: &mut FitsFile, binary_hdu: Option<usize>) -> MsResult<Self> {
79        let hdu = fits.hdu(binary_hdu.unwrap_or(0)).map_err(invalid)?;
80        let telescope = string_key(&hdu, fits, "TELESCOP", "unknown")?;
81        let observer = string_key(&hdu, fits, "OBSERVER", "")?;
82        let object = string_key(&hdu, fits, "OBJECT", "unknown")?;
83        let epoch = optional_key(&hdu, fits, "EPOCH")?
84            .or(optional_key(&hdu, fits, "EQUINOX")?)
85            .unwrap_or(2000.0);
86        let date = optional_key(&hdu, fits, "DATE-OBS")?
87            .or(optional_key(&hdu, fits, "DATE")?)
88            .unwrap_or_else(|| "2000-01-01".to_owned());
89        let release_date = if binary_hdu.is_some() {
90            optional_key(&hdu, fits, "DATE-MAP")?.unwrap_or_default()
91        } else {
92            date.clone()
93        };
94        let frame = optional_key(&hdu, fits, "SPECSYS")?;
95        let rest_frequency = optional_key(&hdu, fits, "RESTFREQ")?
96            .or(optional_key(&hdu, fits, "RESTFRQ")?)
97            .unwrap_or(0.0);
98        let alternate_value = optional_key(&hdu, fits, "ALTRVAL")?;
99        let alternate_pixel = optional_key(&hdu, fits, "ALTRPIX")?;
100        let velocity_reference = optional_key(&hdu, fits, "VELREF")?;
101        let history = history(fits)?;
102        Ok(Self {
103            telescope,
104            observer,
105            object,
106            epoch,
107            date,
108            release_date,
109            frame,
110            rest_frequency,
111            alternate_value,
112            alternate_pixel,
113            velocity_reference,
114            history,
115        })
116    }
117}

119pub(super) struct Extensions {
120    pub(super) array: Array,
121    pub(super) spectral: Option<Vec<SpectralWindow>>,
122    pub(super) sources: Option<BTreeMap<i32, Source>>,
123    pub(super) system_power: Option<Table>,
124    pub(super) velocity_type: Option<String>,
125}
126
127impl Extensions {
128    pub(super) fn read(
129        fits: &mut FitsFile,
130        header: &Header,
131        layout: &GroupLayout,
132    ) -> MsResult<Self> {
133        let mut array = None;
134        let mut spectral = None;
135        let mut sources = None;
136        let mut system_power = None;
137        let mut velocity_type = None;
138        let hdu_count = fits.num_hdus().map_err(invalid)?;
139        for index in 1..hdu_count {
140            let Some(name) = binary_extension_name(fits, index)? else {
141                continue;
142            };
143            let recognized = (array.is_none() && name.contains("AN"))
144                || (spectral.is_none() && name.contains("FQ"))
145                || (sources.is_none() && name.contains("SU"))
146                || (system_power.is_none() && name.contains("SY"));
147            if !recognized {
148                continue;
149            }
150            let hdu = binary_extension(fits, index)?;
151            let HduInfo::TableInfo { num_rows, .. } = &hdu.info else {
152                continue;
153            };
154            let table = BinaryTable {
155                fits,
156                hdu: &hdu,
157                rows: *num_rows,
158            };
159            if array.is_none() && name.contains("AN") {
160                array = Some(read_array(table, &header.telescope)?);
161            } else if spectral.is_none() && name.contains("FQ") {
162                spectral = Some(read_spectral(table, layout)?);
163            } else if sources.is_none() && name.contains("SU") {
164                velocity_type = optional_key(table.hdu, table.fits, "VELTYP")?;
165                sources = Some(read_sources(table)?);
166            } else if system_power.is_none() && name.contains("SY") {
167                system_power = Some(read_system_power(table)?);
168            }
169        }
170        let array = array.ok_or_else(|| invalid("Cannot find the required UVFITS AN table"))?;
171        fits.primary_hdu().map_err(invalid)?;
172        Ok(Self {
173            array,
174            spectral,
175            sources,
176            system_power,
177            velocity_type,
178        })
179    }
180}

182struct BinaryTable<'a> {
183    fits: &'a mut FitsFile,
184    hdu: &'a FitsHdu,
185    rows: usize,
186}
187
188impl BinaryTable<'_> {
189    fn strings(&mut self, name: &str) -> MsResult<Vec<String>> {
190        fixed_text_column(self.fits, self.hdu, name)
191    }
192
193    fn column<T: ReadsCol>(&mut self, name: &str, width: usize) -> MsResult<Vec<T>> {
194        let HduInfo::TableInfo {
195            column_descriptions,
196            ..
197        } = &self.hdu.info
198        else {
199            return Err(invalid("Expected a UVFITS binary table"));
200        };
201        let column = column_descriptions
202            .iter()
203            .find(|column| column.name == name)
204            .ok_or_else(|| invalid(format!("Cannot find UVFITS column {name}")))?;
205        let expected = self
206            .rows
207            .checked_mul(width)
208            .ok_or_else(|| invalid("FITS column size overflow"))?;
209        if column.data_type.repeat != width {
210            return Err(invalid(format!(
211                "UVFITS column {name} needs {width} values per row"
212            )));
213        }
214        let values = self.hdu.read_col(self.fits, name).map_err(invalid)?;
215        if values.len() != expected {
216            return Err(invalid(format!(
217                "UVFITS column {name} needs {width} values per row"
218            )));
219        }
220        Ok(values)
221    }
222
223    fn has_column(&self, name: &str) -> bool {
224        let HduInfo::TableInfo {
225            column_descriptions,
226            ..
227        } = &self.hdu.info
228        else {
229            return false;
230        };
231        column_descriptions.iter().any(|column| column.name == name)
232    }
233
234    fn source_vector(&mut self, name: &str, width: usize) -> MsResult<Vec<f64>> {
235        let values = self.hdu.read_col::<f64>(self.fits, name).map_err(invalid)?;
236        if self.rows.checked_mul(width) == Some(values.len()) {
237            Ok(values.chunks_exact(width).map(|row| row[0]).collect())
238        } else if values.len() == self.rows {
239            Ok(values)
240        } else {
241            Err(invalid(format!("Inconsistent UVFITS source column {name}")))
242        }
243    }
244}

246#[doc(hidden)]
247pub fn fixed_text_column(fits: &mut FitsFile, hdu: &FitsHdu, name: &str) -> MsResult<Vec<String>> {
248    let HduInfo::TableInfo {
249        column_descriptions,
250        num_rows,
251    } = &hdu.info
252    else {
253        return Err(invalid("Expected a FITS binary table"));
254    };
255    let (index, _) = column_descriptions
256        .iter()
257        .enumerate()
258        .find(|(_, column)| column.name == name)
259        .ok_or_else(|| invalid(format!("Cannot find FITS column {name}")))?;
260    let column = i32::try_from(index + 1).map_err(invalid)?;
261    fits.hdu(hdu.number).map_err(invalid)?;
262    let width = fixed_text_width(fits, column, name)?;
263    let len = width
264        .checked_mul(*num_rows)
265        .ok_or_else(|| invalid("FITS text column size overflow"))?;
266    let mut storage = Vec::new();
267    storage.try_reserve_exact(len).map_err(invalid)?;
268    storage.resize(len, 0);
269    let mut data = storage
270        .chunks_exact_mut(width)
271        .map(<[_]>::as_mut_ptr)
272        .collect::<Vec<_>>();
273    let mut status = 0;
274    // SAFETY: the current HDU is a table and every output string has room
275    // for its entire fixed field plus the terminating NUL byte.
276    unsafe {
277        sys::ffgcvs(
278            fits.as_raw(),
279            column,
280            1,
281            1,
282            *num_rows as i64,
283            ptr::null_mut(),
284            data.as_mut_ptr(),
285            ptr::null_mut(),
286            &mut status,
287        );
288    }
289    errors::check_status(status).map_err(invalid)?;
290    Ok(storage
291        .chunks_exact(width)
292        .map(|row| {
293            let value = row
294                .iter()
295                .take_while(|&&byte| byte != 0)
296                .map(|&byte| byte as u8)
297                .collect::<Vec<_>>();
298            String::from_utf8_lossy(&value).trim_end().to_owned()
299        })
300        .collect())
301}

303fn fixed_text_width(fits: &mut FitsFile, column: i32, name: &str) -> MsResult<usize> {
304    let mut typ = 0;
305    let mut repeat = 0;
306    let mut width = 0;
307    let mut status = 0;
308    // SAFETY: the live handle has the caller's table HDU selected, and all
309    // output pointers refer to initialized writable values of the right types.
310    unsafe {
311        sys::ffgtclll(
312            fits.as_raw(),
313            column,
314            &mut typ,
315            &mut repeat,
316            &mut width,
317            &mut status,
318        )
319    };
320    errors::check_status(status).map_err(invalid)?;
321    if typ != sys::TSTRING as i32 || repeat <= 0 || width <= 0 {
322        return Err(invalid(format!(
323            "FITS column {name} must contain fixed-width text"
324        )));
325    }
326    usize::try_from(repeat.max(width))
327        .ok()
328        .and_then(|size| size.checked_add(1))
329        .ok_or_else(|| invalid("FITS text width overflow"))
330}

332fn read_array(mut table: BinaryTable<'_>, telescope: &str) -> MsResult<Array> {
333    let positions = table.column::<f64>("STABXYZ", 3)?;
334    let numbers = table.column::<i32>("NOSTA", 1)?;
335    let stations = table.strings("ANNAME")?;
336    let mounts = table.column::<i32>("MNTSTA", 1)?;
337    let axis_offsets = table.column::<f32>("STAXOF", 1)?;
338    let angle_a = table.column::<f32>("POLAA", 1)?;
339    let angle_b = table.column::<f32>("POLAB", 1)?;
340    let origin = ["ARRAYX", "ARRAYY", "ARRAYZ"].map(|key| table.hdu.read_key(table.fits, key));
341    let [x, y, z] = origin;
342    let origin = [
343        x.map_err(invalid)?,
344        y.map_err(invalid)?,
345        z.map_err(invalid)?,
346    ];
347    let name = string_key(table.hdu, table.fits, "ARRNAM", "Unknown")?;
348    let left_handed = string_key(table.hdu, table.fits, "XYZHAND", "")? == "LEFT";
349    let frame_defined = optional_key::<String>(table.hdu, table.fits, "FRAME")?.is_some();
350    let date = string_key(table.hdu, table.fits, "RDATE", "")?;
351    let time_system = string_key(table.hdu, table.fits, "TIMSYS", "TAI")?;
352    let gst = optional_key::<f64>(table.hdu, table.fits, "GSTIA0")?
353        .unwrap_or(0.0)
354        .to_radians();
355    let degrees_per_day = optional_key(table.hdu, table.fits, "DEGPDY")?.unwrap_or(0.0);
356    let diameters = if table.has_column("DIAMETER") {
357        table.column::<f32>("DIAMETER", 1)?
358    } else {
359        Vec::new()
360    };
361    let use_diameters = diameters.iter().any(|&diameter| diameter > 0.0);
362    let mut antennas = BTreeMap::new();
363    for (index, number) in numbers.into_iter().enumerate() {
364        if number < 1 {
365            return Err(invalid("Invalid station number in UVFITS AN table"));
366        }
367        let station = stations[index]
368            .split('\0')
369            .next()
370            .unwrap_or_default()
371            .trim_end();
372        let position: [f64; 3] = positions[3 * index..3 * index + 3].try_into().unwrap();
373        let diameter = if use_diameters {
374            diameters[index]
375        } else {
376            diameter(telescope, number, station)
377        };
378        antennas.insert(
379            number - 1,
380            Antenna {
381                number,
382                station: station.to_owned(),
383                position,
384                mount: if telescope == "SMA" { 0 } else { mounts[index] },
385                axis_offset: f64::from(axis_offsets[index]),
386                angles: [angle_a[index], angle_b[index]].map(|angle| f64::from(angle).to_radians()),
387                diameter: f64::from(diameter),
388            },
389        );
390    }
391    Ok(Array {
392        name,
393        origin,
394        left_handed,
395        frame_defined,
396        antennas,
397        date,
398        gst,
399        degrees_per_day,
400        time_system,
401    })
402}

404fn diameter(telescope: &str, number: i32, name: &str) -> f32 {
405    match telescope {
406        "ATCA" => 22.0,
407        "SMA" => 6.0,
408        "HATCREEK" | "BIMA" | "ATA" => 6.1,
409        "GMRT" => 45.0,
410        "IRAM_PDB" | "IRAM PDB" => 15.0,
411        "OVRO" | "CARMA" if number <= 6 => 10.4,
412        "OVRO" | "CARMA" => 6.1,
413        "ALMA" if name.starts_with("CM") => 7.0,
414        "ALMA"
415            if ["DA", "DV", "PM"]
416                .iter()
417                .any(|prefix| name.starts_with(prefix)) =>
418        {
419            12.0
420        }
421        _ => 25.0,
422    }
423}

425fn read_system_power(mut table: BinaryTable<'_>) -> MsResult<Table> {
426    let no_if = table
427        .hdu
428        .read_key::<i32>(table.fits, "NO_IF")
429        .map_err(invalid)?;
430    let no_pol = table
431        .hdu
432        .read_key::<i32>(table.fits, "NO_POL")
433        .map_err(invalid)?;
434    if no_if != 1 || !matches!(no_pol, 1 | 2) {
435        return Err(invalid(
436            "UVFITS SYSPOWER requires one IF and one or two polarizations",
437        ));
438    }
439    let time = table.column::<f64>("TIME", 1)?;
440    let intervals = table.column::<f32>("TIME INTERVAL", 1)?;
441    let antennas = table.column::<i32>("ANTENNA NO.", 1)?;
442    let windows = table.column::<i32>("FREQ ID", 1)?;
443    let power_names = ["SWITCHED_DIFF", "SWITCHED_SUM", "REQUANTIZER_GAIN"];
444    let power = ["POWER DIF", "POWER SUM", "POST GAIN"].map(|source| {
445        (1..=no_pol)
446            .map(|hand| table.column::<f32>(&format!("{source}{hand}"), 1))
447            .collect::<MsResult<Vec<_>>>()
448    });
449    let [dif, sum, gain] = power;
450    let power = [dif?, sum?, gain?];
451    let mut columns = vec![
452        ColumnSchema::scalar("ANTENNA_ID", PrimitiveType::Int32),
453        ColumnSchema::scalar("FEED_ID", PrimitiveType::Int32),
454        ColumnSchema::scalar("SPECTRAL_WINDOW_ID", PrimitiveType::Int32),
455        ColumnSchema::scalar("TIME", PrimitiveType::Float64),
456        ColumnSchema::scalar("INTERVAL", PrimitiveType::Float32),
457    ];
458    columns.extend(
459        power_names.map(|name| ColumnSchema::array_variable(name, PrimitiveType::Float32, Some(1))),
460    );
461    let mut output = Table::with_schema(TableSchema::new(columns).map_err(invalid)?);
462    for row in 0..table.rows {
463        let mut values = RecordValue::new(vec![
464            RecordField::new(
465                "ANTENNA_ID",
466                Value::Scalar(ScalarValue::Int32(antennas[row] - 1)),
467            ),
468            RecordField::new("FEED_ID", Value::Scalar(ScalarValue::Int32(0))),
469            RecordField::new(
470                "SPECTRAL_WINDOW_ID",
471                Value::Scalar(ScalarValue::Int32(windows[row] - 1)),
472            ),
473            RecordField::new(
474                "TIME",
475                Value::Scalar(ScalarValue::Float64(time[row] * 86_400.0)),
476            ),
477            RecordField::new(
478                "INTERVAL",
479                Value::Scalar(ScalarValue::Float32(intervals[row] * 86_400.0)),
480            ),
481        ]);
482        for (name, hands) in power_names.into_iter().zip(&power) {
483            let array = Array1::from_iter(hands.iter().map(|hand| hand[row]));
484            values.upsert(name, Value::Array(ArrayValue::Float32(array.into_dyn())));
485        }
486        output.add_row(values)?;
487    }
488    Ok(output)
489}

491fn read_spectral(
492    mut table: BinaryTable<'_>,
493    layout: &GroupLayout,
494) -> MsResult<Vec<SpectralWindow>> {
495    let windows = layout.windows;
496    let _ids = table.column::<i32>("FRQSEL", 1)?;
497    let offsets = table.column::<f64>("IF FREQ", windows)?;
498    let widths = table.column::<f32>("CH WIDTH", windows)?;
499    let bandwidths = table.column::<f32>("TOTAL BANDWIDTH", windows)?;
500    Ok(offsets
501        .into_iter()
502        .zip(widths)
503        .zip(bandwidths)
504        .map(|((offset, increment), bandwidth)| SpectralWindow {
505            frequency: layout.frequency.reference + offset,
506            increment: f64::from(increment),
507            bandwidth: f64::from(bandwidth),
508        })
509        .collect())
510}

512fn read_sources(mut table: BinaryTable<'_>) -> MsResult<BTreeMap<i32, Source>> {
513    let numbers = table.column::<i32>("ID. NO.", 1)?;
514    let names = table.strings("SOURCE")?;
515    let qualifiers = table.column::<i32>("QUAL", 1)?;
516    let codes = table.strings("CALCODE")?;
517    let ra = table.column::<f64>("RAEPO", 1)?;
518    let dec = table.column::<f64>("DECEPO", 1)?;
519    let app_ra = table.column::<f64>("RAAPP", 1)?;
520    let app_dec = table.column::<f64>("DECAPP", 1)?;
521    let motion_ra = table.column::<f64>("PMRA", 1)?;
522    let motion_dec = table.column::<f64>("PMDEC", 1)?;
523    let epochs = table.column::<f64>("EPOCH", 1)?;
524    let windows = table
525        .hdu
526        .read_key::<i32>(table.fits, "NO_IF")
527        .map_err(invalid)?;
528    let windows = usize::try_from(windows)
529        .ok()
530        .filter(|&value| value > 0)
531        .ok_or_else(|| invalid("Invalid NO_IF in UVFITS source table"))?;
532    let frequencies = table.source_vector("RESTFREQ", windows)?;
533    let velocities = table.source_vector("LSRVEL", windows)?;
534    let multi_qualifier = qualifiers.windows(2).any(|pair| pair[0] != pair[1]);
535    let mut sources = BTreeMap::new();
536    for (index, number) in numbers.into_iter().enumerate() {
537        if number < 1 {
538            return Err(invalid("Invalid source number in UVFITS SU table"));
539        }
540        let mut name = names[index].trim_end().to_owned();
541        if multi_qualifier {
542            name = format!("{name}_{}", qualifiers[index]);
543        }
544        sources.insert(
545            number - 1,
546            Source {
547                name,
548                code: codes[index].trim_end().to_owned(),
549                epoch: epochs[index],
550                direction_deg: [ra[index], dec[index]],
551                apparent_deg: [app_ra[index], app_dec[index]],
552                motion_per_day: [motion_ra[index], motion_dec[index]],
553                rest_frequency: frequencies[index],
554                velocity: velocities[index],
555            },
556        );
557    }
558    Ok(sources)
559}

561fn string_key(hdu: &FitsHdu, fits: &mut FitsFile, key: &str, fallback: &str) -> MsResult<String> {
562    Ok(optional_key::<String>(hdu, fits, key)?
563        .map_or_else(|| fallback.to_owned(), |value| value.trim_end().to_owned()))
564}

566#[doc(hidden)]
567pub fn history(fits: &mut FitsFile) -> MsResult<Vec<String>> {
568    use fitsio::{errors, sys};
569    let mut result = Vec::new();
570    let (mut status, mut count, mut space) = (0, 0, 0);
571    // SAFETY: FFghsp reads the current HDU and writes to these live integers.
572    unsafe { sys::ffghsp(fits.as_raw(), &mut count, &mut space, &mut status) };
573    errors::check_status(status).map_err(invalid)?;
574    for number in 1..=count {
575        let mut card = [0; 81];
576        // SAFETY: FFgrec writes one 80-byte card and a trailing null byte.
577        unsafe { sys::ffgrec(fits.as_raw(), number, card.as_mut_ptr(), &mut status) };
578        errors::check_status(status).map_err(invalid)?;
579        let card = card.map(|byte| byte as u8);
580        if let Some(message) = card.strip_prefix(b"HISTORY ") {
581            let length = message
582                .iter()
583                .position(|&byte| byte == 0)
584                .unwrap_or(message.len());
585            let message = String::from_utf8_lossy(&message[..length]);
586            result.push(message.trim_end().to_owned());
587        }
588    }
589    Ok(result)
590}
