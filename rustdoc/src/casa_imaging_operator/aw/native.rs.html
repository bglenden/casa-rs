15const SPEED_OF_LIGHT_M_PER_S: f64 = 299_792_458.0;

17/// One generated, independently cropped plane, `size × size` with x
18/// fastest and the kernel origin at `size / 2`.
19#[derive(Debug)]
20pub struct NativeAwPlane<'a> {
21    /// Square pixel extent after support-buffer cropping.
22    pub size: usize,
23    /// Symmetric support in non-oversampled grid pixels.
24    pub support: usize,
25    /// Complex32 samples with the CASA sampled-area normalisation applied.
26    pub values: &'a [Complex32],
27}
28
29/// Imaging and weight cells produced from the same scientific request.
30#[derive(Debug)]
31pub struct NativeAwPair<'a> {
32    /// A times W times the optional anti-aliasing term.
33    pub imaging: NativeAwPlane<'a>,
34    /// Frequency/conjugate-frequency beam product, without the W phase.
35    pub weight: NativeAwPlane<'a>,
36}
37
38/// Reusable one-cell numerical workspace: six square `Complex32` planes and
39/// one FFT plan, independent of catalog size.
40pub struct EvlaAwWorkspace {
41    size: usize,
42    fft: PlaneFft<f32>,
43    jones: Vec<Complex32>,
44    imaging: Vec<Complex32>,
45    weight: Vec<Complex32>,
46    realized: Option<[(usize, usize); 2]>,
47}
48
49impl EvlaAwWorkspace {
50    /// Allocate the working grid of `request`'s size; the same allocation
51    /// serves every cell of that size.
52    ///
53    /// # Errors
54    ///
55    /// The request is invalid or the FFT cannot be planned.
56    pub fn new(request: EvlaAwCellRequest) -> Result<Self, NativeAwGenerationError> {
57        request
58            .validate()
59            .map_err(|_| NativeAwGenerationError::InvalidGrid)?;
60        let n = request.size * request.size;
61        Ok(Self {
62            size: request.size,
63            fft: PlaneFft::<f32>::new([request.size; 2], false)
64                .map_err(|_| NativeAwGenerationError::WorkspaceMismatch)?,
65            jones: vec![Complex32::default(); 4 * n],
66            imaging: vec![Complex32::default(); n],
67            weight: vec![Complex32::default(); n],
68            realized: None,
69        })
70    }
71
72    /// Bytes of the working planes.
73    #[must_use]
74    pub fn resident_bytes(&self) -> usize {
75        6 * self.size * self.size * size_of::<Complex32>()
76    }
77
78    /// Borrow the last successfully realized pair, with no pixel copies.
79    #[must_use]
80    pub fn pair(&self) -> Option<NativeAwPair<'_>> {
81        let [
82            (imaging_size, imaging_support),
83            (weight_size, weight_support),
84        ] = self.realized?;
85        Some(NativeAwPair {
86            imaging: NativeAwPlane {
87                size: imaging_size,
88                support: imaging_support,
89                values: &self.imaging[..imaging_size * imaging_size],
90            },
91            weight: NativeAwPlane {
92                size: weight_size,
93                support: weight_support,
94                values: &self.weight[..weight_size * weight_size],
95            },
96        })
97    }
98
99    /// Generate a paired EVLA A/W cell without CASA or cache access
100    /// (`AWConvFunc::makeConvFunction2`): the sky Mueller element at the
101    /// cell frequency times the W screen and the optional prolate
102    /// spheroidal, and its weight partner at the conjugate frequency; both
103    /// transformed, the imaging plane peak-normalised, both cropped to
104    /// their support buffers and normalised by their sampled area.
105    ///
106    /// # Errors
107    ///
108    /// The request does not match the workspace, the receiver band is
109    /// unsupported, or a value or normalisation is not finite.
110    pub fn generate(
111        &mut self,
112        model: &EvlaApertureModel,
113        request: EvlaAwCellRequest,
114    ) -> Result<NativeAwPair<'_>, NativeAwGenerationError> {
115        self.realized = None;
116        request
117            .validate()
118            .map_err(|_| NativeAwGenerationError::InvalidGrid)?;
119        if self.size != request.size {
120            return Err(NativeAwGenerationError::WorkspaceMismatch);
121        }
122        let Self {
123            fft,
124            jones,
125            imaging,
126            weight,
127            ..
128        } = self;
129        fill_sky_mueller(model, request, request.frequency_hz, fft, jones, imaging)?;
130        fill_sky_mueller(
131            model,
132            request,
133            request.conjugate_frequency_hz,
134            fft,
135            jones,
136            weight,
137        )?;
138        let origin = (request.size / 2) as isize;
139        let inner_half = (request.size / request.oversampling / 2) as isize;
140        for y in 0..request.size {
141            let iy = y as isize - origin;
142            let m = request.sky_increment_rad[1] * iy as f64;
143            for x in 0..request.size {
144                let ix = x as isize - origin;
145                let p = if request.prolate_spheroidal {
146                    if (-inner_half..inner_half).contains(&ix)
147                        && (-inner_half..inner_half).contains(&iy)
148                    {
149                        (grdsf(ix as f64 / inner_half as f64) as f32)
150                            * (grdsf(iy as f64 / inner_half as f64) as f32)
151                    } else {
152                        0.0
153                    }
154                } else {
155                    1.0
156                };
157                let pixel = x + request.size * y;
158                weight[pixel] =
159                    Complex32::new(p * p, 0.0) * (imaging[pixel] * weight[pixel].conj());
160                let l = request.sky_increment_rad[0] * ix as f64;
161                let rsq = l * l + m * m;
162                let mut screen = Complex32::new(p, 0.0);
163                if request.w_wavelengths > 0.0 && rsq < 1.0 {
164                    let phase =
165                        std::f64::consts::TAU * request.w_wavelengths * ((1.0 - rsq).sqrt() - 1.0);
166                    let (sin, cos) = phase.sin_cos();
167                    screen *= Complex32::new(cos as f32, sin as f32);
168                }
169                imaging[pixel] *= screen;
170            }
171        }
172        // CASA's copy back from the FFT image excludes the final row and
173        // column; the pre-transform edges survive in the spent Jones planes.
174        let edges = &mut jones[..4 * request.size];
175        save_edges(imaging, request.size, &mut edges[..2 * request.size]);
176        save_edges(weight, request.size, &mut edges[2 * request.size..]);
177        transform(fft, imaging)?;
178        transform(fft, weight)?;
179        restore_edges(imaging, request.size, &edges[..2 * request.size]);
180        restore_edges(weight, request.size, &edges[2 * request.size..]);
181        let mut peak = imaging[0];
182        for value in &imaging[1..] {
183            if value.norm_sqr() > peak.norm_sqr() {
184                peak = *value;
185            }
186        }
187        if peak.norm() == 0.0 || !peak.norm().is_finite() {
188            return Err(NativeAwGenerationError::InvalidNumerics);
189        }
190        for value in imaging.iter_mut() {
191            *value /= peak;
192        }
193        self.realized = Some([
194            crop_normalize(imaging, request)?,
195            crop_normalize(weight, request)?,
196        ]);
197        Ok(self.pair().expect("both crops succeeded"))
198    }
199}

201/// The sky Mueller element `request.mueller` at `frequency`: the four
202/// Jones aperture planes transformed, normalised by the centre power, the
203/// selected plane conjugated and squared
204/// (`VLACalcIlluminationConvFunc::applyPB`); unity without the aperture.
205fn fill_sky_mueller(
206    model: &EvlaApertureModel,
207    request: EvlaAwCellRequest,
208    frequency: f64,
209    fft: &mut PlaneFft<f32>,
210    jones: &mut [Complex32],
211    output: &mut [Complex32],
212) -> Result<(), NativeAwGenerationError> {
213    if !request.aperture {
214        output.fill(Complex32::new(1.0, 0.0));
215        return Ok(());
216    }
217    let wavelength = SPEED_OF_LIGHT_M_PER_S / f64::from(frequency as f32);
218    let cell_m = wavelength / (request.size as f64 * request.sky_increment_rad[0].abs());
219    model.fill_aperture(
220        EvlaApertureGrid::new(
221            request.size,
222            cell_m,
223            3,
224            frequency,
225            request.parallactic_angle_rad,
226        )?,
227        jones,
228    )?;
229    let n = request.size * request.size;
230    for plane in jones.chunks_exact_mut(n) {
231        transform(fft, plane)?;
232    }
233    let center = request.size / 2 * (request.size + 1);
234    let mut norm_squared = 0.0_f32;
235    for plane in jones.chunks_exact(n) {
236        norm_squared = (f64::from(norm_squared)
237            + f64::from((plane[center] * plane[center]).norm()) / 2.0)
238            as f32;
239    }
240    if !norm_squared.is_finite() || norm_squared <= 0.0 {
241        return Err(NativeAwGenerationError::InvalidNumerics);
242    }
243    let normalization = norm_squared.sqrt();
244    let selected = if request.mueller == 0 { 0 } else { 3 };
245    for (output, value) in output
246        .iter_mut()
247        .zip(&jones[selected * n..(selected + 1) * n])
248    {
249        let value = (*value / normalization).conj();
250        *output = value * value.conj();
251    }
252    Ok(())
253}

255/// A centred forward transform of a `[y][x]` plane in `f32`
256/// (`FFT2D::c2cFFT`).
257fn transform(
258    fft: &mut PlaneFft<f32>,
259    values: &mut [Complex32],
260) -> Result<(), NativeAwGenerationError> {
261    fft.transform(values, false)
262        .map_err(|_| NativeAwGenerationError::WorkspaceMismatch)
263}
264
265fn save_edges(values: &[Complex32], size: usize, edges: &mut [Complex32]) {
266    edges[..size].copy_from_slice(&values[(size - 1) * size..]);
267    for y in 0..size {
268        edges[size + y] = values[y * size + size - 1];
269    }
270}
271
272fn restore_edges(values: &mut [Complex32], size: usize, edges: &[Complex32]) {
273    values[(size - 1) * size..].copy_from_slice(&edges[..size]);
274    for y in 0..size {
275        values[y * size + size - 1] = edges[size + y];
276    }
277}

279/// `AWConvFunc::resizeCF` and `cfArea`: the support from the last ring
280/// above `1e-3` of the centre, the crop to the support buffer, and the
281/// normalisation by the sampled area over `[−support, support)`.
282fn crop_normalize(
283    values: &mut [Complex32],
284    request: EvlaAwCellRequest,
285) -> Result<(usize, usize), NativeAwGenerationError> {
286    let origin = request.size / 2;
287    let threshold = (f64::from(values[origin * (request.size + 1)].norm()) * 1e-3) as f32;
288    let mut radius = None;
289    'radii: for r in (2..=origin - 2).rev() {
290        for pixel in 0..90 * r {
291            let angle = std::f64::consts::TAU * pixel as f64 / r as f64;
292            let x = (origin as f64 + r as f64 * angle.sin()) as usize;
293            let y = (origin as f64 + r as f64 * angle.cos()) as usize;
294            if values[x + request.size * y].norm() > threshold {
295                radius = Some(r);
296                break 'radii;
297            }
298        }
299    }
300    let radius = radius.ok_or(NativeAwGenerationError::InvalidNumerics)?;
301    let mut support = (0.5_f32 + radius as f32 / request.oversampling as f32) as usize + 1;
302    if support * request.oversampling + (request.oversampling as f32 / 2.0 + 0.5) as usize > origin
303    {
304        support = origin / request.oversampling - 1;
305    }
306    if support == 0 {
307        return Err(NativeAwGenerationError::InvalidNumerics);
308    }
309    let extent = (request.oversampling * (support + 2)) as isize;
310    let bottom = (((origin as isize - extent) / 2) * 2).max(0) as usize;
311    let top = ((((origin as isize + extent) / 2) * 2 - 1) as usize).min(request.size - 1);
312    let size = top - bottom + 1;
313    let mut area = Complex32::default();
314    for ix in -(support as isize)..support as isize {
315        for iy in -(support as isize)..support as isize {
316            let x = (origin as isize + ix * request.oversampling as isize) as usize;
317            let y = (origin as isize + iy * request.oversampling as isize) as usize;
318            area += values[x + request.size * y];
319        }
320    }
321    if area.norm() == 0.0 || !area.norm().is_finite() {
322        return Err(NativeAwGenerationError::InvalidNumerics);
323    }
324    for y in 0..size {
325        for x in 0..size {
326            values[x + size * y] = values[x + bottom + request.size * (y + bottom)] / area;
327        }
328    }
329    if values[..size * size]
330        .iter()
331        .any(|v| !v.re.is_finite() || !v.im.is_finite())
332    {
333        return Err(NativeAwGenerationError::InvalidNumerics);
334    }
335    Ok((size, support))
336}

356    #[test]
357    fn the_workspace_is_reused_and_deterministic() {
358        let model = EvlaApertureModel::new(surface());
359        let request = EvlaAwCellRequest {
360            size: 128,
361            sky_increment_rad: [-0.001, 0.001],
362            frequency_hz: 3e9,
363            conjugate_frequency_hz: 3.1e9,
364            w_wavelengths: 100.0,
365            parallactic_angle_rad: 0.31,
366            mueller: 0,
367            oversampling: 4,
368            prolate_spheroidal: false,
369            aperture: true,
370        };
371        let mut workspace = EvlaAwWorkspace::new(request).expect("workspace");
372        let pair = workspace.generate(&model, request).expect("pair");
373        let first = (pair.imaging.values.to_vec(), pair.weight.values.to_vec());
374        let pair = workspace.generate(&model, request).expect("pair");
375        assert_eq!(pair.imaging.values, first.0);
376        assert_eq!(pair.weight.values, first.1);
377        let mut invalid = request;
378        invalid.size *= 2;
379        assert!(workspace.generate(&model, invalid).is_err());
380        assert!(workspace.pair().is_none());
381    }

401        let mut workspace = EvlaAwWorkspace::new(request).expect("workspace");
402        let pair = workspace.generate(&model, request).expect("pair");
403        for plane in [pair.imaging, pair.weight] {

405            for ix in -(plane.support as isize)..plane.support as isize {
406                for iy in -(plane.support as isize)..plane.support as isize {
407                    let x = (plane.size as isize / 2 + ix * 4) as usize;
408                    let y = (plane.size as isize / 2 + iy * 4) as usize;
409                    let value = plane.values[x + plane.size * y];

411                }
412            }
413            assert!(
414                (area - 1.0).norm() < 2e-6,
415                "sampled-area normalisation {area}"
416            );
417        }

420    #[test]
421    fn the_centred_f32_transform_obeys_the_impulse_and_phase_laws() {
422        let size = 16;
423        let mut fft = PlaneFft::<f32>::new([size; 2], false).expect("plan");
424        let mut values = vec![Complex32::default(); size * size];
425        values[8 + size * 8] = Complex32::new(1.0, 0.0);
426        transform(&mut fft, &mut values).expect("transform");
427        assert!(values.iter().all(|v| *v == Complex32::new(1.0, 0.0)));
428        values.fill(Complex32::default());
429        values[9 + size * 8] = Complex32::new(1.0, 0.0);
430        transform(&mut fft, &mut values).expect("transform");
431        for y in 0..size {
432            for x in 0..size {
433                let expected = Complex32::from_polar(
434                    1.0,
435                    -std::f32::consts::TAU * (x as f32 - 8.0) / size as f32,
436                );
437                assert!((values[x + size * y] - expected).norm() <= 4.0 * f32::EPSILON);
438            }
439        }
440    }
