2//! A component sampled at an image's pixel directions, as CASA's
3//! `SkyCompRep::sample` and the shapes' `sample` methods sample it
4//! (`components/ComponentModels`) when `ia.modify` adds a component list to
5//! an image.
6//!
7//! Each shape gives every pixel direction a scale:
8//! - **a point:** 1 when the direction lies within half a pixel of the
9//!   component along both axes, else 0 (`PointShape::dirIsInPixel`);
10//! - **a Gaussian:** the pixel's area times the unit-flux Gaussian at the
11//!   direction's offset, out to four major axes (`GaussianShape::sample`);
12//! - **a disk:** the pixel's area over the disk's, for a direction inside
13//!   the ellipse (`DiskShape::_calcSample`).
14//!
15//! The offsets are casacore's: the separation and position angle from the
16//! component to the direction (`MVDirection::separation` and
17//! `MVDirection::positionAngle`), the angle counted from north through east.
18//! A pixel gains the component's Stokes flux in Jy, sampled by its spectral
19//! model at each frequency
20//! ([`SpectralModel::sample_stokes`](crate::SpectralModel::sample_stokes)),
21//! times the scale.
22//!
23//! The component's reference direction is first converted into the frame of
24//! the pixel directions with no measures frame, as CASA converts it; a
25//! conversion that needs an epoch or a position (to AZEL, say) is refused. A
26//! limb-darkened disk is refused, as its visibility is.
27//!
28//! The frequencies are taken in the frame of the spectral model's reference
29//! frequency, as the visibility takes them: a caller sampling an image in
30//! another frame converts them first (CASA converts the reference frequency
31//! into the image's frame, `refFreqInFrame`, which gives the same ratio).
32

43/// The pixels a component is sampled on: the frame of their directions and
44/// the size of a pixel along each axis.
45#[derive(Debug, Clone, Copy, PartialEq)]
46pub struct PixelGrid {
47    /// The frame of the pixel directions (the image's direction coordinate).
48    pub frame: DirectionRef,
49    /// The pixel's size along the longitude axis, rad (the increment's
50    /// magnitude).
51    pub longitude_size_rad: f64,
52    /// The pixel's size along the latitude axis, rad.
53    pub latitude_size_rad: f64,
54}
55
56impl PixelGrid {
57    /// The pixel's solid angle, sr, as CASA forms it
58    /// (`pixelLatSize.radian() * pixelLongSize.radian()`).
59    fn area(&self) -> f64 {
60        self.latitude_size_rad * self.longitude_size_rad
61    }
62}
63
64/// `MVPosition::getLong`: the longitude of `xyz`, 0 at a pole.
65fn longitude(xyz: [f64; 3]) -> f64 {
66    if xyz[0] != 0.0 || xyz[1] != 0.0 {
67        xyz[1].atan2(xyz[0])
68    } else {
69        0.0
70    }
71}
72
73/// `MVDirection::separation`: the angle from the unit vector `from` to
74/// `to` (normalised first).
75fn separation(from: [f64; 3], to: [f64; 3]) -> f64 {
76    let length = (to[0] * to[0] + to[1] * to[1] + to[2] * to[2]).sqrt();
77    let length = if length > 0.0 { length } else { 1.0 };
78    let half = ((from[0] - to[0] / length).powi(2)
79        + (from[1] - to[1] / length).powi(2)
80        + (from[2] - to[2] / length).powi(2))
81    .sqrt()
82        / 2.0;
83    2.0 * half.min(1.0).asin()
84}
85
86/// `MVDirection::positionAngle`: the position angle of `to` seen from the
87/// unit vector `from`, north through east.
88fn position_angle(from: [f64; 3], to: [f64; 3]) -> f64 {
89    let longitude_difference = longitude(from) - longitude(to);
90    let sin_lat1 = from[2];
91    let sin_lat2 = to[2];
92    let cos_lat2 = (1.0 - sin_lat2 * sin_lat2).abs().sqrt();
93    let s1 = -cos_lat2 * longitude_difference.sin();
94    let c1 = (1.0 - sin_lat1 * sin_lat1).abs().sqrt() * sin_lat2
95        - sin_lat1 * cos_lat2 * longitude_difference.cos();
96    if s1 != 0.0 || c1 != 0.0 {
97        s1.atan2(c1)
98    } else {
99        0.0
100    }
101}
102
103/// casacore's `Gaussian2DParam::fwhm2int`, `1/√(ln 16)`.
104fn fwhm_to_width() -> f64 {
105    1.0 / 16.0_f64.ln().sqrt()
106}

108impl ComponentShape {
109    /// The shape's scale at each of `directions` (direction cosines in
110    /// `grid.frame`): see the module documentation.
111    ///
112    /// # Errors
113    ///
114    /// [`ComponentError::Direction`] when the reference direction cannot be
115    /// converted into `grid.frame` without a measures frame, and
116    /// [`ComponentError::ImageSample`] for a limb-darkened disk.
117    pub fn pixel_scales(
118        &self,
119        directions: &[[f64; 3]],
120        grid: &PixelGrid,
121    ) -> Result<Vec<f64>, ComponentError> {
122        let centre = self.direction_in(grid.frame)?;
123        Ok(match self.kind() {
124            ShapeKind::Point => {
125                let near = grid.latitude_size_rad.max(grid.longitude_size_rad);
126                directions
127                    .iter()
128                    .map(|&direction| {
129                        let separation = separation(centre, direction).abs();
130                        if separation > near {
131                            return 0.0;
132                        }
133                        let angle = position_angle(centre, direction);
134                        let inside = (separation * angle.sin()).abs()
135                            <= grid.longitude_size_rad / 2.0
136                            && (separation * angle.cos()).abs() <= grid.latitude_size_rad / 2.0;
137                        if inside { 1.0 } else { 0.0 }
138                    })
139                    .collect()
140            }
141            ShapeKind::Gaussian { axes, .. } => {
142                // `Gaussian2D` with its major axis along y, unit flux.
143                let width = fwhm_to_width();
144                let major = axes.major_axis();
145                let ratio = axes.minor_axis() / major;
146                let height = 1.0 / ((major * major * ratio * PI).abs() * width * width);
147                let (sin_pa, cos_pa) = axes.position_angle().sin_cos();
148                let area = grid.area();
149                let cutoff = 4.0 * major;
150                directions
151                    .iter()
152                    .map(|&direction| {
153                        let separation = separation(centre, direction);
154                        if separation > cutoff {
155                            return 0.0;
156                        }
157                        let angle = -position_angle(centre, direction);
158                        let (x, y) = (separation * angle.sin(), separation * angle.cos());
159                        let across = (cos_pa * x + sin_pa * y) / (major * ratio * width);
160                        let along = (-sin_pa * x + cos_pa * y) / (major * width);
161                        area * height * (-(across * across + along * along)).exp()
162                    })
163                    .collect()
164            }
165            ShapeKind::Disk { axes, .. } => {
166                let major_radius = axes.major_axis() / 2.0;
167                let minor_radius = axes.minor_axis() / 2.0;
168                let value = grid.area() / (FRAC_PI_4 * axes.major_axis() * axes.minor_axis());
169                directions
170                    .iter()
171                    .map(|&direction| {
172                        let separation = separation(centre, direction);
173                        if separation > major_radius {
174                            return 0.0;
175                        }
176                        let angle = position_angle(centre, direction) - axes.position_angle();
177                        let along = (separation * angle.cos()).abs();
178                        let across = (separation * angle.sin()).abs();
179                        let inside = along <= major_radius
180                            && across <= minor_radius
181                            && across
182                                <= minor_radius * (1.0 - (along / major_radius).powi(2)).sqrt();
183                        if inside { value } else { 0.0 }
184                    })
185                    .collect()
186            }
187            ShapeKind::LimbDarkenedDisk { .. } => {
188                return Err(ComponentError::ImageSample {
189                    shape: "limb-darkened disk",
190                    reason: "casa-rs does not yet sample its brightness profile".to_string(),
191                });
192            }
193        })
194    }
195
196    /// The reference direction's cosines in `frame`.
197    fn direction_in(&self, frame: DirectionRef) -> Result<[f64; 3], ComponentError> {
198        let direction: &MDirection = self.reference_direction();
199        direction
200            .convert_to(frame, &MeasFrame::default())
201            .map(|converted| converted.cosines())
202            .map_err(|error| ComponentError::Direction {
203                from: direction.refer().as_str().to_string(),
204                to: frame.as_str().to_string(),
205                reason: error.to_string(),
206            })
207    }
208}

210impl SkyComponent {
211    /// Add the component, sampled at `directions` (cosines in
212    /// `grid.frame`) and `frequencies_hz`, to `samples`, indexed
213    /// `[Stokes I/Q/U/V, direction, frequency]` (`SkyCompRep::sample`).
214    ///
215    /// # Errors
216    ///
217    /// Those of [`ComponentShape::pixel_scales`],
218    /// [`Flux::stokes_jy_values`](crate::Flux::stokes_jy_values) and
219    /// [`SpectralModel::sample_stokes`](crate::SpectralModel::sample_stokes).
220    ///
221    /// # Panics
222    ///
223    /// When `samples` is not `[4, directions.len(), frequencies_hz.len()]`.
224    pub fn sample_pixels(
225        &self,
226        samples: &mut Array3<f64>,
227        directions: &[[f64; 3]],
228        grid: &PixelGrid,
229        frequencies_hz: &[f64],
230    ) -> Result<(), ComponentError> {
231        assert_eq!(
232            samples.shape(),
233            [4, directions.len(), frequencies_hz.len()],
234            "samples are [Stokes, direction, frequency]"
235        );
236        let iquv = self.flux().stokes_jy_values()?;
237        let scales = self.shape().pixel_scales(directions, grid)?;
238        let spectra = frequencies_hz
239            .iter()
240            .map(|&frequency| self.spectrum().sample_stokes(frequency, iquv))
241            .collect::<Result<Vec<_>, _>>()?;
242        for (direction, &scale) in scales.iter().enumerate() {
243            if scale == 0.0 {
244                continue;
245            }
246            for (frequency, stokes) in spectra.iter().enumerate() {
247                for (index, value) in stokes.iter().enumerate() {
248                    samples[[index, direction, frequency]] += scale * value;
249                }
250            }
251        }
252        Ok(())
253    }
254}
