2//! Sparse tile patches can borrow cells already in numeric tile order.
3
4use super::{
5    ArrayD, ArrayValue, CasacoreDataType, Cow, StorageError, TilePixel, encode_tiled_array,
6    is_fortran_contiguous, tile_values_as_bytes,
7};
8
9pub(super) fn encode_row(
10    value: &ArrayValue,
11    big_endian: bool,
12) -> Result<(Cow<'_, [u8]>, CasacoreDataType), StorageError> {
13    if big_endian == cfg!(target_endian = "big") {
14        let borrowed = match value {
15            ArrayValue::Complex32(array) => {
16                fortran_bytes(array).map(|bytes| (bytes, CasacoreDataType::TpComplex))
17            }
18            ArrayValue::Float32(array) => {
19                fortran_bytes(array).map(|bytes| (bytes, CasacoreDataType::TpFloat))
20            }
21            _ => None,
22        };
23        if let Some((bytes, data_type)) = borrowed {
24            return Ok((Cow::Borrowed(bytes), data_type));
25        }
26    }
27    encode_tiled_array(value, big_endian).map(|(bytes, data_type)| (Cow::Owned(bytes), data_type))
28}
29
30fn fortran_bytes<T: TilePixel>(array: &ArrayD<T>) -> Option<&[u8]> {
31    if !is_fortran_contiguous(array.shape(), array.strides()) {
32        return None;
33    }
34    array.as_slice_memory_order().map(tile_values_as_bytes)
35}

37#[cfg(test)]
38mod tests {
39    use casa_core::Complex32;
40    use ndarray::{IxDyn, ShapeBuilder};
41
42    use super::*;
43
44    #[test]
45    fn sparse_numeric_cells_preserve_layout_and_endianness() {
46        let shape = IxDyn(&[2, 3]);
47        let floats = vec![
48            1.25,
49            f32::from_bits(0x7fc0_1234),
50            -0.0,
51            4.0,
52            -2.5,
53            f32::INFINITY,
54        ];
55        let complex = floats
56            .iter()
57            .map(|&real| Complex32::new(real, -real))
58            .collect::<Vec<_>>();
59        let float_row_major = ArrayD::from_shape_vec(shape.clone(), floats.clone()).unwrap();
60        let complex_row_major = ArrayD::from_shape_vec(shape.clone(), complex.clone()).unwrap();
61        let float_fortran = ArrayD::from_shape_vec(shape.clone().f(), floats).unwrap();
62        let complex_fortran = ArrayD::from_shape_vec(shape.f(), complex).unwrap();
63
64        for (value, fortran) in [
65            (ArrayValue::Float32(float_row_major), false),
66            (ArrayValue::Complex32(complex_row_major), false),
67            (ArrayValue::Float32(float_fortran), true),
68            (ArrayValue::Complex32(complex_fortran), true),
69        ] {
70            for big_endian in [false, true] {
71                let (reference, reference_type) = encode_tiled_array(&value, big_endian).unwrap();
72                let (encoded, actual_type) = encode_row(&value, big_endian).unwrap();
73                assert_eq!(encoded, reference);
74                assert_eq!(actual_type, reference_type);
75                assert_eq!(
76                    matches!(encoded, Cow::Borrowed(_)),
77                    fortran && big_endian == cfg!(target_endian = "big")
78                );
79            }
80        }
81    }
82}
