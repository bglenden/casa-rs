13/// Model grids ready for degridding: the forward FFT of the corrected,
14/// polarization-expanded model, one full-grid block per
15/// `(plane, grid pol, Taylor term)` in the operator's precision.
16#[derive(Clone, Debug, PartialEq)]
17pub struct PreparedModelGrids {
18    layout: AccumulatorLayout,
19    storage: GridStorage,
20}
21
22impl PreparedModelGrids {
23    pub(crate) fn new(layout: AccumulatorLayout, storage: GridStorage) -> Self {
24        debug_assert_eq!(storage.len(), layout.cells());
25        debug_assert!(layout.is_full_grid());
26        Self { layout, storage }
27    }
28
29    /// Layout of the grids (data terms only, full grid).
30    #[must_use]
31    pub const fn layout(&self) -> &AccumulatorLayout {
32        &self.layout
33    }
34
35    /// Cell precision.
36    #[must_use]
37    pub const fn precision(&self) -> GridPrecision {
38        self.storage.precision()
39    }
40
41    /// The cells.
42    #[must_use]
43    pub const fn storage(&self) -> &GridStorage {
44        &self.storage
45    }
46
47    /// Cells of block `(plane_local, pol, term)` at precision `T`.
48    #[must_use]
49    pub fn block<T: GridScalar>(
50        &self,
51        plane_local: usize,
52        pol: usize,
53        term: usize,
54    ) -> &[Complex<T>] {
55        let offset = self.layout.block_offset(plane_local, pol, term);
56        &T::cells(&self.storage)[offset..offset + self.layout.block_cells()]
57    }
58}

60/// What one dispatch does with a sample block.
61pub enum Work<'a> {
62    /// Accumulate the block into `acc` in `mode`.
63    Grid {
64        /// Data, PSF or weight terms.
65        mode: Mode,
66        /// Target accumulator; it must hold `mode`.
67        acc: &'a mut GridAccumulator,
68    },
69    /// Predict raw visibilities (`npol` per placement, sample-major,
70    /// phase-centre phasor applied) from `model`.
71    Predict {
72        /// Prepared model grids covering every plane in the block.
73        model: &'a PreparedModelGrids,
74        /// `npol × placements` predicted visibilities.
75        out: &'a mut [Complex32],
76    },
77    /// Predict, subtract from the block and grid the residual data terms in
78    /// one dispatch; optionally hand back the raw residual samples for a
79    /// final pass that writes the corrected or model column.
80    ResidualGrid {
81        /// Prepared model grids covering every plane in the block.
82        model: &'a PreparedModelGrids,
83        /// Target accumulator; it must hold the data terms.
84        acc: &'a mut GridAccumulator,
85        /// `npol × placements` raw residual visibilities, when wanted.
86        residual_out: Option<&'a mut [Complex32]>,
87    },
88}

90/// A gridding backend: CPU or device. One implementation of the kernel
91/// contract in plan section 5.3, selected by the runtime.
92pub trait GridBackend: Send {
93    /// Apply `work` to one block with kernel set `cf`.
94    ///
95    /// Every placement must lie on a plane the target holds and have its
96    /// whole support inside the target's tile; the block's polarization
97    /// count must equal the kernel set's visibility polarizations.
98    fn apply(
99        &mut self,
100        block: &SampleBlock<'_>,
101        cf: &dyn ConvolutionFunctionSet,
102        work: Work<'_>,
103    ) -> Result<(), OperatorError>;
104}
