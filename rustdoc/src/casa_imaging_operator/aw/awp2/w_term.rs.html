12pub(super) struct WTerm {
13    side: usize,
14    cell_rad: f64,
15    fft: PlaneFft<f32>,
16    screen: Vec<Complex32>,
17    aperture: Vec<Complex32>,
18}
19
20impl WTerm {
21    pub(super) fn new(side: usize, cell_rad: f64) -> Result<Self, OperatorError> {
22        Ok(Self {
23            side,
24            cell_rad,
25            fft: PlaneFft::new([side; 2], false)?,
26            screen: vec![Complex32::default(); side * side],
27            aperture: vec![Complex32::default(); side * side],
28        })
29    }
30
31    // WPConvFunc::makeWConvFuncs crops all transformed W planes to the
32    // support of the largest W. AWConvFunc uses max(cropped W, cropped A).
33    pub(super) fn convolution_side(&mut self, maximum_w: f64) -> Result<usize, OperatorError> {
34        self.transform_screen(maximum_w)?;
35        let support = screen_support(&self.screen, self.side);
36        Ok((2 * (support + 2)).min(self.side))
37    }
38
39    pub(super) fn prepare(
40        &mut self,
41        w: f64,
42        w_crop_side: usize,
43        convolution_side: usize,
44        fft: &mut PlaneFft<f32>,
45    ) -> Result<Vec<Complex32>, OperatorError> {
46        self.transform_screen(w)?;
47        let mut screen = vec![Complex32::default(); convolution_side * convolution_side];
48        insert_centre(
49            &mut screen,
50            convolution_side,
51            &self.screen,
52            self.side,
53            w_crop_side,
54        );
55        fft.transform(&mut screen, true)?;
56        let normalisation = (convolution_side * convolution_side) as f32;
57        screen.iter_mut().for_each(|value| *value /= normalisation);
58        Ok(screen)
59    }
60
61    pub(super) fn convolve(
62        &mut self,
63        image: &[Complex32],
64        image_side: usize,
65        screen: &[Complex32],
66        side: usize,
67        fft: &mut PlaneFft<f32>,
68    ) -> Result<&[Complex32], OperatorError> {
69        let aperture = &mut self.aperture[..side * side];
70        aperture.fill(Complex32::default());
71        insert_centre(aperture, side, image, image_side, image_side);
72        fft.transform(aperture, true)?;
73        let normalisation = (side * side) as f32;
74        for (beam, phase) in aperture.iter_mut().zip(screen) {
75            *beam = (*beam / normalisation) * phase;
76        }
77        fft.transform(aperture, false)?;
78        Ok(aperture)
79    }
80
81    fn transform_screen(&mut self, w: f64) -> Result<(), OperatorError> {
82        fill_screen(&mut self.screen, self.side, self.cell_rad, w);
83        self.fft.transform(&mut self.screen, false)
84    }
85}

87/// WPConvFunc::makeSkyWFunc, without the spheroidal taper used by WProjectFT.
88fn fill_screen(screen: &mut [Complex32], side: usize, cell_rad: f64, w: f64) {
89    let half = (side / 2) as f64;
90    let two_pi_w = std::f64::consts::TAU * w;
91    for (y, row) in screen.chunks_exact_mut(side).enumerate() {
92        let m = (y as f64 - half) * cell_rad;
93        for (x, value) in row.iter_mut().enumerate() {
94            let l = (x as f64 - half) * cell_rad;
95            let r_squared = l * l + m * m;
96            *value = if r_squared < 1.0 {
97                let phase = two_pi_w * ((1.0 - r_squared).sqrt() - 1.0);
98                let (sin, cos) = phase.sin_cos();
99                Complex32::new(cos as f32, sin as f32)
100            } else {
101                Complex32::default()
102            };
103        }
104    }
105}

107fn screen_support(screen: &[Complex32], side: usize) -> usize {
108    let (minimum, maximum) = screen
109        .iter()
110        .map(|value| value.norm())
111        .fold((f32::INFINITY, 0.0_f32), |(minimum, maximum), amplitude| {
112            (minimum.min(amplitude), maximum.max(amplitude))
113        });
114    let centre = side / 2;
115    (0..centre)
116        .find(|offset| {
117            let sample = screen[(centre + offset) * side + centre].norm();
118            (sample - minimum) / (maximum - minimum) < 0.01
119        })
120        .map_or(centre, |offset| offset + 1)
121}

123pub(super) fn insert_centre(
124    target: &mut [Complex32],
125    target_side: usize,
126    source: &[Complex32],
127    source_side: usize,
128    crop_side: usize,
129) {
130    let source_start = (source_side - crop_side) / 2;
131    let target_start = (target_side - crop_side) / 2;
132    for row in 0..crop_side {
133        let from = (source_start + row) * source_side + source_start;
134        let to = (target_start + row) * target_side + target_start;
135        target[to..to + crop_side].copy_from_slice(&source[from..from + crop_side]);
136    }
137}

139pub(super) struct AwSupport {
140    /// W=0 determines the cutoff and the normalisation of every W cell
141    /// at this frequency (AWConvFunc::supportResizeAWConv).
142    peak: f32,
143    inverse_sum: f32,
144}
145
146impl AwSupport {
147    pub(super) fn new(plane: &[Complex32], side: usize, aperture_support: usize) -> Self {
148        let peak = plane.iter().map(|value| value.norm()).fold(0.0, f32::max);
149        let support = diagonal_support_relative(plane, side, 0.01 * peak)
150            .unwrap_or(5)
151            .max(5)
152            .max(aperture_support);
153        let sum = central_sum(plane, side, support.min(side / 2), false);
154        Self {
155            peak,
156            inverse_sum: if sum > 0.0 { sum.recip() } else { 1.0 },
157        }
158    }
159
160    pub(super) fn half_support(
161        &self,
162        plane: &[Complex32],
163        side: usize,
164        aperture_support: usize,
165    ) -> usize {
166        diagonal_support_relative(plane, side, 0.01 * self.peak)
167            .unwrap_or(5)
168            .max(5)
169            .max(aperture_support)
170    }
171
172    pub(super) fn inverse_sum(&self) -> f32 {
173        self.inverse_sum
174    }
175
176    pub(super) fn normalize(&self, plane: &mut [Complex32]) {
177        plane
178            .iter_mut()
179            .for_each(|value| *value *= self.inverse_sum);
180    }
181}

183#[cfg(test)]
184mod tests {
185    use super::*;
186
187    #[test]
188    fn nonzero_w_support_uses_the_zero_plane_peak_and_sum() {
189        let side = 32;
190        let mut zero = vec![Complex32::default(); side * side];
191        let centre = (side / 2) * (side + 1);
192        zero[centre] = Complex32::new(10.0, 0.0);
193        let normalisation = AwSupport::new(&zero, side, 6);
194        let mut nonzero = vec![Complex32::default(); side * side];
195        nonzero[centre] = Complex32::new(1.0, 0.0);
196        nonzero[centre - 8 * (side + 1)] = Complex32::new(0.12, 0.0);
197        nonzero[centre - 12 * (side + 1)] = Complex32::new(0.09, 0.0);
198        assert_eq!(normalisation.half_support(&nonzero, side, 6), 11);
199        normalisation.normalize(&mut nonzero);
200        assert_eq!(nonzero[centre], Complex32::new(0.1, 0.0));
201    }
202
203    #[test]
204    fn w_screen_uses_the_casa_phase_and_largest_plane_crop() {
205        let w = 8_608.479_602_376_43;
206        let cell_rad = 9.375e-5;
207        let mut term = WTerm::new(1024, cell_rad).unwrap();
208        term.transform_screen(w).unwrap();
209        let centre = term.screen[512 * 1024 + 512];
210        assert!((centre - Complex32::new(499.056_34, -11_472.457)).norm() < 0.02);
211        assert_eq!(screen_support(&term.screen, 1024), 97);
212        assert_eq!(term.convolution_side(53_802.997_514_852_69).unwrap(), 1024);
213    }
