5/// Typed plot payload prepared for one render/export step.
6#[derive(Debug, Clone, PartialEq)]
7pub enum MsPlotPayload {
8    /// Reused `listobs` payload for metadata-oriented presets.
9    ListObs(ListObsPlotPayload),
10    /// Generic scatter payload for raw visibility plots.
11    Scatter(MsScatterPlotPayload),
12    /// Iterated multi-panel scatter payload for raw visibility plots.
13    ScatterGrid(MsScatterGridPayload),
14    /// Multi-plot scatter page payload for stacked/common paired views.
15    ScatterPage(MsScatterPagePayload),
16}
17
18impl MsPlotPayload {}

61/// One grouped scatter series.

78    /// Frequencies in the PlotMS reporting frame, aligned with `points` when present.

80    /// Shared display IDs indexed by PlotMS averaging order.
81    pub plotms_averaged_ids: Option<MsScatterPlotMsAverageIds>,
82}

84/// MeasurementSet identifiers shared by the series of an averaged PlotMS plot.
85#[derive(Debug, Clone, PartialEq)]
86pub enum MsScatterPlotMsAverageIds {
87    /// Scan and field IDs for scalar bins spanning original scans or fields.
88    ScanAndField(Arc<[(i32, i32)]>),
89    /// Output antenna IDs for per-antenna averages.
90    Antenna(Arc<[i32]>),
91    /// Shared `(scan, field, antenna)` IDs when antenna averaging spans scans or fields.
92    ScanFieldAndAntenna(Arc<[(i32, i32, i32)]>),
93    /// Shared spectral coordinates and identifiers for each averaged output row.
94    SpectralWindow {
95        /// Spectral coordinates for each averaged visibility buffer.
96        groups: Arc<[Option<MsScatterPlotMsAverageSpw>]>,
97        /// Buffer index for each output row in PlotMS export order.
98        row_groups: Arc<[usize]>,
99        /// Output antenna for per-antenna rows in the same order, when requested.
100        antenna_ids: Option<Arc<[i32]>>,
101    },
102}

115/// One plotted point's originating visibility sample range.
116#[derive(Debug, Clone, PartialEq, Eq)]
117pub struct MsScatterPointRef {
118    /// MAIN row index.
119    pub row: usize,
120    /// Correlation slot within the row FLAG/DATA cell.
121    pub corr: usize,
122    /// Inclusive channel start for this plotted point.
123    pub chan_start: usize,
124    /// Exclusive channel end for this plotted point.
125    pub chan_end: usize,
126}
127
128impl MsScatterPointRef {
129    pub fn plotms_order(&self, iterator_slot: Option<usize>) -> (usize, usize, usize) {
130        (
131            iterator_slot.unwrap_or(self.row),
132            self.chan_start,
133            self.corr,
134        )
135    }

228pub(super) fn render_scatter_manifest(payload: &MsPlotPayload) -> Result<String, String> {
229    match payload {
230        MsPlotPayload::Scatter(payload) => {
231            let mut out = String::new();
232            out.push_str("# msexplore-manifest-v1\n");
233            out.push_str(&format!("# title={}\n", payload.title));
234            out.push_str(&format!("# x_axis={}\n", payload.x_axis.as_str()));
235            out.push_str(&format!("# y_axis={}\n", payload.y_axis.as_str()));
236            out.push_str(&format!("# legendposition={}\n", payload.legend_position));
237            for line in &payload.header_lines {
238                out.push_str(&format!("# header_line={line}\n"));
239            }
240            if let Some(secondary_y_axis) = payload.secondary_y_axis {
241                out.push_str(&format!(
242                    "# secondary_y_axis={}\n",
243                    secondary_y_axis.as_str()
244                ));
245                out.push_str("series_key\tseries_label\ty_axis\tx\ty\n");
246            } else {
247                out.push_str("series_key\tseries_label\tx\ty\n");
248            }
249            for series in &payload.series {
250                for (x, y) in &series.points {
251                    if payload.secondary_y_axis.is_some() {
252                        out.push_str(&format!(
253                            "{}\t{}\t{}\t{:.12}\t{:.12}\n",
254                            series.color_group,
255                            series.label,
256                            series.y_axis.as_str(),
257                            x,
258                            y
259                        ));
260                    } else {
261                        out.push_str(&format!(
262                            "{}\t{}\t{:.12}\t{:.12}\n",
263                            series.color_group, series.label, x, y
264                        ));
265                    }
266                }
267            }
268            Ok(out)
269        }
270        MsPlotPayload::ScatterGrid(payload) => {
271            let mut out = String::new();
272            out.push_str("# msexplore-manifest-v1\n");
273            out.push_str(&format!("# title={}\n", payload.title));
274            out.push_str(&format!("# x_axis={}\n", payload.x_axis.as_str()));
275            out.push_str(&format!("# y_axis={}\n", payload.y_axis.as_str()));
276            out.push_str(&format!("# iteraxis={}\n", payload.iteraxis.as_str()));
277            out.push_str(&format!("# gridrows={}\n", payload.gridrows));
278            out.push_str(&format!("# gridcols={}\n", payload.gridcols));
279            out.push_str(&format!("# legendposition={}\n", payload.legend_position));
280            out.push_str(&format!("# share_x_bounds={}\n", payload.share_x_bounds));
281            out.push_str(&format!("# share_y_bounds={}\n", payload.share_y_bounds));
282            for line in &payload.header_lines {
283                out.push_str(&format!("# header_line={line}\n"));
284            }
285            out.push_str("panel_key\tpanel_label\tseries_key\tseries_label\tx\ty\n");
286            for panel in &payload.panels {
287                for series in &panel.series {
288                    for (x, y) in &series.points {
289                        out.push_str(&format!(
290                            "{}\t{}\t{}\t{}\t{:.12}\t{:.12}\n",
291                            panel.key, panel.label, series.color_group, series.label, x, y
292                        ));
293                    }
294                }
295            }
296            Ok(out)
297        }
298        MsPlotPayload::ScatterPage(payload) => {
299            let mut out = String::new();
300            out.push_str("# msexplore-manifest-v1\n");
301            out.push_str(&format!("# title={}\n", payload.title));
302            out.push_str(&format!("# exprange={}\n", payload.exprange));
303            out.push_str(&format!("# gridrows={}\n", payload.gridrows));
304            out.push_str(&format!("# gridcols={}\n", payload.gridcols));
305            for line in &payload.header_lines {
306                out.push_str(&format!("# header_line={line}\n"));
307            }
308            out.push_str(
309                "plotindex\trowindex\tcolindex\tplot_title\tx_axis\ty_axis\tseries_key\tseries_label\tx\ty\n",
310            );
311            for item in &payload.items {
312                for series in &item.plot.series {
313                    for (x, y) in &series.points {
314                        out.push_str(&format!(
315                            "{}\t{}\t{}\t{}\t{}\t{}\t{}\t{}\t{:.12}\t{:.12}\n",
316                            item.plotindex,
317                            item.rowindex,
318                            item.colindex,
319                            item.plot.title,
320                            item.plot.x_axis.as_str(),
321                            series.y_axis.as_str(),
322                            series.color_group,
323                            series.label,
324                            x,
325                            y
326                        ));
327                    }
328                }
329            }
330            Ok(out)
331        }
332        MsPlotPayload::ListObs(_) => {
333            Err("text manifest export requires a scatter payload".to_string())
334        }
335    }
336}

338pub(super) fn render_listobs_manifest(payload: &ListObsPlotPayload) -> Result<String, String> {
339    match payload {
340        ListObsPlotPayload::UvCoverage(payload) => {
341            let mut out = String::new();
342            out.push_str("# msexplore-manifest-v1\n");
343            out.push_str("# payload=listobs_uv_coverage\n");
344            out.push_str("# title=UV Coverage\n");
345            out.push_str("# x_axis=u_lambda\n");
346            out.push_str("# y_axis=v_lambda\n");
347            out.push_str(&format!("# mirror={}\n", payload.mirror));
348            out.push_str(&format!(
349                "# axis_extent_lambda={:.12}\n",
350                payload.axis_extent_lambda
351            ));
352            out.push_str(&format!("# tracks={}\n", payload.tracks.len()));
353            out.push_str(&format!("# summary={}\n", payload.summary));
354            out.push_str("series_key\tseries_label\tmirrored\tx\ty\n");
355            for (track_index, track) in payload.tracks.iter().enumerate() {
356                let series_key = format!("track-{track_index}");
357                for (x, y) in &track.points {
358                    out.push_str(&format!(
359                        "{}\t{}\tfalse\t{:.12}\t{:.12}\n",
360                        series_key, track.label, x, y
361                    ));
362                    if payload.mirror {
363                        out.push_str(&format!(
364                            "{}\t{}\ttrue\t{:.12}\t{:.12}\n",
365                            series_key, track.label, -x, -y
366                        ));
367                    }
368                }
369            }
370            Ok(out)
371        }
372        _ => Err("text export is currently available for UV coverage and raw msexplore scatter plots only".to_string()),
373    }
