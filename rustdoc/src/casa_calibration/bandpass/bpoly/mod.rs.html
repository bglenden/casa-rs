37pub(super) fn modern_gather(
38    ms: &MeasurementSet,
39    request: &BandpassSolveRequest,
40    context: CalibrationSolveContext,
41    groups: impl Iterator<Item = SolveAccumulator>,
42    activity: &mut ToolSolveActivity,
43) -> Result<BandpassSolveReport, BandpassSolveError> {
44    let mut stats = SolveStatsBuilder::new(
45        2,
46        crate::solve::spectral_window_ids(ms)?,
47        &context.available_antennas,
48    );
49    for group in groups {
50        if group.gather_attempted() {
51            return Err(BandpassSolveError::UnsupportedConfiguration {
52                reason: "the modern visibility iterator cannot fit BPOLY".into(),
53            });
54        }
55        stats.interval(&IntervalStages::of(&group, 0));
56    }
57    stats.track_activity(Some(activity));
58    Ok(empty_report(
59        request,
60        context.refant_id,
61        context.selected_data,
62        stats.finish(),
63    ))
64}

66pub(super) fn solve_polynomial(
67    ms: &MeasurementSet,
68    request: &BandpassSolveRequest,
69    context: CalibrationSolveContext,
70    groups: impl Iterator<Item = SolveAccumulator>,
71    bin_widths: &BTreeMap<i32, usize>,
72    mut activity: Option<&mut ToolSolveActivity>,
73    mode: crate::output::CalibrationOutputMode,
74) -> Result<BandpassSolveReport, BandpassSolveError> {
75    let cuts = request
76        .cuts
77        .expect("polynomial solver with configured cuts");
78    let window = ms
79        .spectral_window()
80        .map_err(|source| BandpassSolveError::OpenMeasurementSet {
81            path: ms
82                .path()
83                .map(|path| path.display().to_string())
84                .unwrap_or_default(),
85            source,
86        })?;
87    let descriptions = context
88        .rows
89        .iter()
90        .map(|row| (row.data_desc_id, row))
91        .collect::<BTreeMap<_, _>>();
92    let mut frequencies_by_description = BTreeMap::new();
93    for (id, row) in descriptions {
94        let spw = row.data_spw_id;
95        let frequencies = window.chan_freq(spw as usize).map_err(|source| {
96            BandpassSolveError::OpenMeasurementSet {
97                path: ms
98                    .path()
99                    .map(|path| path.display().to_string())
100                    .unwrap_or_default(),
101                source,
102            }
103        })?;
104        let (channels, _) = crate::solve::grouping::polynomial::channel_selection(
105            row.channel_selection.as_ref(),
106            frequencies.len(),
107        )?;
108        let spacing = frequencies
109            .windows(2)
110            .next()
111            .map_or(1.0, |pair| (pair[1] - pair[0]).abs());
112        frequencies_by_description.insert(
113            id,
114            (
115                super::metadata::average_frequencies(
116                    &frequencies,
117                    &channels,
118                    bin_widths.get(&spw).copied().unwrap_or(1),
119                ),
120                spacing,
121            ),
122        );
123    }
124    let combined_window = request.combine.spectral_windows.then(|| {
125        context
126            .rows
127            .iter()
128            .map(|row| row.data_spw_id)
129            .min()
130            .expect("selected windows")
131    });
132    if combined_window.is_some()
133        && frequencies_by_description
134            .values()
135            .map(|(channels, _)| channels.len())
136            .min()
137            != frequencies_by_description
138                .values()
139                .map(|(channels, _)| channels.len())
140                .max()
141    {
142        if activity.is_some() {
143            return Err(BandpassSolveError::UnsupportedConfiguration {
144                reason:
145                    "spectral windows with different selected channel counts cannot be combined"
146                        .into(),
147            });
148        }
149        tracing::warn!(
150            "Spectral windows with different selected channel counts cannot be combined"
151        );
152        return Ok(empty_report(
153            request,
154            context.refant_id,
155            context.selected_data,
156            SolveStats::default(),
157        ));
158    }
159    if let Some(activity) = activity.as_deref_mut() {
160        activity.initialize(window.row_count());
161    }
162    let mut channel_count = 0;
163    let mut fitted = Vec::new();
164    for group in groups {
165        let Some(polynomial) = &group.polynomial else {
166            continue;
167        };
168        let frequencies = polynomial
169            .descriptions()
170            .iter()
171            .map(|id| {
172                let (frequencies, spacing) = &frequencies_by_description[id];
173                (*id, (frequencies.as_slice(), *spacing))
174            })
175            .collect::<BTreeMap<_, _>>();
176        let output_window = combined_window.unwrap_or(group.spw_id);
177        if frequencies.is_empty() {
178            if let Some(activity) = activity.as_deref_mut() {
179                activity.count_mapped_interval(group.spw_id, output_window, false, false);
180            }
181            continue;
182        }
183        let mut rows = match fit::fit_interval(
184            &group,
185            &frequencies,
186            &context.available_antennas,
187            context.refant_id,
188            request,
189            cuts,
190        ) {
191            Ok(rows) => rows,
192            Err(error @ BandpassSolveError::FitPolynomial { .. }) if activity.is_none() => {
193                tracing::warn!("{error}");
194                break;
195            }
196            Err(error) => return Err(error),
197        };
198        if let Some(activity) = activity.as_deref_mut() {
199            activity.count_mapped_interval(group.spw_id, output_window, true, !rows.is_empty());
200        }
201        if !rows.is_empty() {
202            if let Some(spw) = combined_window {
203                for row in &mut rows {
204                    row.spw_id = spw;
205                }
206            }
207            channel_count = frequencies
208                .values()
209                .map(|(channels, _)| channels.len())
210                .max()
211                .unwrap_or(0);
212            fitted.extend(rows);
213        }
214    }
215    if fitted.is_empty() {
216        return Ok(empty_report(
217            request,
218            context.refant_id,
219            context.selected_data,
220            SolveStats::default(),
221        ));
222    }
223    let report =
224        write_polynomial_rows(ms, request, context.refant_id, &fitted, channel_count, mode)?;
225    Ok(BandpassSolveReport {
226        selected_data: context.selected_data,
227        ..report
228    })
229}

231pub(super) fn write_bpoly_caltable(
232    ms: &MeasurementSet,
233    request: &BandpassSolveRequest,
234    refant_id: i32,
235    rows: &[BandpassSolutionRow],
236    mode: crate::output::CalibrationOutputMode,
237) -> Result<BandpassSolveReport, BandpassSolveError> {
238    let fitted_rows = fit_bpoly_rows(
239        ms,
240        rows,
241        request.amplitude_degree,
242        request.phase_degree,
243        request.cuts,
244        &request.selection,
245    )?;
246    let channel_count = rows.first().map(|row| row.channel_count).unwrap_or(0);
247    write_polynomial_rows(ms, request, refant_id, &fitted_rows, channel_count, mode)
248}

250#[expect(
251    clippy::too_many_lines,

253)]
254fn write_polynomial_rows(
255    ms: &MeasurementSet,
256    request: &BandpassSolveRequest,
257    refant_id: i32,
258    fitted_rows: &[BPolySolutionRow],
259    channel_count: usize,
260    mode: crate::output::CalibrationOutputMode,
261) -> Result<BandpassSolveReport, BandpassSolveError> {
262    // Existing casa-rs BPOLY tables used rank-one placeholder arrays and
263    // two extra diagnostic columns. Keep that layout when appending to one.
264    let casa_layout = mode != crate::output::CalibrationOutputMode::Append
265        || !request.output_table.exists()
266        || !Table::open(TableOptions::new(&request.output_table))
267            .map_err(|source| BandpassSolveError::SaveCalibrationTable {
268                path: request.output_table.display().to_string(),
269                source: Box::new(source),
270            })?
271            .schema()
272            .is_some_and(|schema| schema.contains_column(COL_PARAMERR));
273    let columns = legacy_polynomial_columns(casa_layout);
274    let schema = TableSchema::new(columns).expect("valid BPOLY schema");
275    let mut writer = CalibrationTableWriter::with_mode(
276        ms,
277        CalibrationTableDescriptor {
278            output: &request.output_table,
279            schema,
280            subtype: "BPOLY",
281            parameter_type: None,
282            measurement_set_name: ms
283                .path()
284                .map(|path| path.display().to_string())
285                .unwrap_or_else(|| "<in-memory>".to_string()),
286            include_polarization_basis: false,
287            time_extra_precision_column: Some(COL_TIME_EXTRA_PREC),
288        },
289        mode,
290    )?;
291    {
292        let table = writer.table_mut();
293        table
294            .keywords_mut()
295            .upsert(LEGACY_CAL_DESC_KEYWORD, Value::table_ref("././CAL_DESC"));
296        table.keywords_mut().upsert(
297            LEGACY_CAL_HISTORY_KEYWORD,
298            Value::table_ref("././CAL_HISTORY"),
299        );
300        set_fixed_unit_keyword(table, COL_VALID_DOMAIN, &["Hz"]);
301        set_fixed_unit_keyword(table, "REF_FREQUENCY", &["Hz"]);
302        set_measinfo_keyword(
303            table,
304            "REF_FREQUENCY",
305            "frequency",
306            Some(if casa_layout { "LSRK" } else { "TOPO" }),
307        );
308        set_fixed_unit_keyword(table, "REF_DIRECTION", &["rad", "rad"]);
309        set_measinfo_keyword(table, "REF_DIRECTION", "direction", Some("J2000"));
310    }
311
312    let cal_desc_ids = fitted_rows
313        .iter()
314        .map(|row| row.spw_id)
315        .collect::<BTreeSet<_>>()
316        .into_iter()
317        .enumerate()
318        .map(|(cal_desc_id, spw_id)| {
319            (
320                spw_id,
321                i32::try_from(cal_desc_id).expect("small BPOLY CAL_DESC id set"),
322            )
323        })
324        .collect::<BTreeMap<_, _>>();
325
326    let window_names = ms
327        .spectral_window()
328        .ok()
329        .map(|window| {
330            cal_desc_ids
331                .keys()
332                .map(|&id| (id, window.freq_group_name(id as usize).unwrap_or_default()))
333                .collect::<BTreeMap<_, _>>()
334        })
335        .unwrap_or_default();
336    let unit = Complex32::new(1.0, 0.0);
337    let (gain_shape, reference_shape, direction_shape, diagnostic_shape): (
338        &[usize],
339        &[usize],
340        &[usize],
341        &[usize],
342    ) = if casa_layout {
343        (&[1, 1, 1, 1], &[1, 1, 1], &[2, 1, 1, 1], &[1, 1, 1, 1])
344    } else {
345        (&[1], &[1], &[2], &[2])
346    };
347    for row in fitted_rows {
348        let from_casa_task = request.cuts.is_some();
349        let status_weight = if casa_layout { 0.0 } else { 1.0 };
350        let unset = if casa_layout { -1 } else { 0 };
351        let reference_frequency_hz = (row.valid_domain_hz[0] + row.valid_domain_hz[1]) / 2.0;
352        let flat_amp = row
353            .amp_coefficients
354            .iter()
355            .flat_map(|coefficients| coefficients.iter().copied())
356            .collect::<Vec<_>>();
357        let flat_phase = row
358            .phase_coefficients
359            .iter()
360            .flat_map(|coefficients| coefficients.iter().copied())
361            .collect::<Vec<_>>();
362        let amp_shape = IxDyn(&[1, 1, 1, flat_amp.len()]).f();
363        let phase_shape = IxDyn(&[1, 1, 1, flat_phase.len()]).f();
364        let mut fields = vec![
365            RecordField::new(
366                COL_TIME,
367                Value::Scalar(ScalarValue::Float64(row.time_seconds)),
368            ),
369            RecordField::new(
370                COL_TIME_EXTRA_PREC,
371                Value::Scalar(ScalarValue::Float64(0.0)),
372            ),
373            RecordField::new(
374                COL_FIELD_ID,
375                Value::Scalar(ScalarValue::Int32(row.field_id)),
376            ),
377            RecordField::new(
378                COL_ANTENNA1,
379                Value::Scalar(ScalarValue::Int32(row.antenna_id)),
380            ),
381            RecordField::new("FEED1", Value::Scalar(ScalarValue::Int32(unset))),
382            RecordField::new(
383                COL_INTERVAL,
384                Value::Scalar(ScalarValue::Float64(row.interval_seconds)),
385            ),
386            RecordField::new(
387                COL_SCAN_NUMBER,
388                Value::Scalar(ScalarValue::Int32(row.scan_number)),
389            ),
390            RecordField::new(
391                COL_OBSERVATION_ID,
392                Value::Scalar(ScalarValue::Int32(row.observation_id)),
393            ),
394            RecordField::new(COL_ARRAY_ID, Value::Scalar(ScalarValue::Int32(unset))),
395            RecordField::new("PROCESSOR_ID", Value::Scalar(ScalarValue::Int32(-1))),
396            RecordField::new("STATE_ID", Value::Scalar(ScalarValue::Int32(-1))),
397            RecordField::new("PHASE_ID", Value::Scalar(ScalarValue::Int32(-1))),
398            RecordField::new("PULSAR_BIN", Value::Scalar(ScalarValue::Int32(-1))),
399            RecordField::new("PULSAR_GATE_ID", Value::Scalar(ScalarValue::Int32(-1))),
400            RecordField::new("FREQ_GROUP", Value::Scalar(ScalarValue::Int32(unset))),
401            RecordField::new(
402                "FREQ_GROUP_NAME",
403                Value::Scalar(ScalarValue::String(
404                    window_names.get(&row.spw_id).cloned().unwrap_or_default(),
405                )),
406            ),
407            RecordField::new(
408                "FIELD_NAME",
409                Value::Scalar(ScalarValue::String(String::new())),
410            ),
411            RecordField::new(
412                "FIELD_CODE",
413                Value::Scalar(ScalarValue::String(String::new())),
414            ),
415            RecordField::new(
416                "SOURCE_NAME",
417                Value::Scalar(ScalarValue::String(String::new())),
418            ),
419            RecordField::new(
420                "SOURCE_CODE",
421                Value::Scalar(ScalarValue::String(String::new())),
422            ),
423            RecordField::new(
424                "CALIBRATION_GROUP",
425                Value::Scalar(ScalarValue::Int32(unset)),
426            ),
427            RecordField::new(
428                COL_CAL_DESC_ID,
429                Value::Scalar(ScalarValue::Int32(
430                    *cal_desc_ids
431                        .get(&row.spw_id)
432                        .expect("BPOLY CAL_DESC entry for solved SPW"),
433                )),
434            ),
435            RecordField::new("CAL_HISTORY_ID", Value::Scalar(ScalarValue::Int32(unset))),
436            RecordField::new(
437                "GAIN",
438                Value::Array(ArrayValue::Complex32(ArrayD::from_elem(
439                    IxDyn(gain_shape).f(),
440                    if casa_layout {
441                        Complex32::default()
442                    } else {
443                        unit
444                    },
445                ))),
446            ),
447            RecordField::new(
448                "SIDEBAND_REF",
449                Value::Scalar(ScalarValue::Complex32(Complex32::new(1.0, 0.0))),
450            ),
451            RecordField::new(
452                "REF_ANT",
453                Value::Array(ArrayValue::Int32(ArrayD::from_elem(
454                    IxDyn(reference_shape).f(),
455                    if from_casa_task || casa_layout {
456                        refant_id
457                    } else {
458                        row.antenna_id
459                    },
460                ))),
461            ),
462            RecordField::new(
463                "REF_FEED",
464                Value::Array(ArrayValue::Int32(ArrayD::zeros(IxDyn(reference_shape).f()))),
465            ),
466            RecordField::new(
467                "REF_RECEPTOR",
468                Value::Array(ArrayValue::Int32(ArrayD::zeros(IxDyn(reference_shape).f()))),
469            ),
470            RecordField::new(
471                "REF_FREQUENCY",
472                Value::Array(ArrayValue::Float64(ArrayD::from_elem(
473                    IxDyn(reference_shape).f(),
474                    reference_frequency_hz,
475                ))),
476            ),
477            RecordField::new(
478                "MEAS_FREQ_REF",
479                Value::Scalar(ScalarValue::Int32(i32::from(casa_layout))),
480            ),
481            RecordField::new(
482                "REF_DIRECTION",
483                Value::Array(ArrayValue::Float64(ArrayD::zeros(
484                    IxDyn(direction_shape).f(),
485                ))),
486            ),
487            RecordField::new("MEAS_DIR_REF", Value::Scalar(ScalarValue::Int32(0))),
488            RecordField::new(
489                "TOTAL_SOLUTION_OK",
490                Value::Scalar(ScalarValue::Bool(!casa_layout)),
491            ),
492            RecordField::new("TOTAL_FIT", Value::Scalar(ScalarValue::Float32(0.0))),
493            RecordField::new(
494                "TOTAL_FIT_WEIGHT",
495                Value::Scalar(ScalarValue::Float32(status_weight)),
496            ),
497            RecordField::new(
498                "SOLUTION_OK",
499                Value::Array(ArrayValue::Bool(ArrayD::from_elem(
500                    IxDyn(diagnostic_shape).f(),
501                    !casa_layout,
502                ))),
503            ),
504            RecordField::new(
505                "FIT",
506                Value::Array(ArrayValue::Float32(ArrayD::zeros(
507                    IxDyn(diagnostic_shape).f(),
508                ))),
509            ),
510            RecordField::new(
511                "FIT_WEIGHT",
512                Value::Array(ArrayValue::Float32(ArrayD::from_elem(
513                    IxDyn(diagnostic_shape).f(),
514                    status_weight,
515                ))),
516            ),
517            RecordField::new(
518                COL_FLAG,
519                Value::Array(ArrayValue::Bool(ArrayD::from_elem(
520                    IxDyn(diagnostic_shape).f(),
521                    casa_layout,
522                ))),
523            ),
524            RecordField::new(
525                COL_PARAMERR,
526                Value::Array(ArrayValue::from_f32_vec(vec![0.0, 0.0])),
527            ),
528            RecordField::new(
529                COL_SNR,
530                Value::Array(ArrayValue::Float32(ArrayD::from_elem(
531                    IxDyn(diagnostic_shape).f(),
532                    status_weight,
533                ))),
534            ),
535            RecordField::new(
536                COL_WEIGHT,
537                Value::Array(ArrayValue::from_f32_vec(vec![1.0, 1.0])),
538            ),
539            RecordField::new(
540                "POLY_TYPE",
541                Value::Scalar(ScalarValue::String("CHEBYSHEV".to_string())),
542            ),
543            RecordField::new(
544                "POLY_MODE",
545                Value::Scalar(ScalarValue::String("A&P".to_string())),
546            ),
547            RecordField::new(
548                COL_SCALE_FACTOR,
549                Value::Scalar(ScalarValue::Complex32(row.scale_factor)),
550            ),
551            RecordField::new(
552                COL_N_POLY_AMP,
553                Value::Scalar(ScalarValue::Int32(
554                    i32::try_from(row.amp_coefficients.first().map_or(0, Vec::len))
555                        .expect("small BPOLY amp coefficient count"),
556                )),
557            ),
558            RecordField::new(
559                COL_N_POLY_PHASE,
560                Value::Scalar(ScalarValue::Int32(
561                    i32::try_from(row.phase_coefficients.first().map_or(0, Vec::len))
562                        .expect("small BPOLY phase coefficient count"),
563                )),
564            ),
565            RecordField::new(
566                COL_PHASE_UNITS,
567                Value::Scalar(ScalarValue::String(
568                    if from_casa_task || casa_layout {
569                        "RADIANS"
570                    } else {
571                        "RAD"
572                    }
573                    .to_string(),
574                )),
575            ),
576            RecordField::new(
577                COL_VALID_DOMAIN,
578                Value::Array(ArrayValue::Float64(
579                    ArrayD::from_shape_vec(
580                        IxDyn(&[2]).f(),
581                        vec![row.valid_domain_hz[0], row.valid_domain_hz[1]],
582                    )
583                    .expect("BPOLY valid domain should reshape"),
584                )),
585            ),
586            RecordField::new(
587                COL_POLY_COEFF_AMP,
588                Value::Array(ArrayValue::Float64(
589                    ArrayD::from_shape_vec(amp_shape, flat_amp)
590                        .expect("BPOLY amp coefficients should reshape"),
591                )),
592            ),
593            RecordField::new(
594                COL_POLY_COEFF_PHASE,
595                Value::Array(ArrayValue::Float64(
596                    ArrayD::from_shape_vec(phase_shape, flat_phase)
597                        .expect("BPOLY phase coefficients should reshape"),
598                )),
599            ),
600        ];
601        if casa_layout {
602            fields.retain(|field| !matches!(field.name.as_str(), COL_PARAMERR | COL_WEIGHT));
603        }
604        writer.append(RecordValue::new(fields))?;
605    }
606
607    writer.copy_subtables(&[
608        (SubtableId::Observation, "OBSERVATION"),
609        (SubtableId::Antenna, "ANTENNA"),
610        (SubtableId::Field, "FIELD"),
611        (SubtableId::History, "HISTORY"),
612        (SubtableId::SpectralWindow, "SPECTRAL_WINDOW"),
613    ])?;
614
615    let descriptions = writer.output_path().join("CAL_DESC");
616    write_legacy_cal_desc(
617        descriptions.clone(),
618        &cal_desc_ids,
619        |spw| {
620            fitted_rows
621                .iter()
622                .find(|row| row.spw_id == spw)
623                .map_or(1, |row| row.amp_coefficients.len() as i32)
624        },
625        casa_layout,
626        ms.path()
627            .and_then(|path| path.file_name())
628            .and_then(|name| name.to_str())
629            .unwrap_or_default(),
630    )
631    .map_err(|source| BandpassSolveError::SaveCalibrationTable {
632        path: descriptions.display().to_string(),
633        source: Box::new(source),
634    })?;
635    let history = writer.output_path().join("CAL_HISTORY");
636    write_legacy_history(&history).map_err(|source| BandpassSolveError::SaveCalibrationTable {
637        path: history.display().to_string(),
638        source: Box::new(source),
639    })?;
640    writer.save_main()?;
641
642    let field_ids = fitted_rows
643        .iter()
644        .map(|row| row.field_id)
645        .collect::<BTreeSet<_>>();
646    let spw_ids = fitted_rows
647        .iter()
648        .map(|row| row.spw_id)
649        .collect::<BTreeSet<_>>();
650    Ok(BandpassSolveReport {
651        output_table: request.output_table.clone(),
652        table_subtype: "BPOLY".to_string(),
653        refant_antenna_id: refant_id,
654        field_ids: field_ids.iter().copied().collect(),
655        spectral_window_ids: spw_ids.iter().copied().collect(),
656        solution_row_count: fitted_rows.len(),

747fn fit_bpoly_rows(
748    ms: &MeasurementSet,
749    rows: &[BandpassSolutionRow],
750    amplitude_degree: usize,
751    phase_degree: usize,
752    cuts: Option<BandpassSolveCuts>,
753    selection: &casa_ms::MsSelection,
754) -> Result<Vec<BPolySolutionRow>, BandpassSolveError> {
755    let spectral_window =
756        ms.spectral_window()
757            .map_err(|source| BandpassSolveError::OpenMeasurementSet {
758                path: ms
759                    .path()
760                    .map(|path| path.display().to_string())
761                    .unwrap_or_else(|| "<in-memory>".to_string()),
762                source,
763            })?;
764    rows.iter()
765        .map(|row| {
766            let spw_row = usize::try_from(row.spw_id).map_err(|_| {
767                BandpassSolveError::UnsupportedParameterShape {
768                    path: "SPECTRAL_WINDOW row id".to_string(),
769                    shape: vec![row.spw_id as usize],
770                }
771            })?;
772            let channel_frequencies_hz = spectral_window.chan_freq(spw_row).map_err(|source| {
773                BandpassSolveError::OpenMeasurementSet {
774                    path: ms
775                        .path()
776                        .map(|path| path.display().to_string())
777                        .unwrap_or_else(|| "<in-memory>".to_string()),
778                    source,
779                }
780            })?;
781            let selected = selection.channel_selection_for_spw(row.spw_id);
782            let channel_frequencies_hz = crate::solve::grouping::selected_channels(
783                selected.as_ref(),
784                channel_frequencies_hz.len(),
785            )?
786            .into_iter()
787            .map(|channel| channel_frequencies_hz[channel])
788            .collect::<Vec<_>>();
789            fit_bpoly_row(
790                row,
791                &channel_frequencies_hz,
792                [amplitude_degree, phase_degree],
793                cuts,
794            )
795        })
796        .collect()
797}
798
799/// `row`'s channel solutions fitted with `BPOLY`'s polynomials over the
800/// window, of `[amplitude, phase]` degrees: each receptor's log amplitudes
801/// and unwrapped phases at its unflagged channels, leaving out the edges
802/// `cuts` masks. A receptor with no such channel keeps a unit gain.
803pub(super) fn fit_bpoly_row(
804    row: &BandpassSolutionRow,
805    channel_frequencies_hz: &[f64],
806    degrees: [usize; 2],
807    cuts: Option<BandpassSolveCuts>,
808) -> Result<BPolySolutionRow, BandpassSolveError> {
809    if channel_frequencies_hz.len() != row.channel_count || row.channel_count == 0 {
810        return Err(BandpassSolveError::UnsupportedParameterShape {
811            path: "BPOLY fit channel grid".to_string(),
812            shape: vec![channel_frequencies_hz.len(), row.channel_count],
813        });
814    }
815    let valid_domain_hz = [
816        channel_frequencies_hz.first().copied().unwrap_or(0.0),
817        channel_frequencies_hz.last().copied().unwrap_or(0.0),
818    ];
819    // Both receptors take as many coefficients, as the table holds one count.
820    let [amplitude_count, phase_count] =
821        degrees.map(|degree| degree.saturating_add(1).min(row.channel_count));
822    let receptor_count = 2;
823    let mut amp_coefficients = Vec::with_capacity(receptor_count);
824    let mut phase_coefficients = Vec::with_capacity(receptor_count);
825    for receptor in 0..receptor_count {
826        let mut frequencies_hz = Vec::with_capacity(row.channel_count);
827        let mut log_amplitudes = Vec::with_capacity(row.channel_count);
828        let mut phases = Vec::with_capacity(row.channel_count);
829        for (channel, frequency_hz) in channel_frequencies_hz.iter().copied().enumerate() {
830            let index = receptor + receptor_count * channel;
831            let masked =
832                cuts.is_some_and(|cuts| cuts.masks_edge_channel(channel, row.channel_count));
833            if row.flags[index] || masked {
834                continue;
835            }
836            let gain = row.gains[index];
837            frequencies_hz.push(frequency_hz);
838            log_amplitudes.push(gain.norm().max(f32::EPSILON).ln() as f64);
839            phases.push((gain.im as f64).atan2(gain.re as f64));
840        }
841        let unwrapped_phases = unwrap_phases(&phases);
842        amp_coefficients.push(fit_legacy_bpoly_coefficients(
843            &frequencies_hz,
844            &log_amplitudes,
845            valid_domain_hz,
846            amplitude_count,
847            row,
848        )?);
849        phase_coefficients.push(fit_legacy_bpoly_coefficients(
850            &frequencies_hz,
851            &unwrapped_phases,
852            valid_domain_hz,
853            phase_count,
854            row,
855        )?);
856    }
857    Ok(BPolySolutionRow {
858        time_seconds: row.time_seconds,
859        interval_seconds: row.interval_seconds,
860        field_id: row.field_id,
861        spw_id: row.spw_id,
862        antenna_id: row.antenna_id,
863        valid_domain_hz,
864        scan_number: row.scan_number,
865        observation_id: row.observation_id,
866        scale_factor: Complex32::new(1.0, 0.0),
867        amp_coefficients,
868        phase_coefficients,
869    })
870}

872/// `coefficient_count` coefficients of the polynomial through `values` at
873/// `frequencies_hz`. Fewer points than coefficients fit as many
874/// coefficients as there are points, and the rest are zero.
875fn fit_legacy_bpoly_coefficients(
876    frequencies_hz: &[f64],
877    values: &[f64],
878    valid_domain_hz: [f64; 2],
879    coefficient_count: usize,
880    row: &BandpassSolutionRow,
881) -> Result<Vec<f64>, BandpassSolveError> {
882    let mut coefficients = vec![0.0_f64; coefficient_count];
883    let fitted = coefficient_count.min(frequencies_hz.len());
884    if fitted == 0 {
885        return Ok(coefficients);
886    }
887
888    let mut basis = vec![0.0_f64; fitted];
889    let mut rows = Vec::with_capacity(frequencies_hz.len());
890    for (frequency_hz, value) in frequencies_hz.iter().copied().zip(values.iter().copied()) {
891        legacy_bpoly_basis(
892            valid_domain_hz[0],
893            valid_domain_hz[1],
894            frequency_hz,
895            &mut basis,
896        );
897        rows.push((basis.clone(), value, 1.0));
898    }
899    let solved = solve_weighted_least_squares(&rows, fitted).ok_or_else(|| {
900        BandpassSolveError::FitPolynomial {
901            antenna_id: row.antenna_id,
902            field_id: row.field_id,
903            spw_id: row.spw_id,
904            reason: "least-squares system was singular".to_string(),
905        }
906    })?;
907    coefficients[..fitted].copy_from_slice(&solved);
908    Ok(coefficients)
909}

911fn legacy_bpoly_basis(x_start: f64, x_end: f64, x: f64, basis: &mut [f64]) {
912    if basis.is_empty() {
913        return;
914    }
915    basis.fill(0.0);
916    basis[0] = 0.5_f64;
917    if basis.len() == 1 || (x_end - x_start).abs() <= f64::EPSILON {
918        return;
919    }
920    let xcap = ((x - x_start) - (x_end - x)) / (x_end - x_start);
921    let mut t_prev = 1.0_f64;
922    let mut t_curr = xcap;
923    basis[1] = t_curr;
924    for slot in basis.iter_mut().skip(2) {
925        let t_next = 2.0_f64 * xcap * t_curr - t_prev;
926        *slot = t_next;
927        t_prev = t_curr;
928        t_curr = t_next;
929    }
930}
931
932fn unwrap_phases(phases: &[f64]) -> Vec<f64> {
933    let mut unwrapped = Vec::with_capacity(phases.len());
934    let mut offset = 0.0_f64;
935    let mut previous = None;
936    for phase in phases.iter().copied() {
937        if let Some(previous_phase) = previous {
938            let delta = phase - previous_phase;
939            if delta > std::f64::consts::PI {
940                offset -= 2.0_f64 * std::f64::consts::PI;
941            } else if delta < -std::f64::consts::PI {
942                offset += 2.0_f64 * std::f64::consts::PI;
943            }
944        }
945        let adjusted = phase + offset;
946        unwrapped.push(adjusted);
947        previous = Some(phase);
948    }
949    unwrapped
950}
