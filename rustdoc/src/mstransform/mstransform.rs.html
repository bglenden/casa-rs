20fn main() -> ExitCode {
21    TaskFrontEnd::new_with_deferred_schema(
22        "mstransform",
23        mstransform_task_schema_bundle(),
24        &["mstransform", "split"],
25        run,
26    )
27    .expect("the transform contract carries mstransform and split")

30        CasaLoggingConfig::from_env(),
31        std::env::set_current_dir,
32        &mut std::io::stdout(),
33        std::io::stderr(),
34    )
35}

37/// Both surfaces run the one transform; CASA's tasks return None.
38fn run(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
39    let (request, options) = transform_request(task)?;
40    // The output the transform refuses before it starts is the argument's
41    // fault; anything else failed the transform.
42    let report = mstransform_task(&request, &options).map_err(|error| match error {
43        MsTransformError::OutputExists { .. } | MsTransformError::SameInputOutput { .. } => {
44            TaskFailure::validation(error.to_string()).at("outputvis")
45        }
46        MsTransformError::InvalidSpw { .. } => TaskFailure::validation(error.to_string()).at("spw"),
47        MsTransformError::MissingDataColumn { path, column } => TaskFailure::execution(format!(
48            "Desired column ({column}) not found in the input MS ({path})."
49        )),
50        MsTransformError::ChannelWidthCount => {
51            let parameter = channel_parameter(task);
52            TaskFailure::validation(format!(
53                "Number of {parameter} is different from the number of spw"
54            ))
55            .at(parameter)
56        }
57        other => TaskFailure::execution(other.to_string()),
58    })?;
59    ms_task_success(CasaValue::None, &report)
60}

62fn transform_request(
63    task: &TaskInvocation<'_>,
64) -> Result<(MsTransformRequest, MsTransformTaskOptions), TaskFailure> {
65    let (width, channel_widths) = channel_width(task)?;
66    let request = MsTransformRequest {
67        input_ms: task.path("vis")?,
68        output_ms: task.path("outputvis")?,
69        spw: task_selection_text(task, "spw")?,
70        width,
71        data_column: data_column(&task.required_text("datacolumn")?)?,
72        selection: selection(task)?,
73        keep_flags: task.flag("keepflags")?,
74    };
75    let options = MsTransformTaskOptions {
76        use_weight_spectrum: task.surface() == "mstransform" && task.flag("usewtspectrum")?,
77        channel_widths,
78        time_averaging: time_average_options(task)?,
79    };
80    Ok((request, options))
81}

83fn time_average_options(
84    task: &TaskInvocation<'_>,
85) -> Result<Option<TimeAverageOptions>, TaskFailure> {
86    let split = task.surface() == "split";
87    if !split && !task.flag("timeaverage")? {
88        return Ok(None);
89    }
90    let interval = task.required_text("timebin")?;
91    let quantity = Quantity::read(&interval).map_err(|_| {
92        TaskFailure::execution(format!(
93            "Error in QuantumHolder::fromString with input string {interval:?}: Illegal input units or format"
94        )).at("timebin")
95    })?;
96    let seconds = Unit::new("s").expect("seconds are a defined unit");
97    let radians = Unit::new("rad").expect("radians are a defined unit");
98    let interval_seconds = quantity
99        .get_value_in(&seconds)
100        .or_else(|_| {
101            quantity
102                .get_value_in(&radians)
103                .map(|angle| angle * 43_200.0 / std::f64::consts::PI)
104        })
105        .unwrap_or_else(|_| quantity.get_si_value());
106    if interval_seconds <= 0.0 || !interval_seconds.is_finite() {
107        return if split {
108            Ok(None)
109        } else {
110            Err(
111                TaskFailure::validation("Parameter timebin must be > '0s' to do time averaging")
112                    .at("timebin"),
113            )
114        };
115    }
116    let parameter = if split { "combine" } else { "timespan" };
117    let span = match task.value(parameter) {
118        None => "",
119        Some(ParameterValue::String(span)) => span,
120        Some(_) => return Err(TaskFailure::execution(
121            "RecordRep::get_pointer - incorrect data type String   used for field timespan with type unknown",
122        ).at(parameter)),
123    };
124    Ok(Some(TimeAverageOptions {
125        interval_seconds,
126        span_scans: span.contains("scan"),
127        span_states: span.contains("state"),
128        span_fields: span.contains("field"),
129        max_uvw_distance: if split {
130            0.0
131        } else {
132            task.number("maxuvwdistance")?
133        },
134    }))
135}

140    } else {
141        "chanbin"

145fn channel_width(task: &TaskInvocation<'_>) -> Result<(usize, Option<Vec<usize>>), TaskFailure> {
146    let name = channel_parameter(task);
147    let split = name == "width";
148    if !split && !task.flag("chanaverage")? {
149        return Ok((1, None));
150    }
151    let invalid_type = || TaskFailure::validation(format!("invalid {name} type")).at(name);
152    let empty = || {
153        let message = if split {
154            "list index out of range"
155        } else {
156            "chanbin must contain at least one width"
157        };
158        TaskFailure::validation(message).at(name)
159    };
160    let value = task.value(name).ok_or_else(invalid_type)?;
161    match value {
162        ParameterValue::Array(values) if values.is_empty() => Err(empty()),
163        ParameterValue::Array(values) => {
164            let widths = values
165                .iter()
166                .map(|value| channel_integer(value).ok_or_else(invalid_type))
167                .collect::<Result<Vec<_>, _>>()?;
168            if split && !widths.iter().any(|&width| width > 1) {
169                return Ok((1, None));
170            }
171            let widths = widths
172                .into_iter()
173                .map(|width| {
174                    usize::try_from(width)
175                        .ok()
176                        .filter(|&width| width > 0)
177                        .ok_or_else(|| {
178                            TaskFailure::validation(format!(
179                                "{name} must contain positive integers"
180                            ))
181                            .at(name)
182                        })
183                })
184                .collect::<Result<Vec<_>, _>>()?;
185            Ok((widths[0], Some(widths)))
186        }
187        ParameterValue::EmptyArray { .. } => Err(empty()),
188        ParameterValue::String(text) if split => {
189            let width = if text.bytes().all(|digit| digit.is_ascii_digit()) {
190                text.parse::<usize>().unwrap_or(1)
191            } else {
192                1
193            };
194            Ok((width.max(1), None))
195        }
196        value => {
197            // CLI flags for variant parameters are passed as text. The
198            // Python task validates the accepted argument types separately.
199            let width = match value {
200                ParameterValue::String(text) => text.parse().ok(),
201                value => channel_integer(value),
202            }
203            .ok_or_else(invalid_type)?;
204            if !split && width <= 1 && !matches!(value, ParameterValue::NumpyScalar { .. }) {
205                return Err(TaskFailure::validation(
206                    "Parameter chanbin must be > 1 to do channel averaging",
207                )
208                .at(name));
209            }
210            Ok((usize::try_from(width).unwrap_or(1).max(1), None))
211        }
212    }
213}
214
215fn channel_integer(value: &ParameterValue) -> Option<i64> {
216    match value {
217        ParameterValue::Integer(value) => Some(*value),
218        ParameterValue::Bool(value) => Some(i64::from(*value)),
219        ParameterValue::NumpyScalar {
220            dtype: NumpyScalarType::Int32 | NumpyScalarType::Int64,
221            value,
222        }
223        | ParameterValue::ScalarArray(value) => channel_integer(value),
224        _ => None,
225    }
226}

228/// The row selection; the TaQL expression is split's `msselect`, a
229/// casa-rs control, and mstransform's `taql`.
230fn selection(task: &TaskInvocation<'_>) -> Result<MsSelection, TaskFailure> {
231    let selector = |name: &str| -> Result<Option<String>, TaskFailure> {
232        let text = task_selection_text(task, name)?;
233        Ok((!text.trim().is_empty()).then_some(text))
234    };
235    let taql_name = if task.surface() == "split" {
236        "msselect"
237    } else {
238        "taql"
239    };
240    let mut selection = MsSelection {
241        field: selector("field")?,
242        scan: selector("scan")?,
243        antenna: selector("antenna")?,
244        timerange: selector("timerange")?,
245        observation: task.value("observation").and_then(variant_selector_text),
246        intent: task.text("intent")?.filter(|text| !text.is_empty()),
247        correlation: variant_selector(task, "correlation")?,
248        array: variant_selector(task, "array")?,
249        uvrange: variant_selector(task, "uvrange")?,
250        feed: variant_selector(task, "feed")?,
251        msselect: selector(taql_name)?,
252        ..MsSelection::default()
253    };
254    // Earlier releases accepted absolute MJD seconds as a numeric pair.
255    // CASA reads bare numbers in its clock grammar as seconds within the minute.
256    if let Some((start, end)) = selection.timerange.as_deref().and_then(absolute_time_range) {
257        selection.timerange = None;
258        selection = selection.time_range(start, end);
259    }
260    Ok(selection)
261}

263fn variant_selector(task: &TaskInvocation<'_>, name: &str) -> Result<Option<String>, TaskFailure> {
264    let Some(value) = task.value(name) else {
265        return Ok(None);
266    };
267    if task.surface() == "split" && matches!(name, "array" | "feed") {
268        return Ok(variant_selector_text(value));
269    }
270    if task.surface() == "split" && name == "correlation" {
271        match value {
272            ParameterValue::EmptyArray { .. } => return Ok(None),
273            ParameterValue::Array(values) => {
274                let labels = values
275                    .iter()
276                    .map(|value| match value {
277                        ParameterValue::String(text) => Ok(text.as_str()),
278                        _ => Err(TaskFailure::validation("correlation expects strings").at(name)),
279                    })
280                    .collect::<Result<Vec<_>, _>>()?;
281                let text = labels.join(", ").to_uppercase();
282                return Ok((!text.is_empty()).then_some(text));
283            }
284            _ => {}
285        }
286    }
287    let record_type = match value {
288        ParameterValue::String(text) => return Ok((!text.is_empty()).then(|| text.clone())),
289        ParameterValue::NumpyScalar {
290            dtype: NumpyScalarType::Int32 | NumpyScalarType::Int64,
291            ..
292        } => {
293            return Ok(None);
294        }
295        ParameterValue::Bool(_) => "Bool",
296        ParameterValue::Integer(_) => "Int",
297        _ => "unknown",
298    };
299    Err(TaskFailure::execution(format!(
300        "RecordRep::get_pointer - incorrect data type String   used for field {name} with type {record_type:<8}"
301    )).at(name))
302}
303
304fn variant_selector_text(value: &ParameterValue) -> Option<String> {
305    match value {
306        ParameterValue::NumpyScalar {
307            dtype: NumpyScalarType::Int32 | NumpyScalarType::Int64,
308            value,
309        }
310        | ParameterValue::ScalarArray(value) => variant_selector_text(value),
311        ParameterValue::String(text) => (!text.is_empty()).then(|| text.clone()),
312        other => Some(python_repr(other)),
313    }
314}

316/// Resolve CASA's visibility layout, including the existing corrected_data alias.
317fn data_column(value: &str) -> Result<TransformDataColumn, TaskFailure> {
318    let text = value.trim().to_ascii_lowercase();
319    match text.as_str() {
320        "data" => Ok(TransformDataColumn::Data),
321        "corrected" | "corrected_data" => Ok(TransformDataColumn::CorrectedData),
322        "model" => Ok(TransformDataColumn::ModelData),
323        "data,model,corrected" => Ok(TransformDataColumn::DataModelCorrected),
324        "float_data" => Ok(TransformDataColumn::FloatData),
325        "lag_data" => Ok(TransformDataColumn::LagData),
326        "float_data,data" => Ok(TransformDataColumn::FloatDataAndData),
327        "lag_data,data" => Ok(TransformDataColumn::LagDataAndData),
328        "all" => Ok(TransformDataColumn::All),
329        _ => Err(TaskFailure::validation(format!(
330            "unsupported datacolumn {value:?}; CASA takes 'corrected', 'data', 'model', \
331             'data,model,corrected', 'float_data', 'lag_data', 'float_data,data', \
332             'lag_data,data' or 'all'"
333        ))
334        .at("datacolumn")),
335    }
336}

338fn absolute_time_range(value: &str) -> Option<(f64, f64)> {
339    let (start, end) = value.split_once('~')?;
340    let start = start.trim().parse::<f64>().ok()?;
341    let end = end.trim().parse::<f64>().ok()?;
342    (start.is_finite()
343        && end.is_finite()
344        && (start > f64::from(i32::MAX) || end > f64::from(i32::MAX)))
345    .then_some((start, end))
346}

356    /// The request a `surface` command line resolves to.
357    fn request_for(surface: &str, args: &[&str]) -> Result<MsTransformRequest, TaskFailure> {
358        let args = ["--vis", "in.ms", "--outputvis", "out.ms"]
359            .iter()
360            .chain(args)
361            .map(OsString::from)
362            .collect::<Vec<_>>();
363        let bundle = casa_provider_contracts::builtin_surface_bundle(surface).unwrap();
364        let line = TaskCommandLine::parse(&bundle, &args).unwrap();
365        let session = ParameterRuntime::default()
366            .open_session(line.open_session_request(bundle))
367            .unwrap();
368        transform_request(&TaskInvocation::new(&session, std::path::Path::new(".")))
369            .map(|(request, _)| request)
370    }

372    #[test]
373    fn each_task_averages_channels_by_its_own_casa_parameters() {
374        let width = |surface, args| request_for(surface, args).unwrap().width;
375        assert_eq!(width("split", &["--width", "8"]), 8);
376        assert_eq!(width("mstransform", &["--chanbin", "4"]), 1);
377        assert_eq!(
378            width("mstransform", &["--chanaverage", "--chanbin", "4"]),
379            4
380        );
381        let error = request_for("mstransform", &["--chanaverage", "--chanbin", "0"]).unwrap_err();
382        assert_eq!(error.parameter.as_deref(), Some("chanbin"));
383        // CASA's mstransform: "Parameter chanbin must be > 1 to do channel
384        // averaging"; split's width=1 is no averaging (#997 F8).
385        let error = request_for("mstransform", &["--chanaverage"]).unwrap_err();
386        assert_eq!(error.parameter.as_deref(), Some("chanbin"));
387        assert!(error.message.contains("must be > 1"), "{}", error.message);
388        assert_eq!(width("split", &["--width", "1"]), 1);
389    }
390
391    #[test]
392    fn datacolumn_takes_casas_values_and_default() {
393        for surface in ["split", "mstransform"] {
394            let request = request_for(surface, &[]).unwrap();
395            assert_eq!(request.data_column, TransformDataColumn::CorrectedData);
396            let data = request_for(surface, &["--datacolumn", "data"]).unwrap();
397            assert_eq!(data.data_column, TransformDataColumn::Data);
398            let model = request_for(surface, &["--datacolumn", "model"]).unwrap();
399            assert_eq!(model.data_column, TransformDataColumn::ModelData);
400        }
401    }
