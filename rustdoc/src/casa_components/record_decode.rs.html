19impl ComponentList {
20    /// Decode the `nelements` and `componentN` records written by CASA's
21    /// `ComponentList::toRecord` and stored in virtual models.
22    ///
23    /// # Errors
24    /// Missing fields or component values that CASA's model cannot represent.
25    pub fn from_record(record: &RecordValue) -> Result<Self, ComponentError> {
26        Self::decode_record(record, component)
27    }
28
29    /// Decode a supplied component list using the defaults of CASA `cl.fromrecord`.
30    ///
31    /// # Errors
32    /// A supplied component field is malformed or cannot be represented.
33    pub fn from_import_record(record: &RecordValue) -> Result<Self, ComponentError> {
34        let reference =
35            MDirection::from_angles(0., std::f64::consts::FRAC_PI_2, DirectionRef::J2000);
36        let default = SkyComponent::new(
37            Flux::stokes_jy(1., 0., 0., 0.),
38            ComponentShape::point(reference),
39            SpectralModel::new(
40                MFrequency::new(1e9, FrequencyRef::LSRK),
41                SpectralKind::Constant,
42            ),
43        )
44        .to_record()?;
45        Self::decode_record(record, |input| import_component(input, &default))
46    }
47
48    fn decode_record(
49        record: &RecordValue,
50        decode: impl Fn(&RecordValue) -> Result<SkyComponent, ComponentError>,
51    ) -> Result<Self, ComponentError> {
52        let count = match record.get("nelements") {
53            Some(Value::Scalar(ScalarValue::Int32(count))) => usize::try_from(*count).ok(),
54            Some(Value::Scalar(ScalarValue::Int64(count))) => usize::try_from(*count).ok(),
55            Some(Value::Scalar(ScalarValue::UInt32(count))) => usize::try_from(*count).ok(),
56            _ => None,
57        }
58        .ok_or_else(|| invalid("nelements"))?;
59        let mut list = Self::new();
60        for index in 0..count {
61            list.push(decode(subrecord(record, &format!("component{index}"))?)?);
62        }
63        Ok(list)
64    }
65}

67fn import_component(
68    input: &RecordValue,
69    default: &RecordValue,
70) -> Result<SkyComponent, ComponentError> {
71    let mut record = input.clone();
72    for name in ["flux", "shape", "spectrum"] {
73        if record.get(name).is_none() {
74            record.upsert(
75                name,
76                default.get(name).expect("default component field").clone(),
77            );
78        }
79    }
80    let Some(Value::Record(flux)) = record.get_mut("flux") else {
81        return Err(invalid("flux"));
82    };
83    let basis = match flux.get("polarisation") {
84        Some(_) => Polarisation::parse_casa(string(flux, "polarisation")?)?,
85        None => {
86            flux.upsert(
87                "polarisation",
88                Value::Scalar(ScalarValue::String("Stokes".into())),
89            );
90            Polarisation::Stokes
91        }
92    };
93    let flux_value = flux.get("value").ok_or_else(|| invalid("flux.value"))?;
94    let scalar = match flux_value {
95        Value::Array(ArrayValue::Float64(values)) if values.len() == 1 => {
96            values.iter().next().copied()
97        }
98        _ => number(flux, "value").ok(),
99    };
100    if basis == Polarisation::Stokes
101        && let Some(intensity) = scalar
102    {
103        flux.upsert(
104            "value",
105            Value::Array(ArrayValue::from_f64_vec(vec![intensity, 0., 0., 0.])),
106        );
107    }
108    if let Some(error) = flux.get("error") {
109        let valid_type = matches!(error, Value::Array(ArrayValue::Complex64(_)))
110            || (basis == Polarisation::Stokes
111                && matches!(error, Value::Array(ArrayValue::Float64(_))));
112        if !valid_type || complex_values(Some(error)).is_none() {
113            return Err(invalid("flux.error"));
114        }
115    }
116    let Some(Value::Record(shape)) = record.get_mut("shape") else {
117        return Err(invalid("shape"));
118    };
119    if shape.get("direction").is_none() {
120        shape.upsert(
121            "direction",
122            subrecord(default, "shape")?
123                .get("direction")
124                .expect("default direction")
125                .clone(),
126        );
127    }
128    let Some(Value::Record(spectrum)) = record.get_mut("spectrum") else {
129        return Err(invalid("spectrum"));
130    };
131    if spectrum.get("frequency").is_none() {
132        spectrum.upsert(
133            "frequency",
134            subrecord(default, "spectrum")?
135                .get("frequency")
136                .expect("default frequency")
137                .clone(),
138        );
139    }
140    if SpectralType::parse_casa(string(spectrum, "type")?)? == SpectralType::SpectralIndex {
141        number(spectrum, "index")?;
142    }
143    if record.get("label").is_some() {
144        string(&record, "label")?;
145    }
146    component(&record)
147}

149impl SkyComponent {
150    /// Decode a complete CASA component record, including its flux basis,
151    /// shape and direction errors, spectrum, label and optional parameters.
152    ///
153    /// This reads the records produced by [`Self::to_record`]. Tool-specific
154    /// defaults and diagnostics belong to the calling tool.
155    ///
156    /// # Errors
157    /// Missing fields or values the component model cannot represent.
158    pub fn from_record(record: &RecordValue) -> Result<Self, ComponentError> {
159        component(record)
160    }
161}
162
163fn component(record: &RecordValue) -> Result<SkyComponent, ComponentError> {
164    let flux = subrecord(record, "flux")?;
165    let values = complex_values(flux.get("value")).ok_or_else(|| invalid("flux.value"))?;
166    let basis = Polarisation::parse_casa(string(flux, "polarisation")?)?;
167    let mut flux_model = Flux::new(values, string(flux, "unit")?, basis)?;
168    if let Some(errors) = complex_values(flux.get("error")) {
169        flux_model.set_errors(errors);
170    }
171    let mut component = SkyComponent::new(
172        flux_model,
173        shape(subrecord(record, "shape")?)?,
174        spectrum(subrecord(record, "spectrum")?)?,
175    );
176    if let Ok(label) = string(record, "label") {
177        component = component.with_label(label);
178    }
179    if let Some(Value::Array(ArrayValue::Float64(parameters))) = record.get("optionalParameters") {
180        component = component.with_optional_parameters(parameters.iter().copied().collect());
181    }
182    Ok(component)
183}

185fn shape(record: &RecordValue) -> Result<ComponentShape, ComponentError> {
186    let shape_type = ShapeType::parse_casa(string(record, "type")?)?;
187    let direction_record = subrecord(record, "direction")?;
188    let direction =
189        direction_from_record(direction_record).map_err(|error| invalid(error.to_string()))?;
190    let radians = Unit::new("rad").expect("rad is a unit");
191    let angle = |name| {
192        quantity(subrecord(record, name)?)?
193            .get_value_in(&radians)
194            .map_err(|error| invalid(error.to_string()))
195    };
196    let mut shape = if shape_type == ShapeType::Point {
197        ComponentShape::point(direction)
198    } else {
199        let axes = [
200            angle("majoraxis")?,
201            angle("minoraxis")?,
202            angle("positionangle")?,
203        ];
204        let names = ["majoraxiserror", "minoraxiserror", "positionangleerror"];
205        let errors = names.map(|name| {
206            if record.get(name).is_some() {
207                angle(name)
208            } else {
209                Ok(0.0)
210            }
211        });
212        let [major, minor, position] = errors;
213        ComponentShape::from_parameters(
214            shape_type,
215            direction,
216            &axes,
217            Some(&[major?, minor?, position?]),
218        )?
219    };
220    if let Ok(error) = subrecord(direction_record, "error") {
221        shape.set_direction_error(DirectionError::new(
222            quantity(subrecord(error, "latitude")?)?,
223            quantity(subrecord(error, "longitude")?)?,
224        )?);
225    }
226    Ok(shape)
227}

229fn spectrum(record: &RecordValue) -> Result<SpectralModel, ComponentError> {
230    let frequency_record = subrecord(record, "frequency")?;
231    let mut frequency =
232        frequency_from_record(frequency_record).map_err(|error| invalid(error.to_string()))?;
233    let mut kind = SpectralKind::casa_default(SpectralType::parse_casa(string(record, "type")?)?);
234    crate::table::apply_spectral_record(&mut kind, record)?;
235    if matches!(kind, SpectralKind::Tabular(_)) {
236        let frame = frequency_from_record(subrecord(record, "freqRef")?)
237            .map_err(|error| invalid(error.to_string()))?
238            .refer();
239        let hz = number(record, "referenceFreq")?;
240        number(record, "minFreq")?;
241        number(record, "maxFreq")?;
242        frequency = casa_types::measures::frequency::MFrequency::new(hz, frame);
243    }
244    let mut model = SpectralModel::new(frequency, kind);
245    let unit = string(subrecord(frequency_record, "m0")?, "unit")?;
246    let unit = Unit::new(unit).map_err(|error| invalid(error.to_string()))?;
247    if unit.conformant(&Unit::new("Hz").expect("Hz")) {
248        model.set_frequency_unit(unit)?;
249    }
250    Ok(model)
251}

253fn subrecord<'a>(record: &'a RecordValue, name: &str) -> Result<&'a RecordValue, ComponentError> {
254    match record.get(name) {
255        Some(Value::Record(value)) => Ok(value),
256        _ => Err(invalid(name)),
257    }
258}
259
260fn string<'a>(record: &'a RecordValue, name: &str) -> Result<&'a str, ComponentError> {
261    match record.get(name) {
262        Some(Value::Scalar(ScalarValue::String(value))) => Ok(value),
263        _ => Err(invalid(name)),
264    }
265}
266
267fn number(record: &RecordValue, name: &str) -> Result<f64, ComponentError> {
268    match record.get(name) {
269        Some(Value::Scalar(ScalarValue::Float64(value))) => Ok(*value),
270        Some(Value::Scalar(ScalarValue::Float32(value))) => Ok(f64::from(*value)),
271        Some(Value::Scalar(ScalarValue::Int32(value))) => Ok(f64::from(*value)),
272        _ => Err(invalid(name)),
273    }
274}
275
276fn quantity(record: &RecordValue) -> Result<Quantity, ComponentError> {
277    Quantity::new(number(record, "value")?, string(record, "unit")?)
278        .map_err(|error| invalid(error.to_string()))
279}

281fn complex_values(value: Option<&Value>) -> Option<[Complex64; 4]> {
282    let data = match value? {
283        Value::Array(ArrayValue::Float64(values)) => values
284            .iter()
285            .map(|&value| Complex64::new(value, 0.0))
286            .collect::<Vec<_>>(),
287        Value::Array(ArrayValue::Float32(values)) => values
288            .iter()
289            .map(|&value| Complex64::new(f64::from(value), 0.0))
290            .collect(),
291        Value::Array(ArrayValue::Complex64(values)) => values.iter().copied().collect(),
292        Value::Array(ArrayValue::Complex32(values)) => values
293            .iter()
294            .map(|value| Complex64::new(f64::from(value.re), f64::from(value.im)))
295            .collect(),
296        _ => return None,
297    };
298    data.try_into().ok()
299}
300
301fn invalid(field: impl Into<String>) -> ComponentError {
302    ComponentError::Record(field.into())
303}

311    #[test]
312    fn component_record_frequency_honors_its_unit() {
313        let mut model = SpectralModel::new(
314            MFrequency::new(1.5e9, FrequencyRef::LSRK),
315            SpectralKind::Constant,
316        );
317        model.set_frequency_unit(Unit::new("GHz").unwrap()).unwrap();
318        let original = SkyComponent::new(
319            Flux::stokes_jy(1.0, 0.0, 0.0, 0.0),
320            ComponentShape::point(MDirection::from_angles(0.0, 0.0, DirectionRef::J2000)),
321            model,
322        );
323        let decoded = SkyComponent::from_record(&original.to_record().unwrap()).unwrap();
324        assert_eq!(decoded.spectrum().reference_frequency().hz(), 1.5e9);
325    }

329    #[test]
330    fn resolves_component_record_offsets_like_casa_6_7_6_14() {
331        let component = SkyComponent::new(
332            Flux::stokes_jy(1.0, 0.0, 0.0, 0.0),
333            ComponentShape::point(MDirection::from_angles(
334                10_f64.to_radians(),
335                20_f64.to_radians(),
336                DirectionRef::J2000,
337            )),
338            SpectralModel::new(
339                MFrequency::new(1.4e9, FrequencyRef::LSRK),
340                SpectralKind::Constant,
341            ),
342        );
343        let mut record = component.to_record().unwrap();
344        let Value::Record(shape) = record.get_mut("shape").unwrap() else {
345            unreachable!()
346        };
347        let Value::Record(direction) = shape.get_mut("direction").unwrap() else {
348            unreachable!()
349        };
350        direction.push(RecordField::new(
351            "offset",
352            Value::Record(casa_types::measures::direction_to_record(
353                &MDirection::from_angles(
354                    1_f64.to_radians(),
355                    2_f64.to_radians(),
356                    DirectionRef::J2000,
357                ),
358            )),
359        ));
360        let Value::Record(spectrum) = record.get_mut("spectrum").unwrap() else {
361            unreachable!()
362        };
363        let Value::Record(frequency) = spectrum.get_mut("frequency").unwrap() else {
364            unreachable!()
365        };
366        frequency.push(RecordField::new(
367            "offset",
368            Value::Record(casa_types::measures::frequency_to_record(&MFrequency::new(
369                1e9,
370                FrequencyRef::LSRK,
371            ))),
372        ));
373        let decoded = SkyComponent::from_record(&record).unwrap();
374        // Independent CASA cl.add(record), rename/done/open values, preserved
375        // in the Python componentlist offset fixture.
376        let (longitude, latitude) = decoded.shape().reference_direction().as_angles();
377        assert!((longitude - 0.093_570_138_597_165_76).abs() < 1e-15);
378        assert!((latitude - 0.192_564_780_043_098_24).abs() < 1e-15);
379        assert_eq!(decoded.spectrum().reference_frequency().hz(), 2.4e9);
380    }
