2//! Image-plane beams that mix the sky's Stokes planes before gridding.
3use super::{BeamPattern, PredictError, SkyModel, jones::JonesImage};
4use casa_imaging_runtime::Admitted;
5use ndarray::Array2;
6use std::sync::Arc;
7
8pub(crate) enum ImageBeam {
9    Jones(Arc<JonesImage>),
10    Squinted {
11        pattern: Arc<Admitted<BeamPattern>>,
12        offset: [f64; 2],
13        reference_hz: f64,
14    },
15}
16
17impl ImageBeam {
18    pub(crate) fn apply_plane(
19        &self,
20        model: &SkyModel,
21        plane: usize,
22        images: Vec<Array2<f32>>,
23        pointing: [f64; 2],
24        angle: f64,
25    ) -> Result<Vec<Array2<f32>>, PredictError> {
26        match self {
27            Self::Jones(image) => image.apply_plane(model, plane, images, pointing, angle),
28            Self::Squinted {
29                pattern,
30                offset,
31                reference_hz,
32            } => {
33                let frequency = model.plane_frequency_hz(plane);
34                let [right, left] = [1., -1.].map(|sign| {
35                    // BeamSquint::getPointingDirection quantizes the
36                    // frequency scale before rotating the saved AZEL offset.
37                    let scale = f64::from((-reference_hz / frequency) as f32) * sign;
38                    let [sx, sy] = offset.map(|v| v * scale);
39                    let (sin, cos) = (-angle).sin_cos();
40                    let (dx, dy) = (-(sx * cos + sy * sin), -sx * sin + sy * cos);
41                    let [ra, dec] = pointing;
42                    let direction = [
43                        ra + dx.sin().atan2(dx.cos() * (dec + dy).cos()),
44                        (dx.cos() * (dec + dy).sin()).asin(),
45                    ];
46                    model.pixel_of(direction)
47                });
48                let centres = [right?, left?, model.pixel_of(pointing)?];
49                Ok(radial_plane(model, plane, images, &pattern.value, centres))
50            }
51        }
52    }
53}
54
55fn radial_plane(
56    model: &SkyModel,
57    plane: usize,
58    images: Vec<Array2<f32>>,
59    pattern: &BeamPattern,
60    centres: [[f64; 2]; 3],
61) -> Vec<Array2<f32>> {
62    use casa_imaging_model::PolarizationCoordinate::{StokesI, StokesQ, StokesU, StokesV};
63    let frequency = model.plane_frequency_hz(plane);
64    let at = pattern.at_frequency(frequency);
65    let increments = model.increment_rad.map(f64::to_degrees);
66    let [nx, ny] = model.shape();
67    let mut out: Vec<_> = (0..4).map(|_| Array2::zeros((ny, nx))).collect();
68    for y in 0..ny {
69        for x in 0..nx {
70            let powers = centres.map(|centre| {
71                let rx = ((x as f64 - centre[0]) * increments[0]).powi(2) as f32;
72                let ry = ((y as f64 - centre[1]) * increments[1]).powi(2) as f32;
73                let radius = (f64::from((rx + ry).sqrt()) * 60. * frequency / 1e9) as f32;
74                at.power(f64::from(radius))
75            });
76            let mut stokes = [0.; 4];
77            for (pol, image) in model.polarizations().iter().zip(&images) {
78                let slot = match pol {
79                    StokesI => 0,
80                    StokesQ => 1,
81                    StokesU => 2,
82                    StokesV => 3,
83                    _ => unreachable!("sky models contain Stokes planes"),
84                };
85                stokes[slot] = image[[y, x]];
86            }
87            let [i, q, u, v] = stokes;
88            let rr = (i + v) * powers[0];
89            let ll = (i - v) * powers[1];
90            // PBMath1D applies the central beam to RL/LR rather than
91            // the product of displaced voltage patterns.
92            for (output, value) in
93                out.iter_mut()
94                    .zip([(rr + ll) / 2., q * powers[2], u * powers[2], (rr - ll) / 2.])
95            {
96                output[[y, x]] = value;
97            }
98        }
99    }
100    out
