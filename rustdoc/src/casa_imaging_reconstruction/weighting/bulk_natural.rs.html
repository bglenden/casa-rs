11/// Row-independent natural-weighting setup and its initial numerical reduction.
12pub struct BulkNaturalWeighting {
13    sum: Option<WeightingSumWeightPhase>,
14    finite: FiniteValuePolicy,
15    centres: Vec<f64>,
16    boundaries: Vec<f64>,
17    constant: bool,
18    samples: u64,
19    rows: u64,
20}

22/// Reusable selected-channel/correlation outputs. Values remain source-owned;
23/// these arrays contain only flags and the effective numerical weights.
24pub struct NaturalRowPreparation {
25    weights: Vec<f32>,
26    flags: Vec<bool>,
27    weight_flags: Vec<bool>,
28}
29
30impl NaturalRowPreparation {
31    /// Allocate the exact admitted row shape once, then reuse it across rows.
32    pub fn new(rows: usize, channels: usize, correlations: usize) -> Result<Self, WeightingError> {
33        let capacity = rows
34            .checked_mul(channels)
35            .and_then(|n| n.checked_mul(correlations))
36            .filter(|&n| n > 0 && n <= isize::MAX as usize / 6)
37            .ok_or(WeightingError::ResidencyOverflow)?;
38        Ok(Self {
39            weights: vec![0.0; capacity],
40            flags: vec![false; capacity],
41            weight_flags: vec![false; capacity],
42        })
43    }
44
45    /// Effective natural weights in selected channel/correlation order.
46    pub fn weights(&self) -> &[f32] {
47        &self.weights
48    }
49    /// Numerical input rejection flags.
50    pub fn flags(&self) -> &[bool] {
51        &self.flags
52    }
53    /// CASA complete-correlation weight-group flags.
54    pub fn weight_flags(&self) -> &[bool] {
55        &self.weight_flags
56    }
57
58    /// Borrow all three flat outputs for disjoint row-chunk preparation.
59    pub fn buffers_mut(&mut self) -> (&mut [f32], &mut [bool], &mut [bool]) {
60        (&mut self.weights, &mut self.flags, &mut self.weight_flags)
61    }
62}

64impl BulkNaturalWeighting {
65    /// Associate a subsequent exact source traversal with the retained weights.
66    /// The runtime must validate the fresh source completion against this proof.
67    pub fn replay_summary(
68        state: &WeightingAlgorithmState,
69        proof: FrozenWeightingBinding,
70        rows: u64,
71    ) -> Result<WeightingReplaySummary, WeightingError> {
72        if proof.generation != state.generation_id
73            || proof.problem != state.problem
74            || proof.commitment != state.commitment
75            || proof.weighted_sample_count != state.sample_count
76            || rows == 0
77        {
78            return Err(WeightingError::CoverageMismatch);
79        }
80        let sequence = state.next_replay.fetch_add(1, Ordering::Relaxed);
81        let replay_id = WeightingReplayId::new()?;
82        Ok(WeightingReplaySummary {
83            replay_id,
84            generation: state.generation_id,
85
86            sample_count: state.sample_count,
87            block_count: rows,
88            replay_sequence: sequence,
89
90            residency: state.generation_residency,
91        })
92    }
93    /// Begin the ordinary constant-MFS or linear-cube natural-weighting route.
94    pub fn new(
95        problem: &CompiledProblem,
96        plan: &WeightingPlan,
97        initial: bool,
98    ) -> Result<Self, WeightingError> {
99        if problem.weighting().scheme() != WeightingScheme::Natural
100            || problem.weighting().uv_taper().is_some()
101            || problem.visibility_transform().is_some()
102        {
103            return Err(WeightingError::ProblemMismatch);
104        }
105        let constant = matches!(
106            problem.reconstruction().basis(),
107            ReconstructionBasis::Constant
108        );
109        if !constant
110            && (!matches!(
111                problem.reconstruction().basis(),
112                ReconstructionBasis::ChannelLocal { .. }
113            ) || problem.science().spectral().sampling().kernel()
114                != casa_imaging_model::SpectralKernel::Linear)
115        {
116            return Err(WeightingError::ProblemMismatch);
117        }
118        let axis = problem.geometry().spectral();
119        let centres = (0..axis.output_channels())
120            .map(|i| {
121                axis.channel_centre_hz(i)
122                    .ok_or(WeightingError::ProblemMismatch)
123            })
124            .collect::<Result<Vec<_>, _>>()?;
125        let boundaries = (0..=axis.output_channels())
126            .map(|i| {
127                axis.channel_boundary_hz(i)
128                    .ok_or(WeightingError::ProblemMismatch)
129            })
130            .collect::<Result<Vec<_>, _>>()?;
131        Ok(Self {
132            sum: initial
133                .then(|| begin_weighting_generation(problem, plan)?.finish(problem))
134                .transpose()?,
135            finite: problem.numerics().finite_values(),
136            centres,
137            boundaries,
138            constant,
139            samples: 0,
140            rows: 0,
141        })
142    }

144    /// Apply the existing CASA natural-weight/flag rules in a tight row loop.
145    /// `initial` controls only the once-per-source reduction, never the science.
146    #[allow(clippy::too_many_arguments)]
147    pub fn prepare_row(
148        &self,
149        row: SelectedNumericRow<'_>,
150        frequencies: &[f64],
151        boundaries: &[[f64; 2]],
152        weights_out: &mut [f32],
153        flags_out: &mut [bool],
154        weight_flags_out: &mut [bool],
155        initial: bool,
156    ) -> Result<f64, WeightingError> {
157        let correlations = row.correlations.len();
158        let count = row
159            .channels
160            .len()
161            .checked_mul(correlations)
162            .ok_or(WeightingError::ResidencyOverflow)?;
163        if !row.has_exact_shape()
164            || weights_out.len() != count
165            || flags_out.len() != count
166            || weight_flags_out.len() != count
167            || frequencies.len() != row.channels.len()
168            || boundaries.len() != frequencies.len()
169        {
170            return Err(WeightingError::CoverageMismatch);
171        }
172        let mut sum = 0.0;
173        for (channel, coordinate) in row.channels.iter().enumerate() {
174            let source = (coordinate.channel_index - row.first_stored_channel) as usize
175                * row.stored_correlations;
176            let weights = match row.weights {
177                SelectedNumericWeights::PerRow(values) => values,
178                SelectedNumericWeights::PerChannel(values) => {
179                    &values[source..source + row.stored_correlations]
180                }
181            };
182            let (base, group_flag, parallel_flag) =
183                numeric_channel_input_weight(row, source, self.finite)?;
184            let mapped = initial
185                && (self.constant
186                    || !crate::spectral_sampling::linear_terms(
187                        &self.centres,
188                        &self.boundaries,
189                        boundaries[channel],
190                        frequencies[channel],
191                    )
192                    .is_empty());
193            for (ordinal, product) in row.correlations.iter().enumerate() {
194                let corr = product.correlation_index() as usize;
195                let index = source + corr;
196                let destination = channel * correlations + ordinal;
197                let value = match row.visibility {
198                    SelectedNumericVisibility::Float32(values) => {
199                        SelectedVisibilitySample::Float32(values[index])
200                    }
201                    SelectedNumericVisibility::Complex32(values) => {
202                        SelectedVisibilitySample::Complex32([values[index].re, values[index].im])
203                    }
204                };
205                let rejected_group = if product.correlation_type().contributes_to_stokes_i() {
206                    parallel_flag
207                } else {
208                    row.flags[index]
209                };
210                weights_out[destination] = if rejected_group { 0.0 } else { base as f32 };
211                flags_out[destination] = !crate::spectral_operator::accept_polarization_value(
212                    value,
213                    weights[corr],
214                    row.row.row_flag || row.flags[index],
215                    self.finite,
216                )
217                .map_err(|_| WeightingError::CoverageMismatch)?;
218                weight_flags_out[destination] = group_flag || rejected_group || row.row.row_flag;
219                if mapped && !rejected_group {
220                    sum += base;
221                }
222            }
223        }
224        Ok(sum)
225    }

227    /// Publish chunk-local discovery totals only after every chunk has joined.
228    pub fn commit_prepared_chunk(
229        &mut self,
230        sum: f64,
231        samples: u64,
232        rows: u64,
233    ) -> Result<(), WeightingError> {
234        let next_samples = self
235            .samples
236            .checked_add(samples)
237            .ok_or(WeightingError::ResidencyOverflow)?;
238        let next_rows = self
239            .rows
240            .checked_add(rows)
241            .ok_or(WeightingError::ResidencyOverflow)?;
242        let phase = self.sum.as_mut().ok_or(WeightingError::CoverageMismatch)?;
243        add_weight(&mut phase.sum_weights[0], sum)?;
244        self.samples = next_samples;
245        self.rows = next_rows;
246        Ok(())
247    }
248
249    /// Bind the numerical reduction to a successfully completed exact source
250    /// traversal. This versioned identity associates owners, not product content.
251    pub fn finish(
252        mut self,
253        sample_count: u64,
254    ) -> Result<
255        (
256            WeightingAlgorithmState,
257            WeightingReplaySummary,
258            FrozenWeightingBinding,
259        ),
260        WeightingError,
261    > {
262        if self.samples == 0 || self.samples != sample_count {
263            return Err(WeightingError::CoverageMismatch);
264        }
265        let mut sum = self.sum.take().ok_or(WeightingError::CoverageMismatch)?;
266        sum.sum_sample_count = self.samples;
267        sum.density_sample_count = self.samples;
268        let state = sum.finish()?;
269        state.next_replay.store(1, Ordering::Relaxed);
270        let replay_id = WeightingReplayId::new()?;
271        let mut residency = state.generation_residency;
272        residency.weighted_block_bytes = 0;
273        residency.weighted_sample_bytes = 0;
274        residency.simultaneous_selected_weighted_bytes = 0;
275        let summary = WeightingReplaySummary {
276            replay_id,
277            generation: state.generation_id,
278
279            sample_count: self.samples,
280            block_count: self.rows,
281            replay_sequence: 0,
282
283            residency,
284        };
285        let proof = FrozenWeightingBinding {
286            problem: state.problem,
287            commitment: state.commitment,
288            generation: state.generation_id,
289
290            selected_sample_count: sample_count,
291
292            weighted_sample_count: self.samples,
293        };
294        Ok((state, summary, proof))
295    }
