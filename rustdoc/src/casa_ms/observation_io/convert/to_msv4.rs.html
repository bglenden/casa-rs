31/// Convert the store at `source` (MSv2 or MSv4) into a new MSv4
32/// processing set at `destination`.
33///
34/// Every partition becomes one dataset with the selected data groups
35/// ([`ConversionOptions::data_groups`]) and the antennas its baselines name
36/// (others are [`Loss::UnusedAntennas`]).
37/// An MSv4's datasets keep their names. An MSv2's are named as XRADIO's
38/// converter names them: the file name without `.ms`, then the partition's
39/// number in the order the partitions' first MAIN rows appear, padded with
40/// zeros to the width of the largest. Its antennas (for the telescopes
41/// XRADIO renames, by the first observation's telescope), fields, sources
42/// and spectral windows are renamed as that converter renames them
43/// (`NAME_STATION`, `NAME_FIELD_ID`, `NAME_SOURCE_ID`,
44/// `NAME_SPECTRAL_WINDOW_ID`), so the two converters' stores agree; a
45/// source XRADIO finds no row for in one spectral window, which it names
46/// `Unknown` there, is still named here. Two partitions XRADIO would merge,
47/// rows of different feeds, are two datasets here, and rows with
48/// `STATE_ID` -1 have no intents here where XRADIO gives them the last
49/// `STATE` row's; the datasets numbered after such a difference can hold
50/// other data than XRADIO's of the same name. A dataset whose fields
51/// follow an ephemeris holds it as XRADIO's `field_and_source_ephemeris`,
52/// the samples on `time_ephemeris` from the last before its first
53/// time to the first two minutes past its last (the samples no dataset
54/// holds are a reported loss), with the phase centre at each time as XRADIO
55/// states it. Partitions are
56/// converted [`ConversionOptions::workers`] at a time; nothing appears at
57/// `destination` until every dataset is complete. An MSv2 is partitioned
58/// by the keys every partition has, XRADIO's `partition_scheme=[]`, its
59/// pointing is converted, and every dataset is chunked as
60/// [`ConversionOptions::chunks`] says; [`to_msv4_with`] chooses otherwise.
61///
62/// # Errors
63///
64/// All before anything is written: [`ObservationError::InvalidRequest`]
65/// naming the first loss not accepted, or an ephemeris that does not cover
66/// a dataset's times; [`ObservationError::Unsupported`] for fields whose
67/// direction is a polynomial in time, a dataset whose fields follow
68/// different ephemerides (or only some one), or radial velocities that are
69/// not geocentric. Otherwise, any error opening or reading the source or
70/// writing the destination, which then is not created.
71pub fn to_msv4(
72    source: &Path,
73    destination: &Path,
74    options: &ConversionOptions,
75) -> Result<ConversionReport, ObservationError> {
76    let chunks = options.chunks;
77    to_msv4_with(
78        source,
79        destination,
80        options,
81        &Msv4Conversion {
82            partition_scheme: PartitionScheme::default(),
83            pointing: true,
84            chunk_shape: &move |_| chunks,
85        },
86    )
87}

89/// The chunk shape `[time, baseline, frequency, polarization]` asked of a
90/// dataset of the given extents (in the same order), 0 for a whole axis;
91/// each axis is then clipped to the dataset's extent.
92pub type ChunkShape = dyn Fn([usize; 4]) -> [u64; 4] + Sync;

94/// What an MSv4 conversion chooses beyond [`ConversionOptions`], as
95/// XRADIO's converter's `partition_scheme`, `with_pointing` and
96/// `main_chunksize` choose it.
97pub struct Msv4Conversion<'a> {
98    /// The keys an MSv2 source is further partitioned by, each partition a
99    /// dataset; an MSv4 source keeps its datasets.
100    pub partition_scheme: PartitionScheme,
101    /// Whether the source's pointing is converted; without it, its
102    /// pointing is a reported loss ([`Loss::Table`]).
103    pub pointing: bool,
104    /// The chunk shape asked of each dataset by its extents: a fixed shape,
105    /// or [`xradio_chunks`](super::xradio_chunks), as XRADIO's converter
106    /// chunks it.
107    pub chunk_shape: &'a ChunkShape,
108}

110/// [`to_msv4`], partitioned, with or without pointing, and chunked as
111/// `conversion` chooses; [`ConversionOptions::chunks`] is not read.
112///
113/// # Errors
114///
115/// As [`to_msv4`].
116pub fn to_msv4_with(
117    source: &Path,
118    destination: &Path,
119    options: &ConversionOptions,
120    conversion: &Msv4Conversion<'_>,
121) -> Result<ConversionReport, ObservationError> {
122    let Named {
123        observation,
124        catalogue: named,
125        datasets,
126    } = open_named(source, conversion.partition_scheme)?;
127    let catalogue = observation.catalogue();
128    let groups = losses::selected_groups(catalogue, options.data_groups.as_deref())?;
129    let mut found = losses::inventory(source, catalogue, &groups, losses::Store::Msv4)?;
130    let (pointing, lost) = if conversion.pointing {
131        pointing_plan::plan(
132            source,
133            observation.as_ref(),
134            losses::Store::Msv4,
135            options.batch_bytes,
136        )?
137    } else {
138        pointing_plan::omitted(source)?
139    };
140    found.extend(lost);
141    found.extend(losses::unused_antennas(catalogue));
142    found.extend(preflight_fields(observation.as_ref(), &groups[0])?);
143    found.sort();
144    losses::refuse_unaccepted(&found, &options.accept)?;
145    let set = Msv4ProcessingSetWriter::create(destination)?;
146    let ids = catalogue.partition_ids();
147    let next = AtomicUsize::new(0);
148    let results = Mutex::new(Vec::with_capacity(ids.len()));
149    std::thread::scope(|scope| {
150        for _ in 0..options.workers.clamp(1, ids.len().max(1)) {
151            scope.spawn(|| {
152                loop {
153                    let index = next.fetch_add(1, Ordering::Relaxed);
154                    let Some(id) = ids.get(index) else {
155                        break;
156                    };
157                    let result = partition(
158                        &Destination {
159                            set: &set,
160                            named: &named,
161                            pointing: &pointing,
162                            chunk_shape: conversion.chunk_shape,
163                        },
164                        observation.as_ref(),
165                        *id,
166                        &groups,
167                        &datasets[id.0 as usize],
168                        options,
169                    );
170                    let failed = result.is_err();
171                    results
172                        .lock()
173                        .unwrap_or_else(std::sync::PoisonError::into_inner)
174                        .push(result);
175                    if failed {
176                        // Stop handing out partitions; the others finish
177                        // what they hold.
178                        next.store(ids.len(), Ordering::Relaxed);
179                    }
180                }
181            });
182        }
183    });
184    let mut report = ConversionReport {
185        losses: found,
186        ..ConversionReport::default()
187    };
188    for result in results
189        .into_inner()
190        .unwrap_or_else(std::sync::PoisonError::into_inner)
191    {
192        let converted = result?;
193        report.partitions += 1;
194        report.batches += converted.batches;
195        report.samples += converted.samples;
196        report.elements += converted.counters.elements;
197        report.held_elements += converted.counters.held_elements;
198    }
199    set.finish()?;
200    Ok(report)
201}

203/// Refuse, before anything is written, what a dataset's field dataset
204/// cannot hold ([`check_fields`]), and report the samples of each ephemeris
205/// the partitions' fields follow that no dataset holds, one loss an
206/// ephemeris: a dataset holds those from the last before its first time to
207/// the first that reaches past its last time's TDB ([`kept_samples`]),
208/// much as XRADIO's converter keeps those either side of a dataset's times.
209/// Only the samples the source states count: an extrapolated one is not
210/// written, and reading the destination extrapolates it again.
211fn preflight_fields(
212    observation: &dyn Observation,
213    group: &str,
214) -> Result<Vec<Loss>, ObservationError> {
215    let catalogue = observation.catalogue();
216    let mut followed: Vec<(&Arc<FieldEphemeris>, Vec<bool>)> = Vec::new();
217    for id in catalogue.partition_ids() {
218        let grid = observation.grid(id, group)?;
219        let times = unix_times(&grid);
220        let entry = catalogue.partition(id);
221        if let Some((_, data_group)) = entry.data_group(group) {
222            check_fields(catalogue, &grid, &times, data_group.uvw_frame.effective)?;
223        }
224        for field in &entry.fields {
225            let Some(ephemeris) = catalogue.fields[field.0 as usize].ephemeris.as_ref() else {
226                continue;
227            };
228            let index = followed
229                .iter()
230                .position(|(known, _)| Arc::ptr_eq(known, ephemeris))
231                .unwrap_or_else(|| {
232                    let stated = ephemeris.times_mjd().len() - ephemeris.extrapolated();
233                    followed.push((ephemeris, vec![false; stated]));
234                    followed.len() - 1
235                });
236            for held in &mut followed[index].1[kept_samples(ephemeris, &times)] {
237                *held = true;
238            }
239        }
240    }
241    Ok(followed
242        .into_iter()
243        .filter_map(|(ephemeris, held)| {
244            let lost = held.iter().filter(|held| !**held).count();
245            (lost > 0).then(|| {
246                Loss::Table(format!(
247                    "the ephemeris of {}: samples no dataset keeps ({lost}; a dataset keeps \
248                     those from the last before its first time to the first 120 s or more past \
249                     its last)",
250                    ephemeris.name()
251                ))
252            })
253        })
254        .collect())
255}

266/// Open `source`, with the catalogue its datasets are declared with and
267/// each partition's dataset name: an MSv4's own, or an MSv2's, partitioned
268/// by `partition_scheme`, with XRADIO's names.
269fn open_named(source: &Path, partition_scheme: PartitionScheme) -> Result<Named, ObservationError> {
270    let options = OpenOptions {
271        partition_scheme,
272        ..OpenOptions::default()
273    };
274    if source.join("zarr.json").is_file() {
275        let observation = Msv4Observation::open_with(source, &options)?;
276        let catalogue = observation.catalogue().as_ref().clone();
277        let datasets = catalogue
278            .partition_ids()
279            .into_iter()
280            .map(|id| observation.dataset_name(id).to_string())
281            .collect();
282        return Ok(Named {
283            observation: Box::new(observation),
284            catalogue,
285            datasets,
286        });
287    }
288    let observation = Msv2Observation::open_with(source, &options)?;
289    let ids = observation.catalogue().partition_ids();
290    let windows: Vec<u32> = ids
291        .iter()
292        .map(|id| observation.spectral_window_id(*id))
293        .collect();
294    let first_rows: Vec<Option<u32>> = ids.iter().map(|id| observation.first_row(*id)).collect();
295    let fields = observation.measurement_set().field().map_err(storage)?;
296    let sources = (0..observation.catalogue().fields.len())
297        .map(|row| fields.source_id(row).map_err(storage))
298        .collect::<Result<Vec<_>, _>>()?;
299    let named = xradio_names(observation.catalogue(), &windows, &sources);
300    let datasets = dataset_names(source, &first_rows);
301    Ok(Named {
302        observation: Box::new(observation),
303        catalogue: named,
304        datasets,
305    })
306}

308/// Each partition's dataset name, as XRADIO's converter names an MSv2's:
309/// the file name without `.ms`, then the partition's number in the order
310/// of `first_rows`, the partitions' first MAIN rows, padded with zeros to
311/// the width of the largest. The path is made absolute, and resolved when
312/// that leaves no file name, so `.`, `..` and a trailing `/` name the
313/// MeasurementSet they stand for, where XRADIO's converter names `.`'s
314/// datasets `_0`, `..`'s `.._0` and `/`'s `_0` (here `dataset_0`, `/`
315/// having no name).
316fn dataset_names(source: &Path, first_rows: &[Option<u32>]) -> Vec<String> {
317    let mut path = std::path::absolute(source).unwrap_or_else(|_| source.to_path_buf());
318    if path.file_name().is_none() {
319        path = std::fs::canonicalize(source).unwrap_or(path);
320    }
321    let stem = path.file_name().map_or_else(
322        || "dataset".to_string(),
323        |name| name.to_string_lossy().replace(".ms", ""),
324    );
325    let mut order: Vec<usize> = (0..first_rows.len()).collect();
326    order.sort_by_key(|index| (first_rows[*index].unwrap_or(u32::MAX), *index));
327    let width = first_rows.len().saturating_sub(1).to_string().len();
328    let mut names = vec![String::new(); first_rows.len()];
329    for (number, index) in order.into_iter().enumerate() {
330        names[index] = format!("{stem}_{number:0width$}");
331    }
332    names
333}

335/// `catalogue` with XRADIO's names, as its converter names an MSv2's:
336/// antennas `NAME_STATION` for the telescopes it names so
337/// ([`STATION_NAMED`]), fields `NAME_ID`, sources `NAME_SOURCEID` by each
338/// field's `SOURCE_ID` (`sources`), and spectral windows `NAME_ID`
339/// (`spw_ID` unnamed) by the `SPECTRAL_WINDOW` row of each partition
340/// (`windows`).
341fn xradio_names(catalogue: &Catalogue, windows: &[u32], sources: &[i32]) -> Catalogue {
342    let mut named = catalogue.clone();
343    let telescope = catalogue
344        .observations
345        .first()
346        .map_or("", |observation| observation.telescope.as_str());
347    if STATION_NAMED.contains(&telescope) {
348        for antenna in &mut named.antennas {
349            antenna.name = format!("{}_{}", antenna.name, antenna.station);
350        }
351    }
352    for ((index, field), source) in named.fields.iter_mut().enumerate().zip(sources) {
353        field.name = format!("{}_{index}", field.name);
354        if let Some(name) = &mut field.source_name {
355            *name = format!("{name}_{source}");
356        }
357    }
358    for (partition, window) in named.partitions.iter_mut().zip(windows) {
359        partition.spectral_window = match partition.spectral_window.as_str() {
360            "" | "none" => format!("spw_{window}"),
361            name => format!("{name}_{window}"),
362        };
363    }
364    named
365}

367/// What every partition is written into: the processing set, the
368/// catalogue its datasets are declared with, and the pointing carried.
369struct Destination<'a> {
370    set: &'a Msv4ProcessingSetWriter,
371    named: &'a Catalogue,
372    pointing: &'a PointingPlan,
373    chunk_shape: &'a ChunkShape,
374}
375
376/// Convert partition `id` into dataset `name` of the destination: its
377/// pointing, then one traversal per selected group in lockstep, each batch
378/// of the same whole times.
379fn partition(
380    destination: &Destination<'_>,
381    observation: &dyn Observation,
382    id: PartitionId,
383    groups: &[String],
384    name: &str,
385    options: &ConversionOptions,
386) -> Result<Converted, ObservationError> {
387    let Destination {
388        set,
389        named,
390        pointing,
391        chunk_shape,
392    } = *destination;
393    let grid = observation.grid(id, &groups[0])?;
394    let entry = observation.catalogue().partition(id);
395    let extents = [
396        grid.times_s.len(),
397        grid.baselines.len(),
398        entry.frequencies_hz.len(),
399        entry.correlations.len(),
400    ];
401    let requested = chunk_shape(extents);
402    let chunks: [u64; 4] = std::array::from_fn(|axis| {
403        let extent = extents[axis].max(1) as u64;
404        match requested[axis] {
405            0 => extent,
406            chunk => chunk.min(extent),
407        }
408    });
409    let storage = Msv4StorageOptions {
410        dataset_name: name.to_string(),
411        creator: options.creator.clone(),
412        creation_date: options.creation_date.clone(),
413        chunks,
414        codec: options.codec,
415    };
416    let names: Vec<&str> = groups.iter().map(String::as_str).collect();
417    let mut writer = set.dataset(named, id, &names, &grid, &storage)?;
418    // The pointing of the antennas the partition's baselines name, as
419    // planned; dropped before the batches are read.
420    if let Some(read) = pointing.read_for(id) {
421        writer.write_pointing(named, &observation.pointing(read)?)?;
422    }
423    let budget = (options.batch_bytes, Some(writer.chunk_times()));
424    let read = lockstep::read(observation, id, groups, Roles::ALL, budget, |batches| {
425        writer.write_groups(batches)
426    })?;
427    Ok(Converted {
428        batches: read.batches,
429        samples: read.samples,
430        counters: writer.finish()?,
431    })
432}
