2//! The MeasurementSet task binaries on the shared task front end (#931).
3//!
4//! Each binary's `main` builds a [`TaskFrontEnd`] from its provider contract
5//! and a callback that reads the resolved run into its typed request; these
6//! are the reads and the reply the binaries share. The front end owns the
7//! command line, `--json-run`, Last and logging.
8//!
9//! [`TaskFrontEnd`]: casa_task_runtime::TaskFrontEnd
10
11use casa_provider_contracts::CasaValue;
12use casa_task_runtime::{TaskFailure, TaskInvocation, TaskSuccess};
13use serde::Serialize;
14
15use crate::FlagMerge;

17/// A MeasurementSet task's reply: `value` is what the task returns (`&()`
18/// for `None`), read as plain JSON until each task returns CASA's own
19/// types (#931 PR 4c), and `result` the provider's report, which a command
20/// line prints as JSON.
21pub fn ms_task_success(
22    value: &impl Serialize,
23    result: &impl Serialize,
24) -> Result<TaskSuccess, TaskFailure> {
25    let serialize =
26        |error: serde_json::Error| TaskFailure::execution(format!("serialize the result: {error}"));
27    let text = serde_json::to_string_pretty(result).map_err(serialize)?;
28    let value = serde_json::to_value(value).map_err(serialize)?;
29    Ok(TaskSuccess::new(CasaValue::from_plain_json(&value), result)?.with_text(text + "\n"))
30}

32/// Read `name`, a flag-version merge: `replace`, `or` or `and`, any case.
33pub fn task_flag_merge(task: &TaskInvocation<'_>, name: &str) -> Result<FlagMerge, TaskFailure> {
34    let text = task.required_text(name)?;
35    match text.to_ascii_lowercase().as_str() {
36        "replace" => Ok(FlagMerge::Replace),
37        "or" => Ok(FlagMerge::Or),
38        "and" => Ok(FlagMerge::And),
39        _ => Err(TaskFailure::validation(format!(
40            "parameter {name:?} must be replace, or or and, found {text:?}"
41        ))
42        .at(name)),
43    }
44}

46/// Read `name`, a CASA selection string; empty when unset.
47pub fn task_selection_text(task: &TaskInvocation<'_>, name: &str) -> Result<String, TaskFailure> {
48    Ok(task.text(name)?.unwrap_or_default())
