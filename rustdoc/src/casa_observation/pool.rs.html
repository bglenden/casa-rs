2//! The bounded pool owned batches are filled into, and the owned batch that
3//! gives its buffers back when it is dropped.
4
5use std::sync::{Arc, Condvar, Mutex, PoisonError};
6use std::time::{Duration, Instant};
7
8use crate::ObservationError;
9use crate::batch::VisibilityBatch;
10use crate::buffers::BatchBuffers;

12/// A bounded set of [`BatchBuffers`] that owned batches are filled into:
13/// at most `capacity` buffers of at most `envelope` bytes each, free or
14/// lent, whatever batches, roles and pieces they are filled with. That
15/// product is what the caller admits for the pool; the envelope for a
16/// traversal is its estimate's
17/// [`peak_buffer_bytes`](crate::Estimate::peak_buffer_bytes), and each live
18/// cursor's decode scratch is admitted separately.
19///
20/// [`Traversal::next_owned`](crate::Traversal::next_owned) takes a free
21/// buffer only for a batch, and waits while none is free: that wait is the
22/// backpressure a slow consumer applies to the producer. A [`FilledBatch`]
23/// gives its buffer back when it is dropped, so a batch shared through an
24/// `Arc` comes back when its last user lets go. Cloning the pool shares it,
25/// so several cursors, or a producer and the work units packing pieces, can
26/// draw on one bound.
27///
28/// Buffers are allocated as they are first filled and keep their capacity
29/// after that, within the envelope: a fill whose batch needs more is
30/// refused before anything is allocated, a fill frees what a buffer kept
31/// from earlier fills when keeping it would exceed the envelope, and a
32/// returned buffer over it is emptied.
33///
34/// A take waits for a free buffer without limit, unless the handle was made
35/// [`Self::waiting_at_most`] a duration: it then fails with
36/// [`ObservationError::PoolExhausted`] instead. Without a limit, a thread
37/// that takes from a pool must not also be the only holder of every batch
38/// the pool has lent, or it waits forever.
39#[derive(Clone, Debug)]
40pub struct BufferPool {
41    shared: Arc<PoolShared>,
42    wait: Option<Duration>,
43}
44
45#[derive(Debug)]
46struct PoolShared {
47    free: Mutex<Vec<BatchBuffers>>,
48    returned: Condvar,
49    capacity: usize,
50    envelope: usize,
51}
52
53impl BufferPool {
54    /// A pool of `capacity` buffers of at most `envelope` bytes each.
55    ///
56    /// # Panics
57    ///
58    /// If `capacity` is zero: no batch could ever be filled.
59    #[must_use]
60    pub fn new(capacity: usize, envelope: usize) -> Self {
61        assert!(capacity > 0, "a buffer pool holds at least one buffer");
62        Self {
63            shared: Arc::new(PoolShared {
64                free: Mutex::new(
65                    (0..capacity)
66                        .map(|_| BatchBuffers::with_envelope(envelope))
67                        .collect(),
68                ),
69                returned: Condvar::new(),
70                capacity,
71                envelope,
72            }),
73            wait: None,
74        }
75    }
76
77    /// A handle on the same buffers whose takes wait at most `limit` for a
78    /// free one (zero: never wait). A runtime that must stay cancellable
79    /// takes with a short limit and checks for cancellation between tries;
80    /// [`Traversal::next_owned`](crate::Traversal::next_owned) refused this
81    /// way has consumed nothing, so asking again delivers the same batch.
82    #[must_use]
83    pub fn waiting_at_most(&self, limit: Duration) -> Self {
84        Self {
85            shared: Arc::clone(&self.shared),
86            wait: Some(limit),
87        }
88    }
89
90    /// Buffers in the pool, free or lent.
91    #[must_use]
92    pub fn capacity(&self) -> usize {
93        self.shared.capacity
94    }
95
96    /// The most bytes one buffer holds.
97    #[must_use]
98    pub fn envelope(&self) -> usize {
99        self.shared.envelope
100    }
101
102    /// Buffers free now.
103    #[must_use]
104    pub fn free(&self) -> usize {
105        self.lock().len()
106    }
107
108    /// Bytes the free buffers keep, from their capacities.
109    #[must_use]
110    pub fn free_bytes(&self) -> usize {
111        self.lock().iter().map(BatchBuffers::capacity_bytes).sum()
112    }
113
114    /// Fill a free buffer, waiting for one, through `fill`; the buffer goes
115    /// back to the pool if `fill` fails or finishes no batch.
116    ///
117    /// # Errors
118    ///
119    /// [`ObservationError::PoolExhausted`] when the handle's wait limit
120    /// passed with no buffer free (`fill` is then not called); as `fill`
121    /// (which refuses a batch over the envelope through
122    /// [`BatchBuffers::begin`]); or [`ObservationError::InvalidBatch`] when
123    /// it returned without finishing a batch.
124    pub fn fill(
125        &self,
126        fill: impl FnOnce(&mut BatchBuffers) -> Result<(), ObservationError>,
127    ) -> Result<FilledBatch, ObservationError> {
128        let mut filled = FilledBatch {
129            buffers: self.take()?,
130            pool: self.clone(),
131        };
132        fill(&mut filled.buffers)?;
133        if filled.buffers.header().is_none() {
134            return Err(ObservationError::InvalidBatch {
135                reason: "the cursor reported a batch it did not finish".to_string(),
136            });
137        }
138        Ok(filled)
139    }
140
141    /// Copy `batch` (typically a compute window of a larger batch) into a
142    /// free buffer of this pool, waiting for one ([`BatchBuffers::pack`]):
143    /// a piece with storage of its own, and the payload elements copied.
144    ///
145    /// # Errors
146    ///
147    /// As [`BatchBuffers::pack`].
148    pub fn pack(
149        &self,
150        batch: &VisibilityBatch<'_>,
151    ) -> Result<(FilledBatch, usize), ObservationError> {
152        let mut copied = 0;
153        let filled = self.fill(|buffers| {
154            copied = buffers.pack(batch)?;
155            Ok(())
156        })?;
157        Ok((filled, copied))
158    }
159
160    fn lock(&self) -> std::sync::MutexGuard<'_, Vec<BatchBuffers>> {
161        // The buffers hold plain data: a panic elsewhere leaves them usable.
162        self.shared
163            .free
164            .lock()
165            .unwrap_or_else(PoisonError::into_inner)
166    }
167
168    fn take(&self) -> Result<BatchBuffers, ObservationError> {
169        let deadline = self.wait.map(|limit| Instant::now() + limit);
170        let mut free = self.lock();
171        loop {
172            if let Some(buffers) = free.pop() {
173                return Ok(buffers);
174            }
175            free = match deadline {
176                None => self
177                    .shared
178                    .returned
179                    .wait(free)
180                    .unwrap_or_else(PoisonError::into_inner),
181                Some(deadline) => {
182                    let left = deadline.saturating_duration_since(Instant::now());
183                    if left.is_zero() {
184                        return Err(ObservationError::PoolExhausted {
185                            capacity: self.shared.capacity,
186                        });
187                    }
188                    self.shared
189                        .returned
190                        .wait_timeout(free, left)
191                        .unwrap_or_else(PoisonError::into_inner)
192                        .0
193                }
194            };
195        }
196    }
197
198    fn give(&self, mut buffers: BatchBuffers) {
199        buffers.clear();
200        if buffers.capacity_bytes() > self.shared.envelope {
201            // Only a fill that failed after growing an array gets here.
202            buffers = BatchBuffers::with_envelope(self.shared.envelope);
203        }
204        self.lock().push(buffers);
205        self.shared.returned.notify_one();
206    }
207}

209/// A batch together with the pooled buffers it was filled into: an owned
210/// value a work unit can take, or share read-only through an `Arc`.
211///
212/// [`VisibilityBatch`] borrows its buffers, so it cannot outlive or leave
213/// the caller that filled them. A `FilledBatch` moves with its storage
214/// instead, which is how a batch's lifetime is separated from the
215/// traversal's: the cursor fills the next buffers while earlier batches are
216/// still in use elsewhere, a slow user keeps exactly its own batch's
217/// allocation alive, and dropping the batch returns that allocation to its
218/// [`BufferPool`].
219#[derive(Debug)]
220pub struct FilledBatch {
221    buffers: BatchBuffers,
222    pool: BufferPool,
223}
224
225impl FilledBatch {
226    /// The batch: views of the owned buffers and of the catalogue they were
227    /// checked against.
228    #[must_use]
229    pub fn batch(&self) -> VisibilityBatch<'_> {
230        self.buffers
231            .batch()
232            .expect("the pool hands out only buffers holding a finished batch")
233    }
234
235    /// Bytes the buffers hold, from their capacities.
236    #[must_use]
237    pub fn capacity_bytes(&self) -> usize {
238        self.buffers.capacity_bytes()
239    }
240}
241
242impl Drop for FilledBatch {
243    fn drop(&mut self) {
244        self.pool.give(std::mem::take(&mut self.buffers));
245    }
246}

269    /// Fill `times` times of the test catalogue's two baselines, two
270    /// channels and two products with `roles`' payload.
271    fn shaped(
272        buffers: &mut BatchBuffers,
273        times: usize,
274        roles: Roles,
275    ) -> Result<(), ObservationError> {
276        let header = BatchHeader {
277            partition: PartitionId(0),
278            data_group: 0,
279            times,
280            first_channel: 0,
281            channels: 2,
282            correlations: CorrelationAxis::All,
283            roles,
284            weights_per_channel: true,
285            weight_rule: crate::WeightRule::Stored,
286            interval: Sharing::PerSample,
287            exposure: Sharing::PerSample,
288        };
289        let mut fill = buffers.begin(header, &catalogue())?;
290        let arrays = fill.arrays();
291        let cell = times * 2 * 4;
292        arrays
293            .time_offsets_s
294            .extend((0..times).map(|time| time as f64));
295        arrays.field.resize(times, FieldId(0));
296        if roles.values {
297            arrays.values.resize(cell, Complex32::default());
298        }
299        if roles.weights {
300            arrays.weights.resize(cell, 1.0);
301        }
302        fill.finish()
303    }

305    #[test]
306    fn a_batch_returns_its_buffers_and_a_full_pool_waits_for_one() {
307        let pool = BufferPool::new(1, usize::MAX);
308        let first = pool
309            .fill(|buffers| fill(buffers, |_, _| {}))
310            .expect("first");
311        assert_eq!(pool.free(), 0);
312        assert_eq!(first.batch().samples(), 4);
313        let (sent, received) = mpsc::channel();
314        std::thread::scope(|scope| {
315            scope.spawn(|| {
316                let second = pool
317                    .fill(|buffers| fill(buffers, |_, _| {}))
318                    .expect("second");
319                sent.send(second.batch().samples()).expect("send");
320            });
321            std::thread::sleep(Duration::from_millis(20));
322            assert!(
323                received.try_recv().is_err(),
324                "the second fill waits for a buffer"
325            );
326            drop(first);
327            assert_eq!(received.recv().expect("second"), 4);
328        });
329        assert_eq!(pool.free(), 1, "every buffer is back");
330        assert!(
331            pool.free_bytes() > 0,
332            "returned buffers keep their capacity"
333        );
334    }

336    /// A buffer filled with one batch's values and then another's weights
337    /// keeps both allocations unless bounded. A pool buffer never holds more
338    /// than its envelope, lent or free, whatever sequence of roles and
339    /// shapes it serves; a batch over the envelope is refused before
340    /// anything is allocated.
341    #[test]
342    fn a_pool_buffer_never_holds_more_than_its_envelope() {
343        let values = Roles {
344            values: true,
345            ..NO_ROLES
346        };
347        let weights = Roles {
348            weights: true,
349            ..NO_ROLES
350        };
351        let alone = |times, roles| {
352            let mut buffers = BatchBuffers::default();
353            shaped(&mut buffers, times, roles).expect("fill");
354            buffers.capacity_bytes()
355        };
356        let (a, b) = (alone(16, values), alone(32, weights));
357        let mut unbounded = BatchBuffers::default();
358        shaped(&mut unbounded, 16, values).expect("values");
359        shaped(&mut unbounded, 32, weights).expect("weights");
360        assert!(
361            unbounded.capacity_bytes() > a.max(b),
362            "unbounded buffers keep both: {} B for fills of {a} B and {b} B",
363            unbounded.capacity_bytes()
364        );
365        let envelope = a.max(b);
366        let pool = BufferPool::new(1, envelope);
367        for (times, roles) in [(16, values), (32, weights), (16, values), (8, values)] {
368            let filled = pool
369                .fill(|buffers| shaped(buffers, times, roles))
370                .expect("fill");
371            assert!(
372                filled.capacity_bytes() <= envelope,
373                "lent: {times} {roles:?}"
374            );
375            let (piece, _) = BufferPool::new(1, envelope)
376                .pack(&filled.batch().channel_window(0..1).expect("window"))
377                .expect("pack");
378            assert!(piece.capacity_bytes() <= envelope);
379            drop(filled);
380            assert!(pool.free_bytes() <= envelope, "free: {times} {roles:?}");
381        }
382        let over = pool.fill(|buffers| shaped(buffers, 64, values));
383        assert!(matches!(over, Err(ObservationError::InvalidRequest { .. })));
384        assert!(pool.free_bytes() <= envelope);
385        assert_eq!(pool.free(), 1);
386    }

388    /// A handle with a wait limit fails when every buffer is held, on the
389    /// thread that holds them, instead of waiting forever; with one back,
390    /// it fills.
391    #[test]
392    fn a_limited_take_reports_an_exhausted_pool() {
393        let pool = BufferPool::new(2, usize::MAX);
394        let limited = pool.waiting_at_most(Duration::from_millis(10));
395        let held: Vec<FilledBatch> = (0..2)
396            .map(|_| {
397                limited
398                    .fill(|buffers| fill(buffers, |_, _| {}))
399                    .expect("fill")
400            })
401            .collect();
402        let started = Instant::now();
403        let mut called = false;
404        assert!(matches!(
405            limited.fill(|_| {
406                called = true;
407                Ok(())
408            }),
409            Err(ObservationError::PoolExhausted { capacity: 2 })
410        ));
411        assert!(!called, "nothing is filled");
412        assert!(started.elapsed() < Duration::from_secs(5));
413        assert!(matches!(
414            pool.waiting_at_most(Duration::ZERO)
415                .fill(|buffers| fill(buffers, |_, _| {})),
416            Err(ObservationError::PoolExhausted { .. })
417        ));
418        drop(held);
419        assert!(limited.fill(|buffers| fill(buffers, |_, _| {})).is_ok());
420    }

423    fn a_failed_fill_gives_the_buffer_back() {
424        let pool = BufferPool::new(1, usize::MAX);
425        assert!(
426            pool.fill(|buffers| fill(buffers, |_, a| {
427                a.flags.pop();
428            }))
429            .is_err()

435        assert_eq!(pool.free(), 1);
436        let packed = {
437            let source = pool.fill(|buffers| fill(buffers, |_, _| {})).expect("fill");
438            let pieces = BufferPool::new(1, usize::MAX);
439            let (piece, copied) = pieces
440                .pack(&source.batch().channel_window(0..1).expect("window"))
441                .expect("pack");
442            assert!(std::sync::Arc::ptr_eq(
443                piece.batch().catalogue,
444                source.batch().catalogue
445            ));
446            copied
447        };
448        assert_eq!(packed, 8);
449        assert_eq!(pool.free(), 1);
