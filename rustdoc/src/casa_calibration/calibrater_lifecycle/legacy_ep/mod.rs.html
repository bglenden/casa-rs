69    pub(super) fn new(
70        table: &Path,
71        cache: &Path,
72        angle_increment_deg: f64,
73        (interval, preaverage_seconds): (&str, f64),
74        mode: CalibrationOutputMode,
75        reference: RefAntSelector,
76    ) -> MsResult<(Self, String)> {
77        if !angle_increment_deg.is_finite() {
78            return Err(invalid("EP painc must be a finite angle"));
79        }
80        let cache = EpCache::open(cache)?;
81        let interval_quantity = Quantity::read(interval).map_err(invalid)?;
82        let (interval_seconds, interval_label) = interval_quantity
83            .get_value_in(&Unit::new("s").expect("known unit"))
84            .map_or_else(
85                |_| (interval_quantity.value(), format!("{interval}s")),
86                |seconds| (seconds, interval.into()),
87            );
88        if !interval_seconds.is_finite() {
89            return Err(invalid("EP solint must be finite"));
90        }
91        let solver = Self {
92            table: table.to_owned(),
93            cache,
94            angle_increment_deg,
95            interval_seconds,
96            preaverage_seconds: if preaverage_seconds < 0. {
97                if interval_seconds > 0. {
98                    interval_seconds
99                } else {
100                    f64::MAX
101                }
102            } else {
103                preaverage_seconds
104            },
105            mode,
106            reference,
107            model: None,
108        };
109        Ok((solver, interval_label))
110    }

112    pub(super) fn set_model(
113        &mut self,
114        attachment: &CalibrationAttachment,
115        path: &Path,
116    ) -> Result<(), AttachmentError> {
117        if !path.is_dir() {
118            return Err(invalid(format!("Table {} does not exist", path.display())).into());
119        }
120        let image = PagedImage::<f32>::open(path).map_err(invalid)?;
121        let (shape, cell_rad, coordinates) = model_geometry(&image)?;
122        let host = HostResources::detect().map_err(invalid)?;
123        let memory = reserve_model(&host, shape[0])?;
124        let pixels = image.get().map_err(invalid)?;
125        if pixels.iter().any(|value| !value.is_finite()) {
126            return Err(invalid("EP model contains non-finite IQUV values").into());
127        }
128        let buffer = attachment.with_write(|ms, selection, _| Ok(first_buffer(ms, selection)?))?;
129        match buffer.correlations.as_slice() {
130            [StokesType::RR | StokesType::LL]
131            | [StokesType::RR, StokesType::LL]
132            | [StokesType::LL, StokesType::RR] => {}
133            _ => return Err(invalid("legacy EP needs circular parallel-hand data").into()),
134        }
135        let products = convolution(
136            &self.cache,
137            (self.angle_increment_deg, 1),
138            &host,
139            &buffer,
140            shape,
141            cell_rad,
142            &coordinates,
143        )?;
144        let fourier =
145            LegacyEpConvolution::prepare_model(&pixels, &products.average_beam, products.sampling)
146                .map_err(invalid)?;
147        tracing::debug!(table = %self.table.display(), support = products.support[0],
148            convolution_samples = products.convolution.len(), weight_samples = products.weights.len(),
149            "legacy EP prepared its Fourier model");
150        self.model = Some(PreparedEpModel {
151            fourier,
152            products: Arc::new(products),
153            correlation_planes: buffer.correlations,
154            side: shape[0],
155            cell_rad,
156            scale: cell_rad.map(|increment| increment * shape[0] as f64),
157            reference_frequency_hz: buffer.frequency_hz,
158            initial_angle_rad: buffer.angle_rad,
159            coordinates,
160            _memory: memory,
161        });
162        Ok(())
163    }

166pub(crate) fn convolution(
167    cache: &EpCache,
168    (angle_increment_deg, w_planes): (f64, usize),
169    host: &HostResources,
170    buffer: &FirstBuffer,
171    shape: [usize; 4],
172    cell_rad: [f64; 2],
173    beam_coordinates: &CoordinateSystem,
174) -> MsResult<EpCacheProducts> {
175    let read = || {
176        cache.read(
177            host,
178            buffer.angle_rad,
179            angle_increment_deg,
180            &shape,
181            &buffer.correlations,
182            w_planes,
183        )
184    };
185    {
186        let _lock = NativeCacheLock::shared(cache.path()).map_err(invalid)?;
187        if let Some(products) = read()? {
188            return Ok(products);
189        }
190    }
191    let _lock = NativeCacheLock::exclusive(cache.path()).map_err(invalid)?;
192    cache.recover()?;
193    if let Some(products) = read()? {
194        return Ok(products);
195    }
196    let screen_memory = admit(
197        host,
198        &ResourcePolicy::Balanced,
199        &Demand {
200            phase: "legacy EP convolution screen",
201            memory: LegacyEpConvolution::SCREEN_WORKSPACE_BYTES,
202        },
203    )
204    .map_err(invalid)?;
205    let aperture = EvlaApertureModel::new(legacy_vla_surface()?);
206    let (products, kernel_memory, w_scale) = if w_planes == 1 {
207        let products = LegacyEpConvolution::vla_l_band(
208            &aperture,
209            shape[0],
210            cell_rad[0],
211            buffer.frequency_hz,
212            buffer.angle_rad,
213        )
214        .map_err(invalid)?;
215        (products, None, None)
216    } else {
217        let plan = LegacyEpWPlanePlan::new(
218            &aperture,
219            shape[0],
220            cell_rad,
221            buffer.frequency_hz,
222            buffer.angle_rad,
223            w_planes,
224        )
225        .map_err(invalid)?;
226        let memory = admit(
227            host,
228            &ResourcePolicy::Balanced,
229            &Demand {
230                phase: "legacy EP cold convolution cells",
231                memory: plan.retained_kernel_bytes(),
232            },
233        )
234        .map_err(invalid)?;
235        let w_scale = plan.w_scale();
236        (plan.build().map_err(invalid)?, Some(memory), Some(w_scale))
237    };
238    cache.publish(
239        host,
240        (buffer.angle_rad, cell_rad),
241        beam_coordinates,
242        shape,
243        &buffer.correlations,
244        &products,
245    )?;
246    drop((products, kernel_memory, screen_memory));
247    // The persisted angle is single precision and written as decimal.
248    let mut products = cache
249        .read(
250            host,
251            buffer.angle_rad,
252            angle_increment_deg.max(0.001),
253            &shape,
254            &buffer.correlations,
255            w_planes,
256        )?
257        .ok_or_else(|| invalid("legacy EP generated an unusable convolution cache"))?;
258    products.w_scale = w_scale;
259    Ok(products)
260}

262pub(crate) fn reserve_model(host: &HostResources, side: usize) -> MsResult<Reservation> {
263    // Model and average-beam pixels, four retained Fourier grids, and
264    // the image-sized Jones, power and FFT workspace of the average beam.
265    let memory = (side as u64)
266        .saturating_mul(side as u64)
267        .saturating_mul(128)
268        .saturating_add(4 << 20);
269    admit(
270        host,
271        &ResourcePolicy::Balanced,
272        &Demand {
273            phase: "legacy EP model",
274            memory,
275        },
276    )
277    .map_err(invalid)
278}

280pub(crate) fn model_geometry(
281    image: &PagedImage<f32>,
282) -> MsResult<([usize; 4], [f64; 2], CoordinateSystem)> {
283    let [side, ny, 4, 1] = *image.shape() else {
284        return Err(invalid("legacy EP needs a single-channel IQUV image"));
285    };
286    if side != ny || side < 4 || !side.is_multiple_of(2) {
287        return Err(invalid(
288            "legacy EP needs a square image with even pixel lengths",
289        ));
290    }
291    let mut coordinates = image.coordinates().clone();
292    if coordinates.n_coordinates() != 3 {
293        return Err(invalid(
294            "legacy EP needs direction, IQUV and frequency coordinates",
295        ));
296    }
297    let CoordinateModel::Direction(direction) = coordinates.coordinate(0) else {
298        return Err(invalid("legacy EP model is missing direction axes"));
299    };
300    let cell_rad = direction
301        .increment()
302        .try_into()
303        .map_err(|_| invalid("EP direction coordinate must have two increments"))?;
304    let [dx, dy]: [f64; 2] = cell_rad;
305    if !dx.is_finite() || !dy.is_finite() || dx == 0. || (dx.abs() - dy.abs()).abs() > 1e-15 {
306        return Err(invalid("legacy EP needs square angular pixels"));
307    }
308    let circular = [
309        StokesType::RR,
310        StokesType::RL,
311        StokesType::LR,
312        StokesType::LL,
313    ];
314    if !matches!(coordinates.coordinate(1), CoordinateModel::Stokes(stokes)
315            if stokes.stokes() == [StokesType::I, StokesType::Q, StokesType::U, StokesType::V])
316        || coordinates.find_coordinate(CoordinateType::Spectral) != Some(2)
317    {
318        return Err(invalid(
319            "legacy EP needs direction, IQUV and frequency coordinates",
320        ));
321    }
322    coordinates.replace_coordinate(1, StokesCoordinate::new(circular.to_vec()));
323    Ok(([side, ny, 4, 1], cell_rad, coordinates))
324}

326pub(crate) fn first_buffer(
327    ms: &MeasurementSet,
328    selected: &ResolvedMsSelection,
329) -> MsResult<FirstBuffer> {
330    let telescope = ms.observation()?.string(0, "TELESCOPE_NAME")?;
331    if telescope != "VLA" {
332        return Err(invalid(
333            "legacy EP aperture model is implemented for the (E)VLA only",
334        ));
335    }
336    let row = first_cross_row(ms, selected)?;
337    let codes = ms.polarization()?.corr_type(row.polarization_id as usize)?;
338    let circular = [
339        StokesType::RR,
340        StokesType::RL,
341        StokesType::LR,
342        StokesType::LL,
343    ];
344    let correlations = codes
345        .iter()
346        .map(|&code| {
347            circular
348                .iter()
349                .find(|value| **value as i32 == code)
350                .copied()
351                .ok_or_else(|| invalid("legacy EP requires circular correlations"))
352        })
353        .collect::<MsResult<Vec<_>>>()?;
354    if correlations.is_empty() {
355        return Err(invalid("legacy EP first buffer has no correlations"));
356    }
357    let spw = row.spectral_window_id;
358    let frequencies = ms.spectral_window()?.chan_freq(spw as usize)?;
359    let channels = match selected.channel_selections.get(&spw) {
360        Some(selection) => selection.indices(frequencies.len())?,
361        None => (0..frequencies.len()).collect(),
362    };
363    let frequency_hz = channels
364        .into_iter()
365        .map(|channel| frequencies[channel])
366        .fold(0_f64, f64::max);
367    if !(1.0e9..=2.0e9).contains(&frequency_hz) {
368        return Err(invalid(
369            "legacy EP aperture model requires VLA L-band observations",
370        ));
371    }
372    let engine = MsCalEngine::new(ms)?;
373    let angle_rad =
374        f64::from(
375            engine.feed_parallactic_angle(row.time_mjd_seconds, row.field_id as usize, 0)? as f32,
376        );
377    Ok(FirstBuffer {
378        correlations,
379        frequency_hz,
380        angle_rad,
381    })
382}

384fn first_cross_row<'a>(
385    ms: &MeasurementSet,
386    selected: &'a ResolvedMsSelection,
387) -> MsResult<&'a ResolvedMsSelectionRow> {
388    let mut first = None;
389    for rows in selected.selected_rows.chunks(8192) {
390        let indices = rows.iter().map(|row| row.row_index).collect::<Vec<_>>();
391        let columns = ms
392            .main_table()
393            .required_scalar_columns_owned_for_rows(&["ARRAY_ID"], &indices)?;
394        let Some(RequiredScalarColumnValues::Int32(arrays)) = columns.get("ARRAY_ID") else {
395            return Err(invalid("legacy EP requires ARRAY_ID"));
396        };
397        for (row, &array) in rows.iter().zip(arrays) {
398            let time = row.time_mjd_seconds;
399            let key = (array, row.field_id, row.data_desc_id, time.to_bits());
400            if first.as_ref().is_none_or(|(old, _)| key < *old) {
401                first = Some((key, None));
402            }
403            if let Some((held, cross)) = first.as_mut()
404                && key == *held
405                && cross.is_none()
406                && !row.flag_row
407                && row.antenna1_id != row.antenna2_id
408            {
409                *cross = Some(row);
410            }
411        }
412    }
413    first.and_then(|(_, cross)| cross).ok_or_else(|| {
414        invalid("legacy EP needs an unflagged cross-correlation in its first visibility buffer")
415    })
416}

418pub(crate) fn legacy_vla_surface() -> MsResult<EvlaDishSurface> {
419    use std::io::Read;
420    let relative = "nrao/VLA/VLA.surface";
421    let path = std::env::var_os("CASARS_EP_VLA_SURFACE")
422        .map(PathBuf::from)
423        .or_else(|| {
424            [
425                std::env::var_os("CASA_RS_MEASURESPATH").map(PathBuf::from),
426                std::env::var_os("HOME").map(|root| PathBuf::from(root).join(".casa/data")),
427            ]
428            .into_iter()
429            .flatten()
430            .map(|root| root.join(relative))
431            .find(|path| path.is_file())
432        })
433        .ok_or_else(|| {
434            invalid("legacy EP needs nrao/VLA/VLA.surface from the CASA runtime data")
435        })?;
436    let mut text = String::new();
437    std::fs::File::open(path)
438        .and_then(|file| file.take(1_048_577).read_to_string(&mut text))
439        .map_err(invalid)?;
440    if text.len() > 1_048_576 {
441        return Err(invalid("legacy EP aperture surface exceeds 1 MiB"));
442    }
443    EvlaDishSurface::from_surface_text(&text).map_err(invalid)
444}

507    fn compare_retained(
508        label: &str,
509        model: &PreparedEpModel,
510        dataset: &Path,
511        image: &Path,
512        selection: &casa_ms::MsSelection,
513    ) {
514        let stored = PagedImage::<f32>::open(image).unwrap();
515        let (shape, increment, _) = model_geometry(&stored).unwrap();
516        let ms = MeasurementSet::open(dataset).unwrap();
517        let selected = resolve_casa_calibration_selection(&ms, selection).unwrap();
518        let first = first_cross_row(&ms, &selected).unwrap();
519        let channels = ms
520            .spectral_window()
521            .unwrap()
522            .chan_freq(first.spectral_window_id as usize)
523            .unwrap();
524        let rows = selected
525            .selected_rows
526            .iter()
527            .filter(|row| {
528                row.field_id == first.field_id
529                    && row.data_desc_id == first.data_desc_id
530                    && row.time_mjd_seconds == first.time_mjd_seconds
531                    && !row.flag_row
532                    && row.antenna1_id != row.antenna2_id
533            })
534            .map(|row| row.row_index)
535            .collect::<Vec<_>>();
536        let values = |column| {
537            ms.main_table()
538                .column_accessor(column)
539                .unwrap()
540                .array_cells_owned_uncached(&rows)
541                .unwrap()
542        };
543        let mut norm = 0.;
544        let mut difference = 0.;
545        let mut samples = 0;
546        let mut imaginary = 0_f32;
547        for (uvw, stored) in values("UVW").into_iter().zip(values("MODEL_DATA")) {
548            let Some(ArrayValue::Float64(uvw)) = uvw else {
549                panic!("UVW")
550            };
551            let Some(ArrayValue::Complex32(expected)) = stored else {
552                panic!("MODEL_DATA")
553            };
554            let [u, v, _] = uvw.iter().copied().collect::<Vec<_>>().try_into().unwrap();
555            let wavelength = 299_792_458. / channels[0];
556            let position = [u, v].map(|value| -value / wavelength);
557            for channel in 1..channels.len() {
558                assert!(
559                    expected
560                        .index_axis(ndarray::Axis(1), channel)
561                        .iter()
562                        .all(|v| v.norm() == 0.)
563                );
564            }
565            for (correlation, polarization) in model.correlation_planes.iter().enumerate() {
566                let actual = evaluate_sample(
567                    model,
568                    shape[0],
569                    correlation,
570                    *polarization,
571                    [position[0] * increment[0], position[1] * increment[1]],
572                );
573                let expected = expected[[correlation, 0]];
574                difference += f64::from((actual - expected).norm_sqr());
575                norm += f64::from(expected.norm_sqr());
576                imaginary = imaginary.max(expected.im.abs());
577                samples += 1;
578            }
579        }
580        assert!(samples > 600 && imaginary > 0.1);
581        let error = (difference / norm).sqrt();
582        eprintln!(
583            "legacy EP {label}: first buffer {samples} complex samples, normalized RMS {error:.9e}"
584        );
585        assert!(error < 1e-3, "{error}");
586    }

588    fn evaluate_sample(
589        model: &PreparedEpModel,
590        side: usize,
591        correlation: usize,
592        polarization: StokesType,
593        position: [f64; 2],
594    ) -> Complex32 {
595        let centre = side / 2 + 1;
596        let position = position.map(|value| (value * side as f64 + centre as f64) as f32);
597        let loc = position.map(|value| value.round() as isize - 1);
598        let products = &model.products;
599        let sampling = products.sampling as isize;
600        let subpixel = std::array::from_fn::<_, 2, _>(|axis| {
601            (((loc[axis] + 1) as f32 - position[axis]) * sampling as f32).round() as isize
602        });
603        let support = products.support[0] as isize;
604        if loc
605            .iter()
606            .any(|&value| value < support || value + support >= side as isize)
607        {
608            return Complex32::default();
609        }
610        let first = products.convolution.shape()[0] as isize / 2;
611        let grid = &model.fourier[polarization as usize - StokesType::RR as usize];
612        let mut value = Complex32::default();
613        let mut norm = Complex32::default();
614        for y in -support..=support {
615            for x in -support..=support {
616                let kx = (first + x * sampling + subpixel[0]) as usize;
617                let ky = (first + y * sampling + subpixel[1]) as usize;
618                let weight = products.convolution[[kx, ky, correlation, 0]];
619                value += weight * grid[(loc[0] + x) as usize + side * (loc[1] + y) as usize];
620                norm += weight;
621            }
622        }
623        value / norm
624    }
