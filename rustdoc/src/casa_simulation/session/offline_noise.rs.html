2//! Prepare an additive noise field once; optional A Noise output uses that field.
3use super::corruption::calibration_table_root;
4use super::error::{SimulatorError, SimulatorResult};
5use super::plan::NoiseTerm;
6use super::{NoiseSpec, Simulator};
7use crate::corrupt::NoiseLevel;
8use crate::corrupt::jones::TermBytes;
9use crate::corrupt::offline::OfflineNoise;
10use casa_calibration::{simulated_table_bytes, write_simulated_noise_table};
11use casa_imaging_runtime::{Demand, admit};
12use casa_ms::MeasurementSet;
13use std::sync::Arc;
14
15impl Simulator {
16    pub(super) fn set_offline_noise(
17        &mut self,
18        spec: &NoiseSpec,
19        level: NoiseLevel,
20        domain: Arc<crate::corrupt::noise_domain::NoiseDomain>,
21        snapshot: Option<Arc<crate::corrupt::tsys_snapshot::TsysSnapshot>>,
22    ) -> SimulatorResult<()> {
23        self.require_rows()?;
24        let path = calibration_table_root(&spec.table)
25            .map(|root| self.table_path(&format!("{root}.A.cal")))
26            .transpose()?
27            .flatten();
28        let _dimensions = self.admit_term(TermBytes {
29            kept: 0,
30            forming: (self.spectral_windows.len() * 32 + 1024) as u64,
31        })?;
32        let (mut counts, _) = self.integration_counts();
33        counts.resize(self.spectral_windows.len(), 0);
34        let channels = self
35            .spectral_windows
36            .iter()
37            .map(|w| w.channel_count)
38            .collect::<Vec<_>>();
39        let kept = OfflineNoise::bytes(self.antennas.len(), &counts, &channels);
40        let reservation = self.admit_term(TermBytes { kept, forming: 0 })?;
41        let mut field = OfflineNoise::new(
42            self.antennas.len(),
43            &counts,
44            &channels,
45            reservation,
46            domain.metadata(),
47        );
48        let sigma = if let NoiseLevel::Simple(sigma) = &level {
49            *sigma
50        } else {
51            1.
52        };
53        for (index, scan) in self.scans.iter().enumerate() {
54            for time in self.integration_times_s(index) {
55                field.push(
56                    scan.spectral_window_id,
57                    time,
58                    scan.field_id as i32,
59                    self.seed,
60                    sigma,
61                );
62            }
63        }
64        let field = Arc::new(field.sorted());
65        if let Some(path) = path {
66            // Active terms and recorded applications own their fields separately.
67            // Only the latest snapshot is written to a given destination.
68            self.offline_tables.retain(|(saved, _)| saved != &path);
69            self.offline_tables.push((path, Arc::clone(&field)));
70        }
71        self.noise_terms.push(NoiseTerm {
72            domain,
73            snapshot,
74            seed: self.seed,
75            level: match level {
76                NoiseLevel::Simple(_) => NoiseLevel::Fixed(field),
77                scale => NoiseLevel::ScaledFixed {
78                    field,
79                    scale: Arc::new(scale),
80                },
81            },
82            applications: 0,
83        });
84        Ok(())
85    }
86
87    pub(super) fn write_offline_tables(&self) -> SimulatorResult<()> {
88        let Some(memory) = self
89            .offline_tables
90            .iter()
91            .map(|(_, term)| simulated_table_bytes(term.table_rows(), term.table_values()))
92            .max()
93        else {
94            return Ok(());
95        };
96        let resources = &self.options.resources;
97        let _building = admit(
98            resources.host(),
99            resources.policy(),
100            &Demand {
101                phase: "writing an additive simulation noise table",
102                memory,
103            },
104        )
105        .map_err(SimulatorError::admission)?;
106        let ms = MeasurementSet::open(self.path())
107            .map_err(|error| SimulatorError::raised(error.to_string()))?;
108        for (path, field) in &self.offline_tables {
109            write_simulated_noise_table(&ms, path, field.solutions(), Some(field.metadata))
110                .map_err(|error| SimulatorError::raised(format!("write A Noise table: {error}")))?;
111        }
112        Ok(())
113    }
114}
