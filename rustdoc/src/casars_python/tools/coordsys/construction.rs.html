21fn raised(message: impl Into<String>) -> PyErr {
22    PyRuntimeError::new_err(message.into())
23}

25pub(super) fn string_arguments(
26    value: Option<&Bound<'_, PyAny>>,
27    argument: &str,
28) -> PyResult<Vec<String>> {
29    let Some(value) = value
30        .filter(|value| value.is_instance_of::<PyList>() || value.is_instance_of::<PyString>())
31    else {
32        return Ok(Vec::new());
33    };
34    value
35        .extract::<StringList>()
36        .map(|names| names.0)
37        .map_err(|_| PyTypeError::new_err(format!("list _{argument} must contain strings")))
38}

40fn stokes_coordinate(names: Vec<String>) -> PyResult<Option<StokesCoordinate>> {
41    let names = match names.as_slice() {
42        [name] if name.is_empty() || name == " " => return Ok(None),
43        [name] => name
44            .split(if name.contains(',') { ',' } else { ' ' })
45            .filter(|value| !value.is_empty())
46            .map(str::to_string)
47            .collect::<Vec<_>>(),
48        _ => names,
49    };
50    let values = stokes_values(names.iter().map(String::as_str))?;
51    Ok((!values.is_empty()).then(|| StokesCoordinate::new(values)))
52}

54pub(super) fn stokes_values<'a>(
55    names: impl IntoIterator<Item = &'a str>,
56) -> PyResult<Vec<StokesType>> {
57    let values = names
58        .into_iter()
59        .map(|name| StokesType::from_name(name).unwrap_or(StokesType::Undefined))
60        .collect::<Vec<_>>();
61    if values
62        .iter()
63        .enumerate()
64        .any(|(index, value)| values[..index].contains(value))
65    {
66        return Err(raised("You have specified the same Stokes more than once"));
67    }
68    Ok(values)
69}

71fn native_coordinate_field(name: &str) -> Option<(&str, usize)> {
72    let offset = name.find(|character: char| character.is_ascii_digit())?;
73    let (kind, index) = name.split_at(offset);
74    let index = index.parse().ok()?;
75    match kind {
76        "direction" | "spectral" | "stokes" | "linear" | "tabular" | "quality" | "coordsys"
77        | "coordinate" => Some((kind, index)),
78        _ => None,
79    }
80}

82pub(in crate::tools) fn validate_coordinate_indices(record: &RecordValue) -> PyResult<usize> {
83    let mut indices = Vec::new();
84    for field in record.fields() {
85        let Some((_, index)) = native_coordinate_field(&field.name) else {
86            continue;
87        };
88        indices.push(index);
89    }
90    indices.sort_unstable();
91    if !indices.iter().copied().eq(0..indices.len()) {
92        return Err(raised(
93            "Coordinate record numbering must be contiguous and unique",
94        ));
95    }
96    Ok(indices.len())
97}

99/// Append a coordinate and its axis-mapping fields, keeping the existing record intact.
100pub(in crate::tools) fn append_coordinate(
101    coordinates: &mut CoordinateSystem,
102    record: &mut RecordValue,
103    coordinate: impl Into<CoordinateModel>,
104) -> PyResult<()> {
105    let coordinate = coordinate.into();
106    let index = coordinates.n_coordinates();
107    let first_pixel = coordinates.n_pixel_axes() as i32;
108    let first_world = coordinates.n_world_axes() as i32;
109    let n_pixel = coordinate.n_pixel_axes();
110    let n_world = coordinate.n_world_axes();
111    let replacements = coordinate
112        .to_world(&vec![0.0; n_pixel])
113        .map_err(|error| raised(error.to_string()))?;
114    let kind = coordinate.coordinate_type();
115    let name = kind.record_prefix();
116    let mut axis_record = coordinate.to_record();
117    match kind {
118        CoordinateType::Spectral => {
119            axis_record.upsert("conversion", default_spectral_conversion().into());
120        }
121        CoordinateType::Linear => {
122            axis_record.remove("coordinate_type");
123            axis_record.remove("naxes");
124        }
125        _ => {}
126    }
127    record.upsert(format!("{name}{index}"), axis_record.into());
128    record.upsert(
129        format!("pixelmap{index}"),
130        ArrayValue::from_i32_vec((first_pixel..first_pixel + n_pixel as i32).collect()).into(),
131    );
132    record.upsert(
133        format!("worldmap{index}"),
134        ArrayValue::from_i32_vec((first_world..first_world + n_world as i32).collect()).into(),
135    );
136    record.upsert(
137        format!("pixelreplace{index}"),
138        ArrayValue::from_f64_vec(vec![0.0; n_pixel]).into(),
139    );
140    record.upsert(
141        format!("worldreplace{index}"),
142        ArrayValue::from_f64_vec(replacements).into(),
143    );
144    coordinates.add_coordinate(coordinate);
145    Ok(())
146}

148impl CoordinateState {
149    fn append(&mut self, coordinate: impl Into<CoordinateModel>) -> PyResult<()> {
150        append_coordinate(&mut self.coordinates, &mut self.record, coordinate)
151    }
152
153    fn add_defaults(
154        &mut self,
155        direction: bool,
156        spectral: bool,
157        stokes: Vec<String>,
158        linear: i32,
159        tabular: bool,
160    ) -> PyResult<()> {
161        validate_coordinate_indices(&self.record)?;
162        if direction {
163            self.append(defaults::direction_coordinate())?;
164        }
165        if let Some(stokes) = stokes_coordinate(stokes)? {
166            self.append(stokes)?;
167        }
168        if spectral {
169            self.append(defaults::spectral_coordinate())?;
170        }
171        if linear > 0 {
172            let names = (0..linear)
173                .map(|axis| format!("LinAxis{axis}"))
174                .collect::<Vec<_>>();
175            self.append(defaults::linear_coordinate(&names, &[]))?;
176        }
177        if tabular {
178            self.append(TabularCoordinate::linear(0.0, 1.0, 0.0, "TabAxis1", "km"))?;
179        }
180        Ok(())
181    }
182}

184fn evla_observation() -> ObsInfo {
185    let (sin_lon, cos_lon) = (-1.878_288_434_411_257_6_f64).sin_cos();
186    let (sin_lat, cos_lat) = (0.591_675_343_072_337_6_f64).sin_cos();
187    let radius = 6_373_580.000_000_001_f64;
188    current_epoch(
189        ObsInfo::new("EVLA")
190            .with_observer("Karl Jansky")
191            .with_telescope_position(MPosition::new_itrf(
192                cos_lon * cos_lat * radius,
193                sin_lon * cos_lat * radius,
194                sin_lat * radius,
195            )),
196    )
197}

199fn normalize_stokes_records(record: &mut RecordValue) -> PyResult<()> {
200    for field in record.fields_mut() {
201        let Some((kind, _)) = native_coordinate_field(&field.name) else {
202            continue;
203        };
204        let Value::Record(subrecord) = &mut field.value else {
205            continue;
206        };
207        let declared_system = matches!(subrecord.get("coordinate_type"),
208            Some(Value::Scalar(ScalarValue::String(kind))) if kind == "System");
209        if kind == "coordsys" || declared_system {
210            *subrecord = restore(std::mem::take(subrecord), "unknown".into())?.record;
211            if declared_system {
212                subrecord.upsert(
213                    "coordinate_type",
214                    Value::Scalar(ScalarValue::String("System".into())),
215                );
216            }
217            continue;
218        }
219        if kind != "stokes"
220            && !matches!(subrecord.get("coordinate_type"), Some(Value::Scalar(ScalarValue::String(kind))) if kind == "Stokes")
221        {
222            continue;
223        }
224        let Some(Value::Array(ArrayValue::String(values))) = subrecord.get("stokes") else {
225            return Err(raised("Incorrect data type for field stokes"));
226        };
227        if values.ndim() != 1 || values.is_empty() {
228            return Err(raised("The stokes field must be a nonempty vector"));
229        }
230        let values = stokes_values(values.iter().map(String::as_str))?;
231        subrecord.upsert(
232            "stokes",
233            ArrayValue::from_string_vec(
234                values
235                    .into_iter()
236                    .map(|value| value.name().into())
237                    .collect(),
238            )
239            .into(),
240        );
241    }
242    Ok(())
243}

245fn replacement(value: &Value, key: &str) -> PyResult<Value> {
246    let values = match value {
247        Value::Array(ArrayValue::Float64(values)) if values.ndim() == 1 => {
248            return Ok(value.clone());
249        }
250        Value::Array(ArrayValue::Float32(values)) if values.ndim() == 1 => {
251            values.iter().copied().map(f64::from).collect()
252        }
253        Value::Array(ArrayValue::Int32(values)) if values.ndim() == 1 => {
254            values.iter().copied().map(f64::from).collect()
255        }
256        Value::Array(ArrayValue::Int64(values)) if values.ndim() == 1 => {
257            values.iter().map(|value| *value as f64).collect()
258        }
259        _ => {
260            return Err(raised(format!(
261                "Incorrect data type for field {key}, expected a numeric vector"
262            )));
263        }
264    };
265    Ok(ArrayValue::from_f64_vec(values).into())
266}

268pub(super) fn restore(mut record: RecordValue, parent_name: String) -> PyResult<CoordinateState> {
269    validate_coordinate_indices(&record)?;
270    normalize_stokes_records(&mut record)?;
271
272    let mut coordinates =
273        CoordinateSystem::from_record(&record).map_err(|error| raised(error.to_string()))?;
274    validate_restored_linear_coordinates(&coordinates, &record)?;
275    let info = coordinates.obs_info_mut();
276    if record.get("observer").is_none() && info.observer.is_empty() {
277        info.observer = "UNKNOWN".into();
278    }
279    if record.get("telescope").is_none() && info.telescope.is_empty() {
280        info.telescope = "UNKNOWN".into();
281    }
282    if info.date.is_none() {
283        info.date = Some(MEpoch::from_mjd(0.0, EpochRef::LAST));
284    }
285
286    let placements = coordinates
287        .coordinates_in_record(&record)
288        .map_err(|error| raised(error.to_string()))?;
289    let mut restored = coordinates.obs_info().to_record();
290    for (model_index, placement) in placements.into_iter().enumerate() {
291        let field = placement.field;
292        let index = field.trim_start_matches(|character: char| !character.is_ascii_digit());
293        for prefix in ["pixelmap", "worldmap", "pixelreplace", "worldreplace"] {
294            let key = format!("{prefix}{index}");
295            let value = record
296                .get(&key)
297                .ok_or_else(|| raised(format!("RecordInterface: unknown field {key}")))?;
298            let value = if prefix.ends_with("replace") {
299                replacement(value, &key)?
300            } else {
301                value.clone()
302            };
303            restored.push(RecordField::new(key, value));
304        }
305        let model = coordinates.coordinate(model_index);
306        let value = if matches!(
307            model,
308            CoordinateModel::Stokes(_) | CoordinateModel::Quality(_)
309        ) {
310            let mut canonical = model.to_record();
311            if field.starts_with("coordinate") {
312                canonical.upsert(
313                    "coordinate_type",
314                    Value::Scalar(ScalarValue::String(model.coordinate_type().to_string())),
315                );
316            }
317            canonical.into()
318        } else if let Some(Value::Record(original)) = record.get(&field)
319            && let Some(normalized) = super::canonical_restored_coordinate(original, model)?
320        {
321            Value::Record(normalized)
322        } else {
323            record.get(&field).expect("coordinate field exists").clone()
324        };
325        restored.push(RecordField::new(field, value));
326    }
327    Ok(CoordinateState {
328        coordinates,
329        record: restored,
330        parent_name,
331    })
332}

334#[pymethods]
335#[allow(clippy::too_many_arguments)]
336impl PyCoordsys {
337    #[pyo3(signature = (direction=false, spectral=false, stokes=None, linear=0, tabular=false), text_signature = "($self, direction=False, spectral=False, stokes=[], linear=0, tabular=False)")]
338    fn newcoordsys(
339        &self,
340        py: Python<'_>,
341        direction: bool,
342        spectral: bool,
343        stokes: Option<&Bound<'_, PyAny>>,
344        linear: i32,
345        tabular: bool,
346    ) -> PyResult<Py<Self>> {
347        let stokes = string_arguments(stokes, "stokes")?;
348        let child = py.detach(|| {
349            let mut state = self.state();
350            let parent_name = std::mem::take(&mut state.parent_name);
351            let unset = ObsInfo::new("UNKNOWN")
352                .with_observer("UNKNOWN")
353                .with_date(MEpoch::from_mjd(0.0, EpochRef::LAST));
354            *state = CoordinateState::new(CoordinateSystem::new().with_obs_info(unset));
355            state.parent_name = parent_name;
356            state
357                .add_defaults(direction, spectral, stokes, linear, tabular)
358                .map_err(|_| raised("Unable to create new coordsys"))?;
359            *state.coordinates.obs_info_mut() = evla_observation();
360            for field in state.coordinates.obs_info().to_record().into_fields() {
361                state.record.upsert(field.name, field.value);
362            }
363            let mut child = state.clone();
364            child.parent_name = "unknown".into();
365            Ok::<_, PyErr>(child)
366        })?;
367        Py::new(
368            py,
369            Self {
370                state: Mutex::new(child),
371            },
372        )
373    }
374
375    #[pyo3(signature = (direction=false, spectral=false, stokes=None, linear=0, tabular=false), text_signature = "($self, direction=False, spectral=False, stokes=[], linear=0, tabular=False)")]
376    fn addcoordinate(
377        &self,
378        py: Python<'_>,
379        direction: bool,
380        spectral: bool,
381        stokes: Option<&Bound<'_, PyAny>>,
382        linear: i32,
383        tabular: bool,
384    ) -> PyResult<bool> {
385        let stokes = string_arguments(stokes, "stokes")?;
386        py.detach(|| {
387            self.state()
388                .add_defaults(direction, spectral, stokes, linear, tabular)?;
389            Ok(true)
390        })
391    }
392
393    fn fromrecord(&self, py: Python<'_>, record: &Bound<'_, PyDict>) -> PyResult<bool> {
394        let mut record = casa_record(py, Some(record))?;
395        py.detach(|| {
396            let mut state = self.state();
397            let parent_name = match record.remove("parentName") {
398                Some(Value::Scalar(ScalarValue::String(name))) => name,
399                None => state.parent_name.clone(),
400                _ => return Err(raised("Incorrect data type String for field parentName")),
401            };
402            *state = restore(record, parent_name)?;
403            Ok(true)
404        })
405    }
406
407    fn epoch(&self, py: Python<'_>) -> PyResult<Py<PyAny>> {
408        let record = py.detach(|| {
409            let state = self.state();
410            let missing = MEpoch::from_mjd(0.0, EpochRef::LAST);
411            epoch_to_record(
412                state
413                    .coordinates
414                    .obs_info()
415                    .date
416                    .as_ref()
417                    .unwrap_or(&missing),
418            )
419        });
420        CasaValue::from_record(&record).into_python(py)
421    }
