31/// Convert the store at `source` (MSv2 or MSv4) into a new MSv4
32/// processing set at `destination`.
33///
34/// Every partition becomes one dataset with the selected data groups
35/// ([`ConversionOptions::data_groups`]) and the antennas its baselines name
36/// (others are [`Loss::UnusedAntennas`]).
37/// An MSv4's datasets keep their names. An MSv2's are named as XRADIO's
38/// converter names them: the file name without `.ms`, then the partition's
39/// number in the order the partitions' first MAIN rows appear, padded with
40/// zeros to the width of the largest. Its antennas (for the telescopes
41/// XRADIO renames, by the first observation's telescope), fields, sources
42/// and spectral windows are renamed as that converter renames them
43/// (`NAME_STATION`, `NAME_FIELD_ID`, `NAME_SOURCE_ID`,
44/// `NAME_SPECTRAL_WINDOW_ID`), so the two converters' stores agree; a
45/// source XRADIO finds no row for in one spectral window, which it names
46/// `Unknown` there, is still named here. Two partitions XRADIO would merge,
47/// rows of different feeds, are two datasets here, and rows with
48/// `STATE_ID` -1 have no intents here where XRADIO gives them the last
49/// `STATE` row's; the datasets numbered after such a difference can hold
50/// other data than XRADIO's of the same name. A dataset whose fields
51/// follow an ephemeris holds it as XRADIO's `field_and_source_ephemeris`,
52/// the samples on `time_ephemeris` from the last before its first
53/// time to the first two minutes past its last (the samples no dataset
54/// holds are a reported loss), with the phase centre at each time as XRADIO
55/// states it. Partitions are
56/// converted [`ConversionOptions::workers`] at a time; nothing appears at
57/// `destination` until every dataset is complete.
58///
59/// # Errors
60///
61/// All before anything is written: [`ObservationError::InvalidRequest`]
62/// naming the first loss not accepted, or an ephemeris that does not cover
63/// a dataset's times; [`ObservationError::Unsupported`] for fields whose
64/// direction is a polynomial in time, a dataset whose fields follow
65/// different ephemerides (or only some one), or radial velocities that are
66/// not geocentric. Otherwise, any error opening or reading the source or
67/// writing the destination, which then is not created.
68pub fn to_msv4(
69    source: &Path,
70    destination: &Path,
71    options: &ConversionOptions,
72) -> Result<ConversionReport, ObservationError> {
73    let Named {
74        observation,
75        catalogue: named,
76        datasets,
77    } = open_named(source)?;
78    let catalogue = observation.catalogue();
79    let groups = losses::selected_groups(catalogue, options.data_groups.as_deref())?;
80    let mut found = losses::inventory(source, catalogue, &groups, losses::Store::Msv4)?;
81    let (pointing, lost) = pointing_plan::plan(
82        source,
83        observation.as_ref(),
84        losses::Store::Msv4,
85        options.batch_bytes,
86    )?;
87    found.extend(lost);
88    found.extend(losses::unused_antennas(catalogue));
89    found.extend(preflight_fields(observation.as_ref(), &groups[0])?);
90    found.sort();
91    losses::refuse_unaccepted(&found, &options.accept)?;
92    let set = Msv4ProcessingSetWriter::create(destination)?;
93    let ids = catalogue.partition_ids();
94    let next = AtomicUsize::new(0);
95    let results = Mutex::new(Vec::with_capacity(ids.len()));
96    std::thread::scope(|scope| {
97        for _ in 0..options.workers.clamp(1, ids.len().max(1)) {
98            scope.spawn(|| {
99                loop {
100                    let index = next.fetch_add(1, Ordering::Relaxed);
101                    let Some(id) = ids.get(index) else {
102                        break;
103                    };
104                    let result = partition(
105                        &Destination {
106                            set: &set,
107                            named: &named,
108                            pointing: &pointing,
109                        },
110                        observation.as_ref(),
111                        *id,
112                        &groups,
113                        &datasets[id.0 as usize],
114                        options,
115                    );
116                    let failed = result.is_err();
117                    results
118                        .lock()
119                        .unwrap_or_else(std::sync::PoisonError::into_inner)
120                        .push(result);
121                    if failed {
122                        // Stop handing out partitions; the others finish
123                        // what they hold.
124                        next.store(ids.len(), Ordering::Relaxed);
125                    }
126                }
127            });
128        }
129    });
130    let mut report = ConversionReport {
131        losses: found,
132        ..ConversionReport::default()
133    };
134    for result in results
135        .into_inner()
136        .unwrap_or_else(std::sync::PoisonError::into_inner)
137    {
138        let converted = result?;
139        report.partitions += 1;
140        report.batches += converted.batches;
141        report.samples += converted.samples;
142        report.elements += converted.counters.elements;
143        report.held_elements += converted.counters.held_elements;
144    }
145    set.finish()?;
146    Ok(report)
147}

149/// Refuse, before anything is written, what a dataset's field dataset
150/// cannot hold ([`check_fields`]), and report the samples of each ephemeris
151/// the partitions' fields follow that no dataset holds, one loss an
152/// ephemeris: a dataset holds those from the last before its first time to
153/// the first that reaches past its last time's TDB ([`kept_samples`]),
154/// much as XRADIO's converter keeps those either side of a dataset's times.
155/// Only the samples the source states count: an extrapolated one is not
156/// written, and reading the destination extrapolates it again.
157fn preflight_fields(
158    observation: &dyn Observation,
159    group: &str,
160) -> Result<Vec<Loss>, ObservationError> {
161    let catalogue = observation.catalogue();
162    let mut followed: Vec<(&Arc<FieldEphemeris>, Vec<bool>)> = Vec::new();
163    for id in catalogue.partition_ids() {
164        let grid = observation.grid(id, group)?;
165        let times = unix_times(&grid);
166        let entry = catalogue.partition(id);
167        if let Some((_, data_group)) = entry.data_group(group) {
168            check_fields(catalogue, &grid, &times, data_group.uvw_frame.effective)?;
169        }
170        for field in &entry.fields {
171            let Some(ephemeris) = catalogue.fields[field.0 as usize].ephemeris.as_ref() else {
172                continue;
173            };
174            let index = followed
175                .iter()
176                .position(|(known, _)| Arc::ptr_eq(known, ephemeris))
177                .unwrap_or_else(|| {
178                    let stated = ephemeris.times_mjd().len() - ephemeris.extrapolated();
179                    followed.push((ephemeris, vec![false; stated]));
180                    followed.len() - 1
181                });
182            for held in &mut followed[index].1[kept_samples(ephemeris, &times)] {
183                *held = true;
184            }
185        }
186    }
187    Ok(followed
188        .into_iter()
189        .filter_map(|(ephemeris, held)| {
190            let lost = held.iter().filter(|held| !**held).count();
191            (lost > 0).then(|| {
192                Loss::Table(format!(
193                    "the ephemeris of {}: samples no dataset keeps ({lost}; a dataset keeps \
194                     those from the last before its first time to the first 120 s or more past \
195                     its last)",
196                    ephemeris.name()
197                ))
198            })
199        })
200        .collect())
201}

203/// A source opened for conversion, with what its datasets are named.
204struct Named {
205    observation: Box<dyn Observation>,
206    /// The catalogue the datasets are declared with.
207    catalogue: Catalogue,
208    /// Each partition's dataset name.
209    datasets: Vec<String>,
210}
211
212/// Open `source`, with the catalogue its datasets are declared with and
213/// each partition's dataset name: an MSv4's own, or an MSv2's with
214/// XRADIO's names.
215fn open_named(source: &Path) -> Result<Named, ObservationError> {
216    let options = OpenOptions::default();
217    if source.join("zarr.json").is_file() {
218        let observation = Msv4Observation::open_with(source, &options)?;
219        let catalogue = observation.catalogue().as_ref().clone();
220        let datasets = catalogue
221            .partition_ids()
222            .into_iter()
223            .map(|id| observation.dataset_name(id).to_string())
224            .collect();
225        return Ok(Named {
226            observation: Box::new(observation),
227            catalogue,
228            datasets,
229        });
230    }
231    let observation = Msv2Observation::open_with(source, &options)?;
232    let ids = observation.catalogue().partition_ids();
233    let windows: Vec<u32> = ids
234        .iter()
235        .map(|id| observation.spectral_window_id(*id))
236        .collect();
237    let first_rows: Vec<Option<u32>> = ids.iter().map(|id| observation.first_row(*id)).collect();
238    let fields = observation.measurement_set().field().map_err(storage)?;
239    let sources = (0..observation.catalogue().fields.len())
240        .map(|row| fields.source_id(row).map_err(storage))
241        .collect::<Result<Vec<_>, _>>()?;
242    let named = xradio_names(observation.catalogue(), &windows, &sources);
243    let datasets = dataset_names(source, &first_rows);
244    Ok(Named {
245        observation: Box::new(observation),
246        catalogue: named,
247        datasets,
248    })
249}

251/// Each partition's dataset name, as XRADIO's converter names an MSv2's:
252/// the file name without `.ms`, then the partition's number in the order
253/// of `first_rows`, the partitions' first MAIN rows, padded with zeros to
254/// the width of the largest. The path is made absolute, and resolved when
255/// that leaves no file name, so `.`, `..` and a trailing `/` name the
256/// MeasurementSet they stand for, where XRADIO's converter names `.`'s
257/// datasets `_0`, `..`'s `.._0` and `/`'s `_0` (here `dataset_0`, `/`
258/// having no name).
259fn dataset_names(source: &Path, first_rows: &[Option<u32>]) -> Vec<String> {
260    let mut path = std::path::absolute(source).unwrap_or_else(|_| source.to_path_buf());
261    if path.file_name().is_none() {
262        path = std::fs::canonicalize(source).unwrap_or(path);
263    }
264    let stem = path.file_name().map_or_else(
265        || "dataset".to_string(),
266        |name| name.to_string_lossy().replace(".ms", ""),
267    );
268    let mut order: Vec<usize> = (0..first_rows.len()).collect();
269    order.sort_by_key(|index| (first_rows[*index].unwrap_or(u32::MAX), *index));
270    let width = first_rows.len().saturating_sub(1).to_string().len();
271    let mut names = vec![String::new(); first_rows.len()];
272    for (number, index) in order.into_iter().enumerate() {
273        names[index] = format!("{stem}_{number:0width$}");
274    }
275    names
276}

278/// `catalogue` with XRADIO's names, as its converter names an MSv2's:
279/// antennas `NAME_STATION` for the telescopes it names so
280/// ([`STATION_NAMED`]), fields `NAME_ID`, sources `NAME_SOURCEID` by each
281/// field's `SOURCE_ID` (`sources`), and spectral windows `NAME_ID`
282/// (`spw_ID` unnamed) by the `SPECTRAL_WINDOW` row of each partition
283/// (`windows`).
284fn xradio_names(catalogue: &Catalogue, windows: &[u32], sources: &[i32]) -> Catalogue {
285    let mut named = catalogue.clone();
286    let telescope = catalogue
287        .observations
288        .first()
289        .map_or("", |observation| observation.telescope.as_str());
290    if STATION_NAMED.contains(&telescope) {
291        for antenna in &mut named.antennas {
292            antenna.name = format!("{}_{}", antenna.name, antenna.station);
293        }
294    }
295    for ((index, field), source) in named.fields.iter_mut().enumerate().zip(sources) {
296        field.name = format!("{}_{index}", field.name);
297        if let Some(name) = &mut field.source_name {
298            *name = format!("{name}_{source}");
299        }
300    }
301    for (partition, window) in named.partitions.iter_mut().zip(windows) {
302        partition.spectral_window = match partition.spectral_window.as_str() {
303            "" | "none" => format!("spw_{window}"),
304            name => format!("{name}_{window}"),
305        };
306    }
307    named
308}

310/// What every partition is written into: the processing set, the
311/// catalogue its datasets are declared with, and the pointing carried.
312struct Destination<'a> {
313    set: &'a Msv4ProcessingSetWriter,
314    named: &'a Catalogue,
315    pointing: &'a PointingPlan,
316}
317
318/// Convert partition `id` into dataset `name` of the destination: its
319/// pointing, then one traversal per selected group in lockstep, each batch
320/// of the same whole times.
321fn partition(
322    destination: &Destination<'_>,
323    observation: &dyn Observation,
324    id: PartitionId,
325    groups: &[String],
326    name: &str,
327    options: &ConversionOptions,
328) -> Result<Converted, ObservationError> {
329    let Destination {
330        set,
331        named,
332        pointing,
333    } = *destination;
334    let grid = observation.grid(id, &groups[0])?;
335    let entry = observation.catalogue().partition(id);
336    let extents = [
337        grid.times_s.len(),
338        grid.baselines.len(),
339        entry.frequencies_hz.len(),
340        entry.correlations.len(),
341    ];
342    let chunks: [u64; 4] = std::array::from_fn(|axis| {
343        let extent = extents[axis].max(1) as u64;
344        match options.chunks[axis] {
345            0 => extent,
346            chunk => chunk.min(extent),
347        }
348    });
349    let storage = Msv4StorageOptions {
350        dataset_name: name.to_string(),
351        creator: options.creator.clone(),
352        creation_date: options.creation_date.clone(),
353        chunks,
354        codec: options.codec,
355    };
356    let names: Vec<&str> = groups.iter().map(String::as_str).collect();
357    let mut writer = set.dataset(named, id, &names, &grid, &storage)?;
358    // The pointing of the antennas the partition's baselines name, as
359    // planned; dropped before the batches are read.
360    if let Some(read) = pointing.read_for(id) {
361        writer.write_pointing(named, &observation.pointing(read)?)?;
362    }
363    let budget = (options.batch_bytes, Some(writer.chunk_times()));
364    let read = lockstep::read(observation, id, groups, Roles::ALL, budget, |batches| {
365        writer.write_groups(batches)
366    })?;
367    Ok(Converted {
368        batches: read.batches,
369        samples: read.samples,
370        counters: writer.finish()?,
371    })
372}

378    #[test]
379    fn datasets_are_numbered_by_first_row_and_padded_to_the_largest() {
380        let rows = [Some(10), Some(0), None];

383            ["pilot_1", "pilot_0", "pilot_2"]

386        let names = dataset_names(Path::new("pilot.ms/"), &eleven);
387        assert_eq!(names[10], "pilot_00");
388        assert_eq!(names[0], "pilot_10");
389        for path in [".", ".."] {
390            let named = dataset_names(Path::new(path), &[Some(0)]);
391            assert!(
392                !named[0].starts_with("dataset") && !named[0].starts_with('_'),
393                "{path}: {named:?}"
