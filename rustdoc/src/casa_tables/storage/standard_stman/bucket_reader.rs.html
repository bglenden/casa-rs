4use std::io::BufReader;

6use super::*;

8/// Retain a single bucket for rows or payloads that share it.
9#[derive(Default)]
10pub(super) struct LastBucket {
11    number: Option<u32>,
12    data: Vec<u8>,
13}
14
15impl LastBucket {
16    pub(super) fn read(
17        &mut self,
18        file: &mut File,
19        header: &SsmHeader,
20        number: u32,
21    ) -> Result<&[u8], StorageError> {
22        if self.number != Some(number) {
23            self.data = read_bucket(file, header, number)?;
24            self.number = Some(number);
25        }
26        Ok(&self.data)
27    }
28}
29
30/// One operation owns this buffer; no cached bytes survive a table read.
31pub(super) struct ReadBuckets {
32    file: BufReader<File>,
33    length: u64,
34    position: Option<u64>,
35    bucket: Vec<u8>,
36}
37
38impl ReadBuckets {
39    pub(super) fn new(file: File) -> Result<Self, StorageError> {
40        let length = file.metadata()?.len();
41        Ok(Self {
42            file: BufReader::with_capacity(32 * 1024, file),
43            length,
44            position: None,
45            bucket: Vec::new(),
46        })
47    }
48
49    pub(super) fn read(&mut self, header: &SsmHeader, number: u32) -> Result<&[u8], StorageError> {
50        let offset = SSM_HEADER_SIZE + u64::from(number) * u64::from(header.bucket_size);
51        let end = offset + u64::from(header.bucket_size);
52        if end > self.length {
53            return Err(StorageError::FormatMismatch(format!(
54                "SSM bucket {number} out of range: offset={offset}, bucket_size={}, file_len={}",
55                header.bucket_size, self.length
56            )));
57        }
58        let size = header.bucket_size as usize;
59        if size > self.bucket.len() {
60            self.bucket
61                .try_reserve_exact(size - self.bucket.len())
62                .map_err(|error| {
63                    StorageError::FormatMismatch(format!("cannot allocate SSM bucket: {error}"))
64                })?;
65        }
66        let relative = self
67            .position
68            .take()
69            .and_then(|position| i64::try_from(i128::from(offset) - i128::from(position)).ok());
70        if let Some(relative) = relative {
71            self.file.seek_relative(relative)?;
72        } else {
73            self.file.seek(SeekFrom::Start(offset))?;
74        }
75        self.bucket.resize(size, 0);
76        self.file.read_exact(&mut self.bucket)?;
77        self.position = Some(end);
78        Ok(&self.bucket)
79    }
80}

82#[cfg(test)]
83mod tests {
84    use super::*;
85
86    #[test]
87    fn forward_backward_and_invalid_bucket_reads_preserve_bytes() {
88        let root = tempfile::tempdir().unwrap();
89        let path = root.path().join("buckets");
90        let bytes = (0..512 + 1024 * 128)
91            .map(|index| (index % 251) as u8)
92            .collect::<Vec<_>>();
93        std::fs::write(&path, &bytes).unwrap();
94        let header = SsmHeader {
95            bucket_size: 128,
96            nr_buckets: 1024,
97            pers_cache_size: 0,
98            nr_free_buckets: 0,
99            first_free_bucket: -1,
100            nr_idx_buckets: 0,
101            first_idx_bucket: -1,
102            idx_bucket_offset: 0,
103            last_string_bucket: -1,
104            index_length: 0,
105            nr_indices: 0,
106            big_endian: false,
107            io_order: ByteOrder::LittleEndian,
108        };
109        let mut reader = ReadBuckets::new(File::open(path).unwrap()).unwrap();
110        for number in [2, 0, 2, 1, 1000, 500, 11, 0] {
111            let start = SSM_HEADER_SIZE as usize + number as usize * 128;
112            assert_eq!(
113                reader.read(&header, number).unwrap(),
114                &bytes[start..start + 128]
115            );
116        }
117        assert!(reader.read(&header, 1024).is_err());
118        assert_eq!(reader.read(&header, 0).unwrap(), &bytes[512..640]);
119    }
120}
