103    /// Recognize FITS using the expanded spelling and read it at its lexical
104    /// absolute path. Report an unrecognized input with the caller's spelling.
105    ///
106    /// # Errors
107    /// As [`Self::open_attachment`].
108    pub fn open_attachment_named(
109        path: &Path,
110        probe: &Path,
111        name: &Path,
112    ) -> Result<Self, ImageError> {
113        let (mut length, mut tight_value) = attachment_file_length(probe, name)?;
114        if probe != path {
115            if !path.is_file() {
116                return Err(ImageError::CasaSelection(format!(
117                    "{} does not exist or is not readable",
118                    path.display()
119                )));
120            }
121            (length, tight_value) = attachment_file_length(path, name).map_err(|_| {
122                ImageError::CasaSelection(format!(
123                    "Cannot open file {} (or other I/O error)",
124                    path.display()
125                ))
126            })?;
127        }
128        if tight_value {
129            return Err(initial_record_error(path));
130        }
131        let mut fits = open_disk_file(path).map_err(|_| {
132            if length.is_some() {
133                ImageError::CasaSelection(format!(
134                    "FITSImgParser::setup - {} Error opening FITS input.",
135                    path.display()
136                ))
137            } else {
138                unable_to_open_image(name)
139            }
140        })?;
141        let count = if length.is_some() {
142            2
143        } else {
144            fits.num_hdus().map_err(fits_error)?
145        };
146        let hdu = (0..count)
147            .find_map(|index| fits.hdu(index).ok().filter(has_image_data))
148            .ok_or_else(|| missing_attachment_data(path))?;
149        if let Some(length) = length {
150            validate_initial_image_record(path, &mut fits, hdu.number, length)?;
151        }
152        Self::read_metadata(fits, hdu, 0, false)
153    }

163    fn open_impl(
164        fits: fitsio::errors::Result<FitsFile>,
165        name: &Path,
166        which_rep: usize,
167        which_hdu: i64,
168        group_linear_axes: bool,
169    ) -> Result<Self, ImageError> {
170        let mut fits = fits.map_err(|_| unreadable_fits(name))?;
171        let missing_hdu = || {
172            ImageError::CasaSelection(format!(
173                "Exception: No image at specified location in file {}.",
174                name.display()
175            ))
176        };
177        let count = fits.num_hdus().map_err(fits_error)?;
178        let hdu = if which_hdu < 0 {
179            (0..count).find_map(|index| fits.hdu(index).ok().filter(has_image_data))
180        } else {
181            usize::try_from(which_hdu)
182                .ok()
183                .filter(|&index| index < count)
184                .and_then(|index| fits.hdu(index).ok())
185                .filter(has_image_data)
186        }
187        .ok_or_else(missing_hdu)?;
188        Self::read_metadata(fits, hdu, which_rep, group_linear_axes)
189    }

191    fn read_metadata(
192        mut fits: FitsFile,
193        hdu: FitsHdu,
194        which_rep: usize,
195        group_linear_axes: bool,
196    ) -> Result<Self, ImageError> {
197        let HduInfo::ImageInfo {
198            ref shape,
199            image_type,
200        } = hdu.info
201        else {
202            unreachable!("image HDU selected above");
203        };
204        let cards = read_hdu_cards(&mut fits)?;
205        let physical = FitsHeader::from_cards(
206            &cards
207                .iter()
208                .filter(|card| card.as_bytes().get(8) == Some(&b'='))
209                .map(String::as_str)
210                .collect::<Vec<_>>(),
211        );
212        let mut header = select_wcs(&physical, which_rep);
213        let shape = if is_tile_compressed(&physical) {
214            // The physical NAXIS cards describe the compression table rows.
215            let shape: Vec<_> = shape.iter().rev().copied().collect();
216            header.set("NAXIS", FitsValue::Integer(shape.len() as i64));
217            for (axis, &length) in shape.iter().enumerate() {
218                header.set(
219                    format!("NAXIS{}", axis + 1),
220                    FitsValue::Integer(length as i64),
221                );
222            }
223            shape
224        } else {
225            read_fits_axis_shape(&mut fits, &hdu)?
226        };
227        if header.get("RADESYS").is_none() {
228            let system = match header.get_string("RADECSYS") {
229                Some(system) => system,
230                None => match header
231                    .get_float("EQUINOX")
232                    .or_else(|| header.get_float("EPOCH"))
233                {
234                    Some(epoch) if epoch < 1984.0 => "FK4",
235                    Some(_) => "FK5",
236                    None => "ICRS",
237                },
238            };
239            header.set("RADESYS", FitsValue::String(system.into()));
240        }
241        let mut has_wcs = false;
242        for axis in 1..=shape.len() {
243            for (prefix, default) in [("CRVAL", 0.0), ("CRPIX", 0.0), ("CDELT", 1.0)] {
244                let name = format!("{prefix}{axis}");
245                if header.get(&name).is_none() {
246                    header.set(name, FitsValue::Float(default));
247                } else {
248                    has_wcs = true;
249                }
250            }
251            has_wcs |= header.get(&format!("CTYPE{axis}")).is_some();
252        }
253        let aligned = align_fits_pc_rows(&header, shape.len());
254        let (coordinate_header, mut world_order) =
255            aligned.as_ref().map_or((&header, None), |(header, order)| {
256                (header, Some(order.as_slice()))
257            });
258        let mut use_fallback = false;
259        let (mut coordinates, axis_order) =
260            match from_fits_header_with_axis_order(coordinate_header, &shape) {
261                Err(CoordinateError::InvalidFitsAxisUnit { .. }) => {
262                    use_fallback = true;
263                    world_order = None;
264                    (
265                        fallback_fits_linear_coordinates(&shape),
266                        (0..shape.len()).collect(),
267                    )
268                }
269                result => result?,
270            };
271        if !has_wcs && !use_fallback {
272            let ndim = shape.len();
273            coordinates = CoordinateSystem::new().with_obs_info(coordinates.obs_info().clone());
274            coordinates.add_coordinate(
275                LinearCoordinate::new(ndim, vec![String::new(); ndim], vec![String::new(); ndim])
276                    .with_reference_pixel(vec![-1.0; ndim]),
277            );
278        } else if group_linear_axes && !use_fallback {
279            coordinates = group_linear_wcs(&coordinates, &axis_order, coordinate_header);
280        }
281        let obs = coordinates.obs_info_mut();
282        if obs.observer.is_empty() {
283            obs.observer = "UNKNOWN".into();
284        }
285        if obs.telescope.is_empty() {
286            obs.telescope = "UNKNOWN".into();
287        }
288        if obs.date.is_none() {
289            obs.date = Some(MEpoch::from_mjd(0.0, EpochRef::LAST));
290        }
291        if !use_fallback
292            && let [Some(x), Some(y), Some(z)] =
293                ["OBSGEO-X", "OBSGEO-Y", "OBSGEO-Z"].map(|key| header.get_float(key))
294        {
295            obs.telescope_position = Some(MPosition::new_itrf(x, y, z));
296        }
297        let coordinate_record =
298            fits_coordinate_record(&coordinates, &axis_order, world_order, coordinate_header)?;
299        let coordinates = CoordinateSystem::from_record(&coordinate_record).map_err(|error| {
300            if matches!(
301                error,
302                casa_coordinates::CoordinateError::UnsupportedAxisLayout(_)
303            ) {
304                ImageError::CasaSelection(format!("{error} (casa-rs issue #1002)"))
305            } else {
306                error.into()
307            }
308        })?;
309        let mut source = PrimaryHdu {
310            fits,
311            hdu,
312            image_type,
313            shape,
314            header,
315            coordinates,
316        };
317        let history = history_from_cards(&cards, true);
318        let mut info = image_info_from_fits_header(&source.header).to_record();
319        if let Some((beams, records)) = source.beam_set()? {
320            info.remove("restoringbeam");
321            if beams.is_single() {
322                if let Some(beam) = records.get("*0") {
323                    info.upsert("restoringbeam", beam.clone());
324                }
325            } else {
326                let mut grid = beams.to_record();
327                for field in records.fields() {
328                    grid.upsert(&field.name, field.value.clone());
329                }
330                info.upsert("perplanebeams", Value::Record(grid));
331            }
332        } else if let Some(beam) = primary_restoring_beam(&source.header) {
333            info.upsert("restoringbeam", Value::Record(beam.to_record()));
334        } else if info.get("restoringbeam").is_none()
335            && let Some(beam) = restoring_beam_from_history(&history)
336        {
337            info.upsert("restoringbeam", Value::Record(beam.to_record()));
338        }
339        Ok(Self {
340            coordinate_record,
341            units: brightness_unit(&source.header),
342            image_info: info,
343            misc_info: miscellaneous(&physical),
344            history,
345            source,
346        })
347    }

383    /// Copy single-precision pixels and the FITS validity mask into an image.
384    /// The copy uses bounded contiguous blocks along the FITS axis order.
385    ///
386    /// # Errors
387    /// Returns an error when the image's shape disagrees or reading or
388    /// writing fails.
389    pub fn copy_pixels(
390        &mut self,
391        image: &mut impl MutableImageInterface<f32>,
392        zero_blanks: bool,
393    ) -> Result<(), ImageError> {
394        let axes: Vec<_> = (0..self.source.shape.len()).collect();
395        self.copy_pixels_with_axes(image, zero_blanks, &axes)
396    }

398    // Destination axes in source order, with singleton axes appended beyond
399    // the FITS rank before applying the same permutation to pixels and masks.
400    /// Streams FITS pixels and validity into a destination whose axes are indexed by the given source
401    /// permutation. Appended axes have extent one. Invalid permutations, mismatched extents, and read
402    /// or write failures return an image error.
403    pub fn copy_pixels_with_axes(
404        &mut self,
405        image: &mut impl MutableImageInterface<f32>,
406        zero_blanks: bool,
407        axes: &[usize],
408    ) -> Result<(), ImageError> {
409        let mut shape = self.source.shape.clone();
410        let ndim = shape.len();
411        if axes.len() < ndim {
412            return Err(ImageError::InvalidMetadata(
413                "FITS axes were dropped during import".into(),
414            ));
415        }
416        let mut remaining = vec![true; axes.len()];
417        for &axis in axes {
418            if !remaining.get_mut(axis).is_some_and(std::mem::take) {
419                return Err(ImageError::InvalidMetadata(
420                    "invalid FITS axis order".into(),
421                ));
422            }
423        }
424        shape.resize(axes.len(), 1);
425        let expected: Vec<_> = axes.iter().map(|&axis| shape[axis]).collect();
426        if image.shape() != expected {
427            return Err(ImageError::ShapeMismatch {
428                expected,
429                got: image.shape().to_vec(),
430            });
431        }
432        let total = shape
433            .iter()
434            .try_fold(1usize, |product, &length| product.checked_mul(length))
435            .ok_or_else(|| ImageError::InvalidMetadata("FITS image shape is too large".into()))?;
436        let quota = 1_048_576;
437        let mut stride = 1;
438        let mut axis = 0;
439        while axis + 1 < ndim && stride * shape[axis] <= quota {
440            stride *= shape[axis];
441            axis += 1;
442        }
443        let mut block_shape = vec![1; axes.len()];
444        block_shape[..axis].copy_from_slice(&shape[..axis]);
445        let mut origin = vec![0; axes.len()];
446        let mut output_origin = origin.clone();
447        let mut start = 0;
448        let mut mask = None;
449        while start < total {
450            let mut offset = start;
451            for (position, length) in origin.iter_mut().zip(&shape) {
452                *position = offset % length;
453                offset /= length;
454            }
455            for (position, &source_axis) in output_origin.iter_mut().zip(axes) {
456                *position = origin[source_axis];
457            }
458            let run = (quota / stride).min(shape[axis] - origin[axis]);
459            block_shape[axis] = run;
460            let count = run * stride;
461            let (mut values, valid) = self.source.read_pixels(start, count)?;
462            if valid.iter().any(|bit| !bit) {
463                let mask = mask.get_or_insert_with(|| vec![true; total]);
464                mask[start..start + count].copy_from_slice(&valid);
465                for (value, valid) in values.iter_mut().zip(valid) {
466                    if !valid {
467                        *value = if zero_blanks { 0.0 } else { f32::NAN };
468                    }
469                }
470            }
471            let block = ArrayD::from_shape_vec(IxDyn(&block_shape).f(), values)
472                .map_err(|error| ImageError::InvalidMetadata(error.to_string()))?;
473            image.put_slice(&block.permuted_axes(IxDyn(axes)), &output_origin)?;
474            start += count;
475        }
476        if let Some(mask) = mask {
477            let mask = ArrayD::from_shape_vec(IxDyn(&shape).f(), mask)
478                .map_err(|error| ImageError::InvalidMetadata(error.to_string()))?;
479            image.put_mask("mask0", &mask.permuted_axes(IxDyn(axes)))?;
480            image.set_default_mask("mask0")?;
481        }
482        Ok(())
483    }
484}

486fn primary_restoring_beam(header: &FitsHeader) -> Option<StoredGaussianBeam> {
487    let major = header.get_float("BMAJ")?;
488    let minor = header.get_float("BMIN")?;
489    let angle = header.get_float("BPA")?;
490    if !major.is_finite() || !minor.is_finite() || major <= 0. || minor <= 0. {
491        return None;
492    }
493    // The primary cards use degrees, while CASA's native beam record uses
494    // arcseconds for widths and keeps the angle in its original degrees.
495    StoredGaussianBeam::new(
496        Quantity::new(3600. * major.max(minor), "arcsec").ok()?,
497        Quantity::new(3600. * major.min(minor), "arcsec").ok()?,
498        Quantity::new(angle, "deg").ok()?,
499    )
500    .ok()
501}

517fn validate_initial_image_record(
518    path: &Path,
519    fits: &mut FitsFile,
520    hdu: usize,
521    length: u64,
522) -> Result<(), ImageError> {
523    let mut start = 0;
524    let mut status = 0;
525    // SAFETY: the file is open on the selected image HDU. CFITSIO writes
526    // the data offset and status to the two live output pointers.
527    unsafe {
528        fitsio::sys::ffghadll(
529            fits.as_raw(),
530            std::ptr::null_mut(),
531            &mut start,
532            std::ptr::null_mut(),
533            &mut status,
534        );
535    }
536    check_status(status).map_err(fits_error)?;
537    let start = u64::try_from(start).map_err(|_| missing_attachment_data(path))?;
538    let available = length.saturating_sub(start);
539    if available >= FITS_RECORD_BYTES {
540        return Ok(());
541    }
542    if hdu == 0 && available != 0 {
543        return Err(initial_record_error(path));
544    }
545    Err(missing_attachment_data(path))
546}

552/// A read-only pixel stream for a specific FITS image HDU.
553pub struct FitsSectionReader {
554    file: ThreadsafeFitsFile,
555    shape: Vec<usize>,
556    bitpix: i32,
557}
558
559impl FitsSectionReader {
560    /// Open one image HDU and retain it for successive reads. `shape` is
561    /// the logical image shape, in FITS pixel axis order.
562    ///
563    /// # Errors
564    /// Returns an error for missing or incompatible image data, including
565    /// a source whose shape changed since its metadata was read.
566    pub fn open(path: &Path, hdu_index: usize, shape: &[usize]) -> Result<Self, ImageError> {
567        let mut file = open_disk_file(path).map_err(fits_error)?;
568        let hdu = file.hdu(hdu_index).map_err(fits_error)?;
569        let HduInfo::ImageInfo { shape: actual, .. } = hdu.info else {
570            return Err(ImageError::InvalidMetadata(
571                "selected FITS HDU is not an image".into(),
572            ));
573        };
574        if shape.is_empty() || shape.len() > 9 || !actual.iter().rev().eq(shape) {
575            return Err(ImageError::InvalidMetadata(
576                "FITS pixel source has incompatible image dimensions".into(),
577            ));
578        }
579        let (mut bitpix, mut status) = (0, 0);
580        // SAFETY: the handle is on the selected image and both outputs are
581        // valid. CFITSIO reports the logical BITPIX for compressed images.
582        unsafe { fitsio::sys::ffgidt(file.as_raw(), &mut bitpix, &mut status) };
583        check_status(status).map_err(fits_error)?;
584        Ok(Self {
585            file: file.threadsafe(),
586            shape: shape.to_vec(),
587            bitpix,
588        })
589    }
590
591    /// Read a strided section as `f32`, converting integer blanks to NaN.
592    /// Only the requested region is allocated.
593    ///
594    /// # Errors
595    /// Returns an error if reading fails or the slice is outside the image.
596    pub fn read(
597        &self,
598        start: &[usize],
599        count: &[usize],
600        stride: &[usize],
601    ) -> Result<ArrayD<f32>, ImageError> {
602        let mut file = self.file.lock().map_err(fits_error)?;
603        read_section(&mut file, &self.shape, self.bitpix, start, count, stride)
604    }
605}

621impl SharedFitsSectionReader {
622    /// Starts a worker that lazily opens the requested zero-based HDU on its first read. The caller
623    /// supplies its axis extents. An error starting the worker is returned immediately.
624    pub fn new(path: PathBuf, hdu: usize, shape: Vec<usize>) -> Result<Self, ImageError> {
625        let (requests, receive) = mpsc::channel::<Option<SectionRequest>>();
626        let worker = thread::Builder::new()
627            .name("FITS expression reader".into())
628            .spawn(move || {
629                let mut reader = None;
630                while let Ok(Some(request)) = receive.recv() {
631                    let result = (|| {
632                        if reader.is_none() {
633                            reader = Some(FitsSectionReader::open(&path, hdu, &shape)?);
634                        }
635                        reader.as_ref().expect("open FITS section reader").read(
636                            &request.start,
637                            &request.shape,
638                            &request.stride,
639                        )
640                    })();
641                    let _ = request.result.send(result);
642                }
643            })
644            .map_err(|error| ImageError::Io(error.to_string()))?;
645        Ok(Self {
646            requests,
647            worker: Some(worker),
648        })
649    }
650
651    /// Reads owned float pixels from a zero-based section with per-axis extents and strides. Failed
652    /// reads and a disconnected worker return an image error.
653    pub fn read(
654        &self,
655        start: &[usize],
656        shape: &[usize],
657        stride: &[usize],
658    ) -> Result<ArrayD<f32>, ImageError> {
659        let (result, receive) = mpsc::sync_channel(1);
660        self.requests
661            .send(Some(SectionRequest {
662                start: start.to_vec(),
663                shape: shape.to_vec(),
664                stride: stride.to_vec(),
665                result,
666            }))
667            .map_err(|_| ImageError::Io("FITS expression reader has closed".into()))?;
668        receive.recv().map_err(|_| {
669            ImageError::Io("FITS expression reader ended before returning data".into())
670        })?
671    }
672}
673
674impl Drop for SharedFitsSectionReader {
675    fn drop(&mut self) {
676        let _ = self.requests.send(None);
677        if let Some(worker) = self.worker.take() {
678            let _ = worker.join();
679        }
680    }
681}

774    fn get_mask_slice(
775        &self,
776        start: &[usize],
777        shape: &[usize],
778        stride: &[usize],
779    ) -> Result<Option<ArrayD<bool>>, ImageError> {
780        let invalid =
781            || ImageError::InvalidMetadata("FITS pixel section is outside the image".into());
782        if start.len() != self.ndim() || shape.len() != self.ndim() || stride.len() != self.ndim() {
783            return Err(invalid());
784        }
785        for (axis, &count) in shape.iter().enumerate() {
786            if count == 0
787                || stride[axis] == 0
788                || (count - 1)
789                    .checked_mul(stride[axis])
790                    .and_then(|span| start[axis].checked_add(span))
791                    .is_none_or(|last| last >= self.shape[axis])
792            {
793                return Err(invalid());
794            }
795        }
796        let mut full_mask = None;
797        let cursors = TraversalCursorIter::new(
798            shape.to_vec(),
799            self.nice_cursor_shape(),
800            TraversalSpec::tiles(),
801        );
802        for cursor in cursors {
803            let cursor = cursor?;
804            let source = cursor
805                .position
806                .iter()
807                .zip(start)
808                .zip(stride)
809                .map(|((&offset, &first), &step)| first + offset * step)
810                .collect::<Vec<_>>();
811            let pixels = (self.read)(&source, &cursor.shape, stride)?;
812            if pixels.iter().any(|value| value.is_nan()) {
813                let mask =
814                    full_mask.get_or_insert_with(|| ArrayD::from_elem(IxDyn(shape).f(), true));
815                let mut window = mask.slice_each_axis_mut(|axis| {
816                    let dim = axis.axis.index();
817                    let offset = cursor.position[dim];
818                    Slice::from(offset..offset + cursor.shape[dim])
819                });
820                Zip::from(&mut window)
821                    .and(&pixels)
822                    .for_each(|valid, &pixel| *valid = !pixel.is_nan());
823            }
824        }
825        Ok(full_mask)
826    }

829fn read_section(
830    file: &mut FitsFile,
831    shape: &[usize],
832    bitpix: i32,
833    start: &[usize],
834    count: &[usize],
835    stride: &[usize],
836) -> Result<ArrayD<f32>, ImageError> {
837    let invalid = || ImageError::InvalidMetadata("FITS pixel section is outside the image".into());
838    let dims = shape.len();
839    if start.len() != dims || count.len() != dims || stride.len() != dims {
840        return Err(invalid());
841    }
842    let mut begin = Vec::with_capacity(dims);
843    let mut end = Vec::with_capacity(dims);
844    let mut increment = Vec::with_capacity(dims);
845    let mut size = 1usize;
846    for axis in 0..dims {
847        if stride[axis] == 0 || count[axis] == 0 {
848            return Err(invalid());
849        }
850        let last = (count[axis] - 1)
851            .checked_mul(stride[axis])
852            .and_then(|width| start[axis].checked_add(width))
853            .filter(|last| *last < shape[axis])
854            .ok_or_else(invalid)?;
855        begin.push(c_long::try_from(start[axis] + 1).map_err(|_| invalid())?);
856        end.push(c_long::try_from(last + 1).map_err(|_| invalid())?);
857        increment.push(c_long::try_from(stride[axis]).map_err(|_| invalid())?);
858        size = size.checked_mul(count[axis]).ok_or_else(invalid)?;
859    }
860    let mut values = vec![0.0_f32; size];
861    let mut null = f32::NAN;
862    // FITS float pixels already encode blanks as NaN. Asking CFITSIO to
863    // substitute them would also turn valid infinities into blanks.
864    let null_value = if bitpix < 0 {
865        std::ptr::null_mut()
866    } else {
867        (&mut null as *mut f32).cast()
868    };
869    let (mut any_null, mut status) = (0, 0);
870    // SAFETY: the FITS handle is on the selected image and the coordinate
871    // vectors and output buffer match the checked section above.
872    unsafe {
873        fitsio::sys::ffgsv(
874            file.as_raw(),
875            fitsio::sys::TFLOAT as i32,
876            begin.as_mut_ptr(),
877            end.as_mut_ptr(),
878            increment.as_mut_ptr(),
879            null_value,
880            values.as_mut_ptr().cast(),
881            &mut any_null,
882            &mut status,
883        );
884    }
885    check_status(status).map_err(fits_error)?;
886    ArrayD::from_shape_vec(IxDyn(count).f(), values)
887        .map_err(|error| ImageError::InvalidMetadata(error.to_string()))
888}

890fn group_linear_wcs(
891    input: &CoordinateSystem,
892    axis_order: &[usize],
893    header: &FitsHeader,
894) -> CoordinateSystem {
895    let mut output = CoordinateSystem::new().with_obs_info(input.obs_info().clone());
896    let mut index = 0;
897    let mut offset = 0;
898    while index < input.n_coordinates() {
899        let mut axes = Vec::new();
900        while index < input.n_coordinates() {
901            let CoordinateModel::Linear(axis) = input.coordinate(index) else {
902                break;
903            };
904            axes.push(axis);
905            index += 1;
906        }
907        if axes.is_empty() {
908            let coordinate = input.coordinate(index);
909            offset += coordinate.n_pixel_axes();
910            output.add_coordinate(coordinate.clone());
911            index += 1;
912            continue;
913        }
914        let count = axes.iter().map(|axis| axis.n_pixel_axes()).sum();
915        let (transform, increments) = read_linear_wcs(header, &axis_order[offset..offset + count]);
916        offset += count;
917        let names = axes.iter().flat_map(|axis| axis.axis_names()).collect();
918        let units = axes.iter().flat_map(|axis| axis.axis_units()).collect();
919        output.add_coordinate(
920            LinearCoordinate::new(count, names, units)
921                .with_reference_value(
922                    axes.iter()
923                        .flat_map(|axis| axis.reference_value())
924                        .collect(),
925                )
926                .with_reference_pixel(
927                    axes.iter()
928                        .flat_map(|axis| axis.reference_pixel())
929                        .collect(),
930                )
931                .with_increment(increments)
932                .with_pc_matrix(transform),
933        );
934    }
935    output
936}

938fn read_linear_wcs(header: &FitsHeader, axes: &[usize]) -> (Array2<f64>, Vec<f64>) {
939    let matrix = |key: &str, row: usize, column: usize| {
940        header.get_float(&format!("{key}{}_{}", axes[row] + 1, axes[column] + 1))
941    };
942    let dimensions = (axes.len(), axes.len());
943    let cd = Array2::from_shape_fn(dimensions, |(row, column)| matrix("CD", row, column));
944    // casacore represents linear CD matrices directly with unit increments.
945    if cd.iter().any(Option::is_some) {
946        return (cd.mapv(|value| value.unwrap_or(0.0)), vec![1.0; axes.len()]);
947    }
948    let pc = Array2::from_shape_fn(dimensions, |(row, column)| {
949        matrix("PC", row, column).unwrap_or(if row == column { 1.0 } else { 0.0 })
950    });
951    let increments = axes
952        .iter()
953        .map(|axis| {
954            header
955                .get_float(&format!("CDELT{}", axis + 1))
956                .unwrap_or(1.0)
957        })
958        .collect();
959    (pc, increments)
960}

962fn fits_coordinate_record(
963    coordinates: &CoordinateSystem,
964    axis_order: &[usize],
965    world_order: Option<&[usize]>,
966    header: &FitsHeader,
967) -> Result<RecordValue, ImageError> {
968    let original = coordinates.to_record();
969    let mut record = original.clone();
970    let mut descriptors = CoordinateSystem::record_coordinates(&original)?;
971    let coordinate_kind = |field: &str| match field.trim_end_matches(|ch: char| ch.is_ascii_digit())
972    {
973        "direction" => 0,
974        "stokes" => 1,
975        "spectral" => 2,
976        _ => 3,
977    };
978    descriptors.sort_by_key(|coordinate| coordinate_kind(&coordinate.field));
979    let mapping_fields = ["pixelmap", "pixelreplace", "worldmap", "worldreplace"];
980    for descriptor in &descriptors {
981        let kind = descriptor
982            .field
983            .trim_end_matches(|ch: char| ch.is_ascii_digit());
984        let index = &descriptor.field[kind.len()..];
985        record.remove(&descriptor.field);
986        for field in mapping_fields {
987            record.remove(&format!("{field}{index}"));
988        }
989    }
990    for (index, coordinate) in descriptors.iter().enumerate() {
991        let kind = coordinate
992            .field
993            .trim_end_matches(|ch: char| ch.is_ascii_digit());
994        let old_index = &coordinate.field[kind.len()..];
995        for field in mapping_fields {
996            if let Some(value) = original.get(&format!("{field}{old_index}")) {
997                let value = if matches!(field, "pixelmap" | "worldmap")
998                    && let Value::Array(ArrayValue::Int32(map)) = value
999                {
1000                    Value::Array(ArrayValue::Int32(map.mapv(|axis| {
1001                        let pixel = axis_order[axis as usize];
1002                        if field == "worldmap" {
1003                            world_order.map_or(pixel, |order| order[pixel]) as i32
1004                        } else {
1005                            pixel as i32
1006                        }
1007                    })))
1008                } else {
1009                    value.clone()
1010                };
1011                record.upsert(format!("{field}{index}"), value);
1012            }
1013        }
1014        if let Some(Value::Record(mut fields)) = original.get(&coordinate.field).cloned() {
1015            let pixel_axes = coordinate
1016                .pixel_axes
1017                .iter()
1018                .map(|axis| axis.map(|axis| axis_order[axis]))
1019                .collect::<Vec<_>>();
1020            if kind == "direction" {
1021                set_fits_pole_and_projection(&mut fields, &pixel_axes, header);
1022                normalize_direction_matrix(&mut fields);
1023            } else if kind == "linear" {
1024                fields.remove("naxes");
1025                fields.remove("coordinate_type");
1026            } else if kind == "spectral"
1027                && let Some(Some(axis)) = pixel_axes.first()
1028            {
1029                let ctype = header
1030                    .get_string(&format!("CTYPE{}", axis + 1))
1031                    .unwrap_or("");
1032                if ctype.eq_ignore_ascii_case("VOPT") {
1033                    fields.upsert("nativeType", Value::Scalar(ScalarValue::Int32(2)));
1034                }
1035                if let Some(Value::Record(tabular)) = fields.get_mut("tabular") {
1036                    tabular.upsert(
1037                        "pc",
1038                        Value::Array(ArrayValue::Float64(ndarray::arr2(&[[1.0]]).into_dyn())),
1039                    );
1040                }
1041                if header.get("SPECSYS").is_none()
1042                    && header.get("VELREF").is_none()
1043                    && !ctype.to_ascii_uppercase().ends_with("-LSR")
1044                {
1045                    fields.upsert(
1046                        "system",
1047                        Value::Scalar(ScalarValue::String("Undefined".into())),
1048                    );
1049                }
1050                let system = fields.get("system").expect("spectral coordinate system");
1051                let mut conversion = RecordValue::default();
1052                conversion.upsert("system", system.clone());
1053                conversion.upsert(
1054                    "epoch",
1055                    Value::Record(epoch_to_record(&MEpoch::from_mjd(0.0, EpochRef::LAST))),
1056                );
1057                conversion.upsert(
1058                    "position",
1059                    Value::Record(position_to_record(&MPosition::new_itrf(0.0, 0.0, 0.0))),
1060                );
1061                conversion.upsert(
1062                    "direction",
1063                    Value::Record(direction_to_record(&MDirection::from_angles(
1064                        0.0,
1065                        std::f64::consts::FRAC_PI_2,
1066                        DirectionRef::J2000,
1067                    ))),
1068                );
1069                fields.upsert("conversion", Value::Record(conversion));
1070            }
1071            record.upsert(format!("{kind}{index}"), Value::Record(fields));
1072        }
1073    }
1074    Ok(record)
1075}

1077fn normalize_direction_matrix(record: &mut RecordValue) {
1078    let Some(Value::Array(ArrayValue::Float64(mut matrix))) = record.get("pc").cloned() else {
1079        return;
1080    };
1081    let Some(Value::Array(ArrayValue::Float64(mut increments))) = record.get("cdelt").cloned()
1082    else {
1083        return;
1084    };
1085    // The saved PC matrix is transposed with respect to the WCS transform.
1086    // casacore moves each world-axis row's norm into its CDELT.
1087    for (axis, increment) in matrix.axis_iter_mut(Axis(1)).zip(&mut increments) {
1088        let norm = axis
1089            .iter()
1090            .map(|coefficient| coefficient * coefficient)
1091            .sum::<f64>()
1092            .sqrt();
1093        if norm != 0.0 && norm != 1.0 {
1094            *increment *= norm;
1095            axis.into_iter()
1096                .for_each(|coefficient| *coefficient /= norm);
1097        }
1098    }
1099    record.upsert("pc", Value::Array(ArrayValue::Float64(matrix)));
1100    record.upsert("cdelt", Value::Array(ArrayValue::Float64(increments)));
1101}

1103fn set_fits_pole_and_projection(
1104    direction: &mut RecordValue,
1105    pixel_axes: &[Option<usize>],
1106    header: &FitsHeader,
1107) {
1108    let Some(Value::Scalar(ScalarValue::String(projection))) = direction.get("projection") else {
1109        return;
1110    };
1111    let sin = projection == "SIN";
1112    if sin
1113        || matches!(
1114            projection.as_str(),
1115            "TAN" | "STG" | "ARC" | "ZEA" | "AZP" | "ZPN"
1116        )
1117    {
1118        // The pole for a zenithal WCS is the center of the projection.
1119        if let Some(Value::Array(ArrayValue::Float64(reference))) = direction.get("crval")
1120            && let Some(latitude) = reference.get(IxDyn(&[1]))
1121        {
1122            direction.upsert(
1123                "latpole",
1124                Value::Scalar(ScalarValue::Float64(latitude.to_degrees())),
1125            );
1126        }
1127    }
1128    if sin && let Some(Some(axis)) = pixel_axes.get(1) {
1129        let values = [1, 2].map(|index| {
1130            header
1131                .get_float(&format!("PV{}_{index}", axis + 1))
1132                .unwrap_or(0.0)
1133        });
1134        direction.upsert(
1135            "projection_parameters",
1136            Value::Array(ArrayValue::Float64(ndarray::arr1(&values).into_dyn())),
1137        );
1138    }
1139}

1149fn read_hdu_cards(fits: &mut FitsFile) -> Result<Vec<String>, ImageError> {
1150    let (mut count, mut more, mut status) = (0, 0, 0);
1151    // SAFETY: the fitsio file is open on the image HDU and all outputs are live.
1152    unsafe { fitsio::sys::ffghsp(fits.as_raw(), &mut count, &mut more, &mut status) };
1153    check_status(status).map_err(fits_error)?;
1154    let mut cards = Vec::with_capacity(count as usize);
1155    for index in 1..=count {
1156        let mut buffer = [0; 81];
1157        // SAFETY: CFITSIO writes one card, at most 80 bytes and a terminator.
1158        unsafe { fitsio::sys::ffgrec(fits.as_raw(), index, buffer.as_mut_ptr(), &mut status) };
1159        check_status(status).map_err(fits_error)?;
1160        // SAFETY: a successful ffgrec NUL-terminates its output.
1161        cards.push(
1162            unsafe { CStr::from_ptr(buffer.as_ptr()) }
1163                .to_string_lossy()
1164                .into_owned(),
1165        );
1166    }
1167    Ok(cards)
1168}

1201fn alternate_wcs_key(name: &str) -> Option<(&str, u8)> {
1202    let last = *name.as_bytes().last()?;
1203    if !last.is_ascii_uppercase() {
1204        return None;
1205    }
1206    let base = &name[..name.len() - 1];
1207    wcs_key(base).then_some((base, last))
1208}

1210fn select_wcs(header: &FitsHeader, which_rep: usize) -> FitsHeader {
1211    if which_rep == 0 {
1212        return header.clone();
1213    }
1214    let mut alternatives: Vec<u8> = header
1215        .iter()
1216        .filter_map(|field| alternate_wcs_key(&field.name))
1217        .filter_map(|(base, suffix)| base.starts_with("CTYPE").then_some(suffix))
1218        .collect();
1219    alternatives.sort_unstable();
1220    alternatives.dedup();
1221    let Some(suffix) = alternatives.get(which_rep.min(alternatives.len()).saturating_sub(1)) else {
1222        return header.clone();
1223    };
1224    let mut selected = FitsHeader::new();
1225    for field in header.iter() {
1226        if let Some((base, candidate)) = alternate_wcs_key(&field.name) {
1227            if candidate == *suffix {
1228                selected.set(base, field.value.clone());
1229            }
1230        } else if !wcs_key(&field.name) {
1231            selected.set(&field.name, field.value.clone());
1232        }
1233    }
1234    selected
1235}

1258                    | "PCOUNT"
1259                    | "GCOUNT"
1260                    | "TELESCOP"
1261                    | "OBSERVER"
1262                    | "DATE"
1263                    | "DATE-OBS"
1264                    | "OBSGEO-X"
1265                    | "OBSGEO-Y"
1266                    | "OBSGEO-Z"
1267                    | "MJD-OBS"
1268                    | "TIMESYS"
1269                    | "BMAJ"
1270                    | "BMIN"
1271                    | "BPA"
1272                    | "IMTYPE"
1273            )
1274        {
1275            continue;
1276        }
1277        let value = match &field.value {
1278            FitsValue::Integer(value) => {
1279                i32::try_from(*value).map_or(ScalarValue::Int64(*value), ScalarValue::Int32)
1280            }
1281            FitsValue::Float(value) => ScalarValue::Float64(*value),
1282            FitsValue::Logical(value) => ScalarValue::Bool(*value),
1283            FitsValue::String(value) if name == "OBJECT" && value.is_empty() => {
1284                ScalarValue::String(" ".into())
1285            }
1286            FitsValue::String(value) => ScalarValue::String(value.clone()),
1287        };
1288        record.upsert(name.to_ascii_lowercase(), Value::Scalar(value));
1289    }
1290    record
1291}
