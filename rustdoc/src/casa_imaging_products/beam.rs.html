11/// Fit a restoring beam from one real PSF plane stored x-major.
12///
13/// # Errors
14///
15/// Returns [`ProductsError::BeamFitFailed`] when CASA's Gaussian fit does
16/// not converge.
17pub fn fit_restoring_beam(
18    psf: &[f32],
19    shape: [usize; 2],
20    cell_size_rad: [f64; 2],
21    cutoff: f32,
22) -> Result<RestoringBeam, ProductsError> {
23    casa_imaging_deconvolution::fit_restoring_beam(psf, shape, cell_size_rad, cutoff)
24        .map_err(|error| ProductsError::BeamFitFailed(error.to_string()))
25}

27/// What fitting one plane's PSF gives.
28#[derive(Clone, Copy, Debug, PartialEq)]
29pub(crate) enum PlaneBeam {
30    /// The plane's own beam.
31    Fitted(RestoringBeam),
32    /// The plane has no PSF, and so no beam.
33    Blank,
34    /// `FitGaussianPSF` refuses the plane, as its peak is outside the inner
35    /// quarter. `SIImageStore::makeImageBeamSet` warns and carries on.
36    Refused,
37}
38
39/// [`fit_restoring_beam`], as `SIImageStore::makeImageBeamSet` catches the
40/// refusal; any other failure stays a failure.
41///
42/// # Errors
43///
44/// As [`fit_restoring_beam`], but for a peak outside the inner quarter.
45pub(crate) fn fit_plane_beam(
46    psf: &[f32],
47    shape: [usize; 2],
48    cell_size_rad: [f64; 2],
49    cutoff: f32,
50) -> Result<PlaneBeam, ProductsError> {
51    match casa_imaging_deconvolution::fit_restoring_beam(psf, shape, cell_size_rad, cutoff) {
52        Ok(beam) => Ok(PlaneBeam::Fitted(beam)),
53        Err(error @ PsfBeamFitError::PeakOutsideInnerQuarter) => {
54            tracing::warn!(
55                "Error Gaussian fit to PSF : {error} : Setting restoring beam to largest valid beam."
56            );
57            Ok(PlaneBeam::Refused)
58        }
59        Err(error) => Err(ProductsError::BeamFitFailed(error.to_string())),
60    }
61}

63/// The beams of `fits`, `planes` of them to a domain: a refused plane takes
64/// its domain's largest (`getMaxAreaBeam`), or, where no plane has one, the
65/// tiny beam `SIImageStore::makeImageBeamSet` stores instead of a beam of no
66/// area. A plane with no PSF keeps none.
67pub(crate) fn settle_beams(fits: &[PlaneBeam], planes: usize) -> Box<[Option<RestoringBeam>]> {
68    let mut settled = Vec::with_capacity(fits.len());
69    for domain in fits.chunks(planes.max(1)) {
70        let largest = domain
71            .iter()
72            .filter_map(|fit| match fit {
73                PlaneBeam::Fitted(beam) => Some(*beam),
74                _ => None,
75            })
76            .max_by(|left, right| {
77                (left.major_fwhm_rad() * left.minor_fwhm_rad())
78                    .total_cmp(&(right.major_fwhm_rad() * right.minor_fwhm_rad()))
79            });
80        settled.extend(domain.iter().map(|fit| match fit {
81            PlaneBeam::Fitted(beam) => Some(*beam),
82            PlaneBeam::Blank => None,
83            PlaneBeam::Refused => Some(largest.unwrap_or_else(|| {
84                RestoringBeam::new(CASA_TINY_PSF_BEAM_RAD, CASA_TINY_PSF_BEAM_RAD, 0.0)
85                    .expect("a positive round beam")
86            })),
87        }));
88    }
89    settled.into_boxed_slice()
90}

92/// The restoring beam a request fixed, as the products restore with it and
93/// record it.
94///
95/// # Errors
96///
97/// [`ProductsError::BeamFitFailed`] if the model's validated beam is not a
98/// restoring beam, which cannot happen for a beam the model accepted.
99pub(crate) fn fixed_restoring_beam(
100    beam: FixedRestoringBeam,
101) -> Result<RestoringBeam, ProductsError> {
102    RestoringBeam::new(
103        beam.major_fwhm_rad(),
104        beam.minor_fwhm_rad(),
105        beam.position_angle_rad(),
106    )
107    .map_err(|error| ProductsError::BeamFitFailed(error.to_string()))
108}

110#[cfg(test)]

114    fn round(fwhm: f64) -> RestoringBeam {
115        RestoringBeam::new(fwhm, fwhm, 0.0).expect("a beam")

118    /// A lobe at `centre` of a plane of `[64, 32]`, x-major.
119    fn lobe(centre: [f64; 2]) -> Vec<f32> {
120        let mut psf = vec![0.0_f32; 64 * 32];
121        for x in 0..64 {
122            for y in 0..32 {
123                let [dx, dy] = [x as f64 - centre[0], y as f64 - centre[1]];
124                psf[x * 32 + y] = (-(dx * dx + dy * dy) / 8.0).exp() as f32;
125            }
126        }
127        psf
128    }

130    /// Only the inner-quarter refusal is caught: a fit that fails otherwise
131    /// stays a failure. The plane's axes are told apart on a non-square one.
132    #[test]
133    fn only_a_peak_outside_the_inner_quarter_is_left_to_another_plane() {
134        let cells = [1.0e-5, 1.0e-5];
135        let fit = |psf: &[f32]| fit_plane_beam(psf, [64, 32], cells, 0.35);
136        assert!(matches!(fit(&lobe([32.0, 16.0])), Ok(PlaneBeam::Fitted(_))));
137        // Inside both quarters of the plane, though y = 20 is outside x's.
138        assert!(matches!(fit(&lobe([32.0, 20.0])), Ok(PlaneBeam::Fitted(_))));
139        assert_eq!(fit(&lobe([8.0, 16.0])).expect("a fit"), PlaneBeam::Refused);
140        assert_eq!(fit(&lobe([32.0, 26.0])).expect("a fit"), PlaneBeam::Refused);
141        assert!(matches!(
142            fit(&vec![f32::NAN; 64 * 32]),
143            Err(ProductsError::BeamFitFailed(_))
144        ));
145    }
146
147    /// `makeImageBeamSet`: each domain's refused planes take its largest
148    /// beam, a domain with none a tiny one, and a blank plane keeps none.
149    #[test]
150    fn a_refused_plane_takes_its_domains_largest_beam() {
151        let (small, large) = (round(1.0e-5), round(2.0e-5));
152        let settled = settle_beams(
153            &[
154                PlaneBeam::Fitted(small),
155                PlaneBeam::Refused,
156                PlaneBeam::Fitted(large),
157                PlaneBeam::Blank,
158                PlaneBeam::Refused,
159                PlaneBeam::Blank,
160            ],
161            3,
162        );
163        assert_eq!(settled[..4], [Some(small), Some(large), Some(large), None]);
164        let tiny = settled[4].expect("a beam");
165        assert!(tiny.major_fwhm_rad() < 1.0e-11);
166        assert_eq!(settled[5], None);
167    }
