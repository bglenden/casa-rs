2//! `ia.modify(cl.torecord(), subtract=False)`: a component list added to an
3//! image, as CASA's `ComponentImager::modify` adds it through a
4//! `ComponentListImage` (`imageanalysis/Images`).
5//!
6//! The components are sampled at each pixel's direction
7//! ([`SkyComponent::sample_pixels`]) and frequency, a Stokes plane taking the
8//! Stokes parameter it is labelled with. A point goes whole into its nearest
9//! pixel: the pixel whose centre it lies within half a pixel of, looked for
10//! at the rounded pixel and then the eight around it. A point whose pixel
11//! is off the image is left out, as is one more than a pixel beyond its
12//! edge. A point no pixel holds, and every other shape, is sampled at every
13//! pixel.
14//!
15//! Without a spectral axis every component is sampled at the first one's
16//! reference frequency. With one, the frequencies are the image's world
17//! frequencies, in its conversion layer's frame when it has one, and a
18//! component whose spectrum is in another frame is sampled at them
19//! expressed in its own: CASA converts the reference frequency into the
20//! image's frame instead (`refFreqInFrame`), which gives the same ratio.
21//! The conversion uses the conversion layer's epoch, position and
22//! direction. An image without a usable frame, one with no conversion layer
23//! or with the placeholder `ia.fromshape` leaves (an epoch of 0 at the
24//! geocentre), is sampled with the frequencies unconverted, where CASA
25//! converts through the placeholder (`docs/CASA (C++) bugs.md`).
26//!
27//! An image in Jy/beam gains the components times the
28//! beam's area in pixels; any other unit without "pixel" in it is taken as
29//! Jy/pixel, with CASA's warning. As in CASA, a spectral or Stokes axis that
30//! is the image's first is not treated as one, and the image's mask is not
31//! consulted.
32//!
33//! The arithmetic is CASA's single precision: the components' sum in double
34//! precision, rounded, plus the points' (summed in single precision), added
35//! to the image.

51/// The image's grid, as the components are placed on it.
52struct Placement<'a> {
53    direction: &'a DirectionCoordinate,
54    grid: PixelGrid,
55    shape: [usize; 2],
56    frequencies_hz: Vec<f64>,
57    /// The frame of `frequencies_hz` and, when it is usable, the measures
58    /// frame to convert a component's reference frequency into it; `None`
59    /// without a spectral axis.
60    spectral_frame: Option<(FrequencyRef, Option<MeasFrame>)>,
61}
62
63impl Placement<'_> {
64    /// The frequencies at which `component` is sampled: the image's,
65    /// expressed in the frame of the component's spectrum (see the module
66    /// documentation).
67    fn frequencies_for(&self, component: &SkyComponent) -> Result<Vec<f64>, SkyModelError> {
68        let reference = component.spectrum().reference_frequency();
69        let unconverted = Ok(self.frequencies_hz.clone());
70        let Some((image_frame, measures)) = &self.spectral_frame else {
71            return unconverted;
72        };
73        let rest = |frame: FrequencyRef| frame == FrequencyRef::REST;
74        if reference.refer() == *image_frame || rest(reference.refer()) || rest(*image_frame) {
75            return unconverted;
76        }
77        let Some(measures) = measures else {
78            return unconverted;
79        };
80        let converted = reference.convert_to(*image_frame, measures).map_err(|_| {
81            SkyModelError::Model(format!(
82                "Cannot transform frequency from {} to {}",
83                image_frame.as_str(),
84                reference.refer().as_str()
85            ))
86        })?;
87        let ratio = reference.hz() / converted.hz();
88        Ok(self
89            .frequencies_hz
90            .iter()
91            .map(|frequency| frequency * ratio)
92            .collect())
93    }
94
95    /// The direction cosines of pixel `(x, y)`.
96    fn pixel_direction(&self, x: f64, y: f64) -> Result<[f64; 3], SkyModelError> {
97        let world = self.direction.to_world(&[x, y]).map_err(|_| {
98            SkyModelError::Model(format!(
99                "Unable to convert to world direction at pixel [{x}, {y}]"
100            ))
101        })?;
102        Ok(MDirection::from_angles(world[0], world[1], self.grid.frame).cosines())
103    }
104
105    /// Where a point goes (`_findPointSoures`): its pixel and its sampled
106    /// values `[Stokes, frequency]`, `None` when its pixel is off the
107    /// image, or an error that it lies in no pixel (to be sampled at all of
108    /// them).
109    fn point_pixel(&self, point: &SkyComponent) -> Result<PointPlacement, SkyModelError> {
110        let direction = point
111            .shape()
112            .reference_direction()
113            .convert_to(self.grid.frame, &MeasFrame::default())
114            .map_err(|error| SkyModelError::Model(error.to_string()))?;
115        let (lon, lat) = direction.as_angles();
116        let pixel = self.direction.to_pixel(&[lon, lat])?;
117        let target = [(pixel[0] + 0.5).floor(), (pixel[1] + 0.5).floor()];
118        let [nx, ny] = self.shape.map(|n| n as f64);
119        if target[0] < -1.0 || target[1] < -1.0 || target[0] > nx || target[1] > ny {
120            return Ok(PointPlacement::Outside);
121        }
122        let mut candidates = vec![target];
123        for dx in -1..=1 {
124            for dy in -1..=1 {
125                if dx != 0 || dy != 0 {
126                    candidates.push([target[0] + f64::from(dx), target[1] + f64::from(dy)]);
127                }
128            }
129        }
130        let frequencies = self.frequencies_for(point)?;
131        for [x, y] in candidates {
132            let mut values = Array3::zeros((4, 1, frequencies.len()));
133            let at = self.pixel_direction(x, y)?;
134            point.sample_pixels(&mut values, &[at], &self.grid, &frequencies)?;
135            if values.iter().any(|value| *value != 0.0) {
136                let inside = x >= 0.0 && y >= 0.0 && x < nx && y < ny;
137                return Ok(if inside {
138                    PointPlacement::At([x as usize, y as usize], values)
139                } else {
140                    PointPlacement::Outside
141                });
142            }
143        }
144        Ok(PointPlacement::Nowhere)
145    }
146}
147
148enum PointPlacement {
149    At([usize; 2], Array3<f64>),
150    Outside,
151    Nowhere,
152}

154/// `ia.modify(model, subtract=False)` with the components of `list`: see
155/// the module documentation.
156///
157/// # Errors
158///
159/// An empty list, an image with no direction coordinate, a Stokes plane
160/// other than I, Q, U or V, a component casa-rs does not sample, or an image
161/// that cannot be read or written.
162pub fn add_components(image_path: &Path, list: &ComponentList) -> Result<(), SkyModelError> {
163    if list.is_empty() {
164        return Err(SkyModelError::Model(
165            "There are no components in the model componentlist".to_string(),
166        ));
167    }
168    let mut image = PagedImage::<f32>::open(image_path)?;
169    let coordinates = image.coordinates().clone();
170    let shape = image.shape().to_vec();
171    let Some((direction_index, direction_axes)) =
172        pixel_axes(&coordinates, CoordinateType::Direction)
173    else {
174        return Err(SkyModelError::Model(
175            "Image does not have a direction coordinate".to_string(),
176        ));
177    };
178    let CoordinateModel::Direction(direction) = coordinates.coordinate(direction_index) else {
179        unreachable!("the coordinate found is a direction coordinate");
180    };
181    let increment = direction.increment();
182    let mut placement = Placement {
183        direction,
184        grid: PixelGrid {
185            frame: direction.direction_ref(),
186            longitude_size_rad: increment[0].abs(),
187            latitude_size_rad: increment[1].abs(),
188        },
189        shape: [shape[direction_axes[0]], shape[direction_axes[1]]],
190        frequencies_hz: Vec::new(),
191        spectral_frame: None,
192    };
193    let spectral =
194        pixel_axes(&coordinates, CoordinateType::Spectral).filter(|(_, axes)| axes[0] > 0);
195    placement.frequencies_hz = match &spectral {
196        Some((index, axes)) => {
197            let CoordinateModel::Spectral(coordinate) = coordinates.coordinate(*index) else {
198                unreachable!("the coordinate found is a spectral coordinate");
199            };
200            let usable = coordinate
201                .conversion_frame()
202                .filter(|frame| usable_frame(frame))
203                .cloned();
204            placement.spectral_frame = Some((coordinate.world_frequency_ref(), usable));
205            let to_hz = qa_value_in_hz(&coordinate.axis_units()[0])?;
206            (0..shape[axes[0]])
207                .map(|channel| Ok(coordinate.to_world(&[channel as f64])?[0] * to_hz))
208                .collect::<Result<_, SkyModelError>>()?
209        }
210        None => vec![list.components()[0].spectrum().reference_frequency().hz()],
211    };
212    let stokes = stokes_planes(&coordinates, &shape)?;
213
214    let mut extended = Vec::new();
215    let mut points: BTreeMap<[usize; 2], Vec<f32>> = BTreeMap::new();
216    let nfreq = placement.frequencies_hz.len();
217    let nstokes = stokes.planes.len();
218    for component in list.iter() {
219        if !matches!(component.shape().kind(), ShapeKind::Point) {
220            extended.push(component);
221            continue;
222        }
223        match placement.point_pixel(component)? {
224            PointPlacement::At(pixel, values) => {
225                let sums = points
226                    .entry(pixel)
227                    .or_insert_with(|| vec![0.0; nfreq * nstokes]);
228                for frequency in 0..nfreq {
229                    for (plane, &parameter) in stokes.planes.iter().enumerate() {
230                        let sum = &mut sums[frequency * nstokes + plane];
231                        *sum = (f64::from(*sum) + values[[parameter, 0, frequency]]) as f32;
232                    }
233                }
234            }
235            PointPlacement::Outside => {}
236            PointPlacement::Nowhere => extended.push(component),
237        }
238    }
239
240    let [nx, ny] = placement.shape;
241    let mut directions = Vec::with_capacity(nx * ny);
242    for y in 0..ny {
243        for x in 0..nx {
244            directions.push(placement.pixel_direction(x as f64, y as f64)?);
245        }
246    }
247    let mut samples = Array3::<f64>::zeros((4, nx * ny, nfreq));
248    for component in extended {
249        let frequencies = placement.frequencies_for(component)?;
250        component.sample_pixels(&mut samples, &directions, &placement.grid, &frequencies)?;
251    }
252
253    let scale = unit_scale(&image, direction)?;
254    let mut pixels = image.get()?;
255    let frequency_axis = spectral.as_ref().map(|(_, axes)| axes[0]);
256    let mut position = vec![0; shape.len()];
257    for y in 0..ny {
258        for x in 0..nx {
259            let point = points.get(&[x, y]);
260            for frequency in 0..nfreq {
261                for (plane, &parameter) in stokes.planes.iter().enumerate() {
262                    position[direction_axes[0]] = x;
263                    position[direction_axes[1]] = y;
264                    if let Some(axis) = frequency_axis {
265                        position[axis] = frequency;
266                    }
267                    if let Some(axis) = stokes.axis {
268                        position[axis] = plane;
269                    }
270                    let mut value = samples[[parameter, y * nx + x, frequency]] as f32;
271                    if let Some(point) = point {
272                        value += point[frequency * nstokes + plane];
273                    }
274                    if let Some(scale) = scale {
275                        value = (f64::from(value) * scale) as f32;
276                    }
277                    pixels[IxDyn(&position)] += value;
278                }
279            }
280        }
281    }
282    image.put_slice(&pixels, &vec![0; shape.len()])?;
283    image.save()?;
284    Ok(())
285}

287/// The image's Stokes axis and, for each of its planes, the Stokes
288/// parameter (0 to 3 for I, Q, U, V) it takes; one plane of I without one.
289struct StokesPlanes {
290    axis: Option<usize>,
291    planes: Vec<usize>,
292}
293
294fn stokes_planes(
295    coordinates: &casa_coordinates::CoordinateSystem,
296    shape: &[usize],
297) -> Result<StokesPlanes, SkyModelError> {
298    let Some((index, axes)) =
299        pixel_axes(coordinates, CoordinateType::Stokes).filter(|(_, axes)| axes[0] > 0)
300    else {
301        return Ok(StokesPlanes {
302            axis: None,
303            planes: vec![0],
304        });
305    };
306    let CoordinateModel::Stokes(coordinate) = coordinates.coordinate(index) else {
307        unreachable!("the coordinate found is a Stokes coordinate");
308    };
309    let planes = (0..shape[axes[0]])
310        .map(|plane| {
311            let name = coordinate
312                .stokes()
313                .get(plane)
314                .map_or("", |stokes| stokes.name());
315            ["I", "Q", "U", "V"]
316                .iter()
317                .position(|parameter| *parameter == name)
318                .ok_or_else(|| {
319                    SkyModelError::Model(format!(
320                        "Unsupported stokes value {name} at pixel {plane}"
321                    ))
322                })
323        })
324        .collect::<Result<_, _>>()?;
325    Ok(StokesPlanes {
326        axis: Some(axes[0]),
327        planes,
328    })
329}

331/// Whether a conversion layer's measures frame is one to convert with: an
332/// epoch and a position, neither the placeholder a spectral coordinate made
333/// from a shape holds (MJD 0, the geocentre), and a direction.
334fn usable_frame(frame: &MeasFrame) -> bool {
335    let epoch = frame
336        .epoch()
337        .is_some_and(|epoch| epoch.value().as_mjd() != 0.0);
338    let position = frame
339        .position()
340        .is_some_and(|position| position.as_itrf().iter().any(|value| *value != 0.0));
341    epoch && position && frame.direction().is_some()
342}

344/// Hz per unit of a spectral axis in `unit`.
345fn qa_value_in_hz(unit: &str) -> Result<f64, SkyModelError> {
346    let one = casa_types::quanta::Quantity::with_unit(
347        1.0,
348        casa_types::quanta::casacore::casacore_unit(unit).map_err(SimutilError::from)?,
349    );
350    Ok(qa::value_in(&one, "Hz")?)
351}

353/// The factor the components are scaled by for the image's brightness
354/// unit: the beam's area in pixels for Jy/beam with a beam, none otherwise.
355fn unit_scale(
356    image: &PagedImage<f32>,
357    direction: &DirectionCoordinate,
358) -> Result<Option<f64>, SkyModelError> {
359    let units = image.units().to_string();
360    if units.contains("pixel") {
361        return Ok(None);
362    }
363    if !units.contains("beam") {
364        tracing::warn!(
365            "Image units [{units}] are not dimensionally equivalent to Jy/pixel or Jy/beam. Assuming they are equivalent Jy/pixel"
366        );
367        return Ok(None);
368    }
369    let info = image.image_info()?;
370    if info.beam_set.has_multi_beam() {
371        return Err(SkyModelError::Model(
372            "casa-rs does not yet add components to an image with a beam per plane".to_string(),
373        ));
374    }
375    let Some(beam) = info.beam_set.single_beam().filter(|beam| !beam.is_null()) else {
376        tracing::warn!(
377            "No beam defined even though the image units contain a beam. Will assume the beam area is one pixel"
378        );
379        return Ok(None);
380    };
381    let increment = direction.increment();
382    Ok(Some(beam.area() / (increment[0] * increment[1]).abs()))
383}
