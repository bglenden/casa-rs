3//! CASA's `startmodel` (#841 Phase 6b): the image the model starts from
4//! (`SIImageStore::setModelImageOne`). As CASA takes it:
5//! - it is in Jy/pixel (`SynthesisImagerVi2::defineImage` refuses others);
6//! - its masked pixels are zero;
7//! - it is regridded linearly onto the main image's grid and scaled by the
8//!   ratio of the pixel areas, so its flux is kept (`regridToModelImage`:
9//!   `ImageRegrid` with `Interpolate2D::LINEAR`). CASA copies a start model
10//!   already on that grid; casa-rs regrids it too, which changes it only by
11//!   rounding;
12//! - its direction frame is the image's: casa-rs does not convert frames;
13//! - its Stokes planes are copied by position, as CASA leaves that axis
14//!   out of the regrid. Different labels produce a warning, not a refusal
15//!   or a polarization conversion (owner decision, #980, 2026-10-31);
16//! - a cube's start model has the image's spectral axis. An MFS image's has
17//!   one plane, at any frequency: CASA's regrid replicates a degenerate
18//!   spectral axis, and an MFS model holds one plane whatever frequency it
19//!   is stated at.
20//!
21//! It is the model's first generation, so the first major cycle subtracts
22//! its prediction and the restored image includes it. Compile reads its
23//! header only; its pixels are read and regridded once, when the initial
24//! pass seeds the model ([`StartModel::main_samples`]).
25

38/// An explicit start model or same-grid restart model checked against the main domain.
39#[derive(Clone, Debug)]
40pub(crate) struct StartModel {
41    source: Source,
42    input_shape: Vec<usize>,
43    template: RegridTemplate,
44}
45
46/// Where a start model's pixels come from.
47#[derive(Clone, Debug)]
48enum Source {
49    /// An explicit `startmodel`, regridded onto the image.
50    Regrid(PathBuf),
51    /// An earlier run's model on the image's own grid, read as it is: one
52    /// image, or one per Taylor term, `None` for a term with none (CASA
53    /// makes it zero).
54    Direct(Box<[Option<PathBuf>]>),
55}
56
57impl StartModel {
58    /// Check the image at `path` as the start model of `main`, an image of
59    /// `polarizations` Stokes planes and `planes` spectral planes.
60    pub(super) fn prepare(
61        path: &Path,
62        main: &PreparedImageDomain,
63        polarizations: usize,
64        planes: usize,
65    ) -> Result<Self, PrepareError> {
66        let [nx, ny] = main.image_size;
67        Self::onto(
68            path,
69            RegridTemplate {
70                coordinates: main.coordinates.clone(),
71                shape: vec![nx, ny, polarizations, planes],
72            },
73        )
74    }
75
76    /// Check the image at `path` as a start model of `template`'s image.
77    pub(crate) fn onto(path: &Path, mut template: RegridTemplate) -> Result<Self, PrepareError> {
78        let refuse = |reason: String| PrepareError::StartModel {
79            path: path.to_path_buf(),
80            reason,
81        };
82        let opened = AnyPagedImage::open(path).map_err(|error| refuse(error.to_string()))?;
83        let (units, coordinates, shape) = match &opened {
84            AnyPagedImage::Float32(image) => header(image),
85            AnyPagedImage::Float64(image) => header(image),
86            AnyPagedImage::Complex32(_) | AnyPagedImage::Complex64(_) => {
87                return Err(refuse("a start model is a real image".into()));
88            }
89        };
90        if units != "Jy/pixel" {
91            return Err(refuse(format!(
92                "its unit is {units:?}; a start model is in Jy/pixel, as CASA's tclean requires"
93            )));
94        }
95        let (ours, theirs) = (stokes(&coordinates), stokes(&template.coordinates));
96        // SIImageStore::regridToModelImage regrids only direction and
97        // spectral axes. Match its positional Stokes copying even when
98        // the labels differ. This template is private to seeding pixels;
99        // published coordinates still come from the compiled image domain.
100        if ours != theirs
101            && let (Some(input), Some(output)) = (
102                coordinates.find_coordinate(CoordinateType::Stokes),
103                template.coordinates.find_coordinate(CoordinateType::Stokes),
104            )
105        {
106            template
107                .coordinates
108                .replace_coordinate(output, coordinates.coordinate(input).clone());
109        }
110        check_regrid(&coordinates, &shape, &template).map_err(|error| {
111            refuse(format!(
112                "{error}; the non-regridded axis shapes and spectral coordinates must match"
113            ))
114        })?;
115        if ours != theirs {
116            tracing::warn!(startmodel = ?path, input_stokes = ?ours, image_stokes = ?theirs,
117                "startmodel Stokes labels differ: copying planes by position as CASA does; no polarization conversion is performed");
118        }
119        Ok(Self {
120            source: Source::Regrid(path.to_path_buf()),
121            input_shape: shape,
122            template,
123        })
124    }
125
126    /// Reuse an earlier run's model, whose images `terms` (one, or one per
127    /// Taylor term with `None` for a missing term) the caller checked on
128    /// the requested grid, without regridding. `template` is the model's
129    /// whole grid: `[nx, ny, stokes, planes]`, its planes the channels of
130    /// the one image or the Taylor terms.
131    pub(super) fn restart(terms: Box<[Option<PathBuf>]>, template: RegridTemplate) -> Self {
132        Self {
133            source: Source::Direct(terms),
134            input_shape: template.shape.clone(),
135            template,
136        }
137    }
138
139    /// Bytes [`Self::main_samples`] holds at most besides the model: the
140    /// image read (up to 8 bytes a pixel) with its mask, its copy with the
141    /// masked pixels zero, `regrid`'s own read of that copy, the regridded
142    /// image with its mask and the samples copied out of it, and `regrid`'s
143    /// planes: an input plane of `f64` with its mask, and an output plane of
144    /// `f64`, a flag and an input position each. The direction axes are the
145    /// first two of both images, as `check_regrid` requires. Direct restart
146    /// holds only the input image and the emitted samples, within this charge.
147    pub(crate) fn seeding_bytes(&self) -> u64 {
148        let input: usize = self.input_shape.iter().product();
149        let output: usize = self.template.shape.iter().product();
150        let input_plane = self.input_shape[0] * self.input_shape[1];
151        let output_plane = self.template.shape[0] * self.template.shape[1];
152        let position = size_of::<Option<[f64; 2]>>();
153        (input * (8 + 1 + 4 + 4)
154            + output * (4 + 1 + 4)
155            + input_plane * (8 + 1)
156            + output_plane * (8 + 1 + position)) as u64
157    }
158
159    /// The image's pixels along x and y.
160    pub(crate) fn image_size(&self) -> [usize; 2] {
161        [self.template.shape[0], self.template.shape[1]]
162    }
163
164    /// The image's polarizations.
165    pub(crate) fn polarizations(&self) -> usize {
166        self.template.shape[2]
167    }
168
169    /// Bytes making the samples physical holds besides
170    /// [`Self::seeding_bytes`]: the physical samples, and one plane's
171    /// sensitivity read in `f64`.
172    pub(crate) fn conversion_bytes(&self) -> u64 {
173        let output: usize = self.template.shape.iter().product();
174        let output_plane = self.template.shape[0] * self.template.shape[1];
175        (output * 4 + output_plane * 8) as u64
176    }
177
178    /// The main domain's model samples in the model's canonical order
179    /// (spectral plane, polarization, y, x, x fastest), either read directly
180    /// for restart or regridded for an explicit start model.
181    pub(crate) fn main_samples(&self) -> Result<Vec<f32>, ImageError> {
182        let path = match &self.source {
183            Source::Regrid(path) => path,
184            Source::Direct(terms) => {
185                let total: usize = self.template.shape.iter().product();
186                let per_term = total / terms.len();
187                let mut samples = Vec::with_capacity(total);
188                for term in terms {
189                    match term {
190                        Some(path) => {
191                            let image = casa_images::PagedImage::<f32>::open(path)?;
192                            // CASA's restart prediction uses StokesImageUtil::From's raw
193                            // lattice iterator; unlike explicit startmodel, it does not
194                            // zero a pixel mask.
195                            let values = image.get()?;
196                            // Iterate logical axes in model order (term or channel,
197                            // Stokes, y, x), independent of image storage order.
198                            samples.extend(values.reversed_axes().iter().copied());
199                        }
200                        None => samples.resize(samples.len() + per_term, 0.0),
201                    }
202                }
203                return Ok(samples);
204            }
205        };
206        let zeroed = match AnyPagedImage::open(path)? {
207            AnyPagedImage::Float32(image) => zeroed_copy(&image)?,
208            AnyPagedImage::Float64(image) => zeroed_copy(&image)?,
209            AnyPagedImage::Complex32(_) | AnyPagedImage::Complex64(_) => {
210                return Err(ImageError::InvalidMetadata(
211                    "a start model is a real image".into(),
212                ));
213            }
214        };
215        let regridded = regrid(&zeroed, &self.template, RegridMethod::Linear)?;
216        Ok(regridded
217            .data
218            .as_slice_memory_order()
219            .expect("regrid writes a contiguous Fortran-order image")
220            .to_vec())
221    }
222}

224/// The Stokes of `coordinates`' Stokes coordinate, in plane order; `None`
225/// without one.
226fn stokes(coordinates: &CoordinateSystem) -> Option<Vec<StokesType>> {
227    (0..coordinates.n_coordinates()).find_map(|index| match coordinates.coordinate(index) {
228        CoordinateModel::Stokes(stokes) => Some(stokes.stokes().to_vec()),
229        _ => None,
230    })
231}

233/// The unit, coordinates and shape [`StartModel::prepare`] reads.
234fn header<T: ImagePixel>(
235    image: &casa_images::PagedImage<T>,
236) -> (String, CoordinateSystem, Vec<usize>) {
237    (
238        image.units().to_string(),
239        image.coordinates().clone(),
240        image.shape().to_vec(),
241    )
242}

244/// `image` in memory, in single precision, with its masked pixels zero and
245/// no mask, as `setModelImageOne` copies it before regridding.
246fn zeroed_copy<T>(image: &casa_images::PagedImage<T>) -> Result<TempImage<f32>, ImageError>
247where
248    T: ImagePixel + Into<f64> + Copy,
249{
250    let mut values: ArrayD<f32> = image.get()?.mapv(|value| Into::<f64>::into(value) as f32);
251    if let Some(mask) = image.default_mask()? {
252        Zip::from(&mut values).and(&mask).for_each(|value, &valid| {
253            if !valid {
254                *value = 0.0;
255            }
256        });
257    }
258    let mut copy = TempImage::<f32>::new(
259        image.shape().to_vec(),
260        image.coordinates().clone(),
261        TempStoragePolicy::Memory,
262    )?;
263    copy.set_units(image.units())?;
264    copy.put_slice(&values, &vec![0; values.ndim()])?;
265    Ok(copy)
266}

268#[cfg(test)]
269mod tests {
270    use super::*;
271    use casa_coordinates::{
272        DirectionCoordinate, Projection, ProjectionType, SpectralCoordinate, StokesCoordinate,
273    };
274    use casa_types::measures::{direction::DirectionRef, frequency::FrequencyRef};
275    use ndarray::IxDyn;
276
277    #[test]
278    fn direct_restart_reads_raw_x_fastest_pixels_while_startmodel_zeros_the_mask() {
279        let root = tempfile::tempdir().unwrap();
280        let path = root.path().join("existing.model");
281        let shape = vec![4, 3, 1, 1];
282        let mut coordinates = CoordinateSystem::new();
283        coordinates.add_coordinate(DirectionCoordinate::new(
284            DirectionRef::J2000,
285            Projection::new(ProjectionType::SIN),
286            [1.0, 0.5],
287            [-1e-5, 1e-5],
288            [2.0, 1.0],
289        ));
290        coordinates.add_coordinate(StokesCoordinate::new(vec![StokesType::I]));
291        coordinates.add_coordinate(SpectralCoordinate::new(
292            FrequencyRef::LSRK,
293            1e9,
294            1e6,
295            0.0,
296            1e9,
297        ));
298        let mut image =
299            casa_images::PagedImage::<f32>::create(shape.clone(), coordinates.clone(), &path)
300                .unwrap();
301        image.set_units("Jy/pixel").unwrap();
302        let pixels =
303            ArrayD::from_shape_fn(IxDyn(&shape), |index| (index[0] + 10 * index[1]) as f32);
304        image.put_slice(&pixels, &[0; 4]).unwrap();
305        image.make_mask("external", true, false).unwrap();
306        image.save().unwrap();
307        drop(image);
308        let template = RegridTemplate {
309            coordinates,
310            shape: shape.clone(),
311        };
312        let explicit = StartModel::onto(&path, template.clone()).unwrap();
313        assert!(
314            explicit
315                .main_samples()
316                .unwrap()
317                .iter()
318                .all(|&value| value == 0.0)
319        );
320        // The task gate still refuses masked-model continuation until publication
321        // can preserve the mask; the seed reader must nevertheless match CASA.
322        let direct = StartModel::restart(Box::new([Some(path.clone())]), template.clone());
323        assert_eq!(
324            direct.main_samples().unwrap(),
325            vec![
326                0.0, 1.0, 2.0, 3.0, 10.0, 11.0, 12.0, 13.0, 20.0, 21.0, 22.0, 23.0
327            ]
328        );
329        // A Taylor family reads each term in order; a missing term is zero.
330        let terms = StartModel::restart(
331            Box::new([None, Some(path)]),
332            RegridTemplate {
333                shape: vec![4, 3, 1, 2],
334                ..template
335            },
336        );
337        let samples = terms.main_samples().unwrap();
338        assert_eq!(samples[..12], [0.0; 12]);
339        assert_eq!(samples[12..], direct.main_samples().unwrap()[..]);
340    }
