32/// Tables, selectors, and sampling for CASA's `calibrater.accumulate`.
33pub struct CalibrationAccumulationRequest {
34    /// Existing cumulative input when `interval_seconds` is negative.
35    pub table_in: PathBuf,
36    /// Incremental antenna calibration to multiply into the cumulative table.
37    pub incremental: PathBuf,
38    /// Result table. An empty path uses `table_in`.
39    pub table_out: PathBuf,
40    /// Cumulative fields to modify. Empty selects every field.
41    pub field: String,
42    /// Incremental fields to interpolate. Empty uses all solution fields.
43    pub calibration_field: String,
44    /// CASA time and frequency interpolation expression.
45    pub interpolation: String,
46    /// Negative reuses `table_in`. Zero or positive builds a cumulative time grid.
47    pub interval_seconds: f64,
48    /// Map from cumulative spectral windows to incremental ones. Negative is identity.
49    pub spectral_window_map: Vec<i32>,
50}

52/// Apply incremental calibration to an existing cumulative table or to a new
53/// time grid on the MeasurementSet.
54///
55/// # Errors
56/// Returns a description of selection, table, interpolation, or output errors.
57pub fn accumulate_calibration(
58    ms: &MeasurementSet,
59    selection: &MsSelection,
60    request: &CalibrationAccumulationRequest,
61) -> Result<(), String> {
62    if !request.interval_seconds.is_finite() {
63        return Err("accumulation time must be finite".into());
64    }
65    let destination = if request.table_out.as_os_str().is_empty() {
66        &request.table_in
67    } else {
68        &request.table_out
69    };
70    if destination.as_os_str().is_empty() {
71        return Err("an accumulation output table is required".into());
72    }
73    let fields = ms
74        .resolve_field_selector(&request.field)
75        .map_err(error_text)?;
76    let summary = summarize_table(&request.incremental).map_err(error_text)?;
77    let interpolation = ApplyInterpolationMode::parse_casa(&request.interpolation)?;
78    let mut spec = ApplyCalibrationTableSpec::new(&request.incremental);
79    let spectrum = ms.spectral_window().map_err(error_text)?;
80    let window_count = spectrum.row_count();
81    if request.spectral_window_map.len() > window_count {
82        return Err("spwmap has more entries than the MeasurementSet has spectral windows".into());
83    }
84    spec.spwmap.clone_from(&request.spectral_window_map);
85    spec.interp = interpolation;
86    spec.gainfield = match request.calibration_field.as_str() {
87        "" => None,
88        "nearest" => Some(GainFieldSelector::Nearest),
89        selection => Some(GainFieldSelector::Expression(selection.to_string())),
90    };
91    let ms_fields = (0..ms.field().map_err(error_text)?.row_count())
92        .map(|id| i32::try_from(id).map_err(error_text))
93        .collect::<Result<Vec<_>, _>>()?;
94    let (resolved_gainfields, resolved_nearest_gainfields) =
95        plan::resolve_gainfield(ms, &spec, &summary, &ms_fields).map_err(error_text)?;
96    let nearest = resolved_nearest_gainfields
97        .iter()
98        .map(|mapping| {
99            (
100                mapping.measurement_set_field_id,
101                mapping.calibration_field_id,
102            )
103        })
104        .collect::<BTreeMap<_, _>>();
105    let subtype = summary.table_subtype.clone();
106    let (receptors, channelized) = match subtype.as_str() {
107        "G Jones" => (2, false),
108        "T Jones" => (1, false),
109        "B Jones" => (2, true),
110        "Tf Jones" => (1, true),
111        _ => {
112            return Err(format!(
113                "calibration type {subtype} does not support accumulation"
114            ));
115        }
116    };
117    let plan = ApplyCalibrationTablePlan {
118        applicable_row_indices: None,
119        unavailable_row_indices: Vec::new(),
120        spec,
121        applicable_selected_row_count: 0,
122        summary,
123        resolved_gainfields,
124        resolved_nearest_gainfields,
125        spw_mapping: Vec::new(),
126        calibration_spectral_windows: Vec::new(),
127        interp: interpolation,
128        calwt: false,
129    };
130    let lookup = load_interpolated_calibration_table(&plan, None, ms).map_err(error_text)?;
131    let spectral = spectral_window_path(&request.incremental)
132        .map_err(error_text)?
133        .ok_or("incremental calibration table has no spectral windows")?;
134    let data_centers = (0..window_count)
135        .map(|row| {
136            let channels = spectrum.chan_freq(row).map_err(error_text)?;
137            Ok(channels.iter().sum::<f64>() / channels.len() as f64)
138        })
139        .collect::<Result<Vec<_>, String>>()?;
140    let increment = Increment {
141        available_windows: lookup.interpolation_windows(),
142        data_centers,
143        lookup,
144        frequencies: table_frequencies(&spectral)?,
145        fields: &fields,
146        nearest,
147        spec: plan.spec,
148        interpolation,
149    };
150    if request.interval_seconds < 0.0 {
151        return edit_calibration_table(&request.table_in, destination, |table, stage| {
152            if table.info().sub_type != subtype {
153                return Err("cumulative and incremental calibration table types differ".into());
154            }
155            repair_observation_metadata(table, stage)?;
156            increment.apply(table, stage)
157        });
158    }
159
160    let mut writer = prepare_empty_calibration_table(
161        ms,
162        destination,
163        "Complex",
164        &subtype,
165        (!channelized).then_some(SingleChannelFrequency::Mean),
166    )?;
167    let _destination_lock = if destination.exists() {
168        Some(TableWriteLock::acquire_to_reread(destination, 1).map_err(error_text)?)
169    } else {
170        None
171    };
172    let stage = writer.output_path().to_path_buf();
173    populate_time_grid(
174        ms,
175        selection,
176        request.interval_seconds,
177        receptors,
178        channelized,
179        &mut writer,
180    )?;
181    increment.apply(writer.table_mut(), &stage)?;
182    update_spectral_window_flags(writer.table_mut(), &stage)?;
183    writer.save_main().map_err(error_text)
184}

186struct Increment<'a> {
187    lookup: LoadedCalibrationTable,
188    available_windows: Vec<i32>,
189    data_centers: Vec<f64>,
190    nearest: BTreeMap<i32, i32>,
191    frequencies: Vec<Vec<f64>>,
192    fields: &'a [i32],
193    spec: ApplyCalibrationTableSpec,
194    interpolation: ApplyInterpolationMode,
195}
196
197impl Increment<'_> {
198    fn apply(&self, table: &mut Table, path: &Path) -> Result<(), String> {
199        let mut groups = BTreeMap::<_, Vec<_>>::new();
200        for row in 0..table.row_count() {
201            let window = read_i32(table, row, COL_SPECTRAL_WINDOW_ID)?;
202            let time = read_time(table, row)?;
203            let field = read_i32(table, row, COL_FIELD_ID)?;
204            groups
205                .entry((window, time.to_bits(), field))
206                .or_default()
207                .push(row);
208        }
209        let output_frequencies = table_frequencies(&path.join("SPECTRAL_WINDOW"))?;
210        let selected = groups
211            .keys()
212            .map(|&(window, _, _)| window)
213            .collect::<std::collections::BTreeSet<_>>()
214            .into_iter()
215            .collect::<Vec<_>>();
216        let mapping = plan::resolve_spw_mapping(
217            &self.spec,
218            &self.available_windows,
219            self.frequencies.len(),
220            &selected,
221        )
222        .map_err(error_text)?
223        .into_iter()
224        .map(|mapping| (mapping.data_spw_id, mapping.calibration_spw_id))
225        .collect::<BTreeMap<_, _>>();
226        for ((window, time, field), rows) in groups {
227            if !self.fields.is_empty() && self.fields.binary_search(&field).is_err() {
228                continue;
229            }
230            let target = usize::try_from(window).map_err(error_text)?;
231            let mapped = mapping[&window];
232            if self.available_windows.binary_search(&mapped).is_err() {
233                continue;
234            }
235            let source = usize::try_from(mapped).map_err(error_text)?;
236            let axis = output_frequencies
237                .get(target)
238                .ok_or("cumulative spectral window is missing")?;
239            let data_center = *self
240                .data_centers
241                .get(target)
242                .ok_or("cumulative calibration window is absent from the MeasurementSet")?;
243            for row in rows {
244                let antenna = read_i32(table, row, COL_ANTENNA1)?;
245                let observation = read_i32(table, row, COL_OBSERVATION_ID)?;
246                let scan = read_i32(table, row, COL_SCAN_NUMBER)?;
247                let calfield = self.nearest.get(&field).copied().unwrap_or(-1);
248                let correction = self
249                    .lookup
250                    .complex_solution_for_field(
251                        calfield,
252                        TimeQuery {
253                            observation,
254                            scan,
255                            window: mapped,
256                            antenna,
257                            time: f64::from_bits(time),
258                            frequency_hz: Some(data_center),
259                        },
260                    )
261                    .map_err(error_text)?;
262                let correction = correction
263                    .map(|corr| self.resample(corr, source, axis, data_center))
264                    .transpose()?;
265                multiply_row(table, row, correction)?;
266            }
267        }
268        Ok(())
269    }
270
271    fn resample(
272        &self,
273        solution: ComplexSolution,
274        source: usize,
275        output: &[f64],
276        data_center: f64,
277    ) -> Result<ComplexSolution, String> {
278        if solution.channels == 1 {
279            return Ok(solution);
280        }
281        let input = self
282            .frequencies
283            .get(source)
284            .ok_or("incremental spectral window is missing")?;
285
286        let (relative_input, relative_output);
287        let (input, output) = if matches!(
288            self.interpolation,
289            ApplyInterpolationMode::Casa {
290                relative_frequency: true,
291                ..
292            }
293        ) {
294            relative_input = relative_axis(input);
295            relative_output = relative_axis_at_center(output, data_center);
296            (relative_input.as_slice(), relative_output.as_slice())
297        } else {
298            (input.as_slice(), output)
299        };
300        if input == output {
301            return Ok(solution);
302        }
303        solution
304            .resample_frequency(input, output, self.interpolation.frequency_mode())
305            .map_err(error_text)
306    }
307}

309fn populate_time_grid(
310    ms: &MeasurementSet,
311    selection: &MsSelection,
312    interval: f64,
313    receptors: usize,
314    channelized: bool,
315    writer: &mut crate::writer::CalibrationTableWriter<'_>,
316) -> Result<(), String> {
317    let whole_ms = MsSelection::default();
318    let mut selected =
319        collect_selected_rows(ms, if interval > 0.0 { &whole_ms } else { selection }, true)
320            .map_err(error_text)?;
321    if selected.iter().any(|row| !row.time_seconds.is_finite()) {
322        return Err("accumulation requires finite MeasurementSet times".into());
323    }
324    selected.sort_by(|a, b| {
325        let group = (a.field_id, a.data_desc_id).cmp(&(b.field_id, b.data_desc_id));
326        let time = a.time_seconds.total_cmp(&b.time_seconds);
327        a.array_id.cmp(&b.array_id).then(if interval > 0.0 {
328            time.then(group)
329        } else {
330            group.then(time)
331        })
332    });
333    let spectrum = ms.spectral_window().map_err(error_text)?;
334    let antennas = ms.antenna().map_err(error_text)?.row_count();
335    for group in selected.chunk_by(|a, b| {
336        a.array_id == b.array_id && a.field_id == b.field_id && a.data_desc_id == b.data_desc_id
337    }) {
338        let channels = if channelized {
339            usize::try_from(
340                spectrum
341                    .num_chan(group[0].data_spw_id as usize)
342                    .map_err(error_text)?,
343            )
344            .map_err(error_text)?
345        } else {
346            1
347        };
348        let mut start = 0;
349        while start < group.len() {
350            let first = &group[start];
351            let count = if interval == 0.0 {
352                group.len() - start
353            } else {
354                // MSIter places the first sample ten milliseconds into
355                // each interval so floating-point boundary comparisons agree.
356                let origin = first.time_seconds - 0.01;
357                let bin = |time: f64| ((time - origin) / interval).floor();
358                let first_bin = bin(first.time_seconds);
359                group[start..].partition_point(|row| {
360                    row.time_seconds == first.time_seconds
361                        || (interval >= 2.0 * f64::MIN_POSITIVE
362                            && bin(row.time_seconds) == first_bin)
363                })
364            };
365            let mut samples = group[start..start + count]
366                .iter()
367                .map(|row| row.time_seconds)
368                .peekable();
369            let mut sum = 0.0;
370            let mut times = 0;
371            while let Some(time) = samples.next() {
372                while samples.peek() == Some(&time) {
373                    samples.next();
374                }
375                sum += time;
376                times += 1;
377            }
378            let time = sum / f64::from(times);
379            let shape = IxDyn(&[receptors, channels]);
380            for antenna in 0..antennas {
381                let row = RecordValue::new(vec![
382                    RecordField::new(COL_TIME, Value::Scalar(ScalarValue::Float64(time))),
383                    RecordField::new(COL_INTERVAL, Value::Scalar(ScalarValue::Float64(0.0))),
384                    id(COL_FIELD_ID, first.field_id),
385                    id(COL_SPECTRAL_WINDOW_ID, first.data_spw_id),
386                    id(COL_ANTENNA1, i32::try_from(antenna).map_err(error_text)?),
387                    id(COL_ANTENNA2, -1),
388                    id(COL_SCAN_NUMBER, first.scan_number),
389                    id(COL_OBSERVATION_ID, first.observation_id),
390                    RecordField::new(
391                        COL_CPARAM,
392                        Value::Array(ArrayValue::Complex32(ArrayD::from_elem(
393                            shape.clone(),
394                            Complex32::new(1.0, 0.0),
395                        ))),
396                    ),
397                    RecordField::new(
398                        COL_FLAG,
399                        Value::Array(ArrayValue::Bool(ArrayD::from_elem(shape.clone(), false))),
400                    ),
401                    RecordField::new(
402                        COL_PARAMERR,
403                        Value::Array(ArrayValue::Float32(ArrayD::zeros(shape.clone()))),
404                    ),
405                    RecordField::new(
406                        COL_SNR,
407                        Value::Array(ArrayValue::Float32(ArrayD::from_elem(shape.clone(), 1.0))),
408                    ),
409                ]);
410                writer.append(row).map_err(error_text)?;
411            }
412            start += count;
413        }
414    }
415    Ok(())
416}

418fn id(name: &str, value: i32) -> RecordField {
419    RecordField::new(name, Value::Scalar(ScalarValue::Int32(value)))

422fn table_frequencies(path: &Path) -> Result<Vec<Vec<f64>>, String> {
423    let table = Table::open(TableOptions::new(path)).map_err(error_text)?;
424    let spectrum = casa_ms::subtables::MsSpectralWindow::new(&table);
425    (0..table.row_count())
426        .map(|row| spectrum.chan_freq(row).map_err(error_text))
427        .collect()
428}

430fn read_i32(table: &Table, row: usize, name: &str) -> Result<i32, String> {
431    match table
432        .cell_accessor(row, name)
433        .and_then(|cell| cell.scalar())
434        .map_err(error_text)?
435    {
436        ScalarValue::Int32(value) => Ok(*value),
437        _ => Err(format!("invalid calibration table {name} at row {row}")),
438    }
439}

441fn read_time(table: &Table, row: usize) -> Result<f64, String> {
442    match table
443        .cell_accessor(row, COL_TIME)
444        .and_then(|cell| cell.scalar())
445        .map_err(error_text)?
446    {
447        ScalarValue::Float64(value) => Ok(*value),
448        _ => Err(format!("invalid calibration table TIME at row {row}")),
449    }
450}

452fn multiply_row(
453    table: &mut Table,
454    row: usize,
455    update: Option<ComplexSolution>,
456) -> Result<(), String> {
457    let read = |column| {
458        table
459            .cell_accessor(row, column)
460            .and_then(|cell| cell.array())
461            .cloned()
462            .map_err(error_text)
463    };
464    let ArrayValue::Bool(mut flags) = read(COL_FLAG)? else {
465        return Err("calibration table does not have boolean FLAG cells".into());
466    };
467    let ArrayValue::Complex32(mut values) = read(COL_CPARAM)? else {
468        return Err("calibration table does not have complex CPARAM cells".into());
469    };
470    if values.shape() != flags.shape() || values.ndim() != 2 {
471        return Err("calibration solution and flag arrays disagree".into());
472    }
473    if let Some(update) = update {
474        if update.receptors != values.shape()[0]
475            || (update.channels != 1 && update.channels != values.shape()[1])
476        {
477            return Err("cumulative and incremental solution shapes disagree".into());
478        }
479        for receptor in 0..values.shape()[0] {
480            for channel in 0..values.shape()[1] {
481                let index = receptor * update.channels + channel.min(update.channels - 1);
482                let cell = [receptor, channel];
483                flags[cell] |= update.flags[index];
484                if !flags[cell] {
485                    values[cell] *= update.values[index];
486                }
487            }
488        }
489    } else {
490        flags.fill(true);
491    }
492    let mut writer = table
493        .row_accessor_mut()
494        .prepare(&[COL_CPARAM, COL_FLAG])
495        .map_err(error_text)?;
496    writer.seek(row).map_err(error_text)?;
497    writer
498        .set_value_at(0, Value::Array(ArrayValue::Complex32(values)))
499        .map_err(error_text)?;
500    writer
501        .set_value_at(1, Value::Array(ArrayValue::Bool(flags)))
502        .map_err(error_text)?;
503    writer.flush().map_err(error_text)
504}
