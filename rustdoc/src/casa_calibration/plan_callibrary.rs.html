14/// VI1 queries CLPatchPanel with wildcard scan and intent IDs. A library
15/// contributes one registered context for each visibility buffer.
16pub(super) fn choose_legacy_slices(
17    tables: &mut [PlannedTable],
18    rows: &[ApplyRowPlan],
19    registrations: &[crate::execute::LibraryApplyRegistration],
20) -> LegacyScanOverrides {
21    let mut overrides = LegacyScanOverrides::new();
22    for registration in registrations
23        .iter()
24        .filter(|library| library.tables.len() > 1)
25    {
26        let slices = &mut tables[registration.tables.clone()];
27        let mut assigned = vec![Vec::new(); slices.len()];
28        let mut unavailable = vec![Vec::new(); slices.len()];
29        let mut priorities = HashMap::new();
30        for row in rows {
31            let context = (
32                row.calibration_observation_id,
33                row.observation_id,
34                row.field_id,
35                row.data_spw_id,
36            );
37            let preferred = *priorities
38                .entry(context)
39                .or_insert_with(|| legacy_priority(row, slices));
40            if let Some((index, [scan, _])) = preferred {
41                assigned[index].push(row.row_index);
42                if scan >= 0
43                    && scan != row.scan_number
44                    && slices[index]
45                        .0
46                        .spec
47                        .callibrary
48                        .as_ref()
49                        .is_some_and(|library| library.scan_map.active())
50                {
51                    overrides.insert((registration.tables.start + index, row.row_index), scan);
52                }
53            } else if let Some((index, (fallback, _))) = slices
54                .iter()
55                .enumerate()
56                .find(|(_, (table, _))| table.applies_to(row))
57            {
58                assigned[index].push(row.row_index);
59                if fallback
60                    .unavailable_row_indices
61                    .binary_search(&row.row_index)
62                    .is_ok()
63                {
64                    unavailable[index].push(row.row_index);
65                }
66            }
67        }
68        for (((table, _), mut rows), mut missing) in
69            slices.iter_mut().zip(assigned).zip(unavailable)
70        {
71            rows.sort_unstable();
72            missing.sort_unstable();
73            table.applicable_selected_row_count = rows.len();
74            table.applicable_row_indices = Some(rows);
75            table.unavailable_row_indices = missing;
76        }
77    }
78    overrides
79}

81// Observation, field and SPW already match the buffer. Only the scan and
82// intent contribute to ordering the candidate keys.
83type LegacyPriority = (usize, [i32; 2]);
84
85fn legacy_priority(row: &ApplyRowPlan, slices: &[PlannedTable]) -> Option<LegacyPriority> {
86    let matches = |values: &[i32], id| values.contains(&id) || values.contains(&-1);
87    let mut candidates = Vec::new();
88    for (index, (table, groups)) in slices.iter().enumerate() {
89        if !table.spec.apply_to.matches(row) {
90            continue;
91        }
92        let before = candidates.len();
93        candidates.extend(groups.iter().filter_map(|keys| {
94            if matches(&keys[0], row.calibration_observation_id)
95                && matches(&keys[2], row.field_id)
96                && matches(&keys[4], row.data_spw_id)
97            {
98                Some((index, [*keys[1].iter().min()?, *keys[3].iter().min()?]))
99            } else {
100                None
101            }
102        }));
103        candidates[before..].sort_unstable();
104    }
105    let has_wildcard_scan = candidates.iter().any(|(_, key)| key[0] < 0);
106    let has_concrete_scan = candidates.iter().any(|(_, key)| key[0] >= 0);
107    let has_intent = candidates.iter().any(|(_, key)| key[1] >= 0);
108    if has_wildcard_scan && has_concrete_scan && has_intent {
109        first_registered_key(&candidates)
110    } else {
111        candidates
112            .into_iter()
113            .min_by(|(_, left), (_, right)| compare_legacy_keys(left, right))
114    }
115}
116
117fn first_registered_key(candidates: &[LegacyPriority]) -> Option<LegacyPriority> {
118    // A wildcard in a stored key can make the comparator non-transitive.
119    // Respect registration order when constructing a single buffer lookup.
120    // A CASA index with cyclic keys can also update more than one result.
121    let mut children = vec![[None; 2]; candidates.len()];
122    for (index, (_, key)) in candidates.iter().enumerate().skip(1) {
123        let mut node = 0;
124        loop {
125            let right = !compare_legacy_keys(key, &candidates[node].1).is_lt();
126            let next = &mut children[node][usize::from(right)];
127            match *next {
128                Some(child) => node = child,
129                None => {
130                    *next = Some(index);
131                    break;
132                }
133            }
134        }
135    }
136    let mut first = 0;
137    while let Some(child) = children[first][0] {
138        first = child;
139    }
140    candidates.get(first).copied()
141}
142
143fn compare_legacy_keys(left: &[i32; 2], right: &[i32; 2]) -> std::cmp::Ordering {
144    left.iter()
145        .zip(right)
146        .find_map(|(&left, &right)| {
147            (left >= 0 && right >= 0 && left != right).then(|| left.cmp(&right))
148        })
149        .unwrap_or(std::cmp::Ordering::Equal)
150}

152pub(super) fn selected_rows(
153    ms: &MeasurementSet,
154    spec: &ApplyCalibrationTableSpec,
155    selected: &[ApplyRowPlan],
156    legacy_corruption: bool,
157) -> Result<Option<Vec<usize>>, ApplyPlanError> {
158    let Some(library) = spec
159        .callibrary
160        .as_ref()
161        .filter(|library| library.selection != MsSelection::new())
162    else {
163        return Ok(None);
164    };
165    if let Some(spw) = &library.selection.spw {
166        ms.resolve_spw_selector(spw)
167            .map_err(|source| ApplyPlanError::Selection { source })?;
168    }
169    let resolved = resolve_calibration_selection(ms, &library.selection).map_err(|error| {
170        ApplyPlanError::Selection {
171            source: MsError::InvalidInput(error.to_string()),
172        }
173    })?;
174    let mut rows = resolved.row_indices().collect::<Vec<_>>();
175    rows.sort_unstable();
176    rows.dedup();
177    if legacy_corruption && (library.selection.scan.is_some() || library.selection.intent.is_some())
178    {
179        // VI1 syncs observation, field, and SPW, while CLPatchPanel looks
180        // up scan and intent with wildcard keys. Retain the contexts that
181        // the slice registered and admit data from other scans and intents.
182        return Ok(Some(legacy_slice_rows(library, selected, &rows)));
183    }
184    if library.selection.intent.is_some() || library.selection.observation.is_some() {
185        // Calibrater::correct2 sorts ARRAY/SCAN/FIELD/DDID/TIME. VisCal::syncMeta2
186        // uses the first row's STATE_ID and observation for the entire buffer.
187        let indices = selected.iter().map(|row| row.row_index).collect::<Vec<_>>();
188        let arrays = load_i32_column_rows(ms.main_table(), "ARRAY_ID", &indices)?;
189        let mut groups = HashMap::new();
190        let mut grouped = Vec::new();
191        for (index, row) in selected.iter().enumerate() {
192            let key = (
193                value_at_i32(&arrays, index, "ARRAY_ID")?,
194                row.scan_number,
195                row.field_id,
196                row.data_desc_id,
197                row.time_seconds.to_bits(),
198            );
199            let applies = *groups
200                .entry(key)
201                .or_insert_with(|| rows.binary_search(&row.row_index).is_ok());
202            if applies {
203                grouped.push(row.row_index);
204            }
205        }
206        rows = grouped;
207        rows.sort_unstable();
208    }
209    Ok(Some(rows))
210}

212fn legacy_slice_rows(
213    library: &crate::CallibraryOptions,
214    selected: &[ApplyRowPlan],
215    matched_rows: &[usize],
216) -> Vec<usize> {
217    let context = |row: &ApplyRowPlan| {
218        (
219            library
220                .selection
221                .observation
222                .as_ref()
223                .map(|_| row.calibration_observation_id),
224            library.selection.field.as_ref().map(|_| row.field_id),
225            row.data_spw_id,
226        )
227    };
228    let registered = selected
229        .iter()
230        .filter(|row| matched_rows.binary_search(&row.row_index).is_ok())
231        .map(context)
232        .collect::<BTreeSet<_>>();
233    let mut rows = selected
234        .iter()
235        .filter(|row| registered.contains(&context(row)))
236        .map(|row| row.row_index)
237        .collect::<Vec<_>>();
238    rows.sort_unstable();
239    rows
240}

242/// CLPatchPanel distinguishes absent context axes from absent antennas: the
243/// former prevent the buffer's correction, while an entirely unlinked antenna
244/// group passes through this table without adding flags.
245pub(super) fn arrange_contexts(
246    ms: &MeasurementSet,
247    plan: &mut ApplyCalibrationTablePlan,
248    selected: &[ApplyRowPlan],
249) -> Result<Vec<PatchGroup>, ApplyPlanError> {
250    let Some(library) = &plan.spec.callibrary else {
251        return Ok(Vec::new());
252    };
253    if plan.summary.table_subtype == "BPOLY" {
254        return Ok(Vec::new());
255    }
256    let table = Table::open(TableOptions::new(&plan.spec.path)).map_err(|source| {
257        ApplyPlanError::OpenSubtable {
258            path: plan.spec.path.display().to_string(),
259            subtable: "MAIN",
260            source,
261        }
262    })?;
263    let partition_field =
264        !library.field_map.is_empty() || !plan.resolved_nearest_gainfields.is_empty();
265    let mut contexts = HashMap::<_, BTreeSet<i32>>::new();
266    for row in 0..table.row_count() {
267        let field = get_i32(&table, row, "FIELD_ID")?;
268        if !plan.resolved_gainfields.is_empty()
269            && !plan.resolved_gainfields.iter().any(|f| f.field_id == field)
270        {
271            continue;
272        }
273        let observation = if library.observation_map.active() {
274            get_i32(&table, row, "OBSERVATION_ID")?
275        } else {
276            -1
277        };
278        let scan = if library.scan_map.active() {
279            get_i32(&table, row, "SCAN_NUMBER")?
280        } else {
281            -1
282        };
283        let key = (
284            observation,
285            scan,
286            if partition_field { field } else { -1 },
287            get_i32(&table, row, "SPECTRAL_WINDOW_ID")?,
288        );
289        contexts
290            .entry(key)
291            .or_default()
292            .insert(get_i32(&table, row, "ANTENNA1")?);
293    }
294    let antennas = ms
295        .antenna()
296        .map_err(|source| ApplyPlanError::Selection { source })?
297        .row_count();
298    let requested_antennas = (0..antennas as i32)
299        .map(|id| {
300            library
301                .antenna_map
302                .get(id as usize)
303                .or_else(|| library.antenna_map.last())
304                .copied()
305                .filter(|id| *id >= 0)
306                .unwrap_or(id)
307        })
308        .collect::<BTreeSet<_>>();
309    let groups = keys::groups(ms, plan, selected, &contexts)?;
310    let mut active = Vec::new();
311    let mut unavailable = Vec::new();
312    for row in selected.iter().filter(|row| plan.applies_to(row)) {
313        let field = library
314            .field_map
315            .get(row.field_id as usize)
316            .copied()
317            .unwrap_or_else(|| {
318                plan.resolved_nearest_gainfields
319                    .iter()
320                    .find(|f| f.measurement_set_field_id == row.field_id)
321                    .map_or(-1, |f| f.calibration_field_id)
322            });
323        let window = plan
324            .spw_mapping
325            .iter()
326            .find(|m| m.data_spw_id == row.data_spw_id)
327            .expect("mapped applicable row")
328            .calibration_spw_id;
329        let key = (
330            library.observation_map.map(row.calibration_observation_id),
331            library.scan_map.map(row.scan_number),
332            field,
333            window,
334        );
335        match contexts.get(&key) {
336            None => {
337                unavailable.push(row.row_index);
338                active.push(row.row_index);
339            }
340            Some(ants) if !ants.is_disjoint(&requested_antennas) => active.push(row.row_index),
341            Some(_) => {}
342        }
343    }
344    active.sort_unstable();
345    unavailable.sort_unstable();
346    plan.unavailable_row_indices = unavailable;
347    plan.applicable_selected_row_count = active.len();
348    plan.applicable_row_indices = Some(active);
349    Ok(groups)
350}

352/// Multiple slices of one table may cover disjoint data, but CLPatchPanel
353/// refuses to assign the same calibration table to the same MS context twice.
354pub(super) fn validate_overlaps(tables: &[PlannedTable]) -> Result<(), ApplyPlanError> {
355    let mut deployed = HashMap::<String, Vec<&PatchGroup>>::new();
356    for (table, groups) in tables {
357        let Some(library) = &table.spec.callibrary else {
358            continue;
359        };
360        let literal = if library.table_name.is_empty() {
361            table.spec.path.display().to_string()
362        } else {
363            library.table_name.clone()
364        };
365        let previous = deployed.entry(literal).or_default();
366        for group in groups {
367            if previous.iter().any(|prior| keys::overlap(prior, group)) {
368                return Err(ApplyPlanError::Callibrary { message: "Attempted duplicate MSCalPatchKey: overlapping wildcard contexts of one calibration table".into() });
369            }
370        }
371        previous.extend(groups);
372    }
373    Ok(())
374}

376/// Keep physical IDs in the row plan, alongside the scalar metadata VisCal sees.
377pub(super) fn buffer_observations(
378    rows: &mut [ApplyRowPlan],
379    array_id: impl Fn(usize) -> Result<i32, ApplyPlanError>,
380) -> Result<(), ApplyPlanError> {
381    let mut first = HashMap::new();
382    for (index, row) in rows.iter_mut().enumerate() {
383        let key = (
384            array_id(index)?,
385            row.scan_number,
386            row.field_id,
387            row.data_desc_id,
388            row.time_seconds.to_bits(),
389        );
390        row.calibration_observation_id = *first.entry(key).or_insert(row.observation_id);
391    }
392    Ok(())
