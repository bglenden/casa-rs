59    pub(super) fn prepare_pointing_axes(
60        &mut self,
61        task: &TaskInvocation<'_>,
62        pointing: &PointingAxes,
63        has_calibration_or_overlays: bool,
64    ) -> Result<(), TaskFailure> {
65        let x = self.x_axis.expect("PlotMS resolves the primary X axis");
66        let y = self.y_axis.expect("PlotMS resolves the primary Y axis");
67        let metadata_axis =
68            |axis: MsAxis| axis.is_pointing() || matches!(axis, MsAxis::Time | MsAxis::Channel);
69        let metadata_pair = metadata_axis(x) && metadata_axis(y);
70        let real_pair = x.is_pointing() && y == MsAxis::Real;
71        let single_metadata_pair = metadata_pair && pointing.len() == 1;
72        let averaging = self.plotms_averaging_active();
73        if !(metadata_pair || real_pair)
74            || self.y_axis2.is_some()
75            || has_calibration_or_overlays
76            || raw_symbol_visibility(task).flagged
77            || (self.iteraxis.is_some()
78                && !(self.iteraxis == Some(MsIterationAxis::Scan) && single_metadata_pair))
79            || (averaging
80                && (!single_metadata_pair
81                    || self.avgscan
82                    || self.avgfield
83                    || self.avgbaseline
84                    || self.avgantenna
85                    || self.avgspw
86                    || self.scalar))
87        {
88            return Err(TaskFailure::validation(
89                "This PlotMS pointing axis, averaging, iteration, callibrary, overlay or flag-symbol combination is pending (#1032)",
90            ));
91        }
92        if averaging {
93            calibration::averaging::options(task)?;
94            // CASA's metadata-only iterator requests no visibility column. It reports
95            // the requested averaging in the export but retains the selected raw grid.
96            self.avgtime = None;
97            self.avgchannel = None;
98        }
99        if pointing.len() > 1 {
100            self.y_axis2 = self.y_axis;
101            self.secondary_y_data_column = Some(self.data_column);
102            self.additional_ordinates.extend(
103                (2..pointing.len())
104                    .map(|_| (self.y_axis.expect("pointing ordinate"), self.data_column)),
105            );
106        }
107        Ok(())
108    }

155    pub(super) fn resolve_visibility_storage(
156        &mut self,
157        ms: &MeasurementSet,
158        columns: &mut [MsDataColumn; 3],
159        implicit_derived_model: bool,
160        phasecenter: bool,
161    ) -> Result<bool, TaskFailure> {
162        let axes = [self.x_axis, self.y_axis, self.y_axis2];
163        let (adjusted, use_float) = plotms_columns::resolve_storage_columns(
164            ms,
165            *columns,
166            axes,
167            self.plotms_averaging_active(),
168            self.supports_effective_float_averaging(),
169        )?;
170        if let Some((storage, axis)) = self.derived_average_storage(*columns, adjusted) {
171            plotms_columns::check_averaged_derived_storage(
172                ms,
173                storage,
174                axis,
175                implicit_derived_model,
176            )?;
177        }
178        if phasecenter
179            && use_float
180            && axes.into_iter().zip(adjusted).any(|(axis, column)| {
181                axis.is_some_and(MsAxis::is_visibility_math)
182                    && matches!(column, MsDataColumn::Data | MsDataColumn::Float)
183            })
184        {
185            return Err(TaskFailure::validation(
186                "PlotMS phasecenter with MeasurementSet FLOAT_DATA is pending (#1032)",
187            )
188            .at("phasecenter"));
189        }
190        if adjusted != *columns {
191            let [x, y, secondary_y] = adjusted;
192            self.data_column = y;
193            self.x_data_column = (x != y).then_some(x);
194            if self.secondary_y_data_column.is_some() {
195                self.secondary_y_data_column = Some(secondary_y);
196            }
197            *columns = adjusted;
198        }
199        Ok(use_float)
200    }

232    pub(super) fn validate_plotms_columns(
233        &self,
234        task: &TaskInvocation<'_>,
235        has_callibrary: bool,
236        calibrated_ordinates: bool,
237        has_atmosphere: bool,
238        iterated_averaging: bool,
239    ) -> Result<(), TaskFailure> {
240        let columns = self.plotms_axis_columns();
241        let row_identifier_axes = [self.x_axis, self.y_axis]
242            .into_iter()
243            .flatten()
244            .any(MsAxis::is_row_identifier);
245        let uncalibrated_columns = !has_callibrary && (columns.is_some() || row_identifier_axes);
246        if uncalibrated_columns && let Some(columns) = columns {
247            plotms_columns::check_averaging(
248                self.plotms_averaging_active(),
249                self.plotms_extended_averaging()
250                    && self.secondary_y_data_column.is_some()
251                    && !iterated_averaging
252                    && !self.plotms_independent_average(
253                        !has_atmosphere && (self.avgspw || !self.additional_ordinates.is_empty()),
254                    ),
255                self.supports_channel_derived(columns) || self.supports_extended_derived(columns),
256                columns,
257            )?;
258        }
259        if uncalibrated_columns || calibrated_ordinates {
260            let symbols = task.value("customflaggedsymbol");
261            let ordinate_count =
262                1 + usize::from(self.y_axis2.is_some()) + self.additional_ordinates.len();
263            if (0..ordinate_count).any(|index| calibration::figure::shows_flagged(symbols, index))
264                && (calibrated_ordinates
265                    || !columns.is_some_and(|columns| self.supports_visibility_symbols(columns))
266                        && !self.supports_metadata_flag_symbols())
267            {
268                return Err(TaskFailure::validation(
269                    "displaying flagged points for this MeasurementSet column plot is pending #1032",
270                )
271                .at("customflaggedsymbol"));
272            }
273        }
274        Ok(())
275    }

277    pub(super) fn supports_visibility_ordinate_averaging(&self, ordinary_only: bool) -> bool {
278        let spans = self.avgscan || self.avgfield;
279        let time_average = self.avgtime.is_some()
280            && (!ordinary_only || !spans)
281            && !self.avgbaseline
282            && !self.avgantenna;
283        let other_reduction = !ordinary_only
284            && ((self.avgantenna && (!spans || self.avgtime.is_some()))
285                || ((self.avgbaseline || self.avgspw) && !spans));
286        let supported_reduction = (time_average || other_reduction)
287            && self.freqframe.is_none()
288            && self.plotms_independent_average(!ordinary_only);
289        (!self.plotms_extended_averaging() || supported_reduction)
290            && !self.has_mixed_ordinate_dimensions()
291            && self.plotms_axis_columns().is_some_and(|columns| {
292                columns
293                    .into_iter()
294                    .chain(self.additional_ordinates.iter().map(|&(_, column)| column))
295                    .all(|column| {
296                        matches!(
297                            column,
298                            MsDataColumn::Data | MsDataColumn::Corrected | MsDataColumn::Model
299                        )
300                    })
301            })
302    }

304    pub(super) fn supports_channel_derived(&self, [x, y, _]: [MsDataColumn; 3]) -> bool {
305        let supported_pair = |derived: MsDataColumn, stored| {
306            derived.is_derived() && matches!(stored, MsDataColumn::Data | MsDataColumn::Corrected)
307        };
308        let supported_abscissa = self.x_axis.is_some_and(|axis| {
309            if axis.is_visibility_math() {
310                supported_pair(x, y) || supported_pair(y, x)
311            } else {
312                matches!(axis, MsAxis::Channel | MsAxis::Time) && y.is_derived()
313            }
314        });
315        self.avgchannel.is_some_and(|width| width > 1)
316            && !self.plotms_extended_averaging()
317            && supported_abscissa
318            && self.y_axis.is_some_and(MsAxis::is_visibility_math)
319            && self.y_axis2.is_none()
320            && self.iteraxis.is_none()
321    }

323    pub(super) fn supports_extended_derived(&self, [x, y, _]: [MsDataColumn; 3]) -> bool {
324        let channel_average = self.avgchannel.is_some_and(|width| width > 1);
325        let supported_average = if self.avgantenna || self.avgspw {
326            self.avgtime.is_some()
327                && !self.avgbaseline
328                && !(self.avgantenna && self.avgspw)
329                && !self.avgscan
330                && !self.avgfield
331                && self.freqframe.is_none()
332        } else if self.avgbaseline {
333            self.scalar
334        } else {
335            self.avgtime.is_some()
336        };
337        let visibility_column = |column: MsDataColumn| {
338            column.is_derived()
339                || matches!(
340                    column,
341                    MsDataColumn::Data | MsDataColumn::Corrected | MsDataColumn::Model
342                )
343        };
344        let supported_abscissa = self.x_axis.is_some_and(|axis| {
345            if axis == MsAxis::Time || ((channel_average || self.scalar) && axis == MsAxis::Channel)
346            {
347                y.is_derived()
348            } else {
349                axis.is_visibility_math()
350                    && visibility_column(x)
351                    && visibility_column(y)
352                    && (x.is_derived() || y.is_derived())
353            }
354        });
355        supported_abscissa
356            && self.y_axis.is_some_and(MsAxis::is_visibility_math)
357            && self.y_axis2.is_none()
358            && self.iteraxis.is_none()
359            && supported_average
360    }

362    pub(super) fn derived_average_storage(
363        &self,
364        requested: [MsDataColumn; 3],
365        adjusted: [MsDataColumn; 3],
366    ) -> Option<([MsDataColumn; 3], &'static str)> {
367        if self.supports_channel_derived(requested) {
368            Some((adjusted, "channel"))
369        } else if self.supports_extended_derived(requested) {
370            let axis = if self.scalar && self.avgbaseline {
371                "scalar baseline"
372            } else if self.scalar {
373                "scalar time"
374            } else if self.avgchannel.is_some_and(|width| width > 1) {
375                "time- and channel"
376            } else {
377                "time"
378            };
379            Some((adjusted, axis))
380        } else {
381            None
382        }
383    }

385    fn supports_visibility_symbols(&self, columns: [MsDataColumn; 3]) -> bool {
386        let raw = !self.plotms_averaging_active();
387        let reductions = [self.avgtime.is_some(), self.avgantenna, self.avgspw];
388        let unbinned_reduction = reductions.into_iter().filter(|&enabled| enabled).count() == 1
389            && !(self.avgscan || self.avgfield || self.avgbaseline)
390            && self.avgchannel.is_none_or(|width| width <= 1)
391            && self.freqframe.is_none();
392        self.x_axis
393            .is_some_and(|axis| axis.is_visibility_math() || (raw && axis == MsAxis::Channel))
394            && self.y_axis.is_some_and(MsAxis::is_visibility_math)
395            && self.y_axis2.is_none()
396            && (!self.plotms_extended_averaging() || unbinned_reduction)
397            && (!self.scalar || self.avgchannel.is_none_or(|width| width <= 1))
398            && self.iteraxis.is_none()
399            && (self.color_by == MsColorAxis::None
400                || (raw && matches!(self.color_by, MsColorAxis::Scan | MsColorAxis::Correlation)))
401            && columns[..2]
402                .iter()
403                .all(|column| matches!(column, MsDataColumn::Data | MsDataColumn::Corrected))
404    }

414            && self.iteraxis.is_none()
415            && self.color_by == MsColorAxis::None

432    pub(super) fn plotms_phasecenter(
433        &mut self,
434        value: &str,
435        without_calibration_or_overlays: bool,
436    ) -> Result<MDirection, TaskFailure> {
437        let complex_visibility = |axis: MsAxis, column: MsDataColumn| {
438            axis.is_visibility_math()
439                && matches!(
440                    column,
441                    MsDataColumn::Data | MsDataColumn::Corrected | MsDataColumn::Model
442                )
443        };
444        let x = self.x_axis.is_some_and(|axis| {
445            axis.uses_uvw_metres()
446                || complex_visibility(axis, self.x_data_column.unwrap_or(self.data_column))
447        });
448        let channels = self.avgchannel.is_some_and(|width| width > 1);
449        let time = self.avgtime.is_some();
450        if channels && time {
451            return Err(TaskFailure::validation(
452                "PlotMS phasecenter with simultaneous time and channel averaging is refused by CASA 6.7.6.14",
453            )
454            .at("phasecenter"));
455        }
456        if !x
457            || !self
458                .y_axis
459                .is_some_and(|axis| complex_visibility(axis, self.data_column))
460            || !without_calibration_or_overlays
461            || self.y_axis2.is_some()
462            || !self.additional_ordinates.is_empty()
463            || self.avgscan
464            || self.avgfield
465            || self.avgspw
466            || self.avgbaseline
467            || self.avgantenna
468            || (self.scalar && time)
469            || !matches!(self.iteraxis, None | Some(MsIterationAxis::Scan))
470            || ((channels || time) && (self.iteraxis.is_some() || self.freqframe.is_some()))
471        {
472            return Err(TaskFailure::validation(
473                "This combination of PlotMS phasecenter with axes, averaging, iteration, callibrary or overlays is pending (#1032)",
474            )
475            .at("phasecenter"));
476        }
477        // PlotMS ignores scalar for both raw and channel-only requests.
478        self.scalar = false;
479        plotms_phasecenter::parse(value)
480    }

482    pub(super) fn plotms_independent_average(&self, allow_extended_list: bool) -> bool {
483        let transforms_frequency =
484            self.x_axis == Some(MsAxis::Frequency) && self.freqframe.is_some();
485        let crosses_boundary = self.avgscan || self.avgfield;
486        let time_average = self.avgtime.is_some()
487            && !self.avgbaseline
488            && !self.avgantenna
489            && !(transforms_frequency && crosses_boundary);
490        let baseline_average = self.avgbaseline
491            && !self.avgantenna
492            && (!crosses_boundary || self.scalar)
493            && !transforms_frequency;
494        let list_reduction = allow_extended_list
495            && !self.avgbaseline
496            && (!crosses_boundary || self.avgtime.is_some())
497            && self.freqframe.is_none();
498        let antenna_average = list_reduction && self.avgantenna;
499        let spectral_window_average = list_reduction && self.avgspw && !self.avgantenna;
500        let channel_average = self.avgchannel.is_some_and(|width| width > 1);
501        let ordinary_time = time_average
502            && !self.avgspw
503            && !transforms_frequency
504            && !crosses_boundary
505            && !channel_average;
506        let spw_time_u = spectral_window_average
507            && self.avgtime.is_some()
508            && !crosses_boundary
509            && !channel_average
510            && !self.scalar
511            && self.iteraxis.is_none();
512        let scalar_time = time_average && !transforms_frequency;
513        (spectral_window_average
514            || (!self.avgspw
515                && (time_average || baseline_average || antenna_average)
516                && (!self.scalar || scalar_time || baseline_average || antenna_average)))
517            && !self.has_mixed_ordinate_dimensions()
518            && self.x_axis.is_some_and(|axis| {
519                axis.is_visibility_math()
520                    || matches!(axis, MsAxis::Time | MsAxis::Channel | MsAxis::Frequency)
521                    || (ordinary_time && !self.scalar && axis.uses_uvw_metres())
522                    || (spw_time_u && axis == MsAxis::U)
523            })
524    }

550    fn validate_geometry_controls(&self, task: &TaskInvocation<'_>) -> Result<(), TaskFailure> {
551        if self.y_axis.is_some_and(MsAxis::is_ephemeris) {
552            let symbols = raw_symbol_visibility(task);
553            if symbols.flagged || !symbols.unflagged {
554                return Err(TaskFailure::validation(
555                    "PlotMS flag-symbol filtering with ephemeris axes is pending (#1032)",
556                ));
557            }
558        }
559        let antenna_horizontal = [self.x_axis, self.y_axis, self.y_axis2]
560            .into_iter()
561            .flatten()
562            .chain(self.additional_ordinates.iter().map(|(axis, _)| *axis))
563            .any(MsAxis::is_antenna_horizontal);
564        if antenna_horizontal {
565            let symbols = raw_symbol_visibility(task);
566            if self.x_axis != Some(MsAxis::Channel)
567                || !self.y_axis.is_some_and(MsAxis::is_antenna_horizontal)
568                || self.y_axis2.is_some()
569                || self.plotms_averaging_active()
570                || self.iteraxis.is_some()
571                || symbols.flagged
572                || !symbols.unflagged
573            {
574                return Err(TaskFailure::validation(
575                    "This PlotMS antenna horizontal axis combination is pending (#1032)",
576                ));
577            }
578            return Ok(());
579        }
580        let Some((x, y)) = self
581            .x_axis
582            .zip(self.y_axis)
583            .filter(|_| self.y_axis2.is_none())
584        else {
585            return Ok(());
586        };
587        let array_horizontal = plotms_row_axes::array_horizontal_pair(x, y);
588        let static_coordinates = plotms_row_axes::metadata_coordinate_pair(x, y).is_some();
589        if !array_horizontal && !static_coordinates {
590            return Ok(());
591        }
592        let symbols = raw_symbol_visibility(task);
593        if (symbols.flagged || !symbols.unflagged)
594            && (!self.supports_metadata_flag_symbols() || !task.texts("callib")?.is_empty())
595        {
596            let label = if array_horizontal {
597                "array horizontal axes"
598            } else {
599                "metadata coordinate axes"
600            };
601            return Err(TaskFailure::validation(format!(
602                "PlotMS flag-symbol filtering with {label} is pending (#1032)"
603            )));
604        }
605        Ok(())
606    }
607}

609pub(super) fn raw_symbol_visibility(task: &TaskInvocation<'_>) -> RawSymbolVisibility {
610    let is_custom = |name| calibration::figure::shows_flagged(task.value(name), 0);
611    let has_symbol = |name| {
612        let shape = match task.value(name) {
613            Some(ParameterValue::Array(values)) => values.first(),
614            other => other,
615        };
616        !matches!(shape, Some(ParameterValue::String(value)) if value == "nosymbol")
617    };
618    RawSymbolVisibility {
619        unflagged: !is_custom("customsymbol") || has_symbol("symbolshape"),
620        flagged: is_custom("customflaggedsymbol") && has_symbol("flaggedsymbolshape"),
621    }
622}

624fn right_ordinates(task: &TaskInvocation<'_>) -> Result<Vec<bool>, TaskFailure> {
625    let supplied = task.value("yaxislocation");
626    let names = calibration::axis_names(supplied).map_err(TaskFailure::validation)?;
627    let sides = names
628        .into_iter()
629        .map(|name| match name {
630            "left" => Ok(false),
631            "right" => Ok(true),
632            _ => Err(
633                TaskFailure::validation(format!("invalid yaxislocation value '{name}'"))
634                    .at("yaxislocation"),
635            ),
636        })
637        .collect::<Result<Vec<_>, _>>()?;
638    let is_list = |value: Option<&ParameterValue>| matches!(value, Some(ParameterValue::Array(_)));
639    if is_list(supplied) == is_list(task.value("yaxis")) {
640        Ok(sides)
641    } else {
642        Ok(Vec::new())
643    }
644}
645
646fn requested_ordinates(
647    task: &TaskInvocation<'_>,
648    x_axis: MsAxis,
649) -> Result<Vec<MsAxis>, TaskFailure> {
650    let y_value = task.value("yaxis");
651    let mut ordinates = calibration::axis_names(y_value)
652        .map_err(TaskFailure::validation)?
653        .into_iter()
654        .map(|axis| MsAxis::parse(&axis.to_ascii_lowercase()))
655        .collect::<Result<Vec<_>, _>>()
656        .map_err(TaskFailure::validation)?;
657    if matches!(y_value, Some(ParameterValue::Array(_)))
658        && !ordinates.is_empty()
659        && (x_axis.is_pointing() || ordinates.iter().any(|axis| axis.is_pointing()))
660    {
661        return Err(TaskFailure::validation(
662            "currently not supported: multiple y-axes involving ant-ra or ant-dec",
663        )
664        .at("yaxis"));
665    }
666    if ordinates.is_empty() {
667        ordinates.push(MsAxis::Amplitude);
668    }
669    Ok(ordinates)
670}

672/// Translate the CASA PlotMS spellings to the existing MS exploration request.
673fn plotms_symbol_size(task: &TaskInvocation<'_>) -> Result<Option<u32>, TaskFailure> {
674    // PlotMS applies size lists to the individual ordinate artists.
675    if matches!(task.value("symbolsize"), Some(ParameterValue::Array(_))) {
676        return Ok(None);
677    }
678    Ok(task
679        .optional_number("symbolsize")?
680        .map(|value| value as u32))
681}

683pub(super) fn options_from_plotms(
684    task: &TaskInvocation<'_>,
685    visibility_document: bool,
686) -> Result<(CliOptions, bool), TaskFailure> {
687    let text = |name| {
688        task.text(name)
689            .map(|text| text.filter(|value| !value.is_empty()))
690    };
691    let plot_output = text("plotfile")?.map(PathBuf::from);
692    let export_format = text("expformat")?.or_else(|| {
693        plot_output
694            .as_deref()
695            .and_then(std::path::Path::extension)
696            .and_then(std::ffi::OsStr::to_str)
697            .map(str::to_owned)
698    });
699    let x_axis = text("xaxis")?
700        .map_or(Ok(MsAxis::Time), |axis| {
701            MsAxis::parse(&axis.to_ascii_lowercase())
702        })
703        .map_err(TaskFailure::validation)?;
704    let ordinates = requested_ordinates(task, x_axis)?;
705    let y_axis = ordinates[0];
706    let y_axis2 = ordinates.get(1).copied();
707    let averaging = task.flag("averagedata")?;
708    let average = |name| Ok::<_, TaskFailure>(averaging && task.flag(name)?);
709    let average_text = |name| if averaging { text(name) } else { Ok(None) };
710    let data_column = |column: Option<String>| {
711        column
712            .map(|column| MsDataColumn::parse(&column))
713            .transpose()
714            .map_err(TaskFailure::validation)
715    };
716    let x_column = data_column(text("xdatacolumn")?)?.unwrap_or(MsDataColumn::Data);
717    let y_columns = plotms_columns::ordinate_columns(task, &ordinates)?;
718    let y_column = y_columns[0];
719    let secondary_y = y_columns.get(1).copied().unwrap_or(MsDataColumn::Data);
720    let additional_ordinates = ordinates.into_iter().zip(y_columns).skip(2).collect();
721    let weight_axis_columns = plotms_columns::weight_axis_columns(
722        [Some(x_axis), Some(y_axis), y_axis2],
723        [x_column, y_column, secondary_y],
724    );
725    let secondary_uses_data = y_axis2.is_some_and(MsAxis::is_visibility_math);
726    let data_column = if y_axis.is_visibility_math() {
727        y_column
728    } else if x_axis.is_visibility_math() {
729        x_column
730    } else if secondary_uses_data {
731        secondary_y
732    } else {
733        MsDataColumn::Data
734    };
735    let x_data_column =
736        (x_axis.is_visibility_math() && x_column != data_column).then_some(x_column);
737    let secondary_y = if secondary_uses_data {
738        secondary_y
739    } else {
740        data_column
741    };
742    let secondary_y_data_column =
743        (y_axis2 == Some(y_axis) || secondary_y != data_column).then_some(secondary_y);
744    let avgtime = average_text("avgtime")?
745        .map(|value| number::<f64>(&value))
746        .transpose()
747        .map_err(TaskFailure::validation)?;
748    let document_axes = x_axis.is_visibility_math()
749        || y_axis.is_visibility_math()
750        || (avgtime.is_some_and(|interval| interval != 0.0)
751            && [x_axis, y_axis].contains(&MsAxis::WeightSpectrum))
752        || plotms_row_axes::array_horizontal_pair(x_axis, y_axis)
753        || plotms_row_axes::metadata_coordinate_pair(x_axis, y_axis).is_some()
754        || y_axis.is_antenna_horizontal()
755        || x_axis.is_ephemeris()
756        || y_axis.is_ephemeris()
757        // PlotMS falls back to scalar SIGMA for undefined spectral cells;
758        // the raw explorer deliberately requires the requested stored cell.
759        || x_axis == MsAxis::SigmaSpectrum
760        || y_axis == MsAxis::SigmaSpectrum
761        || y_axis2.is_some()
762        || x_axis.is_row_identifier()
763        || y_axis.is_row_identifier()
764        || x_axis.is_pointing()
765        || y_axis.is_pointing();
766    let visibility_document = visibility_document || document_axes;
767    let plot_format = if visibility_document {
768        MsExportFormat::Txt
769    } else {
770        export_format
771            .as_deref()
772            .map(MsExportFormat::parse)
773            .transpose()
774            .map_err(|_| TaskFailure::validation("MeasurementSet figure export is pending #1032"))?
775            .unwrap_or(MsExportFormat::Txt)
776    };
777    let color_by = text("coloraxis")?
778        .map_or(Ok(MsColorAxis::None), |axis| {
779            MsColorAxis::parse(&axis.to_ascii_lowercase())
780        })
781        .map_err(TaskFailure::validation)?;
782    let correlation = text("correlation")?.or(text("polarization")?);
783    let transform = task.flag("transform")?;
784    let options = CliOptions {
785        ms_path: task.path("vis")?,
786        summary_format: MeasurementSetSummaryOutputFormat::Text,
787        summary_output: None,
788        overwrite: task.flag("overwrite")?,
789        selection: MsSelection {
790            selectdata: task.flag("selectdata")?,
791            field: text("field")?,
792            spw: text("spw")?,
793            timerange: text("timerange")?,
794            uvrange: text("uvrange")?,
795            antenna: text("antenna")?,
796            scan: text("scan")?,
797            correlation,
798            observation: text("observation")?,
799            array: text("array")?,
800            intent: text("intent")?,
801            feed: text("feed")?,
802            msselect: text("msselect")?,
803            ..MsSelection::default()
804        },
805        page_spec: None,
806        preset: None,
807        x_axis: Some(x_axis),
808        y_axis: Some(y_axis),
809        y_axis2,
810        data_column,
811        x_data_column,
812        secondary_y_data_column,
813        additional_ordinates,
814        right_ordinates: right_ordinates(task)?,
815        weight_axis_columns,
816        color_by,
817        avgchannel: average_text("avgchannel")?
818            .map(|value| number::<usize>(&value))
819            .transpose()
820            .map_err(TaskFailure::validation)?
821            .filter(|&width| !document_axes || width != 0),
822        avgtime: avgtime.filter(|&interval| !document_axes || interval != 0.0),
823        avgscan: average("avgscan")?,
824        avgfield: average("avgfield")?,
825        avgbaseline: average("avgbaseline")?,
826        avgantenna: average("avgantenna")?,
827        avgspw: average("avgspw")?,
828        scalar: average("scalar")?,
829        freqframe: if transform { text("freqframe")? } else { None },
830        restfreq: if transform { text("restfreq")? } else { None },
831        veldef: if transform {
832            task.required_text("veldef")?
833        } else {
834            "RADIO".into()
835        },
836        iteraxis: text("iteraxis")?
837            .map(|value| MsIterationAxis::parse(&value.to_ascii_lowercase()))
838            .transpose()
839            .map_err(TaskFailure::validation)?,
840        gridrows: count(task, "gridrows")?.max(1),
841        gridcols: count(task, "gridcols")?.max(1),
842        xselfscale: task.flag("xselfscale")?,
843        yselfscale: task.flag("yselfscale")?,
844        xsharedaxis: task.flag("xsharedaxis")?,
845        ysharedaxis: task.flag("ysharedaxis")?,
846        title: text("title")?,
847        xlabel: text("xlabel")?,
848        ylabel: text("ylabel")?,
849        symbol_size: plotms_symbol_size(task)?,
850        showlegend: task.flag("showlegend")?,
851        legendposition: text("legendposition")?
852            .map_or(Ok(MsLegendPosition::UpperRight), |value| {
853                legend_position(&value)
854            })
855            .map_err(TaskFailure::validation)?,
856        showmajorgrid: task.flag("showmajorgrid")?,
857        showminorgrid: task.flag("showminorgrid")?,
858        headeritems: text("headeritems")?,
859        max_points: usize::MAX,
860        flag_action: None,
861        flag_xmin: None,
862        flag_xmax: None,
863        flag_ymin: None,
864        flag_ymax: None,
865        flag_plotindex: None,
866        flag_panel: None,
867        flag_extcorr: false,
868        flag_extchannel: false,
869        flag_selected: false,
870        flag_apply: false,
871        flag_output: None,
872        plot_output,
873        plot_format,
874    };
875    options.validate_plotms_weight_averaging()?;
876    options.validate_geometry_controls(task)?;
877    Ok((options, visibility_document))
878}
