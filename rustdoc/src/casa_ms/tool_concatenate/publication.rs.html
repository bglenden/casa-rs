79fn concatenate(
80    target: &Path,
81    donor: &Path,
82    request: &TestConcatenation<'_>,
83    mode: Mode<'_>,
84    retained: &mut BTreeMap<PathBuf, TableWriteLock>,
85) -> MsResult<()> {
86    let physical = if let Mode::Physical(mode) = mode {
87        Some(mode)
88    } else {
89        None
90    };
91    let virtual_mode = if let Mode::Virtual(mode) = mode {
92        Some(mode)
93    } else {
94        None
95    };
96    let mut scan_file = virtual_mode
97        .and_then(|mode| mode.scan_file)
98        .map(ScanFile::inspect)
99        .transpose()?;
100    let target = target.canonicalize().map_err(io)?;
101    let donor = donor.canonicalize().map_err(io)?;
102    if target.starts_with(&donor) || donor.starts_with(&target) {
103        return Err(io("concatenation requires separate input MeasurementSets"));
104    }
105    for root in [&target, &donor] {
106        check_write_access(&root.join("table.dat")).map_err(io)?;
107    }
108    let mut held = WriteGuards::new(retained);
109    held.acquire(BTreeSet::from([target.clone(), donor.clone()]))?;
110    let (attached, incoming) = rayon::join(
111        || InputTables::capture(&target),
112        || InputTables::capture(&donor),
113    );
114    let (attached, target_main) = attached?;
115    let (incoming, donor_main) = incoming?;
116    let mut leaf_writers = attached.write_paths(true);
117    leaf_writers.extend(incoming.write_paths(false));
118    let pointing = physical.is_some_and(|mode| mode.handling < 2);
119    if pointing {
120        leaf_writers.extend(
121            attached
122                .subtable_path(SubtableId::Pointing)
123                .map(Path::to_path_buf),
124        );
125    }
126    held.acquire(leaf_writers)?;
127    validate_inputs(&attached, &incoming)?;
128
129    let (target_ms, donor_ms) = rayon::join(
130        || attached.open(target_main, false),
131        || incoming.open(donor_main, true),
132    );
133    let target_ms = target_ms?;
134    let donor_ms = donor_ms?;
135    let _donor_ephemerides = incoming.hold_ephemerides()?;
136    let target_calibration = open_calibration(&attached.calibration, false)?;
137    let donor_calibration = open_calibration(&incoming.calibration, true)?;
138    super::layout::check(&target_ms, &donor_ms)?;
139    let named = physical
140        .and_then(|mode| mode.output)
141        .map(|output| NamedOutput::stage(output, &target_ms, &donor_ms))
142        .transpose()?;
143    let plan = MetadataPlan::reconcile(
144        &target_ms,
145        &donor_ms,
146        [&target_calibration, &donor_calibration],
147        request.frequency_hz,
148        request.direction_rad,
149        request.respect_name,
150        virtual_mode.is_some(),
151    )?;
152
153    let virtual_pointing = virtual_mode.is_some()
154        && plan
155            .row_map
156            .antennas
157            .iter()
158            .copied()
159            .ne(0..plan.row_map.antennas.len());
160    if virtual_pointing {
161        held.acquire(
162            incoming
163                .subtable_path(SubtableId::Pointing)
164                .map(Path::to_path_buf)
165                .into_iter()
166                .collect(),
167        )?;
168    }
169    let donor_pointing = incoming
170        .subtable_path(SubtableId::Pointing)
171        .filter(|_| pointing || virtual_pointing)
172        .map(|path| open_pointing(path, virtual_pointing))
173        .transpose()?;
174    let target_pointing = attached
175        .subtable_path(SubtableId::Pointing)
176        .filter(|_| pointing || virtual_pointing)
177        .map(|path| open_pointing(path, pointing))
178        .transpose()?;
179    let attached_history = attached
180        .subtable_path(SubtableId::History)
181        .filter(|_| !plan.row_map.attached_observations.is_empty())
182        .map(|path| -> MsResult<_> {
183            held.acquire(BTreeSet::from([path.to_path_buf()]))?;
184            Ok(Table::open(TableOptions::new(path))?)
185        })
186        .transpose()?;
187    let mut requests = metadata_edits(&plan, &target_ms, &incoming, &target, &held.guards)?;
188    if let Some(history) = &attached_history {
189        prepare_attached_history(
190            history,
191            &plan.row_map.attached_observations,
192            &incoming,
193            &target,
194            &mut requests,
195        )?;
196    }
197    if let Some((left, right)) = target_pointing.as_ref().zip(donor_pointing.as_ref()) {
198        let (table, opposite, root) = if virtual_pointing {
199            (right, &attached, &donor)
200        } else {
201            (left, &incoming, &target)
202        };
203        let path = output_path(table, opposite, root)?;
204        let edit = if virtual_pointing {
205            prepare_virtual_pointing(left, right, &plan, path.clone(), &held.guards)?
206        } else {
207            prepare_pointing(left, right, &plan, path.clone(), &held.guards)?
208        };
209        if let Some(edit) = edit {
210            insert_edit(&mut requests, path, edit)?;
211        }
212    }
213    let history = required_table(&donor_ms, SubtableId::History)?;
214    let history_path = output_path(history, &attached, &donor)?;
215    let logs = prepare_history(history, request, mode)?;
216    insert_edit(
217        &mut requests,
218        history_path,
219        EditRequest::Append(history, Cow::Owned(logs)),
220    )?;
221    if let Some(mode) = physical.filter(|mode| mode.handling % 2 == 0 && named.is_none())
222        && let Some(rows) = main_rows::prepare(
223            &target_ms,
224            &donor_ms,
225            target_ms.main_table(),
226            &plan,
227            mode.weight_scale,
228        )?
229    {
230        insert_edit(
231            &mut requests,
232            target.clone(),
233            EditRequest::Main(target_ms.main_table(), Box::new(rows)),
234        )?;
235    }
236    let scan_text = if let Some(mode) = virtual_mode {
237        prepare_virtual_main(
238            &target_ms,
239            &donor_ms,
240            &plan,
241            mode,
242            scan_file.as_mut(),
243            &mut requests,
244        )?
245    } else {
246        None
247    };
248    let named_held = named
249        .as_ref()
250        .map(|_| held.guards.keys().cloned().collect::<BTreeSet<_>>());
251    // Canonical paths are distinct. Iterate the map to borrow each guard
252    // once and hold it across all fallible edits and synchronization.
253    let work = held
254        .guards
255        .iter_mut()
256        .filter_map(|(path, guard)| requests.remove(path).map(|request| (guard, request)))
257        .collect::<Vec<_>>();
258    if !requests.is_empty() {
259        return Err(io("concatenation edit is missing its captured write lock"));
260    }
261    let prepare_tables = || {
262        work.into_par_iter()
263            .map(prepare_table_edit)
264            .collect::<MsResult<Vec<_>>>()
265    };
266    let edits = if let Some((file, text)) = scan_file.as_mut().zip(scan_text.as_deref()) {
267        rayon::join(prepare_tables, || file.stage(text)).0?
268    } else {
269        prepare_tables()?
270    };
271    validate_inputs(&attached, &incoming)?;
272    publish_batch_then(edits, || {
273        if let Some(named) = named {
274            let mode = physical.expect("named physical request");
275            named.finish(
276                &target_ms,
277                &donor_ms,
278                &plan,
279                mode,
280                &named_held.expect("named output lock paths"),
281            )?;
282        }
283        if let Some(scan_file) = scan_file {
284            scan_file.publish()?;
285        }
286        Ok(())
287    })
288}

290fn validate_inputs(attached: &InputTables, incoming: &InputTables) -> MsResult<()> {
291    let (attached, incoming) = rayon::join(|| attached.validate(), || incoming.validate());
292    attached.and(incoming)

295fn prepare_attached_history<'a>(
296    history: &'a Table,
297    observations: &[usize],
298    incoming: &InputTables,
299    target: &Path,
300    requests: &mut BTreeMap<PathBuf, EditRequest<'a>>,
301) -> MsResult<()> {
302    let mut columns = BTreeMap::new();
303    virtual_rows::remap_ids(history, "OBSERVATION_ID", observations, &mut columns)?;
304    if columns.is_empty() {
305        return Ok(());
306    }
307    let path = output_path(history, incoming, target)?;
308    let values = MainReplacement {
309        columns,
310        channel_selection: None,
311    };
312    insert_edit(
313        requests,
314        path,
315        EditRequest::Columns(history, Box::new(values)),
316    )
317}

319fn prepare_table_edit<'a>(
320    (guard, edit): (&'a mut TableWriteLock, EditRequest<'_>),
321) -> MsResult<TableEdit<'a>> {
322    match edit {
323        EditRequest::Append(table, rows) => Ok(TableEdit::Append(
324            table.prepare_record_append(&rows, guard)?,
325        )),
326        EditRequest::Replace(staged) => Ok(TableEdit::Replace(*staged, guard)),
327        EditRequest::Columns(table, values) => {
328            let keyword = values
329                .channel_selection
330                .as_ref()
331                .map(|value| ("MODEL_DATA", "CHANNEL_SELECTION", value));
332            Ok(TableEdit::Append(table.prepare_column_replacement(
333                &values.columns,
334                keyword,
335                guard,
336            )?))
337        }
338        EditRequest::Main(table, rows) => Ok(TableEdit::Append(
339            table.prepare_record_append_with_options(&rows.records, rows.options(), guard)?,
340        )),
341    }
342}

344fn metadata_edits<'a>(
345    plan: &'a MetadataPlan<'a>,
346    target_ms: &'a crate::MeasurementSet,
347    incoming: &InputTables,
348    target: &Path,
349    held: &BTreeMap<PathBuf, TableWriteLock>,
350) -> MsResult<BTreeMap<PathBuf, EditRequest<'a>>> {
351    let mut requests = BTreeMap::new();
352    for change in &plan.changes {
353        let table = required_table(target_ms, change.table)?;
354        let path = output_path(table, incoming, target)?;
355        let field = (change.table == SubtableId::Field).then_some(plan);
356        let request = metadata_edit(table, &path, &change.rows, field, held)?;
357        insert_edit(&mut requests, path, request)?;
358    }
359    for (table, rows) in &plan.calibration {
360        let path = output_path(table, incoming, target)?;
361        let request = metadata_edit(table, &path, rows, None, held)?;
362        insert_edit(&mut requests, path, request)?;
363    }
364    Ok(requests)
365}

367fn metadata_edit<'a>(
368    table: &'a Table,
369    path: &Path,
370    rows: &'a TableRowsChange,
371    field: Option<&MetadataPlan<'_>>,
372    held: &BTreeMap<PathBuf, TableWriteLock>,
373) -> MsResult<EditRequest<'a>> {
374    let replacement = !rows.updated.is_empty()
375        || !rows.removed.is_empty()
376        || field.is_some_and(|plan| plan.add_ephemeris_column || !plan.ephemerides.is_empty());
377    if replacement {
378        Ok(EditRequest::Replace(Box::new(StagedTable::prepare_change(
379            path.to_owned(),
380            rows,
381            field,
382            held,
383        )?)))
384    } else {
385        Ok(EditRequest::Append(
386            table,
387            matching_fields(table, &rows.appended),
388        ))
389    }
390}

400fn prepare_virtual_main<'a>(
401    target: &'a crate::MeasurementSet,
402    donor: &'a crate::MeasurementSet,
403    plan: &MetadataPlan<'_>,
404    mode: &VirtualConcatenation<'_>,
405    auxiliary: Option<&mut ScanFile>,
406    edits: &mut BTreeMap<PathBuf, EditRequest<'a>>,
407) -> MsResult<Option<String>> {
408    let left = target.main_table();
409    let right = donor.main_table();
410    let row_map = &plan.row_map;
411    let read_bounds = auxiliary.map(|file| {
412        move || file.starting_bounds(left, mode.attached_rows, &row_map.attached_observations)
413    });
414    let (columns, scan) =
415        virtual_rows::prepare(target, donor, &plan.row_map, mode.weight_scale, read_bounds)?;
416    if !columns.columns.is_empty() || columns.channel_selection.is_some() {
417        let path = right.path().ok_or_else(|| io("donor MAIN has no path"))?;
418        insert_edit(
419            edits,
420            path.to_owned(),
421            EditRequest::Columns(right, Box::new(columns)),
422        )?;
423    }
424    let columns = virtual_rows::attached_replacements(left, row_map, mode.attached_rows)?;
425    if !columns.columns.is_empty() {
426        let path = left.path().ok_or_else(|| io("attached MAIN has no path"))?;
427        insert_edit(
428            edits,
429            path.to_owned(),
430            EditRequest::Columns(left, Box::new(columns)),
431        )?;
432    }
433    Ok(scan)
434}

436fn prepare_virtual_pointing<'a>(
437    target: &Table,
438    donor: &'a Table,
439    plan: &MetadataPlan<'_>,
440    path: PathBuf,
441    held: &BTreeMap<PathBuf, TableWriteLock>,
442) -> MsResult<Option<EditRequest<'a>>> {
443    if donor.row_count() == 0 {
444        return Ok(None);
445    }
446    if target.row_count() == 0 {
447        let change = TableRowsChange {
448            removed: (0..donor.row_count()).collect(),
449            ..TableRowsChange::default()
450        };
451        return Ok(Some(EditRequest::Replace(Box::new(
452            StagedTable::prepare_change(path, &change, None, held)?,
453        ))));
454    }
455    let values = donor.scalar_columns_owned(&["ANTENNA_ID"])?;
456    let mapped = values["ANTENNA_ID"]
457        .iter()
458        .map(|value| {
459            let Some(ScalarValue::Int32(id)) = value else {
460                return Err(io("POINTING entry has no antenna ID"));
461            };
462            let antenna =
463                super::main_rows::lookup(&plan.row_map.antennas, *id, "POINTING.ANTENNA_ID")?;
464            Ok(Some(Value::Scalar(ScalarValue::Int32(
465                super::spectral::row_id(*antenna)?,
466            ))))
467        })
468        .collect::<MsResult<Vec<_>>>()?;
469    Ok(Some(EditRequest::Columns(
470        donor,
471        Box::new(MainReplacement {
472            columns: BTreeMap::from([("ANTENNA_ID".into(), mapped)]),
473            channel_selection: None,
474        }),
475    )))
476}

478fn prepare_pointing<'a>(
479    target: &'a Table,
480    donor: &Table,
481    plan: &MetadataPlan<'_>,
482    path: PathBuf,
483    held: &BTreeMap<PathBuf, TableWriteLock>,
484) -> MsResult<Option<EditRequest<'a>>> {
485    if target.row_count() == 0 {
486        return Ok(None);
487    }
488    if donor.row_count() == 0 {
489        let change = TableRowsChange {
490            removed: (0..target.row_count()).collect(),
491            ..TableRowsChange::default()
492        };
493        return Ok(Some(EditRequest::Replace(Box::new(
494            StagedTable::prepare_change(path, &change, None, held)?,
495        ))));
496    }
497    let rows = donor
498        .rows()?
499        .iter()
500        .map(|record| {
501            let Some(Value::Scalar(ScalarValue::Int32(antenna))) = record.get("ANTENNA_ID") else {
502                return Err(io("POINTING entry has no antenna ID"));
503            };
504            let id = usize::try_from(*antenna)
505                .ok()
506                .and_then(|id| plan.row_map.antennas.get(id))
507                .ok_or_else(|| io(format!("POINTING antenna ID {antenna} is invalid")))?;
508            let mut row = record.clone();
509            row.upsert(
510                "ANTENNA_ID",
511                Value::Scalar(ScalarValue::Int32(super::spectral::row_id(*id)?)),
512            );
513            Ok(row)
514        })
515        .collect::<MsResult<Vec<_>>>()?;
516    Ok(Some(EditRequest::Append(
517        target,
518        Cow::Owned(matching_fields(target, &rows).into_owned()),
519    )))
520}

539fn prepare_history(
540    table: &Table,
541    request: &TestConcatenation<'_>,
542    operation: Mode<'_>,
543) -> MsResult<Vec<RecordValue>> {
544    if matches!(operation, Mode::Virtual(_)) {
545        let message = format!(
546            "{} virtually concatenated with {}",
547            request.donor_name,
548            request.attached_name.display()
549        );
550        let command = format!(
551            "msfile='{}' freqTol='{}' dirTol='{}' respectname='{}'",
552            request.donor_name,
553            request.frequency_text,
554            request.direction_text,
555            u8::from(request.respect_name),
556        );
557        return history_rows(table, &[message], &command, "ms::virtconcatenate()", "ms");
558    }
559    if let Mode::Physical(mode) = operation {
560        let message = format!(
561            "{} appended to {}",
562            request.donor_name,
563            request.attached_name.display()
564        );
565        let destination = mode
566            .output
567            .map(|output| output.to_string_lossy())
568            .unwrap_or_default();
569        let command = format!(
570            "msfile='{}' freqTol='{}' dirTol='{}' respectname='{}' handling= {} destmsfile='{}'",
571            request.donor_name,
572            request.frequency_text,
573            request.direction_text,
574            u8::from(request.respect_name),
575            mode.handling,
576            destination,
577        );
578        return history_rows(table, &[message], &command, "ms::concatenate()", "ms");
579    }
580    let message = format!(
581        "Subtables from {} appended to those from {}",
582        request.donor_name,
583        request.attached_name.display()
584    );
585    let command = format!(
586        "msfile= {} freqTol='{}' dirTol='{}'' respectname='{}",
587        request.donor_name,
588        request.frequency_text,
589        request.direction_text,
590        u8::from(request.respect_name),
591    );
592    history_rows(table, &[message], &command, "ms::testconcatenate()", "ms")
593}

595fn output_path(table: &Table, opposite: &InputTables, own_root: &Path) -> MsResult<PathBuf> {
596    let path = table.path().ok_or_else(|| io("a subtable has no path"))?;
597    if path == own_root || opposite.contains(path) {
598        return Err(io(format!(
599            "concatenation would change a shared table at {}",
600            path.display()
601        )));
602    }
603    check_write_access(&path.join("table.dat")).map_err(io)?;
604    Ok(path.to_path_buf())
605}

607enum EditRequest<'a> {
608    Append(&'a Table, Cow<'a, [RecordValue]>),
609    Main(&'a Table, Box<MainAppend>),
610    Columns(&'a Table, Box<MainReplacement>),

614fn insert_edit<'a>(
615    edits: &mut BTreeMap<PathBuf, EditRequest<'a>>,
616    path: PathBuf,
617    edit: EditRequest<'a>,
618) -> MsResult<()> {
619    use std::collections::btree_map::Entry;
620    match edits.entry(path) {
621        Entry::Occupied(entry) => Err(io(format!("several edits share {}", entry.key().display()))),
622        Entry::Vacant(entry) => {
623            entry.insert(edit);
624            Ok(())
625        }
626    }
627}

629fn publish_batch_then(
630    mut edits: Vec<TableEdit<'_>>,
631    finish: impl FnOnce() -> MsResult<()>,
632) -> MsResult<()> {
633    let publish = edits
634        .par_iter_mut()
635        .map(TableEdit::publish)
636        .collect::<Vec<_>>();
637    let result: MsResult<()> = publish
638        .into_iter()
639        .collect::<MsResult<()>>()
640        .and_then(|()| {
641            edits
642                .par_iter_mut()
643                .map(TableEdit::synchronize)
644                .collect::<Vec<_>>()
645                .into_iter()
646                .collect()
647        })
648        .and_then(|()| finish());
649    if let Err(error) = result {
650        let failures: Vec<_> = edits
651            .into_iter()
652            .rev()
653            .filter_map(|edit| edit.restore().err())
654            .map(|error| error.to_string())
655            .collect();
656        return if failures.is_empty() {
657            Err(error)
658        } else {
659            Err(io(format!(
660                "{error}; recovery errors: {}",
661                failures.join(", ")
662            )))
663        };
664    }
665    edits.into_par_iter().for_each(TableEdit::finish);
666    Ok(())
667}

673
674impl TableEdit<'_> {
675    fn publish(&mut self) -> MsResult<()> {
676        match self {
677            Self::Append(append) => Ok(append.publish()?),
678            Self::Replace(edit, original) => edit.publish(original),
679        }
680    }
681
682    fn synchronize(&mut self) -> MsResult<()> {
683        match self {
684            Self::Append(append) => Ok(append.synchronize()?),
685            Self::Replace(..) => Ok(()),
686        }
687    }
688
689    fn restore(self) -> MsResult<()> {
690        match self {
691            Self::Append(append) => Ok(append.restore()?),
692            Self::Replace(edit, original) if edit.published => {
693                if let Err(error) = edit.restore(original) {
694                    return Err(io(format!(
695                        "{error}; original subtable retained at {}",
696                        edit.directory.keep().display()
697                    )));
698                }
699                Ok(())
700            }
701            Self::Replace(..) => Ok(()),
702        }
703    }
704
705    fn finish(self) {
706        if let Self::Replace(mut edit, original) = self {
707            *original = edit.staged.take().expect("replacement table lock");
708        }
709    }
710}

712struct StagedTable {
713    live: PathBuf,
714    private: PathBuf,
715    directory: TempDir,
716    _nested_locks: Vec<TableWriteLock>,
717    staged: Option<TableWriteLock>,
718    published: bool,
719}
720
721impl StagedTable {
722    fn prepare_change(
723        live: PathBuf,
724        change: &TableRowsChange,
725        field: Option<&MetadataPlan<'_>>,
726        held: &BTreeMap<PathBuf, TableWriteLock>,
727    ) -> MsResult<Self> {
728        if !matches!(
729            Table::immediate_part_names(&live, &live)?,
730            TablePartNames::Plain(_)
731        ) {
732            return Err(io(format!(
733                "{}. Concatenation requires writable physical subtables",
734                live.display()
735            )));
736        }
737        let nested_locks = lock_table_tree_for_replacement_with_held(&live, held)?;
738        let directory = tempfile::Builder::new()
739            .prefix(".casa-rs-ms-concat-")
740            .tempdir_in(live.parent().expect("physical table parent"))
741            .map_err(io)?;
742        let private = directory.path().join("table");
743        copy_locked_table_tree(&live, &private).map_err(io)?;
744        require_owned_payload(&private)?;
745        let mut staged = Self {
746            live,
747            private,
748            directory,
749            _nested_locks: nested_locks,
750            staged: None,
751            published: false,
752        };
753        let mut lock = TableWriteLock::acquire_to_reread(&staged.private, 1)?;
754        lock.record_write();
755        staged.write(change, field)?;
756        lock.release()?;
757        staged.lock_staged()?;
758        Ok(staged)
759    }
760
761    fn write(&self, change: &TableRowsChange, field: Option<&MetadataPlan<'_>>) -> MsResult<()> {
762        let mut table = Table::open(TableOptions::new(&self.private))?;
763        if let Some(plan) = field {
764            if plan.add_ephemeris_column {
765                table.add_columns_in_place(
766                    vec![ColumnAddition {
767                        schema: ColumnSchema::scalar("EPHEMERIS_ID", PrimitiveType::Int32)
768                            .with_comment("Ephemeris id, pointer to EPHEMERIS table"),
769                        default: Some(Value::Scalar(ScalarValue::Int32(-1))),
770                        keywords: RecordValue::default(),
771                        manager_group: None,
772                        manager_type: None,
773                    }],
774                    None,
775                )?;
776            }
777            for (source, name) in &plan.ephemerides {
778                let source = source.canonicalize().map_err(io)?;
779                let _reader = held_reader(&source)?;
780                copy_locked_table_tree(&source, &self.private.join(name)).map_err(io)?;
781            }
782        }
783        for (row, record) in &change.updated {
784            for field in record.fields() {
785                let value = table.cell_accessor(*row, &field.name)?.value()?;
786                if value != Some(&field.value) {
787                    table
788                        .cell_accessor_mut(*row, &field.name)?
789                        .set(field.value.clone())?;
790                }
791            }
792        }
793        if !change.updated.is_empty() {
794            table.flush()?;
795            table = Table::open(TableOptions::new(&self.private))?;
796        }
797        let rows = matching_fields(&table, &change.appended);
798        table.append_records_and_close(&rows)?;
799        if !change.removed.is_empty() {
800            let mut table = Table::open(TableOptions::new(&self.private))?;
801            let kept: Vec<_> = (0..table.row_count())
802                .filter(|index| change.removed.binary_search(index).is_err())
803                .collect();
804            table.retain_rows_in_place(&kept)?;
805        }
806        Ok(())
807    }
808
809    fn lock_staged(&mut self) -> MsResult<()> {
810        self.staged = Some(TableWriteLock::acquire_to_reread(&self.private, 1)?);
811        Ok(())
812    }
813
814    fn publish(&mut self, original: &TableWriteLock) -> MsResult<()> {
815        if !original.matches_table_path(&self.live) {
816            return Err(io(format!(
817                "{} changed during concatenation",
818                self.live.display()
819            )));
820        }
821        exchange_directories(&self.private, &self.live).map_err(io)?;
822        self.published = true;
823        if !original.matches_table_path(&self.private)
824            || !self
825                .staged
826                .as_ref()
827                .expect("staged table lock")
828                .matches_table_path(&self.live)
829        {
830            return Err(io(format!(
831                "{} changed at publication",
832                self.live.display()
833            )));
834        }
835        self.staged
836            .as_mut()
837            .expect("staged table lock")
838            .follow_directory_rename(&self.live)?;
839        Ok(())
840    }
841
842    fn restore(&self, original: &TableWriteLock) -> MsResult<()> {
843        if !original.matches_table_path(&self.private)
844            || !self
845                .staged
846                .as_ref()
847                .expect("staged table lock")
848                .matches_table_path(&self.live)
849        {
850            return Err(io(format!(
851                "{} changed after publication",
852                self.live.display()
853            )));
854        }
855        exchange_directories(&self.private, &self.live).map_err(io)
856    }
857}
858
859fn matching_fields<'a>(table: &Table, records: &'a [RecordValue]) -> Cow<'a, [RecordValue]> {
860    let schema = table.schema().expect("MeasurementSet subtable schema");
861    if records
862        .iter()
863        .flat_map(RecordValue::fields)
864        .all(|field| schema.contains_column(&field.name))
865    {
866        return Cow::Borrowed(records);
867    }
868    Cow::Owned(
869        records
870            .iter()
871            .map(|row| {
872                RecordValue::new(
873                    row.fields()
874                        .iter()
875                        .filter(|field| schema.contains_column(&field.name))
876                        .cloned()
877                        .collect(),
878                )
879            })
880            .collect(),
881    )
882}
883
884fn require_owned_payload(path: &Path) -> MsResult<()> {
885    for entry in fs::read_dir(path).map_err(io)? {
886        let entry = entry.map_err(io)?;
887        let kind = entry.file_type().map_err(io)?;
888        if kind.is_symlink() {
889            return Err(io(format!(
890                "subtable payload {} is symbolic",
891                entry.path().display()
892            )));
893        }
894        if kind.is_dir() {
895            require_owned_payload(&entry.path())?;
896        }
897    }
898    Ok(())
899}
