4//! The command-line and task surface of native CASA-RS imaging, on the
5//! shared task front end (#931).
6//!
7//! The binary hosts CASA's `tclean` (the imager re-surfaced, #931). The
8//! front end resolves tclean's parameters through the provider-contracts
9//! catalog: command-line flags and a `--json-run` request become the same
10//! resolved values, and those values, and nothing else, make the
11//! [`ImagingRequest`] the application runs, read as CASA's modes read them
12//! ([`ImagingRequest::from_tclean`]). A run's reply carries its
13//! [`ImagerRunTaskResult`], and returns CASA's summary record
14//! ([`tclean_record`]).
15

19mod task_contract;

34fn imager_surface() -> Result<SurfaceContractBundle, String> {
35    casa_provider_contracts::builtin_surface_bundle("tclean")
36}
37
38fn workspace() -> Result<PathBuf, String> {
39    std::env::current_dir().map_err(|error| format!("the working directory: {error}"))
40}
41
42/// The request `resolved`, every tclean parameter, makes. A value the
43/// catalog accepts but casa-rs cannot honour in the request's modes, or that
44/// the request's own read refuses (`stokes = "IX"`), is refused at its
45/// parameter.
46fn request_from(
47    resolved: Map<String, Value>,
48) -> Result<(ImagingParameters, ImagingRequest), TaskFailure> {
49    let request = ImagingRequest::from_tclean(&resolved).map_err(|refusal| {
50        let failure = TaskFailure::validation(refusal.message);
51        if refusal.parameter.is_empty() {
52            failure
53        } else {
54            failure.at(refusal.parameter)
55        }
56    })?;
57    Ok((ImagingParameters(resolved), request))
58}
59
60/// Resolve `parameters`, catalog-named and sparse, over the catalog's
61/// defaults: the resolved parameters and the request they make.
62pub fn resolve_request(
63    parameters: &ImagingParameters,
64) -> Result<(ImagingParameters, ImagingRequest), String> {
65    let resolved = resolve_plain_values(imager_surface()?, workspace()?, &parameters.0)
66        .map_err(|error| format!("resolve tclean parameters: {error}"))?;
67    request_from(resolved).map_err(|failure| failure.message)
68}

70/// Run `request`, resolved as `parameters`: the run summary, with the
71/// parameters as its echo, is written beside the products
72/// (`<imagename>.summary.json`); SIGINT cancels the run.
73pub fn run(
74    parameters: ImagingParameters,
75    request: &ImagingRequest,
76) -> Result<ImagerRunTaskResult, String> {
77    let started = Instant::now();
78    let mut summary = request.imagename.clone().into_os_string();
79    summary.push(".summary.json");
80    let context = RunContext {
81        host: HostResources::detect().map_err(|error| error.to_string())?,
82        policy: request.resource_policy(),
83        cancel: interrupt::token().clone(),
84        summary: Some(SummaryTarget {
85            path: summary.into(),
86            request: Value::Object(parameters.0.clone()),
87        }),
88    };
89    let outcome = execute(request, context).map_err(|error| error.to_string())?;
90    Ok(ImagerRunTaskResult::from_outcome(
91        parameters,
92        request,
93        &outcome,
94        started.elapsed(),
95    ))
96}

98/// The front end of the `casars-imager` binary, which hosts `tclean`.
99pub fn task_front_end()
100-> TaskFrontEnd<impl Fn(&TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure>> {
101    TaskFrontEnd::new(
102        "casars-imager",
103        imager_task_schema_bundle(),
104        &["tclean"],
105        run_task,
106    )
107    .expect("the imager contract carries its surface")
108}
109
110/// The resolved parameters of `task`, a tclean run, and the request they
111/// make; a request they cannot make is refused.
112pub fn task_request(
113    task: &TaskInvocation<'_>,
114) -> Result<(ImagingParameters, ImagingRequest), TaskFailure> {
115    request_from(task.plain_values())
116}

118/// Run `task`, one resolved imager request, as the `casars-imager` binary
119/// does: the execution callback of [`task_front_end`]. SIGINT cancels it:
120/// the run stops at its next safe point, publishes nothing, and replies that
121/// it was interrupted.
122pub fn run_task(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
123    let (parameters, request) = task_request(task)?;
124    // The record echoes some of tclean's values: a dirty run's threshold as
125    // the caller gave it, a number as a number and text as text, and
126    // tclean's default 0.0 where it was not given.
127    let mut values = parameters.0.clone();
128    values.insert(
129        "threshold".to_string(),
130        task.session()
131            .given("threshold")
132            .map_or(Value::from(0.0), ParameterValue::to_plain_json),
133    );
134    interrupt::install().map_err(|error| {
135        TaskFailure::execution(format!("install the interrupt handler: {error}"))
136    })?;
137    let result = run(parameters, &request).map_err(|error| {
138        if interrupt::token().is_cancelled() {
139            TaskFailure::interrupted(format!("{error}; nothing was published"))
140        } else {
141            TaskFailure::execution(error)
142        }
143    })?;
144    let text = format!(
145        "Wrote CASA-compatible products at prefix {} ({} gridded samples, {} major cycles, \
146         {} reported minor iterations, {} actual components, stop={:?})\n",
147        request.imagename.display(),
148        result.run.gridded_samples,
149        result.run.major_cycles,
150        result.run.minor_iterations,
151        result.run.actual_minor_iterations,
152        result.run.clean_stop_reason,
153    );
154    let value = tclean_record(&request, &result.run, &values);
155    Ok(TaskSuccess::new(value, &result)?.with_text(text))
156}
