53/// A request compiled against its MeasurementSet, before the problem
54/// compiler binds the observation.
55pub(crate) struct Prepared {
56    pub(crate) specification: ProblemSpecification,
57    pub(crate) geometry: GeometryInput,
58    pub(crate) model_lifecycle: ModelLifecycleRequirements,
59    pub(crate) masks: ImageDomainReconstructionMaskPlans,
60    /// A direction-dependent gridder's residual units, which bind every
61    /// solver.
62    pub(crate) minor_cycle_image_response: Option<MinorCycleImageResponse>,
63    pub(crate) observation: SelectedObservationResolutionRequest,
64    pub(crate) write_model_column: bool,
65    pub(crate) write_corrected_data: bool,
66    /// What the native run deploys once the availability check passes.
67    pub(crate) deployment: Deployment,
68}

70/// The MeasurementSet as [`prepare`] surveyed it: the table, its geometry
71/// engine, the source budget, the selection survey and the spectral axis.
72struct Surveyed<'a> {
73    ms: &'a MeasurementSet,
74    engine: &'a MsCalEngine,
75    budget: SelectedObservationContentBudget,
76    survey: &'a Survey,
77    spectral: &'a PreparedSpectralAxis,
78}

80/// Compile `request` against its MeasurementSet.
81pub(crate) fn prepare(
82    request: &ImagingRequest,
83    context: &RunContext,
84) -> Result<Prepared, PrepareError> {
85    let ms = MeasurementSet::open(&request.vis)?;
86    let budget = casa_imaging_runtime::bootstrap_source_budget();
87    // The one Measures provider of the run: compile's geometry and the bound
88    // observation's both evaluate with it.
89    let measures = casa_ms::open_measures_runtime()?;
90    let engine = MsCalEngine::with_measures(&ms, Arc::clone(&measures))?;
91    let survey = selection::survey(request, &ms, budget, &engine)?;
92    let centre = direction::resolve_centre(request, &ms, &survey, &engine, budget)?;
93    let frame = FrameContext {
94        anchor_time_mjd_seconds: survey.first_time_mjd_seconds,
95        time_bounds_mjd_seconds: survey.time_bounds_mjd_seconds,
96        field_id: centre.field_id,
97        phase: centre.phase.clone(),
98        direction: centre.direction,
99        engine: &engine,
100    };
101    let moving_rest_frame = moving_rest_frame(request, &centre, &frame)?;
102    let spectral =
103        spectral::prepare_axis(request, &ms, &survey, &frame, moving_rest_frame.as_ref())?;
104    let surveyed = Surveyed {
105        ms: &ms,
106        engine: &engine,
107        budget,
108        survey: &survey,
109        spectral: &spectral,
110    };
111    let (continuum_transform, channels) = continuum_selection(request, &surveyed, &centre)?;
112    let observation_info = direction::observation_info(&ms, &survey, &centre, &engine)?;
113    let domains = domains::prepare_domains(
114        request,
115        &centre,
116        ImageSpectralCoordinate {
117            frequency_reference: spectral.output_frequency_reference,
118            reference_frequency_hz: spectral.reference_frequency_hz,
119            increment_hz: spectral.increment_hz,
120            rest_frequency_hz: spectral.image_rest_frequency_hz,
121        },
122        &observation_info,
123    )?;
124    let primary_beam = primary_beam_model(request, &ms)?;
125    let deployment = deployment(request, context, &surveyed, &domains, primary_beam)?;
126    let instrument = instrument::scientific_instrument_model(&request.gridder, &ms)?;
127    let specification = specification::specification(
128        request,
129        &spectral,
130        specification_inputs(
131            request,
132            &surveyed,
133            instrument,
134            primary_beam.is_some(),
135            continuum_transform,
136        )?,
137    )?;
138    let reconstruction_planes = match request.deconvolver {
139        Deconvolver::Mtmfs => request.nterms,
140        _ => spectral.output_channels,
141    };
142    let model_samples = domains::model_samples(&domains, reconstruction_planes, &request.stokes)?;
143    let geometry = geometry(request, &domains, &centre, &spectral);
144    let observation = observation_request(
145        request,
146        &ms,
147        survey,
148        &channels,
149        measures,
150        centre.ephemeris,
151        budget,
152    )?;
153    Ok(Prepared {
154        specification,
155        geometry,
156        model_lifecycle: ModelLifecycleRequirements::new(
157            ModelBounds::new(model_samples, model_samples, f64::MAX, f64::MAX)?,
158            NumericPrecision::F64,
159        ),
160        masks: domains::mask_plans(&domains)?,
161        minor_cycle_image_response: match &request.gridder {
162            Gridder::Mosaic { .. } | Gridder::Awproject(_) => Some(MinorCycleImageResponse::new(
163                specification::normalization(request),
164                specification::primary_beam_validity(request)?,
165            )?),
166            Gridder::Standard | Gridder::Wproject { .. } => None,
167        },
168        observation,
169        write_model_column: request.savemodel,
170        write_corrected_data: request.save_continuum_residual,
171        deployment,
172    })
173}

175/// The rest frame of a moving source (`cubesource`): the observatory frame
176/// at the anchor time with the ephemeris's radial velocity.
177fn moving_rest_frame(
178    request: &ImagingRequest,
179    centre: &Centre,
180    frame: &FrameContext<'_>,
181) -> Result<Option<MeasFrame>, PrepareError> {
182    if request.specmode != SpecMode::Cubesource {
183        return Ok(None);
184    }
185    let ephemeris = centre
186        .ephemeris
187        .as_ref()
188        .ok_or(PrepareError::MovingPhaseCentre)?;
189    let velocity = frame.engine.ephemeris_radial_velocity(
190        frame.anchor_time_mjd_seconds,
191        frame.field_id,
192        ephemeris,
193    )?;
194    Ok(Some(
195        frame
196            .engine
197            .spectral_frame_observatory_direction(
198                frame.anchor_time_mjd_seconds,
199                centre.phase.clone(),
200            )?
201            .with_radial_velocity(velocity),
202    ))
203}

205/// The continuum transform of a `fitspw` request and the source channels
206/// the observation reads: the fit channels with the output channels, or
207/// the output channels alone.
208fn continuum_selection(
209    request: &ImagingRequest,
210    surveyed: &Surveyed<'_>,
211    centre: &Centre,
212) -> Result<(Option<SequentialContinuumTransform>, WindowChannels), PrepareError> {
213    let Some(fitspw) = &request.fitspw else {
214        return Ok((None, surveyed.spectral.selected_source_channels.clone()));
215    };
216    let window = selection::one_window(&surveyed.survey.spectral_windows, "continuum subtraction")?;
217    let (transform, selected) = specification::continuum_transform(
218        fitspw,
219        request.fitorder,
220        i32::try_from(centre.field_id).expect("the centre's field id is a stored i32 FIELD_ID"),
221        window,
222        surveyed.spectral.window_channels(window.spw_id),
223    )?;
224    Ok((Some(transform), BTreeMap::from([(window.spw_id, selected)])))
225}

227/// The specification's inputs beyond the request: the instrument, the
228/// W and AW projections over the observed W range, the cube's density
229/// padding and the continuum transform.
230fn specification_inputs(
231    request: &ImagingRequest,
232    surveyed: &Surveyed<'_>,
233    instrument: Option<casa_imaging_model::InstrumentModel>,
234    primary_beam: bool,
235    continuum_transform: Option<SequentialContinuumTransform>,
236) -> Result<SpecificationInputs, PrepareError> {
237    let windows = &surveyed.survey.spectral_windows;
238    Ok(SpecificationInputs {
239        instrument,
240        uncorrected_mask: if primary_beam && request.pblimit >= 0.0 {
241            UncorrectedImageMaskPolicy::PrimaryBeam
242        } else {
243            UncorrectedImageMaskPolicy::None
244        },
245        w_projection: match &request.gridder {
246            Gridder::Wproject { wprojplanes } => Some(specification::w_projection(
247                *wprojplanes,
248                windows,
249                surveyed.survey.w,
250            )?),
251            _ => None,
252        },
253        aw_projection: match &request.gridder {
254            Gridder::Awproject(aw) => Some(specification::aw_projection(
255                aw,
256                windows,
257                surveyed.survey.w,
258            )?),
259            _ => None,
260        },
261        cube_density_padding: cube_density_padding(request, surveyed)?,
262        continuum_transform,
263    })
264}

266/// The selected-observation request: the survey's rows and the channels
267/// read, the visibility and weight columns, and the Measures provider and
268/// centre ephemeris compile evaluated with.
269fn observation_request(
270    request: &ImagingRequest,
271    ms: &MeasurementSet,
272    survey: Survey,
273    channels: &WindowChannels,
274    measures: Arc<dyn MeasuresProvider>,
275    ephemeris: Option<SelectedObservationEphemeris>,
276    budget: SelectedObservationContentBudget,
277) -> Result<SelectedObservationResolutionRequest, PrepareError> {
278    let weight_column = if survey.weight_spectrum_complete {
279        OwnerWeightColumn::WeightSpectrum
280    } else {
281        OwnerWeightColumn::Weight
282    };
283    Ok(SelectedObservationResolutionRequest::new(
284        request.vis.display().to_string(),
285        selection::observation_selection(ms, survey, channels)?,
286        selection::visibility_column(ms, request.datacolumn)?,
287        weight_column,
288        budget,
289        measures,
290    )
291    .with_ephemeris(ephemeris))
292}

294/// The image geometry: the domains, the centre laws and the spectral axis.
295fn geometry(
296    request: &ImagingRequest,
297    domains: &[PreparedImageDomain],
298    centre: &Centre,
299    spectral: &PreparedSpectralAxis,
300) -> GeometryInput {
301    let direction_dependent = matches!(
302        request.gridder,
303        Gridder::Mosaic { .. } | Gridder::Awproject(_)
304    );
305    GeometryInput::new(
306        domains.iter().map(PreparedImageDomain::spec).collect(),
307        CentreLaws::new(
308            centre.law.clone(),
309            instrument::pointing_centre_law(&request.gridder),
310        ),
311        if direction_dependent {
312            UvwCoordinateLaw::MosaicPhaseTrackingCentre
313        } else {
314            UvwCoordinateLaw::PhaseTrackingCentre
315        },
316        spectral.coordinate(),
317    )
318}

320/// The primary beam the products are written with: the mosaic
321/// sensitivity of a direction-dependent gridder, else the telescope's
322/// analytic beam when `.pb` or `.image.pbcor` is asked for.
323fn primary_beam_model(
324    request: &ImagingRequest,
325    ms: &MeasurementSet,
326) -> Result<Option<AnalyticPrimaryBeamModel>, PrepareError> {
327    Ok(match request.gridder {
328        Gridder::Mosaic { .. } | Gridder::Awproject(_) => {
329            Some(AnalyticPrimaryBeamModel::MosaicSensitivity)
330        }
331        Gridder::Standard | Gridder::Wproject { .. } if request.write_pb || request.pbcor => {
332            Some(instrument::standard_primary_beam_model(ms)?)
333        }
334        Gridder::Standard | Gridder::Wproject { .. } => None,
335    })
336}

338/// CASA's padded per-channel density grid of a linearly interpolated
339/// cube with per-channel density.
340fn cube_density_padding(
341    request: &ImagingRequest,
342    surveyed: &Surveyed<'_>,
343) -> Result<Option<usize>, PrepareError> {
344    let (survey, spectral) = (surveyed.survey, surveyed.spectral);
345    if !specification::cube_density_padded(request, spectral) {
346        return Ok(None);
347    }
348    let window = selection::one_window(&survey.spectral_windows, "cube density")?;
349    Ok(Some(
350        surveyed.ms.selected_observation_cube_density_padding(
351            &survey.row_selection,
352            SelectedObservationSpectralWindow::borrow_selected(
353                u32::try_from(window.spw_id).expect("SPW ids are nonnegative stored i32 values"),
354                window.frequency_reference,
355                &window.frequencies_hz,
356                &window.channel_widths_hz,
357                spectral.window_channels(window.spw_id),
358            ),
359            survey.fields.iter().copied(),
360            spectral.output_frequency_reference,
361            [
362                spectral.reference_frequency_hz,
363                spectral.reference_frequency_hz
364                    + (spectral.output_channels - 1) as f64 * spectral.increment_hz,
365            ],
366            spectral.output_channels,
367            surveyed.engine,
368            surveyed.budget.row_io_budget(),
369        )?,
370    ))
371}

373/// The native run's runtime, product publication and AW catalog, still to
374/// be deployed: no directory exists and no AW cell is generated until
375/// [`Deployment::deploy`].
376pub(crate) struct Deployment {
377    /// The runtime; its spill directory is made canonical on deployment.
378    runtime: ApplicationRuntime,
379    publication: ApplicationPublication,
380    /// Every domain's output directory.
381    output_directories: Vec<PathBuf>,
382    aw: Option<AwPlan>,
383}
384
385/// The AW catalog a run opens and, for a native cache, the cells to bring
386/// to the request's policy.
387struct AwPlan {
388    root: PathBuf,
389    native: Option<(NativeAwCachePolicy, NativeAwRequestInput)>,
390    indexing: AwIndexing,
391    resident_bytes: usize,
392}
393
394impl Deployment {
395    /// Create every domain's output directory, settle the spill directory
396    /// beside the main image and bring a native AW cache to the request's
397    /// policy.
398    pub(crate) fn deploy(self) -> Result<ApplicationNative, PrepareError> {
399        for directory in &self.output_directories {
400            std::fs::create_dir_all(directory).map_err(|source| PrepareError::OutputDirectory {
401                path: directory.clone(),
402                source,
403            })?;
404        }
405        let mut runtime = self.runtime;
406        runtime.spill_directory = runtime.spill_directory.canonicalize().map_err(|source| {
407            PrepareError::OutputDirectory {
408                path: runtime.spill_directory.clone(),
409                source,
410            }
411        })?;
412        Ok(ApplicationNative {
413            runtime,
414            publication: self.publication,
415            aw_catalog: self.aw.map(AwPlan::deploy).transpose()?,
416        })
417    }
418}
419
420impl AwPlan {
421    /// Under every policy the request's cells are compared with the
422    /// directory: reuse needs every one, generation fills in the absent
423    /// ones, regeneration clears the cache first; the loader then checks
424    /// each cell's sky increment against the image (plan section 5.6).
425    fn deploy(self) -> Result<AwCatalogDeployment, PrepareError> {
426        if let Some((policy, input)) = &self.native {
427            let (present, expected) = AwCatalog::native_cells_present(&self.root, input);
428            match policy {
429                NativeAwCachePolicy::ReuseOnly if present != expected => {
430                    return Err(PrepareError::AwCacheIncomplete { present, expected });
431                }
432                NativeAwCachePolicy::ReuseOnly => {}
433                NativeAwCachePolicy::GenerateMissing => {
434                    if present != expected {
435                        AwCatalog::generate_native(&self.root, input, true)?;
436                    }
437                }
438                NativeAwCachePolicy::Regenerate => {
439                    AwCatalog::clear_native(&self.root)?;
440                    AwCatalog::generate_native(&self.root, input, false)?;
441                }
442            }
443        }
444        Ok(AwCatalogDeployment {
445            root: self.root,
446            indexing: self.indexing,
447            resident_bytes: self.resident_bytes,
448        })
449    }
450}

452/// The native run's deployment: the runtime with the paged cube state
453/// beside the main image, the product controls and sink, and the AW plan.
454fn deployment(
455    request: &ImagingRequest,
456    context: &RunContext,
457    surveyed: &Surveyed<'_>,
458    domains: &[PreparedImageDomain],
459    primary_beam: Option<AnalyticPrimaryBeamModel>,
460) -> Result<Deployment, PrepareError> {
461    let parent = |path: &Path| {
462        path.parent()
463            .unwrap_or_else(|| Path::new("."))
464            .to_path_buf()
465    };
466    let mut controls = ContinuumProductControls::new(request.psfcutoff as f32).map_err(|_| {
467        PrepareError::PsfCutoff {
468            psfcutoff: request.psfcutoff,
469        }
470    })?;
471    if let Some(model) = primary_beam {
472        controls = controls.with_primary_beam_model(model);
473    }
474    Ok(Deployment {
475        runtime: ApplicationRuntime {
476            host: context.host,
477            resource_policy: context.policy,
478            backend: request.backend,
479            grid_precision: request.gridprecision,
480            cancel: context.cancel.clone(),
481            spill_directory: parent(&request.imagename),
482            summary: context.summary.clone(),
483        },
484        publication: ApplicationPublication {
485            controls,
486            sink: CasaImageProductSink::for_domains(
487                domains.iter().map(PreparedImageDomain::output),
488            ),
489        },
490        output_directories: domains
491            .iter()
492            .map(|domain| parent(&domain.output))
493            .collect(),
494        aw: match &request.gridder {
495            Gridder::Awproject(aw) => Some(aw_plan(request, aw, surveyed)?),
496            _ => None,
497        },
498    })
499}

501/// The AW catalog to open: a CASA cache read as it is, or a native cache
502/// in CASA's format with the input its cells are generated from.
503fn aw_plan(
504    request: &ImagingRequest,
505    aw: &AwProjection,
506    surveyed: &Surveyed<'_>,
507) -> Result<AwPlan, PrepareError> {
508    let (root, native) = match &aw.cf_source {
509        AwCfSource::CasaImport { cfcache } => (cfcache.clone(), None),
510        AwCfSource::NativeEvla {
511            native_cf_cache,
512            native_cf_policy,
513            ..
514        } => {
515            let first_row = surveyed
516                .survey
517                .first_cross_row
518                .ok_or(PrepareError::NoCrossCorrelationRow)?;
519            let input = native_aw::resolve(request, aw, surveyed, first_row)?;
520            input.validate()?;
521            (native_cf_cache.clone(), Some((*native_cf_policy, input)))
522        }
523    };
524    Ok(AwPlan {
525        root,
526        native,
527        indexing: AwIndexing {
528            conjugate_beams: true,
529            image_reference_hz: surveyed.spectral.reference_frequency_hz,
530        },
531        resident_bytes: aw.cf_resident_mb.checked_mul(1 << 20).ok_or(
532            PrepareError::CfResidentSize {
533                megabytes: aw.cf_resident_mb,
534            },
535        )?,
536    })
