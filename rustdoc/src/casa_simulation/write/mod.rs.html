2//! Writing a session's MeasurementSet at `close`.
3//!
4//! One pass creates the MeasurementSet in a staging directory, fills the
5//! subtables, streams MAIN through the bounded write session a batch of
6//! column-shaped rows at a time, and publishes the result. The layout is
7//! casa-rs's conventional one: IncrementalStMan and StandardStMan scalars,
8//! TiledShapeStMan visibility columns with CASA's tile shapes, and
9//! TiledColumnStMan for UVW. Following plan decision 6, MAIN has DATA but
10//! no MODEL_DATA or CORRECTED_DATA. DATA is the sum of the predictions of
11//! every `predict` step covering the row's scan (zero without one) plus the
12//! noise of every `corrupt` step covering it.
13//!
14//! Before anything is written, what the pass holds is admitted against the
15//! session's resource policy ([`resources`]); each batch's geometry,
16//! prediction ([`prediction`]) and noise are then computed on the session's
17//! worker team ([`pipeline`]).

56fn write_error(what: &str) -> impl Fn(casa_ms::MeasurementSetWriteError) -> SimulatorError + '_ {
57    move |error| SimulatorError::raised(format!("{what}: {error}"))
58}

60/// Create the session's MeasurementSet.
61pub(crate) fn write_measurement_set(session: &Simulator) -> SimulatorResult<()> {
62    let window = ReservationWindow::open();
63    let cell = cell_shape(session)?;
64    let shape = cell.map(|cell| main_shape(session, cell));
65    let resources = &session.options().resources;
66    let prediction = match cell {
67        Some(cell) => PredictionStage::new(session, cell, resources.prediction_backend())?,
68        None => None,
69    };
70    // Admitted before anything is written, so a refusal leaves nothing.
71    let admission = shape
72        .as_ref()
73        .map(|shape| resources::admit_main_write(resources, shape, prediction.as_ref()))
74        .transpose()?;
75    let target = MeasurementSetCreateTarget::prepare(session.path(), true)
76        .map_err(write_error("prepare the MeasurementSet"))?;
77    let staging = target.staging_path().to_path_buf();
78    let mut ms = MeasurementSet::create(
79        &staging,
80        MeasurementSetBuilder::new()
81            .with_main_column(OptionalMainColumn::Data)
82            .with_optional_subtable(SubtableId::Source),
83    )
84    .map_err(|error| SimulatorError::raised(format!("create the MeasurementSet: {error}")))?;
85    std::fs::create_dir_all(&staging).map_err(|error| {
86        SimulatorError::raised(format!("create {}: {error}", staging.display()))
87    })?;
88    let info = ms.main_table().info().clone();
89    ms.main_table_mut().set_info(TableInfo {
90        table_type: info.table_type,
91        sub_type: "simulator".to_string(),
92        readme: MAIN_README.iter().map(|line| (*line).to_string()).collect(),
93    });
94    subtables::write_subtables(&mut ms, session)?;
95    let generated = subtables::subtable_overrides(session)?;
96    match shape.zip(admission) {
97        Some((shape, admission)) => {
98            let main = MainTarget {
99                staging: &staging,
100                generated: &generated,
101                window: &window,
102            };
103            write_main(&mut ms, session, &main, &shape, (admission, prediction))?;
104        }
105        None => ms
106            .save()
107            .map_err(|error| SimulatorError::raised(format!("save the MeasurementSet: {error}")))?,
108    }
109    target
110        .commit()
111        .map_err(write_error("publish the MeasurementSet"))
112}

114/// The cell shape every scan shares: (correlations, channels).
115fn cell_shape(session: &Simulator) -> SimulatorResult<Option<(usize, usize)>> {
116    let mut shape = None;
117    for scan in session.scans() {
118        let window = &session.spectral_windows()[scan.spectral_window_id];
119        let this = (window.correlations.len(), window.channel_count);
120        match shape {
121            None => shape = Some(this),
122            Some(first) if first != this => {
123                return Err(SimulatorError::raised(
124                    "observing spectral windows of different shapes in one MeasurementSet is not \
125                     yet implemented in casa-rs",
126                ));
127            }
128            Some(_) => {}
129        }
130    }
131    Ok(shape)
132}

134/// MAIN as the session's scans make it, with cells of `(correlations,
135/// channels)`.
136fn main_shape(session: &Simulator, (correlations, channels): (usize, usize)) -> MainShape<'_> {
137    let scans = session.scans();
138    MainShape {
139        rows: scans.iter().map(Scan::row_count).sum(),
140        correlations,
141        channels,
142        telescope: session.telescope(),
143        largest_integration: scans.iter().map(Scan::baseline_count).max().unwrap_or(0),
144        antennas: scans
145            .iter()
146            .map(|scan| scan.antenna_count)
147            .max()
148            .unwrap_or(0),
149        generated_bytes: subtables::generated_bytes(session),
150    }
151}

161/// Write MAIN and save the MeasurementSet, holding `admission` until done,
162/// predicting with `prediction` when the session predicts.
163fn write_main<'s>(
164    ms: &mut MeasurementSet,
165    session: &'s Simulator,
166    target: &MainTarget<'_>,
167    shape: &MainShape<'_>,
168    (admission, mut prediction): (MainAdmission, Option<PredictionStage<'s>>),
169) -> SimulatorResult<()> {
170    let started = Instant::now();
171    let batch_rows = admission.plan.batch_rows().max(1);
172    if let Some(stage) = &mut prediction {
173        stage.start(admission.block_rows, admission.cores.streaming_workers);
174    }
175    let backend = prediction.as_ref().map_or_else(
176        || session.options().resources.resolved_backend(false),
177        PredictionStage::backend,
178    );
179    let mut writer = MainWriter::start(target.staging, admission.plan.clone())?;
180    let geometry = ArrayGeometry::new(
181        session.telescope(),
182        session.antennas(),
183        session.options().measures.clone(),
184    )?;
185    let team = WorkerTeam::new(admission.cores.streaming_workers)
186        .map_err(|error| SimulatorError::raised(format!("the worker team: {error}")))?;
187    let scans = session.scans();
188    let mut pipeline = Pipeline::new(
189        scans,
190        &geometry,
191        &team,
192        (prediction, session.data_plan()),
193        (shape.correlations, shape.channels),
194    );
195    let integrations = scans
196        .iter()
197        .enumerate()
198        .filter(|(_, scan)| scan.baseline_count() > 0)
199        .flat_map(|(index, scan)| scan.integration_times_s().map(move |time| (index, time)));
200    let setup = started.elapsed();
201    pipeline.write_rows(&mut writer, integrations, shape.rows, batch_rows)?;
202    let finishing = Instant::now();
203    let written = writer.finish(ms, shape.rows, target.generated)?;
204    let finish = finishing.elapsed();
205    let times = &pipeline.times;
206    tracing::info!(
207        rows = shape.rows,
208        batch_rows,
209        block_rows = admission.block_rows,
210        batches = times.batches,
211        workers = admission.cores.streaming_workers,
212        policy_workers = admission.policy_cores.streaming_workers,
213        backend = ?backend,
214        reserved_bytes = admission.reservation.memory(),
215        peak_reserved_bytes = target.window.close(),
216        geometry_s = times.geometry.as_secs_f64(),
217        zero_s = times.zero.as_secs_f64(),
218        predict_s = times.predict.as_secs_f64(),
219        block_fill_s = pipeline.block_fill_time().as_secs_f64(),
220        cells_s = times.cells.as_secs_f64(),
221        send_wait_s = times.send.as_secs_f64(),
222        setup_s = setup.as_secs_f64(),
223        finish_s = finish.as_secs_f64(),
224        writer_write_s = written.write_seconds,
225        writer_finalize_s = written.finalize_seconds,
226        bytes_written = written.bytes_written,
227        total_s = started.elapsed().as_secs_f64(),
228        "simulation pass"
229    );
230    Ok(())
231}

233/// The resolved MAIN scalar columns.
234struct ScalarColumns {
235    antenna1: ScalarColumnId,
236    antenna2: ScalarColumnId,
237    flag_row: ScalarColumnId,
238    time: ScalarColumnId,
239    time_centroid: ScalarColumnId,
240    constants: [(ScalarColumnId, &'static str); 11],
241}
242
243/// MAIN's write session and the scalar runs of the integration being
244/// pushed.
245struct MainWriter {
246    session: MeasurementSetWriteSession,
247    columns: ScalarColumns,
248    antenna1: Vec<i32>,
249    antenna2: Vec<i32>,
250    flag_row: Vec<bool>,
251}
252
253impl MainWriter {
254    fn start(staging: &std::path::Path, plan: MeasurementSetWritePlan) -> SimulatorResult<Self> {
255        let session = MeasurementSetWriteSession::start(staging, plan)
256            .map_err(write_error("start the MAIN write"))?;
257        let id = |name: &str| {
258            session
259                .scalar_column(name)
260                .map_err(write_error("MAIN scalar column"))
261        };
262        let constant = |name: &'static str| id(name).map(|column| (column, name));
263        let columns = ScalarColumns {
264            antenna1: id("ANTENNA1")?,
265            antenna2: id("ANTENNA2")?,
266            flag_row: id("FLAG_ROW")?,
267            time: id("TIME")?,
268            time_centroid: id("TIME_CENTROID")?,
269            constants: [
270                constant("ARRAY_ID")?,
271                constant("DATA_DESC_ID")?,
272                constant("EXPOSURE")?,
273                constant("FEED1")?,
274                constant("FEED2")?,
275                constant("FIELD_ID")?,
276                constant("INTERVAL")?,
277                constant("OBSERVATION_ID")?,
278                constant("PROCESSOR_ID")?,
279                constant("SCAN_NUMBER")?,
280                constant("STATE_ID")?,
281            ],
282        };
283        Ok(Self {
284            session,
285            columns,
286            antenna1: Vec::new(),
287            antenna2: Vec::new(),
288            flag_row: Vec::new(),
289        })
290    }
291
292    /// Append the scalar columns of one integration of `scan` at stored
293    /// TIME `time_s`, whose rows are `rows`.
294    fn push_scalars(
295        &mut self,
296        scan: &Scan,
297        time_s: f64,
298        rows: &[RowGeometry],
299    ) -> SimulatorResult<()> {
300        self.antenna1.clear();
301        self.antenna2.clear();
302        self.flag_row.clear();
303        for row in rows {
304            self.antenna1.push(row.antenna1 as i32);
305            self.antenna2.push(row.antenna2 as i32);
306            self.flag_row.push(row.flagged);
307        }
308        let count = rows.len();
309        let error = write_error("MAIN scalars");
310        let columns = &self.columns;
311        let session = &mut self.session;
312        session
313            .push_scalar_column(
314                columns.antenna1,
315                StreamedScalarValues::Int32(&self.antenna1),
316            )
317            .map_err(&error)?;
318        session
319            .push_scalar_column(
320                columns.antenna2,
321                StreamedScalarValues::Int32(&self.antenna2),
322            )
323            .map_err(&error)?;
324        session
325            .push_scalar_column(columns.flag_row, StreamedScalarValues::Bool(&self.flag_row))
326            .map_err(&error)?;
327        for column in [columns.time, columns.time_centroid] {
328            session
329                .push_repeated_scalar(column, Some(ScalarValue::Float64(time_s)), count)
330                .map_err(&error)?;
331        }
332        for (column, name) in columns.constants {
333            let value = match name {
334                "DATA_DESC_ID" => ScalarValue::Int32(scan.spectral_window_id as i32),
335                "EXPOSURE" | "INTERVAL" => ScalarValue::Float64(scan.integration_s),
336                "FIELD_ID" => ScalarValue::Int32(scan.field_id as i32),
337                "OBSERVATION_ID" => ScalarValue::Int32(scan.observation_id),
338                "SCAN_NUMBER" => ScalarValue::Int32(scan.scan_number),
339                "STATE_ID" => ScalarValue::Int32(scan.state_id as i32),
340                // ARRAY_ID, FEED1, FEED2 and PROCESSOR_ID.
341                _ => ScalarValue::Int32(0),
342            };
343            session
344                .push_repeated_scalar(column, Some(value), count)
345                .map_err(&error)?;
346        }
347        Ok(())
348    }
349
350    /// An empty batch, with the buffers of one the writer has finished with
351    /// when there is one.
352    fn visibility_columns(&self) -> VisibilityColumns {
353        self.session.visibility_columns()
354    }
355
356    /// Queue a full batch for the writer's thread; blocks while its queue is
357    /// full.
358    fn send(&self, batch: VisibilityColumns) -> SimulatorResult<()> {
359        self.session
360            .send_visibility_columns(batch)
361            .map_err(write_error("MAIN visibility batch"))
362    }
363
364    fn finish(
365        self,
366        ms: &mut MeasurementSet,
367        row_count: usize,
368        generated: &casa_ms::SubtableColumnOverrides,
369    ) -> SimulatorResult<MeasurementSetWriteTelemetry> {
370        let mut overrides = ColumnOverrides::for_row_count(row_count);
371        for column in ["DATA", "FLAG", "FLAG_CATEGORY", "UVW", "WEIGHT", "SIGMA"] {
372            overrides.insert_deferred(column);
373        }
374        self.session
375            .save_and_finish_with_subtables(ms, &overrides, generated)
376            .map_err(write_error("finish the MAIN write"))
377    }
