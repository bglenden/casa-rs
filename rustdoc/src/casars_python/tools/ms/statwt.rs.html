8use casa_ms::MsSelection;
9use pyo3::prelude::*;
10use pyo3::types::{PyBool, PyDict, PyDictMethods, PyList, PyListMethods, PyString};

14#[pymethods]
15impl PyMsTool {
16    #[pyo3(signature = (items = None, onlyparse = false), text_signature = "($self, items={}, onlyparse=False)")]
17    fn msselect(
18        &self,
19        py: Python<'_>,
20        items: Option<&Bound<'_, PyDict>>,
21        #[pyo3(from_py_with = strict_bool)] onlyparse: bool,
22    ) -> PyResult<bool> {
23        let Some(request) = items
24            .and_then(|items| selection_request(items).ok())
25            .flatten()
26        else {
27            return Ok(false);
28        };
29        py.detach(|| {
30            let mut attached = self.attached.lock().map_err(runtime_error)?;
31            let Some(attached) = attached.as_mut() else {
32                return Ok(false);
33            };
34            let ms = StatWtMs::open(&attached.path, false).map_err(runtime_error)?;
35            let Ok(resolved) = ms.select(&request) else {
36                return Ok(false);
37            };
38            if onlyparse {
39                return Ok(true);
40            }
41            let mut selected = resolved.iter().map(|row| row.row_index).collect::<Vec<_>>();
42            if let Some(previous) = &attached.selected_rows {
43                selected.retain(|row| previous.binary_search(row).is_ok());
44            }
45            let nonempty = !selected.is_empty();
46            attached.selected_rows = Some(selected);
47            Ok(nonempty)
48        })
49    }
50
51    #[allow(clippy::too_many_arguments)]
52    #[pyo3(signature = (combine = "", timebin = None, slidetimebin = false, chanbin = None, minsamp = 2,
53        statalg = "classic", fence = -1.0, center = "mean", lside = true, zscore = -1.0, maxiter = -1,
54        fitspw = "", excludechans = false, wtrange = None, preview = false, datacolumn = "corrected"),
55        text_signature = "($self, combine='', timebin=1, slidetimebin=False, chanbin='spw', minsamp=2, statalg='classic', fence=-1.0, center='mean', lside=True, zscore=-1.0, maxiter=-1, fitspw='', excludechans=False, wtrange=[], preview=False, datacolumn='corrected')")]
56    fn statwt<'py>(
57        &self,
58        py: Python<'py>,
59        combine: &str,
60        timebin: Option<&Bound<'py, PyAny>>,
61        #[pyo3(from_py_with = strict_bool)] slidetimebin: bool,
62        chanbin: Option<&Bound<'py, PyAny>>,
63        minsamp: i64,
64        statalg: &str,
65        fence: f64,
66        center: &str,
67        #[pyo3(from_py_with = strict_bool)] lside: bool,
68        zscore: f64,
69        maxiter: i64,
70        fitspw: &str,
71        #[pyo3(from_py_with = strict_bool)] excludechans: bool,
72        wtrange: Option<Vec<f64>>,
73        #[pyo3(from_py_with = strict_bool)] preview: bool,
74        datacolumn: &str,
75    ) -> PyResult<Bound<'py, PyDict>> {
76        let options = StatWtOptions {
77            data_column: StatWtDataColumn::parse(datacolumn).map_err(runtime_error)?,
78            algorithm: StatWtAlgorithm::parse(statalg, fence, center, lside, zscore, maxiter)
79                .map_err(runtime_error)?,
80            minimum_samples: usize::try_from(minsamp).unwrap_or(0),
81            combine: StatWtCombine::parse(combine),
82            time_bin: read_bin(
83                timebin,
84                StatWtTimeBin::Timestamps(1),
85                StatWtTimeBin::Timestamps,
86                StatWtTimeBin::duration,
87            )?,
88            sliding: slidetimebin,
89            channel_bin: read_bin(
90                chanbin,
91                StatWtChannelBin::Window,
92                StatWtChannelBin::Channels,
93                StatWtChannelBin::parse,
94            )?,
95            fit_spw: fitspw.to_owned(),
96            exclude_channels: excludechans,
97            weight_range: statwt_weight_range(wtrange.as_deref().unwrap_or_default())
98                .map_err(runtime_error)?,
99        };
100        let report = py.detach(|| {
101            let mut attached = self.attached.lock().map_err(runtime_error)?;
102            let Some(attached) = attached.as_mut() else {
103                return Ok(None);
104            };
105            if !preview && !attached.writable {
106                return Err(runtime_error("MeasurementSet is not writable"));
107            }
108            let mut transient = None;
109            let locks = if preview {
110                None
111            } else if !attached.write_locks.is_empty() {
112                Some(&mut attached.write_locks)
113            } else {
114                Some(
115                    transient
116                        .insert(StatWtMs::write_locks(&attached.sources).map_err(runtime_error)?),
117                )
118            };
119            let mut ms = StatWtMs::open(&attached.path, false).map_err(runtime_error)?;
120            let mut selected = ms.select(&MsSelection::default()).map_err(runtime_error)?;
121            if let Some(rows) = &attached.selected_rows {
122                selected.retain(|row| rows.binary_search(&row.row_index).is_ok());
123            }
124            let report =
125                estimate_statwt(&mut ms, &selected, &options, locks).map_err(runtime_error)?;
126            drop(ms);
127            if let Some(locks) = transient {
128                for lock in locks.into_values() {
129                    lock.release().map_err(runtime_error)?;
130                }
131            }
132            Ok(Some(report))
133        })?;
134        let result = PyDict::new(py);
135        if let Some(report) = report {
136            result.set_item("mean", report.mean)?;
137            result.set_item("variance", report.variance)?;
138        }
139        Ok(result)
140    }
141}

143fn read_bin<T>(
144    value: Option<&Bound<'_, PyAny>>,
145    default: T,
146    count: impl FnOnce(usize) -> T,
147    quantity: impl FnOnce(&str) -> casa_ms::MsResult<T>,
148) -> PyResult<T> {
149    let Some(value) = value else {
150        return Ok(default);
151    };
152    if let Ok(text) = value.cast::<PyString>() {
153        return quantity(text.to_str()?).map_err(runtime_error);
154    }
155    if let Ok(list) = value.cast::<PyList>()
156        && list.is_empty()
157    {
158        return Ok(default);
159    }
160    if !value.is_instance_of::<PyBool>()
161        && let Ok(size) = value.extract::<usize>()
162    {
163        return Ok(count(size));
164    }
165    Err(runtime_error(
166        "statwt bin requires an integer count or a string quantity",
167    ))
168}

170fn selection_request(items: &Bound<'_, PyDict>) -> PyResult<Option<MsSelection>> {
171    let mut request = MsSelection::default();
172    let mut named = false;
173    for (key, value) in items {
174        let key = key.extract::<&str>()?;
175        let slot = match key {
176            "spw" => &mut request.spw,
177            "time" => &mut request.timerange,
178            "field" => &mut request.field,
179            "baseline" | "antenna" => &mut request.antenna,
180            "scan" => &mut request.scan,
181            "scanintent" | "state" => &mut request.intent,
182            "polarization" => &mut request.correlation,
183            "uvdist" => &mut request.uvrange,
184            "observation" => &mut request.observation,
185            "array" => &mut request.array,
186            "taql" => &mut request.msselect,
187            _ => continue,
188        };
189        let value = value.extract::<String>()?;
190        *slot = (!value.is_empty()).then_some(value);
191        named |= slot.is_some();
192    }
193    Ok(named.then_some(request))
194}
