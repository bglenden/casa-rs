6//! the preset name, the selector names, then each request option. A plot

83fn preset_name(preset: MsPlotPreset) -> &'static str {
84    PRESET_NAMES
85        .iter()
86        .find_map(|(name, candidate)| (*candidate == preset).then_some(*name))
87        .expect("every preset has a Python name")
88}

90/// The preset a name selects: case-insensitive, `-` for `_`, and
91/// `flagrow_vs_time` for `flag_row_vs_time`.
92fn parse_preset(preset: &Bound<'_, PyAny>) -> PyResult<MsPlotPreset> {
93    let normalized = preset
94        .call_method0("strip")?
95        .call_method0("upper")?
96        .call_method1("replace", ("-", "_"))?
97        .extract::<String>()?;
98    let normalized = if normalized == "FLAGROW_VS_TIME" {
99        "FLAG_ROW_VS_TIME".to_string()
100    } else {
101        normalized
102    };
103    PRESET_NAMES
104        .iter()
105        .find_map(|(name, candidate)| {
106            (name.to_ascii_uppercase() == normalized).then_some(*candidate)
107        })
108        .ok_or_else(|| {
109            let allowed = PRESET_NAMES.map(|(name, _)| name).join(", ");
110            match preset.repr() {
111                Ok(repr) => PyValueError::new_err(format!(
112                    "unknown MeasurementSet plot preset {repr}; expected one of: {allowed}"
113                )),
114                Err(error) => error,
115            }
116        })
117}

119fn optional_text(value: Option<Bound<'_, PyAny>>) -> PyResult<Option<String>> {
120    value
121        .filter(|value| !value.is_none())
122        .map(|value| value.extract::<String>())
123        .transpose()
124}

126fn flag(name: &str, value: &Bound<'_, PyAny>) -> PyResult<bool> {
127    if let Ok(flag) = value.cast::<PyBool>() {
128        return Ok(flag.is_true());
129    }
130    match value.extract::<i64>() {
131        Ok(0) => Ok(false),
132        Ok(1) => Ok(true),
133        _ => Err(PyTypeError::new_err(format!("{name} must be a bool"))),
134    }
135}
136
137fn count(value: &Bound<'_, PyAny>) -> PyResult<u64> {
138    value.extract::<u64>().map_err(|error| {
139        if error.is_instance_of::<PyOverflowError>(value.py()) {
140            PyValueError::new_err(format!(
141                "u64 requires 0 <= value < {}",
142                u128::from(u64::MAX) + 1
143            ))
144        } else {
145            error
146        }
147    })
148}

150fn platform_count(py: Python<'_>, name: &str, value: u64) -> PyResult<usize> {
151    usize::try_from(value).map_err(|_| reason_error(py, format!("{name} exceeds platform usize")))
152}

154/// The selectors of a request, refusing names msexplore does not select by.
155fn selectors<'py>(selection: &Bound<'py, PyAny>) -> PyResult<Bound<'py, PyDict>> {
156    let py = selection.py();
157    let selectors = if selection.is_truthy()? {
158        py.get_type::<PyDict>()
159            .call1((selection,))?
160            .cast_into::<PyDict>()?
161    } else {
162        PyDict::new(py)
163    };
164    let mut unexpected = selectors
165        .keys()
166        .iter()
167        .map(|key| key.extract::<String>())
168        .collect::<PyResult<Vec<_>>>()?;
169    unexpected.retain(|key| !SELECTORS.contains(&key.as_str()));
170    if !unexpected.is_empty() {
171        unexpected.sort();
172        return Err(PyValueError::new_err(format!(
173            "unknown MeasurementSet selectors: {}",
174            unexpected.join(", ")
175        )));
176    }
177    Ok(selectors)
178}

180/// The request that `casars.data.measurement_set_plot`'s arguments make.
181///
182/// `options` holds the keyword arguments other than the selection, by name.
183fn plot_request(
184    measurement_set: &Bound<'_, PyAny>,
185    preset: &Bound<'_, PyAny>,
186    selection: &Bound<'_, PyAny>,
187    options: &Bound<'_, PyDict>,
188) -> PyResult<MeasurementSetPlotRequest> {
189    let py = options.py();
190    let preset = parse_preset(preset)?;
191    let selectors = selectors(selection)?;
192    let dataset_path = py
193        .import("os")?
194        .call_method1("fspath", (measurement_set,))?
195        .extract::<String>()?;
196    let mut request = MeasurementSetPlotRequest::new(dataset_path, preset);
197    for (name, slot) in [
198        ("field", &mut request.field),
199        ("spw", &mut request.spw),
200        ("timerange", &mut request.timerange),
201        ("uvrange", &mut request.uvrange),
202        ("antenna", &mut request.antenna),
203        ("scan", &mut request.scan),
204        ("correlation", &mut request.correlation),
205        ("array", &mut request.array),
206        ("observation", &mut request.observation),
207        ("intent", &mut request.intent),
208        ("feed", &mut request.feed),
209        ("msselect", &mut request.msselect),
210    ] {
211        *slot = optional_text(selectors.get_item(name)?)?;
212    }
213    let option = |name: &str| {
214        options
215            .get_item(name)?
216            .ok_or_else(|| PyKeyError::new_err(name.to_string()))
217    };
218    request.data_column = option("data_column")?.extract::<String>()?;
219    request.color_by = optional_text(Some(option("color_by")?))?;
220    let avgchannel = Some(option("avgchannel")?)
221        .filter(|value| !value.is_none())
222        .map(|value| count(&value))
223        .transpose()?;
224    request.avgtime = Some(option("avgtime")?)
225        .filter(|value| !value.is_none())
226        .map(|value| value.extract::<f64>())
227        .transpose()?;
228    for (name, slot) in [
229        ("avgscan", &mut request.avgscan),
230        ("avgfield", &mut request.avgfield),
231        ("avgbaseline", &mut request.avgbaseline),
232        ("avgantenna", &mut request.avgantenna),
233        ("avgspw", &mut request.avgspw),
234        ("scalar", &mut request.scalar),
235    ] {
236        *slot = flag(name, &option(name)?)?;
237    }
238    request.iteraxis = optional_text(Some(option("iteraxis")?))?;
239    let max_plot_points = count(&option("max_plot_points")?)?;
240    request.avgchannel = avgchannel
241        .map(|value| platform_count(py, "avgchannel", value))
242        .transpose()?;
243    request.max_plot_points = usize::try_from(max_plot_points).unwrap_or(usize::MAX);
244    Ok(request)
245}

247fn axis_dict<'py>(py: Python<'py>, axis: PlotDocumentAxis) -> PyResult<Bound<'py, PyDict>> {
248    let dict = PyDict::new(py);
249    dict.set_item("id", axis.id)?;
250    dict.set_item("label", axis.label)?;
251    dict.set_item("unit", axis.unit)?;
252    dict.set_item("lower", axis.lower)?;
253    dict.set_item("upper", axis.upper)?;
254    Ok(dict)
255}

257fn layer_dict<'py>(py: Python<'py>, layer: PlotDocumentLayer) -> PyResult<Bound<'py, PyDict>> {
258    let dict = PyDict::new(py);
259    dict.set_item("title", layer.title)?;
260    dict.set_item("color_group", layer.color_group)?;
261    dict.set_item("y_axis_id", layer.y_axis_id)?;
262    dict.set_item("x", layer.x_values.into_pyarray(py))?;
263    dict.set_item("y", layer.y_values.into_pyarray(py))?;
264    let provenance = PyList::empty(py);
265    for point in layer.provenance {
266        let entry = PyDict::new(py);
267        entry.set_item("row", point.row)?;
268        entry.set_item("corr", point.corr)?;
269        entry.set_item("chan_start", point.chan_start)?;
270        entry.set_item("chan_end", point.chan_end)?;
271        provenance.append(entry)?;
272    }
273    dict.set_item("provenance", provenance)?;
274    Ok(dict)
275}

277fn panel_dict<'py>(
278    py: Python<'py>,
279    id: String,
280    title: String,
281    axes: Vec<PlotDocumentAxis>,
282    layers: Vec<PlotDocumentLayer>,
283) -> PyResult<Bound<'py, PyDict>> {
284    let dict = PyDict::new(py);
285    dict.set_item("id", id)?;
286    dict.set_item("title", title)?;
287    let axes = axes
288        .into_iter()
289        .map(|axis| axis_dict(py, axis))
290        .collect::<PyResult<Vec<_>>>()?;
291    dict.set_item("axes", axes)?;
292    let layers = layers
293        .into_iter()
294        .map(|layer| layer_dict(py, layer))
295        .collect::<PyResult<Vec<_>>>()?;
296    dict.set_item("layers", layers)?;
297    Ok(dict)
298}

300fn plot_dict(py: Python<'_>, plot: MeasurementSetPlot) -> PyResult<Bound<'_, PyDict>> {
301    let document = plot.document;
302    let panels = if document.panels.is_empty() {
303        vec![panel_dict(
304            py,
305            document.id,
306            document.title,
307            document.axes,
308            document.layers,
309        )?]
310    } else {
311        document
312            .panels
313            .into_iter()
314            .map(|panel| panel_dict(py, panel.id, panel.title, panel.axes, panel.layers))
315            .collect::<PyResult<Vec<_>>>()?
316    };
317    let dict = PyDict::new(py);
318    dict.set_item("title", plot.title)?;
319    dict.set_item("summary", plot.summary)?;
320    dict.set_item("header_lines", document.header_lines)?;
321    dict.set_item("show_legend", document.show_legend)?;
322    dict.set_item("panels", panels)?;
323    dict.set_item("measurement_set", plot.dataset_path)?;
324    dict.set_item("preset", preset_name(plot.preset))?;
325    dict.set_item("selection_summary", plot.selection_summary)?;
326    Ok(dict)
327}

329/// Build one MeasurementSet preset plot, releasing the GIL while it reads.
330#[pyfunction]
331pub(crate) fn build_measurement_set_plot<'py>(
332    py: Python<'py>,
333    measurement_set: &Bound<'py, PyAny>,
334    preset: &Bound<'py, PyAny>,
335    selection: &Bound<'py, PyAny>,
336    options: &Bound<'py, PyDict>,
337) -> PyResult<Bound<'py, PyDict>> {
338    let request = plot_request(measurement_set, preset, selection, options)?;
339    let plot = py
340        .detach(|| build_plot(&request))
341        .map_err(|error| reason_error(py, error.to_string()))?;
342    plot_dict(py, plot)
343}
