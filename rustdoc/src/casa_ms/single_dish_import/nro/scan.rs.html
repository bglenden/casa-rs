11const JST_SECONDS: f64 = 9.0 * 3600.0;
12
13pub(super) fn start_time(record: Record<'_>) -> MsResult<f64> {
14    let time = record.text(8, 24)?;
15    if !time.is_ascii() || time.len() < 14 || !time[..12].bytes().all(|byte| byte.is_ascii_digit())
16    {
17        return Err(invalid("scan timestamp"));
18    }
19    let calendar = format!(
20        "{}/{}/{}/{}:{}:{}",
21        &time[..4],
22        &time[4..6],
23        &time[6..8],
24        &time[8..10],
25        &time[10..12],
26        &time[12..],
27    );
28    let seconds_unit = Unit::new("s").map_err(|_| invalid("time unit"))?;
29    let time = Quantity::read(&calendar).map_err(|_| invalid("scan timestamp"))?;
30    let seconds = time
31        .get_value_in(&seconds_unit)
32        .map_err(|_| invalid("scan timestamp"))?;
33    Ok(seconds - JST_SECONDS)
34}

36pub(super) fn decode(record: Record<'_>, header: &Header, start: f64) -> MsResult<RawIntegration> {
37    let array = array(record, header)?;
38    let midpoint = start + 0.5 * header.interval;
39    let state = if record.text(32, 8)? == "ON" {
40        ("OBSERVE_TARGET#ON_SOURCE,POSITION_SWITCH", 1)
41    } else {
42        ("OBSERVE_TARGET#OFF_SOURCE,POSITION_SWITCH", 2)
43    };
44    Ok(RawIntegration {
45        row: Integration {
46            time: midpoint,
47            interval: header.interval,
48            scan: record.i32(4),
49            antenna: array.beam,
50            field: 0,
51            feed: 0,
52            window: array.window,
53            basis: Basis::Linear,
54            state,
55            direction: Direction {
56                position: [record.f64(56), record.f64(64)],
57                rate: [0.; 2],
58            },
59            weather: Some(Weather([
60                (f64::from(record.f32(124)) + 273.15) as f32,
61                record.f32(128),
62                record.f32(132),
63                record.f32(136),
64                record.f32(140),
65            ])),
66        },
67        plane: Plane {
68            index: array.plane,
69            data: record.spectrum(header.n_channels, array.scale),
70            flag: None,
71            flag_row: false,
72            system_temperature: vec![record.f32(148)],
73            calibration_temperature: Vec::new(),
74        },
75    })
76}

78fn array(record: Record<'_>, header: &Header) -> MsResult<Array> {
79    let number = record
80        .text(120, 4)?
81        .get(1..)
82        .and_then(|digits| digits.parse::<usize>().ok())
83        .and_then(|id| id.checked_sub(1))
84        .ok_or_else(|| invalid("scan array number"))?;
85    let array = header
86        .arrays
87        .get(number)
88        .and_then(|array| *array)
89        .ok_or_else(|| invalid("scan array number"))?;
90    let identifier = record.text(264, 8)?;
91    if identifier
92        .get(..2)
93        .and_then(|value| value.parse::<i32>().ok())
94        != Some(array.beam + 1)
95        || identifier.get(2..4) != Some(array.name)
96        || identifier
97            .get(4..6)
98            .and_then(|value| value.parse::<i32>().ok())
99            != Some(array.window + 1)
100    {
101        return Err(invalid("inconsistent scan array mapping"));
102    }
103    Ok(array)
104}
