11/// CASA's polynomial order, fitting method and error weighting controls.
12#[derive(Debug, Clone, Copy)]
13pub struct CalAnalysisFit<'a> {
14    /// AVERAGE, LINEAR or QUADRATIC, or the corresponding initial.
15    pub order: &'a str,
16    /// LSQ or ROBUST, or the corresponding initial.
17    pub method: &'a str,
18    /// Weight by the inverse square of each solution uncertainty.
19    pub weighted: bool,
20}
21
22#[derive(Clone, Copy)]
23pub(super) struct FitOptions {
24    degree: usize,
25    robust: bool,
26    weighted: bool,
27}
28
29impl FitOptions {
30    pub(super) fn parse(options: CalAnalysisFit<'_>) -> Result<Self, String> {
31        let degree = match initial(options.order, 'A') {
32            'A' => 0,
33            'L' => 1,
34            'Q' => 2,
35            _ => return Err("Invalid fit order".into()),
36        };
37        let robust = match initial(options.method, 'L') {
38            'L' => false,
39            'R' => true,
40            _ => return Err("Invalid fit type".into()),
41        };
42        Ok(Self {
43            degree,
44            robust,
45            weighted: options.weighted,
46        })
47    }
48
49    pub(super) fn record(
50        self,
51        x: &[f64],
52        values: &[f64],
53        errors: &[f64],
54        flags: &mut [bool],
55    ) -> [(&'static str, Value); 12] {
56        let fit = self.run(x, values, errors, flags);
57        let valid = fit.is_some();
58        let fit = fit.unwrap_or_default();
59        let order = ["AVERAGE", "LINEAR", "QUADRATIC"][self.degree];
60        let method = if self.robust { "ROBUST" } else { "LSQ" };
61        let weight = if self.weighted { "YES" } else { "NO" };
62        let string = |text: &str| Value::Scalar(ScalarValue::String(text.to_string()));
63        let float = |value: f64| Value::Scalar(ScalarValue::Float64(value));
64        let floats = |data: Vec<f64>| Value::Array(ArrayValue::from_f64_vec(data));
65        [
66            ("order", string(order)),
67            ("type", string(method)),
68            ("weight", string(weight)),
69            ("validFit", Value::Scalar(ScalarValue::Bool(valid))),
70            ("pars", floats(fit.coefficients)),
71            ("vars", floats(fit.variances)),
72            ("covars", floats(fit.covariances)),
73            ("model", floats(fit.model)),
74            ("res", floats(fit.residuals)),
75            ("redChi2", float(fit.reduced_chi_square)),
76            ("resMean", float(fit.residual_mean)),
77            ("resVar", float(fit.residual_variance)),
78        ]
79    }
80
81    fn run(self, x: &[f64], values: &[f64], errors: &[f64], flags: &mut [bool]) -> Option<Fit> {
82        if self.robust {
83            robust::trim(x, values, errors, flags, self.degree, self.weighted)?;
84        }
85        let active = x
86            .iter()
87            .zip(values)
88            .zip(errors)
89            .zip(flags.iter())
90            .filter_map(|(((&x, &y), &sigma), &flag)| (!flag).then_some((x, y, sigma)))
91            .collect::<Vec<_>>();
92        let dof = active.len().checked_sub(self.degree)?;
93        if dof == 0 {
94            return None;
95        }
96        let x_scale = scale(active.iter().map(|&(x, _, _)| x))?;
97        let y_scale = scale(active.iter().map(|&(_, y, _)| y))?;
98        let mut system = normal::System::new(self.degree + 1);
99        let samples = active
100            .into_iter()
101            .filter_map(|(x, y, error)| {
102                let sigma = if self.weighted { error } else { 1. } / y_scale;
103                (sigma != 0. && sigma != -1.).then_some((x / x_scale, y / y_scale, sigma))
104            })
105            .collect::<Vec<_>>();
106        for &(x, y, sigma) in &samples {
107            system.accumulate(x, y, sigma)?;
108        }
109        let solution = system.solve()?;
110        // Form the small residuals directly. Subtracting the corresponding
111        // normal-equation quadratic forms can lose their entire sum.
112        let metric = samples
113            .iter()
114            .map(|&(x, y, sigma)| ((y - polynomial(&solution.coefficients, x)) / sigma).powi(2))
115            .sum::<f64>()
116            / dof as f64;
117        let covariance_scale = if self.weighted { 1. } else { metric };
118        let factors = [y_scale, y_scale / x_scale, y_scale / (x_scale * x_scale)];
119        let coefficients = solution
120            .coefficients
121            .iter()
122            .zip(factors)
123            .map(|(&coefficient, factor)| coefficient * factor)
124            .collect::<Vec<_>>();
125        let mut variances = Vec::new();
126        let mut covariances = Vec::new();
127        for (row, values) in solution.covariance.iter().enumerate() {
128            variances.push(values[row] * factors[row].powi(2) * covariance_scale);
129            for column in row + 1..values.len() {
130                covariances
131                    .push(values[column] * factors[row] * factors[column] * covariance_scale);
132            }
133        }
134        Some(Fit::new(
135            x,
136            values,
137            coefficients,
138            variances,
139            covariances,
140            if self.weighted { metric } else { 1. },
141        ))
142    }
143}

145fn scale(values: impl Iterator<Item = f64>) -> Option<f64> {
146    let mut min = f64::INFINITY;
147    let mut max = f64::NEG_INFINITY;
148    for value in values {
149        if !value.is_finite() {
150            return None;
151        }
152        min = min.min(value);
153        max = max.max(value);
154    }
155    let scale = min.abs() + max.abs();
156    scale
157        .is_finite()
158        .then_some(if scale == 0. { 1. } else { scale })
159}

161#[derive(Default)]
162struct Fit {
163    coefficients: Vec<f64>,
164    variances: Vec<f64>,
165    covariances: Vec<f64>,
166    model: Vec<f64>,
167    residuals: Vec<f64>,
168    reduced_chi_square: f64,
169    residual_mean: f64,
170    residual_variance: f64,
171}
172
173impl Fit {
174    fn new(
175        x: &[f64],
176        values: &[f64],
177        coefficients: Vec<f64>,
178        variances: Vec<f64>,
179        covariances: Vec<f64>,
180        reduced_chi_square: f64,
181    ) -> Self {
182        let model = x
183            .iter()
184            .map(|&x| polynomial(&coefficients, x))
185            .collect::<Vec<_>>();
186        let residuals = values
187            .iter()
188            .zip(&model)
189            .map(|(y, fitted)| y - fitted)
190            .collect::<Vec<_>>();
191        let residual_mean = residuals.iter().sum::<f64>() / residuals.len() as f64;
192        let residual_variance = residuals
193            .iter()
194            .map(|value| (value - residual_mean).powi(2))
195            .sum::<f64>()
196            / residuals.len().saturating_sub(1).max(1) as f64;
197        Self {
198            coefficients,
199            variances,
200            covariances,
201            model,
202            residuals,
203            reduced_chi_square,
204            residual_mean,
205            residual_variance,
206        }
207    }
208}
209
210fn polynomial(coefficients: &[f64], x: f64) -> f64 {
211    coefficients
212        .iter()
213        .enumerate()
214        .map(|(power, coefficient)| coefficient * x.powi(power as i32))
215        .sum()
216}
