2//! One hypercube of a streamed `TiledShapeStMan` column, written a row at a
3//! time.
4//!
5//! A row fills its slot in every tile across the cell, in the band of tiles
6//! that holds it along the row axis. Tiles are numbered with the row axis
7//! slowest, so a band is contiguous in the cube's file. A cube whose band
8//! fits the writer's budget assembles each band in memory and writes it with
9//! one call when its last row arrives. Otherwise each tile row is written in
10//! place as it is pushed, a Bool tile row by reading back the bytes it
11//! shares with its neighbours. The cubes of a column share one tile-row
12//! scratch, held by the column's writer ([`ShapeCubeWorkingSet`]).
13

43/// The bytes a cube works in while it writes a row.
44///
45/// The encoded row and a Bool tile row's packed bytes live only while the
46/// row is written; the tile row is the writer's scratch, shared by every
47/// cube and sized for the largest. A column holds, at once, its largest
48/// encoded row, its largest tile row and its largest packed Bool tile row,
49/// which may come from different cubes ([`Self::max`], [`Self::bytes`]).
50#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
51pub(super) struct ShapeCubeWorkingSet {
52    /// One row's cell, encoded.
53    pub(super) row_bytes: usize,
54    /// One tile row of the cell.
55    pub(super) tile_row_bytes: usize,
56    /// One Bool tile row's packed bits and the byte they may share.
57    pub(super) packed_bool_bytes: usize,
58}
59
60impl ShapeCubeWorkingSet {
61    /// The larger of each part of `self` and `other`.
62    pub(super) fn max(self, other: Self) -> Self {
63        Self {
64            row_bytes: self.row_bytes.max(other.row_bytes),
65            tile_row_bytes: self.tile_row_bytes.max(other.tile_row_bytes),
66            packed_bool_bytes: self.packed_bool_bytes.max(other.packed_bool_bytes),
67        }
68    }
69
70    /// Every part, held at once.
71    pub(super) fn bytes(self) -> Option<usize> {
72        self.row_bytes
73            .checked_add(self.tile_row_bytes)?
74            .checked_add(self.packed_bool_bytes)
75    }
76}

78/// What a cube of one layout is: the bytes it works in per row, its tiles
79/// along each axis of the cube (the row axis last), and one tile's bytes.
80struct Geometry {
81    working_set: ShapeCubeWorkingSet,
82    tiles_per_dim: Vec<usize>,
83    tile_cell_nelem: usize,
84    bucket_size: usize,
85}
86
87impl Geometry {
88    /// The geometry of a cube of `layout`, which it checks: rows, cell
89    /// extents and a tile of the cell's rank and the row axis, none empty.
90    fn of(
91        layout: &StreamedTiledShapeCubeLayout,
92        value_type: StreamedTiledShapeValueType,
93    ) -> Result<Self, StorageError> {
94        if layout.row_count == 0 {
95            return Err(StorageError::FormatMismatch(format!(
96                "streamed TiledShape layout {:?} has no rows",
97                layout.cell_shape
98            )));
99        }
100        if layout.cell_shape.is_empty() || layout.cell_shape.contains(&0) {
101            return Err(StorageError::FormatMismatch(format!(
102                "streamed TiledShape cell dimensions must be positive: {:?}",
103                layout.cell_shape
104            )));
105        }
106        if layout.tile_shape.len() != layout.cell_shape.len() + 1 || layout.tile_shape.contains(&0)
107        {
108            return Err(StorageError::FormatMismatch(format!(
109                "streamed TiledShape tile rank must be cell rank + row axis; cell={:?} tile={:?}",
110                layout.cell_shape, layout.tile_shape
111            )));
112        }
113        let elem_size = value_type.elem_size();
114        let overflow = |what: &str| {
115            StorageError::FormatMismatch(format!("streamed TiledShape {what} overflows usize"))
116        };
117        let product = |extents: &[usize], what: &str| {
118            extents
119                .iter()
120                .try_fold(1usize, |acc, &dim| acc.checked_mul(dim))
121                .ok_or_else(|| overflow(what))
122        };
123        let cell_rank = layout.cell_shape.len();
124        let cell_nelem = product(&layout.cell_shape, "cell size")?;
125        let tile_cell_nelem = product(&layout.tile_shape[..cell_rank], "tile size")?;
126        let row_bytes = cell_nelem
127            .checked_mul(elem_size)
128            .ok_or_else(|| overflow("row byte size"))?;
129        let tile_row_bytes = tile_cell_nelem
130            .checked_mul(elem_size)
131            .ok_or_else(|| overflow("tile-row size"))?;
132        let packed_bool_bytes = if value_type == StreamedTiledShapeValueType::Bool {
133            tile_cell_nelem.div_ceil(8) + 1
134        } else {
135            0
136        };
137        let working_set = ShapeCubeWorkingSet {
138            row_bytes,
139            tile_row_bytes,
140            packed_bool_bytes,
141        };
142        working_set.bytes().ok_or_else(|| overflow("working set"))?;
143        let mut cube_shape = layout.cell_shape.clone();
144        cube_shape.push(layout.row_count);
145        let tiles_per_dim: Vec<usize> = cube_shape
146            .iter()
147            .zip(layout.tile_shape.iter())
148            .map(|(&cube, &tile)| cube.div_ceil(tile))
149            .collect();
150        product(&tiles_per_dim, "tile count")?;
151        let (bucket_size, _) =
152            compute_tile_layout(&[value_type.casacore_type()], &layout.tile_shape);
153        Ok(Self {
154            working_set,
155            tiles_per_dim,
156            tile_cell_nelem,
157            bucket_size,
158        })
159    }
160
161    /// Tiles across the cell: one band's.
162    fn non_row_tile_count(&self) -> usize {
163        self.tiles_per_dim[..self.tiles_per_dim.len() - 1]
164            .iter()
165            .product()
166    }
167}

169/// The row map of `runs` runs of rows: an entry of three `u32`s for each
170/// run of consecutive rows in one cube.
171fn row_map_bytes(runs: usize) -> usize {
172    runs.saturating_mul(3 * size_of::<u32>())
173}
174
175/// The budget from which a column writer of `rows` rows, working in
176/// `working` bytes, assembles its cubes' bands (`bands` bytes) in memory:
177/// both, beside the largest row map the rows can need.
178pub(super) fn buffered_bytes(working: usize, bands: usize, rows: usize) -> usize {
179    working
180        .saturating_add(bands)
181        .saturating_add(row_map_bytes(rows))
182}

184impl StreamingTiledShapeWriter {
185    /// The bytes a writer of `value_type` cells in `layouts` works in: the
186    /// least budget it writes its rows in, when they come as at most `runs`
187    /// runs of consecutive rows of one cell shape (undefined rows one more
188    /// shape), which is its largest encoded row, tile row and packed Bool
189    /// tile row beside the row map of `runs` entries; and the budget from
190    /// which it assembles its bands in memory ([`buffered_bytes`]), which it
191    /// then holds.
192    ///
193    /// # Errors
194    ///
195    /// A layout the writer refuses, or sizes that overflow.
196    pub fn resident_bytes(
197        value_type: StreamedTiledShapeValueType,
198        layouts: &[StreamedTiledShapeCubeLayout],
199        runs: usize,
200    ) -> Result<(usize, usize), StorageError> {
201        let mut working_set = ShapeCubeWorkingSet::default();
202        let (mut bands, mut rows) = (0usize, 0usize);
203        for layout in layouts {
204            let geometry = Geometry::of(layout, value_type)?;
205            working_set = working_set.max(geometry.working_set);
206            bands = bands.saturating_add(geometry.non_row_tile_count() * geometry.bucket_size);
207            rows = rows.saturating_add(layout.row_count);
208        }
209        let working = working_set.bytes().ok_or_else(|| {
210            StorageError::FormatMismatch("streamed TiledShape working set overflows usize".into())
211        })?;
212        Ok((
213            working.saturating_add(row_map_bytes(runs.min(rows))),
214            buffered_bytes(working, bands, rows),
215        ))
216    }
217}

219impl DirectShapeCubeWriter {
220    /// Create the cube's file at `temp_data_path`, sized for every planned
221    /// row, and return the writer with the bytes it works in per row.
222    pub(super) fn create(
223        temp_data_path: PathBuf,
224        layout: StreamedTiledShapeCubeLayout,
225        value_type: StreamedTiledShapeValueType,
226    ) -> Result<(Self, ShapeCubeWorkingSet), StorageError> {
227        let Geometry {
228            working_set,
229            tiles_per_dim,
230            tile_cell_nelem,
231            bucket_size,
232        } = Geometry::of(&layout, value_type)?;
233        let cell_rank = layout.cell_shape.len();
234        let cell_tiles_per_dim = &tiles_per_dim[..cell_rank];
235        let non_row_tile_count = cell_tiles_per_dim.iter().product::<usize>();
236        let tile_plans = (0..non_row_tile_count)
237            .map(|tile_index| {
238                let tile_pos = linear_to_nd(tile_index, cell_tiles_per_dim);
239                let cell_start: Vec<usize> = tile_pos
240                    .iter()
241                    .enumerate()
242                    .map(|(axis, &pos)| pos * layout.tile_shape[axis])
243                    .collect();
244                let actual_extent: Vec<usize> = cell_start
245                    .iter()
246                    .enumerate()
247                    .map(|(axis, &start)| {
248                        std::cmp::min(layout.tile_shape[axis], layout.cell_shape[axis] - start)
249                    })
250                    .collect();
251                StreamedPrimitiveTilePlan {
252                    cell_start,
253                    actual_extent,
254                }
255            })
256            .collect();
257        let file_bytes = tiles_per_dim
258            .iter()
259            .product::<usize>()
260            .checked_mul(bucket_size)
261            .ok_or_else(|| {
262                StorageError::FormatMismatch("streamed TiledShape file size overflows usize".into())
263            })?;
264        let file = OpenOptions::new()
265            .create(true)
266            .truncate(true)
267            .read(true)
268            .write(true)
269            .open(&temp_data_path)?;
270        file.set_len(file_bytes as u64)?;
271        let rows_per_tile = *layout.tile_shape.last().expect("validated tile row axis");
272        let cube = Self {
273            temp_data_path,
274            file,
275            cell_shape: layout.cell_shape,
276            row_count: layout.row_count,
277            rows_written: 0,
278            tile_shape: layout.tile_shape,
279            tile_plans,
280            non_row_tile_count,
281            rows_per_tile,
282            tile_cell_nelem,
283            bucket_size,
284            band: Vec::new(),
285        };
286        Ok((cube, working_set))
287    }
288
289    /// Bytes of one band: every tile across the cell, one tile along the row
290    /// axis.
291    pub(super) fn band_bytes(&self) -> usize {
292        self.non_row_tile_count * self.bucket_size
293    }
294
295    /// Assemble each band in memory from now on.
296    pub(super) fn buffer_bands(&mut self) {
297        self.band = vec![0; self.band_bytes()];
298    }
299
300    /// Write the next row's cell, `encoded` in the cube's cell order, one
301    /// tile row at a time through `tile_row`, the column's scratch of at
302    /// least this cube's tile-row bytes.
303    pub(super) fn write_row(
304        &mut self,
305        value_type: StreamedTiledShapeValueType,
306        encoded: &[u8],
307        tile_row: &mut [u8],
308    ) -> Result<(), StorageError> {
309        let elem_size = value_type.elem_size();
310        let cell_rank = self.cell_shape.len();
311        let row_tile_index = self.rows_written / self.rows_per_tile;
312        let row_in_tile = self.rows_written % self.rows_per_tile;
313        let tile_row_bytes = self.tile_cell_nelem * elem_size;
314        let tile_row = &mut tile_row[..tile_row_bytes];
315        for (cell_tile_index, plan) in self.tile_plans.iter().enumerate() {
316            tile_row.fill(0);
317            copy_cube_to_tile(
318                encoded,
319                &self.cell_shape,
320                tile_row,
321                &self.tile_shape[..cell_rank],
322                &plan.cell_start,
323                &plan.actual_extent,
324                elem_size,
325            );
326            let band_offset = cell_tile_index * self.bucket_size;
327            let bucket_start = row_tile_index * self.band_bytes() + band_offset;
328            if value_type == StreamedTiledShapeValueType::Bool {
329                let bit_offset = row_in_tile * self.tile_cell_nelem;
330                let byte_offset = bit_offset / 8;
331                let bit_in_byte = bit_offset % 8;
332                let byte_count = (bit_in_byte + self.tile_cell_nelem).div_ceil(8);
333                if self.band.is_empty() {
334                    let mut packed = vec![0u8; byte_count];
335                    self.file
336                        .seek(SeekFrom::Start((bucket_start + byte_offset) as u64))?;
337                    self.file.read_exact(&mut packed)?;
338                    write_bool_bits_from_bytes(&mut packed, bit_in_byte, tile_row);
339                    self.file
340                        .seek(SeekFrom::Start((bucket_start + byte_offset) as u64))?;
341                    self.file.write_all(&packed)?;
342                } else {
343                    let start = band_offset + byte_offset;
344                    write_bool_bits_from_bytes(
345                        &mut self.band[start..start + byte_count],
346                        bit_in_byte,
347                        tile_row,
348                    );
349                }
350            } else {
351                let row_offset = row_in_tile * tile_row_bytes;
352                if self.band.is_empty() {
353                    self.file
354                        .seek(SeekFrom::Start((bucket_start + row_offset) as u64))?;
355                    self.file.write_all(tile_row)?;
356                } else {
357                    let start = band_offset + row_offset;
358                    self.band[start..start + tile_row_bytes].copy_from_slice(tile_row);
359                }
360            }
361        }
362        self.rows_written += 1;
363        if !self.band.is_empty() && self.rows_written.is_multiple_of(self.rows_per_tile) {
364            self.write_band(row_tile_index)?;
365        }
366        Ok(())
367    }
368
369    /// Write the last band if rows ended inside it. Its rows past the last
370    /// are zero, as they are in the file.
371    pub(super) fn finish_band(&mut self) -> Result<(), StorageError> {
372        if !self.band.is_empty() && !self.rows_written.is_multiple_of(self.rows_per_tile) {
373            self.write_band(self.rows_written / self.rows_per_tile)?;
374        }
375        Ok(())
376    }
377
378    fn write_band(&mut self, row_tile_index: usize) -> Result<(), StorageError> {
379        self.file
380            .seek(SeekFrom::Start((row_tile_index * self.band.len()) as u64))?;
381        self.file.write_all(&self.band)?;
382        self.band.fill(0);
383        Ok(())
384    }

403    /// The pushes: `Some(0)` a row of the first shape, `Some(1)` of the
404    /// second, `None` an undefined row.
405    const ORDER: [Option<usize>; 14] = [
406        Some(0),
407        Some(0),

422    fn write(
423        dir: &std::path::Path,
424        name: &str,
425        value_type: StreamedTiledShapeValueType,
426        budget: usize,
427    ) -> (StreamedTiledShapeColumn, Vec<Vec<u8>>) {
428        let mut writer =
429            StreamingTiledShapeWriter::create(dir.join(name), value_type, layouts(), budget, true)
430                .expect("create writer");
431        let shapes = [[2usize, 5], [2, 4]];
432        for (row, shape) in ORDER.iter().enumerate() {
433            let Some(shape) = shape else {
434                writer.push_undefined_row().expect("undefined row");
435                continue;
436            };
437            let cell = shapes[*shape];
438            let samples = cell[0] * cell[1];
439            if value_type == StreamedTiledShapeValueType::Bool {
440                let values = (0..samples)
441                    .map(|sample| (row * 7 + sample * 3) % 5 < 2)
442                    .collect::<Vec<_>>();
443                writer.push_bool_row(&cell, &values).expect("bool row");
444            } else {
445                let values = (0..samples)
446                    .map(|sample| Complex32::new(row as f32, sample as f32 + 0.5))
447                    .collect::<Vec<_>>();
448                writer
449                    .push_complex32_row(&cell, &values)
450                    .expect("complex row");
451            }
452        }
453        let column = writer.finish().expect("finish");
454        let files = column
455            .cubes
456            .iter()
457            .map(|cube| std::fs::read(&cube.temp_data_path).expect("cube file"))
458            .collect();
459        (column, files)
460    }

462    #[test]
463    fn band_buffered_cubes_write_the_files_of_in_place_rows() {
464        let dir = tempfile::tempdir().expect("tempdir");
465        for value_type in [
466            StreamedTiledShapeValueType::Complex32,
467            StreamedTiledShapeValueType::Bool,
468        ] {
469            // A budget that holds the working set and the largest row map,
470            // but not the bands, keeps every tile row in place.
471            let elem = value_type.elem_size();
472            let packed = if value_type == StreamedTiledShapeValueType::Bool {
473                6usize.div_ceil(8) + 1
474            } else {
475                0
476            };
477            let working = 10 * elem + 6 * elem + packed;
478            let in_place_budget = working + ORDER.len() * 3 * size_of::<u32>();
479            let (direct, direct_files) = write(
480                dir.path(),
481                &format!("{value_type:?}.direct"),
482                value_type,
483                in_place_budget,
484            );
485            let (banded, banded_files) = write(
486                dir.path(),
487                &format!("{value_type:?}.banded"),
488                value_type,
489                1 << 20,
490            );
491            assert!(
492                banded.max_resident_bytes() > direct.max_resident_bytes(),
493                "{value_type:?}: the large budget buffers bands"
494            );
495            assert_eq!(direct_files.len(), 2);
496            assert_eq!(direct_files, banded_files, "{value_type:?} cube files");
497            assert_eq!(direct.row_map, banded.row_map);
498            assert_eq!(direct.cube_map, banded.cube_map);
499            assert_eq!(direct.pos_map, banded.pos_map);
500        }
501    }

503    /// [`StreamingTiledShapeWriter::resident_bytes`] states the least
504    /// budget a writer is created with and the budget from which it
505    /// buffers its bands.
506    #[test]
507    fn resident_bytes_are_the_least_and_the_band_buffering_budgets() {
508        let dir = tempfile::tempdir().expect("tempdir");
509        for value_type in [
510            StreamedTiledShapeValueType::Complex32,
511            StreamedTiledShapeValueType::Bool,
512        ] {
513            let (working, buffered) =
514                StreamingTiledShapeWriter::resident_bytes(value_type, &layouts(), 0)
515                    .expect("bytes");
516            let refused = StreamingTiledShapeWriter::create(
517                dir.path().join(format!("{value_type:?}.less")),
518                value_type,
519                layouts(),
520                working - 1,
521                true,
522            );
523            assert!(refused.is_err(), "{value_type:?}: below the working set");
524            let peak = |name: &str, budget| {
525                write(dir.path(), name, value_type, budget)
526                    .0
527                    .max_resident_bytes()
528            };
529            let (below, at) = (
530                peak(&format!("{value_type:?}.below"), buffered - 1),
531                peak(&format!("{value_type:?}.at"), buffered),
532            );
533            assert!(below < at, "{value_type:?}: bands buffer from {buffered}");
534            assert!(
535                at <= buffered,
536                "{value_type:?}: held {at}, beyond {buffered}"
537            );
538        }

541    /// The least budget holds the row map of the runs it is sized for:
542    /// [`ORDER`]'s rows come as twelve runs of one shape, which a writer of
543    /// the least budget for twelve writes and one for eleven refuses.
544    #[test]
545    fn the_least_budget_holds_the_row_map_of_its_runs() {
546        let dir = tempfile::tempdir().expect("tempdir");
547        let value_type = StreamedTiledShapeValueType::Complex32;
548        let runs = 1 + ORDER.windows(2).filter(|pair| pair[0] != pair[1]).count();
549        assert_eq!(runs, 12);
550        let push_all = |name: &str, budget: usize| {
551            let mut writer = StreamingTiledShapeWriter::create(
552                dir.path().join(name),
553                value_type,
554                layouts(),
555                budget,
556                true,
557            )?;
558            let shapes = [[2usize, 5], [2, 4]];
559            for shape in ORDER {
560                match shape {
561                    Some(shape) => {
562                        let cell = shapes[shape];
563                        let values = vec![Complex32::new(1.0, 0.0); cell[0] * cell[1]];
564                        writer.push_complex32_row(&cell, &values)?;
565                    }
566                    None => writer.push_undefined_row()?,
567                }
568            }
569            writer.finish().map(|_| ())
570        };
571        let least = |runs| {
572            StreamingTiledShapeWriter::resident_bytes(value_type, &layouts(), runs)
573                .expect("bytes")
574                .0
575        };
576        push_all("enough", least(runs)).expect("every run fits");
577        assert!(push_all("short", least(runs - 1)).is_err(), "a run short");
578    }
