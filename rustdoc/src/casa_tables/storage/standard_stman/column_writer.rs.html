2//! The StandardStMan file writer for any mix of columns, with bounded
3//! memory.
4//!
5//! The columns are written one after another, each over all its rows, which
6//! fixes where their shared storage goes: indirect arrays and records are
7//! appended to the array file (`table.fNi`) and long strings to the string
8//! buckets in that column-major order. Each column's slice of a data bucket
9//! is written at its place in the file as soon as it is full, and each
10//! string bucket once the next one starts, even within one long string;
11//! the index is generated as it is streamed into its buckets
12//! ([`super::index`]). So the writer holds one slice, one string bucket,
13//! one index bucket and the array file's buffer, whatever the row count and
14//! however long a string, and while it stores a string array cell, that
15//! cell's encoding (its elements' lengths and bytes). The file is laid out
16//! as casacore's `SSMBase` reads it: the 512-byte header, the data buckets,
17//! the string buckets, then the index buckets.
18

35/// The cells a StandardStMan file is written from, by column and row.
36pub(super) trait SsmCells {
37    /// Rows to write.
38    fn row_count(&self) -> usize;
39
40    /// The cell of `column` in `row`, `None` when it is undefined.
41    fn cell(&self, column: usize, row: usize) -> Result<Option<Cow<'_, Value>>, StorageError>;
42}
43
44impl SsmCells for ColumnSources<'_> {
45    fn row_count(&self) -> usize {
46        ColumnSources::row_count(self)
47    }
48
49    fn cell(&self, column: usize, row: usize) -> Result<Option<Cow<'_, Value>>, StorageError> {
50        ColumnSources::cell(self, column, row)
51    }
52}

54/// Row records as cells: each column's field, found at its index in the
55/// record when `field_indices` gives one and by name otherwise.
56pub(super) struct RowCells<'r> {
57    rows: &'r [RecordValue],
58    field_indices: Option<&'r [usize]>,
59    names: Vec<&'r str>,
60}
61
62impl<'r> RowCells<'r> {
63    pub(super) fn new(
64        rows: &'r [RecordValue],
65        field_indices: Option<&'r [usize]>,
66        col_descs: &'r [ColumnDescContents],
67    ) -> Self {
68        Self {
69            rows,
70            field_indices,
71            names: col_descs
72                .iter()
73                .map(|desc| desc.col_name.as_str())
74                .collect(),
75        }
76    }
77}
78
79impl SsmCells for RowCells<'_> {
80    fn row_count(&self) -> usize {
81        self.rows.len()
82    }
83
84    fn cell(&self, column: usize, row: usize) -> Result<Option<Cow<'_, Value>>, StorageError> {
85        let name = self.names[column];
86        let record = self.rows.get(row).ok_or_else(|| {
87            StorageError::FormatMismatch(format!(
88                "SSM row writer missing row {row} for column {name}"
89            ))
90        })?;
91        let fields = record.fields();
92        let field = self
93            .field_indices
94            .and_then(|indices| indices.get(column))
95            .and_then(|&index| fields.get(index))
96            .filter(|field| field.name == name)
97            .or_else(|| fields.iter().find(|field| field.name == name));
98        Ok(field.map(|field| Cow::Borrowed(&field.value)))
99    }
100}

102/// Bits a row of each column takes in a data bucket: the bits of its
103/// elements; 12 bytes for a variable-length string's reference; 8 bytes for
104/// an indirect cell's array-file offset.
105fn column_bits(col_descs: &[ColumnDescContents]) -> Vec<usize> {
106    col_descs
107        .iter()
108        .map(|desc| {
109            if is_ssm_fixed_string(desc) {
110                desc.max_length as usize * ssm_cell_element_count(desc) * 8
111            } else if is_ssm_variable_string(desc) {
112                96
113            } else if is_ssm_indirect(desc) {
114                64
115            } else {
116                ssm_cell_element_count(desc) * canonical_element_size(desc.data_type).1
117            }
118        })
119        .collect()
120}

122/// Write a string bucket at its place in the file: bucket `number` of
123/// buckets of `bucket_bytes` bytes.
124fn write_string_bucket(
125    file: &mut File,
126    (number, bucket_bytes): (usize, usize),
127    bucket: &[u8],
128) -> Result<(), StorageError> {
129    file.seek(SeekFrom::Start(bucket_offset(number, bucket_bytes)))?;
130    file.write_all(bucket)?;
131    Ok(())
132}

134/// Write a StandardStMan data file from `cells`, column by column, and
135/// return the DM data blob for `table.dat`.
136///
137/// The bytes are those of a writer that assembles every bucket in memory
138/// and writes the columns in order.
139pub(super) fn write_ssm_file_columns(
140    file_path: &Path,
141    col_descs: &[ColumnDescContents],
142    cells: &impl SsmCells,
143    big_endian: bool,
144) -> Result<Vec<u8>, StorageError> {
145    let nrrow = cells.row_count();
146    let ncol = col_descs.len();
147    // The shared array file of indirect columns (SSM's version 0, without
148    // reference counts).
149    let mut array_writer = if col_descs.iter().any(is_ssm_indirect) {
150        let mut path = file_path.as_os_str().to_os_string();
151        path.push("i");
152        Some(StManArrayFileWriter::create(
153            &PathBuf::from(path),
154            big_endian,
155            0,
156        )?)
157    } else {
158        None
159    };
160    let bits = column_bits(col_descs);
161    let (rows_per_bucket, bucket_size) = new_ssm_bucket_layout(&bits, nrrow);
162    let bucket_rows = rows_per_bucket as usize;
163    let bucket_bytes = bucket_size as usize;
164    let mut column_offsets = Vec::with_capacity(ncol);
165    let mut offset = 0u32;
166    for &column_bits in &bits {
167        column_offsets.push(offset);
168        offset += (bucket_rows * column_bits).div_ceil(8) as u32;
169    }
170    let nr_data_buckets = nrrow.div_ceil(bucket_rows);
171
172    let mut file = File::create(file_path)?;
173    let mut strings = StringBuckets::after(nr_data_buckets);
174    let mut slice = Vec::new();
175    for (column, desc) in col_descs.iter().enumerate() {
176        let indirect = is_ssm_indirect(desc);
177        let nrelem = if indirect {
178            1
179        } else {
180            ssm_cell_element_count(desc)
181        };
182        for bucket in 0..nr_data_buckets {
183            slice.clear();
184            slice.resize((bucket_rows * bits[column]).div_ceil(8), 0);
185            let first_row = bucket * bucket_rows;
186            for row in first_row..nrrow.min(first_row + bucket_rows) {
187                let value = cells.cell(column, row)?;
188                let row_in_bucket = row - first_row;
189                if indirect {
190                    let writer = array_writer
191                        .as_mut()
192                        .expect("indirect columns have an array file");
193                    let offset = write_indirect_cell(writer, desc, value.as_deref())?;
194                    let cell = &mut slice[row_in_bucket * 8..];
195                    if big_endian {
196                        write_i64_be(cell, offset);
197                    } else {
198                        write_i64_le(cell, offset);
199                    }
200                } else {
201                    write_cell_to_bucket(
202                        &mut slice,
203                        0,
204                        row_in_bucket,
205                        desc,
206                        desc.data_type,
207                        nrelem,
208                        value.as_deref(),
209                        &mut |data| {
210                            strings.stream(data, bucket_bytes, |number, bucket| {
211                                write_string_bucket(
212                                    &mut file,
213                                    (nr_data_buckets + number, bucket_bytes),
214                                    bucket,
215                                )
216                            })
217                        },
218                        nr_data_buckets,
219                        big_endian,
220                    )?;
221                }
222            }
223            file.seek(SeekFrom::Start(
224                bucket_offset(bucket, bucket_bytes) + u64::from(column_offsets[column]),
225            ))?;
226            file.write_all(&slice)?;
227        }
228    }
229    // The open string bucket, the last.
230    strings.release(|number, bucket| {
231        write_string_bucket(&mut file, (nr_data_buckets + number, bucket_bytes), bucket)
232    })?;
233    if let Some(writer) = array_writer.as_mut() {
234        writer.finish()?;
235    }
236    let nr_string_buckets = strings.len();
237    write_index_and_header(
238        &mut file,
239        IndexLayout {
240            nrrow,
241            ncol,
242            rows_per_bucket,
243            bucket_size,
244            nr_data_buckets,
245            nr_string_buckets,
246        },
247        big_endian,
248    )?;
249    serialize_ssm_dm_blob("SSM", &column_offsets, ncol)
250}

252/// Append an indirect cell to the array file and return its offset there:
253/// a record as its serialized bytes, an array as itself, and 0 for an
254/// undefined cell.
255fn write_indirect_cell(
256    writer: &mut StManArrayFileWriter,
257    desc: &ColumnDescContents,
258    value: Option<&Value>,
259) -> Result<i64, StorageError> {
260    match value {
261        Some(Value::Record(record)) if desc.is_record() => {
262            let bytes = super::super::table_control::serialize_record_to_uchar(record)?;
263            let payload = Value::Array(ArrayValue::UInt8(
264                ArrayD::from_shape_vec(IxDyn(&[bytes.len()]), bytes).map_err(|err| {
265                    StorageError::FormatMismatch(format!(
266                        "SSM record column '{}': invalid byte payload shape: {err}",
267                        desc.col_name
268                    ))
269                })?,
270            ));
271            writer.write_array(&payload, CasacoreDataType::TpUChar)
272        }
273        Some(value @ Value::Array(_)) if !desc.is_record() => writer.write_array(
274            value,
275            CasacoreDataType::from_primitive_type(desc.require_primitive_type()?, false),
276        ),
277        Some(Value::Array(_) | Value::Record(_)) => Err(StorageError::FormatMismatch(format!(
278            "SSM indirect column '{}' received incompatible value type",
279            desc.col_name
280        ))),
281        _ => Ok(0),
282    }
283}

441            .expect("read dir")
442            .map(|entry| {

445                    entry.file_name().to_string_lossy().to_string(),

502    /// The string buckets lie between the data and index buckets; one that
503    /// links to a next bucket is full, every one states its used and unused
504    /// bytes, and the header names the last, where casacore appends the
505    /// next string (`SSMBase::itsLastStringBucket`).
506    #[test]
507    fn string_buckets_are_linked_and_the_header_names_the_last() {
508        let root = tempfile::tempdir().expect("tempdir");
509        let col_descs = schema_columns()
510            .iter()
511            .map(ColumnDescContents::from_column_schema)
512            .collect::<Vec<_>>();
513        let records = (0..1500).map(row).collect::<Vec<_>>();
514        let path = root.path().join("table.f0");
515        write_ssm_file_columns(
516            &path,
517            &col_descs,
518            &RowCells::new(&records, None, &col_descs),
519            false,
520        )
521        .expect("column writer");
522        let mut file = File::open(&path).expect("open");
523        let header = super::super::parse_ssm_header(&mut file).expect("header");
524        let indices = super::super::parse_ssm_indices(&mut file, &header).expect("index");
525        // The data buckets are the index's; the string buckets follow.
526        let first = indices[0].bucket_number.len() as i32;
527        let last = header.last_string_bucket;
528        assert_eq!(last, header.first_idx_bucket - 1);
529        assert!(last > first, "several string buckets");
530        let capacity = header.bucket_size as i32 - 16;
531        let ints = |bucket: &[u8], at: usize| {
532            i32::from_be_bytes(bucket[at..at + 4].try_into().expect("four bytes"))
533        };
534        let mut linked = 0;
535        for number in first..=last {
536            let bucket =
537                super::super::read_bucket(&mut file, &header, number as u32).expect("bucket");
538            let (used, unused, next) = (ints(&bucket, 4), ints(&bucket, 8), ints(&bucket, 12));
539            assert_eq!(used + unused, capacity, "bucket {number}");
540            if next >= 0 {
541                assert_eq!(next, number + 1, "bucket {number} links the next");
542                assert_eq!(used, capacity, "bucket {number} links on, so it is full");
543                linked += 1;
544            }
545            if number == last {
546                assert_eq!(next, -1, "the last string bucket links nowhere");
547            }
548        }
549        assert!(linked >= 2, "long strings continue in linked buckets");
550    }
