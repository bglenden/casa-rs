24const COPY_BLOCK_PIXELS: usize = 1_048_576;
25
26struct AddAxes<'a> {
27    outfile: &'a str,
28    direction: bool,
29    spectral: bool,
30    stokes: &'a str,
31    linear: bool,
32    tabular: bool,
33    overwrite: bool,
34    silent: bool,
35}
36
37impl AddAxes<'_> {
38    fn coordinates(&self, record: &mut RecordValue, ndim: usize) -> PyResult<CoordinateSystem> {
39        let expected = validate_coordinate_indices(record)?;
40        let mut coordinates = CoordinateSystem::from_record(record).map_err(raised)?;
41        let original_count = coordinates.n_coordinates();
42        if original_count != expected || coordinates.n_pixel_axes() != ndim {
43            return Err(raised("Image coordinates do not describe all pixel axes"));
44        }
45        // New image records use CASA's typed field names. Translate the
46        // older generic spelling without rebuilding coordinate metadata.
47        for placement in CoordinateSystem::record_coordinates(record).map_err(raised)? {
48            if let Some(index) = placement.field.strip_prefix("coordinate")
49                && let Some(Value::Record(axis)) = record.get(&placement.field)
50                && let Some(Value::Scalar(ScalarValue::String(kind))) = axis.get("coordinate_type")
51            {
52                let native_field = format!("{}{index}", kind.to_ascii_lowercase());
53                let axis = axis.clone();
54                record.remove(&placement.field);
55                record.upsert(native_field, axis.into());
56            }
57        }
58        let added: [Option<CoordinateModel>; 5] = [
59            self.direction
60                .then(|| defaults::direction_coordinate().into()),
61            self.spectral
62                .then(|| defaults::spectral_coordinate().into()),
63            (!self.stokes.is_empty()).then(|| {
64                let stokes = StokesType::from_name(self.stokes).unwrap_or(StokesType::Undefined);
65                StokesCoordinate::new(vec![stokes]).into()
66            }),
67            self.linear
68                .then(|| defaults::linear_coordinate(&["Axis1".into()], &[]).into()),
69            self.tabular
70                .then(|| TabularCoordinate::linear(0.0, 1.0, 0.0, "Tabular", "").into()),
71        ];
72        for coordinate in added.into_iter().flatten() {
73            let kind = coordinate.coordinate_type();
74            if coordinates.find_coordinate(kind).is_some() {
75                if !self.silent {
76                    return Err(raised(format!("Image already contains a {kind}Coordinate")));
77                }
78            } else {
79                append_coordinate(&mut coordinates, record, coordinate)?;
80            }
81        }
82        if coordinates.n_coordinates() == original_count && !self.silent {
83            return Err(raised("Exception: No degenerate axes specified."));
84        }
85        Ok(coordinates)
86    }
87
88    fn history(&self) -> String {
89        format!(
90            "ia.adddegaxes(outfile={:?}, direction={}, spectral={}, stokes={:?}, linear={}, tabular={}, overwrite={}, silent={})",
91            self.outfile,
92            self.direction,
93            self.spectral,
94            self.stokes,
95            self.linear,
96            self.tabular,
97            self.overwrite,
98            self.silent
99        )
100    }
101}

103fn expand<T>(mut block: ArrayD<T>, ndim: usize) -> ArrayD<T> {
104    while block.ndim() < ndim {
105        let axis = block.ndim();
106        block = block.insert_axis(Axis(axis));
107    }
108    block
109}

111fn copy_pixels<T: ImagePixel>(
112    source: &dyn ImageInterface<T>,
113    input_shape: &[usize],
114    output: &mut Storage<T>,
115    output_ndim: usize,
116    read: impl Fn(&dyn ImageInterface<T>, &[usize], &[usize], &[usize]) -> PyResult<ArrayD<T>>,
117) -> PyResult<()> {
118    let mut capacity = COPY_BLOCK_PIXELS;
119    let chunk_shape: Vec<_> = input_shape
120        .iter()
121        .map(|length| {
122            let extent = capacity.min(*length).max(1);
123            capacity /= extent;
124            extent
125        })
126        .collect();
127    let names = source.mask_names();
128    for name in &names {
129        validate_name(name)?;
130    }
131    let stride = vec![1; input_shape.len()];
132    // Walk the first pixel axis fastest, matching the stored array order.
133    let grid: Vec<_> = input_shape
134        .iter()
135        .zip(&chunk_shape)
136        .rev()
137        .map(|(&length, &extent)| length.div_ceil(extent))
138        .collect();
139    for index in indices(IxDyn(&grid)) {
140        let start: Vec<_> = index
141            .slice()
142            .iter()
143            .rev()
144            .zip(&chunk_shape)
145            .map(|(index, extent)| index * extent)
146            .collect();
147        let lengths: Vec<_> = input_shape
148            .iter()
149            .zip(&start)
150            .zip(&chunk_shape)
151            .map(|((&length, &start), &extent)| extent.min(length - start))
152            .collect();
153        let block = read(source, &start, &lengths, &stride)?;
154        let mut position = start.clone();
155        position.resize(output_ndim, 0);
156        let pixels = expand(block, output_ndim);
157        with_storage!(output, image => image.put_slice(&pixels, &position)).map_err(raised)?;
158        for name in &names {
159            let block = source
160                .get_named_mask_slice(name, &start, &lengths, &stride)
161                .map_err(raised)?;
162            let block = expand(block, output_ndim);
163            with_storage!(output, image => image.put_mask_slice(name, &block, &position))
164                .map_err(raised)?;
165        }
166    }
167    if let Some(default) = source.default_mask_name() {
168        with_storage!(output, image => image.set_default_mask(&default)).map_err(raised)?;
169    }
170    Ok(())
171}

173fn native_pixels<T: ImagePixel>(
174    source: &dyn ImageInterface<T>,
175    start: &[usize],
176    lengths: &[usize],
177    stride: &[usize],
178) -> PyResult<ArrayD<T>> {
179    source.get_slice(start, lengths, stride).map_err(raised)
180}

182fn copy_storage<T: ImagePixel>(
183    source: &Storage<T>,
184    path: Option<&Path>,
185    input_shape: &[usize],
186    shape: &[usize],
187    coordinates: &CoordinateSystem,
188    read: impl Fn(&dyn ImageInterface<T>, &[usize], &[usize], &[usize]) -> PyResult<ArrayD<T>>,
189) -> PyResult<Storage<T>> {
190    if let Storage::Paged(image) = source
191        && let Some(groups) = image.table().keywords().get("ATTRGROUPS")
192        && !matches!(groups, Value::Record(groups) if groups.is_empty())
193    {
194        return not_yet_implemented_in("image", "adddegaxes with attribute groups", 1625);
195    }
196    let mut output =
197        Storage::create(shape.to_vec(), coordinates.clone(), path, None).map_err(raised)?;
198    with_storage!(source, original => {
199        copy_pixels(original.as_ref(), input_shape, &mut output, shape.len(), read)?;
200        with_storage!(&mut output, image => {
201            image.set_units(original.units()).map_err(raised)?;
202            image.set_misc_info(original.misc_info()).map_err(raised)
203        })
204    })?;
205    let rows = match source {
206        Storage::Paged(image) => {
207            image_history_rows(image.name().expect("persistent image")).map_err(raised)?
208        }
209        Storage::Temp { image, .. } => image.history_rows().map_err(raised)?.into_owned(),
210    };
211    with_storage!(&mut output, image => image.append_history_entries(&rows)).map_err(raised)?;
212    Ok(output)
213}

215#[pymethods]
216impl PyImageTool {
217    #[allow(clippy::too_many_arguments)]
218    #[pyo3(signature = (outfile="", direction=false, spectral=false, stokes="", linear=false, tabular=false, overwrite=false, silent=false), text_signature = "($self, outfile='', direction=False, spectral=False, stokes='', linear=False, tabular=False, overwrite=False, silent=False)")]
219    fn adddegaxes(
220        &self,
221        py: Python<'_>,
222        outfile: &str,
223        direction: bool,
224        spectral: bool,
225        stokes: &str,
226        linear: bool,
227        tabular: bool,
228        overwrite: bool,
229        silent: bool,
230    ) -> PyResult<Py<Self>> {
231        let request = AddAxes {
232            outfile,
233            direction,
234            spectral,
235            stokes,
236            linear,
237            tabular,
238            overwrite,
239            silent,
240        };
241        let mut logs = Vec::new();
242        let result: PyResult<Option<_>> = py.detach(|| {
243            let state = self.state();
244            let Some(attachment) = state.image.as_ref() else {
245                return Ok(None);
246            };
247            if !outfile.is_empty() && !overwrite && Path::new(outfile).symlink_metadata().is_ok() {
248                return Err(raised(format!(
249                    "File {outfile} exists, and the user does not want to remove it."
250                )));
251            }
252            publish_image(outfile, overwrite, |path| {
253                let mut original = attachment.lock();
254                if !outfile.is_empty() {
255                    for source in original
256                        .path()
257                        .into_iter()
258                        .chain(attachment.operand_paths().iter().map(PathBuf::as_path))
259                    {
260                        ensure_outfile_does_not_alias_input(Path::new(outfile), source)
261                            .map_err(raised)?;
262                    }
263                }
264                attachment.require_pixel_read_lock(&mut original)?;
265                let mut record = original.coordinate_record();
266                let input_shape = original.shape();
267                let coordinates = request.coordinates(&mut record, input_shape.len())?;
268                let mut image_info = image_info_record(&original).cloned().unwrap_or_default();
269                attachment.beam_view().apply_to(&mut image_info);
270                let mut shape = input_shape.to_vec();
271                shape.resize(coordinates.n_pixel_axes(), 1);
272                let target = if outfile.is_empty() {
273                    "(temp)image".to_owned()
274                } else {
275                    format!("image '{outfile}'")
276                };
277                logs.push(LogLine::info(format!(
278                    "Creating {target} of shape {shape:?}"
279                )));
280                let mut copy = match &*original {
281                    ImageHandle::Float(source) => ImageHandle::Float(copy_storage(
282                        source,
283                        path,
284                        input_shape,
285                        &shape,
286                        &coordinates,
287                        |source, start, lengths, stride| match original.fits() {
288                            // CASA copies the FITS values, including NaN, but the implicit FITS mask has no name to copy.
289                            Some(fits) => fits.read(start, lengths, stride),
290                            None => native_pixels(source, start, lengths, stride),
291                        },
292                    )?),
293                    ImageHandle::Double(source) => ImageHandle::Double(copy_storage(
294                        source,
295                        path,
296                        input_shape,
297                        &shape,
298                        &coordinates,
299                        native_pixels,
300                    )?),
301                    ImageHandle::Complex(source) => ImageHandle::Complex(copy_storage(
302                        source,
303                        path,
304                        input_shape,
305                        &shape,
306                        &coordinates,
307                        native_pixels,
308                    )?),
309                    ImageHandle::DComplex(source) => ImageHandle::DComplex(copy_storage(
310                        source,
311                        path,
312                        input_shape,
313                        &shape,
314                        &coordinates,
315                        native_pixels,
316                    )?),
317                };
318                copy.set_coordinate_record(record)?;
319                if !image_info.is_empty() {
320                    with_pixels!(&mut copy, storage => with_storage!(storage, image =>
321                        image.set_image_info_record(image_info)
322                    ))
323                    .map_err(raised)?;
324                }
325                if state.history {
326                    copy.record_history("adddegaxes", request.history())?;
327                }
328                copy.save().map_err(raised)?;
329                Ok(copy)
330            })
331            .map(Some)
332        });
333        emit(py, "casars.tools.image", &logs)?;
334        Py::new(py, result?.map_or_else(Self::new, Self::attached))
335    }
